Propose a near‑optimal step‑function (non‑negative vector) for the
evaluate_sequence() benchmark.

Algorithm
---------
1. **Deterministic sweep** – evaluate several parametric families
   (inverse‑distance, inverse‑product‑distance, exponential,
   Hann / Hamming / Blackman windows) for a handful of lengths.
   The best vector found becomes the initial seed.

2. **Differential Evolution (DE)** on the probability simplex.
   • Population size ~80 (larger than the reference code) for more
     diversity.
   • Two mutation strategies are mixed:
        • classic “rand/1” and “best/1” (chosen randomly each trial).
   • Mutation factor F is drawn uniformly from [0.5, 1.0] each time,
     crossover probability CR = 0.9.
   • Whenever the global best does not improve for many generations we
     re‑initialise half the population to escape stagnation.
   • All trial vectors are projected onto the simplex with the
     O(n log n) Wang & Carreira‑Perpiñán algorithm.

3. **Projected sub‑gradient polishing**.
   • Starting from the best DE solution we repeatedly take a step in the
     negative sub‑gradient of the “max‑convolution” objective.
   • Step size is adaptively decayed; when progress stalls a tiny random
     “re‑heat” perturbation is applied.
   • This stage seeks the local optimum that DE may have missed.

The routine respects the 1000 s budget (a 5 s safety margin) and prints
progress.  It returns a plain Python list of non‑negative floats whose
sum is 1 (any scaling would yield the same evaluated score because the
objective is homogeneous).

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection – Wang & Carreira‑Perpiñán (O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution) using a real FFT."""
    n = a.shape[0]
    size = 2 * n                # zero‑padding length
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # keep only the first (2n‑1) entries (the rest are zeros from padding)
    return conv[: 2 * n - 1]


def score_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector (any sum)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_b = conv.max()
    sum_a = a.sum()
    # the function protects against too‑small sums, but our vectors never reach that.
    return float(2.0 * n * max_b / (sum_a ** 2))


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices belonging to (almost) active shifts contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - eps)[0]

    n = a.shape[0]
    grad = np.zeros_like(a)

    for k in active:
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        i_vals = np.arange(i0, i1 + 1, dtype=np.int64)
        j_vals = k - i_vals
        # each (i,j) pair contributes a[j] + a[i] to the gradient of a[i];
        # because the convolution counts ordered pairs, the derivative is 2*a[j].
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Parametric families (used for the deterministic sweep)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """Edge‑heavy: a[i] ∝ (min(i, n‑1‑i)+1)^{-p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """Edge‑heavy: a[i] ∝ ((i+1)*(n‑i))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def window_distribution(n: int, kind: str) -> np.ndarray:
    """Classic signal‑processing windows (Hann, Hamming, Blackman)."""
    i = np.arange(n, dtype=np.float64)
    if kind == "hann":
        w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    elif kind == "hamming":

             + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    else:
        raise ValueError(f"Unknown window kind: {kind}")
    w = np.maximum(w, 1e-14)
    w /= w.sum()
    return w

            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            current /= current.sum()
            current_score = score_vector(current)
            step = 0.3
            no_improve = 0
            print(f"Subgrad it {it:6d}: re‑heat")

    # ------------------------------------------------------------------
    # 6. Return the best vector found (as a plain list)
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {len(best_vec)})")
    return best_vec.tolist()
