"""
Propose a step‑function (non‑negative sequence) that approximately
minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

by Adam optimisation on a soft‑max (probability‑simplex) parametrisation.
The algorithm uses a temperature‑controlled soft‑max to smooth the
`max` operation, decays the temperature, and finally polishes the best
candidate with a pure max‑entry sub‑gradient and Euclidean projection
onto the simplex.

The implementation runs without any external files, respects the 1000 s
budget, and keeps the best sequence found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n, dtype=np.float64) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using real FFT (fast for the sizes we need).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    eps_temp = max(max_conv * 0.5, 1e-12)   # start a bit above half the peak
    eps_min = max(max_conv * 1e-6, 1e-12)  # never go to zero
    eps_decay = 0.9995

    # Tracking the best solution
    best_seq = a.copy()
    best_val = 2.0 * n * max_conv
    best_iter = 0
    stagn = 0               # iterations without improvement
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop (soft‑max smoothed max)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---------- obtain current a from logits ----------
        a = np.exp(x_vec - np.max(x_vec))
        a_sum = a.sum()
        if a_sum == 0:
            # numerical disaster – re‑initialise a tiny uniform vector
            a = np.full_like(x_vec, 1.0 / n)
        else:
            a /= a_sum

        # ---------- convolution and current score ----------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ---------- keep the best ----------
        if cur_score < best_val - 1e-13:
            best_val = cur_score
            best_seq = a.copy()
            best_iter = it
            stagn = 0
            print(f"[iter {it:6d}] new_best={best_val:.12f}  lr={lr:.2e}  eps={eps_temp:.2e}")
        else:
            stagn += 1

        # ---------- temperature schedule ----------
        eps_temp = max(eps_temp * eps_decay, eps_min)

        # ---------- soft‑max weights ----------
        # w_i ∝ exp((conv_i - max_conv) / eps_temp)
        # use np.clip for safety against overflow/underflow
        w_exp = np.exp(np.clip((conv - max_conv) / eps_temp, -700, 700))
        w_sum = w_exp.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_exp / w_sum

        # ---------- gradient wrt a (soft‑max weighted sum) ----------
        rev_a = a[::-1]
        # full convolution rev_a * w, then take the central n entries
        conv_rev = _full_convolution_fft(rev_a, w)
        g_a = 2.0 * conv_rev[n - 1 : n - 1 + n]   # same shape as a

        # ---------- chain rule: gradient wrt logits ----------
        # For a = softmax(x) we have ∂L/∂x_i = a_i * (g_a_i - <g_a, a>)
        avg = np.dot(g_a, a)                     # scalar <g_a, a>
        g_x = a * (g_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)

        it_float = float(it)
        m_hat = m / (1.0 - beta1 ** it_float)
        v_hat = v / (1.0 - beta2 ** it_float)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        x_vec -= step

        # ---------- optional perturbation when stuck ----------
        if stagn > 3000:
            # Add a small isotropic Gaussian perturbation to the logits
            noise = rng.normal(scale=0.02, size=n)
            x_vec += noise
            stagn = 0
            # temporarily boost learning rate to escape
            lr = min(1.0, lr * 1.3)
            print(f"[perturb] iteration {it}")

        # ---------- slowly decay the base learning rate ----------
        lr = max(1e-4, lr * 0.9999)

        # occasional re‑normalisation of logits for numerical stability
        if it % 5000 == 0:
            x_vec -= np.max(x_vec)

    # ------------------------------------------------------------------
    # Pure max‑entry sub‑gradient polishing on the best candidate.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    for _ in range(3000):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        # pick the first maximal index (any works)
        idx = int(np.flatnonzero(conv >= mx - 1e-15)[0])

        # Sub‑gradient of the single maximal entry:
        # g_i = 2 * a_j where i + j = idx (with 0 ≤ i,j < n)
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j]

        # Small projected step (explicit projection onto the simplex)
        a = _project_simplex(a - 5e-6 * g)

    # ------------------------------------------------------------------
    # Evaluate with the official function and return the clamped list.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in a.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}, "
          f"best_iter={best_iter}, best_val={best_val:.12f}")
    return final_seq
