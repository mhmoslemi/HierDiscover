"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
* The feasible domain is the probability simplex:
  a_i ≥ 0, Σ a_i = 1 (the objective is scale‑invariant).
* The objective 2·n·max(a*a) is approximated by a smooth
  log‑sum‑exp surrogate.  The surrogate and its gradient are
  evaluated by FFT‑based convolutions (O(n log n)).
* A continuation scheme (large → tiny smoothing ε) drives a
  L‑BFGS‑B optimisation from many random starts.
* The best iterate is refined by a greedy hill‑climbing phase:
  repeatedly transfer a small random amount of mass from a
  source index to a target index; the convolution and its 
  maximum are recomputed cheaply with FFT after each trial.
* The best exact value seen (via ``evaluate_sequence``) is
  printed and the corresponding sequence is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself (length 2·len(x)‑1)
    using an FFT of length that is a power of two.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)               # eliminate tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors (length len(x)+len(y)‑1)
    using an FFT of length that is a power of two.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp of the auto‑convolution) and gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Return (L_ε(a), ∂L_ε/∂w) for a soft‑max parametrisation
    a_i = exp(w_i) / Σ exp(w).
    """
    # soft‑max (guarantees a_i ≥ 0, Σa_i = 1)
    w_shift = w - np.max(w)               # for numerical stability
    a = np.exp(w_shift)
    a = a / a.sum()

    # full auto‑convolution a*a
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth approximation of the max (log‑sum‑exp)
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # = log Σ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form, see paper)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # p_k ≈ soft‑max over conv entries

    # g_i = 4·n·∑_k p_k·a_{k‑i}   (indices outside [0,n‑1] give 0)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)         # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]       # middle n entries
    grad_a = 4.0 * n * seg[::-1]          # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)           # ∂L/∂w

    return L, grad_w


# ----------------------------------------------------------------------
# Fast exact evaluation (avoids the O(n²) numpy.convolve)
# ----------------------------------------------------------------------
def _fast_evaluate(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²) – the exact objective used by evaluate_sequence."""
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 1e-12:                      # safeguard – should never happen
        return np.inf
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (greedy mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, cur_val: float,
                deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing: repeatedly move a small random fraction of mass
    from a random source index to a random target index.
    The move is kept only if the exact objective improves.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        src = best_a[i]
        if src <= 1e-12:
            continue
        # transfer at most 5 % of the source mass
        delta = src * 0.05 * rng.random()
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _fast_evaluate(cand)
        if val < best_val:
            best_a = cand
            best_val = val
            # occasional progress report
            print(f"[hill‑climb] improved to {best_val:.6f}")
    return best_a, best_val
