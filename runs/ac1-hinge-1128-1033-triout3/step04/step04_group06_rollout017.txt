"""
Hybrid sequence optimizer.

1. Start from the best known construction (`height_sequence_1`) if it exists,
   otherwise from a modest Dirichlet draw.
2. Repeatedly solve a linear program:
       maximize   Σ g_j
       subject to p * g  ≤  M   (point‑wise convolution bound)
                  g ≥ 0
   where M = max(p * p).  The LP finds a distribution g that can carry more
   total mass without increasing the current maximum convolution value.
3. Mix the current distribution p with the normalised LP solution g
   (convex combination with weight t = 0.5).  The theoretical argument
   guarantees a monotone decrease of max(p * p).  Iterate while time permits.
4. When the LP phase ends, perform a fast random‑hill‑climbing phase:
   propose small convex combinations of p with random Dirichlet samples,
   accept only if the evaluation improves.
5. Keep the best sequence seen and return it as a plain list of floats.
   The evaluation function is scale‑invariant, so we work on the probability
   simplex (Σ a_i = 1).

The algorithm runs entirely within the given time budget (default 1000 s)
and uses only NumPy, SciPy and the provided `evaluate_sequence`.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Fast full linear convolution via a real FFT (used many times)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x with itself (real FFT)."""
    n = x.shape[0]
    total = 2 * n - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:total]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights that (hopefully) minimizes
    `evaluate_sequence`.  The routine performs a LP‑refinement followed by
    a simple stochastic hill‑climb, always respecting the 1000 s time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny margin so we finish before the hard deadline
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Initial distribution on the probability simplex
    # ------------------------------------------------------------------
    try:
        # reuse the best construction found previously, if present
        init = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        p = init / init.sum()
    except Exception:
        # otherwise start from a modest Dirichlet draw
        n = 2000
        p = np.random.dirichlet(alpha=np.full(n, 0.5))
        p = p / p.sum()
    else:
        n = p.shape[0]

    # ------------------------------------------------------------------
    # 2. Helper to evaluate the current probability vector
    # ------------------------------------------------------------------
    def _eval(p_vec: np.ndarray) -> float:
        conv = _full_convolution_fft(p_vec)
        return float(2.0 * n * conv.max())

    best_p = p.copy()
    cur_val = _eval(p)
    best_val = cur_val
    print(f"[init] n={n}, evaluation={cur_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Phase 1 – LP refinement (up to max_iter_lp iterations)
    # ------------------------------------------------------------------
    max_iter_lp = 20
    for it in range(max_iter_lp):
        now = time.time()
        if now >= deadline:
            break

        # recompute convolution and the current bound M
        conv = _full_convolution_fft(p)
        M = conv.max()
        cur_val = float(2.0 * n * M)

        if cur_val < best_val:
            best_val = cur_val
            best_p = p.copy()
            print(f"[LP {it:02d}] new best = {best_val:.6f}")

        # Build the sparse matrix describing the linear constraints:
        # (p * g)[k] = Σ_i p_i g_{k-i}  ≤  M   for k = 0,…,2n‑2
        rows = 2 * n - 1
        # each diagonal offset i (0≤i<n) contains the constant p[i]
        diag_data = []
        offsets = np.arange(n)               # 0,…,n‑1
        for i in range(n):
            length = rows - i                # length of the i‑th diagonal
            diag_data.append(np.full(length, p[i]))
        A_ub = sp.diags(diag_data, offsets, shape=(rows, n), format="csc")
        b_ub = np.full(rows, M, dtype=float)

        # Linear program: maximise Σ g_j  ⇔ minimise −Σ g_j
        c = -np.ones(n)                      # objective coefficients

        try:
            res = opt.linprog(c,
                              A_ub=A_ub,
                              b_ub=b_ub,
                              bounds=[(0, None)] * n,
                              method="highs")
        except Exception as e:
            print(f"[LP] linprog raised an exception: {e}")
            break

        if not res.success:
            print(f"[LP] linprog failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        # If we cannot increase the total mass, we are at a stationary point
        if sum_g <= 1.0 + 1e-12:
            print("[LP] No increase in Σg – terminating LP phase")
            break

        # Normalise g to the simplex and mix it with the current p
        g_norm = g / sum_g
        t_mix = 0.5                           # mixing weight (any 0<t≤1 works)
        new_p = (1.0 - t_mix) * p + t_mix * g_norm
        new_p = np.maximum(new_p, 0.0)
        s = new_p.sum()
        if s == 0:
            new_p = np.full_like(new_p, 1.0 / n)
        else:
            new_p = new_p / s

        p = new_p

    # ------------------------------------------------------------------
    # 4. Phase 2 – Random hill climbing for the remaining time
    # ------------------------------------------------------------------
    hill_iter = 0
    # a small step size that will be slowly annealed
    tau = 0.02
    while True:
        now = time.time()
        if now >= deadline:
            break

        # propose a tiny convex combination with a fresh random Dirichlet vector
        q = np.random.dirichlet(alpha=np.full(n, 0.5))
        cand = (1.0 - tau) * p + tau * q
        cand = cand / cand.sum()

        cand_val = _eval(cand)

        # accept only if we improve
        if cand_val < cur_val:
            p = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_p = cand.copy()
                print(f"[hill {hill_iter:07d}] new best = {best_val:.6f}")

        hill_iter += 1
        # slowly reduce the step size so we can fine‑tune later
        if hill_iter % 5000 == 0:
            tau = max(0.001, tau * 0.9)

    # ------------------------------------------------------------------
    # 5. Final reporting and return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_p.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_p.tolist()
