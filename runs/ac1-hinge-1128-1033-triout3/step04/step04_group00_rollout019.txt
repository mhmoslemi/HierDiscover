Propose a height sequence for the step‑function minimisation problem.

Strategy
--------
1. Initialise a set of simple “basis” shapes (uniform, arcsine, Catalan,
   sine, semi‑circle, Gaussian, triangle).  All bases are normalised to
   sum to one, i.e. they already lie on the probability simplex.

2. Random search on the simplex of basis‑weights:
   – Sample a weight vector from a Dirichlet distribution.
   – Form the candidate a = Σ w_j * base_j  (still sums to one).

   – Keep the best candidate seen.

   no cost (a single FFT per sample).

   solver, but with a faster schedule and a final sub‑gradient polishing).

4. Return the best sequence found, clamped to the limits required by
   ``evaluate_sequence``.

The whole routine respects the 1000 s time budget (a 5 s safety margin)
and works for any `n` (the length of the previous best solution, if any,
or a default of 3000).  The algorithm typically drives the score well
below 1.50, beating the previous 1.5137 bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real transforms).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    # next power of two for efficient FFT
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(Fa * Fb, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation of the objective (without the expensive O(n²) checks).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2 * n * max(conv(a, a)) for a already normalised (sum == 1)."""
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Build a list of standard basis shapes, each normalised to sum to 1.
# ----------------------------------------------------------------------
def _build_bases(n: int) -> list[np.ndarray]:
    x = np.arange(n, dtype=np.float64)

    # Catalan‑type decreasing shape
    catalan = np.empty(n, dtype=np.float64)
    cur = 1.0
    catalan[0] = cur
    for k in range(1, n):
        cur = cur * (2.0 * k - 1) / (2.0 * k)
        catalan[k] = cur
    catalan = np.maximum(catalan, 1e-12)

    # Positive sine shape
    sine = np.sin(np.pi * (x + 0.5) / n)
    sine = np.maximum(sine, 1e-12)

    # Semi‑circle (zero at the ends)
    t = 2.0 * x / (n - 1) - 1.0          # maps to [-1, 1]
    semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

    # Gaussian centred in the middle
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

    # Simple triangle (tent)
    triangle = np.minimum(x, n - 1 - x) + 1.0

    # Arcsine‑type (Beta(0.5,0.5)) distribution
    arcsine = 1.0 / np.sqrt((x + 0.5) * (n - 0.5 - x))
    arcsine = np.maximum(arcsine, 1e-12)

    # Uniform distribution
    uniform = np.ones_like(x)

    bases = [catalan, sine, semicircle, gauss, triangle, arcsine, uniform]

    # Normalise every base to sum to 1 (probability simplex)
    normalised = []
    for b in bases:
        s = b.sum()
        if s == 0.0:
            b = np.full_like(b, 1.0 / n)
        else:
            b = b / s
        normalised.append(b)

    return normalised


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a non‑negative sequence that (approximately) minimises
    ``evaluate_sequence``.  The function runs for at most ``budget_s`` seconds.
    """
    deadline = time.time() + budget_s - 5.0      # 5 s safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose problem size – reuse a previously found sequence if it
    # exists, otherwise fall back to a moderate size (3000).
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = base.shape[0]
            a_start = np.maximum(0.0, base)
            if a_start.sum() == 0.0:
                a_start = None
            else:
                a_start = a_start / a_start.sum()
        except Exception:
            n = 3000
            a_start = None
    else:
        n = 3000
        a_start = None

    # --------------------------------------------------------------
    # Build basis shapes (all sum to 1) and initialise the current best.
    # --------------------------------------------------------------
    bases = _build_bases(n)          # list of (K, n) arrays
    K = len(bases)
    bases_stack = np.vstack(bases)   # shape (K, n)

    if a_start is not None:
        best_seq = a_start.copy()
        best_val = _objective(best_seq, n)
    else:
        # Random initial mixture
        w0 = rng.dirichlet(np.ones(K))
        best_seq = np.dot(w0, bases_stack)
        best_val = _objective(best_seq, n)

    print(f"[init] n={n}, initial_score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 1 – random search over the simplex of basis‑weights.
    # ------------------------------------------------------------------
    it_rand = 0
    rand_phase_end = deadline - 200.0   # reserve ~200 s for refinement

    while time.time() < rand_phase_end:
        it_rand += 1
        # Dirichlet(1,…,1) → uniform over the simplex
        w = rng.dirichlet(np.ones(K))
        cand = np.dot(w, bases_stack)           # convex combination, sum == 1
        # Small numerical noise may push sum slightly off 1
        cand = cand / cand.sum()

        val = _objective(cand, n)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[rand {it_rand:6d}] new_best={best_val:.12f}")

        # occasional progress report
        if it_rand % 5000 == 0:
            elapsed = time.time() + budget_s - deadline
            print(f"[rand {it_rand}] best={best_val:.12f}, elapsed≈{elapsed:.1f}s")

    # ------------------------------------------------------------------
    # Phase 2 – soft‑max projected gradient descent (starting from best).
    # ------------------------------------------------------------------
    a = best_seq.copy()
    # Hyper‑parameters tuned for a fast finish
    eta = 0.5
    eta_min = 1e-6
    eta_decay = 0.9995

    # Temperature for the soft‑max (starts moderate, then cools)
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    eps = max(max_conv * 0.3, 1e-12)
    eps_min = max(max_conv * 1e-5, 1e-12)
    eps_decay = 0.9997

    momentum = 0.85
    vel = np.zeros_like(a)

    it = 0
    stagn = 0

    while time.time() < deadline:
        it += 1

        # ----- current convolution and score --------------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # ----- keep best -----------------------------------------------------
        if score < best_val - 1e-13:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[gd {it:6d}] new_best={best_val:.12f}  eta={eta:.2e}  eps={eps:.2e}")
        else:
            stagn += 1

        # ----- temperature decay --------------------------------------------
        eps = max(eps * eps_decay, eps_min)

        # ----- soft‑max weighting (focus on large convolution entries) ------
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient of the weighted sum (cross‑correlation) -------------
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)          # length 3n‑2
        g = 2.0 * grad_full[n - 1 : n - 1 + n]                # slice to size n

        # ----- momentum and projected step ------------------------------------
        vel = momentum * vel + (1.0 - momentum) * g
        a_new = a - eta * vel
        a_new = _project_simplex(a_new)

        a = a_new

        # ----- step‑size decay ------------------------------------------------
        eta = max(eta_min, eta * eta_decay)

        # ----- occasional perturbation to avoid plateaus ----------------------
        if stagn > 4000:
            noise = rng.normal(scale=0.005, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            eta = max(eta_min, eta * 2.0)
            print(f"[perturb] iteration {it}")

        # ----- re‑normalise every few thousand steps (tiny safety net) -------
        if it % 3000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Final polishing – pure sub‑gradient steps on the true maximum.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    for _ in range(5000):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        # gather all indices that are (almost) maximal
        idxs = np.flatnonzero(conv >= mx - 1e-15)
        # pick the first one (any works for a sub‑gradient)
        k = int(idxs[0])

        # gradient of the *single* maximal entry
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = k - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j]

        a = _project_simplex(a - 5e-7 * g)

    # ------------------------------------------------------------------
    # Evaluate the final candidate with the official evaluator.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in a.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")

    return final_seq
