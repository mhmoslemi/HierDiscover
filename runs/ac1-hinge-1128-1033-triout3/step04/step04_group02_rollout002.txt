"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Approach
--------
* Parameterise the step heights by a soft‑max vector *a* (Σa = 1, a ≥ 0).
  The objective is scale‑invariant, therefore fixing Σa = 1 is safe.
* Approximate the non‑smooth ``max`` in the objective by a
  log‑sum‑exp smoothing:
        F_ε(a) = 2·n·ε·log∑_k exp((a∗a)_k / ε) .
  Its gradient can be written in closed form and evaluated with two FFT‑based
  convolutions, which makes each L‑BFGS iteration cheap (O(n log n)).
* Run a continuation scheme: solve the smoothed problem for a sequence of
  decreasing ε (log‑space) with L‑BFGS‑B, keeping the best exact objective
  seen.
* Perform a few random restarts (different Dirichlet initialisations) to
  escape possible local minima.
* Finish with a targeted hill‑climbing phase that repeatedly moves a small
  amount of mass from indices that contribute most to the current maximum
  of the auto‑convolution to indices that contribute little.  The exact
  objective (``2·n·max(conv)/sum(a)^2``) is evaluated directly – it is fast
  and avoids the security checks of ``evaluate_sequence``.
* The best sequence found before the time budget expires is returned
  (as a plain ``list[float]``).
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (useful for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself (length 2·len‑1) using FFT.
    The result is clipped to non‑negative values to avoid tiny negatives.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors (length len(x)+len(y)-1) using FFT.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. the soft‑max parameters
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Return (F_ε, ∂F_ε/∂w) for a given parameter vector `w` and smoothing `eps`.

    The soft‑max enforces a ≥ 0, Σa = 1:
        a_i = exp(w_i) / Σ_j exp(w_j)

    The smoothed objective is
        F_ε(a) = 2·n·ε·log∑_k exp((a∗a)_k / ε)

    Gradient is obtained by the chain rule (soft‑max + convolution).
    """
    # ----- soft‑max (feasible a) -----
    w_shift = w - np.max(w)          # improve numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                    # Σa = 1, a ≥ 0

    # ----- auto‑convolution a*a -----
    conv = _full_convolution(a)       # length 2·n‑1
    max_conv = conv.max()

    # ----- smoothed max via log‑sum‑exp -----
    #   F = 2·n·max_conv + 2·n·eps·log∑_k exp((conv_k‑max_conv)/eps)
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)      # log∑_k exp((conv_k‑max)/eps)
    F = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a (closed form) -----
    #   p_k = exp(conv_k/eps) / Σ_j exp(conv_j/eps)   (soft‑max weights)
    weights = np.exp(shifted)
    p = weights / weights.sum()       # shape (2·n‑1,)

    #   grad_a_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]                               # reversed p
    conv_ap = _conv_fft(a, p_rev)                  # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]                # middle n entries
    grad_a = 4.0 * n * seg[::-1]                   # shape (n,)

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)                    # ∂F/∂w

    return F, grad_w


def _smooth_obj_wrapper(w: np.ndarray, eps: float, n: int):
    """Helper for ``scipy.optimize.minimize`` – returns (value, grad)."""
    return _smooth_obj_and_grad(w, eps, n)


# ----------------------------------------------------------------------
# Exact objective (used for cheap hill‑climbing)
# ----------------------------------------------------------------------


def _exact_objective(a: np.ndarray, n: int) -> float:
    """
    Compute the exact evaluation value:
        2·n·max(a∗a) / (sum(a)^2)

    The sequence is assumed to be non‑negative; the formula matches
    ``evaluate_sequence`` when the sum is > 0.01 and all entries are finite.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = np.convolve(a, a)
    max_c = conv.max()
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Random simplex (Dirichlet) sampler
# ----------------------------------------------------------------------


def _random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Sample a random point on the probability simplex using iid exponentials.
    """
    x = rng.exponential(scale=1.0, size=n)
    x_sum = x.sum()
    if x_sum == 0:
        return np.full(n, 1.0 / n)
    return x / x_sum


# ----------------------------------------------------------------------
# Hill‑climbing refinement (targeted mass transfers)
# ----------------------------------------------------------------------


def _hill_climb(a_start: np.ndarray,
                best_val: float,
                deadline: float,
                n: int,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform targeted random mass transfers while time permits.
    Each move keeps Σa constant and never makes an entry negative.
    Accept a move only when it strictly improves the exact objective.
    """
    cur_a = a_start.copy()
    cur_val = best_val

    # pre‑compute a small number for transfer size scaling
    TRANSFER_SCALE = 0.10   # up to 10 % of the source mass per move

    while time.time() < deadline:
        # Occasionally recompute the convolution and the index of the current peak
        conv = _full_convolution(cur_a)
        k_max = int(np.argmax(conv))

        # Contribution of each index to the peak at k_max:
        #   contrib[i] = a_i * a_{k_max-i}  (zero if the partner index is out of range)
        idx_partner = k_max - np.arange(n)
        valid = (idx_partner >= 0) & (idx_partner < n)
        contrib = np.zeros_like(cur_a)
        contrib[valid] = cur_a[valid] * cur_a[idx_partner[valid]]

        # Choose a source index with probability proportional to its contribution
        total_contrib = contrib.sum()
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = rng.choice(n, p=src_probs)
        else:
            # fallback – uniform choice
            i = rng.integers(0, n)

        # Choose a destination index with low contribution (bottom 15 % quantile)
        low_thr = np.quantile(contrib, 0.15)
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size == 0:
            j = rng.integers(0, n)
        else:
            j = rng.choice(low_candidates)

        if i == j:
            continue

        # Transfer a random fraction of the source mass
        delta = cur_a[i] * TRANSFER_SCALE * rng.random()
        if delta <= 0.0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # Evaluate the exact objective (fast, no security wrappers)
        val = _exact_objective(new_a, n)

        if val < cur_val - 1e-12:          # accept only strict improvements
            cur_a = new_a
            cur_val = val
            print(f"[hill] improved objective = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimize a non‑negative step‑function under the ``evaluate_sequence`` metric.

    The function respects the 1000 s time budget, performs a continuation
    L‑BFGS optimisation of a log‑sum‑exp surrogate, tries a few random restarts,
    and finally refines the best solution with a targeted hill‑climbing phase.
    The returned list is the best sequence found before the deadline.
    """
    # ------------------------------------------------------------------
    # basic setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # leave a tiny safety margin

    # ------------------------------------------------------------------
    # problem size – larger n gives a tighter theoretical bound
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # initialise a (try to recycle a previous construction if available)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)    # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        # fallback – uniform start
        a0 = np.full(n, 1.0 / n)

    # enforce feasibility (non‑negative, sum = 1)
    a0 = np.maximum(a0, 0.0)
    a0_sum = a0.sum()
    if a0_sum == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0_sum

    # soft‑max parameters for the L‑BFGS stage
    w = np.log(a0 + 1e-12)

    # keep track of the best solution seen so far
    best_a = a0.copy()
    best_val = _exact_objective(best_a, n)
    print(f"[init] n={n}, initial exact objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # continuation schedule for the smoothing parameter ε
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(5e-8), num=14)

    # ------------------------------------------------------------------
    # helper that runs the whole ε‑schedule starting from a given w
    # ------------------------------------------------------------------
    def _run_continuation(start_w: np.ndarray) -> tuple[np.ndarray, float]:
        w_local = start_w.copy()
        local_best_a = best_a.copy()
        local_best_val = best_val

        for eps in eps_schedule:
            if time.time() >= deadline:
                break

            # L‑BFGS‑B on the smoothed surrogate (allow more iterations than before)
            max_iter_stage = 400
            res = minimize(_smooth_obj_wrapper,
                           w_local,
                           args=(eps, n),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter_stage,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})

            w_local = res.x

            # compute the current feasible a and evaluate the exact metric
            w_shift = w_local - np.max(w_local)
            a_cur = np.exp(w_shift)
            a_cur = a_cur / a_cur.sum()

            cur_val = _exact_objective(a_cur, n)
            if cur_val < local_best_val:
                local_best_val = cur_val
                local_best_a = a_cur.copy()
                print(f"[eps {eps:.1e}] new best exact = {local_best_val:.6f}")

        return w_local, local_best_a, local_best_val

    # ------------------------------------------------------------------
    # first (deterministic) run – start from the recycled / uniform vector
    # ------------------------------------------------------------------
    w, best_a_candidate, best_val_candidate = _run_continuation(w)
    if best_val_candidate < best_val:
        best_val = best_val_candidate
        best_a = best_a_candidate.copy()

    # ------------------------------------------------------------------
    # a few random restarts (only if time permits)
    # ------------------------------------------------------------------
    NUM_RESTARTS = 2
    for r in range(NUM_RESTARTS):
        if time.time() >= deadline:
            break
        # random simplex point → soft‑max parameters
        a_rand = _random_simplex(n, rng)
        w_rand = np.log(a_rand + 1e-12)
        w, cand_a, cand_val = _run_continuation(w_rand)
        if cand_val < best_val:
            best_val = cand_val
            best_a = cand_a.copy()
            print(f"[restart {r+1}] new global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # final hill‑climbing refinement (targeted mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hill_climb(best_a, best_val, deadline, n, rng)

    # ------------------------------------------------------------------
    # final report (use the official evaluate_sequence for sanity)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] objective reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
