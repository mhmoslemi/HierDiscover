Propose a step‑function (non‑negative height sequence) that minimizes the
objective defined by ``evaluate_sequence``.

The algorithm works on the probability simplex (the heights sum to 1) by
parameterising the vector with logits (soft‑max).  It repeatedly performs the
following for the allotted time budget:

* Compute the full auto‑convolution a*a (via a real FFT).
* Form a smooth approximation of the max‑operator by a temperature‑scaled
  soft‑max over the convolution entries.
* Obtain a (sub‑)gradient of the smoothed maximum using a single
  convolution with the weight vector.
* Update the logits with Adam (bias‑corrected moments) and a cosine learning‑
  rate schedule.
* Occasionally apply a random “jitter’’ that moves a fraction of mass from
  one coordinate to another – this helps the optimiser escape plateaux.
* Periodically restart from a fresh random Dirichlet point; the best sequence
  ever observed is kept globally.

After the time runs out we return the best sequence found (as a plain
Python list).  The code also prints progress and the final value reported
by ``evaluate_sequence`` if it is available in the surrounding environment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution x*y using the FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    # next power of two for an efficient FFT
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # tiny negative values caused by round‑off are harmless – make them zero
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max (weights w) w.r.t. a
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i}.
    Implemented as a single convolution: reverse w and convolve it with a.
    """
    n = a.shape[0]
    u = w[::-1]                                 # reverse weight vector
    conv = _full_convolution_fft(u, a)          # length = 3n‑2
    # indices that correspond to the original length n:
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g

    constantly keeping the best sequence seen and finally returning it.
    """

    start_time = time.time()

    # Choose a length.  If a previous construction (height_sequence_1)

        prev = np.array(height_sequence_1, dtype=float)

    # ------------------------------------------------------------------

    best_a = None

    # ------------------------------------------------------------------
    def _objective(a_vec):
        conv = _full_convolution_fft(a_vec, a_vec)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Main optimisation loop with timed restarts
    # ------------------------------------------------------------------
    restart_seconds = 30.0            # seconds allocated per restart block
    eps = 1e-12                       # for log‑softmax stability

    while time.time() < deadline:
        now = time.time()
        time_left = deadline - now

        # ----------------------------------------------------------------
        # If enough time remains, perform a fresh restart; otherwise jump
        # to a final fine‑tuning phase.
        # ----------------------------------------------------------------
        if time_left > 2 * restart_seconds:
            # ---------- initialise a new candidate ----------
            if prev is not None and np.random.rand() < 0.2:
                # use the previous construction (possibly padded)
                a = np.zeros(n, dtype=float)
                a[: prev.shape[0]] = np.clip(prev, 0.0, 1000.0)
                # fill the rest with a tiny Dirichlet draw to keep the sum 1
                rem = n - prev.shape[0]
                if rem:
                    extra = np.random.gamma(1.0, 1.0, size=rem)
                    a[prev.shape[0] :] = extra
                a = a / a.sum()
            else:
                # fresh random Dirichlet start (α < 1 gives a more varied shape)
                alpha = 0.8
                g = np.random.gamma(alpha, 1.0, size=n)
                a = g / g.sum()

            # ----------
            # Soft‑max parametrisation (logits) and Adam state
            # ----------
            x = np.log(np.maximum(a, eps))

            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1, beta2 = 0.90, 0.999
            beta1_pow, beta2_pow = 1.0, 1.0
            eps_adam = 1e-8
            base_lr = 0.35                     # relatively aggressive start

            # deadline for this restart phase
            restart_deadline = now + restart_seconds
            iteration = 0

            # ------------------------------------------------------------
            # Restart‑local optimisation loop
            # ------------------------------------------------------------
            while time.time() < restart_deadline:
                iteration += 1

                conv = _full_convolution_fft(a, a)
                M = conv.max()
                cur_val = 2.0 * n * M

                # keep global best if improved
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:6.1f}s] new global best {best_val:.6f} "
                          f"(restart iter {iteration})")

                # -------- temperature schedule for the soft‑max ----------
                progress = (time.time() - now) / restart_seconds
                tau_start = M * 0.5
                tau_end = M * 1e-6
                tau = tau_start * (1.0 - progress) + tau_end * progress

                # -------- compute soft‑max weights w over convolution -----
                if tau > 0.0:
                    exp_args = (conv - M) / tau
                    max_arg = exp_args.max()                # stabilise exponentiation
                    exp_vals = np.exp(exp_args - max_arg)
                    sum_exp = exp_vals.sum()
                    if sum_exp == 0.0 or not np.isfinite(sum_exp):
                        w = np.zeros_like(conv)
                        w[np.argmax(conv)] = 1.0
                    else:
                        w = exp_vals / sum_exp
                else:
                    w = np.zeros_like(conv)
                    w[np.argmax(conv)] = 1.0

                # -------- gradient w.r.t. a (smoothed sub‑gradient) ------
                grad_a = _gradient_from_weights(a, w)

                # -------- chain rule to obtain gradient w.r.t. logits ----
                dot = np.dot(grad_a, a)                     # <grad_a , a>
                grad_x = a * (grad_a - dot)                 # ∇_x (softmax)

                # -------- Adam update ---------------------------------
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                beta1_pow *= beta1
                beta2_pow *= beta2
                m_hat = m / (1.0 - beta1_pow)
                v_hat = v / (1.0 - beta2_pow)

                # cosine decay of learning rate within this restart
                lr = base_lr * 0.5 * (1.0 + np.cos(np.pi * progress))

                # parameter update
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # back‑to the simplex via soft‑max
                max_x = x.max()
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # -------- occasional random jitter (mass transfer) -----
                if np.random.rand() < 0.05:      # ~5 % per iteration
                    i = np.random.randint(n)
                    j = np.random.randint(n - 1)
                    if j >= i:
                        j += 1
                    delta = np.random.rand() * a[i] * 0.5
                    a[i] -= delta
                    a[j] += delta
                    a = np.clip(a, 0.0, None)
                    a = a / a.sum()
                    # reset Adam state to avoid stale momentum after a jump
                    m.fill(0.0)
                    v.fill(0.0)
                    beta1_pow, beta2_pow = 1.0, 1.0
                    x = np.log(np.maximum(a, eps))

                # occasional progress prints
                if iteration % 5000 == 0:
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:6.1f}s] restart iter {iteration}, "
                          f"cur={cur_val:.6f}, best={best_val:.6f}")

            # end of this restart – loop back for another one if time permits
        else:
            # ------------------------------------------------------------
            # Final fine‑tuning phase (use the best solution found so far)
            # ------------------------------------------------------------
            if best_a is None:
                # fallback: uniform distribution
                a = np.full(n, 1.0 / n)
            else:
                a = best_a.copy()

            x = np.log(np.maximum(a, eps))
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1, beta2 = 0.90, 0.999
            beta1_pow, beta2_pow = 1.0, 1.0
            eps_adam = 1e-8
            base_lr = 0.15                     # a gentler step size for polishing

            while time.time() < deadline:
                conv = _full_convolution_fft(a, a)
                M = conv.max()
                cur_val = 2.0 * n * M
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:6.1f}s] new global best (fine) {best_val:.6f}")

                # In the final stage we drive the temperature essentially to zero
                # – i.e. take a hard max – to optimise the true objective.
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0

                grad_a = _gradient_from_weights(a, w)
                dot = np.dot(grad_a, a)
                grad_x = a * (grad_a - dot)

                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                beta1_pow *= beta1
                beta2_pow *= beta2
                m_hat = m / (1.0 - beta1_pow)
                v_hat = v / (1.0 - beta2_pow)

                # slowly decay learning rate to a very small value
                overall_progress = (time.time() - start_time) / (deadline - start_time)
                lr = base_lr * (1.0 - overall_progress) + 1e-6 * overall_progress

                x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
                max_x = x.max()
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # tiny stochastic perturbations keep the optimiser from sticking
                if np.random.rand() < 0.02:
                    i = np.random.randint(n)
                    delta = np.random.rand() * a[i] * 0.1
                    a[i] = max(a[i] - delta, 0.0)
                    a = a / a.sum()
                    x = np.log(np.maximum(a, eps))
                    m.fill(0.0)
                    v.fill(0.0)
                    beta1_pow, beta2_pow = 1.0, 1.0

            # final phase completed – break out of the outer loop
            break

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # Report the evaluation using the provided function (if available)
    try:
        ev = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {ev:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
