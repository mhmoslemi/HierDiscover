def conv_max(p):
    p_fft = np.fft.rfft(p, n=m_fft)
    conv_full = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2*n-1]
    return float(conv_full.max())

# Simplex projection (Euclidean projection onto {x ≥ 0, Σx = 1})
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution using a padded FFT."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _conv_max(p: np.ndarray, m_fft: int) -> float:

    p_fft = np.fft.rfft(p, n=m_fft)
