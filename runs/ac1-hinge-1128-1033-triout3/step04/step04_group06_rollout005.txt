"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(a*a) / (sum(a)**2)

where `a*a` denotes the full linear auto‑convolution.
The search consists of three stages, all respecting the 1000 s budget:

1. **Exponentiated‑gradient (EG) descent** with a soft‑max
   approximation of the max‑norm.  This rapidly drives the objective
   down from a random start.

2. **Linear‑program refinement** (the method described in the
   statement).  For a current sequence `f`, we solve

       max   Σ_j b_j
       s.t.  b_j ≥ 0,
             (f * b)_k ≤ M   for all k,
   where `M = max(f*f)`.  The LP yields a direction that provably
   cannot increase the objective; we mix the LP solution with `f`
   (step size `t`) and repeat while time permits.

3. **Final EG polishing** (a few extra EG iterations) to squeeze
   out the last improvements.

All heavy work (convolutions, FFTs) is performed with NumPy; the LP
uses SciPy’s HiGHS interior‑point solver on a sparse Toeplitz matrix.
The best sequence found is returned as a plain list of floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using a real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative round‑off
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max smoothed max‑norm (used by EG)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i}
    via a single convolution: reverse w, convolve with a,
    then take the slice corresponding to indices 0..n‑1.
    """
    n = a.shape[0]
    # reverse the weight vector (full length = 2n‑1)
    u = w[::-1]
    # full convolution u * a → length 3n‑2 (we only need the middle part)
    conv = _full_convolution_fft(u, a)
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Direct objective (identical to evaluate_sequence's core computation)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)**2).  Assumes a ≥ 0."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _full_convolution_fft(a, a)
    M = conv.max()
    return float(2.0 * a.shape[0] * M / (s * s))

    # Phase 1 – Exponentiated‑gradient descent (soft‑max)
    # ------------------------------------------------------------------
    momentum = np.zeros_like(a)
    beta = 0.92                           # momentum decay
    jitter_prob = 0.018                   # probability of a random mass move
    eta_start, eta_end = 0.45, 5e-5       # multiplicative step‑size schedule
    tau_factor_start, tau_factor_end = 0.35, 1e-5   # temperature schedule

    # allocate ~45 % of the budget to this phase
    eg_end = start_time + 0.45 * budget_s
    iteration = 0
    eps = 1e-15

    while time.time() < eg_end:
        # convolution & current max
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)           # sum(a)=1 ⇒ denominator = 1
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[EG {iteration:6d}] new best = {best_val:.6f}")

        # progress ∈ [0,1] for schedules inside this phase
        prog = (time.time() - start_time) / (eg_end - start_time)

        # temperature for the soft‑max (annealed)
        tau = (tau_factor_start * (1.0 - prog) +
               tau_factor_end   * prog) * M
        if tau <= 0 or not np.isfinite(tau):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            exp_args = (conv - M) / tau
            np.clip(exp_args, -80, 80, out=exp_args)
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp

        # gradient of the smoothed objective
        grad = _gradient_from_weights(a, w)

        # momentum update (exponential moving average)
        momentum = beta * momentum + (1.0 - beta) * grad

        # step‑size schedule (multiplicative)
        eta = eta_start * (1.0 - prog) + eta_end * prog

        # exponentiated‑gradient update (keep exponents bounded)
        step = -eta * momentum
        np.clip(step, -45, 45, out=step)
        a = a * np.exp(step)

        # occasional random jitter to escape plateaux
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a[src] * 0.5
            a[src] -= delta
            a[dst] += delta

        # project back onto the simplex (simple renormalisation)
        a = np.maximum(a, eps)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        iteration += 1

        # occasional sanity check
        if iteration % 2500 == 0:
            _ = _full_convolution_fft(a, a)   # unused, just to keep NumPy happy

    # ------------------------------------------------------------------
    # Phase 2 – Linear‑program refinement (the “LP step” described
    #          in the problem statement).  This phase will dominate the
    #          remaining time budget.
    # ------------------------------------------------------------------
    # Pre‑compute the index arrays for the Toeplitz constraint matrix.
    # A_ub has shape (2n‑1, n) and entry (k, j) = a_j if 0 ≤ k‑j < n else 0.
    # For each column j we have n consecutive rows j … j+n‑1.
    row_idx = (np.arange(n)[:, None] + np.arange(n)).ravel()
    col_idx = np.repeat(np.arange(n), n)

    # Mix‑in parameter (how aggressively we move toward the LP solution)
    t_start, t_end = 0.25, 0.05

    lp_iteration = 0
    while time.time() < deadline:
        # recompute current max convolution M
        conv = _full_convolution_fft(a, a)
        M = conv.max()

        # Build the sparse constraint matrix A_ub = Toeplitz(a)
        data = np.repeat(a, n)                # each column j repeats a_j n times
        A_ub = _make_sparse_matrix(data, row_idx, col_idx, (2 * n - 1, n))

        # Right‑hand side: M for every convolution index
        b_ub = np.full(2 * n - 1, M, dtype=float)

        # Objective: maximise Σ b_j  ⇔ minimise –Σ b_j
        c = -np.ones(n, dtype=float)

        # Solve the LP (HiGHS is the default high‑performance method)
        try:
            from scipy.optimize import linprog
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs')
        except Exception as exc:          # pragma: no cover
            print("[LP] scipy not available or Linprog failed:", exc)
            break

        if not res.success:
            print(f"[LP {lp_iteration}] linprog failed: {res.message}")
            break

        b0 = res.x
        sum_b0 = b0.sum()
        if sum_b0 == 0.0:
            break

        # Scale to keep the total mass constant (sum = 1)
        b = b0 * (1.0 / sum_b0)

        # Mix the LP direction with the current vector
        prog_lp = (time.time() - eg_end) / (deadline - eg_end)
        t = t_start * (1.0 - prog_lp) + t_end * prog_lp
        a = (1.0 - t) * a + t * b

        # Renormalise (very small numerical drift may appear)
        a = np.maximum(a, eps)
        a = a / a.sum()

        # Update best record
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[LP {lp_iteration:5d}] new best = {best_val:.6f}")

        lp_iteration += 1

    # ------------------------------------------------------------------
    # Phase 3 – Short EG polishing (optional, time‑permitting)
    # ------------------------------------------------------------------
    polish_end = deadline
    while time.time() < polish_end:
        # identical to the EG loop but without the soft‑max temperature
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[polish] new best = {best_val:.6f}")

        # use a very small learning rate now
        eta = 5e-5
        # hard max => one‑hot weight
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        grad = _gradient_from_weights(a, w)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = -eta * momentum
        np.clip(step, -30, 30, out=step)
        a = a * np.exp(step)
        a = np.maximum(a, eps)
        a = a / a.sum()

    # ------------------------------------------------------------------
    # Final reporting (the outer framework will also evaluate)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()


# ----------------------------------------------------------------------
# Small utility to build a CSR matrix from pre‑computed index arrays.
# ----------------------------------------------------------------------
def _make_sparse_matrix(data: np.ndarray,
                        rows: np.ndarray,
                        cols: np.ndarray,
                        shape: tuple[int, int]) -> 'scipy.sparse.csr_matrix':
    """
    Construct a CSR matrix given the data and index vectors.
    Imported lazily to avoid importing scipy at module import time
    when it may not be needed.
    """
    import scipy.sparse
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=shape)
