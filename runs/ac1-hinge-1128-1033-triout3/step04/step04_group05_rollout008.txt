"""
Improved step‑function search.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) and
minimises the quantity 2·n·max_k(a*a)[k] – exactly the value returned by
``evaluate_sequence`` after normalisation.  The variables are kept in log‑space
(`x`, a = softmax(x)) which guarantees feasibility automatically.  A
soft‑max “temperature’’ τ is used to smooth the max‑operator; τ is driven
exponentially to zero so that the final stage uses the true sub‑gradient of
the max.  An Adam optimiser with a linearly decaying learning‑rate performs
the updates, while occasional random “jitter’’ moves help to escape plateaux.
The routine runs until the time‑budget expires (default 1000 s) and returns
the best sequence encountered.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper: full (linear) convolution via FFT (real‑valued signals)
# -------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution x*y (length len(x)+len(y)-1) using a real FFT.
    The result is clipped at zero to avoid tiny negative rounding errors.
    """
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    # next power of two for efficient FFT
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate round‑off negatives (should be non‑negative theoretically)
    np.maximum(conv, 0, out=conv)
    return conv


# -------------------------------------------------------------
# Gradient of the (soft‑max) max‑convolution w.r.t. the coefficients
# -------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a probability vector `a` (length n) and a weight vector `w`
    (length 2n‑1) representing a soft‑max over the convolution entries,
    return the gradient ∂/∂a  Σ_k w_k (a*a)[k].
    Implements the formula   g_i = 2·Σ_k w_k a_{k‑i}
    via a single convolution.
    """
    n = a.shape[0]
    u = w[::-1]                     # reverse the weight vector
    conv = _full_convolution_fft(u, a)   # length = 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)      # indices that correspond to the original n entries
    g = 2.0 * conv[idx]
    return g

# Main entry point required by the competition framework
# -------------------------------------------------------------

    Optimises a non‑negative step‑function (list of heights) within the given
    time‑budget.  Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the surrounding framework
    deadline = start_time + budget_s - 2.0

    # ---------------------------------------------------------
    # Choose a length and an initial point.
    # If a previous construction is available (height_sequence_1)
    # start from it, otherwise fall back to a uniform distribution.
    # ---------------------------------------------------------
    try:

        init_seq = np.asarray(height_sequence_1, dtype=float)
        n = init_seq.shape[0]
        a = np.maximum(init_seq, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
    except Exception:                               # pragma: no cover
        n = 1800
        a = np.full(n, 1.0 / n)

    # ---------------------------------------------------------
    # Log‑space representation (softmax guarantees a_i ≥ 0 and Σa_i = 1)
    # ---------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(a, eps))

    # Adam optimiser state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.90, 0.999

            a = a / a.sum()
            x = np.log(np.maximum(a, eps))

            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0
