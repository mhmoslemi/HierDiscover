1️⃣  **Broad seed exploration** – we generate a diverse pool of candidates:
   * symmetric Beta‑shapes (a_i ∝ x^{p‑1}(1‑x)^{p‑1}) for many exponents p,
   * geometric decays (a_i ∝ r^i + r^{n‑1‑i}),
   * a few random Dirichlet draws (different concentration parameters),
   * the historic construction (if present) and the uniform vector.
   Each seed is normalised to sum = 1 and its exact score is evaluated.
   The best few seeds are kept for local refinement.

2️⃣  **Local optimisation** – for each selected seed we run a cascade of
   optimisation phases, always staying on the probability simplex:
   * **Exact sub‑gradient descent** on the true max‑convolution entry.
     A simple back‑tracking line‑search guarantees descent.
   * **Adam on a smoothed log‑sum‑exp approximation** of the max.
     The smoothing parameter α is increased steadily to turn the smooth
     objective into the true one.
   * **Random‑swap hill‑climber** that moves a tiny amount of mass between
     two positions if it improves the exact objective.
   All intermediate improvements update the global best.

3️⃣  **Return the best vector** – clipped to the limits enforced by
   `evaluate_sequence` (0 ≤ x ≤ 1000) and returned as a plain Python list.

The whole routine respects the 1000 s budget, aborting early if the time
limit is reached.  The optimiser is deliberately simple, fast and
deterministic given the supplied random seed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Full self‑convolution via FFT (pre‑computed size)
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Return the full (length 2·len(p)‑1) convolution p * p."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)
    # keep only the valid part
    return conv[: 2 * p.size - 1]

# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp smoothed max (used by Adam)
# ----------------------------------------------------------------------
def _gradient_from_weights(p: np.ndarray,
                           p_fft: np.ndarray,
                           w: np.ndarray,
                           m_fft: int) -> np.ndarray:
    """Gradient of  f(p)=log‑sum‑exp(α·conv(p,p)) w.r.t. p."""
    # w is a probability vector of length 2·n‑1 (soft‑max weights)
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)
    # keep the part that corresponds to the needed range
    corr = corr[: 3 * p.size - 2]
    idx = 2 * p.size - 2 - np.arange(p.size)
    grad = 2.0 * corr[idx]
    return grad

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        base_seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
    except Exception:  # height_sequence_1 may be undefined
        base_seq = None

    n = len(base_seq) if base_seq is not None else 1153

    # FFT size – next power‑of‑two ≥ 2·n‑1
    m_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Objective helper (exact, without re‑normalisation)
    # --------------------------------------------------------------
    def _score(p: np.ndarray) -> float:
        """Exact evaluation used for comparing candidates (sum(p) should be 1)."""
        conv = _conv_self(p, m_fft)
        max_b = conv.max()
        # sum(p) is 1 (or very close), but keep the denominator for safety
        s = p.sum()
        return float(2.0 * n * max_b / (s * s))

    # --------------------------------------------------------------
    # 1️⃣  Seed generation
    # --------------------------------------------------------------
    seeds = []

    # Uniform
    seeds.append(np.full(n, 1.0 / n))

    # Historic (if available)
    if base_seq is not None:
        seq = np.clip(base_seq, 0.0, 1000.0)
        if seq.sum() > 0.0:
            seeds.append(seq / seq.sum())

    # Symmetric Beta shapes (a_i ∝ x^{p‑1}(1-x)^{p‑1})
    xs = (np.arange(n, dtype=float) + 0.5) / n
    for p in np.linspace(0.2, 1.0, 17):          # 0.2 … 1.0
        xp = np.clip(xs, 1e-12, 1.0 - 1e-12)
        a = (xp ** (p - 1.0)) * ((1.0 - xp) ** (p - 1.0))
        a = a / a.sum()
        seeds.append(a)

    # Skewed Beta shapes (α ≠ β)
    for α, β in [(0.2, 1.0), (1.0, 0.2), (0.3, 0.8), (0.8, 0.3)]:
        xp = np.clip(xs, 1e-12, 1.0 - 1e-12)
        a = (xp ** (α - 1.0)) * ((1.0 - xp) ** (β - 1.0))
        a = a / a.sum()
        seeds.append(a)

    # Geometric decays (both sides)
    for r in np.linspace(0.85, 0.99, 8):
        i = np.arange(n, dtype=float)
        geom = r ** i + r ** (n - 1 - i)
        geom = np.clip(geom, 0.0, None)
        geom = geom / geom.sum()
        seeds.append(geom)

    # Random Dirichlet draws (different concentration parameters)
    for conc in [0.1, 0.3, 0.6, 1.0]:
        for _ in range(4):
            a = rng.gamma(conc, 1.0, size=n)
            if a.sum() == 0:
                continue
            a = a / a.sum()
            seeds.append(a)

    # --------------------------------------------------------------
    # Evaluate all seeds and keep the best few
    # --------------------------------------------------------------
    # We'll keep at most 5 best seeds for intensive refinement
    scored_seeds = []
    for s in seeds:
        score = _score(s)
        scored_seeds.append((score, s))
    scored_seeds.sort(key=lambda x: x[0])
    top_k = min(5, len(scored_seeds))
    best_seeds = [v for (_, v) in scored_seeds[:top_k]]

    # --------------------------------------------------------------
    # Global best tracking
    # --------------------------------------------------------------
    best_score = float("inf")
    best_vec = None

    # Time budget handling
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Helper: exact sub‑gradient descent with backtracking line‑search
    # --------------------------------------------------------------
    def _subgrad_opt(p0: np.ndarray,
                     max_iters: int = 8000,
                     init_step: float = 0.5) -> tuple[np.ndarray, float]:
        p = p0.copy()
        best_local = p.copy()
        best_local_score = _score(p)
        step = init_step
        for it in range(max_iters):
            if time.time() >= deadline:
                break
            conv = _conv_self(p, m_fft)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val / (p.sum() ** 2)

            # keep local best
            if cur_score < best_local_score - 1e-14:
                best_local_score = cur_score
                best_local = p.copy()
                # early exit if we already beat the target
                if best_local_score <= 1.503:
                    break

            # active set (indices attaining the max up to tolerance)
            tol = 1e-13
            active = np.where(conv >= max_val - tol)[0]

            # sub‑gradient (average over active shifts)
            g = np.zeros_like(p)
            for idx in active:
                if idx < n:
                    g[: idx + 1] += 2.0 * p[idx::-1]
                else:
                    start = idx - n + 1
                    g[start:] += 2.0 * p[n - 1 : start - 1 : -1]
            g /= float(len(active))

            # back‑tracking line‑search
            dec = 0.5
            step_local = step
            while step_local > 1e-12:
                p_candidate = _simplex_projection(p - step_local * g)
                cand_score = _score(p_candidate)
                if cand_score < cur_score:
                    p = p_candidate
                    break
                step_local *= dec
            else:
                # no improvement – shrink the global step size and continue
                step *= 0.5
                continue

            # gradually decay the step size
            step *= 0.9997
            if step < 1e-9:
                break
        return best_local, best_local_score

    # --------------------------------------------------------------
    # Helper: Adam optimisation on smoothed log‑sum‑exp max
    # --------------------------------------------------------------
    def _adam_opt(p0: np.ndarray,
                  max_iters: int = 120000,
                  base_lr: float = 0.06,
                  alpha_max: float = 4000.0) -> tuple[np.ndarray, float]:
        p = p0.copy()
        best_local = p.copy()
        best_local_score = _score(p)

        m1 = np.zeros_like(p)
        m2 = np.zeros_like(p)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8

        alpha = 1.0           # smoothing factor
        alpha_mul = 1.002

        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # FFT of current p (re‑used for gradient)
            p_fft = np.fft.rfft(p, n=m_fft)
            conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

            # log‑sum‑exp soft‑max weights
            z = alpha * conv
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)          # subtract max for stability
            w = exp_z / np.sum(exp_z)          # w sums to 1

            # gradient of the smooth objective
            grad = _gradient_from_weights(p, p_fft, w, m_fft)

            # Adam moments
            m1 = beta1 * m1 + (1.0 - beta1) * grad
            m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

            # bias‑corrected estimates
            m1_hat = m1 / (1.0 - beta1 ** (it + 1))
            m2_hat = m2 / (1.0 - beta2 ** (it + 1))

            # step size shrinks as α grows
            lr = base_lr / np.sqrt(alpha)
            step_vec = lr * m1_hat / (np.sqrt(m2_hat) + eps)

            p = p - step_vec
            p = _simplex_projection(p)

            # increase smoothing
            alpha = min(alpha * alpha_mul, alpha_max)

            # occasional exact evaluation (every ≈ 2³ iterations)
            if (it & 1023) == 0:
                cur_score = _score(p)
                if cur_score < best_local_score - 1e-14:
                    best_local_score = cur_score
                    best_local = p.copy()
                    if best_local_score <= 1.503:
                        break
        return best_local, best_local_score

    # --------------------------------------------------------------
    # Helper: tiny random‑swap hill‑climber
    # --------------------------------------------------------------
    def _swap_hillclimb(p0: np.ndarray,
                        max_iters: int = 30000,
                        eps_scale: float = 0.002) -> tuple[np.ndarray, float]:
        p = p0.copy()
        cur_score = _score(p)
        best_local = p.copy()
        best_local_score = cur_score

        for _ in range(max_iters):
            if time.time() >= deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            eps_move = eps_scale * rng.random() * p[i]   # move a fraction of p[i]
            if eps_move <= 0.0:
                continue
            p_new = p.copy()
            p_new[i] -= eps_move
            p_new[j] += eps_move
            # keep numerical exactness
            p_new = _simplex_projection(p_new)
            new_score = _score(p_new)
            if new_score < cur_score - 1e-15:
                p = p_new
                cur_score = new_score
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local = p.copy()
                    if best_local_score <= 1.503:
                        break
        return best_local, best_local_score

    # --------------------------------------------------------------
    # Main refinement loop over the selected seeds
    # --------------------------------------------------------------
    for seed_id, seed_vec in enumerate(best_seeds, 1):
        if time.time() >= deadline:
            break
        print(f"\n=== Refinement of seed {seed_id}/{len(best_seeds)} ===")
        # ensure a clean probability vector
        p = np.maximum(seed_vec, 0.0)
        p = p / p.sum()

        # Phase A – exact sub‑gradient descent
        p_a, score_a = _subgrad_opt(p, max_iters=7000, init_step=0.5)
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_vec = p_a.copy()
            print(f"  Sub‑grad phase → new best {best_score:.6f}")

        # Phase B – Adam on smooth approximation
        p_b, score_b = _adam_opt(p_a, max_iters=100000, base_lr=0.07, alpha_max=3000.0)
        if score_b < best_score - 1e-12:
            best_score = score_b
            best_vec = p_b.copy()
            print(f"  Adam phase   → new best {best_score:.6f}")

        # Phase C – final sub‑gradient polishing (shorter)
        p_c, score_c = _subgrad_opt(p_b, max_iters=3000, init_step=0.15)
        if score_c < best_score - 1e-12:
            best_score = score_c
            best_vec = p_c.copy()
            print(f"  Polishing    → new best {best_score:.6f}")

        # Phase D – random swap hill‑climber
        p_d, score_d = _swap_hillclimb(p_c, max_iters=20000, eps_scale=0.0015)
        if score_d < best_score - 1e-12:
            best_score = score_d
            best_vec = p_d.copy()
            print(f"  Swaps        → new best {best_score:.6f}")

        # Early exit if target already reached
        if best_score <= 1.503:
            print("Target achieved – stopping early.")
            break

    # ------------------------------------------------------------------
    # Fallback: if nothing succeeded (should never happen)
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)
        best_score = _score(best_vec)

    print(f"\nFinal best score: {best_score:.6f}")

    # Clip to the limits imposed by evaluate_sequence (0 ≤ x ≤ 1000)
    return np.clip(best_vec, 0.0, 1000.0).tolist()
