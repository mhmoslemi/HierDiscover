"""
Minimising ``evaluate_sequence`` by a two‑stage optimisation.

Stage 1 – smoothed L‑BFGS‑B.
    * Re‑parameterise the non‑negative step heights by a soft‑max
      (guarantees Σa = 1).
    * Optimise a smooth log‑sum‑exp approximation of the max‑convolution.
    * Several continuation values for the smoothing parameter ε are used.
    * A few different initialisations (uniform, edge‑heavy, interpolated
      previous best, random Dirichlet) are tried – the best result among
      them is kept.

Stage 2 – exact sub‑gradient descent on the true objective.
    * The exact objective for Σa = 1 is 2·n·max(a∗a); this is cheap to
      evaluate with an FFT based full convolution.
    * At each iteration the lag k where the convolution attains its
      maximum is found and the sub‑gradient ∂max/∂a (which is
      a_{k‑i}+a_{i‑k}) is computed analytically.
    * A multiplicative mirror‑descent update ``a ← a·exp(‑η·g)`` keeps
      all entries non‑negative; the vector is renormalised to sum 1.
    * The step size η is reduced if the update does not improve the
      objective.  The process runs until the time budget expires.

Both stages use only FFT based convolutions (O(n log n)), so the
algorithm can afford a fairly large number of steps.  The final
candidate is returned as a plain ``list[float]`` – the external
``evaluate_sequence`` routine will compute the exact score.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (needed for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself (length 2·len‑1).
    Implemented with an FFT of size a power of two for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative noise introduced by floating point errors
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors (length len(x)+len(y)-1)
    via FFT.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. the soft‑max
# parameters w (L‑BFGS works on w)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Return (L_ε(w), ∂L_ε/∂w) for a given vector of soft‑max parameters w.
    ``eps`` is the smoothing parameter.
    """
    # ---- soft‑max → feasible a (Σa = 1, a_i ≥ 0) ----
    w_shift = w - np.max(w)                     # for numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # ---- full self‑convolution (a*a) ----
    conv = _full_convolution(a)                 # length 2·n‑1
    max_conv = conv.max()

    # ---- smoothed max via log‑sum‑exp ----
    # L = 2·n·ε·log(∑_k exp(conv_k/ε))
    # use the standard shifted formulation
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)                # = log ∑ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (closed form) ----
    # p_k = exp(conv_k/ε) / Σ_j exp(conv_j/ε)
    weights = np.exp(shifted)
    p = weights / weights.sum()                 # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)               # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]             # middle n entries
    grad_a = 4.0 * n * seg[::-1]                # shape (n,)

    # ---- chain rule for the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)                 # ∂L/∂w
    return L, grad_w


# ----------------------------------------------------------------------
# Exact sub‑gradient of the true objective at the current maximising lag
# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)[k] at the index ``k`` where the maximum is
    attained.  The gradient w.r.t. a_i equals a_{k‑i} (if in range) plus
    a_{i‑k} (if in range).  Returned vector has shape (n,).
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    # contribution a_{k‑i}
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        i = np.arange(lo, hi + 1)
        g[i] += a[k - i]

    # contribution a_{i‑k}
    lo2 = max(k, 0)
    hi2 = n - 1
    if lo2 <= hi2:
        i2 = np.arange(lo2, hi2 + 1)
        g[i2] += a[i2 - k]

    return g


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact objective (mirrored multiplicative update)
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         start_time: float,
                         deadline: float,
                         n: int,
                         best_a: np.ndarray,
                         best_val: float):
    """
    Perform many cheap sub‑gradient steps using the exact max‑convolution.
    ``a`` is the starting point (already normalised).  The routine updates
    ``best_a``/``best_val`` whenever a better vector is found.
    """
    cur_a = a.copy()
    cur_val = 2.0 * n * _full_convolution(cur_a).max()   # exact objective
    if cur_val < best_val:
        best_val = cur_val
        best_a = cur_a.copy()
        print(f"[sgd init] new best = {best_val:.6f}")

    # initial multiplicative step size (log‑space)
    eta = 0.5

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_conv = conv.max()
        k = int(conv.argmax())

        # compute sub‑gradient at the worst lag
        g = _grad_max_conv(cur_a, k)

        # attempt a multiplicative update; halve η until improvement
        improved = False
        eta_try = eta
        while eta_try > 1e-8:
            cand = cur_a * np.exp(-eta_try * g)
            s = cand.sum()
            if s == 0.0:
                eta_try *= 0.5
                continue
            cand /= s                               # renormalise to Σ=1
            cand_val = 2.0 * n * _full_convolution(cand).max()
            if cand_val < cur_val - 1e-12:          # strict improvement
                cur_a = cand
                cur_val = cand_val
                improved = True
                break
            eta_try *= 0.5

        if not improved:
            # no improvement → shrink the global step size a bit
            eta *= 0.7
            if eta < 1e-10:
                break
        else:
            # successful step → possibly enlarge η a little (but keep it modest)
            eta = min(eta * 1.05, 0.8)

        # record global best
        if cur_val < best_val:
            best_val = cur_val
            best_a = cur_a.copy()
            print(f"[sgd] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing (small random mass transfers) – a cheap extra polish
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       n: int):
    """
    Very simple random mass‑transfer hill‑climber that respects Σa = 1.
    Used only after the main optimisation stages to squeeze out remaining
    improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a fraction of mass from i to j (up to 5 %)
        delta = cur_a[i] * 0.05 * rng.random()
        if delta <= 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_val = 2.0 * n * _full_convolution(new_a).max()
        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    The strategy combines soft‑max L‑BFGS optimisation,
    exact sub‑gradient descent and a final random refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    # --------------------------------------------------------------
    # Problem size – larger n yields a lower theoretical bound.
    # --------------------------------------------------------------
    n = 8000                                         # chosen to push the bound closer to 1

    # --------------------------------------------------------------
    # Build a small collection of diverse starting points.
    # --------------------------------------------------------------
    starts = []

    # 1) uniform distribution
    starts.append(np.full(n, 1.0 / n))

    # 2) edge‑heavy shape a_i ∝ 1/√((i+1)*(n-i))
    idx = np.arange(n)
    edge_shape = 1.0 / np.sqrt((idx + 1) * (n - idx))
    edge_shape /= edge_shape.sum()
    starts.append(edge_shape)

    # 3) interpolation of a previously found solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, old)
            interp = np.maximum(interp, 0.0)
            interp /= interp.sum()
            starts.append(interp)
        else:
            starts.append(old / old.sum())
    except Exception:
        pass

    # 4) a few random Dirichlet samples (concentration = 0.3 gives a more
    #    edge‑biased distribution)
    rng = np.random.default_rng(seed + 1234)
    for _ in range(2):
        rnd = rng.dirichlet(np.full(n, 0.3))
        starts.append(rnd)

    # --------------------------------------------------------------
    # Helpers for evaluating the exact objective (no call to evaluate_sequence)
    # --------------------------------------------------------------
    def exact_eval(a_vec: np.ndarray) -> float:
        """Return 2·n·max(a∗a) for a vector with Σa = 1."""
        return 2.0 * n * _full_convolution(a_vec).max()

    # ------------------------------------------------------------------
    # MAIN LOOP over different starts
    # ------------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # continuation schedule for the smooth approximation (log‑space)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(5e-8), num=14)

    for idx_start, a0 in enumerate(starts):
        if time.time() >= deadline:
            break

        # ensure feasibility
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / a0_sum

        w = np.log(a0 + 1e-12)                      # soft‑max parameters

        # L‑BFGS on the smoothed surrogate for each ε
        for eps in eps_schedule:
            if time.time() >= deadline:
                break

            # objective function returning (value, gradient) for L‑BFGS
            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 150,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})

            w = res.x

            # exact evaluation of the current iterate
            w_shift = w - np.max(w)
            a_cur = np.exp(w_shift)
            a_cur /= a_cur.sum()
            cur_val = exact_eval(a_cur)

            if cur_val < best_val:
                best_val = cur_val
                best_a = a_cur.copy()
                print(f"[start {idx_start}] eps={eps:.1e} → new best = {best_val:.6f}")

        # after the ε‑schedule, use the current a as a new starting point
        # for the exact sub‑gradient phase
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Exact sub‑gradient descent (fast because only FFT based convolutions)
        # ------------------------------------------------------------------
        a_start = a_cur.copy()
        best_a, best_val = _subgradient_descent(a_start,
                                                start_time,
                                                deadline,
                                                n,
                                                best_a,
                                                best_val)

    # ----------------------------------------------------------------------
    # Final cheap random hill‑climbing polish (only if time permits)
    # ----------------------------------------------------------------------
    if best_a is not None and time.time() < deadline:
        best_a, best_val = _hill_climb_refine(best_a,
                                              best_val,
                                              deadline,
                                              n)

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain list of floats)
    # ----------------------------------------------------------------------
    if best_a is None:
        # Fallback – uniform distribution (should never happen)
        best_a = np.full(n, 1.0 / n)

    # Final sanity check (optional)
    final_score = exact_eval(best_a)
    print(f"[final] best evaluation (exact) = {final_score:.6f}")

    return best_a.tolist()
