"""
Propose a candidate step‑function (non‑negative vector) that minimises

    2 * n * max_k (a * a)[k]

where * is the linear convolution and sum(a)=1 (the objective is
scale‑invariant, so we normalise to a probability simplex).

Algorithm
---------
1. **Deterministic sweep** over several symmetric parametric families
   (U‑shapes, shifted U‑shapes, product‑distance, exponential, cosine and
   simple mixtures).  A modest set of lengths and parameter grids is tried.
   The best vector from this stage seeds the stochastic phase.

2. **Stochastic refinement** on the simplex using a *smoothed* maximum.
   For a temperature τ we replace  max(conv)  by the smooth approximation

        Sτ(a) = τ·log Σ_k exp(conv_k/τ) ,

   whose gradient is a simple correlation that can be obtained with FFTs.
   We perform a mirror‑descent step w.r.t. the KL‑divergence
   (exponentiated‑gradient update) and accept the move with a simulated‑annealing
   rule.  τ, the step size η and the annealing temperature decay
   geometrically, and occasional random “re‑heat’’ moves avoid stagnation.

3. The best vector observed during the whole run is returned as a plain
   Python list.  All numbers are non‑negative and sum ≈ 1, satisfying the
   security checks of `evaluate_sequence`.

The routine respects the supplied time budget (default 1000 s) and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1. Helper utilities (FFT based convolution / correlation)
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a * a using a real FFT.
    Returns a NumPy array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlate w (length 2n-1) with a (length n) and return the
    'valid' part of length n:
        (w ⋆ a)[i] = Σ_j w[i + j] * a[j] .
    Computed via FFT for O(n log n) speed.
    """
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())           # length for FFT (≥ 3n‑2)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(W * np.conj(A), n=L)
    # valid slice starts at index (n‑1) and has length n
    return corr[n - 1 : n - 1 + n]


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact score returned by evaluate_sequence for a vector with sum(a)=1.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 2. Parametric families (all symmetric, generate probability vectors)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))‑p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


def mix_distribution(n: int, p1: float, p2: float, lam_mix: float) -> np.ndarray:
    """
    Simple convex mixture of a min‑distance U‑shape and a product‑distance U‑shape.
    lam_mix ∈ [0,1] gives weight on the first component.
    """
    a1 = u_min_distribution(n, p1)
    a2 = u_prod_distribution(n, p2)
    a = lam_mix * a1 + (1.0 - lam_mix) * a2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 3. Smoothed objective & gradient (soft‑max of convolution)
# ----------------------------------------------------------------------


def smooth_max_and_gradient(a: np.ndarray, tau: float):
    """
    Compute a τ‑smoothed approximation of max(conv) and its gradient
    w.r.t. a on the probability simplex.

    Returns
    -------
    smooth_val : float
        τ·log Σ_k exp(conv_k/τ)  (adds the max for numerical stability).
    conv : ndarray (2n‑1)
        Full self‑convolution of a (needed for the true score).
    grad : ndarray (n,)
        Gradient of the smooth max with respect to a.
    """
    n = a.shape[0]
    # Full self‑convolution
    conv = full_convolution_fft(a)                     # length 2n‑1
    max_c = conv.max()
    # Stable soft‑max weights
    shifted = conv - max_c
    # Prevent division by zero when τ is extremely tiny
    if tau < 1e-12:
        # pure max – weight concentrated on the argmax
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        smooth_val = max_c
    else:
        exps = np.exp(shifted / tau)
        sum_exps = exps.sum()
        # Guard against underflow when all exps vanish
        if sum_exps == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            smooth_val = max_c
        else:
            w = exps / sum_exps
            smooth_val = max_c + tau * np.log(sum_exps)

    # Gradient:  ∂S/∂a_i = Σ_k w_k * 2 a_{k‑i}
    # which is 2 * correlation of w with a (valid part).
    corr = correlation_fft(w, a)               # length n
    grad = 2.0 * corr
    return smooth_val, conv, grad


def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) step on the simplex:
        a' ∝ a * exp(-η * grad)
    The result is renormalised to sum = 1.
    """
    # Clip exponent to avoid overflow/underflow
    delta = -eta * grad
    delta = np.clip(delta, -50, 50)
    a_new = a * np.exp(delta)
    s = a_new.sum()
    if s == 0.0:
        # fallback – uniform distribution
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new


# ----------------------------------------------------------------------
# 4. Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the given
    time budget (seconds).  Returns the sequence as a plain Python list.
    The routine combines a deterministic parametric sweep with a
    smoothed‑gradient stochastic refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 4.0      # small safety margin

    # --------------------------------------------------------------
    # 1. Deterministic sweep over parametric families
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    lengths = [8000, 12000, 16000, 20000]        # moderate sizes fitting the budget
    exponents = np.arange(0.2, 5.01, 0.2)       # for U‑shapes
    shift_cs = [0.5, 1.0, 2.0, 5.0]            # shift constants
    lam_vals = np.logspace(-5, -0.3, 25)       # exponential decay
    cos_exps = np.arange(0.5, 8.5, 0.5)        # cosine powers
    mix_lams = [0.0, 0.25, 0.5, 0.75, 1.0]    # mixture weights

    print("=== Deterministic sweep start ===")
    for n in lengths:
        if time.time() > deadline:
            break
        # --- basic U‑shape families ---
        for p in exponents:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

            for c in shift_cs:
                a = u_min_shift_distribution(n, p, c)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"new best (shift) n={n:5d} p={p:4.2f} c={c:.1f}  score={sc:.6f}")

        # --- exponential family ---
        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (exp)   n={n:5d} lam={lam:.5f}  score={sc:.6f}")

        # --- cosine family ---
        for p in cos_exps:
            a = cos_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (cos)   n={n:5d} p={p:3.1f}  score={sc:.6f}")

        # --- mixture of two U‑shapes ---
        for p1 in (0.5, 1.0, 2.0):
            for p2 in (0.5, 1.0, 2.0):
                for lm in mix_lams:
                    a = mix_distribution(n, p1, p2, lm)
                    sc = score_from_vector(a)
                    if sc < best_score:
                        best_score, best_seq = sc, a.copy()
                        print(f"new best (mix)  n={n:5d} p1={p1:.1f} p2={p2:.1f} lam={lm:.2f} score={sc:.6f}")

        # quick time check
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 2. incorporate any externally provided construction (if available)
    # --------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0.0:
                external /= s
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score, best_seq = sc, external.copy()
                    print(f"using external height_sequence_1  score={sc:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 3. fallback in the unlikely case that nothing was generated
    # --------------------------------------------------------------
    if best_seq is None:
        n = 8000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f"fallback random seed  n={n}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Stochastic refinement using smoothed max (exponential‑gradient)
    # --------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]
    current = np.maximum(current, 0.0)
    current /= current.sum()
    _, current_conv, _ = smooth_max_and_gradient(current, tau=1e-3)   # cheap init
    current_score = float(2.0 * n * current_conv.max())

    # Hyper‑parameters (tuned empirically)
    temperature = 1.0                # for simulated‑annealing acceptance
    temp_decay = 0.9997
    eta = 0.7                        # step size for exponential‑gradient
    eta_decay = 0.9998
    tau = 0.02                       # initial smoothing temperature
    tau_decay = 0.9999
    min_tau = 5e-5

    iteration = 0
    last_improve = 0

    print("\n=== Stochastic refinement start ===")
    while time.time() < deadline:
        iteration += 1

        # ---- 4a. smooth objective & its gradient ----
        smooth_val, conv_cur, grad = smooth_max_and_gradient(current, tau)

        # ---- 4b. exponentiated‑gradient candidate ----
        cand = exponential_update(current, grad, eta)

        # ---- 4c. evaluate true objective for the candidate ----
        cand_score = score_from_vector(cand)

        # ---- 4d. acceptance test (simulated annealing) ----
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ---- 4e. schedule updates ----
        temperature *= temp_decay
        eta *= eta_decay
        tau = max(min_tau, tau * tau_decay)

        # ---- 4f. occasional re‑heat to escape stagnation ----
        if iteration - last_improve > 150000:
            bump = np.random.rand(n) * 1e-5
            current = np.maximum(current + bump, 0.0)
            current /= current.sum()
            current_score = score_from_vector(current)
            temperature = 1.0
            eta = 0.7
            tau = 0.02
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

        # quick deadline check
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 5. Return the best sequence found (as a plain list)
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
