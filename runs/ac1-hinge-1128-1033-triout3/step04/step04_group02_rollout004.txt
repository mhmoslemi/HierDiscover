def propose_candidate(seed=42, budget_s=1000, **kwargs):

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # choose problem size based on time availability: maybe 8000 if enough time.
    n = 8000

    # epsilon schedule define:
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=14)

    best_a = None
    best_val = np.inf

    # try different seeds
    rng = np.random.default_rng(seed)
    restarts = 5
    for restart in range(restarts):
        if time.time() >= deadline:
            break

        # initialization: start from uniform or previous known
        if restart ==0 and 'height_sequence_1' in globals():
            # try to load previous height sequence and interpolate
            try:
                old = np.asarray(height_sequence_1, dtype=float)
                if old.shape[0] != n:
                    old_x = np.linspace(0,1,old.shape[0])
                    new_x = np.linspace(0,1,n)
                    a0 = np.interp(new_x, old_x, old)
                else:
                    a0 = old.copy()
            except Exception:
                a0 = np.full(n, 1/n)
        else:
            # random positive distribution:
            # sample exponentials, then normalize sum=1.
            raw = rng.exponential(scale=1.0, size=n)
            a0 = raw / raw.sum()

        # ensure feasibility
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()

        w = np.log(a0 + 1e-12)

        # smoothing loop
        for eps in eps_schedule:
            if time.time() >= deadline:
                break

            # define function
            def fun_and_grad(w_vec):
                L, g = _smooth_obj_and_grad(w_vec, eps, n)
                return L, g

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 300,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        # evaluate exact
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()
        try:
            val = evaluate_sequence(a.tolist())
        except Exception:
            val = np.inf

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[restart {restart}] new best = {best_val:.6f}")

        # Optionally run hill climbing on this candidate to improve
        if time.time() < deadline:
            a_refined, val_refined = _refine_hill_climb(a, val, deadline, n)
            if val_refined < best_val:
                best_val = val_refined
                best_a = a_refined.copy()
                print(f"[restart {restart} hill] new best = {best_val:.6f}")

    # final hill climbing on overall best for remaining time
    if best_a is not None and time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    print(f"[final] best evaluation = {best_val:.6f}")
    return best_a.tolist()

"""
Optimisation of the step‑function evaluation (``evaluate_sequence``).

The algorithm works as follows:

*  Choose a problem size ``n`` (we use a relatively large value, 8000,
   which gives the optimiser many degrees of freedom).
*  Represent the heights with a soft‑max, i.e.
   a_i = exp(w_i) / Σ_j exp(w_j).  This guarantees a_i ≥ 0 and Σ a_i = 1,
   which is sufficient because the evaluation function is scale‑invariant.
*  minimise a smooth log‑sum‑exp surrogate of the max‑convolution

        L_ε(w) = 2·n·ε·log(∑_k exp(conv_k/ε))

   where ``conv = a*a`` (full auto‑convolution).  The gradient of ``L_ε``
   w.r.t. the soft‑max parameters is obtained analytically using two FFT‑based
   convolutions.
*  A continuation schedule (geometric decay of ε) is used and each stage is
   solved with L‑BFGS‑B.  Several random restarts improve the chance of
   escaping poor local minima.
*  After the continuation phase a cheap stochastic hill‑climbing refinement
   (random mass transfers that keep the total mass constant) is performed
   until the time budget expires.
*  The best exact value seen (via ``evaluate_sequence``) is returned.

The whole routine respects the 1000 s budget and prints progress
information.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # clip tiny negative round‑off errors
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective and its gradient (w.r.t. the soft‑max parameters)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Return (L_ε, ∂L_ε/∂w) for a given parameter vector w and smoothing ε.
    The gradient is computed by the chain rule through the soft‑max.
    """
    # ---- soft‑max (a_i ≥ 0, Σ a_i = 1) ----------
    w_shift = w - np.max(w)               # numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                         # now Σ a = 1

    # ---- full auto‑convolution a*a ----------
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # ---- smoothed max (log‑sum‑exp) ------
    # L = 2·n·ε·log(∑ exp(conv/ε))
    # we use a shifted formulation to avoid overflow.
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # = log∑ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (closed form) ------
    # p_k = exp(conv_k/ε) / Σ_j exp(conv_j/ε)   (soft‑max over the convolution)
    # using the shifted values we have:
    weights = np.exp(shifted)
    p = weights / weights.sum()           # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}  (indices outside [0,n‑1] give 0)
    # Implemented via a convolution with the reversed p.
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)         # length 3·n‑2
    # middle n entries correspond to the required sums
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # shape (n,)

    # ---- chain rule (soft‑max) -----------------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)           # ∂L/∂w
    return L, grad_w


# ----------------------------------------------------------------------
# Hill‑climbing refinement (random mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float, n: int):
    """
    Perform cheap stochastic hill‑climbing: repeatedly move a small random
    fraction of mass from one index to another.  Updates are kept only when
    the exact objective improves.
    """
    cur_a = a.copy()
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Transfer up to 10 % of the source mass.
        delta = cur_a[i] * (0.02 + 0.08 * rng.random())
        if delta == 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # Exact evaluation (may be a bit expensive but we call it only on
        # promising candidates).
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a step‑function under the given time budget and return the
    best sequence of heights found.  The implementation combines a
    continuation L‑BFGS‑B optimisation of a smooth surrogate for the
    max‑convolution with a final stochastic hill‑climbing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final print statements
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a larger n gives more flexibility.
    # --------------------------------------------------------------
    n = 8000   # chosen to stay well within the 1000 s budget

    # --------------------------------------------------------------
    # Helper to initialise a feasible starting vector.
    # --------------------------------------------------------------
    def _initial_a(restart_idx: int):
        # Try to reuse a previously stored construction if available.
        if restart_idx == 0:
            try:
                # ``height_sequence_1`` may be defined in the global scope.
                old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
                if old.shape[0] != n:
                    old_x = np.linspace(0.0, 1.0, old.shape[0])
                    new_x = np.linspace(0.0, 1.0, n)
                    a0 = np.interp(new_x, old_x, old)
                else:
                    a0 = old.copy()
                a0 = np.maximum(a0, 0.0)
                a0 = a0 / a0.sum()
                return a0
            except Exception:
                pass
        # Random positive start (exponential distribution → strictly >0)
        rng = np.random.default_rng(seed + restart_idx + 1)
        raw = rng.exponential(scale=1.0, size=n)
        a0 = raw / raw.sum()
        return a0

    best_a = None
    best_val = np.inf

    # --------------------------------------------------------------
    # Continuation schedule for ε (log‑space).  The last value is tiny,
    # giving a close approximation of the true max.
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=14)

    # --------------------------------------------------------------
    # Perform several independent restarts (each a full L‑BFGS run).
    # --------------------------------------------------------------
    max_restarts = 6
    for restart in range(max_restarts):
        if time.time() >= deadline:
            break

        a0 = _initial_a(restart)
        # soft‑max parameters
        w = np.log(a0 + 1e-12)

        # ----- loop over smoothing parameters -----
        for eps in eps_schedule:
            if time.time() >= deadline:
                break

            # The objective is smooth, we can hand it to SciPy directly.
            def _fun_and_grad(vec):
                return _smooth_obj_and_grad(vec, eps, n)

            # L‑BFGS‑B; we keep the iteration budget modest so that the overall
            # runtime stays within limits.
            res = minimize(_fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 250,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        # ----- evaluate the exact objective for this restart -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()
        try:
            val = evaluate_sequence(a.tolist())
        except Exception:
            val = np.inf

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[restart {restart}] new best (exact) = {best_val:.6f}")

        # ----- optional hill‑climbing on the current candidate -----
        if time.time() < deadline:
            a_ref, val_ref = _refine_hill_climb(a, val, deadline, n)
            if val_ref < best_val:
                best_val = val_ref
                best_a = a_ref.copy()
                print(f"[restart {restart}] post‑refine best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final hill‑climbing on the overall best solution.
    # --------------------------------------------------------------
    if best_a is not None and time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # --------------------------------------------------------------
    # Final reporting
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
