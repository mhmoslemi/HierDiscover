Pseudo:

def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho where u_i > cssv_i / i+1
    rho = np.where(u > cssv / np.arange(1, n+1))[0]
    if rho.size == 0:
        # fallback: all zero, set uniform?
        rho = 0
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function sequence.

The method improves on the previous Adam‑softmax approach by
  * starting from the best known construction (``height_sequence_1``) if it
    exists,
  * using a two‑phase schedule: an initial “soft‐max” phase for exploration,
    followed by a pure “hard‑max” sub‑gradient phase that directly attacks the
    current maximal convolution entry,
  * employing mirrored Adam updates in the log‑space (softmax) parametrisation,
    which automatically respects non‑negativity and the simplex constraint,
  * adding occasional random mass‑transfer moves and a stagnation‑based restart
    to escape plateaux.

The algorithm runs until the 1000 s budget (minus a small safety margin) is
exhausted and returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution via FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution x*y using real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # tiny negative values caused by numerical noise are removed
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of Σ w_k (a*a)[k] w.r.t. a  (w is a weight vector)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute   g = 2 * Σ_k w_k * a_{k-i}   for all i.
    Implemented via a single convolution: reverse w and convolve with a.
    """
    n = a.shape[0]
    # reverse the weight vector
    u = w[::-1]
    conv = _full_convolution_fft(u, a)           # length 3n‑2
    # indices that correspond to the original n‑dimensional gradient
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Projection onto the probability simplex (used only after a random jitter)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho index: the last i where u_i > cssv_i / i
    rho = np.where(u > cssv / ind)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
