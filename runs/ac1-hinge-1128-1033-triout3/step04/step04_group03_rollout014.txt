import cvxpy as cp

def lp_optimize(a, M):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # constraints: convolution (a*b) <= M for all k
    # compute convolution as sum_i a_i * b_{k - i}
    # express as linear constraints:
    conv_expr = []
    for k in range(2*n-1):
        expr = 0
        i_low = max(0, k - (n-1))
        i_high = min(n-1, k)
        # sum over i
        for i in range(i_low, i_high+1):
            j = k - i
            expr += a[i] * b[j]
        conv_expr.append(expr <= M)
    # sum(b) = 1 (or any constant)
    constraints = conv_expr + [cp.sum(b) == 1]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver='ECOS', warm_start=True, max_iters=5000)
    return b.value

"""
Propose a candidate step‑function (non‑negative vector) that minimises

    2 * n * max_k (a * a)[k]   (a * a – full linear convolution)

The algorithm consists of three stages:

1️⃣  Quick deterministic sweep over a few classic parametric families
    (U‑shape based on edge distance, exponential decay and cosine‑based).
    The best vector from this sweep seeds the optimisation.

2️⃣  Optional use of the global variable ``height_sequence_1`` (if it exists)
    – it already contains a very strong construction.

3️⃣  Stochastic refinement on the probability simplex.
    * A smoothed‑max objective is built from the convolution (softmax with
      a temperature β that slowly increases).  Its gradient is obtained by an
      FFT‑based cross‑correlation, giving an inexpensive “soft” sub‑gradient.
    * An exponentiated‑gradient (mirror descent w.r.t. KL) step updates the
      distribution, followed by a simulated‑annealing acceptance test.
    * Every few hundred iterations a targeted mass‑transfer move, guided by
      the hard sub‑gradient of the true max, is tried – this aggressively
      flattens the peak of the convolution.
    * After long stagnation a small random jitter (re‑heat) is applied.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence found (as a plain Python list whose entries sum to one).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w>=0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 2 * n                     # power‑of‑2 not required, but convenient
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector with sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Hard sub‑gradient of max_k (a*a)[k] w.r.t. a (uses only active indices)."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def smooth_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max convolution objective.
    w – softmax weights (length 2n‑1), sum(w)=1.
    Returns a vector of length n.
    """
    n = a.shape[0]
    # cross‑correlation (w ⋆ a)[i] = Σ_j w[i+j] a[j]
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)          # length 3n‑2
    offset = len(w) - 1                       # = 2n‑2
    grad = 2.0 * conv_rev[offset - np.arange(n)]
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent / exponentiated‑gradient step on the simplex:
        a' ∝ a * exp(-η·grad)
    """
    # clip gradient to avoid overflow in the exponent
    grad = np.clip(grad, -1e6, 1e6)
    exp_term = np.exp(-eta * grad)
    a_new = a * exp_term
    s = a_new.sum()
    if s == 0.0 or not np.isfinite(s):
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new

# Parametric families (all symmetric and normalised to sum = 1)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))^-p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for the best non‑negative height sequence within a given time
    budget (default 1000 s).  Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0    # safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over a few parametric families (lightweight)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # length candidates – a compromise between quality and speed
    lengths = [4000, 6000, 8000, 12000]

    # a very compact grid of parameters – enough to find a decent seed
    for n in lengths:
        if time.time() > deadline:
            break

        # ---- U‑shape based on min‑distance ----
        for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"init (U‑min)  n={n:5d} p={p:3.1f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"init (U‑prod) n={n:5d} p={p:3.1f}  score={sc:.6f}")

        # ---- shifted U‑shape ----
        for p in [0.8, 1.2, 2.0]:
            for c in [1.0, 2.0, 5.0]:
                a = u_min_shift_distribution(n, p, c)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"init (shift) n={n:5d} p={p:3.1f} c={c:3.1f} score={sc:.6f}")

        # ---- exponential decay ----
        for lam in [1e-4, 5e-4, 1e-3, 5e-3]:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"init (exp)    n={n:5d} lam={lam:.5f}  score={sc:.6f}")

        # ---- cosine‑based ----
        for p in [0.5, 1.0, 2.0]:
            a = cos_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"init (cos)    n={n:5d} p={p:3.1f}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate the external construction if it exists
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            external /= external.sum()
            sc = score_from_vector(external)
            if sc < best_score:
                best_score, best_seq = sc, external.copy()
                print(f"using external height_sequence_1  score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Random fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f"fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional up‑sampling to a larger length (may improve flatness)
    # ------------------------------------------------------------------
    def upsample(vec, new_n):
        old_n = vec.shape[0]
        x_old = np.linspace(0, 1, old_n)
        x_new = np.linspace(0, 1, new_n)
        up = np.interp(x_new, x_old, vec)
        up = np.maximum(up, 0.0)
        up /= up.sum()
        return up

    # try a single larger length (≈ 1.5×) and keep it if it helps
    for factor in [1.2, 1.5]:
        cand_n = int(round(best_seq.shape[0] * factor))
        if cand_n > best_seq.shape[0] and time.time() < deadline:
            cand = upsample(best_seq, cand_n)
            sc = score_from_vector(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"upsampled to n={cand_n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Stochastic refinement (soft‑max gradient + occasional targeted moves)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    a = np.maximum(a, 0.0)
    a /= a.sum()

    conv = full_convolution_fft(a)
    score = float(2.0 * n * conv.max())

    # hyper‑parameters
    beta = 0.5               # soft‑max temperature (starts low, increases)
    beta_max = 5000.0
    beta_factor = 1.0015

    eta = 0.3                # step size for exponentiated‑gradient
    eta_decay = 0.9999
    eta_min = 1e-5

    temperature = 1.0
    temp_decay = 0.9997

    iteration = 0
    last_improve = 0

    print("\n=== Stochastic refinement start ===")
    while time.time() < deadline:
        iteration += 1

        # ---- 5a. convolution & soft‑max weights ----
        conv = full_convolution_fft(a)
        maxc = conv.max()
        # soft‑max (avoid overflow by subtracting max)
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ---- 5b. smooth gradient (cross‑correlation) ----
        grad = smooth_gradient(a, w)

        # ---- 5c. exponentiated‑gradient candidate ----
        cand = exponential_update(a, grad, eta)

        # ---- 5d. evaluate candidate ----
        cand_conv = full_convolution_fft(cand)
        cand_score = float(2.0 * n * cand_conv.max())

        # ---- 5e. occasional targeted mass‑transfer move (hard sub‑gradient) ----
        if iteration % 200 == 0:
            subgrad = compute_subgradient(a, conv)
            i = int(np.argmax(subgrad))
            j = int(np.argmin(subgrad))
            if a[i] > 1e-12:
                eps = a[i] * 0.25               # move up to 25 % of the source mass
                cand2 = a.copy()
                cand2[i] -= eps
                cand2[j] += eps
                cand2 = np.maximum(cand2, 0.0)
                cand2 /= cand2.sum()
                cand2_conv = full_convolution_fft(cand2)
                cand2_score = float(2.0 * n * cand2_conv.max())
                # keep the better of the two candidates
                if cand2_score < cand_score:
                    cand, cand_conv, cand_score = cand2, cand2_conv, cand2_score

        # ---- 5f. Simulated‑annealing acceptance ----
        accept = False
        if cand_score < score:
            accept = True
        else:
            prob = np.exp((score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a, conv, score = cand, cand_conv, cand_score
            if cand_score < best_score - 1e-12:
                best_score, best_seq, last_improve = cand_score, cand.copy(), iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ---- 5g. schedule updates ----
        beta = min(beta * beta_factor, beta_max)
        eta = max(eta * eta_decay, eta_min)
        temperature *= temp_decay

        # ---- 5h. re‑heat after long stagnation ----
        if iteration - last_improve > 250000:
            jitter = np.random.rand(n) * 1e-5
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            conv = full_convolution_fft(a)
            score = float(2.0 * n * conv.max())
            temperature = 1.0
            eta = 0.3
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

        # ---- 5i. periodic small random perturbation (helps escape tiny plateaus) ----
        if iteration % 5000 == 0:
            perturb = np.random.rand(n) * 5e-6
            a = np.maximum(a + perturb, 0.0)
            a /= a.sum()
            # no need to recompute conv here – it will be done at next loop start

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (plain Python list)
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
