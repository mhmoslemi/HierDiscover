Algorithm (Duchi et al.):

def simplex_proj(v):
    # ensures sum(v) = 1, v >= 0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that (approximately) minimizes
``evaluate_sequence``.

Algorithm
---------
1. Choose a problem size `n` (default 4000).  The objective is scale‑invariant,
   so we enforce the simplex constraint Σ a_i = 1.
2. Multiple *restarts* are performed.
   * Initialise a with a random Dirichlet vector (α=0.5 → sparse solutions).
   * Optimise a smooth surrogate of `max(conv(a,a))` using L‑BFGS‑B on a
     soft‑max parametrisation.  The surrogate is the log‑sum‑exp
     approximation with a continuation schedule for the smoothing parameter.
   * From the final point of L‑BFGS run a projected sub‑gradient descent on the
     true max‑convolution (the sub‑gradient is obtained by a single FFT‑based
     convolution).  After each step we project onto the probability simplex.
   * A cheap hill‑climbing refinement (random mass transfers) is applied until
     the remaining time budget is exhausted.
3. The best sequence found over all restarts is returned.

All heavy operations (convolutions, gradients) are implemented with FFTs,
so even for `n≈4000` many thousands of iterations fit easily into the
1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors (len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smooth surrogate of max‑convolution and its gradient (w.r.t. soft‑max)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L_ε, ∂L_ε/∂w) for a given soft‑max parameter vector `w`
    and smoothing parameter `eps`.  `L_ε = 2·n·ε·log∑_k exp(conv_k/ε)`.
    The gradient is computed analytically via FFTs.
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)          # numeric stability
    a = np.exp(w_shift)
    a /= a.sum()                     # a_i ≥ 0, Σa_i = 1

    # full self‑convolution a*a
    conv = _full_convolution(a)      # length 2·n‑1
    max_conv = conv.max()

    # smoothed max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)     # = log∑ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()      # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}  (indices outside [0,n‑1] give 0)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)   # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]  # middle n entries
    grad_a = 4.0 * n * seg[::-1]     # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=float)
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform solution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the (non‑smoothed) max‑convolution term
# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. `a`.  `mask` should be a 0/1
    array of length 2·n‑1 indicating which convolution entries are
    currently maximal (or within a tiny tolerance).
    Returns a vector of length n.
    """
    n = a.shape[0]
    mask_rev = mask[::-1]
    conv = _conv_fft(a, mask_rev)                 # length 3·n‑2
    seg = conv[n - 1: 2 * n - 1]                  # middle n entries
    grad = 2.0 * seg[::-1]                        # factor 2 from derivative
    return grad
