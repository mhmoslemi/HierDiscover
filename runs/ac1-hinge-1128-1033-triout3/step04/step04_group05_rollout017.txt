gives the optimiser a better starting point).  Within a phase the
objective 2·n·max(conv) (the value returned by ``evaluate_sequence``
for a list with sum = 1) is minimised by optimiser Adam on a
log‑softmax representation of the sequence (automatically satisfying
non‑negativity and the simplex constraint).  The “max’’ is smoothed
by a soft‑max with a temperature that anneals from 0.4·max to
≈10⁻⁶·max, which provides a stable sub‑gradient.  Occasionally a
random “jitter’’ (mass transfer between two positions) is applied
to escape plateaux.  The best sequence seen over the whole budget is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full (linear) convolution via FFT – fast for many calls
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution ``x * y`` (length ``len(x)+len(y)-1``)
    using a real FFT.  The result is clipped at zero to avoid tiny
    negative rounding errors.
    """
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()       # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of a weighted sum of convolution entries
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    For a vector ``a`` and a weight vector ``w`` (same length as
    ``a*a``) compute ``∇_a  Σ_k w_k (a*a)[k]``.
    This is obtained by a single convolution with the reversed weight.
    """
    n = a.shape[0]
    u = w[::-1]                                     # reverse
    conv = _full_convolution_fft(u, a)              # length = 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)                      # n entries that we need
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Upsample a shorter sequence by inserting zeros between entries
# ----------------------------------------------------------------------
def _upsample_sequence(old_a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Insert zeros between the old entries so that the resulting vector
    has length ``new_n`` (``new_n`` must be >= len(old_a)``).  The mass
    of the original vector is preserved; the new vector is then normalised
    to sum‑to‑one.
    """
    old_n = old_a.shape[0]
    if new_n == old_n:
        return old_a.copy()
    new_a = np.zeros(new_n, dtype=old_a.dtype)
    # integer spacing – a simple deterministic mapping
    factor = new_n // old_n
    if factor >= 1:
        for i, v in enumerate(old_a):
            new_a[i * factor] = v
    else:       # fallback – stretch by interpolation (should not happen)
        positions = np.linspace(0, new_n - 1, old_n).astype(int)
        new_a[positions] = old_a
    # renormalise (tiny rounding errors may appear)
    s = new_a.sum()
    if s > 0:
        new_a /= s
    else:       # all zero – revert to uniform
        new_a[:] = 1.0 / new_n
    return new_a


# ----------------------------------------------------------------------
# Core optimisation for a single phase
# ----------------------------------------------------------------------
def _optimise_phase(n: int,
                    init_a: np.ndarray,
                    phase_deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a length‑``n`` sequence starting from ``init_a`` until
    ``phase_deadline`` (absolute ``time.time``).  Returns the best
    sequence seen and its objective value (2·n·max(convolution)).
    """
    # ensure feasibility
    a = init_a.copy()
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    # log‑softmax representation (guarantees simplex)
    eps = 1e-12
    x = np.log(np.maximum(a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.90, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps_adam = 1e-8
    base_lr = 0.30

    # bookkeeping
    conv = _full_convolution_fft(a, a)
    cur_val = float(2.0 * n * conv.max())
    best_val = cur_val
    best_a = a.copy()
    print(f"[init] n={n}, objective={best_val:.6f}")

    phase_start = time.time()
    phase_len = phase_deadline - phase_start
    iteration = 0

    while True:
        now = time.time()
        if now >= phase_deadline:
            break

        iteration += 1

        # ------------------------------------------------------------------
        # convolution and current objective
        # ------------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best (n={n}) = {best_val:.6f}")

        # ------------------------------------------------------------------
        # schedule (progress in [0,1])
        # ------------------------------------------------------------------
        progress = (now - phase_start) / phase_len

        # temperature for the soft‑max (relative to the current maximum)
        tau_start = M * 0.40
        tau_end   = M * 1e-6
        tau = tau_start * (1.0 - progress) + tau_end * progress
        if tau <= 0.0:
            tau = 1e-12

        # ------------------------------------------------------------------
        # soft‑max weights over the convolution entries
        # ------------------------------------------------------------------
        exp_args = (conv - M) / tau
        max_arg = exp_args.max()
        exp_vals = np.exp(exp_args - max_arg)          # shift for stability
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_vals / sum_exp

        # ------------------------------------------------------------------
        # gradient w.r.t. a (smoothed sub‑gradient of the max)
        # ------------------------------------------------------------------
        grad_a = _gradient_from_weights(a, w)

        # chain rule for the log‑softmax parametrisation
        dot = np.dot(grad_a, a)               # <grad_a , a>
        grad_x = a * (grad_a - dot)

        # ------------------------------------------------------------------
        # Adam update (learning‑rate decays linearly)
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = base_lr * (1.0 - progress) + 1e-5 * progress
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑to probability simplex
        max_x = x.max()
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ------------------------------------------------------------------
        # occasional random mass‑transfer (helps to escape plateaux)
        # ------------------------------------------------------------------
        # probability of a jitter increases early in the phase
        jitter_prob = 0.09 * (1.0 - progress) + 0.02
        if np.random.rand() < jitter_prob:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta
            a = a / a.sum()
            max_x = a.max()
            x = np.log(np.maximum(a, eps))
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises the evaluation function.  The search runs three phases,
    each with a larger vector size, re‑using the best solution from the
    previous phase via zero‑insertion up‑sampling.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so the framework does not kill us early
    global_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Phase configuration: (length, fraction of total time)
    # ------------------------------------------------------------------
    phases = [
        (1200, 0.30),   # coarse search, cheap
        (2400, 0.35),   # medium size
        (4800, 0.35)    # fine‑grained large vector
    ]

    best_overall_a = None
    best_overall_val = np.inf

    for n, frac in phases:
        phase_deadline = start_time + budget_s * frac - 2.0
        if phase_deadline > global_deadline:
            phase_deadline = global_deadline
        # initialise – either uniform or up‑sampled previous best
        if best_overall_a is None:
            init_a = np.full(n, 1.0 / n)
        else:
            init_a = _upsample_sequence(best_overall_a, n)

        # run the optimiser for the allocated slice of time
        best_a_phase, best_val_phase = _optimise_phase(
            n, init_a, phase_deadline
        )

        # keep the best solution seen so far
        if best_val_phase < best_overall_val:
            best_overall_val = best_val_phase
            best_overall_a = best_a_phase.copy()
            print(f"[phase complete] new global best = {best_overall_val:.6f}")

        # stop early if we have exhausted the budget
        if time.time() >= global_deadline:
            break

    # Final reporting (the framework will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_overall_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
