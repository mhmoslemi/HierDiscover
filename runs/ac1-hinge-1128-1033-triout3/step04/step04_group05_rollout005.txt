"""
Propose a low‑max‑auto‑correlation step function.

The algorithm is a hybrid of a conditional‑gradient (Frank‑Wolfe) step
based on a linear program and a smoothed‑max Adam optimisation.
* First we take a feasible vector (uniform or the previously found
  `height_sequence_1`), normalise it to a probability distribution and
  perform one LP that maximises the total mass while keeping the
  convolution with the current vector bounded by its current maximum.
  The resulting direction is mixed with the current vector (mixing
  coefficient `t=0.2`).  This usually gives an immediate drop of the
  objective.
* Then we run a long Adam optimisation on the relaxed objective
  `softmax_tau( a * a )`.  The soft‑max temperature `tau` decays
  exponentially so that the algorithm focuses on the true maximum
  later in the run.  Random “jitter’’ moves are added occasionally to
  escape plateaux.
* Throughout the run we keep the best sequence seen (according to the
  original `evaluate_sequence`) and return it when the time budget
  expires.

All numbers stay non‑negative and are kept on the simplex
(`∑ a_i = 1`) via a soft‑max parameterisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution using FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # tiny negative values can appear from numerical noise – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max (soft‑max) of the convolution.
# For a weight vector w (∑ w = 1) the gradient w.r.t. a is
#    grad_a = 2 * (reverse(w) * a)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Compute ∇_a  Σ_k w_k (a*a)[k] = 2·(rev(w) * a)."""
    n = a.shape[0]
    # reverse the weight vector
    u = w[::-1]
    conv = _full_convolution_fft(u, a)          # length = 3n‑2
    # extract the n entries that correspond to the original indices
    start = 2 * n - 2
    idx = start - np.arange(n)                  # decreasing indices
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (used only for a reset).
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >=0, sum x = 1}."""
    # Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (hopefully) minimise
    `evaluate_sequence`.  The routine respects the 1000 s time budget.
    """
    # ------------------------------------------------------------------
    # Basic set‑up
    # ------------------------------------------------------------------
    import numpy as np
    import time
    try:
        import scipy.optimize
        import scipy.sparse as sp
        have_scipy = True
    except Exception:          # pragma: no cover
        have_scipy = False

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # Choose an initial sequence
    # ------------------------------------------------------------------
    # If a previous construction is available use it, otherwise start
    # from a uniform distribution.
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=np.float64)
        # keep a moderate length (the previous best used 1800)
        target_n = 1800
        if init.size > target_n:
            init = init[:target_n]
        elif init.size < target_n:
            init = np.concatenate([init,
                                   np.zeros(target_n - init.size,
                                            dtype=np.float64)])
    else:
        target_n = 1800
        init = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    # Enforce non‑negativity and normalise to sum = 1 (the scale is irrelevant)
    init = np.maximum(init, 0.0)
    if init.sum() < 1e-12:
        init = np.full_like(init, 1.0 / init.size)
    a = init / init.sum()
    n = a.shape[0]

    # ------------------------------------------------------------------
    # ONE conditional‑gradient (LP) step to jump away from the uniform
    # point.  This often yields a noticeable reduction of the max‑conv.
    # ------------------------------------------------------------------
    if have_scipy:
        try:
            conv = _full_convolution_fft(a, a)
            M = conv.max()
            # Build the constraint matrix A such that A @ b ≤ M
            rows = []
            cols = []
            data = []
            for k in range(2 * n - 1):
                j_start = max(0, k - (n - 1))
                j_end = min(n - 1, k)
                # indices j where b_j participates in the k‑th convolution entry
                for j in range(j_start, j_end + 1):
                    i = k - j
                    rows.append(k)
                    cols.append(j)
                    data.append(a[i])
            A = sp.csr_matrix((data, (rows, cols)),
                              shape=(2 * n - 1, n), dtype=np.float64)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            # maximise sum(b)  → minimise -sum(b)
            c = -np.ones(n, dtype=np.float64)

            # `highs` is the modern backend in SciPy ≥1.6
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                        bounds=[(0, None)] * n,
                                        method='highs',
                                        options={'presolve': True})
            if res.success:
                b = res.x
                sum_b = b.sum()
                if sum_b > 1e-12:
                    g = b / sum_b                     # unit‑sum direction
                    t = 0.2                            # mixing coefficient
                    a = (1.0 - t) * a + t * g
                    a = a / a.sum()                    # protect against tiny drift
                    print(f"[lp] LP succeeded – sum(b)={sum_b:.6f}, "
                          f"M before={M:.6f}, new M≈{_full_convolution_fft(a,a).max():.6f}")
            else:
                print("[lp] linprog failed:", res.message)
        except Exception as exc:                # pragma: no cover
            print("[lp] exception during LP step:", exc)

    # ------------------------------------------------------------------
    # Switch to log‑space (softmax) representation – this automatically
    # satisfies the simplex constraints.
    # ------------------------------------------------------------------
    eps_num = 1e-12
    x = np.log(np.maximum(a, eps_num))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.90, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps_adam = 1e-8
    base_lr = 0.30

    # bookkeeping of the best solution seen so far
    conv = _full_convolution_fft(a, a)
    cur_val = float(2.0 * n * conv.max())
    best_val = cur_val
    best_a = a.copy()
    print(f"[init] n={n}, initial objective={best_val:.6f}")

    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop (runs until the time budget is exhausted)
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1

        # ---- evaluate current objective
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # ---- schedule parameters
        progress = (now - start_time) / (deadline - start_time)

        # temperature for the soft‑max (exponential decay)
        tau_start = M * 0.5
        tau_end = M * 1e-7
        # exponential decay factor tuned experimentally
        tau = tau_start * (0.5 ** (10 * progress))
        if tau < tau_end:
            tau = tau_end

        # ---- compute soft‑max weights over the convolution entries
        if tau > 0.0:
            exp_args = (conv - M) / tau
            max_arg = exp_args.max()
            exp_vals = np.exp(exp_args - max_arg)
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # ---- gradient w.r.t. a (smoothed sub‑gradient)
        grad_a = _gradient_from_weights(a, w)

        # ---- gradient w.r.t. the logits x (chain rule for softmax)
        dot = np.dot(grad_a, a)          # <grad_a , a>
        grad_x = a * (grad_a - dot)      # ∇_x (softmax) = diag(a)*(grad_a - <grad_a,a>)

        # ---- Adam update for x
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate schedule (linear decay to a tiny value)
        lr = base_lr * (1.0 - progress) + 1e-5 * progress
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- obtain a from the updated logits (softmax)
        max_x = x.max()
        exp_x = np.exp(x - max_x)               # shift for numerical stability
        a = exp_x / exp_x.sum()                  # stays on the simplex

        # ---- occasional random “jitter’’ moves (helps escape plateaux)
        if np.random.rand() < 0.03:              # ~3 % per iteration
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta
            # re‑normalise and wipe the Adam state (the sudden change
            # deserves a fresh start)
            a = a / a.sum()
            x = np.log(np.maximum(a, eps_num))
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0

    # ------------------------------------------------------------------
    # Final reporting (the framework will also call `evaluate_sequence`)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
