def project_simplex(v):
    # assumes v is 1D numpy array
    # based on algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u - (cssv - 1) / np.arange(1, n+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _full_convolution_fft(x, y):
    n = len(x)
    m = len(y)
    total = n + m - 1
    size = 1 << ((total - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Set small negative (due to numeric) to 0
    np.maximum(conv, 0, out=conv)
    return conv
```

def project_simplex(v):
    v = np.asarray(v, dtype=float)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Optimisation of a non‑negative step‑function using projected Adam and a
sub‑gradient of the max‑convolution.  The algorithm runs several
independent restarts within the allotted time budget; each restart
performs Adam updates on the probability simplex, obtains a gradient
from a small set of convolution entries that achieve (or are close to)
the current maximum, and injects occasional random mass‑transfer moves
to escape plateaux.  The best sequence seen over all restarts is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding artefacts
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the (soft) max‑convolution w.r.t. the coefficients a
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented via a single convolution: reverse w and convolve with a.
    """
    n = a.shape[0]
    u = w[::-1]                              # reversed weight vector
    conv = _full_convolution_fft(u, a)       # length = (2n-1) + n - 1 = 3n-2
    start = 2 * n - 2
    idx = start - np.arange(n)               # indices that map back to a_i
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σ a_i = 1, a_i ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector v onto the simplex {x | x ≥ 0, sum x = 1}.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v.copy()

    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u)
    rho = np.nonzero(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  The routine performs several short optimisation
    runs (restarts) inside the time budget; each run uses projected Adam on
    the simplex together with a top‑K sub‑gradient of the max‑convolution
    and occasional random mass‑transfer jitter.  The best sequence seen
    overall is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – the framework kills the process a couple of seconds late
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float('inf')
    max_restarts = 5                # number of independent restarts

    restart = 0
    while time.time() < deadline and restart < max_restarts:
        # Allocate roughly equal time to the remaining restarts
        time_left = deadline - time.time()
        if time_left < 20:          # not enough time for a meaningful run
            break
        restarts_remaining = max_restarts - restart
        phase_time = time_left / restarts_remaining
        phase_deadline = time.time() + phase_time

        # ------------------------------------------------------------------
        # Choose a problem size (larger size offers more freedom)
        # The last two restarts use a longer vector.
        # ------------------------------------------------------------------
        if restart >= max_restarts - 2:
            n = 3600
        else:
            n = 1800

        # ------------------------------------------------------------------
        # Initialise a probability vector a (sum = 1, a_i ≥ 0)
        # Try to reuse a previously known good sequence if present.
        # ------------------------------------------------------------------

                else:

            else:

        except NameError:

        # Add a tiny random perturbation to avoid starting at a fixed point
        a += 0.01 * np.random.rand(n)
        a = _project_simplex(a)

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.90, 0.999
        beta1_pow, beta2_pow = 1.0, 1.0
        eps_adam = 1e-8
        base_lr = 0.5               # initial learning rate

        # ------------------------------------------------------------------
        # Local‑best tracking for this restart
        # ------------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        best_val_restart = 2.0 * n * cur_max
        best_a_restart = a.copy()

        iteration = 0
        # Number of convolution entries that will receive weight for the sub‑gradient.
        K = max(5, int(0.001 * (2 * n - 1)))   # at least 5 indices
        jitter_prob = 0.06                     # chance of a random mass move per iteration
        jitter_max_frac = 0.5                  # largest fraction of mass that can be moved

        # ------------------------------------------------------------------
        # Main optimisation loop for this restart
        # ------------------------------------------------------------------
        while time.time() < phase_deadline:
            now = time.time()

            # ----- convolution and current objective value -----
            conv = _full_convolution_fft(a, a)
            M = conv.max()                     # max_{k} (a * a)[k]
            cur_val = 2.0 * n * M               # sum(a) == 1 → objective = 2·n·M

            # ----- keep global best (via the official evaluator) -----
            if cur_val < best_val_restart:

                except Exception:
                    pass

            # ----- learning‑rate schedule (linear decay) -----
            progress = (now - start_time) / (deadline - start_time)
            lr = base_lr * (1.0 - progress) + 1e-5 * progress

            # ----- sub‑gradient: concentrate on the K largest convolution entries -----
            if K < conv.size:
                # argpartition gives the K largest without full sorting
                idx = np.argpartition(-conv, K - 1)[:K]
            else:
                idx = np.arange(conv.size)
            w = np.zeros_like(conv)
            w[idx] = 1.0 / idx.size

            grad_a = _gradient_from_weights(a, w)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a * grad_a)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            a_step = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_step)

            # ----- occasional random mass‑transfer (helps escape plateaux) -----
            if np.random.rand() < jitter_prob:
                i = np.random.randint(n)
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
                # move a random fraction of the mass from i to j
                delta = np.random.rand() * a[i] * jitter_max_frac
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)
                # Reset Adam moments after a sudden change
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow, beta2_pow = 1.0, 1.0

            iteration += 1

        # ------------------------------------------------------------------
        # End of this restart – evaluate the locally best sequence with the
        # official scorer and, if it improves the global best, store it.
        # ------------------------------------------------------------------
        try:
            final_score = evaluate_sequence(best_a_restart.tolist())
        except Exception:
            final_score = float('inf')

        if final_score < best_score:
            best_score = final_score
            best_seq = best_a_restart.copy()
            print(f"[global] new best after restart {restart}: {best_score:.6f}")

        restart += 1

    # ----------------------------------------------------------------------
    # Fallback in the very unlikely event that nothing was found
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(1800, 1.0 / 1800)

    return best_seq.tolist()
