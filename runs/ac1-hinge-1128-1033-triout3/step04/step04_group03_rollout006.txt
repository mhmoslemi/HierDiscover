def upsample_vector(a, new_len):
    old_len = a.shape[0]
    x_old = np.linspace(0, 1, old_len)
    x_new = np.linspace(0, 1, new_len)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()

def beta_distribution(n, alpha, beta):
    i = np.arange(1, n+1)

    a = (x ** (alpha-1)) * ((1 - x) ** (beta-1))

def compute_subgradient_max(a, conv):
    k = np.argmax(conv)
    n = a.shape[0]
    start_i = max(0, k - (n - 1))
    end_i = min(n - 1, k)

    j_vals = k - i_vals
    grad = np.zeros_like(a)
    grad[i_vals] += 2.0 * a[j_vals]
    return grad

def refine_on_simplex(a, deadline):
    a = a.astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = a.shape[0]
    current = a.copy()
    conv_curr = full_convolution_fft(current)
    current_score = float(2.0 * n * conv_curr.max())
    best = current.copy()
    best_score = current_score
    alpha = 0.2
    alpha_decay = 0.9997
    beta = 0.9
    velocity = np.zeros_like(a)
    stagn_counter = 0
    iter = 0
    while time.time() < deadline:
        iter += 1
        conv_curr = full_convolution_fft(current)
        grad = compute_subgradient_max(current, conv_curr)
        # momentum
        velocity = beta * velocity + (1 - beta) * grad
        # step direction normalized
        norm_l1 = np.abs(velocity).sum() + 1e-15
        step = -alpha * velocity / norm_l1
        cand = current + step
        cand = simplex_projection(cand)
        # evaluate
        conv_cand = full_convolution_fft(cand)
        cand_score = float(2.0 * n * conv_cand.max())
        # accept if improvement
        if cand_score < current_score:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                stagn_counter = 0
                print(f"Iter {iter}: new best {best_score:.6f}")
            else:
                stagn_counter += 1
        else:
            stagn_counter += 1
        # decay step size
        alpha *= alpha_decay
        # occasionally reheat
        if stagn_counter > 5000:
            eps = 1e-4
            noise = np.random.rand(n) * eps
            current = np.maximum(current + noise, 0.0)
            current = current / current.sum()
            current_score = float(2.0 * n * full_convolution_fft(current).max())
            stagn_counter = 0
            alpha = max(alpha, 0.001)
            print(f"Iter {iter}: reheating")
    return best, best_score

def upsample_vector(a, new_len):
    old_len = a.shape[0]
    x_old = np.linspace(0, 1, old_len)
    x_new = np.linspace(0, 1, new_len)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    return a_new

"""
Propose a candidate step‑function (non‑negative vector) that minimises

    2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm consists of three
phases:

1.  A deterministic sweep over several symmetric parametric families
    (power‑law, shifted power‑law, exponential, cosine, beta‑distribution)
    for a set of lengths.  The best vector from this sweep seeds the next
    phase.

2.  Projected gradient descent on the probability simplex using a
    sub‑gradient of the maximal convolution entry (chosen from a single
    maximising index) together with simple momentum and a L1‑normalised
    step.  The step size decays and a small random “re‑heat” move is
    applied after prolonged stagnation.

3.  Optional up‑sampling (linear interpolation) of the current best
    vector to a larger length, followed by another refinement pass.
    The process stops when the time budget expires.

All helper functions are top‑level; the routine respects a 1000 s
budget and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex { w >= 0 , sum(w) = 1 }.
    Implementation follows Wang & Carreira‑Perpiñán (O(n log n)).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1 (the linear convolution).
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact objective value for a vector with sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient for the maximal convolution entry
# ----------------------------------------------------------------------


def compute_subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only the (single) index attaining the maximum is used.
    """
    k = int(np.argmax(conv))
    n = a.shape[0]
    start_i = max(0, k - (n - 1))
    end_i = min(n - 1, k)
    i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
    j_vals = k - i_vals
    grad = np.zeros_like(a)
    grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Parametric families (all symmetric)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))^-p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑distribution on {1,…,n}:
        a_i ∝ (i/(n+1))^{alpha-1} (1‑i/(n+1))^{beta‑1}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    x = i / (n + 1.0)
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Upsampling / interpolation
# ----------------------------------------------------------------------


def upsample_vector(a: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear interpolation from a vector of length len(a) to length new_len,
    followed by renormalisation to the probability simplex.
    """
    old_len = a.shape[0]
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new

# Projected gradient descent on the simplex
# ----------------------------------------------------------------------


def refine_on_simplex(start_vec: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a (momentum‑augmented) projected sub‑gradient descent on the
    probability simplex, aiming to minimise max_k (a*a)[k].
    The routine stops when the time limit is reached.
    Returns the best vector found and its objective value.
    """
    a = np.maximum(start_vec, 0.0)
    a /= a.sum()
    n = a.shape[0]

    current = a.copy()
    conv_curr = full_convolution_fft(current)
    current_score = float(2.0 * n * conv_curr.max())

    best = current.copy()
    best_score = current_score

    # hyper‑parameters
    alpha = 0.12               # initial L1‑step magnitude
    alpha_decay = 0.9997
    beta = 0.9                 # momentum coefficient
    velocity = np.zeros_like(a)

    it = 0
    stagn = 0
    while time.time() < deadline:
        it += 1

        # gradient (sub‑gradient from a single maximal index)
        grad = compute_subgradient_max(current, conv_curr)

        # momentum update
        velocity = beta * velocity + (1.0 - beta) * grad

        # L1‑normalised step (keeps total transferred mass ≈ alpha)
        norm_l1 = np.abs(velocity).sum() + 1e-15
        step = -alpha * velocity / norm_l1

        cand = current + step
        cand = simplex_projection(cand)

        # evaluate candidate
        conv_cand = full_convolution_fft(cand)
        cand_score = float(2.0 * n * conv_cand.max())

        # accept only if we improve the current position
        if cand_score < current_score:
            current = cand
            conv_curr = conv_cand
            current_score = cand_score

            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                stagn = 0
                print(f"Iter {it:6d}: new best score = {best_score:.6f}")
            else:
                stagn += 1
        else:
            stagn += 1

        # decay step size
        alpha *= alpha_decay

        # re‑heat after long stagnation
        if stagn >= 5000:
            eps = 1e-4
            noise = np.random.rand(n) * eps
            current = np.maximum(current + noise, 0.0)
            current /= current.sum()
            conv_curr = full_convolution_fft(current)
            current_score = float(2.0 * n * conv_curr.max())
            stagn = 0
            alpha = max(alpha, 0.005)
            print(f"Iter {it:6d}: re‑heat (stagnation)")

        # safety stop if step size becomes negligible
        if alpha < 1e-7:
            break

    return best, best_score
