"""
propose_candidate
-----------------
Heuristic search for a non‑negative step function (vector) that minimises

    2 * n * max_k (a * a)[k]

where * is linear convolution.  The algorithm consists of three stages:

1. **Deterministic sweep** over several parametric families
   (inverse distance, inverse product distance, symmetric exponential,
   arcsine‑law) on a handful of lengths.  This quickly yields a decent
   seed vector.

2. **Local optimisation** with a projected Adam optimiser on the
   probability simplex (log‑space updates, adaptive moments).  The Adam
   step works directly on the logarithms of the coefficients, guaranteeing
   positivity, and we project onto the simplex after each iteration.
   The optimiser runs until the time budget is exhausted.

3. **Final polishing** using a few iterations of simple projected
   sub‑gradient descent (the same gradient as in the original code) to
   clean up any remaining slack.

All helper functions are defined at top level.  No file IO is performed,
and the routine respects the 1000 s budget.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n                     # zero‑padding length (power‑of‑2 not required)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_vector(a: np.ndarray) -> float:
    """
    Exact value returned by evaluate_sequence for a vector with sum == 1.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    All indices attaining (or almost attaining) the maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]          # derivative of a_i a_j (ordered pairs)
    return grad


# ----------------------------------------------------------------------
# Parametric families (used for the deterministic seed)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i, n‑1‑i)+1)^{‑p}   (symmetric, edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^{‑p}   (edge‑heavy, product‑distance)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i) + exp(-lam*(n‑1‑i))   (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def arcsine_distribution(n: int) -> np.ndarray:
    """a[i] ∝ 1/√((i+½)(n‑i‑½))  – discretised arcsine law (U‑shape)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# Main search routine

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence within the given time budget.
    Returns a plain Python list of floats (∑a ≈ 1).  The routine prints
    progress messages and finally the best score found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over a few lengths & parametric families
    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf

    # Lengths to try (moderate – still fast for the deterministic sweep)
    lengths = [2000, 4000, 6000, 8000, 10000, 12000]

    # Parameter grids (coarse → fine enough for a quick scan)
    p_grid = np.linspace(0.2, 6.0, 30)                  # for power‑law families
    lam_grid = np.logspace(-4, -0.3, 30)                # for exponential family

    print("=== Deterministic sweep start ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # uniform (baseline)
        a = np.ones(n, dtype=np.float64)
        a /= a.sum()
        sc = score_vector(a)
        if sc < best_score:
            best_score, best_vec = sc, a.copy()
            print(f" uniform n={n:5d}  score={sc:.6f}")

        # u‑min family
        for p in p_grid:
            a = u_min_distribution(n, p)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f" u_min n={n:5d} p={p:5.2f}  score={sc:.6f}")

        # u‑prod family
        for p in p_grid:
            a = u_prod_distribution(n, p)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f" u_prod n={n:5d} p={p:5.2f}  score={sc:.6f}")

        # symmetric exponential family
        for lam in lam_grid:
            a = exp_distribution(n, lam)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f" exp n={n:5d} λ={lam:.2e}  score={sc:.6f}")

        # arcsine law (no free parameter)
        a = arcsine_distribution(n)
        sc = score_vector(a)
        if sc < best_score:
            best_score, best_vec = sc, a.copy()
            print(f" arcsine n={n:5d}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate external seed if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            external = np.array(height_sequence_1, dtype=float)
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0:
                external /= s
                sc = score_vector(external)
                if sc < best_score:
                    best_score, best_vec = sc, external.copy()
                    print(f" external seed used, length {len(external)} score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 5000
        best_vec = np.random.rand(n) + 1e-8
        best_vec /= best_vec.sum()
        best_score = score_vector(best_vec)
        print(f"fallback random seed, n={n} score={best_score:.6f}")

    n = best_vec.shape[0]
    print(f"\n=== Selected problem size: n = {n} ===")
    print(f" seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local Adam optimisation on the simplex (log‑space)
    # ------------------------------------------------------------------
    print("\n=== Adam optimisation start ===")
    # initialise log‑probabilities
    log_a = np.log(best_vec + 1e-30)
    m = np.zeros_like(log_a)            # first moment
    v = np.zeros_like(log_a)            # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12                           # initial learning rate
    t = 0

    best_local_vec = best_vec.copy()
    best_local_score = best_score
    last_improve = 0

    while time.time() < deadline:
        t += 1

        # map back to the simplex
        a = np.exp(log_a)
        a_sum = a.sum()
        a /= a_sum

        conv = full_convolution_fft(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_local_score - 1e-12:
            best_local_score = cur_score
            best_local_vec = a.copy()
            last_improve = t
            print(f" Adam it {t:6d}: new best = {best_local_score:.6f}")

        # sub‑gradient of the (non‑smooth) objective
        grad = compute_subgradient(a, conv)

        # Adam update on the log‑variables
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        log_a -= step

        # occasional learning‑rate decay & re‑normalisation
        if t % 1000 == 0:
            lr *= 0.9
            a = np.exp(log_a)
            a /= a.sum()
            log_a = np.log(a + 1e-30)

        # re‑heat if stuck for a long time

            last_improve = t
            print(f" Adam it {t:6d}: re‑heat")

    best_vec = best_local_vec
    best_score = best_local_score

    # ------------------------------------------------------------------
    # 5. Short projected sub‑gradient polishing (fine‑tuning)
    # ------------------------------------------------------------------
    print("\n=== Sub‑gradient polishing start ===")
    a = best_vec.copy()
    cur_score = best_score
    step = 0.08
    step_decay = 0.9995
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution_fft(a)
        grad = compute_subgradient(a, conv)

        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = a   # protect against numerical collapse
        else:
            cand /= cand_sum

                best_vec = cand.copy()

    # ------------------------------------------------------------------
