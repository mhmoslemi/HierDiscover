def _subgradient_descent(a_init, best_val, deadline, n):
    """Perform mirror descent to reduce max auto-convolution using exponentiated gradient."""
    a = a_init.copy()
    best_a = a.copy()
    best_score = best_val
    rng = np.random.default_rng()
    t = 0
    while time.time() < deadline:
        # compute full convolution via FFT
        conv = _full_convolution(a)
        kmax = int(np.argmax(conv))
        # compute gradient g_i = 2 * a_{kmax - i}
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros(n)
        g[mask] = 2.0 * a[j[mask]]
        # step size schedule
        eta = 0.5 / np.sqrt(t + 1)
        # multiplicative update (mirror descent)
        a = a * np.exp(-eta * g)
        # renormalize
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / sum_a
        # evaluate
        val = evaluate_sequence(a.tolist())
        if val < best_score:
            best_score = val
            best_a = a.copy()
            print(f"[subgrad] iter {t} new best = {best_score:.6f}")
        t += 1
        # early break if improvement stagnates?
        if t > 5000:
            break
    return best_a, best_score

"""
Solver for the “step‑function” challenge.

The objective to minimise is

    E(a) = 2·n·max(a * a) / (sum(a)²) ,

where * denotes the discrete convolution of the non‑negative vector a
(step heights).  Because the expression is scale‑invariant we can
restrict to probability vectors (sum = 1) and simply minimise
    2·n·max(a * a).

The algorithm combines several complementary optimisation stages:

1. **Continuation L‑BFGS‑B** – optimise a smooth log‑sum‑exp surrogate
   of the max‑convolution.  The surrogate is
        L_ε(a) = 2·n·ε·log(∑_k exp((a*a)_k / ε)),
   whose gradient can be evaluated analytically using FFTs.
   A schedule of decreasing ε drives the solution towards the true
   objective.

2. **Random restarts** – after each full continuation run the current
   best distribution is perturbed with Gaussian noise and the
   continuation is launched again.  This helps to escape local minima.

3. **Exponentiated‑gradient sub‑gradient descent** – directly minimises
   the exact max‑convolution using the sub‑gradient
        g_i = 2·a_{k*−i},
   where k* is the index of the current maximum.  The update
   ``a ← a·exp(-η·g)`` (mirror descent on the simplex) guarantees
   positivity; a renormalisation restores the unit‑sum constraint.

4. **Final cheap hill‑climbing** – random mass transfers are performed
   until the time budget expires, keeping any improvement.

All heavy work is done with FFT‑based convolutions, which makes the
method fast enough to explore vectors of a few thousand entries within
the 1000 s budget.  The best sequence found is returned as a plain
list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full auto‑convolution of a real vector (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)‑1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smooth surrogate and its gradient (w.r.t. soft‑max parameters)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smooth objective L_ε and its gradient w.r.t. the
    soft‑max parameters w for a vector of length n.

    The parameters are interpreted via a soft‑max:
        a_i = exp(w_i) / Σ_j exp(w_j) .
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)              # improve numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # full auto‑convolution a*a
    conv = _full_convolution(a)          # length 2·n‑1
    max_conv = conv.max()

    # smoothed max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)         # = log∑_k exp((conv_k‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()          # p_k = softmax over conv entries

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)        # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]       # middle n entries
    grad_a = 4.0 * n * seg[::-1]         # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)          # ∂L/∂w
    return L, grad_w


class SmoothObjective:
    """Callable wrapper required by ``scipy.optimize.minimize``."""
    __slots__ = ("eps", "n")

    def __init__(self, eps: float, n: int):
        self.eps = eps
        self.n = n

    def __call__(self, w: np.ndarray):
        """Return (f, grad) for the current ε."""
        return _smooth_obj_and_grad(w, self.eps, self.n)


# ----------------------------------------------------------------------
# Sub‑gradient (mirror‑descent) refinement on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a_start: np.ndarray,
                         best_val: float,
                         deadline: float,
                         n: int):
    """
    Exponentiated‑gradient descent on the exact objective
    ``2·n·max(a*a)`` while staying on the probability simplex.
    Returns the best vector and its (exact) value seen.
    """
    a = a_start.copy()
    # Current exact value (sum(a) = 1 → val = 2·n·max_conv)
    cur_conv = _full_convolution(a)
    cur_val = 2.0 * n * cur_conv.max()
    best_a = a.copy()
    best_score = best_val

    rng = np.random.default_rng()
    t = 0
    while time.time() < deadline:
        # identify the index attaining the maximum convolution value
        kmax = int(np.argmax(cur_conv))

        # sub‑gradient of max_conv w.r.t. a: g_i = 2·a_{kmax‑i}
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros(n, dtype=float)
        g[mask] = 2.0 * a[j[mask]]

        # step size – diminishing as 1/√t
        eta = 0.5 / np.sqrt(t + 1.0)

        # mirror‑descent (exponentiated) update, stays in the interior
        a_new = a * np.exp(-eta * g)
        s = a_new.sum()
        if s == 0 or np.isnan(s):
            a_new = np.full(n, 1.0 / n)
        else:
            a_new = a_new / s

        # evaluate new exact objective
        conv_new = _full_convolution(a_new)
        val_new = 2.0 * n * conv_new.max()

        # accept only if we improve (greedy descent)
        if val_new < cur_val:
            a = a_new
            cur_conv = conv_new
            cur_val = val_new
            if cur_val < best_score:
                best_score = cur_val
                best_a = a.copy()
                print(f"[subgrad] iter {t} new best = {best_score:.6f}")

        t += 1
        # a safety limit – a few thousand iterations are more than enough
        if t >= 5000:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple random hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       n: int):
    """
    Small random perturbations that keep Σa constant.
    Accepts a move only when it strictly improves the exact objective.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a fraction of mass from i to j
        delta = cur_a[i] * 0.05 * rng.random()
        if delta == 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # evaluate exact objective (fast path using convolution)
        conv = _full_convolution(new_a)
        val = 2.0 * n * conv.max()
        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val
