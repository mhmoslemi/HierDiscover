        # Optimization parameters

        v = np.zeros_like(a)

        local_best_seq = a.copy()
        stagn = 0

        while time.time() < deadline:

            max_conv = conv.max()

                best_seq = a.copy()
                print(f"[global] new best {best_score:.12f} t={time.time():.2f}")

                local_best_seq = a.copy()

        restart_id += 1

        if time.time() + 5 >= deadline:
            break

    if best_seq is None:
        # fallback: uniform
        best_seq = np.full(n, 1.0/n)

    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using an Adam‑based projected gradient descent on the simplex.
The algorithm

* builds several “nice’’ initial shapes (U‑shaped, Catalan, Gaussian,
  sine, semi‑circle, plus the previous best if available);
* runs a soft‑max approximation of the max‑convolution objective;
* updates the vector with Adam (per‑coordinate step‑size) and projects
  back onto the probability simplex;
* decays the soft‑max temperature and the learning‑rate, injects
  occasional Gaussian noise to escape plateaus and restarts the whole
  search a few times within the 1000 s budget;
* finishes with a short exact‑max sub‑gradient polishing phase;
* returns the best sequence seen.

The implementation relies only on NumPy (no external solvers) and
convolution is performed with FFT for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative height sequence that (approximately) minimizes
    the evaluation function defined in the problem statement.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (a safety margin of 5 s is kept).
    **kwargs : dict
        Ignored – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence discovered before the deadline.
    """
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0          # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose problem size – reuse length of a previous solution if
    # possible, otherwise fall back to 5000.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(base.shape[0])
        except Exception:
            n = 5000
    else:
        n = 5000

    # --------------------------------------------------------------
    # Pre‑compute a few “nice’’ shapes that will be mixed to obtain
    # starting points.
    # --------------------------------------------------------------
    x = np.arange(n, dtype=np.float64)

    # 1. U‑shaped (large at the ends, low in the middle)
    eps_small = 1e-12
    u_shape = 1.0 / np.sqrt((x + 0.5) * (n - 0.5 - x) + eps_small)

    # 2. Catalan‑type decreasing sequence (classic start)
    catalan = np.empty(n, dtype=np.float64)
    cur = 1.0
    catalan[0] = cur
    for k in range(1, n):
        cur = cur * (2.0 * k - 1) / (2.0 * k)
        catalan[k] = cur

    # 3. Gaussian centred in the middle
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

    # 4. Positive sine wave
    sine = np.sin(np.pi * (x + 0.5) / n)

    # 5. Semi‑circle (zero at ends, high in the centre)
    t = 2.0 * x / (n - 1) - 1.0               # maps to [-1, 1]
    semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

    bases = [u_shape, catalan, gauss, sine, semicircle]

    # Include the previous best, if it exists, as an extra base shape.
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.shape[0] == n:
                bases.append(prev)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_score = np.inf
    best_seq = None

    restart_id = 0
    # ------------------------------------------------------------------
    # Outer restart loop – we keep restarting with fresh random mixtures
    # as long as time permits.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Random mixture of the base shapes => starting point.
        # --------------------------------------------------------------
        weights = rng.random(len(bases))
        weights /= weights.sum()
        a = np.zeros(n, dtype=np.float64)
        for w, b in zip(weights, bases):
            a += w * b
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = rng.random(n)
            s = a.sum()
        a /= s                         # ensure a lies on the simplex

        # --------------------------------------------------------------
        # Adam hyper‑parameters and soft‑max temperature.
        # --------------------------------------------------------------
        lr = 0.6                       # initial learning rate
        lr_min = 1e-5
        lr_decay = 0.9995

        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8

        # first evaluation – needed for the temperature schedule
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        eps = max(max_conv * 1.0, 1e-12)          # start a bit hotter
        eps_min = max(max_conv * 1e-5, 1e-12)
        eps_decay = 0.9985

        m = np.zeros_like(a)          # first moment buffer
        v = np.zeros_like(a)          # second moment buffer
        t_iter = 0

        # Local best inside this restart
        local_best_score = 2.0 * n * max_conv
        local_best_seq = a.copy()

        stagn = 0                     # iterations without improvement

        # --------------------------------------------------------------
        # Inner optimisation loop (Adam + soft‑max).
        # --------------------------------------------------------------
        max_iters_this_restart = 200_000
        while time.time() < deadline and t_iter < max_iters_this_restart:
            # ---- current convolution and score ---------------------------------
            conv = _full_convolution_fft(a, a)
            max_conv = conv.max()
            score = 2.0 * n * max_conv          # sum(a) == 1 by construction

            # ---- update global best --------------------------------------------
            if score < best_score - 1e-12:
                best_score = score
                best_seq = a.copy()
                print(f"[global] new best {best_score:.12f}  restart={restart_id}")

            # ---- update local best -----------------------------------------------
            if score < local_best_score - 1e-12:
                local_best_score = score
                local_best_seq = a.copy()
                stagn = 0
            else:
                stagn += 1

            # ---- temperature decay -----------------------------------------------
            eps = max(eps * eps_decay, eps_min)

            # ---- soft‑max weighting of convolution entries ------------------------
            w = np.exp((conv - max_conv) / eps)   # every entry ≤ 1
            w_sum = w.sum()
            if not np.isfinite(w_sum) or w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = w / w_sum

            # ---- gradient of the weighted sum (cross‑correlation) ---------------
            rev_a = a[::-1]
            conv_rev = _full_convolution_fft(rev_a, w)          # length 3n‑2
            g = 2.0 * conv_rev[n - 1 : n - 1 + n]                # size n

            # ---- Adam update ------------------------------------------------------
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)

            # bias‑corrected moments
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            a_new = a - step
            a_new = _project_simplex(a_new)
            a = a_new

            # ---- learning‑rate decay ---------------------------------------------
            lr = max(lr_min, lr * lr_decay)

            # ---- occasionally perturb to escape plateaus -------------------------
            if stagn > 5000:
                noise = rng.normal(scale=0.02, size=n)
                a = a + noise
                a = _project_simplex(a)
                # reset Adam state so that the perturbation is not immediately damped
                m.fill(0.0)
                v.fill(0.0)
                t_iter = 0
                stagn = 0
                # warm‑up the temperature a touch
                eps = max(eps, max_conv * 0.5)
                print(f"[perturb] restart={restart_id} iter={t_iter}")

        # ------------------------------------------------------------------
        # Polishing phase for the best sequence found in this restart.
        # ------------------------------------------------------------------
        a = local_best_seq.copy()
        for _ in range(1500):
            conv = _full_convolution_fft(a, a)
            mx = conv.max()
            # take the first index attaining the maximum
            idx = int(np.flatnonzero(conv >= mx - 1e-15)[0])

            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            j = idx - i_range
            g = np.zeros_like(a)
            g[i_range] = 2.0 * a[j]

            a = _project_simplex(a - 5e-6 * g)

        # update global best with the polished candidate
        final_conv = _full_convolution_fft(a, a)
        polish_score = 2.0 * n * final_conv.max()
        if polish_score < best_score:
            best_score = polish_score
            best_seq = a.copy()
            print(f"[polish] new best {best_score:.12f}  restart={restart_id}")

        restart_id += 1
        # stop if not enough time left for another full restart
        if time.time() + 5.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but guard against an empty best.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Clamp to the limits required by `evaluate_sequence` and return.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
