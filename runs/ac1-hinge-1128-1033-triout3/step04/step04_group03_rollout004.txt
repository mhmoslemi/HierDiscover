"""
Propose a candidate step‑function (non‑negative vector) that minimises

    2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm combines:

1.  A very fast deterministic seeding using simple parametric families
    (U‑shape, product‑U‑shape, exponential, cosine).  The best seeding for
    each explored length is kept.

2.  A smooth‑max gradient descent on the simplex.  The objective
    max conv is approximated by a soft‑max (log‑sum‑exp) with a temperature
    β that is slowly increased (β‑annealing).  The gradient is obtained
    via an FFT‑based correlation, then a projected additive step is taken.
    Simulated‑annealing acceptance and occasional “re‑heat’’ (random noise)
    avoid stagnation.

3.  Multiple lengths (≈5 k‑20 k) are tried sequentially; the overall best
    sequence found within the allotted time (default 1000 s) is returned.

The code is pure NumPy, runs on a single CPU, and respects the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w>=0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]  # drop trailing zeros


def correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear correlation (a ⋆ w) for non‑negative vectors a and w.
    Returns the slice corresponding to lags 0 … len(a)-1.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(A * np.conj(W), n=size)
    offset = m - 1          # index of lag 0 in the full correlation array
    return corr[offset : offset + n]


def smooth_gradient(a: np.ndarray, beta: float):
    """
    Compute the gradient of the smoothed max ‑conv (log‑sum‑exp)
    for a given temperature β.  Returns (grad, conv, max_val).
    """
    conv = full_convolution_fft(a)                 # length 2n‑1
    max_val = conv.max()
    # soft‑max weights (stable: max shifted to zero)
    exp_input = beta * (conv - max_val)            # max entry becomes 0
    w = np.exp(exp_input)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum                               # probability vector
    # gradient = 2 * (a ⋆ w)
    grad = 2.0 * correlation_fft(a, w)
    return grad, conv, max_val


# ----------------------------------------------------------------------
# Simple parametric families (used for deterministic seeding)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1+cos(π·i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative height sequence that minimises
    2·n·max_k (a*a)[k]  (the value returned by evaluate_sequence when sum(a)=1).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function returns the best sequence found
        before the deadline.

    Returns
    -------
    list[float]
        The best sequence (as a plain Python list).  The sum of the entries
        is approximately one; the external evaluator rescales as needed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1. Deterministic seeding over a handful of lengths & families
    # ------------------------------------------------------------------
    # Lengths that give a good trade‑off between expressivity and speed.
    lengths = [4800, 5808, 7200, 9600, 12000]   # include the previously known length
    exponents = np.arange(0.5, 5.01, 0.25)      # for U‑shape families
    shift_cs = [1.0, 2.5, 5.0]                 # shift values
    lam_vals = np.logspace(-4, -0.3, 20)       # exponential decay factors
    cos_exps = np.arange(0.5, 8.5, 0.5)        # cosine exponents

    print("=== Deterministic seeding start ===")
    for n in lengths:
        if time.time() >= deadline:
            break
        # a few simple seeds – we keep only the best for this length
        local_best = None
        local_score = np.inf

        # Uniform (baseline)
        a = np.full(n, 1.0 / n)
        sc = 2.0 * n * full_convolution_fft(a).max()
        if sc < local_score:
            local_score, local_best = sc, a.copy()

        # Power‑law U‑shapes (min‑distance)
        for p in exponents:
            a = u_min_distribution(n, p)
            sc = 2.0 * n * full_convolution_fft(a).max()
            if sc < local_score:
                local_score, local_best = sc, a.copy()

            a = u_prod_distribution(n, p)
            sc = 2.0 * n * full_convolution_fft(a).max()
            if sc < local_score:
                local_score, local_best = sc, a.copy()

            for c in shift_cs:
                a = u_min_shift_distribution(n, p, c)
                sc = 2.0 * n * full_convolution_fft(a).max()
                if sc < local_score:
                    local_score, local_best = sc, a.copy()

        # Exponential family
        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = 2.0 * n * full_convolution_fft(a).max()
            if sc < local_score:
                local_score, local_best = sc, a.copy()

        # Cosine family
        for p in cos_exps:
            a = cos_distribution(n, p)
            sc = 2.0 * n * full_convolution_fft(a).max()
            if sc < local_score:
                local_score, local_best = sc, a.copy()

        print(f"Length {n:5d} best seed score = {local_score:.6f}")

        # Update global best
        if local_score < best_score:
            best_score, best_seq = local_score, local_best.copy()
            print(f"*** New global best after seeding: {best_score:.6f} (len {n}) ***")

    # ------------------------------------------------------------------
    # 2. Optional external construction (if the variable exists)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size:
            external = np.maximum(external, 0.0)
            if external.sum() > 0.0:
                external /= external.sum()
                sc = 2.0 * len(external) * full_convolution_fft(external).max()
                if sc < best_score:
                    best_score, best_seq = sc, external.copy()
                    print(f"*** New global best from external var: {best_score:.6f} ***")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Stochastic refinement via smooth‑max gradient descent
    # ------------------------------------------------------------------
    # If no seed was found (unlikely), fall back to a random vector.
    if best_seq is None:
        n = 6000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = 2.0 * n * full_convolution_fft(best_seq).max()
        print(f"Fallback seed generated, score = {best_score:.6f}")

    # Main refinement loop – we keep iterating until the time budget expires.
    print("\n=== Stochastic refinement start ===")
    # To allow several restarts with different lengths we reuse the best length.
    n = best_seq.shape[0]

    # initialise optimisation state from the best seed found so far
    a = best_seq.copy()
    conv = full_convolution_fft(a)
    cur_score = 2.0 * n * conv.max()

    # hyper‑parameters
    beta = 0.5            # smooth‑max temperature (starts low)
    beta_max = 30.0
    eta = 0.5             # additive step size
    eta_min = 1e-4
    eta_decay = 0.9997
    temperature = 1.0
    temp_decay = 0.9995
    last_improve = 0
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ----- gradient of the smooth approximation of max(conv) -----
        grad, conv, max_val = smooth_gradient(a, beta)

        # ----- tentative additive update on the simplex -----
        a_cand = a - eta * grad
        a_cand = simplex_projection(a_cand)

        # ----- evaluate candidate -----
        conv_cand = full_convolution_fft(a_cand)
        cand_score = 2.0 * n * conv_cand.max()

        # ----- simulated‑annealing acceptance -----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_cand
            conv = conv_cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new global best = {best_score:.6f}")

        # ----- schedule updates -----
        beta = min(beta * 1.001, beta_max)          # anneal smooth‑max
        eta = max(eta * eta_decay, eta_min)         # slowly shrink step
        temperature = max(temperature * temp_decay, 1e-8)

        # ----- occasional re‑heat to escape plateaux -----
        if iteration - last_improve > 30000:
            # add tiny random noise and re‑normalise
            noise = np.random.rand(n) * 1e-4
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            conv = full_convolution_fft(a)
            cur_score = 2.0 * n * conv.max()
            temperature = 1.0
            eta = 0.5
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

        # ----- safety time check -----
        if iteration % 2000 == 0 and time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
