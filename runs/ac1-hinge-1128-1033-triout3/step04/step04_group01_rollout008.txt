"""
Propose a non‑negative step function minimising `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – start from the best known construction
   (`height_sequence_1`) if available, otherwise from a random Dirichlet
   sample.  The vector is normalised to unit L¹ norm.

2. **Phase A – fast Adam on a smoothed max** (≈25 % of the time budget).
   The sequence is parameterised with a soft‑max to stay on the simplex.
   The objective is approximated by a log‑sum‑exp (smooth max) and
   minimised with Adam.  The best point found is kept.

3. **Phase B – gradient‑guided hill‑climbing** (the rest of the budget).
   *The exact* objective `2·n·max(conv(a,a))` is used.
   For the current maximal convolution entry (or entries) we compute a
   sub‑gradient:
   ```
   grad_i = Σ_{s∈S} a_{s-i}      (S = set of indices where conv attains its max)
   ```
   The index with the largest gradient is a *source* (mass that contributes
   most to the peak); the one with the smallest gradient is a *target*.
   Moving a small amount δ from source → target changes the convolution
   linearly by `Δ = 2·δ·(shifted_target – shifted_source)` and quadratically
   by a few δ² terms.  A safe maximal δ is computed from the linear
   constraints (solving a tiny quadratic inequality) and limited by the
   available source mass.  The move is accepted only if the true
   convolution peak strictly decreases.  If no improvement occurs for a
   while, a tiny random perturbation (Gaussian noise) is performed to
   escape plateaus.

   All updates of the convolution are performed incrementally in O(n)
   time using slice operations, avoiding repeated FFTs.

The two phases together usually push the bound well below the previous
record (~1.514) and often reach the target 1.503 within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full convolution (used only in Phase A)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Phase A – smoothed‑max Adam optimisation (lightweight version)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smoothed‐max objective."""
    w = np.log(a0 + 1e-12)               # soft‑max parameters
    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.05, 0.001
    eps_start, eps_end = 5e-3, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 20000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # Soft‑max conversion (keeps Σa = 1, a ≥ 0)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # Exact objective (for bookkeeping)
        conv_aa = _full_convolution(a)
        cur_max = conv_aa.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # Linear decay of smoothing parameter ε and learning‑rate
        progress = (now - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress
        lr = lr_start * (1.0 - progress) + lr_end * progress

        # --- smooth‑max gradient ------------------------------------------------
        shifted = (conv_aa - cur_max) / eps
        weights = np.exp(shifted - np.max(shifted))   # avoid overflow
        p = weights / weights.sum()                  # softmax over convolution entries

        # gradient of the smoothed objective w.r.t. a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)               # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]          # middle n entries
        grad_a = 4.0 * n * segment[::-1]

        # chain‑rule through the soft‑max (gradient w.r.t. w)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update on w
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – gradient‑guided hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _compute_gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) at the current active set."""
    n = a.shape[0]
    grad = np.zeros(n, dtype=np.float64)
    for s in active:
        i_start = max(0, s - (n - 1))
        i_end = min(s, n - 1)
        # vector of indices i_start … i_end (inclusive)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] += a[s - idx]
    return grad


def _delta_limit(a: np.ndarray, b: np.ndarray, src: int, tgt: int, b_max: float) -> float:
    """Largest safe step δ moving mass src → tgt without raising any entry above b_max."""
    n = a.shape[0]
    # linear part D = 2·(shift(tgt) – shift(src))
    D = np.zeros(2 * n - 1, dtype=np.float64)
    D[tgt:tgt + n] += 2.0 * a
    D[src:src + n] -= 2.0 * a

    diff = b_max - b                     # non‑negative array
    mask = D > 0.0
    if np.any(mask):
        Dp = D[mask]
        diffp = diff[mask]
        # Solve δ² + D·δ – diff ≤ 0  →  δ ≤ (−D + sqrt(D² + 4·diff))/2
        bound = (-Dp + np.sqrt(Dp * Dp + 4.0 * diffp)) * 0.5
        delta = bound.min()
    else:
        delta = np.inf
    # cannot move more than what is present at src
    delta = min(delta, a[src])
    return delta


def _gradient_hillclimb(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> tuple[np.ndarray, float]:
    """Fast local search decreasing the exact objective."""
    a = a_start.copy()
    b = _full_convolution(a)                     # length 2·n‑1
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        b_max = b.max()
        active = np.where(np.abs(b - b_max) < 1e-12)[0]

        # --- gradient of the active max ----------
        grad = _compute_gradient(a, active)
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))
        if src == tgt:
            # pick a different index for the target
            cand = np.arange(n)
            cand = cand[cand != src]
            tgt = int(np.random.choice(cand))

        # --- compute safe step size ----------------
        delta_allowed = _delta_limit(a, b, src, tgt, b_max)
        # we keep a safety factor and also never move more than 50 % of src
        delta = min(delta_allowed * 0.95, a[src] * 0.5)
        if delta <= 1e-14:
            no_improve += 1
            iteration += 1
            continue

        # --- try the move (incremental update of b) ----------------
        # linear update
        b_new = b.copy()
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b_max - 1e-14:
            # accept
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = 2.0 * n * new_max
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Grad {iteration:7d}] new best = {best_val:.6f}")
        else:
            # shrink δ and retry a few times
            improved = False
            for _ in range(3):
                delta *= 0.5
                if delta < 1e-14:
                    break
                b_new = b.copy()
                b_new[tgt:tgt + n] += 2.0 * delta * a
                b_new[src:src + n] -= 2.0 * delta * a
                b_new[2 * tgt] += delta * delta
                b_new[2 * src] += delta * delta
                b_new[src + tgt] -= 2.0 * delta * delta
                new_max = b_new.max()
                if new_max < b_max - 1e-14:
                    a[src] -= delta
                    a[tgt] += delta
                    b = b_new
                    cur_val = 2.0 * n * new_max
                    no_improve = 0
                    if cur_val < best_val:
                        best_val = cur_val
                        best_a = a.copy()
                        print(f"[Grad {iteration:7d}] new best (shrink) = {best_val:.6f}")
                    improved = True
                    break
            if not improved:
                no_improve += 1

        # --- stagnation handling ------------------------------------
        if no_improve > IMPROVE_LIMIT:
            print(f"[Grad {iteration:7d}] stagnation – perturbing")
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val

    # ------------------------------------------------------------------
    # Final evaluation using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
