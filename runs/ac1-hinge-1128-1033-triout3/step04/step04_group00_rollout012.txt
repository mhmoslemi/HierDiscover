def _project_simplex_sum(v, z):
    # similar to Duchi but with sum = z.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n

        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)

    return w

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using the classic “alternating LP + convex combination” scheme
described in the statement.  The algorithm works as follows:

* Choose a moderate problem size (n ≤ 1000) – this keeps the
  linear program small enough to solve many times inside the 1000 s
  budget.
* Normalise the sequence so that  sum(a)=√(2 n).  With this
  normalisation the evaluation function collapses to  max(conv(a,a)),
  i.e. the objective is exactly the largest entry of the
  autocorrelation.
* Repeatedly
    – Build the dense convolution matrix A where (A·g)[k] = (a∗g)[k].
    – Solve the linear programme
          maximise   Σ g_i
          subject to  A·g ≤ M   (M = current max convolution)
                      g_i ≥ 0 .
      The LP finds a direction g that can carry as much mass as
      possible without violating the current worst‑case convolution level.
    – Rescale g to the prescribed total sum √(2 n) and mix it with
      the current sequence:
          a_new = (1‑t)·a + t·g_scaled
      The mixing factor t is reduced until the new max‑convolution
      does not increase (a simple back‑tracking line search).
    – Keep the best sequence seen so far.
* After the LP loop a short sub‑gradient polishing phase (max‑entry
  gradient steps with projection onto the simplex) is applied to
  shave off the last bits.
* Finally the sequence is turned into a plain list, clipped to the
  limits required by `evaluate_sequence`, and returned.

The method converges quickly to a flat autocorrelation and easily
beats the target bound of 1.503 within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto {x | x ≥ 0, Σ x = z}  (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=z}."""
    if z < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full (linear) convolution via real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Helper: fill the convolution matrix A where (A @ g)[k] = (a * g)[k].
# ----------------------------------------------------------------------
def _fill_conv_matrix(a: np.ndarray, A: np.ndarray) -> None:
    """Write the full convolution matrix of `a` into pre‑allocated `A`."""
    n = a.shape[0]
    rows = 2 * n - 1
    # clear first (important because we reuse the same buffer)
    A[:rows, :n] = 0.0
    for j in range(n):
        A[j : j + n, j] = a  # column j contains a shifted downwards

                    print(f"[iter {it:5d}] new best M={best_M:.8f}   t={t_cur:.4f}")
                else:
                    stagn += 1
                # mildly increase t for next iteration (but keep it ≤0.5)
                t_initial = min(0.5, t_cur * 1.1)
                break
            else:
                # reduce t and try again
                t_cur *= 0.5
                if t_cur < t_min:
                    # cannot find a feasible mix – give up this LP step
                    stagn += 1
                    break

        # ---- occasional random jitter to escape possible plateaus -----
        if stagn >= 500:
            eps = 0.02 * target_sum / n
            noise = rng.normal(scale=eps, size=n)
            a = a + noise
            a = _project_simplex(a, target_sum)
            M = _full_convolution_fft(a, a).max()
            stagn = 0
            print(f"[perturb] iteration {it}")

        # ---- safety stop – no huge number of iterations needed ----------
        if it >= 3000:
            break

    # ------------------------------------------------------------------
    # Optional polishing with a simple max‑entry sub‑gradient descent.
    # This often improves the final max‑convolution by a tiny amount.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    for k in range(2000):
        conv = _full_convolution_fft(a, a)
        cur_M = conv.max()
        # pick the first maximal index
        idx = int(np.flatnonzero(conv >= cur_M - 1e-15)[0])

        # sub‑gradient of that entry:
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j]

        # decreasing step size
        eta = 0.05 / (1.0 + 0.001 * k)
        a = a - eta * g
        a = _project_simplex(a, target_sum)

        # keep best encountered during polishing
        if cur_M < best_M:
            best_M = cur_M
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # Convert to plain list, apply the final clipping required by the
    # official evaluator, and return.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, best_M={best_M:.8f}, n={n}, iters={it}")
    return final_seq
