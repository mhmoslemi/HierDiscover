def adam_optimize(a0, max_iters, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.1
    best_local = a.copy()
    best_score_local = score_vector(a)
    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        conv = full_convolution_fft(a)
        grad = compute_subgradient(a, conv)
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        cand = a - step
        cand = np.maximum(cand, 0)
        cand = simplex_projection(cand)
        cand /= cand.sum()
        cand_score = score_vector(cand)
        if cand_score < best_score_local:
            best_score_local = cand_score
            best_local = cand.copy()
        a = cand
        # optional: if no improvement for many iters, break or reheat
    return best_local, best_score_local
```

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimises

        2 * n * max_k (a * a)[k]

where ``*`` is the full linear convolution.  The routine consists of three
stages:

1.  Deterministic sweep over a variety of parametric families (symmetric,
    monotone, truncated) on several problem sizes.  The best few candidates
    are kept as seeds.
2.  A global search with Differential Evolution on the probability simplex.
    The population size is larger than the reference implementation and the
    mutation strategy is “best/1” which accelerates convergence.
3.  A fast projected Adam‑style sub‑gradient polish that directly minimises
    the maximum convolution value while remaining on the simplex.

All heavy work is performed with FFT‑based convolution, which is
``O(n log n)`` and fast enough for the 1000 s budget.  Progress and
intermediate scores are printed; the best vector found (sum≈1) is returned as
a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, exact)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2*n-1)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector with sum==1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a (handles non‑smooth max)."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        # each ordered pair (i,j) appears once in the convolution;
        # derivative of a_i * a_j w.r.t a_i is a_j (and similarly for a_j).
        # The implementation below adds the contribution for both sides,
        # which matches the sub‑gradient used in the reference code.
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Parametric families (used for deterministic seed generation)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shaped: a[i] ∝ (min(i, n‑1‑i)+1)^(-p) (edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shaped product: a[i] ∝ ((i+1)*(n‑i))^(-p)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def decreasing_distribution(n: int, p: float) -> np.ndarray:
    """Monotone decreasing: a[i] ∝ (i+1)^(-p)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = i ** (-p) + 1e-14
    a /= a.sum()
    return a


def truncated_decreasing_distribution(n: int, p: float, cut_frac: float) -> np.ndarray:
    """Monotone decreasing on a leading block of length ceil(cut_frac*n), zeros elsewhere."""
    cut = max(1, int(np.ceil(cut_frac * n)))
    i = np.arange(cut, dtype=np.float64) + 1.0
    a = i ** (-p) + 1e-14
    full = np.zeros(n, dtype=np.float64)
    full[:cut] = a
    full /= full.sum()
    return full


# ----------------------------------------------------------------------
# Differential Evolution on the simplex (best/1 strategy)
# ----------------------------------------------------------------------
def de_optimize(init_population: np.ndarray,
                max_time: float,
                deadline: float,
                best_global_score: float,
                best_global_vec: np.ndarray) -> tuple[np.ndarray, float]:
    """
    DE with a “best/1” mutation scheme.  The routine runs until the global
    deadline is reached and returns the best vector found together with its
    score.
    """
    pop = init_population.copy()
    pop_scores = np.array([score_vector(ind) for ind in pop])
    pop_size, n = pop.shape

    # store the current global best
    best_idx = int(pop_scores.argmin())
    best_vec = pop[best_idx].copy()
    best_score = float(pop_scores[best_idx])

    F = 0.9
    CR = 0.9
    generation = 0
    while time.time() < deadline:
        generation += 1
        # one DE generation
        for i in range(pop_size):
            # best/1: donor = best + F * (a - b)
            idxs = np.arange(pop_size)
            idxs = idxs[idxs != i]
            a, b = np.random.choice(idxs, 2, replace=False)
            donor = best_vec + F * (pop[a] - pop[b])
            donor = np.maximum(donor, 0.0)

            # binomial crossover
            crossover_mask = np.random.rand(n) < CR
            if not np.any(crossover_mask):
                crossover_mask[np.random.randint(n)] = True
            trial = np.where(crossover_mask, donor, pop[i])

            # project back onto the simplex
            trial = simplex_projection(trial)
            trial /= trial.sum()

            trial_score = score_vector(trial)

            # selection
            if trial_score < pop_scores[i]:
                pop[i] = trial
                pop_scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    print(f"DE gen {generation:4d} ind {i:2d}: new best = {best_score:.6f}")

        # occasional time‑check
        if time.time() > deadline - 2.0:   # keep a safety margin
            break
    return best_vec, best_score


# ----------------------------------------------------------------------
# Adam‑style sub‑gradient polish
# ----------------------------------------------------------------------
def adam_polish(start_vec: np.ndarray,
                deadline: float,
                best_score_global: float,
                best_vec_global: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Fast projected Adam optimisation of the max‑convolution term.
    Stops when the time budget is exhausted.
    """
    a = start_vec.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_local = a.copy()
    best_local_score = score_vector(a)

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution_fft(a)
        grad = compute_subgradient(a, conv)

        # Adam update (we minimise, so move opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)
        cand /= cand.sum()
        cand_score = score_vector(cand)

        if cand_score < best_local_score:
            best_local_score = cand_score
            best_local = cand.copy()
            last_improve = it
            if cand_score < best_score_global:
                best_score_global = cand_score
                best_vec_global = cand.copy()
                print(f"Adam it {it:6d}: new global best = {best_score_global:.6f}")

        a = cand

        # simple re‑heat if stuck for a while
        if it - last_improve > 4000:
            bump = np.random.randn(n) * 1e-5
            a = simplex_projection(a + bump)
            a /= a.sum()
            last_improve = it
            print(f"Adam it {it:6d}: re‑heat (stagnation)")

        # safety stop
        if lr * np.linalg.norm(step) < 1e-12:
            break

    return best_vec_global, best_score_global


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the given time
    budget. Returns a plain Python list of floats (∑a ≈ 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for the final return

    # ------------------------------------------------------------------
    # 1. Deterministic sweep – collect a handful of promising seeds
    # ------------------------------------------------------------------
    top_k = 12                     # how many seeds we keep for later optimisation
    seeds = []                     # list of (score, vector)

    def try_candidate(vec):
        """store candidate if it ranks among the best k seen so far."""
        nonlocal seeds
        sc = score_vector(vec)
        if len(seeds) < top_k:
            seeds.append((sc, vec.copy()))
            seeds.sort(key=lambda x: x[0])
        else:
            if sc < seeds[-1][0]:
                seeds[-1] = (sc, vec.copy())
                seeds.sort(key=lambda x: x[0])

    lengths = [2000, 4000, 6000, 8000, 10000]
    p_grid = np.linspace(0.2, 4.0, 22)      # includes sub‑0.5 values
    lam_grid = np.logspace(-5, 0, 20)      # up to 1.0 for exponential decay
    cut_fracs = [0.3, 0.5, 0.7, 0.9]

    print("=== Deterministic sweep start ===")
    for n in lengths:
        if time.time() > deadline:
            break
        # symmetric families (already used in the reference code)
        for p in p_grid:
            try_candidate(u_min_distribution(n, p))
            try_candidate(u_prod_distribution(n, p))
        for lam in lam_grid:
            try_candidate(exp_distribution(n, lam))
        # new monotone families
        for p in p_grid:
            try_candidate(decreasing_distribution(n, p))
        for p in p_grid:
            for cf in cut_fracs:
                try_candidate(truncated_decreasing_distribution(n, p, cf))
        print(f" length {n:5d} sweep done, best so far = {seeds[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. Random Dirichlet seeds (including sparse draws)
    # ------------------------------------------------------------------
    print("\n=== Random Dirichlet seeds ===")
    for _ in range(30):
        n = np.random.choice(lengths)
        alpha = np.random.uniform(0.05, 2.0)   # concentration; <1 favours sparsity
        samp = np.random.gamma(alpha, 1.0, size=n)
        vec = samp / samp.sum()
        try_candidate(vec)
        if time.time() > deadline:
            break
    print(f"After random seeds best = {seeds[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise Differential Evolution population (best‑1 strategy)
    # ------------------------------------------------------------------
    best_vec, best_score = seeds[0][1].copy(), seeds[0][0]
    n = best_vec.shape[0]
    pop_size = max(30, len(seeds) * 2)      # ensure a reasonably sized population
    population = np.empty((pop_size, n), dtype=np.float64)

    # Fill population: best vector + several deterministic variations + random draws
    population[0] = best_vec
    for i, (_, vec) in enumerate(seeds[1:pop_size], start=1):
        population[i] = vec
    for i in range(len(seeds), pop_size):
        rnd = np.random.rand(n)
        population[i] = rnd / rnd.sum()

    # ------------------------------------------------------------------
    # 4. Global DE optimisation (best/1 scheme)
    # ------------------------------------------------------------------
    print("\n=== Differential Evolution (best/1) start ===")
    best_vec, best_score = de_optimize(population, budget_s, deadline,
                                       best_score, best_vec)

    # ------------------------------------------------------------------
    # 5. Adam‑style sub‑gradient polishing (local refinement)
    # ------------------------------------------------------------------
    print("\n=== Adam sub‑gradient polishing start ===")
    best_vec, best_score = adam_polish(best_vec, deadline, best_score, best_vec)

    # ------------------------------------------------------------------
    # 6. Return result
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}  (length {len(best_vec)})")
    return best_vec.tolist()
