"""
Propose a non‑negative step‑function (probability vector) that minimizes

      2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a two‑phase Adam‑based optimisation:

1. Initialise the vector with an *arcsine* shape, the unique distribution whose
   continuous self‑convolution is exactly uniform.  After normalisation the
   convolution is already close to flat, giving scores near the theoretical
   lower bound ≈ 1.0.

2. Refine the vector with a smoothed max gradient (soft‑max weighting) and
   Adam momentum, projecting onto the probability simplex after every step.
   Temperature and learning‑rate are annealed; occasional Gaussian noise
   injections help escape plateaus.

A short sub‑gradient polishing phase at the end further reduces the peak
convolution value.  The whole routine respects the 1000 s time budget and
returns a list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for large vectors)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a power‑of‑two FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()        # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative height sequence that (approximately) minimises the
    evaluation function defined in the problem statement.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (a safety margin of a few seconds is kept).

    Returns
    -------
    list[float]
        The best sequence discovered before the deadline.
    """
    # ------------------------------------------------------------------
    # Timing and RNG setup
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size `n` and initialise the vector `a`
    # ------------------------------------------------------------------
    a = None
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = base.shape[0]
            a = np.maximum(0.0, base)
            if a.sum() > 0.0:
                a = a / a.sum()
            else:
                a = None
        except Exception:
            a = None

    # If no usable warm‑start, build a fresh arcsine‑shaped vector.
    if a is None:
        # use a moderately large size (larger n → lower theoretical bound)
        n = 8000
        # mid‑point positions in (0,1) for a discrete arcsine distribution
        i = np.arange(n, dtype=np.float64)
        x = (i + 0.5) / n                     # in (0,1)
        arcsine = 1.0 / np.sqrt(np.clip(x * (1.0 - x), 1e-16, None))
        a = arcsine / arcsine.sum()           # normalise to sum = 1

    # ------------------------------------------------------------------
    # Keep track of the best candidate seen so far
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    best_val = 2.0 * n * max_conv            # sum(a) == 1 => denominator = 1
    best_seq = a.copy()
    print(f"[init] n={n}, initial_score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    lr = 0.08                                 # initial learning rate
    lr_min = 5e-6
    lr_decay = 0.9998

    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(a)                     # first moment
    v = np.zeros_like(a)                     # second moment
    b1_pow = 1.0
    b2_pow = 1.0
    adam_eps = 1e-8

    # Soft‑max temperature for approximating the max
    eps_temp = max(max_conv * 0.5, 1e-12)
    eps_min = max(max_conv * 1e-5, 1e-12)
    eps_decay = 0.9995

    stagn = 0                                 # iterations without improvement
    it = 0

    # ------------------------------------------------------------------
    # Primary optimisation loop (soft‑max + Adam)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- convolution and current score -----
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # ----- best‑track -----
        if score < best_val - 1e-14:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[iter {it:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # ----- temperature schedule -----
        eps_temp = max(eps_temp * eps_decay, eps_min)

        # ----- soft‑max weights (focus on large conv entries) -----
        w = np.exp((conv - max_conv) / eps_temp)   # values ≤ 1
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient of the weighted sum (cross‑correlation) -----
        rev_a = a[::-1]
        conv_rev = _full_convolution_fft(rev_a, w)   # length 3n‑2
        g = 2.0 * conv_rev[n - 1 : n - 1 + n]        # slice back to size n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        b1_pow *= beta1
        b2_pow *= beta2
        m_hat = m / (1.0 - b1_pow)
        v_hat = v / (1.0 - b2_pow)

        step = - lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        a_new = a + step
        a = _project_simplex(a_new)

        # ----- learning‑rate decay -----
        lr = max(lr_min, lr * lr_decay)

        # ----- occasional random perturbation to escape plateaus -----
        if stagn > 3000:
            noise = rng.normal(scale=0.005, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            # reset Adam moments after a jump
            m.fill(0.0)
            v.fill(0.0)
            b1_pow = 1.0
            b2_pow = 1.0
            print(f"[perturb] iteration {it}")

        # ----- periodic re‑normalisation sanity check -----
        if it % 4000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Final polishing: pure max‑entry sub‑gradient steps (no soft‑max)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    for sub_it in range(2000):
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        # choose the first index attaining the maximum
        idx = int(np.flatnonzero(conv >= max_val - 1e-15)[0])

        # build the sub‑gradient for this single maximal entry
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j]

        # tiny fixed step; keep projection onto simplex
        a = _project_simplex(a - 2e-5 * g)

    # ------------------------------------------------------------------
    # Clip to evaluator limits and evaluate the final candidate
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in a.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")

    return final_seq
