"""
Propose a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`.  The algorithm works in two phases:

* Phase A – smooth‑max Adam optimisation on the simplex.
  The vector a is represented by soft‑max logits w (so Σa = 1, a ≥ 0).
  We minimise a smooth approximation of `max(conv(a,a))` using a
  log‑sum‑exp with an exponentially decaying temperature ε.
  Adam’s moments, occasional random perturbations and restart on
  stagnation help escaping local minima.

* Phase B – targeted hill‑climbing on the exact objective.
  Starting from the best point of Phase A we keep the full self‑convolution
  b = a*a.  In each move a small amount of mass δ is transferred from a
  “source” index i to a “target” index j.  The source is chosen
  proportionally to its contribution to the current maximum entry
  `b[k_max]`; the target is sampled uniformly.  The convolution is
  updated exactly in O(n) by adding the linear effect of the mass
  transfer and the pure quadratic terms.  Only strictly improving moves
  are kept; after a long period without improvement we perturb the
  current best solution and continue.  This focussed search usually
  beats a plain random hill‑climber and is able to push the objective
  below the 1.503 target within the 1000 s budget.

Both phases run inside the allocated time budget; the best sequence seen
is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective helper (exact, sum of a is assumed = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) (sum(a) is 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth approximation of max(conv(a,a)).
    Returns the best (a, value) seen before *deadline*.
    """
    # logits for soft‑max, guaranteeing Σa = 1, a ≥ 0
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.10, 0.001
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---------- soft‑max to obtain a on the simplex ----------
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()                       # Σa = 1, a ≥ 0

        # ---------- exact objective for tracking ----------
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                # early exit – target already reached
                break
        else:
            no_improve_cnt += 1

        # ---------- schedule for ε and learning‑rate ----------
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ---------- smoothed max via log‑sum‑exp ----------
        b_full = _full_convolution(a)           # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)               # safe in float64
        p = weights / weights.sum()             # p ≈ one‑hot when eps tiny

        # ---------- gradient of smooth objective w.r.t. a ----------
        # p_rev is the reverse of p (needed for convolution)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)           # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]     # n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]        # shape (n,)

        # ---------- chain rule through the soft‑max ----------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---------- Adam update ----------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ---------- occasional random perturbation ----------
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # ---------- restart if stagnated ----------
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restarting after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """
    Fast local search that transfers a small amount of mass between
    coordinates while keeping Σa = 1.  The source index is chosen
    proportionally to its contribution to the current maximal
    convolution entry, which yields a more directed descent than a
    completely random source selection.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # parameters for the mass‑transfer moves
    max_frac_start, max_frac_end = 0.5, 0.02
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress based schedule for the allowed transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # ---- locate the current maximal convolution entry ----
        k_max = int(np.argmax(b))

        # ---- compute contributions of each index to b[k_max] ----
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ---- choose a source index (biased towards large contributions) ----
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            continue

        # ---- choose a target index (uniform, different from i) ----
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # ---- amount of mass to move ----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n)) ----
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ---- accept only strictly improving moves ----
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[HillTar {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    # reached the target – exit early
                    break
        else:
            no_improve += 1

        # ---- stagnation handling ----
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillTar {iteration:7d}] perturbing after stagnation")
        iteration += 1

    return best_a

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin of 0.5 s for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – we pick a larger n than the previous 4000; the
    # theoretical optimum 2·n·max(conv) → 1 as n grows, giving more
    # room for improvement.
    # ------------------------------------------------------------------
    n = 8000          # feel free to adjust (memory ≈ 250 KB for the convolution)

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a known construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        # fall back to a simple smooth bump (Gaussian‑like) plus a uniform mix
        x = np.arange(n, dtype=float)
        mu = n / 2.0
        sigma = n / 6.0
        gaussian = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        a0 = gaussian
        a0 /= a0.sum()

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_val = float('inf')
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – smooth‑max Adam optimisation (≈ 40 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.40 * budget_s
    best_a, best_val = _adam_phase(a0, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase B – targeted hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    best_a = _hillclimb_targeted(best_a, n,
                                 start=phase_a_deadline,
                                 deadline=deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=TARGET)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        # should never happen, but we guard against unexpected errors
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
