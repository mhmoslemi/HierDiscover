def softmax_gradient(a, conv, alpha):
    max_val = conv.max()
    # compute weights
    exp_weights = np.exp(alpha * (conv - max_val))
    prob = exp_weights / exp_weights.sum()
    # reverse prob
    prob_rev = prob[::-1]
    N = a.shape[0]
    L = 1 << ((N + (2*N-1) - 2).bit_length())  # next power of 2 >= 3N-2
    # maybe we can pre-allocate L each iteration; it's small overhead.
    fft_a = np.fft.rfft(a, n=L)
    fft_pr = np.fft.rfft(prob_rev, n=L)
    conv_full = np.fft.irfft(fft_a * fft_pr, n=L)
    # get grad segment
    start = 2*N - 2
    grad = 2.0 * conv_full[start : start + N]
    return grad

Optimisation of the max-autoconvolution objective via
smooth‑max gradient descent on the probability simplex.
We start from the best known construction, up‑sample it to a
larger length (≈20000) and then run an Adam‑style projected
gradient descent, where the gradient of a soft‑max surrogate of the
maximum convolution entry is computed efficiently with FFTs.
Occasional random perturbations (“re‑heat”) and a simple simulated
annealing acceptance rule help escape plateaus.
The routine respects the 1000 s budget and returns the best
vector found (normalised to sum ≈ 1).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (real FFT)
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a * a (real FFT).
    Returns a NumPy array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Gradient of a smooth approximation of the max‑convolution entry
# ----------------------------------------------------------------------


def softmax_max_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute ∇_a  ( soft‑max_{k} conv[k] ) using the weighting
    w_k ∝ exp(alpha * (conv[k] - max(conv))/Δ),
    where Δ = max(conv) - min(conv) + eps.
    The gradient is 2 * (a * w_rev) evaluated on the appropriate slice.
    """

    min_val = conv.min()

    if delta == 0.0:
        # all entries equal → gradient is zero; return zero vector.
        return np.zeros_like(a)
    # scaled soft‑max weights
    exp_weights = np.exp(alpha * (conv - max_val) / (delta + 1e-15))
    prob = exp_weights / exp_weights.sum()
    prob_rev = prob[::-1]                     # reverse for correlation
    N = a.shape[0]
    # length needed for a * prob_rev convolution
    L = 1 << ((3 * N - 2).bit_length())
    # FFTs
    fa = np.fft.rfft(a, n=L)
    fp = np.fft.rfft(prob_rev, n=L)
    corr = np.fft.irfft(fa * fp, n=L)
    start = 2 * N - 2
    grad = 2.0 * corr[start : start + N]
    return grad

    # --------------------------------------------------------------

Optimisation of the max‑autoconvolution objective via
projected Adam gradient descent on a smooth‑max surrogate.

The goal is to minimise

    2 * n * max_k (a * a)[k]

            best_seq = a.copy()
            print(f"[det] new best U‑min p={p:.1f} score={score:.6f}")

    for lam in (1e-4, 5e-4, 2e-3, 1e-2):
        a = exponential(TARGET_N, lam)
        conv = full_convolution_fft(a)
        score = 2.0 * TARGET_N * conv.max()
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[det] new best exp lam={lam:.5f} score={score:.6f}")

    # --------------------------------------------------------------
    # 2. Stochastic refinement – projected Adam + soft‑max gradient
    # --------------------------------------------------------------
    a = best_seq.copy()
    N = a.shape[0]

    # Adam state
    m = np.zeros(N, dtype=np.float64)
    v = np.zeros(N, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # hyper‑parameters
    lr0 = 0.8                     # initial learning rate
    lr = lr0
    lr_decay = 0.9995

    alpha_start = 5.0
    alpha_end = 50.0

    iteration = 0
    last_improve = 0

    print("\n=== Stochastic refinement start ===")
    while time.time() < deadline:
        iteration += 1

        # current evaluation
        conv = full_convolution_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * N * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # temperature schedule for the soft‑max
        elapsed = time.time() - start_time
        total = deadline - start_time
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / total)

        # gradient of the smooth‑max surrogate
        grad = softmax_max_gradient(a, conv, alpha)

        # Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # project back onto the simplex
        a_new = simplex_projection(a_new)

        # occasional random “re‑heat” to escape stagnation
        if iteration % 10000 == 0:
            bump = np.random.rand(N) * 1e-5
            a_new = simplex_projection(a_new + bump)
            m.fill(0.0)
            v.fill(0.0)
            lr = lr0
            print(f"Iter {iteration:7d}: re‑heat + reset moments")

        # update the iterate
        a = a_new

        # learning‑rate decay
        lr *= lr_decay

        # strong re‑heat after a long stall
        if iteration - last_improve > 250000:
            a = simplex_projection(a + np.random.rand(N) * 1e-4)
            m.fill(0.0)
            v.fill(0.0)
            lr = lr0
            last_improve = iteration
            print(f"Iter {iteration:7d}: strong re‑heat after stagnation")

    # --------------------------------------------------------------
    # 3. Return the best sequence discovered
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
