"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.  The algorithm is a multi‑restart optimisation:

* each restart works on a chosen length ``n`` (2000–8000);
* the sequence is parameterised by a soft‑max ``a = softmax(w)`` so that
  ``a_i ≥ 0`` and ``∑ a_i = 1`` (the objective is scale‑invariant);
* a smooth log‑sum‑exp surrogate of the max‑convolution
  ``L_ε(a) = 2·n·ε·log∑_k exp((a*a)_k/ε)`` is minimised with L‑BFGS‑B,
  using a continuation schedule for the smoothing parameter ε;
* after the continuation a cheap hill‑climbing refinement (random mass
  transfers) is performed;
* the best sequence among all restarts (within the 1000 s budget) is
  returned.

All heavy work (convolutions, gradient) is performed with FFTs for
speed.  No closures or lambdas are used – every helper is a top‑level
function or class.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself (length 2·len‑1).
    Uses FFT for O(n log n) speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)   # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp of the convolution) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smoothed objective L_ε and its gradient w.r.t. soft‑max
    parameters w for a given smoothing ε and length n.
    Returns (L, ∂L/∂w).
    """
    # soft‑max (ensures a_i ≥ 0, Σa_i = 1)
    w_shift = w - np.max(w)               # numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # full self‑convolution a*a
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smoothed max: L = 2·n·max_conv + 2·n·ε·log∑_k exp((conv_k‑max_conv)/ε)
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form):
    #   p_k = exp(conv_k/ε) / Σ_j exp(conv_j/ε)
    #   ∂L/∂a_i = 4·n·∑_k p_k a_{k‑i}
    weights = np.exp(shifted)
    p = weights / weights.sum()                 # shape (2·n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)               # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]              # middle n entries
    grad_a = 4.0 * n * seg[::-1]                # gradient wrt a, shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)                 # ∂L/∂w
    return L, grad_w


# ----------------------------------------------------------------------
# Simple hill‑climbing (random mass transfers) for final refinement
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Perform cheap random mass‑transfer moves while time permits.
    Keeps Σa = 1 and a_i ≥ 0.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a fraction (≤5 %) of mass from i to j
        delta = cur_a[i] * 0.05 * rng.random()
        if delta == 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Callable class used by scipy.optimize.minimize (no closures)
# ----------------------------------------------------------------------
class _SmoothObj:
    """Callable returning (L, grad) for a given ε and length n."""
    __slots__ = ("eps", "n")

    def __init__(self, eps: float, n: int):
        self.eps = eps
        self.n = n

    def __call__(self, w: np.ndarray):
        return _smooth_obj_and_grad(w, self.eps, self.n)


# ----------------------------------------------------------------------
# Single optimisation run (one restart)
# ----------------------------------------------------------------------
def _run_one_restart(n: int, init_a: np.ndarray, deadline: float):
    """
    Optimise a sequence of length n, starting from ``init_a``.
    Returns (best_a, best_val) found before ``deadline``.
    """
    w = np.log(init_a + 1e-12)                 # soft‑max parameters
    best_a = init_a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, start value = {best_val:.6f}")

    # ε‑schedule (coarse → fine).  The final ε is very small to approximate a true max.
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(5e-9), num=15)

    for eps in eps_schedule:
        if time.time() >= deadline:
            break
        obj = _SmoothObj(eps, n)
        # L‑BFGS‑B with a modest iteration limit (the continuation already provides many steps)
        res = minimize(obj, w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # evaluate exact objective
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()
        try:
            cur_val = evaluate_sequence(a.tolist())
        except Exception:
            continue
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

    # final hill‑climbing refinement (uses the exact evaluation)
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    return best_a, best_val
