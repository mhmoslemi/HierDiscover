"""
propose_candidate
-----------------
Search for a non‑negative step‑function (probability vector) that minimises

        2 * n * max_k (a * a)[k]

where `*` denotes full linear convolution.  The routine works in three
phases:

1.  A deterministic sweep over several parametric families (U‑shaped,
    product‑U, exponential, beta‑type) for a handful of lengths.  The best
    vector found gives a good starting length and a seed.

2.  A smooth optimisation on the probability simplex using a
    soft‑max approximation of the ``max`` and L‑BFGS‑B.  The optimisation
    proceeds through a sequence of increasing smoothing parameters
    (γ = 5, 20, 80, 200 …).  The variables are the unconstrained logits
    `y` with `a = softmax(y)`.  Gradients are obtained analytically via
    the FFT for the convolution and the cross‑correlation needed for the
    soft‑max weighted sum.

3.  A short projected stochastic hill‑climb that moves a tiny amount of
    mass between random coordinates.  It helps to escape any remaining
    tiny local minima.

The whole search respects the 1000 s budget.  The best vector found is
returned as a plain Python list (its entries sum to 1, all are ≥ 0).
"""

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Utility: next power of two (for zero‑padding in FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Full linear convolution using real FFT (fast for large vectors)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return full linear convolution of real vectors x and y using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = _next_pow2(n + m - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
# Scoring utilities (exact equivalent of evaluate_sequence for sum≈1)
# ----------------------------------------------------------------------
def _score_vector(a: np.ndarray) -> float:
    """Exact score for a probability vector (∑a==1)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (produce a probability vector)
# ----------------------------------------------------------------------
def _u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i,n‑1‑i)+1)^(-p)   (edge‑heavy, “U” shape)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def _u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^(-p)   (edge‑heavy, product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def _exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i) + exp(-lam*(n‑1‑i))   (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def _beta_distribution(n: int, s: float) -> np.ndarray:
    """Symmetric Beta(s,s) density discretised on {0,…,n‑1}."""
    # Centres of bins
    x = (np.arange(n) + 0.5) / n
    a = (x ** (s - 1)) * ((1 - x) ** (s - 1)) + 1e-14
    a /= a.sum()
    return a


def _offset_power_distribution(n: int, p: float, off: float) -> np.ndarray:
    """a[i] ∝ (min(i,n‑1‑i)+off)^(-p).  Generalises the simple U‑shape."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d) + off
    a = d ** (-p) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth objective (soft‑max of convolution) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(y: np.ndarray, gamma: float) -> tuple[float, np.ndarray]:
    """
    Returns (f, g) where
        f(y) = 2*n * ( max(conv) + log( sum exp(gamma*(conv-max)) ) / gamma )
    and g = ∇_y f(y).
    ``y`` are unconstrained logits; the probability vector is
        a = softmax(y).
    """
    # soft‑max to obtain a probability vector
    # (subtract max for numerical stability)
    y_shift = y - np.max(y)
    a = np.exp(y_shift)
    a_sum = a.sum()
    a = a / a_sum

    n = a.shape[0]

    # full auto‑convolution a*a
    conv = _full_convolution_fft(a, a)            # length 2n‑1
    max_c = conv.max()

    # unnormalised soft‑max weights over the convolution entries
    # w_unnorm = exp(gamma * (conv - max_c))  -> ∈ (0,1]
    w_unnorm = np.exp(gamma * (conv - max_c))
    w_sum = w_unnorm.sum()
    # soft‑max weights (summing to 1)
    w = w_unnorm / w_sum

    # objective value (exact equivalent of the original max when gamma→∞)
    f = 2.0 * n * (max_c + np.log(w_sum) / gamma)

    # gradient w.r.t. a:
    #   ∂/∂a_i  Σ_k w_k Σ_j a_j a_{k-j}
    #   = 2 * Σ_k w_k a_{k-i}   (cross‑correlation of w and a)
    a_rev = a[::-1]
    # full convolution w * a_rev   -> length (2n-1)+n-1 = 3n-2
    corr_full = _full_convolution_fft(w, a_rev)
    start = n - 1
    grad_a = 2.0 * corr_full[start:start + n]      # shape (n,)

    # chain rule through the soft‑max mapping a = softmax(y)
    dot = np.dot(a, grad_a)                       # scalar = a·grad_a
    grad_y = (2.0 * n) * a * (grad_a - dot)

    return f, grad_y.astype(np.float64)

        for off in offsets:
            for p in offset_exps:
                a = _offset_power_distribution(n, p, off)
                sc = _score_vector(a)
                if sc < best_score:
                    best_score, best_vec = sc, a.copy()
                    print(f"new best (off‑p)  n={n:5d} p={p:4.2f} off={off:.2f}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate any external construction if available
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        if external.sum() > 0.0:
            external /= external.sum()
            sc = _score_vector(external)
            if sc < best_score:
                best_score, best_vec = sc, external.copy()
                print(f"external seed used, length {len(external)} score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 5000
        best_vec = np.random.rand(n)
        best_vec /= best_vec.sum()
        best_score = _score_vector(best_vec)
        print(f"fallback random seed, n={n} score={best_score:.6f}")

    n = best_vec.shape[0]
    print(f"\n=== Selected length for refinement: n = {n} ===")
    print(f"Seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth optimisation on the simplex (soft‑max of max)
    # ------------------------------------------------------------------
    # Initialise logits y = log(a) (adding a tiny epsilon for safety)
    eps = 1e-14
    y = np.log(best_vec + eps)

    # Schedule of smoothing parameters (larger γ → tighter approximation of max)
    gammas = [5.0, 20.0, 80.0, 200.0, 500.0]

    for gamma in gammas:
        if time.time() > deadline:
            break
        print(f"\n--- L‑BFGS‑B with γ = {gamma:.1f} ---")

        # wrapper returning (f, grad) for scipy
        def fun(y_vec):
            return _smooth_obj_and_grad(y_vec, gamma)

        # Run a modest number of iterations – L‑BFGS‑B is very efficient
        res = opt.minimize(fun,
                           y,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 30,
                                    'ftol': 1e-12,
                                    'disp': False})

        y = res.x
        # Extract the refined probability vector
        a = np.exp(y - np.max(y))
        a /= a.sum()
        cur_score = _score_vector(a)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"  → improvement after γ={gamma:.1f}: score={cur_score:.6f}")

        # keep a tiny time‑check
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 5. Very light stochastic hill‑climb (helps for any leftover micros)
    # ------------------------------------------------------------------
    print("\n=== Stochastic hill‑climb polishing ===")
    current = best_vec.copy()
    cur_score = best_score
    n = current.shape[0]
    rng = np.random.default_rng(seed)

    # We move a small mass δ from a random index i to j.
    # δ decays over time to allow fine‑grained adjustments.
    delta = 0.01
    delta_decay = 0.9995
    it = 0
    while time.time() < deadline and delta > 1e-9:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose move
        move = min(delta, current[i])
        if move == 0.0:
            continue
        cand = current.copy()
        cand[i] -= move
        cand[j] += move
        # re‑normalise (tiny numerical drift)
        cand /= cand.sum()
        cand_score = _score_vector(cand)
        if cand_score < cur_score:
            current = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                print(f"Iter {it:6d}: new best = {best_score:.6f}")

        delta *= delta_decay
        if it % 5000 == 0:
            # occasional progress report
            print(f"iter {it:6d} – delta {delta:.2e} – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {len(best_vec)})")
    return best_vec.tolist()
