def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Find non‑negative step heights that minimise the
    relative worst‑case convolution bound.
    Uses exponentiated gradient on the exact max‑subgradient,
    occasional random mass transfers, and a simple learning‑rate
    schedule.  The method runs until the time budget is exhausted
    and returns the best sequence discovered."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Choose length; may be increased if time permits.
    n = 2500   # moderate size, still fast

    # Initial sequence – random positive values normalized to sum 1.
    a = np.exp(np.random.randn(n))
    a /= a.sum()

    best_a = a.copy()
    conv = _full_convolution_fft(a, a)
    best_val = float(2 * n * conv.max())
    print(f"[init] n={n}, objective={best_val:.6f}")

    iter_count = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # Compute convolution and current objective
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2 * n * M)

        # Record best
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iter_count:7d}] new best = {best_val:.6f}")

        # Identify the index where the maximum occurs
        k_max = int(np.argmax(conv))

        # Build sub‑gradient of max_{k} (a*a)[k] with respect to a
        # g_i = 2 * a_{k_max-i} whenever the index lies in [0, n-1].
        g = np.zeros_like(a)
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = 2.0 * a[k_max - idx]

        # Simple learning‑rate schedule (linear decay)
        progress = (now - start) / (deadline - start)
        lr = 0.6 * (1.0 - progress) + 1e-5 * progress

        # Exponentiated‑gradient (multiplicative) update on the simplex
        a = a * np.exp(-lr * g)

        # Occasionally inject a random “mass‑transfer” move to escape plateaux
        if np.random.rand() < 0.02:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            # Move a random fraction (up to 50 %) of the mass at i to j
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta

        # Re‑normalise and guard against numerical drift
        a_sum = a.sum()
        if a_sum <= 0.0 or not np.isfinite(a_sum):
            a = np.full(n, 1.0 / n)
        else:
            a = np.maximum(a, 0.0)   # remove tiny negatives that may appear
            a /= a_sum

        iter_count += 1

    # Final evaluation using the supplied helper (mainly for logging)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()

"""
Optimiser for the “step‑function heights’’ challenge.

The objective returned by ``evaluate_sequence`` (with the natural normalisation
∑a_i = 1) is

    F(a) = 2·n·max_k (a * a)[k] ,

i.e. twice the length times the largest coefficient of the self‑convolution.
The problem is non‑convex; we use a simple projected (exponentiated) gradient
descent on the exact sub‑gradient of the max‑convolution together with a few
heuristics:

*  The gradient of  max_k (a*a)[k]  w.r.t. a is  g_i = 2·a_{k*−i}  for the index
   k* where the maximum is attained (zero outside the support).
*  An exponentiated‑gradient (multiplicative) update keeps a on the probability
   simplex automatically.
*  A linear learning‑rate schedule (≈0.6 → 1e‑5) and occasional random
   “mass‑transfer’’ jumps help to escape plateaux.
*  If no improvement has been seen for a while we re‑initialise from a fresh
   random point (the current best is always remembered).
*  When a previous good construction is available in the global variable
   ``height_sequence_1`` we start from it.

The optimiser runs until the allotted time budget (default 1000 s) is almost
exhausted and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution via FFT (real transforms are enough for real data)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*y (length len(x)+len(y)-1)."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    # next power of two for efficient FFT
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative values caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a non‑negative sequence that minimises the evaluation
    function.  The routine respects the given time budget, prints progress,
    and finally returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin so the surrounding harness does not kill us
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Determine the length of the sequence.
    # Prefer the length of a previously found construction if it exists.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(height_sequence_1, dtype=float)
        n = init_seq.size
        # positive normalisation (protect against accidental zeros)
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = init_seq / init_seq.sum()
    else:
        n = 2500                     # a modestly large size; still fast
        a = np.exp(np.random.randn(n))   # random positive start
        a /= a.sum()

    # ------------------------------------------------------------------
    # Book‑keeping of the best solution seen so far
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv.max())
    best_a = a.copy()
    print(f"[init] n={n}, initial objective={best_val:.6f}")

    # Parameters for the optimiser
    base_lr = 0.60                 # start learning rate (multiplicative update)
    min_lr = 1e-5                  # final learning rate
    restart_interval = 30.0        # seconds without improvement -> restart
    last_improve = start_time
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop (runs until the time budget expires)
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- convolution and objective -----
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)

        # ----- keep the best ever seen -----
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = now
            print(f"[{it:7d}] new best = {best_val:.6f}")

        # ----- restart if we have been stuck for a while -----
        if now - last_improve > restart_interval:
            # re‑initialise from a fresh random point (keep the global best)
            a = np.exp(np.random.randn(n))
            a /= a.sum()
            last_improve = now
            print(f"[{it:7d}] restart after stagnation")

        # ----- sub‑gradient of the max‑convolution -----
        # the maximum occurs at index k_max
        k_max = int(np.argmax(conv))

        # g_i = 2 * a_{k_max-i} for all i where the index is valid
        g = np.zeros_like(a)
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        idx = np.arange(low, high + 1, dtype=int)       # i where contribution exists
        g[idx] = 2.0 * a[k_max - idx]

        # ----- learning‑rate schedule (linear decay) -----
        progress = (now - start_time) / (deadline - start_time)
        lr = base_lr * (1.0 - progress) + min_lr * progress

        # ----- exponentiated‑gradient update (keeps a on the simplex) -----
        a = a * np.exp(-lr * g)

        # ----- occasional random “jitter’’ move (helps escape flat regions) -----
        if np.random.rand() < 0.02:          # ~2 % chance per iteration
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            # move a random fraction (up to 50 %) of the mass from i to j
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta

        # ----- normalise back onto the probability simplex -----
        a_sum = a.sum()
        if a_sum <= 0.0 or not np.isfinite(a_sum):
            a = np.full(n, 1.0 / n)          # fall back to uniform in the unlikely event
        else:
            # protect from tiny negatives that may arise from rounding
            np.maximum(a, 0.0, out=a)
            a /= a_sum

        it += 1

    # ------------------------------------------------------------------
    # Final reporting (the framework will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
