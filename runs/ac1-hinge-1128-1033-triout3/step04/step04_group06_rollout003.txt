Project onto simplex:

def simplex_projection(v):

    # ensures v >= 0 and sum = 1

    u = np.sort(v)[::-1]  # sort descending

    cssv = np.cumsum(u)

    rho = np.where(u > (cssv - 1) / np.arange(1, len(v)+1))[0][-1]

    theta = (cssv[rho] - 1) / (rho + 1)

    w = np.maximum(v - theta, 0)

    return w

"""
Propose a non‑negative step‑function minimizing
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm mixes:
* a soft‑max (KL‑mirror) descent on the simplex (the baseline idea);
* a hard‑max sub‑gradient phase;
* occasional random “jitter”/hill‑climbing moves;
* simplex projection (exact, O(n log n));
* adaptive schedules for the temperature τ and the (multiplicative) learning
  rate η, plus momentum.

The search runs for the whole allotted wall‑clock budget (default 1000 s)
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negatives due to round‑off
    np.maximum(conv, 0, out=conv)
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) =\
        """
        Gradient of  Σ_k w_k (a*a)[k] with respect to a.
        The formula is  g = 2 * (rev(w) * a) .
        """
        n = a.shape[0]
        # reverse w (full length = 2n‑1)
        u = w[::-1]
        conv = _full_convolution_fft(u, a)          # length 3n‑2
        start = 2 * n - 2
        idx = start - np.arange(n)
        return 2.0 * conv[idx]


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σ a_i = 1 , a_i ≥ 0)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex.
    Implements the algorithm of Duchi et al. (2008) – O(n log n).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()

    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the rho index
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to guard against rounding errors
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n)
    return w
