def _simulated_annealing(a, cur_val, deadline, n):
    cur_a = a.copy()
    cur_obj = cur_val # assume cur_val is exact objective from evaluate_sequence? but we can compute fast_obj.
    best_a = cur_a.copy()
    best_obj = cur_obj
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    start = time.time()
    while time.time() < deadline:
        # compute temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / total_time
        T = max(1e-6, 0.01 * (1 - frac))  # linear cooling
        i, j = rng.integers(0, n, size=2)
        if i==j: continue
        delta = cur_a[i] * rng.random() * 0.1  # up to 10% of mass at i
        if delta == 0: continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # ensure non-negative
        if new_a[i] < 0: continue
        new_obj = fast_objective(new_a)
        if new_obj < cur_obj or rng.random() < np.exp(-(new_obj - cur_obj) / T):
            cur_a = new_a
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise a probability vector ``a`` of length ``n`` on the simplex
   (either by interpolating the previous best global ``height_sequence_1``
   or by using the uniform distribution).
2. Optimise a smooth surrogate of the max‑convolution
   ``L_ε(a) = 2·n·ε·log∑_k exp((a*a)_k/ε)``.
   The surrogate is minimised by a soft‑max re‑parameterisation
   ``a_i = exp(w_i)/∑exp(w)`` and L‑BFGS‑B.
   The smoothing parameter ``ε`` is decreased on a log‑scale
   (continuation method) – 25 stages from 1e‑2 down to 1e‑12.
3. After the continuation phase a simulated‑annealing style hill‑climber
   repeatedly transfers a random fraction of mass between two positions,
   always keeping the vector on the simplex.
   The fast FFT‑based objective
   ``2·n·max(a*a) / (∑a)^2`` is used for the inner loop; the exact
   ``evaluate_sequence`` call is performed only when a new best exact
   value is found.
4. The best exact sequence seen before the 1000 s budget expires is
   returned.

The whole procedure runs under the time budget, prints progress and
returns a list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate of the max‑convolution and its gradient
# ----------------------------------------------------------------------
def smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Return (L_ε, ∂L_ε/∂w) for a given soft‑max parameter vector ``w``,
    smoothing parameter ``eps`` and length ``n``.
    """
    # soft‑max -> feasible a (non‑negative, Σa = 1)
    w_shift = w - np.max(w)               # improve numeric stability
    a = np.exp(w_shift)
    a = a / a.sum()

    # a * a (full auto‑convolution)
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth approximation of the max (log‑sum‑exp)
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # = log∑ exp((conv_k‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form, see derivation in the statement)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # shape (2·n‑1,)

    # g = 4·n·(a * p) expressed via convolution with reversed p
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)         # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]       # middle n entries
    grad_a = 4.0 * n * seg[::-1]          # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)           # ∂L/∂w

    return L, grad_w


# ----------------------------------------------------------------------
# Fast evaluation of the exact objective (no clipping, sum‑invariant)
# ----------------------------------------------------------------------
def fast_objective(a: np.ndarray, n: int) -> float:
    """
    Compute 2·n·max(a*a) / (sum(a)^2) using FFT.
    This is exactly the value returned by ``evaluate_sequence`` for
    non‑negative vectors (apart from the cryptic clipping performed there).
    """
    conv = _full_convolution(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Simulated‑annealing style hill‑climbing refinement
# ----------------------------------------------------------------------
def simulated_annealing(a: np.ndarray, cur_best_val: float,
                        deadline: float, n: int, rng: np.random.Generator):
    """
    Random mass‑transfer moves while keeping Σa constant.
    Accepts worsening moves with a temperature that linearly cools.
    Returns the best (vector, Objective, exact value) found.
    """
    cur_a = a.copy()
    cur_obj = fast_objective(cur_a, n)
    best_a = cur_a.copy()
    best_obj = cur_obj
    best_val = cur_best_val

    start = time.time()
    total = max(deadline - start, 1e-9)

    T0 = 0.02          # initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = max(1e-12, T0 * (1.0 - frac))

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a random fraction (≤20%) of the mass from i to j
        delta = cur_a[i] * rng.random() * 0.20
        if delta == 0.0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:                     # safety
            continue

        new_obj = fast_objective(new_a, n)

        # Metropolis acceptance
        if new_obj < cur_obj or rng.random() < np.exp(-(new_obj - cur_obj) / T):
            cur_a = new_a
            cur_obj = new_obj

            if new_obj < best_obj:
                best_obj = new_obj
                best_a = new_a.copy()
                # check the exact (slow) evaluation only when we improve the surrogate
                try:
                    exact_val = evaluate_sequence(best_a.tolist())
                except Exception:
                    continue
                if exact_val < best_val:
                    best_val = exact_val
                    print(f"[SA] new global best = {best_val:.6f}")

    return best_a, best_obj, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a step‑function using a continuation L‑BFGS‑B phase
    followed by a simulated‑annealing refinement.
    Returns the best list of heights found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # problem size – a larger n gives a finer discretisation
    n = 5000

    # ------------------------------------------------------------------
    # Initialise a (interpolate previous best if available)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start

    a0 = np.clip(a0, 0.0, None)
    if a0.sum() < 0.01:
        a0 = np.full(n, 1.0 / n)
    a0 = a0 / a0.sum()
    w = np.log(a0 + 1e-12)        # soft‑max parameters

    # ------------------------------------------------------------------
    # Initial exact evaluation
    # ------------------------------------------------------------------
    best_a = a0.copy()
    try:
        best_val = evaluate_sequence(best_a.tolist())
    except Exception:
        best_val = fast_objective(best_a, n)   # fallback (should not happen)
    best_obj = fast_objective(best_a, n)
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Continuation schedule for ε (log‑space, finer near zero)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-2, -12, num=25)   # 1e‑2 … 1e‑12

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        # L‑BFGS‑B on the smoothed surrogate
        res = minimize(smooth_obj_and_grad,
                       w,
                       args=(eps, n),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # obtain a from w
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        cur_obj = fast_objective(a, n)
        # occasionally evaluate the exact objective
        try:
            cur_val = evaluate_sequence(a.tolist())
        except Exception:
            cur_val = cur_obj

        # keep the best exact value seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            best_obj = cur_obj
            print(f"[eps {eps:.1e}] new best exact = {best_val:.6f}")

        # also retain the best surrogate if it improves (helpful for SA)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # progress report
        print(f"[eps {eps:.1e}] cur exact = {cur_val:.6f}, cur surrogate = {cur_obj:.9f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing refinement (random mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        rng = np.random.default_rng(seed + 9999)
        best_a, _, best_val = simulated_annealing(best_a, best_val,
                                                  deadline, n, rng)

    # ------------------------------------------------------------------
    # Final exact evaluation (for safety)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation = {final_val:.6f}")
        best_val = final_val
    except Exception as exc:    # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # The framework expects a plain list of floats
    return best_a.tolist()
