def _targeted_hillclimb(a_start, n, deadline, best_a, best_val):
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac = 0.5
    IMPROVE_LIMIT = 50000
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        # find max index
        k_max = np.argmax(b)
        # compute shifted version of a for gradient
        # s_i = a[k_max - i] if valid else 0
        # compute contribution c_i = a[i] * s_i
        s = np.zeros(n)
        # compute s efficiently by reversed indexing
        # we can compute it as: s = np.zeros(n); for i in range(n): if 0 <= k_max - i < n: s[i] = a[k_max - i]
        # Use slicing:
        #   start = max(0, k_max - (n-1))
        #   end = min(k_max+1, n)

        c = a * s
        # choose source i = argmax(c)
        i = int(np.argmax(c))
        # choose target j = argmin(s) where j != i
        # we want low s values (contribute less to max)

        # but ensure a[j] is not zero after moving.
        j_candidates = np.where(np.arange(n) != i)[0]
        # we can pick j = argmin(s[j])
        j = int(j_candidates[np.argmin(s[j_candidates])])

        if a[i] <= 1e-14:
            continue

        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0:
            continue

        # update b incrementally as before
        b_new = b.copy()
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i+j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12: # improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration}] new best = {best_val:.6f}")

            print(f"[Hill {iteration}] perturb after stagnation")
        iteration += 1

"""
Propose a step‑function (non‑negative heights) that minimises the
`evaluate_sequence` objective.

Algorithm
---------
1. **Random parametric search** – sample a few thousand densities obtained
   from Beta‑distributions (including mixtures with the uniform density).
   Each sampled shape is normalised to sum 1 and its exact objective is
   evaluated by an FFT‑based convolution.  The best shape becomes the
   starting point for the subsequent optimisation.

2. **Smooth‑max Adam optimisation** – the vector `a` is represented via a
   soft‑max (`a = softmax(w)`) guaranteeing non‑negativity and Σa = 1.
   A smooth approximation of `max(conv(a,a))` (log‑sum‑exp) is minimised.
   The smoothing parameter ε and the learning‑rate are decreased
   geometrically, an Adam optimiser is used and occasional random restarts
   avoid stagnation.

3. **Targeted hill‑climbing** – from the best point found by Adam,
   we perform an exact local search on the true objective.
   In each iteration the shift `k` attaining the current maximum
   convolution value is identified; the gradient of that shift tells us
   which coordinates contribute most to the peak.
   A small amount of mass is transferred from a high‑contributing index
   to a low‑contributing one, and the convolution is updated in O(n).
   Improvements are accepted greedily; after long stagnation a tiny
   random perturbation re‑explores the neighbourhood.

All steps respect the 1000 s time budget.  The final sequence is returned
as a plain list of floats (sum ≈ 1, all entries ≥ 0).
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft


# ----------------------------------------------------------------------
# Convolution utilities (full convolution of two real vectors)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = rfft(x, size)
    conv = irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = rfft(x, size)
    fy = rfft(y, size)
    conv = irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (exact, no security checks – used internally)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)).  Assumes Σa≈1."""
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Parameteric families for the random initial search
# ----------------------------------------------------------------------
def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Return a normalised density on the discrete grid obtained from the
    Beta(α,β) distribution (mid‑point rule).  The shape is evaluated in
    log‑space to avoid overflow for extreme parameters.
    """
    x = (np.arange(n) + 0.5) / n               # points in (0,1)
    # log‑pdf of Beta(α,β)
    log_pdf = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # shift to keep numbers in a reasonable range
    log_pdf -= np.max(log_pdf)
    pdf = np.exp(log_pdf)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0:
        # fall back to uniform if something went wrong
        return np.full(n, 1.0 / n)
    return pdf / pdf_sum


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution objective.
    Returns the best (a, value) seen before *deadline*.
    """
    # soft‑max parameter w (log‑its)
    w = np.log(a0 + 1e-16)

    # Adam hyper‑parameters (geometric schedules)
    beta1 = 0.90
    beta2 = 0.999
    lr_start = 5e-2
    lr_end = 1e-3
    eps_start = 5e-3
    eps_end = 1e-10

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 25000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # --------- obtain feasible a from w (soft‑max) ----------
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(w, 1.0 / n)
        else:
            a /= a_sum

        # --------- exact objective for tracking ----------
        conv_aa = _full_convolution(a)
        max_conv = conv_aa.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # --------- schedule of smoothing ε and learning‑rate ----------
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress   # geometric decay
        lr = lr_start * (lr_end / lr_start) ** progress      # geometric decay

        # --------- smooth‑max (log‑sum‑exp) weights ----------
        shifted = (conv_aa - max_conv) / eps
        # shift is ≤ 0, safe exponential
        weights = np.exp(shifted)
        p = weights / weights.sum()                # analogue of soft‑max over shifts

        # --------- gradient of smoothed objective w.r.t. a ----------
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)              # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]         # middle n entries
        grad_a = 4.0 * n * segment[::-1]           # ∂/∂a of 2·n·softmax

        # --------- chain rule through soft‑max (gradient w.r.t. w) ----------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # --------- Adam update ----------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation (helps escape plateaus)
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stuck for a long time
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] restarting after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _targeted_hillclimb(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> np.ndarray:
    """
    Fast local search that transfers a small amount of mass from a
    “high‑contributing” index to a “low‑contributing” one, with exact
    O(n) updates of the convolution.
    """
    a = a_start.copy()
    b = _full_convolution(a)                     # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac = 0.5                               # max fraction of source mass to move
    IMPROVE_LIMIT = 100_000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- identify the shift attaining the current maximum ----
        k_max = int(np.argmax(b))

        # ---- gradient for that shift: g_i = 2*a_{k_max-i} (zero outside) ----
        # we construct the vector s_i = a_{k_max-i}
        s = np.zeros(n)
        if k_max < n:
            # overlapping region is 0..k_max
            s[:k_max + 1] = a[k_max::-1]
        else:
            # overlapping region is k_max-(n-1) .. n-1
            offset = k_max - (n - 1)
            s[offset:] = a[n - 1:offset - 1:-1]   # reversed tail

        # contribution of each coordinate to the current peak
        contrib = a * s

        # source: index with largest contribution (most responsible for the peak)
        i = int(np.argmax(contrib))
        if a[i] <= 1e-14:
            # nothing to move – pick another iteration
            continue

        # target: index where s is smallest (least influence on the peak)
        # avoid choosing the source again
        mask = np.arange(n) != i
        j = int(np.argmin(np.where(mask, s, np.inf)))

        # amount of mass to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (exact) ----
        b_new = b.copy()
        # cross‑terms (2·δ·a) added / removed by shifting a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ---- accept only improving moves ----
        if new_val < cur_val - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")

        else:
            no_improve += 1

        # ---- occasional perturbation after long stagnation ----
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturbing")

        iteration += 1

    return best_a
