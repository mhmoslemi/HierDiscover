def propose_candidate(seed=42,budget_s=1000):
   np.random.seed(seed)
   start = time.time()
   deadline = start + budget_s - 0.5

   # set n based on time? We'll try n = 8000 (maybe 12000 if time permits)
   n = 8000

   # continuity schedule of epsilon: choose maybe 0.1 to 1e-8.
   eps_schedule = np.logspace(np.log10(1e-2), np.log10(1e-8), num=20)

   best_val = np.inf
   best_a = None

   # We will attempt up to max_restarts = 5 or until time.

   for restart in range(5):
       if time.time() >= deadline: break
       # init a via Dirichlet random:
       rng = np.random.default_rng(seed + restart)
       a0 = rng.exponential(scale=1.0, size=n)
       a0 = a0 / a0.sum()
       w = np.log(a0 + 1e-12)  # initialize softmax param.

       current_best_val = evaluate_sequence(a0.tolist())
       if current_best_val < best_val:
           best_val = current_best_val
           best_a = a0.copy()
           print(f"[restart {restart}] init eval = {current_best_val:.6f}")

       # iterate over eps schedule
       for eps in eps_schedule:
           if time.time() >= deadline: break
           # minimize smoothed objective
           def fun(w_vec):
               L, grad = _smooth_obj_and_grad(w_vec, eps, n)
               return L, grad

           res = minimize(fun, w, method='L-BFGS-B', jac=True,
                          options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False})

           w = res.x
           # compute current exact eval
           a = np.exp(w - np.max(w))
           a = a / a.sum()
           cur_val = evaluate_sequence(a.tolist())
           if cur_val < best_val:
               best_val = cur_val
               best_a = a.copy()
               print(f"[restart {restart} eps {eps:.1e}] new best = {best_val:.6f}")

       # hill climbing refinement
       if time.time() < deadline:
           refined_a, refined_val = _refine_hill_climb(best_a, best_val, deadline, n)
           if refined_val < best_val:
               best_val = refined_val
               best_a = refined_a.copy()
               print(f"[restart {restart} refine] new best = {best_val:.6f}")

   # final output
   print(f"[final] best evaluation = {best_val:.6f}")
   return best_a.tolist()

"""
Propose a step‑function that (hopefully) beats the previous 1.5138 bound.

Algorithm
---------
* Work with a soft‑max parametrisation a_i = exp(w_i)/∑exp(w) → a_i ≥ 0,
  Σa_i = 1 (the evaluation is scale‑invariant, so the exact sum does not
  matter as long as it exceeds the 0.01 safety limit).
* Minimise a smooth surrogate of the max‑convolution:
        Φ_ε(a) = 2·n·[ max(conv(a)) + ε·log∑_k exp((conv_k‑max(conv))/ε) ] .
  Its gradient w.r.t. the soft‑max parameters w can be written with
  two FFT‑based convolutions.
* A continuation schedule drives ε from 1e‑2 down to 1e‑8, solving a
  sequence of L‑BFGS‑B problems (≈ 200 iterations each).  Several random
  restarts are performed; the best exact value seen (via the official
  ``evaluate_sequence``) is kept.
* After the continuation phase a cheap hill‑climbing refinement moves a
  small random fraction of mass between two coordinates while keeping the
  total sum constant.
* The whole routine respects the 1000 s budget and returns the best
  sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective and gradient (w.r.t. soft‑max parameters)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smoothed surrogate Φ_ε and its gradient ∂Φ/∂w.

    Parameters
    ----------
    w : np.ndarray
        Soft‑max parameters (unconstrained).
    eps : float
        Smoothing parameter (ε > 0).
    n : int
        Length of the underlying step function.

    Returns
    -------
    L : float
        Value of Φ_ε.
    grad_w : np.ndarray
        Gradient of Φ_ε w.r.t. w (same shape as w).
    """
    # --- soft‑max (positive, sum = 1) ---
    w_shift = w - np.max(w)          # for numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                    # now a_i ≥ 0, Σa_i = 1

    # --- full self‑convolution ---
    conv = _full_convolution(a)      # length 2·n‑1
    max_conv = conv.max()

    # --- smoothed max (log‑sum‑exp) ---
    # Φ_ε = 2·n·[ max_conv + eps·log Σ exp((conv‑max_conv)/eps) ]
    shifted = (conv - max_conv) / eps
    # log‑sum‑exp in a numerically stable way
    log_sum = logsumexp(shifted)     # = log Σ exp((conv‑max)/eps)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # --- gradient w.r.t. a (closed form) ---
    # p_k = exp((conv_k‑max)/eps) / Σ_j exp((conv_j‑max)/eps)
    weights = np.exp(shifted)
    p = weights / np.sum(weights)    # shape (2·n‑1,)

    # For each i, dΦ/da_i = 4·n·Σ_k p_k·a_{k‑i}
    # This is a convolution of a with the reversed p vector.
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)    # length 3·n‑2
    # Extract the n entries that correspond to offsets i = 0 … n‑1
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]      # shape (n,)

    # --- chain rule (soft‑max) ---
    dot = np.dot(grad_a, a)          # Σ_j grad_a_j·a_j
    grad_w = a * (grad_a - dot)      # ∂Φ/∂w_i

    return float(L), grad_w.astype(np.float64)


# ----------------------------------------------------------------------
# Hill‑climbing refinement (random mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float, n: int) -> tuple[np.ndarray, float]:
    """
    Perform cheap random mass‑transfer moves that keep Σa constant.
    Accept a move only if it strictly improves the exact evaluation.
    """
    cur_a = a.copy()
    cur_best = cur_val
    rng = np.random.default_rng()
    # “temperature” – maximal fraction of the source weight that may be moved
    max_frac = 0.15

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        src = cur_a[i]
        if src <= 0.0:
            continue
        # random fraction in (0, max_frac]
        delta = src * max_frac * rng.random()
        if delta <= 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # evaluate with the official (exact) function
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_best:
            cur_a = new_a
            cur_best = val
            print(f"[refine] improved to {cur_best:.6f}")
    return cur_a, cur_best


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises the step‑function heights via a multi‑restart continuation
    L‑BFGS‑B scheme on a smoothed log‑sum‑exp surrogate,
    followed by a random hill‑climbing refinement.
    Returns the best list of floats found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – larger n usually gives a lower theoretical bound.
    # ------------------------------------------------------------------
    n = 8000       # feel free to adjust (fits easily in the time budget)

    # ------------------------------------------------------------------
    # Global best bookkeeping
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Continuation schedule for ε (log‑space, finer than before)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-2), np.log10(1e-8), num=25)

    # ------------------------------------------------------------------
    # Number of random restarts (limited by remaining time)
    # ------------------------------------------------------------------
    max_restarts = 8
    restart = 0
    while restart < max_restarts and time.time() < deadline:
        # ----- random start (Dirichlet via exponential) -----
        rng = np.random.default_rng(seed + restart + 12345)
        a0 = rng.exponential(scale=1.0, size=n)
        a0 /= a0.sum()                    # Σa = 1  (scale‑invariant)
        w = np.log(a0 + 1e-14)            # soft‑max parameters

        # evaluate the raw start – useful for diagnostics
        try:
            val0 = evaluate_sequence(a0.tolist())
        except Exception:
            val0 = np.inf
        if val0 < best_val:
            best_val = val0
            best_a = a0.copy()
            print(f"[restart {restart}] initial eval = {best_val:.6f}")

        # ----- continuation L‑BFGS‑B over the ε‑schedule -----
        for eps in eps_schedule:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            # L‑BFGS‑B (no bounds needed – soft‑max guarantees feasibility)
            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 200,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})

            w = res.x

            # Exact evaluation of the current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            try:
                cur_val = evaluate_sequence(a.tolist())
            except Exception:
                cur_val = np.inf

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[restart {restart} eps {eps:.1e}] new best = {best_val:.6f}")

        # ----- final hill‑climbing refinement for this restart -----
        if time.time() < deadline and best_a is not None:
            best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

        restart += 1

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
