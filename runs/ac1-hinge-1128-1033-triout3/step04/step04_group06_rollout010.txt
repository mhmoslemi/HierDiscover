"""
Propose a (non‑negative, possibly sparse) step‑function that minimises

    2·n·max(a*a) / (sum(a)**2)

using a fast FFT based convolution, a soft‑max approximation of the
∞‑norm and an Adam‑style additive update on the probability simplex.
The routine performs many short restarts (different initial vectors,
different lengths) inside the allotted time budget and keeps the best
sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, works for any lengths)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two 1‑D vectors using a power‑of‑2 FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()            # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Remove harmless negative round‑off artefacts
    np.maximum(conv, 0, out=conv)
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of ∑_k w_k·(a*a)[k] with respect to a.
    It is computed as 2·(reverse(w) * a) and then sliced to length n.
    """
    n = a.shape[0]
    rev_w = w[::-1]                     # length = 2n‑1
    conv = _full_convolution_fft(rev_w, a)   # length = 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)                 # descending indices
    return 2.0 * conv[idx]

# Core optimiser – Adam on the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _optimize_sequence(a_init: np.ndarray,
                       n: int,
                       budget_s: float,
                       global_deadline: float,
                       current_best: float = float('inf')) -> tuple[np.ndarray, float]:
    """
    Run a single optimisation run for a fixed length ``n``.
    Returns (best_sequence, best_value) obtained within ``budget_s`` seconds.
    """
    a = a_init.copy()
    # normalise to the simplex
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)

    # bookkeeping

    conv_a = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv_a.max())

        current_best = best_val

    # Adam‑style accumulator

    v = np.zeros_like(a)
    eps = 1e-12

    # schedules (linear annealing)
    eta_start = 0.6
    eta_end = 1e-5
    tau_factor_start = 0.5          # τ ≈ factor·max at the beginning
    tau_factor_end = 1e-6           # essentially hard max at the end

    beta1 = 0.9
    beta2 = 0.999

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin
    iteration = 0
    no_imp = 0                                      # counter for stagnation

    while True:

        if now >= deadline or now >= global_deadline - 0.5:
            break

        # current convolution and objective
        conv_a = _full_convolution_fft(a, a)
        max_b = conv_a.max()

        # keep best ever seen (global best for this run)
        if cur_val < best_val - 1e-12:               # a tiny tolerance

        else:

        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # learning‑rate schedule

    return best_a, final_val
