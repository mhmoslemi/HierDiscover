def conv_self(p, m_fft):
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2*len(p)-1]
    return conv

def objective(p, n, m_fft):
    conv = conv_self(p, m_fft)
    return 2.0 * n * np.max(conv)

def gradient_smoothed(p, m_fft, alpha):
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2*len(p)-1]
    # compute softmax weights
    z = alpha * conv
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    w = exp_z / np.sum(exp_z)
    # compute gradient using helper
    grad = _gradient_from_weights(p, p_fft, w, m_fft)
    return grad

"""
Propose a step function (non‑negative sequence) that minimizes

    2 · n · max(conv_self) / (sum(a)²)

The algorithm is a multi‑stage search:

1️⃣  **Generate many diverse seeds** – arcsine‑like shapes, geometric,
    Beta‑like families, mixtures with the uniform vector and a bunch of
    random Dirichlet draws (all projected onto the probability simplex).

2️⃣  **Quick evaluation** of every seed to keep the few best (default 3).

3️⃣  **Refinement of the best seeds**  
    •  Adam optimisation on a log‑sum‑exp (smooth) approximation of the
       max‑convolution. The smoothing factor *α* is increased gradually so
       that the smooth objective converges to the true one.  
    •  Exact sub‑gradient descent on the true max‑entry (weights are 1 on the
       active indices, 0 otherwise).  Both phases use a Euclidean projection
       onto the simplex after every update.

4️⃣  **Local hill‑climbing** – tiny transfers of mass between two random
    positions are accepted if they improve the exact objective.

5️⃣  **Global simulated annealing** started from the best vector found so far;
    moves that worsen the objective are occasionally accepted according to a
    temperature schedule, allowing the search to escape shallow local minima.

All heavy lifting (convolution, smoothed‑max gradient) is done with FFTs.
The routine respects the 1000 s time budget; if the budget expires the best
vector found is returned.  The returned list is already clipped to the range

import time

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full linear self‑convolution of p using FFT of size m_fft."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _gradient_from_weights(p: np.ndarray, p_fft: np.ndarray,
                          w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Gradient of  Σ_i w_i·(p∗p)_i   (where * is linear convolution).
    w must be a probability vector (Σ w_i = 1, non‑negative, length = 2·n‑1).
    """
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * p.size - 2]
    idx = 2 * p.size - 2 - np.arange(p.size)
    grad = 2.0 * corr[idx]
    return grad


# ----------------------------------------------------------------------
# Exact objective (sum is 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(p: np.ndarray, n: int, m_fft: int) -> float:
    conv = _conv_self(p, m_fft)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max entry
# ----------------------------------------------------------------------
def _subgradient_descent(p0: np.ndarray, n: int, m_fft: int,
                         deadline: float,
                         max_iters: int = 30_000,
                         step_start: float = 0.2,
                         decay: float = 0.9995,
                         min_step: float = 1e-9) -> np.ndarray:
    p = p0.copy()
    step = step_start
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # FFT of current p
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
        max_val = conv.max()
        # active set (indices attaining the maximal value up to a tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        # build weighting vector (uniform on the active set)
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / float(active.size)
        # gradient of weighted sum (which equals a sub‑gradient of the max)
        grad = _gradient_from_weights(p, p_fft, w, m_fft)
        # gradient step + projection
        p = p - step * grad
        p = _simplex_projection(p)
        step = max(step * decay, min_step)
    return p


# ----------------------------------------------------------------------
# Adam optimisation on a log‑sum‑exp smooth approximation of max
# ----------------------------------------------------------------------
def _adam_smooth(p0: np.ndarray, n: int, m_fft: int,
                 deadline: float,
                 max_iters: int = 150_000,
                 base_lr: float = 0.04,
                 alpha_start: float = 0.5,
                 alpha_mul: float = 1.005,
                 alpha_max: float = 120.0) -> np.ndarray:
    p = p0.copy()
    m1 = np.zeros_like(p)
    m2 = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = alpha_start
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

        # smooth max via soft‑max (log‑sum‑exp) weights
        z = alpha * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # subtract max for numerical stability
        w = exp_z / np.sum(exp_z)

        grad = _gradient_from_weights(p, p_fft, w, m_fft)

        # Adam moment updates
        m1 = beta1 * m1 + (1.0 - beta1) * grad
        m2 = beta2 * m2 + (1.0 - beta2) * (grad * grad)

        m1_hat = m1 / (1.0 - beta1 ** (it + 1))
        m2_hat = m2 / (1.0 - beta2 ** (it + 1))

        lr = base_lr / np.sqrt(alpha + 1e-12)   # smaller steps as smoothing grows
        step_vec = lr * m1_hat / (np.sqrt(m2_hat) + eps)

        p = p - step_vec
        p = _simplex_projection(p)

        # increase smoothing
        alpha = min(alpha * alpha_mul, alpha_max)

        # occasional exact check (optional – cheap)
        if (it & 1023) == 0:
            # we do not update a global best here; the caller does that
            pass

    return p


# ----------------------------------------------------------------------
# Small random‑swap hill‑climber (only accepts improvements)
# ----------------------------------------------------------------------
def _hillclimb(p0: np.ndarray, n: int, m_fft: int,
               deadline: float,
               rng: np.random.Generator,
               steps: int = 30_000,
               eps_move: float = 0.001) -> np.ndarray:
    p = p0.copy()
    cur_score = _objective(p, n, m_fft)
    best_local = cur_score
    best_vec = p.copy()
    for it in range(steps):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount that can be moved from i to j
        max_move = min(p[i], eps_move)
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move               # 0 ≤ delta ≤ max_move
        new_p = p.copy()
        new_p[i] -= delta
        new_p[j] += delta
        new_score = _objective(new_p, n, m_fft)
        if new_score < cur_score - 1e-15:
            p = new_p
            cur_score = new_score
            if new_score < best_local:
                best_local = new_score
                best_vec = new_p.copy()
    return best_vec


# ----------------------------------------------------------------------
# Global simulated annealing starting from the best vector so far
# ----------------------------------------------------------------------
def _simulated_annealing(p_start: np.ndarray, n: int, m_fft: int,
                         deadline: float,
                         rng: np.random.Generator,
                         steps: int = 70_000,
                         eps_move: float = 0.003) -> np.ndarray:
    p = p_start.copy()
    cur_score = _objective(p, n, m_fft)
    best_score = cur_score
    best_vec = p.copy()

    T0 = 0.015
    T_end = 1e-5
    for it in range(steps):
        if time.time() >= deadline:
            break
        T = T0 * (1.0 - it / steps) + T_end * (it / steps)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_move = min(p[i], eps_move)
        if max_move <= 0.0:
            continue
        delta = (rng.random() - 0.5) * 2.0 * max_move   # can move either direction
        new_p = p.copy()
        new_p[i] -= delta
        new_p[j] += delta
        # keep feasibility (non‑negative)
        if new_p[i] < 0.0 or new_p[j] < 0.0:
            continue
        new_score = _objective(new_p, n, m_fft)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            p = new_p
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = new_p.copy()
                print(f"Anneal iter {it:5d} → new best {best_score:.6f}")

    return best_vec


# ----------------------------------------------------------------------
# Core search routine required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size (length of the step function)
    # ------------------------------------------------------------------
    try:
        base_seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base_seq = None

    n = len(base_seq) if base_seq is not None else 1153

    # FFT size – next power of two that can hold any needed convolution
    m_fft = 1 << ((2 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Build a pool of diverse initial candidates (all are probability vectors)
    # ------------------------------------------------------------------
    init_vectors = []

    i = np.arange(n, dtype=float)
    eps = 1e-12
    base = (i + 0.5) * (n - 0.5 - i) + eps  # positive values for the whole range

    # 1) Arcsine‑type family  a_i ∝ base**exp   (exp negative → edge heavy)
    for exp in np.linspace(-1.5, 0.4, 19):      # includes the classic -0.5
        p = np.power(base, exp, where=base > 0)
        p = np.maximum(p, 0.0)
        s = p.sum()
        if s > 0:
            init_vectors.append(p / s)

    # 2) Geometric decay a_i ∝ r**i
    for r in np.linspace(0.95, 0.9998, 13):
        geom = r ** i
        init_vectors.append(geom / geom.sum())

    # 3) Beta‑like shapes a_i ∝ base**a  (a>0 puts mass in the centre)
    for a in np.linspace(0.2, 2.0, 10):
        p = np.power(base, a, where=base > 0)
        init_vectors.append(p / p.sum())

    # 4) Mixture of arcsine and uniform
    p_uniform = np.full(n, 1.0 / n)
    p_arcsine = np.power(base, -0.5, where=base > 0)
    p_arcsine = p_arcsine / p_arcsine.sum()
    for lam in np.linspace(0.0, 1.0, 11):
        mix = lam * p_arcsine + (1.0 - lam) * p_uniform
        init_vectors.append(mix / mix.sum())

    # 5) Random Dirichlet draws
    for _ in range(12):
        rnd = rng.random(n)
        init_vectors.append(rnd / rnd.sum())

    # 6) Historic sequence, if present
    if base_seq is not None:
        seq = np.clip(base_seq, 0.0, 1000.0)
        if seq.sum() > 0:
            init_vectors.append(seq / seq.sum())

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds, keep the best few
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_vec = None
    scores_and_vecs = []

    deadline = time.time() + budget_s - 1.0   # keep a small safety margin

    for p in init_vectors:
        if time.time() >= deadline:
            break
        sc = _objective(p, n, m_fft)
        scores_and_vecs.append((sc, p.copy()))
        if sc < best_score:
            best_score = sc
            best_vec = p.copy()
            print(f"Initial candidate score: {best_score:.6f}")

    # keep Top‑K seeds for heavy refinement
    scores_and_vecs.sort(key=lambda x: x[0])
    top_k = 3
    top_seeds = [vec for _, vec in scores_and_vecs[:top_k]]

    # ------------------------------------------------------------------
    # Refine each top seed (Adam → exact sub‑gradient → hill‑climber)
    # ------------------------------------------------------------------
    for seed_id, p0 in enumerate(top_seeds, 1):
        if time.time() >= deadline:
            break
        print(f"\n--- Refinement #{seed_id}/{len(top_seeds)} ---")
        # Adam on smooth max
        p_adam = _adam_smooth(p0, n, m_fft, deadline)

        # Exact sub‑gradient polishing
        p_sub = _subgradient_descent(p_adam, n, m_fft, deadline)

        # Tiny random‑swap hill‑climber
        p_hc = _hillclimb(p_sub, n, m_fft, deadline, rng)

        sc = _objective(p_hc, n, m_fft)
        if sc < best_score:
            best_score = sc
            best_vec = p_hc.copy()
            print(f"New best after refinement #{seed_id}: {best_score:.6f}")

        # if we already hit the target we can stop early
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Global simulated annealing from the best vector found so far
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing phase ---")
        p_anneal = _simulated_annealing(best_vec, n, m_fft, deadline, rng)
        # a short final hill‑climb to polish any residual improvement
        p_final = _hillclimb(p_anneal, n, m_fft, deadline, rng, steps=20_000)
        final_score = _objective(p_final, n, m_fft)
        if final_score < best_score:
            best_score = final_score
            best_vec = p_final.copy()
            print(f"Best after annealing: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to the bounds used by evaluate_sequence)
    # ------------------------------------------------------------------
    print(f"\nFinal best score: {best_score:.6f}")
    return np.clip(best_vec, 0.0, 1000.0).tolist()
