def project_simplex(v):
    # Compute projection onto simplex
    # [source: Duchi et al. 2008]
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step‑function (list of heights) that minimises the
`evaluate_sequence` objective.

Algorithm
---------

* Phase A – smoothed‑max Adam optimisation (≈ 50 % of the time).  
  The vector `a` is parametrised by a soft‑max `w` so that Σa=1 and a≥0.
  A log‑sum‑exp surrogate of the max‑convolution is minimised with Adam.
  The smoothing ε is annealed from 5·10⁻³ down to 1·10⁻¹².

* Phase B1 – projected sub‑gradient descent on the exact max‑convolution
  (≈ 30 % of the time).  
  For the current peak index k* we use the exact sub‑gradient  
  g_i = 2·a_{k*‑i} (zero outside the support) and step a ← a−η·g,
  projecting back onto the probability simplex after each step.
  The step‑size η is adaptively reduced on stalls and a small random
  perturbation is applied after long stagnation.

* Phase B2 – deterministic hill‑climbing (≈ 20 % of the time).  
  Using the O(n) incremental update of the convolution, we repeatedly
  transfer a small amount of mass from a source index i (chosen where the
  gradient is largest) to a target index j (where the gradient is smallest)
  and accept only moves that strictly improve the exact objective.
  After many idle steps a random perturbation of the current best solution
  is performed to escape local minima.

The three phases share the best solution seen so far.  At the end the
candidate is evaluated with `evaluate_sequence` and returned as a list of
floats (Σ≈1, all entries ≥0).  The implementation fits comfortably into the
1000 s budget on a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase A – smoothed max Adam optimisation
# ----------------------------------------------------------------------


def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on the log‑sum‑exp surrogate of max(conv(a,a))."""
    # soft‑max parameterisation (Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.05
    lr_end = 0.001
    eps_start = 5e-3
    eps_end = 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 20000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---------- a from soft‑max ----------
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ---------- current exact objective ----------
        conv_aa = _full_convolution(a)
        max_conv = conv_aa.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve_cnt += 1

        # ---------- smoothly‑approximated max ----------
        progress = (now - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        shifted = (conv_aa - max_conv) / eps
        # avoid overflow / underflow
        weights = np.exp(shifted - np.max(shifted))
        p = weights / weights.sum()  # soft‑max over convolution entries

        # gradient of smoothed objective w.r.t. a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]    # middle n entries
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the soft‑max (gradient w.r.t. w)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---------- Adam update ----------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random kick
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stuck
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] restarting after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B1 – projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------


def _grad_descent_phase(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on  max(conv(a,a))  with projection onto the simplex.
    """
    a = a_start.copy()
    cur_val = best_val
    iteration = 0
    eta = 0.2                     # initial step size
    no_improve = 0
    STALL_LIMIT = 20000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # full convolution and current max
        b = _full_convolution(a)
        K = int(np.argmax(b))
        max_b = b[K]
        cur_obj = 2.0 * n * max_b
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[GD {iteration:7d}] new best = {best_val:.6f}")

        # compute sub‑gradient of the peak entry
        g = np.zeros_like(a)
        i_start = max(0, K - (n - 1))
        i_end = min(K, n - 1)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[K - idx]

        # gradient step + simplex projection
        v = a - eta * g
        a_new = _project_simplex(v)

        # evaluate new objective
        b_new = _full_convolution(a_new)
        new_max_b = b_new.max()
        new_obj = 2.0 * n * new_max_b

        if new_obj < cur_obj - 1e-12:
            # accept
            a = a_new
            cur_val = new_obj
            no_improve = 0
            # modestly increase step‑size on success
            eta = min(eta * 1.02, 1.0)
        else:
            # reject – shrink step‑size
            eta *= 0.5
            no_improve += 1
            if eta < 1e-12:
                # re‑initialise step size with a small random nudge
                eta = 0.1 * np.random.rand()
                a = best_a.copy()
                noise = np.random.normal(scale=0.005, size=n)
                a = _project_simplex(a + noise)

        if no_improve > STALL_LIMIT:
            # perturb to escape a flat region
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a + noise)
            no_improve = 0
            print(f"[GD {iteration:7d}] stagnation – perturbing")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B2 – deterministic hill‑climbing on the exact objective
# ----------------------------------------------------------------------


def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> np.ndarray:
    """
    Very fast local search by moving a small amount of mass between coordinates.
    The move direction is guided by the gradient of the current maximal convolution entry.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    max_frac = 0.5
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current peak index and its gradient
        K = int(b.argmax())
        g = np.zeros_like(a)
        i_start = max(0, K - (n - 1))
        i_end = min(K, n - 1)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[K - idx]

        # source: index with large gradient * mass (to have something to move)
        # target: index with small gradient (to reduce the peak)
        if np.random.rand() < 0.9:
            # gradient‑guided choice
            valid_src = np.where(a > 1e-14)[0]
            if valid_src.size == 0:
                continue
            src_scores = g[valid_src] * a[valid_src]  # heuristic
            i = valid_src[np.argmax(src_scores)]
            # choose target with minimal gradient (break ties randomly)
            tgt_candidates = np.where(np.arange(n) != i)[0]
            j = tgt_candidates[np.argmin(g[tgt_candidates])]
        else:
            # fallback to previous heuristic
            i = int(np.argmax(a))
            j = np.random.randint(n)
            while j == i:
                j = np.random.randint(n)

        if a[i] <= 1e-14:
            continue

        # amount of mass to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_obj = 2.0 * n * b_new.max()

        if new_obj < cur_val:
            # accept move
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_obj
            no_improve = 0
            if new_obj < best_val:
                best_val = new_obj
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb the best solution slightly and restart local search
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturbing")

        iteration += 1

    return best_a
