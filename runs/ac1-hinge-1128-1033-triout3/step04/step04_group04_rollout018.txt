import time
import numpy as np
import math

def propose_candidate(seed=42, budget_s=1000):
    """
    Optimize a non‑negative step‑function (probability vector) to minimise
    2 * n * max_k (a * a)[k] using a smooth‑max surrogate and Adam updates.
    The vector is represented via a soft‑max over unconstrained parameters,
    allowing fast gradient evaluation through FFT‑based convolutions.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # ---- choose problem dimension -------------------------------------------------
    # Larger n gives a lower theoretical bound but increases per‑iteration cost.
    # 15000 is a good compromise for a 1000 s budget on a single CPU.
    n = 15000

    # ---- FFT size (next power of two >= 3n-2) ------------------------------------
    def next_pow2(x):
        return 1 << (x - 1).bit_length()
    m = next_pow2(3 * n - 2)

    # ---- initialise a with an edge‑heavy exponential distribution -----------------
    lam = 0.001  # decay rate (tuned empirically)
    idx = np.arange(n, dtype=np.float64)
    a0 = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    a0 += 1e-14
    a0 /= a0.sum()
    x = np.log(a0 + 1e-30)  # unconstrained parameters

    # ---- Adam hyper‑parameters ----------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05      # base learning rate
    m_adam = np.zeros_like(x)
    v_adam = np.zeros_like(x)

    # ---- smoothed‑max schedule ---------------------------------------------------
    beta_start = 0.5
    beta_end = 30.0
    beta = beta_start

    # ---- book‑keeping -------------------------------------------------------------
    best_score = np.inf
    best_vec = None
    last_improve = 0

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max transformation (ensures positivity and sum‑to‑one) ----------
        max_x = np.max(x)
        a = np.exp(x - max_x)
        a_sum = a.sum()
        a /= a_sum

        # ----- convolution a * a --------------------------------------------------
        Fa = np.fft.rfft(a, n=m)
        conv_full = np.fft.irfft(Fa * Fa, n=m)[: 2 * n - 1]

        # ----- current score ------------------------------------------------------
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = float(cur_score)
            best_vec = a.copy()
            last_improve = it
            print(f"Iter {it:6d}: new best score = {best_score:.6f}")

        # ----- smooth‑max weights --------------------------------------------------
        # Increase beta slowly (geometric schedule)
        if beta < beta_end:
            beta = min(beta_end, beta * 1.001)

        # stable softmax over conv
        conv_max = cur_max  # already computed
        w = np.exp(beta * (conv_full - conv_max))
        w_sum = w.sum()
        w /= w_sum

        # ----- sliding window of w (sum over n consecutive positions) ------------
        # cum[i] = sum_{k< i} w[k]
        cum = np.concatenate([[0.0], w.cumsum()])
        # windows_i = sum_{k=i}^{i+n-1} w[k]
        windows = cum[n:] - cum[:-n]               # shape (n,)

        # ----- correlation term grad_corr = Σ_j a_j w_{i+j} ----------------------
        a_rev = a[::-1]
        Fa_rev = np.fft.rfft(a_rev, n=m)
        Fw = np.fft.rfft(w, n=m)
        corr_full = np.fft.irfft(Fa_rev * Fw, n=m)[: 3 * n - 2]
        # slice indices i = 0..n-1 correspond to positions n-1 .. n-1+n-1
        grad_corr = corr_full[n - 1 : n - 1 + n]

        # ----- gradient w.r.t. a (smooth‑max surrogate) ---------------------------
        grad_a = grad_corr + a * windows          # shape (n,)

        # ----- transfer gradient to the unconstrained parameters x -----------------
        # derivative of softmax: ∂a_i/∂x_j = a_i (δ_{ij} - a_j)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update --------------------------------------------------------
        m_adam = beta1 * m_adam + (1 - beta1) * grad_x
        v_adam = beta2 * v_adam + (1 - beta2) * (grad_x * grad_x)
        m_hat = m_adam / (1 - beta1 ** it)
        v_hat = v_adam / (1 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional random re‑heat to escape plateaus -----------------------
        if it - last_improve > 5000:
            # small random perturbation to the parameters
            perturb = np.random.randn(n) * 1e-3
            x += perturb
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            last_improve = it
            print(f"Iter {it:6d}: re‑heat (no improvement for {it-last_improve} it)")

        # ----- progress report ----------------------------------------------------
        if it % 2000 == 0:
            now = time.time()
            elapsed = now - (deadline - budget_s + 5.0)
            print(f"Iter {it:6d} elapsed {elapsed:.1f}s best {best_score:.6f}")

    # ---------------------------------------------------------------------------
    # Return best sequence found (as plain Python list)
    # ---------------------------------------------------------------------------
    if best_vec is None:
        # fallback to a simple Dirichlet sample
        rnd = np.random.rand(n)
        best_vec = rnd / rnd.sum()
    print("\n=== Search finished ===")
    print(f"Best score = {best_score:.6f} after {it} iterations, length {len(best_vec)}")
    return best_vec.tolist()

"""
optimised_step_function.py

Search for a non‑negative step‑function (probability vector) that minimises

    2 * n * max_k (a * a)[k]

The algorithm works inside a 1000 s time budget and returns the best
vector found as a plain Python list.

Key ideas
---------

* Parametrise the vector with unconstrained real numbers `x` and use a
  soft‑max (`a = softmax(x)`) to enforce non‑negativity and the
  normalization ∑a=1.
* Optimise a *smooth‑max* surrogate of the objective:
      fβ(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
  which approaches the true max when β grows.
* Compute the gradient of fβ analytically; it only needs a
  correlation between `a` and the soft‑max weights `w`.
  All convolutions / correlations are performed with a real FFT,
  giving O(n log n) per iteration.
* Use Adam (adaptive moment estimation) for stable, fast descent.
* A simple “re‑heat” (random perturbation) is applied if the best
  score does not improve for a while, helping to escape plateaus.

The implementation respects the required API – only the
`propose_candidate` function is exported and no I/O or external
optimisation libraries are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: smallest power of two ≥ x
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two not less than x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Optimise a step function under the 1000 s budget.

    Returns
    -------
    list[float]
        The best probability vector found (its sum is ≈ 1, all entries ≥ 0).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for the final return

    # ------------------------------------------------------------------
    # 1. Choose a problem size.
    #    n≈15000 gives a good trade‑off between a low theoretical bound
    #    (~1.0) and the per‑iteration cost of the FFT‑based gradient.
    # ------------------------------------------------------------------
    n = 15000

    # FFT length sufficient for linear convolution of a (len=n) with w (len=2n‑1)
    m = next_pow2(3 * n - 2)   # we need up to len(a)+len(w)-1 = 3n‑2

    # ------------------------------------------------------------------
    # 2. Initialise a with an edge‑heavy exponential decay (a good starting
    #    point for the max‑convolution objective).
    # ------------------------------------------------------------------
    lam = 0.001                       # decay rate – tuned empirically
    idx = np.arange(n, dtype=np.float64)
    a0 = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    a0 += 1e-14                      # avoid exact zeros
    a0 /= a0.sum()
    # Unconstrained parameters; the soft‑max later guarantees a∈Δⁿ
    x = np.log(a0 + 1e-30)

    # ------------------------------------------------------------------
    # 3. Adam hyper‑parameters
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                         # base learning rate
    m_adam = np.zeros_like(x)
    v_adam = np.zeros_like(x)

    # ------------------------------------------------------------------
    # 4. Smooth‑max schedule (β grows to focus on the true maximum)
    # ------------------------------------------------------------------
    beta = 0.5                         # initial β
    beta_target = 30.0                 # final β

    # ------------------------------------------------------------------
    # 5. Book‑keeping
    # ------------------------------------------------------------------
    best_score = np.inf
    best_vec = None
    last_improve = 0
    it = 0

    # ------------------------------------------------------------------
    # 6. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- soft‑max: a_i = exp(x_i) / Σ_j exp(x_j)  ------------------
        max_x = np.max(x)
        a = np.exp(x - max_x)           # subtraction for numerical stability
        a /= a.sum()                    # now a is a probability vector

        # ----- convolution a * a (full linear convolution) ---------------
        Fa = np.fft.rfft(a, n=m)
        conv_full = np.fft.irfft(Fa * Fa, n=m)[: 2 * n - 1]

        # ----- current objective value (the one evaluate_sequence returns) ----
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = float(cur_score)
            best_vec = a.copy()
            last_improve = it
            print(f"Iter {it:6d}: new best = {best_score:.6f}")

        # ----- smooth‑max weights w_k ∝ exp(β·(conv_k - max_conv)) ----------
        #   Scale the exponent by the range of conv values so that β
        #   has a reasonable effect even when the values are tiny.
        conv_range = conv_full.max() - conv_full.min()
        if conv_range < 1e-12:
            conv_range = 1e-12
        w = np.exp(beta * (conv_full - cur_max) / conv_range)
        w /= w.sum()                   # now w is a probability distribution

        # ----- increase β slowly (geometric schedule) --------------------
        if beta < beta_target:
            beta = min(beta_target, beta * 1.001)

        # ----- sliding‑window sum of w (Σ_{k=i}^{i+n‑1} w_k) ---------------
        cum = np.concatenate([[0.0], w.cumsum()])   # length 2n
        windows = cum[n:] - cum[:-n]                 # shape (n,)

        # ----- correlation term Σ_j a_j w_{i+j} ---------------------------
        a_rev = a[::-1]
        Fa_rev = np.fft.rfft(a_rev, n=m)
        Fw = np.fft.rfft(w, n=m)
        corr_full = np.fft.irfft(Fa_rev * Fw, n=m)[: 3 * n - 2]
        # For i=0..n‑1 the required values sit at positions n‑1 … n‑1+n‑1
        grad_corr = corr_full[n - 1 : n - 1 + n]     # shape (n,)

        # ----- gradient of the smooth‑max surrogate w.r.t. a -------------
        grad_a = grad_corr + a * windows

        # ----- map ∂f/∂a to ∂f/∂x (soft‑max Jacobian) --------------------
        dot = np.dot(a, grad_a)                     # scalar = a·∂f/∂a
        grad_x = a * (grad_a - dot)                 # shape (n,)

        # ----- Adam update ------------------------------------------------
        m_adam = beta1 * m_adam + (1 - beta1) * grad_x
        v_adam = beta2 * v_adam + (1 - beta2) * (grad_x * grad_x)
        m_hat = m_adam / (1 - beta1 ** it)
        v_hat = v_adam / (1 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- re‑heat if we have been stagnant for a while ---------------
        if it - last_improve > 5000:
            # small random perturbation of the parameters
            x += np.random.randn(n) * 1e-3
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            last_improve = it
            print(f"Iter {it:6d}: re‑heat (no improvement for 5000 it)")

        # ----- light progress report ------------------------------------
        if it % 2000 == 0:
            elapsed = time.time() - start_time
            print(f"Iter {it:6d} – elapsed {elapsed:.1f}s – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Return the best vector (as a plain list) found within the budget
    # ------------------------------------------------------------------
    if best_vec is None:
        # fallback – random Dirichlet vector if, for some reason, nothing was saved
        rnd = np.random.rand(n)
        best_vec = rnd / rnd.sum()
    total_time = time.time() - start_time
    print("\n=== Search finished ===")
    print(f"Best score  = {best_score:.6f}")
    print(f"Iterations  = {it}")
    print(f"Time elapsed = {total_time:.1f}s")
    print(f"Vector length = {len(best_vec)}")
    return best_vec.tolist()
