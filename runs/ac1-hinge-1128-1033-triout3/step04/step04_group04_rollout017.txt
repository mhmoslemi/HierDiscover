"""
Propose a candidate height sequence that (hopefully) improves the
current best upper‑bound 1.514767.  The routine works in three phases:

1.  Deterministic sweep of a few parametric “U‑shaped’’ families on a
    range of lengths (up to 18 000).  This gives a good seed and also
    discovers whether a larger length can yield a smaller value.

2.  A modest Differential‑Evolution (DE) run on the probability simplex
    (population ≈ 35).  The DE is continued until the time budget is
    exhausted, with occasional random re‑initialisation of stagnating
    individuals.

3.  A final polishing phase based on exponentiated‑gradient descent
    on a smooth approximation of the max‑convolution (soft‑max).  The
    update is performed multiplicatively and the vector is projected
    onto the simplex after each step.

All heavy convolution work uses an FFT‑based implementation
(`full_convolution_fft`) which scales as O(n log n) and thus works
well even for n≈18000.  The algorithm respects the given time budget
and prints progress information.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex { w >= 0 , sum(w)=1 }."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution_fft(a: np.ndarray, b: np.ndarray = None) -> np.ndarray:
    """
    Full linear convolution using real FFT.
    If b is None, compute a * a.
    Returns an array of length len(a)+len(b)-1 (or 2*len(a)-1).
    """
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        n = a.shape[0]
        size = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)
        return conv[: 2 * n - 1]
    else:
        b = np.asarray(b, dtype=np.float64)
        n = a.shape[0]
        m = b.shape[0]
        size = _next_pow2(n + m - 1)
        fa = np.fft.rfft(a, n=size)
        fb = np.fft.rfft(b, n=size)
        conv = np.fft.irfft(fa * fb, n=size)
        return conv[: n + m - 1]


def score_vector(a: np.ndarray) -> float:
    """Score assuming sum(a) == 1: 2 * n * max(a*a)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)          # a * a
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (hand‑crafted “U‑shaped’’ distributions)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i,n‑1‑i)+1)^(-p)  (edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^(-p)  (edge‑heavy, product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i) + exp(-lam*(n‑1‑i))  (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# Differential Evolution on the simplex

def differential_evolution(base_vec: np.ndarray,
                           base_score: float,
                           deadline: float) -> Tuple[np.ndarray, float]:
    """
    Simple DE on the probability simplex.  Keeps a modest population (≈35)
    and runs until the time budget expires.  Occasionally re‑injects
    random individuals if progress stalls.
    """
    np.random.seed()                     # let randomness vary across runs
    n = base_vec.shape[0]
    pop_size = 35
    F = 0.8
    CR = 0.9

    # ------------------------------------------------------------------
    # Initialise population – a few curated seeds plus random Dirichlet draws
    # ------------------------------------------------------------------
    population = np.empty((pop_size, n), dtype=np.float64)
    scores = np.empty(pop_size, dtype=np.float64)

    # The best deterministic seed
    population[0] = base_vec
    scores[0] = base_score

    # a few additional seeds from the families (different parameters)
    seed_idx = 1
    for p in (0.8, 2.0, 4.0):
        if seed_idx >= pop_size: break
        population[seed_idx] = u_min_distribution(n, p)
        scores[seed_idx] = score_vector(population[seed_idx])
        seed_idx += 1
        if seed_idx >= pop_size: break
        population[seed_idx] = u_prod_distribution(n, p)
        scores[seed_idx] = score_vector(population[seed_idx])
        seed_idx += 1
    for lam in (1e-3, 5e-2, 2e-1):
        if seed_idx >= pop_size: break
        population[seed_idx] = exp_distribution(n, lam)
        scores[seed_idx] = score_vector(population[seed_idx])
        seed_idx += 1

    # fill remaining slots with uniform Dirichlet draws
    for i in range(seed_idx, pop_size):
        rnd = np.random.rand(n)
        population[i] = rnd / rnd.sum()
        scores[i] = score_vector(population[i])

    # keep a global best copy
    best_idx = int(scores.argmin())
    best_vec = population[best_idx].copy()
    best_score = float(scores[best_idx])

    generation = 0
    no_improve_gens = 0
    print("\n=== DE optimisation start ===")
    while time.time() < deadline:
        generation += 1
        gen_improved = False
        for i in range(pop_size):
            # --- mutation ---
            idxs = np.arange(pop_size)
            idxs = idxs[idxs != i]
            a, b, c = np.random.choice(idxs, 3, replace=False)
            donor = population[a] + F * (population[b] - population[c])
            donor = np.maximum(donor, 0.0)          # keep non‑negative

            # --- crossover ---
            crossover_mask = np.random.rand(n) < CR
            if not np.any(crossover_mask):
                crossover_mask[np.random.randint(n)] = True
            trial = np.where(crossover_mask, donor, population[i])

            # --- simplex projection ---
            trial = simplex_projection(trial)

            # --- evaluation ---
            trial_score = score_vector(trial)

            # --- selection ---
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    gen_improved = True
                    print(f"Gen {generation:5d} ind {i:2d}: new best = {best_score:.6f}")

        # stall detection – replace a fraction of the worst individuals
        if not gen_improved:
            no_improve_gens += 1
        else:
            no_improve_gens = 0

        if no_improve_gens >= 30:          # 30 generations without any improvement
            # re‑heat a portion of the population
            n_replace = max(1, pop_size // 5)
            worst = np.argsort(scores)[-n_replace:]
            for w in worst:
                rnd = np.random.rand(n)
                population[w] = rnd / rnd.sum()
                scores[w] = score_vector(population[w])
            print(f"Gen {generation:5d}: re‑heated {n_replace} individuals")
            no_improve_gens = 0

        # occasional time check
        if time.time() > deadline:
            break

    return best_vec, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing with a smooth max (soft‑max)
# ----------------------------------------------------------------------
def expmax_gradient_polish(vec: np.ndarray,
                           current_score: float,
                           deadline: float) -> Tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on the smooth
    approximation of max_k (a*a)[k].  Uses a soft‑max with temperature τ
    that is slowly increased, and a decaying step size.
    """
    a = vec.copy()
    best = a.copy()
    best_score = current_score

    step = 0.20                     # initial additive step (in exponent)
    step_decay = 0.9995
    tau = 5.0                       # temperature for soft‑max
    tau_inc = 0.12
    it = 0
    last_improve = 0

    n = a.shape[0]
    print("\n=== Exponentiated‑gradient polishing start ===")
    while time.time() < deadline:
        it += 1
        # ----- convolution and soft‑max weights -----
        conv = full_convolution_fft(a)            # a * a
        max_val = conv.max()
        # soft‑max (subtract max to avoid overflow)
        exp_terms = np.exp(tau * (conv - max_val))
        w = exp_terms / exp_terms.sum()           # weight vector, sum = 1

        # ----- gradient of the smooth max (2 * a ⋆ w) -----
        # cross‑correlation a ⋆ w can be obtained via convolution of a_rev and w
        a_rev = a[::-1]
        grad_full = full_convolution_fft(a_rev, w)   # length = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # shape (n,)

        # ----- multiplicative update on the simplex -----
        cand = a * np.exp(-step * grad)              # element‑wise
        cand = simplex_projection(cand)              # back to simplex
        cand_score = score_vector(cand)

        # ----- accept if we improve -----
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            last_improve = it
            print(f"  it {it:6d}: new best = {best_score:.6f}")

        # we always move forward (the direction is a descent direction,
        # but we keep the old a if the move made things worse)
        if cand_score <= current_score:
            a = cand
            current_score = cand_score
        # else keep a unchanged (acts like a back‑track)

        # ----- schedules -----
        step *= step_decay
        tau = min(500.0, tau + tau_inc)

        # ----- termination / re‑heat -----
        if step < 1e-8:
            break
        if it - last_improve > 4000:
            # small random jitter to escape possible plateaus
            jitter = np.random.rand(n) * 1e-6
            a = simplex_projection(a + jitter)
            a /= a.sum()
            current_score = score_vector(a)
            step = 0.20
            tau = 5.0
            last_improve = it
            print(f"  it {it:6d}: re‑heat")

    return best, best_score


# ----------------------------------------------------------------------
# Final sub‑gradient polishing (as used in the baseline)
# ----------------------------------------------------------------------
def subgradient_polish(vec: np.ndarray,
                       current_score: float,
                       deadline: float) -> Tuple[np.ndarray, float]:
    """Short projected sub‑gradient polishing (same as the baseline)."""
    n = vec.shape[0]
    a = vec.copy()
    best = a.copy()
    best_score = current_score

    step = 0.15
    decay = 0.9995
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution_fft(a)
        grad = compute_subgradient(a, conv)      # uses the exact max‑lag sub‑gradient
        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)
        cand_score = score_vector(cand)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            last_improve = it
            print(f"Subgrad it {it:6d}: new best = {best_score:.6f}")

        a = cand
        step *= decay
        if step < 1e-7:
            break
        if it - last_improve > 3000:
            # random re‑heat
            bump = np.random.rand(n) * 1e-5
            a = simplex_projection(a + bump)
            a /= a.sum()
            step = 0.15
            last_improve = it
            print(f"Subgrad it {it:6d}: re‑heat")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative height sequence that minimizes
    2 * n * max_k (a*a)[k]  (with the sum of the sequence normalised to 1).
    The routine respects the supplied time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0          # small safety margin

    # 1. deterministic sweep for a good starting point
    best_vec, best_score = deterministic_sweep(deadline)

    # 2. DE optimisation on the simplex (global exploration)
    if time.time() < deadline:
        best_vec, best_score = differential_evolution(best_vec, best_score, deadline)

    # 3. Exponentiated‑gradient polishing (smooth max)
    if time.time() < deadline:
        best_vec, best_score = expmax_gradient_polish(best_vec, best_score, deadline)

    # 4. Final short sub‑gradient polishing (fine‑tune)
    if time.time() < deadline:
        best_vec, best_score = subgradient_polish(best_vec, best_score, deadline)

    # safety: ensure non‑negative and normalised
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec = np.ones_like(best_vec) / len(best_vec)
    else:
        best_vec = best_vec / s

    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {len(best_vec)})")
    return best_vec.tolist()
