"""
Propose a step‑function (non‑negative heights that sum to 1) that minimizes
``evaluate_sequence``.

Algorithm Overview
------------------
1. **Parameterisation** – The vector `a` is represented by soft‑max
   parameters `w` (so `a_i = exp(w_i)/∑exp(w)`), guaranteeing feasibility.

2. **Smooth surrogate** – Minimise a smoothed version of
   ``max(conv)``:
        Lε(a) = 2·n·[ max(conv) + ε·log∑_k exp((conv_k‑max(conv))/ε) ] .
   The gradient of `Lε` w.r.t. `w` is obtained analytically using FFT
   (two convolutions per evaluation).

3. **Continuation + multi‑restart** – Run a schedule of decreasing ε.
   For each ε we perform a short L‑BFGS‑B optimisation.  The whole
   optimisation is repeated several times from *different random starts*
   (Dirichlet‑distributed initial points) – the best result over all
   restarts is retained.

4. **Projected sub‑gradient refinement** – After the continuation phase
   we use a cheap projected sub‑gradient method on the *exact* max
   (no smoothing).  Each iteration finds the index `k*` where the
   convolution attains its maximum and takes a step in the direction
   `-∇max = -2·a_{k*‑i}`.  The step is projected onto the probability
   simplex (non‑negative, sum = 1) by removing the mean and clipping.

5. **Final random hill‑climb** – A short stochastic hill‑climbing phase
   (random mass transfers) explores the local neighbourhood and may
   capture improvements missed by the gradient steps.

All phases respect the 1000 s budget; the current best sequence (and its
exact evaluation) is printed regularly.

The implementation uses only `numpy` and `scipy`.
"""

# ----------------------------------------------------------------------
# FFT utilities (real data) – used both by the smooth surrogate and the
# sub‑gradient refinement (to compute the exact convolution quickly)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: smallest power of two ≥ `x`."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negatives caused by floating‑point noise.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp surrogate) and its gradient w.r.t.
# the soft‑max parameters `w`.
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Return the smoothed objective ``Lε`` and its gradient w.r.t. ``w``.
    The parameter vector `w` is unconstrained; the corresponding
    probability vector `a` satisfies a_i ≥ 0 and Σ a_i = 1.
    """
    # ---- soft‑max transformation (feasible a) ----
    w_shift = w - np.max(w)           # improve numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                     # Σ a_i = 1, a_i ≥ 0

    # ---- auto‑convolution a*a (full, length 2·n‑1) ----
    conv = _full_convolution(a)

    # ---- smoothed max via log‑sum‑exp ----
    max_conv = conv.max()
    shifted = (conv - max_conv) / eps          # (conv‑max)/ε
    log_sum = logsumexp(shifted)               # log∑ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a ----
    # p_k = exp(conv_k/ε) / Σ_j exp(conv_j/ε)
    # we already have shifted = (conv‑max)/ε, so:
    weights = np.exp(shifted)                  # ∝ exp(conv/ε)
    p = weights / weights.sum()                # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}  (indices outside the range are ignored)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)              # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]            # the centre n entries
    grad_a = 4.0 * n * seg[::-1]               # shape (n,)

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)                    # scalar = <grad_a , a>
    grad_w = a * (grad_a - dot)                # ∂L/∂w (size n)

    return L, grad_w


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – fallback to uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
