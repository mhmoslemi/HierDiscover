"""
propose_candidate
-----------------
Search for a non‑negative step‑function (probability vector) that
minimises

    2 * n * max_k (a * a)[k]

where * denotes the full linear convolution.  The algorithm works in
three stages:

1.  A deterministic sweep over several parametric families
    (U‑shaped power laws, exponential decay, and a symmetric beta‑
    distribution with shape parameter α<1) for a handful of problem
    sizes.  The best vector from this sweep is kept as a seed.

2.  A fast projected gradient descent that directly minimises a
    smooth approximation of the max‑convolution (soft‑max with a
    temperature that is increased gradually).  The gradient of the
    soft‑max is obtained via an FFT‑based convolution, giving an
    O(n log n) step.

3.  The best vector discovered is returned as a plain Python list.
    All vectors are kept on the probability simplex, so the objective
    is scale‑invariant as required by ``evaluate_sequence``.

The implementation respects the 1000‑second time budget, prints progress
and never uses closures or lambdas.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {w >=0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    theta = 0.0
    if rho.size > 0:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

def score_vector(a: np.ndarray) -> float:
    """
    Exact value returned by evaluate_sequence for a vector with sum == 1.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Parametric families for deterministic sweep
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i, n‑1‑i)+1)^{-p} (edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^{-p} (edge‑heavy, product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a

def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i) + exp(-lam*(n‑1‑i)) (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

def beta_u_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on n points.
    For α<1 the distribution is U‑shaped (edge‑heavy).
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # protect against 0**negative which would be inf
    eps = 1e-14
    a = np.power(np.maximum(x, eps), alpha - 1) * np.power(np.maximum(1 - x, eps), alpha - 1)
    a = np.maximum(a, eps)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Gradient helpers for soft‑max approximation of max‑convolution
# ----------------------------------------------------------------------
def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_k = exp(alpha*conv_k) / sum.
    For numerical stability we subtract max(conv) before exponentiation.
    """
    maxv = conv.max()
    exps = np.exp(alpha * (conv - maxv))
    w = exps / exps.sum()
    return w

def weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of sum_k w_k * (a*a)[k] w.r.t. a.
    Uses the identity  ∂/∂a_i (a*a)[k] = 2 * a_{k-i}
    and computes the weighted sum via an FFT‑based convolution.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # length needed for full convolution of w (len 2n-1) and a_rev (len n)
    needed = len(w) + n - 1            # = 3n-2
    sz = 1 << ((needed - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a_rev, n=sz)
    Fw = np.fft.rfft(w, n=sz)
    conv_full = np.fft.irfft(Fa * Fw, n=sz)
    # gradient_i = 2 * conv_full[n-1-i]  for i=0..n-1
    g = 2.0 * conv_full[n-1::-1][:n]
    return g
