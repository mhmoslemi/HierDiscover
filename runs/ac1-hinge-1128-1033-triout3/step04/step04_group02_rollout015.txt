def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    n = a.shape[0]
    grad = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    # indices i range [i_min, i_max]
    # corresponding j = k-i range [k-i_max, k-i_min]
    # We can compute slice
    grad[i_min:i_max+1] = 2 * a[k - np.arange(i_min, i_max+1)]
    return grad

Projection to simplex function:

def project_simplex(v):
    # algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.maximum(v, 0)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Run several random restarts.  For each restart
  - initialise a probability vector ``a`` from a Dirichlet distribution;
    store its log‑soft‑max parameters ``w``.
  - minimise a smooth log‑sum‑exp surrogate of the max‑convolution by a
    continuation of L‑BFGS‑B with decreasing smoothing parameter ε.
  - the surrogate has a closed‑form gradient that uses two FFT based
    convolutions per evaluation.
  - after the continuation phase perform a few hundred sub‑gradient steps
    on the *exact* max‑convolution (gradient = 2·a shifted by the index of
    the maximal entry) with projection onto the probability simplex.
  - finally apply a cheap hill‑climbing refinement (random mass transfers)
    until the time budget is exhausted.
* The best sequence seen by the exact ``evaluate_sequence`` is kept and
  returned.

The whole routine respects the 1000 s budget, prints progress and works
with any previously found global variable ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself (length 2·len‑1).
    Uses an FFT of the next power of two for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative round‑off
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp of the convolution) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smoothed surrogate L_ε and its gradient w.r.t. the
    soft‑max parameters w.

    L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    where conv = a*a (full auto‑convolution) and a = softmax(w).
    """
    # ---- soft‑max, guarantees a_i≥0 and Σa_i = 1 --------------------
    w_shift = w - np.max(w)                 # for numeric stability
    a = np.exp(w_shift)
    a_sum = a.sum()
    a = a / a_sum

    # ---- full auto‑convolution ------------------------------------
    conv = _full_convolution(a)            # length 2·n‑1
    max_conv = conv.max()

    # ---- smoothed max (log‑sum‑exp) -------------------------------
    # Prevent overflow by shifting with max_conv
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)           # = log Σ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (closed form) -------------------------
    # p_k = exp(conv_k/ε) / Σ_j exp(conv_j/ε)
    weights = np.exp(shifted)
    p = weights / weights.sum()            # shape (2·n‑1,)

    # g_i = 4·n·Σ_k p_k·a_{k‑i}   (indices outside [0,n‑1] give 0)
    p_rev = p[::-1]                        # reversed p
    conv_ap = _conv_fft(a, p_rev)          # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]        # middle n entries
    grad_a = 4.0 * n * seg[::-1]           # shape (n,)

    # ---- chain rule through the soft‑max -------------------------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)            # ∂L/∂w
    return L, grad_w


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean projection)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}.
    Algorithm of Duchi et al. (2008) – O(n log n).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries negative → project to uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, n: int,
                         deadline: float, best_val: float):
    """
    Perform a few hundred iterations of sub‑gradient descent on the
    exact objective 2·n·max(conv) (sum(a)=1 is always maintained).
    Returns the best vector and its exact evaluation.
    """
    a = a0.copy()
    step0 = 0.5                       # initial step size (will decay)
    cur_val = best_val

    while time.time() < deadline:
        # compute full convolution once per iteration
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # gradient of max(conv) w.r.t. a: 2·a shifted by k_max
        grad = np.zeros_like(a)
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        # slice update; note that k_max - np.arange(i_min, i_max+1) is a valid index
        grad[i_min:i_max + 1] = 2.0 * a[k_max - np.arange(i_min, i_max + 1)]

        # decaying step size (√ schedule)
        step = step0 / np.sqrt(np.sum(np.abs(grad)) + 1.0)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        a = a_new
        # occasional exact evaluation (every 200 iterations)
        if int(np.sum(np.abs(grad))) % 200 == 0:
            try:
                val = evaluate_sequence(a.tolist())
                if val < cur_val:
                    cur_val = val
                    best_seq = a.copy()
                    print(f"[subgrad] new best = {cur_val:.6f}")
            except Exception:
                pass
    return a, cur_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (random mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Randomly move a small fraction of mass from one index to another.
    Keep changes that improve the exact objective.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer up to 5 % of the mass at i
        delta = cur_a[i] * 0.05 * rng.random()
        if delta <= 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val
