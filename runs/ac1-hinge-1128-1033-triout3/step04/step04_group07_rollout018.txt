"""
Propose a step‐function on a (potentially) larger support and
optimise its L∞ autocorrelation (equivalently minimise
2·n·max(conv_self)/sum²). The method builds many diverse starts
(asymmetric power‑law, geometric, skewed beta, random Dirichlet)
and refines each by:
  * projected sub‑gradient descent on the exact max entry,
  * Adam optimisation on a log‑sum‑exp smooth approximation,
  * a final fine‑grained sub‑gradient pass,
  * a random‑swap hill‑climber.
The best vector found before the time budget expires is returned.
"""
import numpy as np
import time

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution using FFT of length m_fft."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv

def _gradient_from_weights(p: np.ndarray, p_fft: np.ndarray,
                          w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Gradient of the (α·log‑sum‑exp) smoothed max.
    w – soft‑max weights over the 2·n‑1 convolution entries, sum(w)=1.
    """
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * p.size - 2]
    idx = 2 * p.size - 2 - np.arange(p.size)
    grad = 2.0 * corr[idx]
    return grad

def _subgradient(p: np.ndarray, n: int, m_fft: int,
                 tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv_self(p)) w.r.t. p.
    Returns a vector of the same shape as p.
    """
    conv = _conv_self(p, m_fft)
    max_val = conv.max()
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(p)
    for idx in active:
        if idx < n:
            # contribution from pairs (i, idx-i) with i≤idx
            g[: idx + 1] += 2.0 * p[idx::-1]
        else:
            # idx ≥ n : the overlapping part slides back
            start = idx - n + 1
            # elements p[start:] correspond to i where i+ (idx-i) = idx
            g[start:] += 2.0 * p[n - 1 : start - 1 : -1]
    g /= float(len(active))
    return g

def _objective(p: np.ndarray, n: int, m_fft: int) -> float:
    """Exact objective: 2·n·max(conv_self(p)).  The vector is assumed to sum to 1."""
    conv = _conv_self(p, m_fft)
    return float(2.0 * n * conv.max())

    2·n·max(conv_self) / (sum(a)²)

The algorithm builds a diverse set of starting vectors (uniform,
geometric decays, power‑law, skewed Beta shapes and random Dirichlet
draws), then refines each of them with four stages:

1. Projected sub‑gradient descent on the exact maximum convolution entry.
2. Adam optimisation on a log‑sum‑exp smooth approximation of the max,
   slowly increasing the smoothing parameter α.
3. A fine‑grained sub‑gradient pass with a smaller step size.
4. A tiny random‑swap hill‑climber (accept‑only‑improves).

# Helper: projection onto the probability simplex.

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
