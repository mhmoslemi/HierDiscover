"""
Propose a candidate step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the full linear convolution.
The algorithm combines:

* A fast full convolution using a real FFT.
* An Adam‑style exponentiated‑gradient descent on the simplex.
  The gradient is the exact sub‑gradient of the ∞‑norm
  (i.e. only the index attaining the current maximum is used).
* A tiny amount of random “jitter’’ to escape plateaus.
* Periodic bookkeeping of the best solution seen so far.

The routine runs for the allotted time (default 1000 s) and returns the
best sequence (as a plain list of floats) found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution (real FFT, next power‑of‑two length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Numerical noise may produce tiny negatives – clamp to 0
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Evaluation (the function used by the competition framework)
# ----------------------------------------------------------------------
def _fast_evaluation(a: np.ndarray) -> float:
    """Compute 2·n·max(conv(a,a)) / (sum(a)**2) efficiently."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    # The external evaluator protects against sum_a < 0.01;
    # we keep sum_a close to 1, so the division is safe.
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Adam‑style exponentiated‑gradient descent on the exact max‑convolution
    sub‑gradient, with occasional random jitter.  Returns the best sequence
    found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin – finish a couple of seconds earlier
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Initialise the probability vector a (sum = 1)
    # --------------------------------------------------------------
    try:
        # Use the previously discovered construction if it exists
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / a.sum()
    except Exception:
        # Otherwise start from a moderately sparse Dirichlet draw
        n = 2500                         # feel free to enlarge if time permits
        a = np.random.dirichlet(alpha=np.full(n, 0.5))
    n = a.shape[0]

    # --------------------------------------------------------------
    # Optimisation bookkeeping
    # --------------------------------------------------------------
    best_a = a.copy()
    best_val = _fast_evaluation(best_a)
    print(f"[init] n={n}, initial evaluation={best_val:.6f}")

    # Adam moment estimates (initialized to zero)
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment

    # Hyper‑parameters
    eta_start = 0.60              # initial (raw) learning rate
    eta_end   = 1e-4              # final learning rate
    beta1 = 0.90                  # first‑moment decay
    beta2 = 0.999                 # second‑moment decay
    eps = 1e-12                   # for numerical stability
    jitter_prob = 0.012           # probability of a random mass transfer
    max_step_clip = 30.0          # clip the exponent to avoid overflow

    iteration = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- full convolution and current objective value -----------------
        conv = _full_convolution_fft(a, a)          # length 2n‑1
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        cur_val = 2.0 * n * max_val / (a.sum() * a.sum())   # identical to _fast_evaluation(a)

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # ---- progress‑dependent learning‑rate ---------------------------
        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # ---- sub‑gradient of the max‑convolution (hard max) -------------
        # For each i, the contribution to the maximal entry is
        #   a[i] * a[max_idx - i]  (if the partner index lies inside 0…n‑1)
        partner = max_idx - np.arange(n)
        mask = (partner >= 0) & (partner < n)

        # Initialise gradient with the two‑times‑partner rule
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[partner[mask]]

        # When i == partner (i.e. 2*i == max_idx) the term appears only once
        equal_mask = mask & (partner == np.arange(n))
        if np.any(equal_mask):
            g[equal_mask] = a[equal_mask]      # single occurrence

        # ---- Adam updates (first & second moments) ----------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # Bias‑correction
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        # ---- multiplicative (exponentiated) update -----------------------
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        np.clip(step, -max_step_clip, max_step_clip, out=step)   # safety
        a = a * np.exp(-step)          # (a_i ← a_i·exp(-η·ĝ_i))

        # ---- occasional random “jitter’’ --------------------------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # transfer a random fraction (≤50 %) of the mass at src to dst
            delta = a[src] * np.random.rand() * 0.5
            a[src] -= delta
            a[dst] += delta

        # ---- projection back onto the simplex (simple renormalisation) ---
        a = np.maximum(a, eps)                 # stay strictly positive
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        iteration += 1

        # ---- occasional full recomputation (to keep numeric drift tiny)--
        if iteration % 3000 == 0:
            # recompute just to wipe out any accumulated rounding errors
            a = a / a.sum()

    # --------------------------------------------------------------
    # Final reporting (the outer framework will also call evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
