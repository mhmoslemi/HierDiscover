"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.
--------------------------------------------------------------------------

The algorithm has three stages:

1. **Smooth optimisation** –  the original non‑smooth objective
   ``max(conv)`` is replaced by a log‑sum‑exp surrogate.
   The surrogate is minimised with L‑BFGS‑B while the smoothing
   parameter ε is decreased (continuation).  The sequence is
   represented by a soft‑max so that all entries stay non‑negative
   and the ℓ₁‑norm is fixed (the objective is scale‑invariant).

2. **Random restarts** –  the continuation from several random
   perturbations of the best solution helps to escape local minima.

3. **Guided hill‑climbing** –  after the smooth phase we iteratively
   reduce the *true* maximum convolution value.
   We locate the lag `k` where the convolution attains its maximum,
   identify the pair of indices `(i, j)` that contribute most to it,
   and move a small amount of mass from that pair to two other
   positions that do not create a new peak.  The move is accepted
   only if the exact ``evaluate_sequence`` value improves.
   The step size is gradually reduced (simulated‑annealing style).

The whole routine respects a 1000 s time budget; progress and any
new best scores are printed.  The final list of heights is returned
as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative values caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp surrogate) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Given soft‑max parameters `w`, compute the smooth surrogate
    L_ε(a) = 2·n·[max(conv) + ε·log Σ exp((conv‑max)/ε)]
    together with its gradient w.r.t. `w`.
    """
    # ----- soft‑max (feasible a, Σa = 1) -----
    w_shift = w - np.max(w)       # improve numerical stability
    a = np.exp(w_shift)
    a = a / a.sum()

    # ----- full auto‑convolution a*a -----
    conv = _full_convolution(a)   # length 2·n‑1
    max_conv = conv.max()

    # ----- log‑sum‑exp smoothing -----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # = log Σ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a (closed form) -----
    # p_k = exp(conv_k/ε) / Σ_j exp(conv_j/ε)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]                       # reversed p
    conv_ap = _conv_fft(a, p_rev)         # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]       # middle n entries
    grad_a = 4.0 * n * seg[::-1]          # shape (n,)

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)           # ∂L/∂w
    return L, grad_w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Reduce the exact objective by moving mass away from the most
    offending convolution lag.  The procedure stops when the time
    budget expires.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05            # initial fraction of mass to move

    while time.time() < deadline:
        # compute full autocorrelation
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # index (or one of them) where the maximum is attained
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            # should not happen, but guard against it
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose destination indices that do NOT create the same sum k
        # (simple heuristic – pick from the low‑mass tail)
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # ensure the new pair does not reproduce the offending lag
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # determine a feasible mass transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        # safety: keep non‑negativity (numpy may produce -0.)
        if (new_a < 0).any():
            continue

        # evaluate the exact objective
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        # accept only improvements (simple greedy descent)
        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            # tighten step size a little after a success
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            # slowly shrink the step size even on failures
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val
