    n = len(a)
    # Precompute FFT size
    size2 = 1 << ((3*n - 2).bit_length())
    # parameters
    tau = 0.5
    tau_inc = 1.05
    step = 0.1
    step_decay = 0.999
    best_a = a.copy()
    best_score = score_vector(a)
    for it in range(max_iters):
        if time.time() > time_deadline:
            break
        # compute conv via FFT
        conv = full_convolution_fft(a)  # length 2n-1
        # compute softmax weights
        maxc = conv.max()

def gradient_annealing(a, deadline, max_iter=20000, init_step=0.1, step_decay=0.9995,
                     init_tau=0.5, tau_mult=1.02):
    n = a.shape[0]
    best_a = a.copy()
    best_score = score_vector(a)
    # precompute FFT size for gradient
    size_corr = 1 << ((3*n - 2).bit_length())
    # allocate padded arrays for reuse
    a_pad = np.zeros(size_corr, dtype=np.float64)
    w_pad = np.zeros(size_corr, dtype=np.float64)
    step = init_step
    tau = init_tau
    last_improve = 0
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute convolution
        conv = full_convolution_fft(a)
        # compute softmax weights
        maxc = conv.max()
        # stability: subtract maxc
        exp_vals = np.exp(tau * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # compute gradient via correlation
        # fill padded arrays
        a_pad[:n] = a
        w_pad[:2*n-1] = w
        # FFT
        ffta = np.fft.rfft(a_pad)
        fftw = np.fft.rfft(w_pad)
        corr = np.fft.irfft(ffta * np.conj(fftw), n=size_corr)
        start = (2*n-1) - 1  # offset
        grad = corr[start:start+n]
        # gradient step
        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)
        cand /= cand.sum()
        # enforce symmetry
        cand = (cand + cand[::-1]) * 0.5
        cand /= cand.sum()
        cand_score = score_vector(cand)
        # always move (like gradient descent)
        a = cand
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            last_improve = it
            print(f" Anneal iter {it:5d}: new best {best_score:.6f}")
        # update step & tau
        step *= step_decay
        if step < 1e-8:
            step = 1e-8
        tau *= tau_mult
        if tau > 1e4:
            tau = 1e4
        # reheat if stuck
        if it - last_improve > 5000:
            # add tiny random perturbation
            bump = np.random.rand(n) * 1e-5
            a = simplex_projection(a + bump)
            a /= a.sum()
            a = (a + a[::-1]) * 0.5
            last_improve = it
            print(f" Anneal iter {it}: reheat")
    return best_a, best_score

But a loop may overhead; using bit_length: `size_corr = 1 << ((3*n - 2).bit_length())` works because if x is a power of 2, say 8 (1000b) bit_length=4, shift left yields 16 (overshoot). So we need to adjust: If x is power of two, we should not double. So we can compute:

```
size_corr = 1 << ((3*n - 2 - 1).bit_length())
```

Because (x-1).bit_length() returns bits needed for (x-1). For x=8, x-1=7 (111b) bit_length=3 -> 1 << 3 = 8, correct. For x=9, x-1=8 (1000b) bit_length=4 -> 1 << 4 = 16 >9. Good.

Thus:

```
size_corr = 1 << ((3*n - 2 - 1).bit_length())
```

Now for full_convolution_fft: we can compute size = 1 << ((2*n - 1).bit_length()) to have next power-of-two; but we may simply compute as `size = 2*n` which is power-of-two only for certain n; but rfft works with any length; but zero padding to length 2*n yields conv length up to 2*n-1. So we can keep that as earlier.

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimises

        2 * n * max_k (a * a)[k]

where * denotes the full linear convolution.  The routine works in three
stages and respects a 1000 s time budget.

1. **Deterministic sweep** – a small set of parametric families
   (inverse‑distance, inverse‑product‑distance, symmetric exponential)
   is evaluated for several lengths.  The best vector from this cheap
   sweep is taken as a seed.

2. **Annealed soft‑max gradient descent** – the max‑convolution term is
   approximated by a smooth soft‑max (temperature τ).  The gradient of the
   smoothed objective is obtained via FFT‑based cross‑correlation,
   projected onto the probability simplex and symmetrised after each
   step.  τ is increased gradually so that the optimisation first moves
   in a smooth landscape and finally focuses on the true ℓ∞ norm.

3. **Projected Differential Evolution & sub‑gradient polishing** – the
   best vector from stage 2 is refined globally with a classic DE loop
   on the simplex and finally polished by a short projected
   sub‑gradient descent.

All helper functions are top‑level, no closures are used and the code
only relies on NumPy (and optionally SciPy for a fast correlation, but
a pure‑NumPy FFT implementation is provided).  Progress is printed and
the best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex:
        { w >= 0 , sum(w) = 1 }.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_vector(a: np.ndarray) -> float:
    """
    Exact score returned by ``evaluate_sequence`` for a vector that
    already satisfies sum(a) == 1 (and a_i >= 0).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric U‑shaped families (used for the deterministic sweep)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i, n‑1‑i)+1)^{-p} (symmetric, edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^{-p} (edge‑heavy, product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i) + exp(-lam*(n‑1‑i)) (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max annealed gradient descent (stage 2)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    if x <= 1:
        return 1
    return 1 << ((x - 1).bit_length())


def annealed_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 20000,
    init_step: float = 0.12,
    step_decay: float = 0.9995,
    init_tau: float = 0.6,
    tau_mult: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smoothed version of the ℓ∞‑convolution
    objective.  Temperature τ (soft‑max sharpness) grows gradually.
    The vector is kept symmetric (a == a[::-1]) because any optimal
    solution can be symmetrised without worsening the score.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = score_vector(a)

    # size for the cross‑correlation (a ⋆ w) – length = 3n‑2
    corr_len = 3 * n - 2
    fft_size = _next_pow2(corr_len)

    # pre‑allocate padded buffers (avoid repeated allocations)
    a_pad = np.zeros(fft_size, dtype=np.float64)
    w_pad = np.zeros(fft_size, dtype=np.float64)

    step = init_step
    tau = init_tau
    last_improve = 0
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- convolution and soft‑max weights -----
        conv = full_convolution_fft(a)                      # length 2n‑1
        maxc = conv.max()
        # stability trick: subtract maxc before exponentiation
        exp_vals = np.exp(tau * (conv - maxc))
        w = exp_vals / exp_vals.sum()                       # soft‑max weights

        # ----- gradient via FFT‑based cross‑correlation -----
        a_pad[:n] = a
        w_pad[: 2 * n - 1] = w
        ffta = np.fft.rfft(a_pad)
        fftw = np.fft.rfft(w_pad)
        corr = np.fft.irfft(ffta * np.conj(fftw), n=fft_size)
        start = (2 * n - 1) - 1
        grad = corr[start:start + n]                         # linear cross‑correlation

        # ----- projected gradient step -----
        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)
        cand /= cand.sum()

        # force symmetry (does not change the sum)
        cand = (cand + cand[::-1]) * 0.5
        cand /= cand.sum()

        cand_score = score_vector(cand)
        a = cand

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            last_improve = it
            print(f"  Anneal it {it:5d}: new best {best_score:.6f}")

        # ----- schedule updates -----
        step *= step_decay
        if step < 1e-9:
            step = 1e-9
        tau *= tau_mult
        if tau > 1e4:
            tau = 1e4

        # re‑heat if stuck for a long while
        if it - last_improve > 5000:
            bump = np.random.rand(n) * 1e-5
            a = simplex_projection(a + bump)
            a /= a.sum()
            a = (a + a[::-1]) * 0.5
            a /= a.sum()
            last_improve = it
            print(f"  Anneal it {it:5d}: re‑heat")

    return best_a, best_score


# ----------------------------------------------------------------------
# Differential Evolution on the simplex (stage 3)
# ----------------------------------------------------------------------
def differential_evolution(
    seed_vec: np.ndarray,
    deadline: float,
    pop_size: int = 40,
    F: float = 0.75,
    CR: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Classic DE on the probability simplex.  The population is initialised
    with the supplied seed plus a few curated variations and the rest
    with random Dirichlet draws.
    """
    n = seed_vec.shape[0]
    population = np.empty((pop_size, n), dtype=np.float64)
    pop_scores = np.empty(pop_size, dtype=np.float64)

    # curated seeds
    seeds = [seed_vec]
    for p in (0.6, 1.5, 3.0):
        seeds.append(u_min_distribution(n, p))
        seeds.append(u_prod_distribution(n, p))
    for lam in (5e-4, 2e-3, 1e-2):
        seeds.append(exp_distribution(n, lam))

    for i, vec in enumerate(seeds[:pop_size]):
        population[i] = vec
        pop_scores[i] = score_vector(vec)

    # fill remaining slots with random Dirichlet vectors
    for i in range(len(seeds), pop_size):
        rnd = np.random.rand(n)
        population[i] = rnd / rnd.sum()
        pop_scores[i] = score_vector(population[i])

    best_idx = int(pop_scores.argmin())
    best_vec = population[best_idx].copy()
    best_score = float(pop_scores[best_idx])

    generation = 0
    while time.time() < deadline:
        generation += 1
        for i in range(pop_size):
            # --- mutation (DE/rand/1) ---
            idxs = np.arange(pop_size)
            idxs = idxs[idxs != i]
            a, b, c = np.random.choice(idxs, 3, replace=False)
            donor = population[a] + F * (population[b] - population[c])
            donor = np.maximum(donor, 0.0)

            # --- crossover (binomial) ---
            crossover_mask = np.random.rand(n) < CR
            if not np.any(crossover_mask):
                crossover_mask[np.random.randint(n)] = True
            trial = np.where(crossover_mask, donor, population[i])

            # --- projection onto the simplex + symmetrisation ---
            trial = simplex_projection(trial)
            trial /= trial.sum()
            trial = (trial + trial[::-1]) * 0.5
            trial /= trial.sum()

            trial_score = score_vector(trial)

            # --- selection ---
            if trial_score < pop_scores[i]:
                population[i] = trial
                pop_scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    print(
                        f"  DE gen {generation:5d} ind {i:2d}: new best = {best_score:.6f}"
                    )
        # occasional time check
        if time.time() >= deadline:
            break
    return best_vec, best_score


# ----------------------------------------------------------------------
# Short projected sub‑gradient polishing (final stage)
# ----------------------------------------------------------------------
def subgradient_polish(
    a0: np.ndarray,
    deadline: float,
    init_step: float = 0.14,
    step_decay: float = 0.9994,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the exact ℓ∞ objective.
    Only the indices attaining the maximal convolution value receive a
    non‑zero gradient (see ``compute_subgradient``).
    """
    n = a0.shape[0]

    def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-15
        max_idx = np.where(conv >= max_val - eps)[0]
        grad = np.zeros_like(a)
        for k in max_idx:
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
            j_vals = k - i_vals
            # each (i,j) contributes a[j] to ∂/∂a[i] and symmetrically
            grad[i_vals] += a[j_vals]
        return grad

    current = a0.copy()
    current_score = score_vector(current)
    best_vec = current.copy()
    best_score = current_score
    step = init_step
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution_fft(current)
        grad = compute_subgradient(current, conv)

        cand = current - step * grad
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)
        cand /= cand.sum()
        cand = (cand + cand[::-1]) * 0.5
        cand /= cand.sum()
        cand_score = score_vector(cand)

        if cand_score < current_score:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                last_improve = it
                print(f"  Subgrad it {it:6d}: new best = {best_score:.6f}")

        step *= step_decay
        if step < 1e-7:
            break
        if it - last_improve > 3000:
            # small random re‑heat
            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            current /= current.sum()
            current = (current + current[::-1]) * 0.5
            current /= current.sum()
            current_score = score_vector(current)
            step = init_step
            last_improve = it
            print(f"  Subgrad it {it}: re‑heat")
    return best_vec, best_score
