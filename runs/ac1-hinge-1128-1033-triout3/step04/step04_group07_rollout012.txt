"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(conv_self) / (sum(a)²)

The optimizer works in four stages:

1. **L2‑flattening** – minimise the squared error between the
   self‑convolution and the uniform distribution.  This pushes the
   convolution towards a flat shape and gives a very good starting
   point for the true max‑objective.

2. **Smooth‑max (log‑sum‑exp) descent** – use a temperature‑controlled
   smooth approximation of the max and optimise it with Adam.

3. **Exact max sub‑gradient** – finish with a few sub‑gradient steps on
   the true max entry (active‑set method).

4. **Simulated‑annealing hill‑climber** – occasional uphill moves are
   accepted to escape local minima and polish the final vector.

All updates are projected onto the probability simplex
{ x ≥ 0 , Σ x = 1 }, which guarantees a non‑negative vector with
unit sum – exactly what `evaluate_sequence` expects.  The routine
adapts its iteration budget to the supplied time limit (default 1000 s)
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full convolution)
# ----------------------------------------------------------------------
def _conv_full(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution using FFT of size m_fft."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)
    return conv[: 2 * p.size - 1]


def _conv_full_two(a: np.ndarray, b: np.ndarray, m_fft: int) -> np.ndarray:
    """Full convolution a * b using the same FFT size."""
    a_fft = np.fft.rfft(a, n=m_fft)
    b_fft = np.fft.rfft(b, n=m_fft)
    conv = np.fft.irfft(a_fft * b_fft, n=m_fft)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Gradient of the squared‑error loss L2(p) = ½‖p∗p – u‖²
# (u = uniform distribution on the convolution support)
# ----------------------------------------------------------------------
def _grad_l2(p: np.ndarray, target: np.ndarray, m_fft: int) -> np.ndarray:
    """Gradient of ½‖conv(p) – target‖² w.r.t. p."""
    n = p.size
    # forward convolution
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

    # difference
    diff = conv - target
    diff_fft = np.fft.rfft(diff, n=m_fft)

    # reversed p for the gradient
    p_rev = p[::-1]
    p_rev_fft = np.fft.rfft(p_rev, n=m_fft)

    # gradient = 2 * (diff * p_rev) (full conv, then take first n entries)
    grad_full = np.fft.irfft(diff_fft * p_rev_fft, n=m_fft)
    grad = 2.0 * grad_full[:n]
    return grad, conv  # also return conv for convenience


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp smooth max.
# ----------------------------------------------------------------------
def _weights_from_conv(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i ∝ exp(α·conv_i)."""
    z = alpha * conv
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    w = exp_z / np.sum(exp_z)
    return w


def _grad_logsumexp(p: np.ndarray, p_fft: np.ndarray,
                    w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Gradient of α·log‑sum‑exp(α·conv(p)).
    The weights w (len 2·n‑1) satisfy Σ w = 1.
    """
    # reverse w
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    # correlation of p with w_rev (full length 3·n‑2, we need first n entries)
    corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)
    idx = 2 * p.size - 2 - np.arange(p.size)
    grad = 2.0 * corr_full[idx]
    return grad


# ----------------------------------------------------------------------
# Objective (the quantity returned by evaluate_sequence when sum=1)
# ----------------------------------------------------------------------
def _objective(p: np.ndarray, m_fft: int) -> float:
    conv = _conv_full(p, m_fft)
    return float(2.0 * p.size * np.max(conv))

            m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

            m1_hat = m1 / (1.0 - beta1 ** (it + 1))
            m2_hat = m2 / (1.0 - beta2 ** (it + 1))

            step = base_lr / (np.sqrt(m2_hat) + eps_adam)
            p = p - step * m1_hat
            p = _simplex_projection(p)

        # (2) Smooth‑max descent (log‑sum‑exp) using Adam
        # --------------------------------------------------------------
        m1 = np.zeros_like(p)
        m2 = np.zeros_like(p)
        base_lr = 0.08

        alpha_mul = 1.001
        alpha_max = 200.0

            w = _weights_from_conv(conv, alpha)

            grad = _grad_logsumexp(p, p_fft, w, m_fft)

            m1 = beta1 * m1 + (1.0 - beta1) * grad
            m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

            m1_hat = m1 / (1.0 - beta1 ** (it + 1))
            m2_hat = m2 / (1.0 - beta2 ** (it + 1))

            lr = base_lr / np.sqrt(alpha)    # shrink as α grows
            step = lr * m1_hat / (np.sqrt(m2_hat) + eps_adam)

            p = p - step
            p = _simplex_projection(p)

            # raise temperature slowly
            alpha = min(alpha * alpha_mul, alpha_max)

            # periodic exact check
            if (it & 1023) == 0:
                cur = float(2.0 * n * np.max(conv))
                _try_update(p, cur, "SmoothMax")
                if best_score <= 1.503:
                    break

        # (3) Exact max sub‑gradient descent (few hundred steps)
        # --------------------------------------------------------------
        step_sz = 0.12
        decay = 0.9999
        min_step = 1e-8
        for it in range(60_000):
            if time.time() >= deadline:
                break
            conv = _conv_full(p, m_fft)
            max_val = conv.max()
            cur = float(2.0 * n * max_val)
            _try_update(p, cur, "SubGrad")
            if best_score <= 1.503:
                break

            # active set (indices attaining the max up to tolerance)
            tol = 1e-13
            active = np.where(conv >= max_val - tol)[0]

            g = np.zeros_like(p)

        # --------------------------------------------------------------

        temp_end = 1e-5

            i, j = rng.integers(0, n, size=2)

            new_p[i] -= eps_move
            new_p[j] += eps_move
            new_score = _objective(new_p, m_fft)

            # accept if better, otherwise with Metropolis probability
            if new_score < cur_score - 1e-15 or rng.random() < np.exp((cur_score - new_score) / temp):
                p_hc = new_p
                cur_score = new_score
                _try_update(p_hc, cur_score, "HillClimb")
                if best_score <= 1.503:
                    break

        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: use a uniform distribution if nothing else succeeded.
    # ------------------------------------------------------------------
    if best_p is None:
        best_p = np.full(n, 1.0 / n)
        best_score = _objective(best_p, m_fft)

    print(f"\nFinal best score: {best_score:.6f}")
    return np.clip(best_p, 0.0, 1000.0).tolist()
