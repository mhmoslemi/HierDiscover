def cross_correlation(w, a):
    L = 1 << ( (3*len(a)-1).bit_length() )
    # Actually need size >= len(w)+len(a)-1 = 3n-2
    L = 1 << ((len(w) + len(a) - 1).bit_length())

    fa = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fw * np.conj(fa), n=L)
    # corr length = L, but valid region indices [0..len(w)+len(a)-2]
    # Grab the relevant region.
    valid = corr[:len(w)+len(a)-1]
    return valid

def compute_gradient_correlation(w, a):
    n = a.shape[0]
    # compute cross-correlation using FFT
    # length needed: len(w)+len(a)-1 = (2n-1)+n-1 = 3n-2
    L = 1 << ((len(w) + len(a) - 1).bit_length())
    fw = np.fft.rfft(w, n=L)

    # cross-correlation = ifft(fw * conj(fa))
    corr = np.fft.irfft(fw * np.conj(fa), n=L)
    # we only need the part of length len(w)+len(a)-1 from start
    corr = corr[:len(w)+len(a)-1]  # length 3n-2

    # gradient g_i = 2 * corr[n-1 - i]  for i in 0..n-1
    # So:
    idx = (n - 1) - np.arange(n)  # indices decreasing
    grad = 2.0 * corr[idx]
    # Ensure gradient size n
    return grad

"""
Propose a non‑negative step‑function (a probability vector) that minimises

    2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The routine consists of

1.  A relatively cheap deterministic sweep over several symmetric parametric
    families (U‑shapes, shifted U‑shapes, product‑distance, exponential,
    cosine, semi‑circle, symmetric Beta) for a few lengths.  The best vector
    from this sweep seeds the stochastic phase.

2.  Stochastic refinement on the probability simplex using a *smooth‑max*
    surrogate for the ‑∞‑norm of the autocorrelation.  The gradient of the
    smooth‑max is obtained by a single FFT‑based cross‑correlation
    (O(n log n)).  We perform an exponentiated‑gradient (mirror‑descent with
    KL‑divergence) step, enforce symmetry (a = (a+rev(a))/2) and project back
    onto the simplex.  A simulated‑annealing acceptance rule together with
    occasional random mass‑transfer moves and a re‑heat strategy helps escape
    local minima.

All helper functions are top‑level, there are no closures or file I/O, and
the search respects a user‑provided time budget (default 1000 s).  The
returned list sums to one; the external evaluator rescales it as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex { w >= 0 , sum(w) = 1 }."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a * a using a real FFT.
    Returns a NumPy array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector normalised to sum=1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Mirror‑descent / exponentiated‑gradient step on the simplex."""
    # Clip the exponent to avoid overflow/underflow.
    exp_term = np.exp(-eta * grad)
    a_new = a * exp_term
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new


# ----------------------------------------------------------------------
# Parametric families (all symmetric)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))^-p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


def semi_circle_distribution(n: int, p: float = 1.0) -> np.ndarray:
    """Symmetric semi‑circle shape ( (1‑x²)^{p/2} ) with x∈[-1,1]."""
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0          # map to [-1, 1]
    base = np.maximum(0.0, 1.0 - x * x)   # 1‑x², clipped at 0
    a = base ** p + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float = 0.5) -> np.ndarray:
    """
    Symmetric Beta(α,α) discretisation:  a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}.
    For α<1 we obtain U‑shapes, for α>1 a hill‑shape.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (alpha - 1.0)) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient utilities (smooth‑max surrogate)
# ----------------------------------------------------------------------


def smoothmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_k = exp(gamma*conv_k) / Σ exp(gamma*conv).
    The subtraction of the maximum ensures numerical stability.
    """
    max_c = conv.max()
    exp_vals = np.exp(gamma * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    return w


def gradient_via_correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  ( Σ_k w_k (a*a)[k] ) = 2 * (w ⋆ a)[i]  (cross‑correlation).
    This routine uses a single FFT and extracts the required n entries.
    """
    n = a.shape[0]
    m = w.shape[0]                     # m = 2n‑1
    # Choose a power‑of‑two length for the FFT.
    L = 1 << ((m + n - 1).bit_length())
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(wf * np.conj(af), n=L)
    # Keep only the valid part of length m+n‑1 = 3n‑2
    corr = corr[: m + n - 1]
    # g_i = 2 * corr[(n‑1)‑i]   for i = 0 … n‑1
    idx = (n - 1) - np.arange(n)
    grad = 2.0 * corr[idx]
    # Numerical noise may produce tiny negative values; clip to zero.
    grad = np.maximum(grad, 0.0)
    return grad

        # ----- occasional random mass‑transfer move (helps escape) -----
        if iteration % 500 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(cand[i] * 0.5, cand[i] * np.random.rand() * 0.5)
            cand[i] -= delta
            cand[j] += delta
            cand = simplex_projection(cand)

        # ----- evaluate candidate -----
        cand_conv = full_convolution_fft(cand)
        cand_score = float(2.0 * n * cand_conv.max())

        # ----- simulated‑annealing acceptance -----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            # we keep the current conv/value for the next iteration
            conv = cand_conv
            cur_score = cand_score

        # ----- schedule updates -----
        temperature *= temp_decay
        eta *= eta_decay
        gamma = min(gamma * gamma_inc, gamma_max)

        # ----- stagnation handling (re‑heat) -----
        if iteration - last_improve > stagnation_limit:
            # tiny random bump + re‑symmetrisation
            a += np.random.rand(n) * 1e-5
            a = (a + a[::-1]) * 0.5
            a = simplex_projection(a)
            temperature = 1.0
            eta = 0.4
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

    # ------------------------------------------------------------------
    # 5. Final sanity check and return
    # ------------------------------------------------------------------
    # (evaluate_sequence is available in the environment; we could call it
    # here to guard against unexpected clipping, but the vector is already
    # non‑negative and sums to one, which is sufficient.)
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
