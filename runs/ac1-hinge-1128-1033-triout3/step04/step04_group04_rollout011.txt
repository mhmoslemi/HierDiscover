"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimises

    2 * n * max_k (a * a)[k]

where * denotes full linear convolution.  The algorithm works in three
phases:

1.  Build a low‑dimensional “basis” (Gaussian kernels) and represent any
    candidate a as a convex combination of these kernels.  This reduces the
    optimisation dimension from O(n) to O(m) (m≈30‑40) while still being able
    to generate a full‑length sequence.

2.  Differential Evolution on the simplex of the basis coefficients.
    Each individual is a probability vector c (∑c=1, c≥0); the full
    sequence a is obtained as a = Σ_i c_i·g_i.  The DE loop runs until the
    time budget is exhausted, constantly keeping the best candidate.

3.  A short projected sub‑gradient polish directly on the basis
    coefficients (gradient of max‑convolution propagated through the basis).

The final sequence is normalised to sum‑1 and returned as a plain Python
list.  All operations respect the 1000 s budget; progress is printed to
stdout.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w≥0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full convolution via real FFT
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution) using a real FFT."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (used for polishing)
# ----------------------------------------------------------------------
def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Parametric families (used only for seed generation)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the given time
    budget (≈1000 s).  Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # a safety margin

    # --------------------------------------------------------------
    # 0. Configuration
    # --------------------------------------------------------------
    n = 15000                     # length of the final sequence (adjustable)
    m = 38                        # number of basis kernels
    sigma = 0.12                  # width of each Gaussian kernel

    # --------------------------------------------------------------
    # 1. Build basis {g_i}_i=0^{m-1} (each g_i ≥0 , sum(g_i)=1)
    # --------------------------------------------------------------
    positions = np.arange(n) / (n - 1)           # in [0,1]
    centers = np.linspace(0.0, 1.0, m)
    G = np.empty((m, n), dtype=np.float64)      # each row = a kernel
    for idx, c in enumerate(centers):
        g = np.exp(-((positions - c) ** 2) / (2.0 * sigma ** 2))
        g = np.maximum(g, 1e-14)
        g /= g.sum()
        G[idx] = g

    # helper: convert basis coefficients to a full‑length vector
    def a_from_c(c_vec: np.ndarray) -> np.ndarray:
        """Linear combination of basis vectors (returns a of length n)."""
        a = c_vec @ G                     # shape (n,)
        # tiny renorm to fight numerical drift
        s = a.sum()
        if s > 0:
            a /= s
        return a

    # helper: exact score for a basis coefficient vector (sum(c)=1 => sum(a)=1)
    def score_from_c(c_vec: np.ndarray) -> float:
        a = a_from_c(c_vec)
        conv = full_convolution_fft(a)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # 2. Initialise DE population on the simplex of coefficients
    # --------------------------------------------------------------
    pop_size = 60
    population = np.empty((pop_size, m), dtype=np.float64)
    pop_scores = np.empty(pop_size, dtype=np.float64)

    # a few curated seeds (uniform, parametric families, external if possible)
    seeds = []

    # uniform
    seeds.append(np.full(m, 1.0 / m))

    # parametric families (U‑min, U‑prod, exponential)
    param_lengths = [n]  # we generate them directly at our target length
    exponents = [0.5, 1.0, 2.0, 3.0]
    lambdas = [1e-3, 5e-3, 1e-2, 3e-2]

    for p in exponents:
        seeds.append(u_min_distribution(n, p))
        seeds.append(u_prod_distribution(n, p))
    for lam in lambdas:
        seeds.append(exp_distribution(n, lam))

    # map each seed (full‑length) onto the basis via a non‑negative least‑squares
    # then project onto the simplex.
    mapped_seeds = []
    for seq in seeds:
        # solve  G.T @ c ≈ seq   (G.T is n×m)
        # use least‑squares; enforce non‑negativity afterwards
        c, *_ = np.linalg.lstsq(G.T, seq, rcond=None)
        c = np.maximum(c, 0.0)
        c = simplex_projection(c)
        mapped_seeds.append(c)

    # external variable (if present)
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            # up‑sample / down‑sample to length n
            old_n = external.shape[0]
            old_pos = np.arange(old_n) / (old_n - 1) if old_n > 1 else np.array([0.0])
            new_pos = positions
            ext_interp = np.interp(new_pos, old_pos, external)
            ext_interp = np.maximum(ext_interp, 0.0)
            ext_interp /= ext_interp.sum() if ext_interp.sum() > 0 else 1.0
            c_ext, *_ = np.linalg.lstsq(G.T, ext_interp, rcond=None)
            c_ext = np.maximum(c_ext, 0.0)
            c_ext = simplex_projection(c_ext)
            mapped_seeds.append(c_ext)
    except Exception:
        pass

    # Fill the initial population: first use the mapped seeds, the rest random.
    for i in range(pop_size):
        if i < len(mapped_seeds):
            population[i] = mapped_seeds[i]
        else:
            rnd = np.random.rand(m)
            population[i] = rnd / rnd.sum()
        pop_scores[i] = score_from_c(population[i])

    # keep global best
    best_idx = int(pop_scores.argmin())
    best_c = population[best_idx].copy()
    best_score = float(pop_scores[best_idx])
    print(f"Initial best score (DE): {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Differential Evolution on the basis coefficients
    # --------------------------------------------------------------
    F = 0.78
    CR = 0.9
    generation = 0

    while time.time() < deadline:
        generation += 1
        for i in range(pop_size):
            # choose three distinct donors different from i
            idxs = np.arange(pop_size)
            idxs = idxs[idxs != i]
            a_idx, b_idx, c_idx = np.random.choice(idxs, 3, replace=False)
            donor = population[a_idx] + F * (population[b_idx] - population[c_idx])
            donor = np.maximum(donor, 0.0)
            donor = simplex_projection(donor)

            # binomial crossover
            mask = np.random.rand(m) < CR
            if not np.any(mask):
                mask[np.random.randint(m)] = True
            trial = np.where(mask, donor, population[i])
            trial = simplex_projection(trial)

            trial_score = score_from_c(trial)

            if trial_score < pop_scores[i]:
                population[i] = trial
                pop_scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_c = trial.copy()
                    print(f"Gen {generation:4d} ind {i:2d}: new best = {best_score:.6f}")

        # occasional early‑exit check
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 4. Low‑dimensional sub‑gradient polishing (on the coefficients)
    # --------------------------------------------------------------
    print("\n=== Sub‑gradient polishing (basis coefficients) start ===")
    cur_c = best_c.copy()
    cur_score = best_score
    step = 0.07
    step_decay = 0.9995
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        a = a_from_c(cur_c)
        conv = full_convolution_fft(a)
        # sub‑gradient w.r.t. a
        grad_a = compute_subgradient(a, conv)
        # propagate to coefficients
        grad_c = grad_a @ G.T          # shape (m,)

        cand_c = cur_c - step * grad_c
        cand_c = np.maximum(cand_c, 0.0)
        cand_c = simplex_projection(cand_c)

        cand_score = score_from_c(cand_c)

        if cand_score < cur_score:
            cur_c = cand_c
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_c = cand_c.copy()
                last_improve = it
                print(f"Subgrad it {it:6d}: new best = {best_score:.6f}")

        step *= step_decay
        if step < 1e-8:
            break
        if it - last_improve > 3000:
            # re‑heat a little to escape shallow plateaus
            bump = np.random.rand(m) * 1e-4
            cur_c = simplex_projection(cur_c + bump)
            cur_score = score_from_c(cur_c)
            step = 0.07
            last_improve = it
            print(f"Subgrad it {it:6d}: re‑heat")

    # --------------------------------------------------------------
    # 5. Produce final sequence and return
    # --------------------------------------------------------------
    final_a = a_from_c(best_c)
    final_a = np.maximum(final_a, 0.0)
    total = final_a.sum()
    if total < 0.01:
        # extremely unlikely, but fallback to uniform
        final_a = np.full(n, 1.0 / n)
    else:
        final_a /= total

    # final verification using the provided evaluate_sequence (optional)
    try:
        final_score = evaluate_sequence(final_a.tolist())
        print(f"\n=== Search finished ===")
        print(f"Best score (internal estimate) : {best_score:.6f}")
        print(f"Score reported by evaluate_sequence: {final_score:.6f}")
    except Exception:
        print("\n=== Search finished ===")
        print(f"Best score (internal estimate) : {best_score:.6f}")

    return final_a.tolist()
