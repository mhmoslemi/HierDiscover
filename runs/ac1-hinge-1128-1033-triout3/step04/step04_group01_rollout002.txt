def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    n = 4000

    # global best track
    best_a = None
    best_val = np.inf

    while time.time() < deadline:
        # initialization: maybe from global best with noise, else random Dirichlet
        if best_a is not None and np.random.rand() < 0.5:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
        else:
            # Dirichlet with alpha=0.3 to encourage diverse shape
            a = np.random.dirichlet(alpha=np.full(n, 0.3))
        # compute convolution
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Init] new global best = {best_val:.6f}")

        # hillclimb with targeted moves
        max_frac = 0.6
        idle_limit = 150000
        idle = 0
        iter_cnt = 0

        while time.time() < deadline:
            # find current max idx
            k_max = int(np.argmax(b))
            # compute contributions for source
            contributions = np.zeros(n)
            # partners for each i:
            low = max(0, k_max - (n-1))
            high = min(n-1, k_max)
            # we can compute contributions for i in [low, high] where partner exists
            partners = k_max - np.arange(low, high+1)
            contributions[low:high+1] = a[low:high+1] * a[partners]
            # choose source
            total_contrib = contributions.sum()
            if total_contrib > 0 and np.random.rand() < 0.9:
                i = np.random.choice(n, p=contributions/total_contrib)
            else:
                i = np.argmax(a)  # fallback

            if a[i] <= 1e-12:
                continue

            # choose target j where contribution to k_max is zero if possible
            if k_max <= n-1:
                # any j > k_max gives partner negative
                if k_max+1 < n:
                    j_candidates = np.arange(k_max+1, n)
                else:
                    j_candidates = None
            else:
                # k_max > n-1
                max_j = k_max - (n-1) - 1
                if max_j >= 0:
                    j_candidates = np.arange(0, max_j+1)
                else:
                    j_candidates = None

            if j_candidates is not None and j_candidates.size > 0:

                # ensure j != i, else pick random

"""
Improved step‑function search.

The algorithm consists of two stages that are repeatedly run until the
time budget expires.

* **Stage 0 – warm‑start** – if a global variable ``height_sequence_1``
  is present it is interpolated to the working size and normalised.
  Otherwise a Dirichlet draw (α=0.3) provides a random start.

* **Stage 1 – targeted hill‑climbing** – the current convolution
  ``b = a * a`` is kept explicitly.  At each iteration the index
  ``k_max`` of the maximal entry of ``b`` is identified.  The move
  tries to *reduce* this entry:
  – a source index ``i`` is sampled proportionally to its contribution
    ``a[i]·a[k_max‑i]`` to the peak;
  – a target index ``j`` is chosen from those positions that do **not**
    contribute to the peak (i.e. where the partner index ``k_max‑j`` is
    out of range).  If no such index exists a weighted choice favours
    positions with a small partner value.
  – a small amount ``δ`` of mass is transferred from ``i`` to ``j``.
    The convolution can be updated exactly in O(n) time using the
    bilinearity of convolution, avoiding a full FFT recomputation.
  The move is accepted only if the exact objective
  ``2·n·max(b)`` strictly improves.  After a long period without
  improvement the current best solution is perturbed by Gaussian noise
  and the process restarts – this gives the algorithm many chances to
  escape local minima.

The whole loop runs until the 1000 s budget (minus a small safety
margin) is exhausted.  The best sequence seen is finally evaluated
with the provided ``evaluate_sequence`` routine and returned as a list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, full length 2·n‑1)
# ----------------------------------------------------------------------
def full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Return the full linear convolution of `x` with itself.
    Length = 2*len(x)-1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def hillclimb_targeted(a_init: np.ndarray,
                       n: int,
                       deadline: float,
                       global_best_a: np.ndarray,
                       global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Perform the targeted hill‑climbing described in the module docstring.
    Returns the best (a, value) found before *deadline*.
    """
    a = a_init.copy()
    b = full_convolution(a)                # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # keep a reference to the global best (may be the same object)
    best_a = global_best_a
    best_val = global_best_val

    # parameters
    max_frac = 0.55          # max fraction of source mass that can be moved
    idle_limit = 200_000     # after this many non‑improving steps we perturb
    idle = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # --------------------------------------------------------------
        # 1) locate the current peak of the convolution
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))                     # 0 … 2·n‑2

        # --------------------------------------------------------------
        # 2) pick a source index i that participates heavily in the peak
        # --------------------------------------------------------------
        contrib = np.zeros(n, dtype=np.float64)

        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx_range = np.arange(low, high + 1, dtype=int)
        partner = k_max - idx_range                    # partner ∈ [0, n‑1]
        contrib[idx_range] = a[idx_range] * a[partner]

        total_contrib = contrib.sum()
        if total_contrib > 0 and np.random.rand() < 0.9:
            i = int(np.random.choice(n, p=contrib / total_contrib))
        else:
            # fallback – pick the currently largest coordinate
            i = int(np.argmax(a))

        if a[i] <= 1e-14:          # nothing to move
            continue

        # --------------------------------------------------------------
        # 3) pick a target index j that does NOT add to the peak
        # --------------------------------------------------------------
        # indices where the partner k_max‑j lies outside [0, n‑1]
        if k_max <= n - 1:
            # any j > k_max gives a negative partner
            if k_max + 1 < n:
                candidates = np.arange(k_max + 1, n, dtype=int)
            else:
                candidates = None
        else:
            # k_max > n‑1  →  j < k_max-(n‑1) gives negative partner
            max_j = k_max - (n - 1) - 1
            if max_j >= 0:
                candidates = np.arange(0, max_j + 1, dtype=int)
            else:
                candidates = None

        if candidates is not None and candidates.size > 0:
            # make sure we do not pick the source index
            if candidates.size == 1 and candidates[0] == i:
                continue
            j = int(np.random.choice(candidates))
        else:
            # no “zero‑partner” positions – choose j with the smallest
            # partner value a[k_max‑j] (inverse weighting)
            partner_all = k_max - np.arange(n, dtype=int)
            out_of_range = (partner_all < 0) | (partner_all >= n)
            weights = np.where(out_of_range,
                               1.0,
                               1.0 / (a[partner_all.clip(0, n - 1)] + 1e-12))
            weights[i] = 0.0                     # never choose the source
            weights_sum = weights.sum()
            if weights_sum == 0:
                continue
            probs = weights / weights_sum
            j = int(np.random.choice(n, p=probs))

        # --------------------------------------------------------------
        # 4) amount of mass to move
        # --------------------------------------------------------------
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------
        # 5) incremental update of the convolution (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # cross‑terms 2·δ·a shifted by the source/target indices
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # --------------------------------------------------------------
        # 6) accept only improving moves
        # --------------------------------------------------------------
        if new_val < cur_val:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            idle = 0
            it += 1

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {it:9d}] new global best = {best_val:.6f}")

        else:
            idle += 1

        # --------------------------------------------------------------
        # 7) occasional large perturbation after stagnation
        # --------------------------------------------------------------
        if idle > idle_limit:
            # start from the best solution seen so far and add noise
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            idle = 0
            print(f"[Hill {it:9d}] stagnation – perturbing")
            # continue with the perturbed state

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The routine respects the 1000 s
    time budget, prints progress information and finally returns the
    best sequence found as a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – we use n = 4000 (a good compromise)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Warm‑start: reuse the previously known construction if available
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # interpolate to the working size
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        # fallback – random Dirichlet (α = 0.3 gives a reasonably
        # varied initial distribution)
        a0 = np.random.dirichlet(alpha=np.full(n, 0.3))

    a0 /= a0.sum()
    best_a = a0.copy()
    # evaluate the initial candidate
    b0 = full_convolution(best_a)
    best_val = 2.0 * n * b0.max()
    print(f"[Init] initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – repeatedly launch the targeted hill‑climb
    # ------------------------------------------------------------------
    # we keep running new restarts until the global deadline is reached
    while time.time() < deadline:
        # give each restart roughly a few seconds; the inner routine
        # stops automatically when the overall deadline expires
        remaining = deadline - time.time()
        if remaining < 5.0:                 # not enough time for another run
            break

        # start from the current global best with a small perturbation,
        # or from a fresh random Dirichlet with 30 % probability
        if np.random.rand() < 0.6 and best_a is not None:
            a_start = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a_start = np.maximum(a_start + noise, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = np.random.dirichlet(alpha=np.full(n, 0.3))

        # run the targeted hill‑climb on this start
        best_a, best_val = hillclimb_targeted(
            a_start,
            n,
            deadline,
            best_a,
            best_val,
        )

    # ------------------------------------------------------------------
    # Final evaluation using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
