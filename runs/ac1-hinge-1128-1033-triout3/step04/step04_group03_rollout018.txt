"""
Propose a candidate step‑function (non‑negative vector) that minimises

        2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm consists of

1.  A fast deterministic sweep over a few symmetric parametric families
    (power‑law, shifted power‑law, product‑law, exponential, cosine) for
    several lengths.  The best vector from this sweep seeds the refinement.

2.  Two‑phase local optimisation on the probability simplex:
    – Phase A: multiplicative (exponential‑gradient / mirror descent) updates
      driven by a sub‑gradient of the maximal convolution entry.
    – Phase B: additive projected‑gradient descent (Euclidean) using the same
      sub‑gradient, which proves much more effective for fine‑grained
      improvements.  Simulated‑annealing acceptance, occasional random
      mass‑transfer moves and a re‑heat strategy are used to avoid stagnation.

All helper functions are top‑level.  The routine respects the given time
budget (default 1000 s) and returns the best sequence found (as a plain
Python list whose entries sum to one; the external evaluator will rescale
it automatically).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex { w >= 0 , sum(w) = 1 }.
    Implementation follows Wang & Carreira‑Perpiñán (O(n log n)).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a * a using a real FFT.
    Returns a NumPy array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector with sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (used by both phases)
# ----------------------------------------------------------------------


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    For each index k attaining (or almost attaining) the maximum, we add
    the contribution 2*a[j] to the gradient entry i where i+j = k.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent / exponentiated‑gradient step on the simplex:
        a' ∝ a * exp(-η * grad)
    The result is exactly normalised to sum = 1.
    """
    # Clip exponent to avoid overflow/underflow
    exp_term = np.exp(-eta * grad)
    a_new = a * exp_term
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new


# ----------------------------------------------------------------------
# Parametric families (all symmetric) – the same families as in the
# previous state‑of‑the‑art code.
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))^-p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Deterministic sweep over families
# ----------------------------------------------------------------------


def deterministic_sweep(lengths, budget_deadline):
    """
    Scan a modest set of families for several lengths.
    Returns the best sequence found (sum == 1) and its score.
    """
    best_seq = None
    best_score = np.inf

    # parameter grids (coarse but sufficient for a quick sweep)
    exponents = np.arange(0.1, 5.01, 0.2)
    shift_cs = [1.0, 2.0, 5.0]
    lam_vals = np.logspace(-4, -0.3, num=20)
    cos_exps = np.arange(0.5, 8.5, 0.5)

    for n in lengths:
        if time.time() > budget_deadline:
            break
        # --- power‑law families ---
        for p in exponents:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

            for c in shift_cs:
                a = u_min_shift_distribution(n, p, c)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"new best (shift) n={n:5d} p={p:4.2f} c={c:.1f}  score={sc:.6f}")

        # --- exponential family ---
        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (exp)   n={n:5d} lam={lam:.5f}  score={sc:.6f}")

        # --- cosine family ---
        for p in cos_exps:
            a = cos_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (cos)   n={n:5d} p={p:3.1f}  score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within a given time
    budget.  Returns the sequence as a plain Python list (sum ≈ 1.0).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over several lengths (≈20 % of the budget)
    # ------------------------------------------------------------------
    sweep_deadline = time.time() + (budget_s * 0.2)
    candidate_lengths = [2000, 3000, 4000, 5000, 5808, 8000]
    best_seq, best_score = deterministic_sweep(candidate_lengths, sweep_deadline)

    # ------------------------------------------------------------------
    # 2. If a previous construction is available, try it as well
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0.0:
                external /= s
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score, best_seq = sc, external.copy()
                    print(f"using external height_sequence_1  score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback: random Dirichlet if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.random.dirichlet(np.ones(n))
        best_score = score_from_vector(best_seq)
        print(f"fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Phase‑A: multiplicative (exponential‑gradient) refinement
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]
    current = np.maximum(current, 0.0)
    current /= current.sum()
    conv_cur = full_convolution_fft(current)
    cur_score = 2.0 * n * conv_cur.max()

    eta_exp = 0.4         # initial step size for exponential update
    eta_exp_decay = 0.9995
    temperature = 1.0
    temp_decay = 0.9996
    iteration = 0
    last_improve = 0

    print("\n=== Phase‑A (exponential‑gradient) start ===")
    while time.time() < deadline and time.time() < deadline * 0.6:
        iteration += 1
        grad = compute_subgradient(current, conv_cur)

        # exponential‑gradient candidate
        cand_exp = exponential_update(current, grad, eta_exp)

        # occasional random mass‑transfer move
        if iteration % 256 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] > 0.0:
                delta = current[i] * np.random.rand() * 0.5
                cand_rand = current.copy()
                cand_rand[i] -= delta
                cand_rand[j] += delta
                cand_rand = np.maximum(cand_rand, 0.0)
                cand_rand = simplex_projection(cand_rand)
                cand = cand_rand
            else:
                cand = cand_exp
        else:
            cand = cand_exp

        conv_cand = full_convolution_fft(cand)
        cand_score = 2.0 * n * conv_cand.max()

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current, conv_cur, cur_score = cand, conv_cand, cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # schedule updates
        eta_exp *= eta_exp_decay
        temperature *= temp_decay

        # re‑heat after long stagnation
        if iteration - last_improve > 200000:
            bump = np.random.rand(n) * 1e-5
            current = np.maximum(current + bump, 0.0)
            current = simplex_projection(current)
            conv_cur = full_convolution_fft(current)
            cur_score = 2.0 * n * conv_cur.max()
            temperature = 1.0
            eta_exp = 0.4
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation (Phase‑A)")

    # ------------------------------------------------------------------
    # 5. Phase‑B: additive projected‑gradient descent (fine tuning)
    # ------------------------------------------------------------------
    eta_add = 0.02
    eta_add_decay = 0.9999
    temperature = 1.0
    temp_decay = 0.9997
    iteration = 0
    last_improve = 0

    print("\n=== Phase‑B (additive gradient) start ===")
    while time.time() < deadline:
        iteration += 1

        conv_cur = full_convolution_fft(current)
        cur_score = 2.0 * n * conv_cur.max()

        # sub‑gradient of the maximal entry
        grad = compute_subgradient(current, conv_cur)

        # additive candidate
        cand_add = current - eta_add * grad
        cand_add = np.maximum(cand_add, 0.0)
        cand_add = simplex_projection(cand_add)

        # exponential candidate (kept for occasional jumps)
        cand_exp = exponential_update(current, grad, eta_exp)

        # evaluate both
        conv_add = full_convolution_fft(cand_add)
        score_add = 2.0 * n * conv_add.max()
        conv_exp = full_convolution_fft(cand_exp)
        score_exp = 2.0 * n * conv_exp.max()

        if score_add < score_exp:
            cand = cand_add
            cand_score = score_add
        else:
            cand = cand_exp
            cand_score = score_exp

        # occasional random transfer move
        if iteration % 256 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] > 0.0:
                delta = current[i] * np.random.rand() * 0.5
                cand_rand = current.copy()
                cand_rand[i] -= delta
                cand_rand[j] += delta
                cand_rand = np.maximum(cand_rand, 0.0)
                cand_rand = simplex_projection(cand_rand)
                conv_rand = full_convolution_fft(cand_rand)
                score_rand = 2.0 * n * conv_rand.max()
                if score_rand < cand_score:
                    cand, cand_score = cand_rand, score_rand

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # schedule updates
        eta_add *= eta_add_decay
        eta_exp *= eta_exp_decay
        temperature *= temp_decay

        # re‑heat on stagnation
        if iteration - last_improve > 150000:
            bump = np.random.rand(n) * 1e-5
            current = np.maximum(current + bump, 0.0)
            current = simplex_projection(current)
            temperature = 1.0
            eta_add = 0.02
            eta_exp = 0.4
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation (Phase‑B)")

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
