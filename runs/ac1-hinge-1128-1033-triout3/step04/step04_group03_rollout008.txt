The smooth‑max (log‑sum‑exp) approximates the maximal convolution entry, providing a dense gradient that lets the optimizer escape the plateaus observed with a pure sub‑gradient.  
A continuation schedule gradually raises the smoothing parameter β so that the algorithm first moves quickly on a smoothed landscape and finally focuses on the true max.  
Exponential‑gradient (mirror‑descent) updates keep the vector non‑negative and automatically enforce the sum‑to‑one constraint.  
Random “mass‑transfer” jumps and a simple reheating rule are added to avoid stagnation.

```python
"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm:

1.  Starts from the best known construction (global variable
    ``height_sequence_1``) or from a few simple random seeds.
2.  Uses a smooth‑max surrogate  fβ(a)= (1/β)·log Σ exp(β·(a*a)[k])
    whose gradient is a weighted sum of the convolution entries.
    For a given β the gradient can be written as a short convolution,
    which we compute with a single real FFT.
3.  Performs an exponentiated‑gradient (mirror‑descent w.r.t. the KL‑divergence)
    update  a ← a·exp(−η·∇fβ)  followed by renormalisation to the simplex.
4.  Increases β during the run (continuation) so that the optimisation
    initially sees a flattened landscape and finally targets the true max.
5.  Occasionally executes a random mass‑transfer move and a reheating
    step when no improvement has been seen for a long time.

All computation respects the given time budget (default 1000 s) and
returns the best vector found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using a real FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    # next power of two for efficient FFT
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n]


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute a*a (full linear convolution) using a real FFT.
    The length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def gradient_smooth_max(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate

        fβ(a) = (1/β) log Σ_k exp(β·conv_k)

    where conv = a*a.  The gradient is

        ∇ fβ_i = 2 * Σ_j a_j * w_{i+j},
        w_k = exp(β·conv_k) / Σ_l exp(β·conv_l)

    This is evaluated by a single FFT‑based convolution.
    """
    max_c = conv.max()
    # stabilise exponentials
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # reversed a needed for the index shift described in the analysis
    a_rev = a[::-1]
    conv_rev = conv_full_fft(a_rev, w)                 # length = 3*len(a)-2
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]            # slice gives Σ_j a_j w_{i+j}
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent step on the probability simplex:
        a' ∝ a * exp( -η * grad )
    The result is renormalised to sum = 1.
    """
    # clip the gradient to avoid overflow in the exponent
    grad_clip = np.clip(grad, -10.0, 10.0)
    exp_term = np.exp(-eta * grad_clip)
    a_new = a * exp_term
    s = a_new.sum()
    if s == 0.0 or np.isnan(s):
        # fallback to uniform distribution
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact score returned by evaluate_sequence for a vector with sum == 1.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisation core
# ----------------------------------------------------------------------


def _optimize_from_start(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine a single starting vector using the smooth‑max continuation.
    Returns the best vector found and its score.
    """
    a = np.maximum(a_start, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / a.sum()

    n = a.shape[0]
    best_seq = a.copy()
    best_score = score_from_vector(a)

    # continuation schedule for the smoothing parameter β
    beta_schedule = [5, 10, 20, 40, 80, 160, 320, 500]
    eta = 0.5                     # initial step size for the exponential update
    eta_decay = 0.9997            # gentle decay across iterations
    iteration = 0
    last_improve = 0

    print(f"  optimisation start – length {n}, initial score {best_score:.6f}")

    for beta in beta_schedule:
        # a modest upper bound on the number of iterations per β
        max_iters_beta = 25000
        for _ in range(max_iters_beta):
            if time.time() > deadline:
                return best_seq, best_score
            iteration += 1

            # --- compute current convolution and gradient ---
            conv = full_convolution_fft(a)
            grad = gradient_smooth_max(a, conv, beta)

            # --- exponential‑gradient update ---
            a = exponential_update(a, grad, eta)
            eta *= eta_decay

            # --- evaluate current score ---
            conv = full_convolution_fft(a)
            cur_score = float(2.0 * n * conv.max())

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = iteration
                print(f"    iter {iteration:7d} β={beta:5.1f}  new best = {best_score:.6f}")

            # --- occasional random mass‑transfer move (helps escape) ---
            if iteration % 512 == 0:
                i, j = np.random.choice(n, 2, replace=False)
                if a[i] > 0.0:
                    delta = a[i] * np.random.rand() * 0.5
                    a_tmp = a.copy()
                    a_tmp[i] -= delta
                    a_tmp[j] += delta
                    a_tmp = np.maximum(a_tmp, 0.0)
                    a_tmp = a_tmp / a_tmp.sum()
                    conv_tmp = full_convolution_fft(a_tmp)
                    tmp_score = float(2.0 * n * conv_tmp.max())
                    if tmp_score < cur_score:
                        a = a_tmp
                        cur_score = tmp_score
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = a.copy()
                            last_improve = iteration
                            print(f"    iter {iteration:7d} mass‑move improved to {best_score:.6f}")

            # --- stagnation handling (re‑heat) ---
            if iteration - last_improve > 150000:
                # inject a tiny random bump and reset step size
                bump = np.random.rand(n) * 1e-5
                a = np.maximum(a + bump, 0.0)
                a = a / a.sum()
                eta = 0.5
                last_improve = iteration
                print(f"    iter {iteration:7d} re‑heat after stagnation")
    return best_seq, best_score
```
