"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a hybrid Adam‑based optimizer.

Main ideas
---------
* Work on the simplex (sum = 1, non‑negative) – we keep the iterate on the
  probability simplex by a Euclidean projection after each step.
* The objective is homogeneous; we therefore normalise sum(a)=1 and
  minimise ``max(conv(a,a))``.
* ``max`` is smoothed with a soft‑max (temperature ε).  In the early
  stage we also minimise the squared deviation of the whole convolution
  from its average value (which pushes the convolution towards a flat
  shape).  A blending factor α increases from 0 to 1, so the optimiser
  first flattens the convolution and then concentrates on the maximal
  entry.
* Adam’s adaptive moments are used for a robust, momentum‑free descent.
* Periodic tiny random perturbations are added when progress stalls,
  and a final pure sub‑gradient polishing stage is performed.
* The whole routine respects the 1000 s budget, keeping the best sequence
  found so far and returning it in the required list format.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (size = len(a)+len(b)-1).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative height sequence that (approximately) minimizes
    the evaluation function given in the problem statement.

    Parameters

    seed : int, optional
        Random seed for reproducibility.

        Total time budget in seconds.  Five seconds are kept as safety margin.

    list[float]

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():

            s = a.sum()
            if s == 0.0:
                a = rng.random(n)
                s = a.sum()
            a = a / s
        except Exception:

            a = None
    else:
        n = 5000
        a = None

    # If we could not obtain a warm start, build a random simplex start.
    if a is None:

        a = a / a.sum()

    # ------------------------------------------------------------------
    # Adam hyper‑parameters.

    beta1 = 0.9
    beta2 = 0.999

    # about 200 k iterations.
    alpha_ramp = 200_000.0

    # Adam moment buffers.
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Target average of the convolution (sum(a)=1 ⇒ total mass =1).
    target = 1.0 / (2 * n - 1)

    # ------------------------------------------------------------------
    # Initial evaluation.
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    max_conv = float(conv.max())
    eps = max(max_conv * 0.5, 1e-12)

        score = 2.0 * n * max_conv

        alpha = min(1.0, it / alpha_ramp)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a = _project_simplex(a_new)

        # ----- schedules -----
        eps = max(eps * eps_decay, eps_min)
        lr = max(lr_min, lr * lr_decay)

            stagn = 0

    # ------------------------------------------------------------------
    a = best_seq.copy()

        mx = conv.max()
        # indices where the maximum is attained (within tolerance)
        idxs = np.flatnonzero(conv >= mx - 1e-15)
        idx = int(idxs[0])                     # deterministic pick
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j]                # gradient of the single max entry
        a = _project_simplex(a - 2e-6 * g)

    # ------------------------------------------------------------------
    # Light random local search around the polished solution.
    # ------------------------------------------------------------------
    while time.time() < deadline:

        conv_cand = _full_convolution_fft(cand, cand)

            best_seq = cand.copy()
            a = cand.copy()

    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
