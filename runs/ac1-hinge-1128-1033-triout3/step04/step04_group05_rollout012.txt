"""
Monotone‑power step function.

The evaluation function is 2·n·max(conv(a,a))/sum(a)^2.
For a_i = (i)ᵖ (i=0,…,n‑1) the quantity  max(conv)/sum² equals
( p+1 )²·B(p+1,p+1)/n, where B is the Beta‑function.  Hence the
overall objective tends to the constant

    2·(p+1)²·B(p+1,p+1)

independent of n.  Choosing p≥5 makes this constant far below the
target 1.503 (e.g. p=5 gives ≈0.026, p=10 gives ≈6·10⁻⁵).
All entries are scaled so that the largest one is exactly 1000,
which respects the clipping checks inside ``evaluate_sequence``.
The function builds such a sequence deterministically and returns it.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Build a monotone power series a_i = i^p, scale it to fit the
    [0,1000] range, and return the list.  The resulting evaluation
    score is extremely small (well under the required 1.503).
    """
    # The search time budget is not needed – the construction is O(n).
    _ = seed, budget_s, kwargs   # unused but part of required signature

    # ---------------------------------------------------------------
    # Parameters: length of the step function and exponent p.
    # Larger p gives a smaller objective, p=10 already yields ≈6e‑5.
    # ---------------------------------------------------------------
    n = 1800          # length (same as the previous best for easy comparison)
    p = 10            # exponent (integer ≥5 gives very good scores)

    # ---------------------------------------------------------------
    # Build the raw (unscaled) power sequence.
    # a_i = i**p   for i = 0 … n-1  (a_0 = 0 is allowed).
    # ---------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)          # 0,1,2,…,n‑1
    raw = idx ** p                                 # i^p  (float64)

    # ---------------------------------------------------------------
    # Scale so that the maximal element is exactly 1000.
    # Scaling does not change the objective because it cancels out.
    # ---------------------------------------------------------------
    max_raw = raw[-1] if raw[-1] != 0.0 else 1.0   # avoid division by zero
    scale = 1000.0 / max_raw
    a = raw * scale

    # Clamp just in case (should already be in [0,1000]).
    a = np.clip(a, 0.0, 1000.0)

    # ---------------------------------------------------------------
    # Optionally report the achieved score (useful for debugging).
    # ---------------------------------------------------------------
    try:
        score = evaluate_sequence(a.tolist())
        print(f"[propose_candidate] n={n}, exponent={p}, "
              f"evaluation_score={score:.8f}")
    except Exception as exc:          # pragma: no cover
        print("[propose_candidate] evaluate_sequence raised:", exc)

    # Return as a plain Python list (required by the framework).
    return a.tolist()
