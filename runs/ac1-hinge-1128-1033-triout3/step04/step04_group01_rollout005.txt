"""
Fast targeted hill‑climbing for the max‑convolution objective.

The algorithm works in three stages:

1. **Initialisation** – load the previously best known sequence
   (``height_sequence_1``) if it is available, otherwise fall back to a
   uniform distribution.  The vector is normalised to sum = 1.

2. **Product‑guided local search** – repeatedly move a small amount of
   mass from a “source’’ index that contributes most to the current
   maximal convolution entry to a “target’’ index that contributes the
   least.  The move is accepted only if the exact objective
   ``2·n·max(conv(a,a))`` strictly improves.  Stagnation triggers a
   cheap random perturbation of the best solution seen so far.

3. **Optional final smoothing** – a short Adam optimisation on a
   smooth‐max surrogate (few seconds) is used to polish the final
   candidate.

All operations are performed in O(n) time per iteration (n ≈ 4000)
using an incremental update of the convolution, which makes the
search fast enough to explore many millions of moves within the
1000 s time budget.

The function ``propose_candidate`` returns the best sequence of
non‑negative heights found before the deadline.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper: full convolution of a vector with itself (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full convolution ``x * x`` (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Phase A – a very short Adam optimisation (optional polish)
# ----------------------------------------------------------------------
def _adam_refine(a0: np.ndarray,
                 n: int,
                 deadline: float) -> np.ndarray:
    """
    Perform a few Adam steps on a smooth‑max surrogate.
    This is only a polishing step; it runs for at most a few seconds.
    """
    # --- hyper‑parameters (tiny budget) ---
    beta1 = 0.9
    beta2 = 0.999
    lr_start = 0.02
    lr_end = 0.001
    eps_start = 1e-3
    eps_end = 1e-8

    w = np.log(a0 + 1e-12)                     # soft‑max parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- reconstruct a from w ----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ---- evaluate exact objective (for tracking only) ----
        conv_aa = _full_convolution(a)
        cur_val = 2.0 * n * conv_aa.max()

        # ---- smooth‑max parameters ----
        progress = (now - (deadline - 5.0)) / 5.0   # linearly shrink over ~5 s
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # ---- smoothed weights over convolution entries ----
        max_c = conv_aa.max()
        shifted = (conv_aa - max_c) / eps
        wgt = np.exp(shifted - np.max(shifted))    # numeric stability
        p = wgt / wgt.sum()                         # soft‑max over indices

        # ---- gradient of the smooth surrogate w.r.t. a ----
        p_rev = p[::-1]
        # convolution a * p_rev (full length = 3·n‑2)
        size = 1 << ((3 * n - 2) - 1).bit_length()
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p_rev, size)
        conv_ab = np.fft.irfft(fa * fp, size)[:3 * n - 2]
        segment = conv_ab[n - 1: 2 * n - 1]        # middle n entries
        grad_a = 4.0 * n * segment[::-1]

        # ---- chain rule through the soft‑max ----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---- Adam update on the parametrisation ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter to avoid exact plateaus
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.005, size=n)

    # final projection back onto the simplex
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase B – product‑guided hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _product_hillclimb(a_start: np.ndarray,
                       n: int,
                       deadline: float) -> np.ndarray:
    """
    Run a deterministic local search that moves mass from the index
    contributing most to the current maximal convolution entry to the
    index that contributes the least.
    """
    a = a_start.copy()
    b = _full_convolution(a)               # current full convolution
    cur_val = 2.0 * n * b.max()

    best_a = a.copy()
    best_val = cur_val

    max_frac = 0.5                         # up‑to this fraction of source mass may be moved
    stagnation = 0
    IMPROVE_LIMIT = 150_000                # after many idle steps we perturb

    iteration = 0
    # to keep numeric drift under control we recompute b from scratch
    # every so often
    RECOMPUTE_EVERY = 5000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- identify the current maximal convolution index ----
        k_max = int(b.argmax())

        # ---- pick a source index i that contributes most to b[k_max] ----
        if np.random.rand() < 0.15:                      # occasional random source
            i = np.random.choice(n, p=a)
        else:
            # vectorised computation of a_i * a_{k_max-i}
            if k_max < n:
                idx = np.arange(k_max + 1)
                partner = k_max - idx
            else:
                idx = np.arange(k_max - (n - 1), n)
                partner = k_max - idx
            prod = a[idx] * a[partner]
            i = idx[np.argmax(prod)]

        # ---- pick a target index j with minimal impact on the same entry ----
        # Prefer indices where a is (almost) zero.
        zero_mask = a < 1e-12
        if zero_mask.any() and np.random.rand() < 0.6:
            # pick a random zero entry (different from i)
            zeros = np.flatnonzero(zero_mask)
            j = np.random.choice(zeros)
            if j == i:
                # fallback to a normal candidate
                j = np.argmin(a)
        else:
            # minimise the partner term a_{k_max-j}
            mask = (k_max - np.arange(n) >= 0) & (k_max - np.arange(n) < n)
            partner_val = np.where(mask, a[k_max - np.arange(n)], 0.0)
            # the safest target is the one with smallest partner value;
            # break ties by smallest a[j]
            min_partner = partner_val.min()
            cand = np.flatnonzero(partner_val == min_partner)
            if cand.size > 1:
                # among candidates choose the one with smallest a
                j = cand[np.argmin(a[cand])]
            else:
                j = cand[0]
            if j == i:
                # if we happen to pick the source, pick the global minimum a
                j = np.argmin(a)

        # ---- determine the transfer amount ----
        if a[i] <= 0.0:
            continue
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n)) ----
        b_new = b.copy()
        # contributions from the added mass at j
        b_new[j:j + n] += 2.0 * delta * a
        # contributions from the removed mass at i
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        # ---- evaluate the new objective ----
        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ---- accept if we have an improvement ----
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            # small negative numerical artefacts are clipped
            if a[i] < 0.0:
                a[i] = 0.0
            b = b_new
            cur_val = new_val
            stagnation = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")

        else:
            stagnation += 1

        # ---- periodic recomputation to avoid drift ----
        if iteration % RECOMPUTE_EVERY == 0:
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        # ---- stagnation handling – cheap perturbation ----
        if stagnation > IMPROVE_LIMIT:
            a = best_a.copy()
            # add modest Gaussian noise, then renormalise
            a += np.random.normal(scale=0.005, size=n)
            a = np.maximum(a, 0.0)
            total = a.sum()
            if total == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= total
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            stagnation = 0
            print(f"[Hill {iteration:7d}] stagnation – perturbing")

        iteration += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a step‑function (non‑negative heights) under a fixed time budget.
    The function returns a list of floats whose sum is 1 and which
    minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the wrapper that calls us
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand points give a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise the candidate.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is a global that may be supplied by the runner.
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # linearly interpolate to the desired size
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        # fallback: uniform distribution
        a0 = np.full(n, 1.0 / n)

    a0_sum = a0.sum()
    if a0_sum == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 /= a0_sum

    # ------------------------------------------------------------------
    # Phase B – product‑guided hill‑climbing (the bulk of the budget)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0               # keep ~5 s for optional polishing
    best_a = _product_hillclimb(a0, n, hill_deadline)

    # ------------------------------------------------------------------
    # Optional tiny Adam polishing (≈ 5 s) – can squeeze a little extra.
    # ------------------------------------------------------------------
    final_deadline = deadline
    if time.time() < final_deadline:
        best_a = _adam_refine(best_a, n, final_deadline)

    # ------------------------------------------------------------------
    # Final evaluation – use the provided secure wrapper.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
