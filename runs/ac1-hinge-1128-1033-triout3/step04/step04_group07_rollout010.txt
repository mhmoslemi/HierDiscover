def _beta_shape(n, a, b):
    i = np.arange(n, dtype=np.float64)
    # use i+0.5 for half-step increments to avoid zeros.
    shape = ((i + 0.5) ** (a - 1.0)) * ((n - 0.5 - i) ** (b - 1.0))
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape = shape / total
    return shape

    2·n·max(conv_self) / (sum(a)**2)

# --------------------------------------------------------------
# Simplex projection (Euclidean)
# --------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Full self‑convolution of p using a pre‑computed FFT size."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]

def _gradient_from_weights(p: np.ndarray, p_fft: np.ndarray,
                          w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smoothed max.
    w – soft‑max weights (length 2·n‑1) that sum to 1.

    idx = 2 * p.size - 2 - np.arange(p.size)
    grad = 2.0 * corr[idx]
    return grad

    return float(2.0 * n * np.max(conv))

# --------------------------------------------------------------

                   tol: float = 1e-13) -> np.ndarray:
    """Return the sub‑gradient of max(conv_self(p)) at p."""
    max_val = conv.max()
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(p)
    for k in active:

            start = k - n + 1
            g[start:] += 2.0 * p[n - 1 : start - 1 : -1]
    if active.size:
        g /= float(active.size)
    return g

    if s == 0:
