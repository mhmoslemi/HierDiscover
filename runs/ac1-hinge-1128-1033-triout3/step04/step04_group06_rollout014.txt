def propose_candidate(seed=..., budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time=time.time()
    deadline = start_time+budget_s-2.0

    # Choose n based on time / memory. Use n=2500 maybe.

    try: get global variable height_sequence_1 else start new.

    Ensure a is probability simplex (sum 1) and non-negative.

    best_a = a.copy(); best_val = evaluate_sequence(a.tolist()); (but evaluate function includes scaling to 0-1000; fine.)

    Use Adam parameters: lr_start = 0.2, lr_end = 0.001 (annealed). β1=0.9, β2=0.999, eps=1e-8.

    Momentum m, v arrays zeros.

    Set initial temperature factor tau_factor_start = 0.5, tau_factor_end = 1e-5.

    iteration=0

    while time < deadline:

        # Compute conv and obj
        conv = full_convolution_fft(a, a)
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val; best_a = a.copy(); print update.

        # progress = (time - start)/total_time
        progress = (time.time() - start_time) / (deadline - start_time)

        # Temperature
        tau = (tau_factor_start*(1-progress)+tau_factor_end*progress) * M
        # no zero handling (if tau<=0, set w one-hot at max).
        if tau <= 0 or np.isnan(tau) or np.isinf(tau):
            w = np.zeros_like(conv); w[np.argmax(conv)] = 1.0
        else:
            # compute softmax stable
            exp_args = (conv - M) / tau
            np.clip(exp_args, -80, 80, out=exp_args)
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv); w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp

        # Compute gradient g = _gradient_from_weights(a, w)
        g = _gradient_from_weights(a, w)   # shape (n,)
        # Optionally scaling factor: since softmax approximates max, gradient magnitude may be small; we rely on step size

        # Adam step
        lr = lr_start * (1 - progress) + lr_end * progress
        m = β1 * m + (1 - β1) * g
        v = β2 * v + (1 - β2) * (g * g)
        m_hat = m / (1 - β1**(iteration+1))
        v_hat = v / (1 - β2**(iteration+1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Occasionally random jitter: if rand < p_jitter
        if np.random.rand() < 0.01:
            src, dst = np.random.randint(n), np.random.randint(n)
            if src != dst:
                delta = np.random.rand() * a[src] * 0.5
                a[src] -= delta
                a[dst] += delta

        # Project onto simplex: ensure non-negative then renormalize
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum

        iteration +=1

        # occasional recompute conv for sanity
        if iteration % 5000 == 0:
            conv = full_convolution_fft(a, a)   # ignore

    final = evaluate_sequence(best_a.tolist())
    print final result.

    return best_a.tolist()

    """Project vector v onto probability simplex."""
    # (non-negative) sum to 1
    # Using algorithm O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]

    w = np.maximum(v - theta, 0)

"""
Propose a step‑function sequence that (approximately) minimizes

    2 * n * max(convolve(a, a)) / (sum(a)**2)

where `a` is non‑negative and lives on the probability simplex
(sum‑to‑one).  The algorithm is a projected Adam optimizer on a
soft‑max (log‑sum‑exp) approximation of the max‑convolution.  It
starts from a V‑shaped distribution (large values at the ends,
small values in the centre) which is known to flatten the
auto‑convolution.  The temperature τ is annealed so that the
objective gradually switches from a smooth surrogate to the true
∞‑norm.  Random jitter and occasional restarts help escape plateaus.
The best sequence seen before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution via real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of `x` and `y` using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate of max(a*a)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Return ∇_a  Σ_k w_k (a*a)[k] .
    Implements  g_i = 2· Σ_k w_k a_{k-i}  via a single convolution.
    """
    n = a.shape[0]
    u = w[::-1]                                 # reverse the weight vector
    conv = _full_convolution_fft(u, a)          # length = 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)                  # positions that correspond to grad entries
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to unit L1‑norm."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

    print(f"[init] n={n}, initial evaluation={best_val:.6f}")

    # Adam state
    m = np.zeros_like(a)                     # first moment
    v = np.zeros_like(a)                     # second moment
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-12

    # learning‑rate schedule (additive Adam steps)
    eta_start = 0.10
    eta_end   = 0.001

    # temperature schedule (soft‑max approximation of max)
    tau_factor_start = 0.30                 # τ ≈ 0.30·max at the beginning
    tau_factor_end   = 1e-6                 # essentially a hard max at the end

    iteration = 0
    last_improve_iter = 0
    stagnation_limit = 20000                # restart if no improvement for this many iters
    jitter_prob = 0.015                     # occasional random mass transfer

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- current convolution and objective value ----
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)

        # keep track of the best sequence ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve_iter = iteration
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # ---- progress dependent schedules ----
        progress = (now - start_time) / (deadline - start_time)   # ∈[0,1]

        # temperature for the soft‑max
        tau = (tau_factor_start * (1.0 - progress) +
               tau_factor_end   * progress) * M
        if tau <= 0 or not np.isfinite(tau):
            # fall back to a hard sub‑gradient (one‑hot at the max)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # stable soft‑max: shift by the maximum to avoid overflow
            exp_args = (conv - M) / tau
            np.clip(exp_args, -80.0, 80.0, out=exp_args)
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp

        # ---- gradient of the smooth surrogate ----
        g = _gradient_from_weights(a, w)          # shape (n,)

        # ---- Adam update (additive) ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        eta = eta_start * (1.0 - progress) + eta_end * progress
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional random jitter (helps to jump out of plateaus) ----
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a[src] * 0.5
            a[src] -= delta
            a[dst] += delta

        # ---- projection back onto the simplex ----
        a = _project_simplex(a)

        iteration += 1

        # ---- stagnation handling: occasional restart around the best known ----
        if iteration - last_improve_iter > stagnation_limit:
            # perturb the best solution slightly and continue
            noise = np.random.normal(scale=0.02, size=n)
            a = best_a + noise
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = iteration
            print(f"[{iteration:7d}] stagnation restart")

        # occasional sanity‑check (no functional impact)
        if iteration % 10000 == 0:
            _ = _full_convolution_fft(a, a)   # keep the FFT plans warm

    # --------------------------------------------------------------
    # Final reporting (the outer framework will also evaluate)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
