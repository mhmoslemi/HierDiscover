def convolve_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()   # next pow of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx*fy, n=size)
    return conv[:n]

def convolve_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << (n-1).bit_length()
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n]

"""
Propose a step‑function (non‑negative vector) minimising

    2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm consists of

1.  A fast deterministic sweep over several symmetric parametric families
    (U‑shapes, shifted U‑shapes, product‑distance, exponential, cosine,
    Gaussian, arcsine‑type).  The best vector from this sweep seeds the
    optimisation phase.

2.  Optional resampling to larger lengths (the objective improves for longer
    sequences).

3.  Stochastic refinement on the probability simplex:
        – a sub‑gradient step (exponential‑gradient / mirror descent)
        – a smoothed‑max (soft‑max) gradient descent step
        – simulated‑annealing acceptance and occasional random mass‑transfer
          moves.
    Parameters (step sizes, temperature, smoothing factor) decay slowly and
    a re‑heat mechanism avoids long stagnation.

The routine respects the given time budget (default 1000 s) and returns the
best sequence found as a plain Python list (its L1‑norm is 1, the external
evaluator will rescale it as required).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (real FFT, result length 2*len(a)-1)."""
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n]

# ----------------------------------------------------------------------
# Scoring (exact for vectors with sum == 1)
# ----------------------------------------------------------------------
def score_from_vector(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Sub‑gradient of the (hard) max‑convolution
# ----------------------------------------------------------------------
def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

# ----------------------------------------------------------------------
# Exponential‑gradient (mirror descent) update
# ----------------------------------------------------------------------
def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """a' ∝ a * exp(-η * grad)  with exact renormalisation."""
    exp_term = np.exp(-eta * grad)
    a_new = a * exp_term
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new

# ----------------------------------------------------------------------
# Smoothed‑max gradient (soft‑max) and a gradient‑descent step
# ----------------------------------------------------------------------
def softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv) w.r.t. a.
    Returns the vector of size len(a).
    """
    # stabilised soft‑max weights
    max_val = conv.max()
    w = np.exp(alpha * (conv - max_val))
    w = w / w.sum()

    # convolution of weights with reversed a (see derivation in analysis)
    a_rev = a[::-1]
    conv_wa = convolve_fft(w, a_rev)          # length = 3*len(a)-2
    start = len(a) - 1
    grad = 2.0 * conv_wa[start:start + len(a)]
    return grad


def gradient_descent_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Plain projected gradient descent (step opposite to the gradient)."""
    a_new = a - eta * grad
    a_new = simplex_projection(a_new)
    return a_new

# ----------------------------------------------------------------------
# Parametric families (all symmetric)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


def gaussian_distribution(n: int, sigma_frac: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    sigma = sigma_frac * (n - 1)
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2) + 1e-14
    a /= a.sum()
    return a


def arcsine_distribution(n: int, p: float) -> np.ndarray:
    """Product‑distance power‑law (p=0.5 gives the arcsine law)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a

# ----------------------------------------------------------------------

    old_n = v.shape[0]

    x_new = np.linspace(0.0, 1.0, new_n)
    v_new = np.interp(x_new, x_old, v)
    v_new = np.maximum(v_new, 0.0)
    v_new /= v_new.sum()
    return v_new

                cand_rand_conv = full_convolution_fft(cand_rand)
                cand_score = 2.0 * n * cand_conv.max()
                cand_rand_score = 2.0 * n * cand_rand_conv.max()
                if cand_rand_score < cand_score:
                    cand = cand_rand
                    cand_score = cand_rand_score
                else:
                    cand_score = cand_score
            else:
                cand_conv = full_convolution_fft(cand)
                cand_score = 2.0 * n * cand_conv.max()
        else:
            cand_conv = full_convolution_fft(cand)
            cand_score = 2.0 * n * cand_conv.max()

        # ------------------------------------------------------------------
        # 5b. Simulated‑annealing acceptance
        # ------------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_conv = cand_conv
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best global score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 5c. Schedule updates
        # ------------------------------------------------------------------
        temperature *= temp_decay
        eta_EG *= eta_EG_decay
        eta_GD *= eta_GD_decay
        alpha *= alpha_inc

        # ------------------------------------------------------------------
        # 5d. Re‑heat if stagnated for a long time
        # ------------------------------------------------------------------
        if iteration - last_improve > stagnation_limit:
            bump = np.random.rand(n) * 1e-5
            current = np.maximum(current + bump, 0.0)
            current /= current.sum()
            current_conv = full_convolution_fft(current)
            current_score = 2.0 * n * current_conv.max()
            temperature = 1.0
            eta_EG = 0.5
            eta_GD = 0.05
            alpha = 5.0
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

    # ------------------------------------------------------------------
    # 6. Return best sequence found
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
