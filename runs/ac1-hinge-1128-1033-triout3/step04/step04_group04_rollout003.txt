After each update we recompute the convolution (via FFT), the active set,
the gradient and keep track of the best vector seen.  If the optimiser
stalls we add a tiny random perturbation (“re‑heat”).  The routine runs
until the allotted time budget (≈ 1000 s) expires and returns the best
vector found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution using real FFT (returns length 2n-1)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 2 * n                     # zero‑padding length (power of two not required)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
# Helper: compute the objective (the value returned by evaluate_sequence)
# ----------------------------------------------------------------------
def score_vector(a: np.ndarray) -> float:
    """Assumes sum(a) == 1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Parametric families (used for alternative seeds)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a

def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Projection: softmax (guarantees a_i ≥ 0 and ∑a_i = 1)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

# ----------------------------------------------------------------------
# Gradient of max‑convolution w.r.t. the log‑parameters
# ----------------------------------------------------------------------
def grad_log_params(x: np.ndarray, a: np.ndarray, conv: np.ndarray,
                    max_b: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Returns ∂/∂x ( max_k (a*a)[k] ) for a = softmax(x).
    """
    n = a.shape[0]
    G = np.zeros_like(a)

    # Accumulate contributions for each active convolution index
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]

    # Sub‑gradient for the log‑parameters (see derivation in the docstring)
    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad

# ----------------------------------------------------------------------
# Adam optimiser on the log‑probabilities.
# ----------------------------------------------------------------------
def run_adam(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Runs Adam on the log‑parameters starting from `initial_a` for at most
    `time_limit` seconds.  Returns the best vector found and its score.
    """
    start = time.time()
    n = initial_a.shape[0]

    # initialise log‑parameters (avoid log(0))
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.2          # base learning rate in log‑space

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    while time.time() - start < time_limit - 0.5:
        iteration += 1

        # ------- softmax -------
        a = softmax(x)

        # ------- convolution -------
        conv = full_convolution_fft(a)
        max_b = conv.max()

        # active set of (near‑)maximal indices
        tol = max_b * 1e-12 + 1e-15
        active = np.where(conv >= max_b - tol)[0]

        # ------- gradient -------
        grad = grad_log_params(x, a, conv, max_b, active)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # a simple decay of the stepsize (helps stability for long runs)
        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- bookkeeping -------
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")

        # ------- re‑heat if stuck -------
        if iteration - last_improve > 5000:
            # add a tiny random nudge and reset the moments
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Iter {iteration:6d}: re‑heat (no improvement)")

    return best_vec, best_score

# ----------------------------------------------------------------------
# Main entry point required by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Searches for a non‑negative step‑function that minimises the evaluation
    function.  The routine works within the supplied time budget, prints
    progress and finally returns the best vector (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # 5 s safety margin

    # --------------------------------------------------------------
    # 1. Acquire an initial seed (use the global construction if present)
    # --------------------------------------------------------------
    n = 5000                     # fallback length
    initial_vec = None

    # try to use the previously found construction if it exists
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        s = external.sum()
        if s > 0.0:
            external /= s
            n = external.shape[0]
            initial_vec = external
            print(f"Using external seed of length {n}")
    except NameError:
        pass

    # if the external seed is unavailable, run a tiny deterministic sweep
    if initial_vec is None:
        # pick a modest length that fits easily into the remaining budget
        n = 5000
        candidates = []
        for p in [0.7, 1.0, 1.5]:
            candidates.append(u_min_distribution(n, p))
            candidates.append(u_prod_distribution(n, p))
        for lam in [5e-3, 1e-2, 5e-2]:
            candidates.append(exp_distribution(n, lam))
        # also add a uniform vector
        candidates.append(np.ones(n) / n)

        best_score = np.inf
        for a in candidates:
            sc = score_vector(a)
            if sc < best_score:
                best_score = sc
                initial_vec = a
        print(f"Deterministic seed selected (length {n}, score {best_score:.6f})")

    # --------------------------------------------------------------
    # 2. Prepare a small portfolio of starting points
    # --------------------------------------------------------------
    seeds = [initial_vec]

    # a few random Dirichlet draws for diversification
    for _ in range(3):
        r = np.random.rand(n)
        seeds.append(r / r.sum())

    # a couple of handcrafted shapes (different exponents)
    for p in [0.4, 2.0]:
        seeds.append(u_min_distribution(n, p))
        seeds.append(u_prod_distribution(n, p))
    for lam in [2e-3, 2e-2]:
        seeds.append(exp_distribution(n, lam))

    # --------------------------------------------------------------
    # 3. Run Adam optimisation for each seed, partitioning the budget
    # --------------------------------------------------------------
    best_overall_vec = None
    best_overall_score = np.inf

    remaining_time = deadline - time.time()
    per_seed_budget = remaining_time / len(seeds)

    for idx, seed_vec in enumerate(seeds):
        # ensure the seed is a proper probability vector
        a = np.maximum(seed_vec, 0.0)
        if a.sum() == 0.0:
            a = np.ones_like(a)
        a = a / a.sum()

        print(f"\n=== Optimisation start: seed {idx+1}/{len(seeds)} (budget {per_seed_budget:.1f}s) ===")
        vec, sc = run_adam(a, per_seed_budget)

        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_vec = vec.copy()
            print(f"*** New global best from seed {idx+1}: score = {best_overall_score:.6f}")

        # update remaining time for the next seed
        remaining_time = deadline - time.time()
        if remaining_time <= 2.0:
            break
        per_seed_budget = remaining_time / max(1, len(seeds) - idx - 1)

    # --------------------------------------------------------------
    # 4. Final report
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_overall_score:.6f} (length {len(best_overall_vec)})")
    return best_overall_vec.tolist()
