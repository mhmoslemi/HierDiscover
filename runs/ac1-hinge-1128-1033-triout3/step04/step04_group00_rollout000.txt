The function `propose_candidate` searches for a non‑negative sequence `a` that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}.
\]

Because the objective is homogeneous we may safely normalise the vector to
\(\sum a = 1\) and simply minimise `2·n·max(conv(a,a))`.  
The search consists of three stages

1. **Randomised initialisations** – a mixture of a few classic “nice” shapes
   (Catalan, sine, semi‑circle, Gaussian, triangle) plus a few pure random
   vectors.  The best of those candidates is kept as a warm start.

2. **Projected Adam optimisation with a soft‑max approximation of the
   `max`** –  
   * the convolution is computed with a real FFT (full linear convolution).  
   * a temperature `ε` produces a smooth weight `w` over the convolution
     entries `w_i ∝ exp((c_i−c_max)/ε)`.  
   * the gradient of the weighted sum is a cross‑correlation `g = 2·rev(a)*w`.  
   * Adam’s adaptive step is taken and the iterate is projected onto the
     probability simplex (`a_i≥0, ∑a_i=1`).  
   * `ε` and the learning‑rate are annealed; after long periods without
     improvement a small Gaussian perturbation is added to escape plateaus.

3. **Exact max‑entry polishing** – a short sub‑gradient loop that uses the true
   max‑entry gradient (instead of the soft‑max) for a final fine‑tuning.

The whole procedure respects the 1000 s budget, repeatedly checks the remaining
time and stops early if necessary.  The best sequence encountered is returned
after clipping to the limits required by `evaluate_sequence`.

"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

by a multi‑restart projected Adam optimiser with a soft‑max smoothing of the
max‑convolution, followed by an exact sub‑gradient polishing phase.
"""

import time

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution via FFT (full linear convolution).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Base shapes used for random initialisation.
# ----------------------------------------------------------------------
def _initial_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random convex combination of a few smooth base shapes."""
    x = np.arange(n, dtype=np.float64)

    # Catalan‑type decreasing sequence
    catalan = np.empty(n, dtype=np.float64)
    cur = 1.0
    catalan[0] = cur
    for k in range(1, n):
        cur = cur * (2.0 * k - 1) / (2.0 * k)
        catalan[k] = cur

    # Sine (positive on whole interval)
    sine = np.sin(np.pi * (x + 0.5) / n)

    # Semi‑circle (zero at the ends)
    t = 2.0 * x / (n - 1) - 1.0
    semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

    # Gaussian centred in the middle
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

    # Triangular tent
    triangle = np.minimum(x, n - 1 - x) + 1.0

    bases = [catalan, sine, semicircle, gauss, triangle]
    weights = rng.random(len(bases))
    weights /= weights.sum()
    a = sum(w * b for w, b in zip(weights, bases))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

                v.fill(0.0)
                stagn = 0
                lr = max(lr, 0.5)   # give a slight step‑size boost
                print(f"[restart {restart_id:02d}] perturb after stagnation")

            # ----- very occasional re‑normalisation (numerical safety) ---------------
            if t % 5000 == 0:
                a = a / a.sum()

            # ----- early exit for this restart if we have spent too much time -----
            # NOTE: the outer while will break as soon as the global deadline is hit.

        # end of one restart – continue with another one if time permits

    # --------------------------------------------------------------
    # Phase 2 – exact max‑entry sub‑gradient polishing
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        best_seq = np.full(n, 1.0 / n)

    a = best_seq.copy()
    for it in range(2500):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        # indices achieving the maximum (within numerical tolerance)
        idxs = np.flatnonzero(conv >= mx - 1e-15)
        idx = int(idxs[0])               # take the first one

        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j]

        # tiny projected step
        a = _project_simplex(a - 5e-6 * g)

        if (it + 1) % 500 == 0:
            # tiny progress report
            cur_val = 2.0 * n * np.max(_full_convolution_fft(a, a))
            print(f"[polish] it {it+1:04d}  cur_val={cur_val:.12f}")

    # --------------------------------------------------------------
    # Final post‑processing (clipping to evaluator limits)
    # --------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in a.tolist()]

    # Optional verification – not required for the harness but useful for debugging
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}  best_val={best_val:.12f}  n={n}")
    except Exception as exc:
        # In the unlikely event that evaluate_sequence is unavailable
        print(f"[result] unable to evaluate final sequence: {exc}")

    return final_seq
