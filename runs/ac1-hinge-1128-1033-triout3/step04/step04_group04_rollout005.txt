2. **Exponential‑gradient (mirror‑descent) refinement** – Starting from the
   seed we repeatedly compute a sub‑gradient of the max‑convolution
   (the only non‑smooth part of the objective) and perform a
   KL‑mirror‑descent update
        a ← a * exp(‑η·∇)   (projected back to the simplex).
   The step‑size η decays slowly; an occasional “re‑heat’’ injects
   tiny random noise if progress stalls.

3. **Simulated‑annealing polishing** – Small random mass transfers
   between two coordinates are accepted either if they improve the
   score or with a temperature‑controlled probability.  Temperature and
   transfer magnitude decay over time.

All operations respect the non‑negative simplex constraints, use fast
FFT‑based full convolutions, and stop a few seconds before the deadline
so that the best vector is always returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w ≥0, sum w = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution & scoring utilities
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)−1)."""
    n = a.shape[0]
    size = 2 * n                      # zero‑padding length (power‑of‑2 not required)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_vector(a: np.ndarray) -> float:
    """Score for a vector a with sum(a) == 1 → 2·n·max(a*a)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a (only active indices)."""
    n = a.shape[0]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Parametric families (always symmetric, sum‑normalised)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i, n‑1‑i) + 1)^{‑p}   (edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^{‑p}   (edge‑heavy, product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(‑lam·i) + exp(‑lam·(n‑1‑i))   (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# Main search routine
# ----------------------------------------------------------------------

    # ------------------------------------------------------------------

    current_score = best_score
    best_vec = current.copy()

    decay = 0.9999
    iter_no = 0

            last_improve_sa = iter_sa

    # 6. Final report
