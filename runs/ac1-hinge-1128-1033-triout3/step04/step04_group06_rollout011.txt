def simplex_projection(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]

def project_onto_simplex(v):
    """Project vector v onto probability simplex {x >=0, sum=1}."""
    # Use algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Projected Adam optimization of the max‑auto‑convolution.
The algorithm works on the probability simplex (a_i ≥ 0,
∑a_i = 1).  At each iteration it

* evaluates the full convolution a*a via a real FFT,
* builds a (soft‑max) weight vector w that concentrates on the
  indices where the convolution is maximal,
* computes the gradient of the smoothed objective
  g = 2·(w_rev * a)  (exactly the sub‑gradient when τ→0),
* updates an Adam‑style estimate of the first/second moments,
* takes an additive step opposite to the (bias‑corrected) gradient,
* projects the result back onto the simplex,
* occasionally performs a random “jitter’’ move to escape plateaux.

The temperature τ is annealed from a sizeable fraction of the
current maximum toward almost zero, so the algorithm gradually
optimises the true ∞‑norm.  The implementation is fully
self‑contained, uses only NumPy and runs until the time budget
(≈1000 s) expires, returning the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using a real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # tiny negative values caused by round‑off are set to zero
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient from the (soft‑)max weights (same as the baseline)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute  g_i = 2 * Σ_k w_k * a_{k-i}
    via a single convolution: reverse w, convolve with a,
    then take the proper slice.
    """
    n = a.shape[0]
    # reverse weight vector (full length = 2n-1)
    u = w[::-1]
    # full convolution u * a  → length 3n‑2
    conv = _full_convolution_fft(u, a)
    # indices that correspond to the gradient entries:
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, ∑x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Adam‑based projected gradient descent on a (soft‑max) smoothing of
    max (a*a).  Returns the best non‑negative step‑function found
    before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # finish a couple of seconds before the hard deadline
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Initialise the sequence (try to re‑use a previously found one)
    # --------------------------------------------------------------
    try:
        # Global variable may contain a previously good construction
        a0 = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = a0.shape[0]
        a = a0.copy()
    except Exception:
        # No previous construction – start from a modest Dirichlet draw
        n = 2500                     # a size that still keeps FFT cheap
        a = np.random.dirichlet(alpha=np.full(n, 0.5))

    # Normalise to the simplex (protect against pathological draws)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # --------------------------------------------------------------
    # Optimisation bookkeeping
    # --------------------------------------------------------------
    best_a = a.copy()
    conv = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv.max())
    print(f"[init] n={n}, initial evaluation={best_val:.6f}")

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    # Hyper‑parameters (linear schedules)
    alpha_start = 0.6               # initial (additive) step size
    alpha_end   = 0.001             # final step size
    tau_factor_start = 0.6          # τ ≈ factor·M at the beginning
    tau_factor_end   = 1e-6         # final factor (practically zero)

    jitter_prob = 0.02
    iteration = 0
    last_print = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- current convolution, objective value ----
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # ---- progress for schedule ----
        progress = (now - start_time) / (deadline - start_time)

        # ---- temperature for the soft‑max (annealed) ----
        tau_factor = tau_factor_start * (1.0 - progress) + tau_factor_end * progress
        tau = tau_factor * M
        if tau < 1e-12:
            # hard sub‑gradient (arg‑max only)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # stable soft‑max: shift by the maximum to avoid overflow
            exp_args = (conv - M) / tau
            np.clip(exp_args, -80, 80, out=exp_args)
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()
            if sum_exp <= 0.0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp

        # ---- gradient of the smoothed objective ----
        grad = _gradient_from_weights(a, w)       # shape (n,)

        # ---- Adam moment updates ----
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # ---- step‑size schedule (additive) ----
        alpha = alpha_start * (1.0 - progress) + alpha_end * progress
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)   # move opposite to grad

        # ---- gradient‑descent update on the simplex ----
        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        # ---- occasional random jitter (helps escape plateaux) ----
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # move a modest random fraction of the mass at src to dst
            delta = np.random.rand() * a_candidate[src] * 0.2
            a_candidate[src] = max(a_candidate[src] - delta, 0.0)
            a_candidate[dst] = a_candidate[dst] + delta
            a_candidate = _project_simplex(a_candidate)

        a = a_candidate
        iteration += 1

        # occasional progress report (kept sparse to avoid overhead)
        if iteration - last_print >= 5000:
            elapsed = now - start_time
            print(f"[{iteration:7d}] cur={cur_val:.6f}, best={best_val:.6f}, "
                  f"elapsed={elapsed:.1f}s")
            last_print = iteration

    # --------------------------------------------------------------
    # Final reporting (the outer framework will also call it)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
