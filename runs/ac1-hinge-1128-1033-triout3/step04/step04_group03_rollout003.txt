"""
Propose a step‑function (non‑negative vector) that minimizes

        2 * n * max_k (a * a)[k]

where * is the full linear convolution.
The algorithm consists of three phases:

1.  A fast deterministic sweep over a few classic symmetric families
    (U‑shapes, exponential, cosine).  The best vector from this sweep
    seeds the subsequent optimisation.

2.  Several short random restarts that generate a fresh candidate from
    simple parametric families (U‑shape, product‑U‑shape, exponential,
    cosine or a random mixture).

3.  For each restart a smooth‑max approximation of the objective is
    minimised on the probability simplex by Adam (exponentiated‑gradient
    with momentum).  The smoothness parameter α is increased linearly,
    driving the smoothed objective towards the true max.  Gradients are
    evaluated by FFT‑based correlation, which costs O(n log n).  The
    iterate is always projected back onto the simplex.

All work respects the given time budget (default 1000 s).  The function
returns the best sequence found as a plain Python list whose sum is ≈1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex { w >= 0, sum(w)=1 }."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 2 * n                     # power‑of‑two is not required, but a little extra helps
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of 2*n*smoothmax(conv) w.r.t. a,
    where w are the soft‑max weights (sum to 1) over the convolution entries.
    The formula reduces to 2 * correlate(w, a) (valid mode).
    Implemented by FFT to stay O(n log n).
    """
    n = a.shape[0]          # length of a
    L = w.shape[0]          # = 2*n-1
    # size for FFT: at least L + n - 1 = 3*n-2
    size = 1 << int(np.ceil(np.log2(L + n - 1)))
    # FFT of w (real signal)
    fw = np.fft.rfft(w, n=size)
    # FFT of reversed a
    fa_rev = np.fft.rfft(a[::-1], n=size)
    conv = np.fft.irfft(fw * fa_rev, n=size)
    # Extract the "valid" part: indices n-1 … 2n-2 (inclusive)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector with sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Parametric families (all symmetric, non‑negative, sum‑normalised)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))^-p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a

            return cos_distribution(n, p)
        # 'mix' – blend two U‑shapes
        p1 = np.random.uniform(0.1, 10.0)
        p2 = np.random.uniform(0.1, 10.0)
        w = np.random.rand()
        a1 = u_min_distribution(n, p1)
        a2 = u_prod_distribution(n, p2)
        a = w * a1 + (1 - w) * a2
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        return a

    # Hyper‑parameters for Adam
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.4          # base learning rate for Adam
    max_iters_per_restart = 5000

    # Determine how many restarts we can afford given remaining time.
    remaining_time = deadline - time.time()
    # Rough estimate: ~0.02 s per iteration for n≈2·10⁴ (FFT + a few ops).
    # Allocate about 20 ms per iteration → 5000 iters ≈ 100 s.
    # We'll aim for up to 5 restarts but stop early if time runs out.
    max_restarts = 5

    print("\n=== Stochastic refinement (smooth‑max Adam) start ===")
    for restart_idx in range(max_restarts):
        if time.time() > deadline:
            break

        # Choose length – either reuse best length or pick a random one
        if best_seq is not None and np.random.rand() < 0.5:
            n = best_seq.shape[0]
            a = best_seq.copy()
        else:
            n = np.random.choice(lengths)
            a = random_candidate(n)

        # Small jitter to avoid accidental symmetry lock‑up
        a *= (1.0 + 0.02 * np.random.randn(n))
        a = np.maximum(a, 0.0)
        a /= a.sum()

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # Smooth‑max temperature schedule
        alpha = 5.0
        alpha_target = 200.0
        alpha_inc = (alpha_target - alpha) / max_iters_per_restart

        last_report = 0
        for it in range(1, max_iters_per_restart + 1):
            # Time guard
            if time.time() > deadline:
                break

            # 1) convolution of current a
            conv = full_convolution_fft(a)                # length 2n-1
            max_c = conv.max()

            # 2) soft‑max weights over the convolution entries
            exp_args = alpha * (conv - max_c)              # max entry → 0
            w = np.exp(exp_args)
            w_sum = w.sum()
            if w_sum == 0.0:                               # numerical safeguard
                w = np.zeros_like(w)
                w[np.argmax(conv)] = 1.0
                w_sum = 1.0
            w /= w_sum

            # 3) gradient of the smoothed objective (via FFT correlation)
            grad = gradient_from_weights(a, w)             # shape (n,)

            # 4) Adam update (note: we are minimising → step = -lr * m̂ / (√v̂+ε))
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step

            # 5) project back onto the probability simplex
            a = simplex_projection(a)

            # 6) increase smoothness
            alpha += alpha_inc

            # occasional exact evaluation (every ~50 iters)
            if it - last_report >= 50:
                sc = score_from_vector(a)
                last_report = it
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"Restart {restart_idx+1:02d}, iter {it:5d}: new best score = {best_score:.6f} (n={n})")

        # final evaluation for this restart
        sc = score_from_vector(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"Restart {restart_idx+1:02d} finished: improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence found
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    if best_seq is None:
        raise RuntimeError("No candidate was generated – this should never happen.")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # Convert to plain Python list for the caller
    return best_seq.tolist()
