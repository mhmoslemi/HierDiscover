import time, numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _full_convolution(x):
    n = x.shape[0]
    out_len = 2*n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx*fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

def _conv_fft(x, y):
    n = x.shape[0]; m = y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx*fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

def _smooth_obj_and_grad(w, eps, n):
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    conv = _full_convolution(a)
    max_conv = conv.max()
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum
    # compute weights p_k
    weights = np.exp(shifted)
    p = weights / weights.sum()
    # gradient wrt a
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)
    seg = conv_ap[n-1:2*n-1]
    grad_a = 4.0 * n * seg[::-1]
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

def _proj_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    # algorithm from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho if rho > 0 else 0
    w = np.maximum(v - theta, 0)
    return w

def _refine_hill_climb(a, best_val, deadline):
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
* Parameterise a non‑negative vector with a soft‑max:
      a_i = exp(w_i) / Σ_j exp(w_j)      →  Σ a_i = 1,   a_i ≥ 0
* Optimise a *smooth* surrogate of the max‑convolution
      L_ε(a) = 2·n·( max(conv) + ε·log∑_k exp(conv_k/ε) )
  whose gradient w.r.t. the soft‑max parameters can be written in closed form
  and evaluated with two FFT‑based convolutions.
* Use a continuation scheme: start with a large ε (very smooth) and
  gradually decrease ε while performing L‑BFGS‑B optimisation at each stage.
* Perform several random restarts (including the previous best construction
  if available) to avoid being trapped in a local minimum.
* After the continuation phase run a cheap hill‑climbing / simulated‑annealing
  refinement that keeps the total mass constant (projection onto the simplex).
* Return the best exact sequence found before the time budget expires.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negatives that may arise from numerical noise.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective (L_ε) and its gradient w.r.t. the soft‑max parameters
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L_ε, ∂L_ε/∂w) for a soft‑max parameter vector w.
    The gradient is obtained by the chain rule through the soft‑max.
    """
    # ----- soft‑max (feasible a, Σa = 1) -----
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a = a / a.sum()                  # a_i ≥ 0, Σa_i = 1

    # ----- full auto‑convolution a*a -----
    conv = _full_convolution(a)      # length 2·n‑1
    max_conv = conv.max()

    # ----- smooth approximation of the max (log‑sum‑exp) -----
    # L = 2·n·(max_conv + ε·log∑_k exp(conv_k/ε))
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)     # = log∑_k exp((conv_k‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a (closed form) -----
    # p_k = exp(conv_k/ε) / Σ_j exp(conv_j/ε)
    weights = np.exp(shifted)
    p = weights / weights.sum()      # shape (2·n‑1,)

    # g_a = 4·n·∑_k p_k·a_{k‑i}   (indices outside [0,n‑1] give 0)
    p_rev = p[::-1]                           # reversed p
    conv_ap = _conv_fft(a, p_rev)              # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]            # middle n entries
    grad_a = 4.0 * n * seg[::-1]               # shape (n,)

    # ----- chain rule (soft‑max) -----------------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)                # ∂L/∂w
    return L, grad_w


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho if rho > 0 else 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _refine_hill_climb(a: np.ndarray, best_val: float,

        # numerical noise may cause tiny negatives → project onto simplex
        new_a = _proj_simplex(new_a)

            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
