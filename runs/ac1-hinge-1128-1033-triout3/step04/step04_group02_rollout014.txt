def _refine_hill_climb(a, best_val, deadline, n):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:
            # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # transfer fraction of mass from i to j
            delta = cur_a[i] * 0.05 * rng.random()  # up to 5% of i
            if delta <= 0:
                continue
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        # keep nonnegativity; but swap always ok
        # evaluate
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:  # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

"""
Multi‑restart L‑BFGS on a smooth log‑sum‑exp surrogate of the
max‑convolution, followed by a targeted anti‑symmetric re‑ordering
and a fast hill‑climbing refinement (mass transfers + swaps).

The algorithm works under a 1000 s budget and returns the best
non‑negative step‑function (list of floats) found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global parameters used by the L‑BFGS wrapper (required to avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)      # clip tiny negatives from rounding
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    The sequence a is obtained by a_softmax = exp(w_i)/∑exp(w).
    L_ε = 2·n·max(conv) + 2·n·ε·log∑_k exp((conv_k‑max)/ε).
    """
    # soft‑max (guarantees a_i ≥ 0, Σa_i = 1)
    w_shift = w - np.max(w)                # improve numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # auto‑convolution a * a
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log∑_k exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed‑form)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability vector over k

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    # performed efficiently via convolution with the reversed p
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)        # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]       # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.minimize (uses globals for eps,n)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Produce an ordering where large values are paired with small ones
    when the sequence is reversed.  This often lowers the peak of the
    self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        # large (from the right end) goes to the left side,
        # small (from the left end) goes to the symmetric right side.
        new_a[i] = sorted_vals[n - 1 - i]
        new_a[n - 1 - i] = sorted_vals[i]

    if n % 2:
        new_a[half] = sorted_vals[half]   # middle element stays unchanged
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random mass‑transfer + random swaps, keeping Σa constant.
    Accept only strict improvements to keep the process fast.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        # 20 % chance of a pure swap (re‑orders the values)
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            # move up to 5 % of the donor's mass
            delta = cur_a[i] * 0.05 * rng.random()
            if delta <= 0.0:
                continue
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with minimal ``evaluate_sequence`` value.

    The strategy consists of several independent restarts of an L‑BFGS
    optimisation on a smooth surrogate, a targeted anti‑symmetric
    re‑ordering of the resulting vector, and a final hill‑climbing
    phase that mixes mass‑transfers and swaps.  The routine respects the
    1000 s time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – keep it modest for many restarts but still large
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Try to start from a previously discovered construction
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)    # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = np.full(n, 1.0 / n)    # uniform start

    # enforce feasibility required by ``evaluate_sequence``
    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()
    best_a = a0.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # L‑BFGS schedule (log‑space smoothing)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-8), num=13)

    # ------------------------------------------------------------------
    # Main multi‑restart loop
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 30.0:      # leave ~30 s for final refinement
        restart += 1
        print(f"[restart {restart}]")

        # random soft‑max parameters (wide variance → diverse shapes)
        w = np.random.randn(n) * 2.0

        # each restart gets the full epsilon schedule
        for eps in eps_schedule:
            if time.time() >= deadline - 20.0:
                break
            # set globals for the wrapper
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n

            # L‑BFGS‑B on the smoothed objective
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 250,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # exact evaluation of the current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a = a / a.sum()
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # --------------------------------------------------------------
        # Apply the anti‑symmetric re‑ordering heuristic on the best a
        # obtained in this restart
        # --------------------------------------------------------------
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final refinement (mass transfers + swaps) until time runs out
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
