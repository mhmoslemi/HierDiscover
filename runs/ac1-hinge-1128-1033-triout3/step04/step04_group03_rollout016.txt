def conv_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]

def smooth_max_gradient(a, beta):
    # compute conv
    conv = full_convolution_fft(a)
    # compute softmax weights
    xb = beta * conv
    xb_max = xb.max()
    w = np.exp(xb - xb_max)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0/w.shape[0])
    else:
        w = w / w_sum
    # compute gradient using convolution of w and reversed a
    a_rev = a[::-1]
    conv_rev = conv_fft(w, a_rev)
    n = a.shape[0]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    # smooth loss = (1/beta) * log sum(exp(beta*conv))
    smooth_loss = (np.log(w_sum) + xb_max) / beta   # because logsumexp = xb_max + log(sum(exp(xb-xb_max))) = xb_max + log(w_sum)
    return grad, smooth_loss, conv

"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm follows two
phases:

1.  Initialise from several deterministic parametric families and from the
    external variable ``height_sequence_1`` (if present).  The best of those
    seeds becomes the starting point.

2.  Refine the seed by minimising a smooth approximation of the
    ``max`` (log‑sum‑exp) using Adam (adaptive gradient descent) on the
    probability simplex.  The gradient of the smooth loss is obtained by a
    single FFT‑based convolution per iteration.  A simulated‑annealing
    acceptance rule, occasional random mass‑transfer moves and a reheating
    schedule help to escape local minima.

The routine respects the given time budget (default 1000 s) and returns
the best sequence found as a plain Python list (sum ≈ 1)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex { w >= 0 , sum(w) = 1 }.
    Implementation follows Wang & Carreira‑Perpiñán (O(n log n)).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a * a using a real FFT.
    Returns a NumPy array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n                     # power of two is not required but fast enough
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution (x * y) using FFT.  Returns an array of length
    len(x) + len(y) - 1.
    """
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()        # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    z = np.fft.irfft(X * Y, n=size)
    return z[:n]


def smooth_max_gradient(a: np.ndarray, smooth_beta: float):
    """
    Compute the gradient of the smooth (log‑sum‑exp) approximation of
    max_k (a*a)[k] with respect to a, together with the smooth loss value
    and the exact convolution.
    """
    conv = full_convolution_fft(a)                # length 2n‑1
    xb = smooth_beta * conv
    xb_max = xb.max()
    w = np.exp(xb - xb_max)                       # unnormalised softmax weights
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w = w / w_sum                               # softmax probabilities

    # grad = 2 * sum_k w_k * a_{k-i}  (i ranges over a)
    a_rev = a[::-1]
    conv_rev = conv_fft(w, a_rev)                 # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    # smooth loss = (1 / beta) * logsumexp(beta * conv)
    smooth_loss = (xb_max + np.log(w_sum)) / smooth_beta
    return grad, smooth_loss, conv


# ----------------------------------------------------------------------
# Parametric families (all symmetric)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))^-p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within a given
    time budget.  Returns the sequence as a plain Python list; the sum
    is approximately one (the external evaluator rescales it).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise from external construction or simple parametric families
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        if init.sum() == 0.0:
            raise ValueError
        init /= init.sum()
        n = init.shape[0]
        print(f"Using external height_sequence_1 (length {n})")
    except Exception:
        # fall back to a modest length and uniform start
        n = 8000
        init = np.full(n, 1.0 / n)
        print(f"External sequence not available – using uniform start of length {n}")

    # a handful of deterministic seeds (all already normalised)
    seeds = [
        init,
        np.full(n, 1.0 / n),
        u_min_distribution(n, p=1.0),
        u_min_distribution(n, p=2.0),
        u_prod_distribution(n, p=1.0),
        exp_distribution(n, lam=0.01),
        cos_distribution(n, p=2.0),
    ]

    best_seq = None
    best_score = np.inf

    print("\n=== Evaluating initial seeds ===")
    for s in seeds:
        s = simplex_projection(s)               # make sure we are on the simplex
        sc = evaluate_sequence(s.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
            print(f"  seed score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Refine the best seed with smooth‑max Adam optimisation
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_conv = full_convolution_fft(a)
    cur_score = 2.0 * n * cur_conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    eps = 1e-8
    lr = 0.3

    # schedule parameters
    temperature = 1.0
    temp_decay = 0.9997
    smooth_beta = 0.5
    smooth_beta_target = 200.0
    smooth_beta_factor = 1.001

    iteration = 0
    last_improve = 0

    print("\n=== Refined optimisation start ===")
    while time.time() < deadline:
        iteration += 1

        # ---- compute smooth gradient ----
        grad, smooth_loss, conv = smooth_max_gradient(a, smooth_beta)

        # ---- Adam update on the simplex (project after the step) ----
        m = adam_beta1 * m + (1 - adam_beta1) * grad
        v = adam_beta2 * v + (1 - adam_beta2) * (grad ** 2)
        m_hat = m / (1 - adam_beta1 ** iteration)
        v_hat = v / (1 - adam_beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_cand = a - step
        a_cand = simplex_projection(a_cand)

        # ---- evaluate candidate ----
        cand_conv = full_convolution_fft(a_cand)
        cand_score = 2.0 * n * cand_conv.max()

        # ---- simulated‑annealing acceptance ----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_cand
            cur_conv = cand_conv
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ---- occasional random mass‑transfer move (helps escape) ----
        if iteration % 500 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * np.random.rand() * 0.2   # move up to 20 % of the mass at i
                a[i] -= delta
                a[j] += delta
                a = simplex_projection(a)
                cur_conv = full_convolution_fft(a)
                cur_score = 2.0 * n * cur_conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    last_improve = iteration
                    print(f"Iter {iteration:7d} (mass‑move): new best = {best_score:.6f}")

        # ---- schedule updates ----
        temperature *= temp_decay
        smooth_beta = min(smooth_beta_target, smooth_beta * smooth_beta_factor)

        # ---- stagnation handling (re‑heat) ----
        if iteration - last_improve > 20000:
            a += np.random.rand(n) * 1e-4
            a = simplex_projection(a)
            cur_conv = full_convolution_fft(a)
            cur_score = 2.0 * n * cur_conv.max()
            temperature = 1.0
            smooth_beta = 0.5
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation, score={cur_score:.6f}")

        # ---- time check ----
        if time.time() >= deadline:
            break

    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
