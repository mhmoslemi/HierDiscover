"""
Improved optimizer for the ``evaluate_sequence`` challenge.

The algorithm works in four stages (respecting a 1000 s budget):

1. **Smooth surrogate optimisation** – minimise a log‑sum‑exp smooth
   version of the exact objective with L‑BFGS‑B.  The smoothing
   parameter ε is decreased geometrically (ε≈10⁻¹ … 10⁻¹⁶).

2. **Randomised restarts** – the best solution from stage 1 is
   perturbed and optimised again on the finer ε‑tail.  This helps to
   escape the basin of attraction of the first local minimum.

3. **Simulated annealing** – a simple Metropolis move (transfer a
   random fraction of mass from one index to another) is performed
   many thousands of times.  The temperature decays linearly with the
   remaining time; occasional uphill moves let the search explore
   beyond the local optimum found by the gradient steps.

4. **Fine sub‑gradient descent** – a classic projected sub‑gradient
   step on the exact max‑convolution objective finishes the polishing
   phase.

All helper functions are top‑level, no lambdas or hidden closures are
used.  The routine returns a list of non‑negative floats that (almost
certainly) improves the current best bound of ≈1.512.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------------------------------------------------------------
# FFT based convolution utilities
# ---------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·n‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # tiny negative values can appear due to round‑off, clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)
    return conv


# ---------------------------------------------------------------
# Log‑sum‑exp smooth surrogate and gradient (used by L‑BFGS)
# ---------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (objective, gradient) of the ε‑smoothed surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε),

    where a = softmax(w) (∑a = 1, a_i ≥ 0) and conv = a * a.
    """
    # soft‑max – converts unrestricted w to a feasible probability vector
    w_shift = w - np.max(w)                # for numerical stability
    a = np.exp(w_shift)
    a = a / a.sum()

    # full autocorrelation of a
    conv = _full_convolution(a)
    max_c = conv.max()

    # log‑sum‑exp part
    shifted = (conv - max_c) / eps
    weights = np.exp(shifted)
    log_sum = np.logaddexp.reduce(shifted)     # log Σ exp(...)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (see analysis notes)
    p = weights / weights.sum()                # probability over lags
    p_rev = p[::-1]                            # reversed for convolution
    conv_ap = _conv_fft(a, p_rev)              # length = 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]            # keep the middle n entries
    grad_a = 4.0 * n * seg[::-1]               # chain from 2·n·max_c term

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


# ---------------------------------------------------------------
# Project a vector onto the probability simplex {x≥0, Σx = 1}
# ---------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------
# Sub‑gradient descent directly on the exact max‑convolution objective
# ---------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Projected sub‑gradient descent on 2·n·max(conv)  (Σa = 1, a≥0).
    Returns the best vector found before the deadline.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    step = 0.1

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_c = conv.max()
        ks = np.flatnonzero(conv == max_c)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[idx] = 2.0 * a[k - idx]

        # gradient step + projection
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = 2.0 * n * np.max(_full_convolution(a_new))

        if new_val < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            step = min(step * 1.05, 0.5)
        else:
            step = max(step * 0.7, 1e-5)

    return best_a, best_val


# ---------------------------------------------------------------
# Simulated annealing – random mass transfers
# ---------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Simple Metropolis move: pick a source index i and a destination j,
    move a random fraction of the source mass to j, accept according to
    a linearly cooling temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    cur_a = a.copy()
    cur_obj = cur_val

    start = time.time()
    while time.time() < deadline:
        # linear cooling temperature (0.1 → 1e‑6)
        remaining = deadline - time.time()
        total = deadline - start
        T = max(1e-6, 0.1 * remaining / total)

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or cur_a[i] <= 0.0:
            continue

        # transfer a random fraction (up to 30 %) of the source mass
        delta = cur_a[i] * rng.random() * 0.3
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # exact evaluation (no normalisation needed – sum stays 1)
        new_obj = 2.0 * n * np.max(_full_convolution(a_new))

        if new_obj < cur_obj or rng.random() < math.exp(-(new_obj - cur_obj) / T):
            cur_a = a_new
            cur_obj = new_obj
            if cur_obj < best_val:
                best_val = cur_obj
                best_a = cur_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

    return best_a, best_val

# Main candidate proposer required by the evaluation framework
# ---------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Optimises a non‑negative step‑function to minimise
    ``evaluate_sequence``.  The routine respects the given time budget
    and returns the best list of floats found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # -----------------------------------------------------------
    # Problem size – a somewhat larger n gives the optimiser more
    # freedom while still being tractable.
    # -----------------------------------------------------------
    n = 8000

    # -----------------------------------------------------------
    # Initialise the sequence (reuse a previous construction if
    # available, otherwise start from a near‑uniform random vector)
    # -----------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        else:
            a0 = a0.copy()
    except Exception:
        a0 = np.full(n, 1.0 / n) + rng.random(n) * 1e-4

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    best_a = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # -----------------------------------------------------------
    # Phase 1 – smoothed surrogate optimisation (continuation in ε)
    # -----------------------------------------------------------
    eps_schedule = np.logspace(-1, -16, num=32)   # from 0.1 down to 1e‑16
    w = np.log(best_a + 1e-15)

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        res = minimize(fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # evaluate exact objective
        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[smooth eps {eps:.1e}] new best = {best_val:.6f}")

    # -----------------------------------------------------------
    # Phase 2 – randomised restarts (finer ε‑tail)
    # -----------------------------------------------------------
    restarts = 8
    while time.time() < deadline - 5.0 and restarts > 0:
        restarts -= 1
        w = np.log(best_a + 1e-15) + rng.normal(scale=0.4, size=n)

        for eps in eps_schedule[-9:]:    # only the finest eps values
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 250,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart] new best = {best_val:.6f}")

    # -----------------------------------------------------------
    # Phase 3 – simulated annealing for global exploration
    # -----------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _simulated_annealing(best_a,
                                                best_val,
                                                deadline,
                                                rng)

    # -----------------------------------------------------------
    # Phase 4 – fine sub‑gradient polishing
    # -----------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _subgradient_descent(best_a,
                                                best_val,
                                                deadline,
                                                rng)

    # -----------------------------------------------------------
    # Final reporting (using the provided evaluator)
    # -----------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Convert to plain Python list as required by the framework
    return best_a.tolist()
