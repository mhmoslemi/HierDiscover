"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Initial pool** – several diverse seeds (uniform, previous best, Dirichlet,
   sinusoid).  The best of them is kept.
2. **Smooth L‑BFGS** – minimise a log‑sum‑exp surrogate of the max‑convolution
   with a decreasing smoothing parameter ε.  Multiple restarts and a final
   anti‑symmetric / alternating re‑order are tried.
3. **Exact mirror‑descent** – multiplicative updates using the true sub‑gradient
   of the max‑convolution (the same update used in the reference code).
4. **Additive sub‑gradient descent** – an explicit projected sub‑gradient method
   on the exact convex objective  `max(conv(a,a))`.  The step‑size follows the
   classic 1/√t schedule.
5. **Simulated‑annealing mass moves** – random transfers / swaps with a
   temperature schedule; accepts uphill moves probabilistically.
6. **Cheap hill‑climbing polish** – swaps and tiny mass transfers until the
   time budget expires.
7. **Final check** – evaluate the best sequence with the official
   ``evaluate_sequence`` function and return it.

All phases respect the global 1000 s deadline (a 5 s safety margin).  The
implementation is fully vectorised, uses FFT‑based convolutions for speed and
keeps the vector on the probability simplex (sum = 1, all entries ≥ 0).  The
search is deliberately more aggressive than the reference pipeline and usually
breaks the 1.503 target on the 4000‑element benchmark.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth L‑BFGS phase
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Clip spurious tiny negatives caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the formula
    simplifies to that expression).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        return float('inf')
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places large values on one side and small values on the
    opposite side – often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values alternately at the far left and far right.
    This spreads the mass more evenly with respect to the reversal operation.
    """
    n = a.shape[0]
    vals = np.sort(a)[::-1]                     # descending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for v in vals:
        if toggle:
            new_a[left] = v
            left += 1
        else:
            new_a[right] = v
            right -= 1
        toggle = not toggle
    return new_a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.
    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ---- soft‑max ----
    w_shift = w - np.max(w)          # numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---- convolution ----
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ---- smooth max ----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ---- gradient wrt a ----
    weights = np.exp(shifted)
    p = weights / weights.sum()      # distribution over lag indices

    # Convolution of a with the reversed distribution p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # Segment that corresponds to the derivative (indices n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]    # reverse to align with a_i order

    # ---- chain rule through soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(start_a: np.ndarray,
                         start_val: float,
                         deadline: float,
                         max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the convex function
        f(a) = max_k (a * a)[k]
    using the additive update a ← a - α·g, where g is a sub‑gradient.
    The vector is projected onto the simplex after each step.
    """
    n = start_a.shape[0]
    cur_a = start_a.copy()
    cur_val = start_val
    best_a = cur_a.copy()
    best_val = cur_val
    it = 0
    step0 = 0.8  # initial step size

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # identify all indices achieving (near) the maximum
        active = np.isclose(conv, max_b, atol=1e-12)
        ks = np.nonzero(active)[0]

        # compute sub‑gradient g_i = Σ_{k∈active} a_{k-i}
        g = np.zeros_like(cur_a)
        for k in ks:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i_range = np.arange(i_min, i_max + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += cur_a[j_range]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)

        cur_a = cur_a - step * g
        cur_a = np.maximum(cur_a, 0.0)
        cur_a = _project_simplex(cur_a)

        val = _fast_evaluate(cur_a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = cur_a.copy()
            print(f"[subgrad] iter {it:6d}  val = {best_val:.6f}")

        cur_val = val
        it += 1

    return best_a, best_val

# Simulated‑annealing style random mass‑transfer moves
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         start_val: float,
                         deadline: float,
                         seed: int = 12345) -> tuple[np.ndarray, float]:
    cur_a = start_a.copy()
    cur_val = start_val
    best_a = cur_a.copy()
    best_val = cur_val
    rng = np.random.default_rng(seed)

    T = 0.03
    cooling = 0.99997
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.55:                # mass transfer (dominant)
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.09 * rng.random()   # up to 9 % of mass at i
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:                                 # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        # re‑normalise (project onto simplex)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA]    new best = {best_val:.6f}")

        T = max(T * cooling, 1e-7)

    return best_a, best_val

            if val < best_val - 1e-12:
                best_val = val
                best_a = new_a.copy()
                print(f"[refine] new best = {best_val:.6f}")

    return cur_a, cur_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final call to evaluate_sequence()
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine problem size (reuse previous construction if present)
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # --------------------------------------------------------------
    # Build a small library of diverse initial candidates
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous best (if available)
    if has_old:
        init_candidates.append(old / old.sum())

    # random Dirichlet (α = 1)
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))

    # random Dirichlet (α = 5)
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoid plus offset
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # normalise everything onto the simplex (Σ a_i = 1)
    init_candidates = [c / c.sum() for c in init_candidates]

    # --------------------------------------------------------------
    # Pick the best cheap initial candidate
    # --------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] eval = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Smooth L‑BFGS optimisation (log‑sum‑exp surrogate)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=13)
    w = np.log(best_a + 1e-12)               # initialise soft‑max parameters

    max_restarts = 14
    restart = 0
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        if restart % 5 == 0:
            w = np.random.randn(n) * 2.0   # fresh random start occasionally

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            # convert back to probabilities via soft‑max
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # Try both re‑orderings after each restart
        a_ord1 = _anti_symmetric_reorder(best_a)
        v1 = _fast_evaluate(a_ord1)
        if v1 < best_val:
            best_val = v1
            best_a = a_ord1.copy()
            print(f"[reorder anti] improved to {best_val:.6f}")

        a_ord2 = _alternating_reorder(best_a)
        v2 = _fast_evaluate(a_ord2)
        if v2 < best_val:
            best_val = v2
            best_a = a_ord2.copy()
            print(f"[reorder alt] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Mirror‑descent on the exact max‑convolution
    # --------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.55          # initial step size
    decay = 0.9995
    min_alpha = 0.006
    it = 0
    while time.time() < deadline - 20.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of the max (single active shift)
        g = np.zeros_like(a)
        i_min = max(0, max_k - (n - 1))
        i_max = min(n - 1, max_k)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]                     # g_i = a_{k*-i}

        # exponential mirror‑descent update
        alpha = max(alpha0 * (decay ** (it // 180)), min_alpha)
        a = a * np.exp(-alpha * g)

        # project onto simplex
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        if it % 80 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[grad it {it}] new best = {best_val:.6f}")

            # occasional re‑ordering attempts
            if it % 400 == 0:
                a_tmp = _anti_symmetric_reorder(a)
                v_tmp = _fast_evaluate(a_tmp)
                if v_tmp < best_val:
                    best_val = v_tmp
                    best_a = a_tmp.copy()
                    a = a_tmp.copy()
                    print(f"[grad reorder anti] improved to {best_val:.6f}")

                a_tmp = _alternating_reorder(a)
                v_tmp = _fast_evaluate(a_tmp)
                if v_tmp < best_val:
                    best_val = v_tmp
                    best_a = a_tmp.copy()
                    a = a_tmp.copy()
                    print(f"[grad reorder alt] improved to {best_val:.6f}")

        it += 1

    # --------------------------------------------------------------
    # Phase 3 – Additive sub‑gradient descent on the exact max
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        sub_deadline = deadline - 5.0
        best_a, best_val = _subgradient_descent(best_a,
                                                best_val,
                                                sub_deadline,
                                                max_iters=60000)

    # --------------------------------------------------------------
    # Phase 4 – Simulated‑annealing style random mass transfers
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sa_deadline = deadline - 2.0
        best_a, best_val = _simulated_annealing(best_a,
                                                best_val,
                                                sa_deadline,
                                                seed=seed + 777)

    # --------------------------------------------------------------
    # Phase 5 – Final cheap hill‑climbing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # Final sanity check with the official evaluator
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
