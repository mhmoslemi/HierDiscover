def _simplex_projection(v):
    """Project onto simplex {x >= 0, sum x = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]   # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _simplex_projection(v):
    """Project vector v onto the probability simplex (sum=1, nonnegative)."""
    # works for any length
    n = v.shape[0]
    # sort v descending

"""
Hybrid optimisation for the step‑function convolution problem.

The algorithm proceeds in the following stages (all respecting a total
budget of `budget_s` seconds):

1. Load the previous construction (if available) and use it as a baseline.
2. Randomly sample a few parametric families (Beta distributions and
   mixtures of two Betas).  Because the objective is scale‑invariant we
   only need to normalise the vectors to sum to one.  These samples are
   evaluated with a fast FFT‑based evaluator and the best one is kept.
3. Apply a few deterministic reorderings (descending, anti‑symmetric,
   alternating) to the current best vector.
4. Perform a projected sub‑gradient descent on the *exact* max‑convolution
   objective.  For the current max‑index `k` the sub‑gradient is
   `g_i = a_i + a_{k-i}` (if the second term is defined).  After each
   gradient step we project back onto the probability simplex.
5. A lightweight hill‑climbing phase (random swaps / small mass
   transfers) refines the solution until the deadline.
6. Return the final sequence (as a Python list) and print the exact
   score using the provided `evaluate_sequence`.

The method is deliberately different from the previous L‑BFGS + SA
pipeline and is designed to explore a wider set of shapes while keeping
the runtime well within the 1000 s limit.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (identical to the ones used before)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = rfft(a, size)
    conv = irfft(A * A, size)[:out_len]
    # eliminate tiny negatives caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` for a non‑negative vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple re‑ordering heuristics
# ----------------------------------------------------------------------
def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Sort descending."""
    return np.sort(a)[::-1]


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large values left, small values right (pairwise)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = sorted_vals[n - 1 - i]      # largest to the left
        out[n - 1 - i] = sorted_vals[i]      # smallest to the right
    if n % 2:
        out[half] = sorted_vals[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    sorted_vals = np.sort(a)
    left, right = 0, len(sorted_vals) - 1
    out = np.empty_like(a)
    idx = 0
    while left <= right:
        out[idx] = sorted_vals[right]
        idx += 1
        right -= 1
        if left <= right:
            out[idx] = sorted_vals[left]
            idx += 1
            left += 1
    return out


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, a_i ≥ 0)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects an arbitrary vector ``v`` onto the simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)                # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise exactly
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    Returns the best sequence found within the allotted time.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2  # tiny safety margin

    # --------------------------------------------------------------
    # Problem size (kept identical to the previous best)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 0. Load previous construction if possible
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")
    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] baseline evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Random sampling of Beta‑family shapes
    # --------------------------------------------------------------
    xs = (np.arange(n, dtype=float) + 0.5) / n  # mid‑point grid on (0,1)

    time_left = deadline - time.time()
    beta_deadline = time.time() + 0.30 * time_left
    while time.time() < beta_deadline:
        # Log‑uniform sampling of the shape parameters
        log_alpha = np.random.uniform(-2.0, 1.0)   # α ∈ [10⁻², 10¹]
        log_beta  = np.random.uniform(-2.0, 1.0)   # β ∈ [10⁻², 10¹]
        alpha = 10.0 ** log_alpha
        beta = 10.0 ** log_beta

        # Raw (unnormalised) Beta density on the grid
        a_raw = np.power(xs, alpha - 1.0) * np.power(1.0 - xs, beta - 1.0)
        a_raw = np.nan_to_num(a_raw, nan=0.0, posinf=0.0, neginf=0.0)
        if a_raw.sum() == 0.0:
            continue
        a = a_raw / a_raw.sum()

        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[beta] new best = {best_val:.6f} (α={alpha:.4f}, β={beta:.4f})")

    # --------------------------------------------------------------
    # 2. Random mixtures of two Beta families
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    mix_deadline = time.time() + 0.20 * time_left
    while time.time() < mix_deadline:
        w = np.random.rand()
        # Parameters for the two components
        log_a1 = np.random.uniform(-2.0, 1.0)
        log_b1 = np.random.uniform(-2.0, 1.0)
        log_a2 = np.random.uniform(-2.0, 1.0)
        log_b2 = np.random.uniform(-2.0, 1.0)
        a1 = np.power(xs, 10.0 ** log_a1 - 1.0) * np.power(1.0 - xs, 10.0 ** log_b1 - 1.0)
        a2 = np.power(xs, 10.0 ** log_a2 - 1.0) * np.power(1.0 - xs, 10.0 ** log_b2 - 1.0)
        a1 = np.nan_to_num(a1, nan=0.0, posinf=0.0, neginf=0.0)
        a2 = np.nan_to_num(a2, nan=0.0, posinf=0.0, neginf=0.0)

        a_mix_raw = w * a1 + (1.0 - w) * a2
        if a_mix_raw.sum() == 0.0:
            continue
        a_mix = a_mix_raw / a_mix_raw.sum()

        val = _eval_fast(a_mix)
        if val < best_val:
            best_val = val
            best_a = a_mix.copy()
            print(f"[mix] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Deterministic reorderings
    # --------------------------------------------------------------
    for reorder in (_descending_reorder, _anti_symmetric_reorder, _alternating_reorder):
        a_r = reorder(best_a)
        v_r = _eval_fast(a_r)
        if v_r < best_val:
            best_val = v_r
            best_a = a_r.copy()
            print(f"[reorder] {reorder.__name__} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Projected sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        # Number of iterations scales with the remaining budget
        max_iters = int(min(2000, max(200, 1500 * time_left / budget_s)))
        a = best_a.copy()
        step0 = 0.05  # initial step size
        for it in range(max_iters):
            if time.time() >= deadline:
                break
            conv = _fft_convolve_self(a)
            k_max = int(np.argmax(conv))

            i_vals = np.arange(n, dtype=int)
            j_vals = k_max - i_vals
            mask = (j_vals >= 0) & (j_vals < n)

            grad = a.copy()
            grad[mask] += a[j_vals[mask]]

            eta = step0 / (1.0 + 0.01 * it)   # gently decreasing steps
            a_new = _simplex_projection(a - eta * grad)

            cur_val = _eval_fast(a_new)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_new.copy()
                print(f"[sgd] iter {it:04d} new best = {best_val:.6f}")
            a = a_new

    # --------------------------------------------------------------
    # 5. Final hill‑climbing (swap / small transfers)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 2.0:
        cur_a = best_a.copy()
        cur_val = best_val
        rng = np.random.default_rng()
        it = 0
        while time.time() < deadline:
            if rng.random() < 0.25:          # pure swap
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                new_a = cur_a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]
            else:                            # small mass transfer
                i, j = rng.integers(0, n, size=2)
                if i == j or cur_a[i] == 0.0:
                    continue
                delta = cur_a[i] * 0.05 * rng.random()
                new_a = cur_a.copy()
                new_a[i] -= delta
                new_a[j] += delta

            val = _eval_fast(new_a)
            if val < cur_val - 1e-12:
                cur_a = new_a
                cur_val = val
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = cur_a.copy()
                    print(f"[hill] iter {it} new best = {best_val:.6f}")
            it += 1

    # --------------------------------------------------------------
    # 6. Exact evaluation with the provided function
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
