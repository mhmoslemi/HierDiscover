def generate_initial_sequences(n, rng):

   seeds = []

   # Uniform
   a_uni = np.full(n, 1.0/n)
   seeds.append(a_uni)

   # Hann
   a_hann = np.hanning(n) # returns zeros at edges? Hann actually includes zeros at both ends. It returns weights sin^2(π * i / (N-1)). It's positive but sum not 1. We'll normalize.
   a_hann = np.maximum(a_hann, 0) # ensure nonnegative.
   a_hann = a_hann / a_hann.sum()
   seeds.append(a_hann)

   # Hamming
   a_hamm = np.hamming(n)
   a_hamm = np.maximum(a_hamm, 0)
   a_hamm = a_hamm / a_hamm.sum()
   seeds.append(a_hamm)

   # Blackman
   a_black = np.blackman(n)
   a_black = np.maximum(a_black, 0)
   a_black = a_black / a_black.sum()
   seeds.append(a_black)

   # Kaiser with various betas
   for beta in [2,4,6,8,10]:
       a_kaiser = np.kaiser(n, beta) # from numpy
       a_kaiser = np.maximum(a_kaiser, 0)
       a_kaiser = a_kaiser / a_kaiser.sum()
       seeds.append(a_kaiser)

   # Gaussian bump
   x = np.linspace(-1,1,n)
   sigma = 0.3
   a_gauss = np.exp(-0.5*(x/sigma)**2)
   a_gauss = a_gauss / a_gauss.sum()
   seeds.append(a_gauss)

   # Sine distribution: a_i = sin(pi*(i+0.5)/n) positive
   i = np.arange(n)
   a_sine = np.sin(np.pi * (i + 0.5) / n)
   a_sine = np.maximum(a_sine, 0)
   a_sine = a_sine / a_sine.sum()
   seeds.append(a_sine)

   # random
   a_rand = rng.random(n)
   a_rand = a_rand / a_rand.sum()
   seeds.append(a_rand)

   return seeds

"""
Advanced multi‑start optimizer for `evaluate_sequence`.

Strategy
--------
1. **Diverse seeds** –  generate a handful of analytically‑motivated
   initial sequences (uniform, Hann, Hamming, Blackman, several Kaiser
   windows, a Gaussian bump, a sine‑shaped profile and a random draw).
   All are normalised to sum = 1.

2. **Smooth surrogate minimisation** –  For each seed we minimise the
   log‑sum‑exp smooth approximation of the L∞‑convolution norm using
   L‑BFGS‑B.  The smoothing parameter ε is decreased on a log‑scale
   (continuation) to approach the true objective.

3. **Randomised restarts** –  After the continuation a few perturbed
   restarts are performed (Gaussian noise on the soft‑max parameters)
   and the continuation is repeated on the finer ε‑tail.

4. **Aggressive mass‑transfer hill‑climbing** –  A specialised local
   search repeatedly moves a fraction of the mass away from the lag(s)
   where the convolution attains its maximum.  The move is
   accepted only when the exact objective improves.

5. **Projected sub‑gradient descent** –  The exact nonsmooth objective
   2·n·max(conv) is optimised with a sub‑gradient step and Euclidean
   projection onto the simplex.  An adaptive step‑size drives rapid
   convergence.

The whole procedure respects the 1000 s budget and keeps the best
candidate (across all lengths and seeds) seen so far.  The final
sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·n‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative rounding artefacts
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (value, gradient) of the surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε)

    where a = softmax(w) (∑a = 1, a_i ≥ 0).  The gradient is taken with
    respect to w (the log‑softmax parameters).
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)               # prevent overflow
    a = np.exp(w_shift)
    a = a / a.sum()

    # autocorrelation of a (full, length 2·n‑1)
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    # stable log‑sum‑exp
    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted - max_shifted)       # numerically stable
    p = weights / weights.sum()                   # probability distribution over lags

    # convolution a * rev(p)   (size 3·n‑2, we only need central n entries)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                # linear convolution
    seg = conv_ap[n - 1: 2 * n - 1]               # middle n entries, reversed later
    grad_a = 4.0 * n * seg[::-1]                 # derivative of 2·n·max(conv) part

    # chain rule through the soft‑max:  ∂L/∂w = a ⊙ (∂L/∂a - (a·∂L/∂a)·1)
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill‑climbing
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              cur_val: float,
                              deadline: float,
                              rng: np.random.Generator,
                              step_start: float = 0.2):
    """
    Repeatedly move a fraction of mass away from the lag(s) attaining the
    maximal convolution.  Returns the best vector (and its value) found
    before the deadline expires.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_val = cur_val
    step_frac = step_start

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        # choose a lag uniformly among the ties
        k = int(rng.choice(ks))

        # indices that contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]

        if contrib.size == 0:
            continue

        # pick up to 4 most influential pairs
        top_n = min(4, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        # low‑mass positions that are not donors
        low_mass = np.where(cur_a < cur_a.mean())[0]
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            # amount of mass to move (fraction of the *smaller* of the two)
            delta = step_frac * min(new_a[i], new_a[j]) if i != j else step_frac * new_a[i]
            if delta <= 0:
                ok = False
                break
            new_a[i] -= delta
            new_a[j] -= delta
            # distribute to two fresh slots (the two destinations for this pair)
            p = int(dest[2 * d])
            q = int(dest[2 * d + 1])
            new_a[p] += delta
            new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue

        # re‑normalise onto the simplex
        new_a = _project_simplex(new_a)

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-6)

    return best_a, best_val


# ----------------------------------------------------------------------
# Fine sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Sub‑gradient descent on 2·n·max(conv) with Euclidean projection onto the simplex.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_val = cur_val
    eta = 0.2                     # start with a moderate step size

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag (see analysis)
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # take a step and project back onto the simplex
        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            eta = min(eta * 1.08, 0.5)
        else:
            eta = max(eta * 0.75, 1e-5)

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed generation (various windows + random)
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator):
    """Return a list of diverse normalised initial sequences of length n."""
    seeds = []

    # 1) uniform
    seeds.append(np.full(n, 1.0 / n))

    # 2) Hann (np.hanning)
    h = np.hanning(n)
    seeds.append(h / h.sum())

    # 3) Hamming
    h = np.hamming(n)
    seeds.append(h / h.sum())

    # 4) Blackman
    h = np.blackman(n)
    seeds.append(h / h.sum())

    # 5) Kaiser with several betas
    for beta in (2, 4, 6, 8, 10, 12):
        h = np.kaiser(n, beta)
        seeds.append(h / h.sum())

    # 6) Gaussian bump (centered)
    x = np.linspace(-1.0, 1.0, n)
    sigma = 0.3
    g = np.exp(-0.5 * (x / sigma) ** 2)
    seeds.append(g / g.sum())

    # 7) Sine‑shaped positive profile
    i = np.arange(n)
    s = np.sin(np.pi * (i + 0.5) / n)
    seeds.append(s / s.sum())

    # 8) pure random
    r = rng.random(n)
    seeds.append(r / r.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑start optimiser for `evaluate_sequence`.  Returns a list of
    non‑negative floats (the best step‑function heights found before the
    time budget expires).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_val = np.inf
    best_a = None
    best_n = None

    # ------------------------------------------------------------------
    # Problem sizes – we try a couple of lengths (larger lengths give a
    # finer search space but are slower).  2000 and 4000 work well on a
    # single core within the time budget.
    # ------------------------------------------------------------------
    candidate_lengths = [2000, 4000]

    for n in candidate_lengths:
        if time.time() >= deadline - 5:
            break
        print(f"[setup] Beginning optimisation for n = {n}")

        # ------------------------------------------------------------------
        # Build the set of starting points
        # ------------------------------------------------------------------
        seeds = _generate_seeds(n, rng)

        # If a previously found best exists, interpolate it to the current
        # length and use it as an additional seed.
        if best_a is not None:
            try:
                old_n = best_n
                old_x = np.linspace(0.0, 1.0, old_n)
                new_x = np.linspace(0.0, 1.0, n)
                interpolated = np.interp(new_x, old_x, best_a)
                interpolated = np.maximum(interpolated, 0.0)
                interpolated /= interpolated.sum()
                seeds.append(interpolated)
            except Exception:
                pass

        # ------------------------------------------------------------------
        # Optimisation loop over all seeds for this length
        # ------------------------------------------------------------------
        for seed_idx, a0 in enumerate(seeds):
            if time.time() >= deadline - 5:
                break
            print(f"[seed] n={n}  seed {seed_idx + 1}/{len(seeds)}")

            # soft‑max parameters for the smooth surrogate
            w = np.log(a0 + 1e-12)

            # ------------------------------------------------------------------
            # 1. Smooth surrogate minimisation (continuation in ε)
            # ------------------------------------------------------------------
            eps_schedule = np.logspace(-2, -12, num=20)   # 20 values
            cur_w = w.copy()
            cur_best_val = 2.0 * n * np.max(_full_convolution(a0))

            for eps in eps_schedule:
                if time.time() >= deadline:
                    break

                def fun_and_grad(w_vec):
                    return _smooth_obj_and_grad(w_vec, eps, n)

                res = minimize(fun_and_grad,
                               cur_w,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': 200,
                                        'ftol': 1e-12,
                                        'gtol': 1e-12,
                                        'disp': False})
                cur_w = res.x

                a = np.exp(cur_w - np.max(cur_w))
                a = a / a.sum()
                cur_val = 2.0 * n * np.max(_full_convolution(a))

                if cur_val < cur_best_val:
                    cur_best_val = cur_val

                # update global best
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    best_n = n
                    print(f"[global best] new best = {best_val:.6f} (n={n})")

            # ------------------------------------------------------------------
            # 2. Randomised restarts on the finer ε‑tail
            # ------------------------------------------------------------------
            n_restarts = 5
            for r in range(n_restarts):
                if time.time() >= deadline - 5:
                    break
                w_restart = cur_w + rng.normal(scale=0.5, size=n)

                for eps in eps_schedule[-5:]:   # use the smallest 5 eps values
                    if time.time() >= deadline:
                        break
                    def fun_and_grad(w_vec):
                        return _smooth_obj_and_grad(w_vec, eps, n)

                    res = minimize(fun_and_grad,
                                   w_restart,
                                   method='L-BFGS-B',
                                   jac=True,
                                   options={'maxiter': 150,
                                            'ftol': 1e-12,
                                            'gtol': 1e-12,
                                            'disp': False})
                    w_restart = res.x

                a = np.exp(w_restart - np.max(w_restart))
                a = a / a.sum()
                cur_val = 2.0 * n * np.max(_full_convolution(a))

                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    best_n = n
                    print(f"[global best restart] new best = {best_val:.6f} (n={n})")

            # ------------------------------------------------------------------
            # 3. Aggressive mass‑transfer hill‑climbing (exact objective)
            # ------------------------------------------------------------------
            if time.time() < deadline:
                start_val = 2.0 * n * np.max(_full_convolution(best_a if best_n == n else a0))
                best_a_loc, best_val_loc = _mass_transfer_hill_climb(
                    best_a if best_n == n else a0,
                    start_val,
                    deadline,
                    rng,
                    step_start=0.3)
                if best_val_loc < best_val:
                    best_val = best_val_loc
                    best_a = best_a_loc.copy()
                    best_n = n
                    print(f"[global best after hillclimb] new best = {best_val:.6f} (n={n})")

            # ------------------------------------------------------------------
            # 4. Fine sub‑gradient descent
            # ------------------------------------------------------------------
            if time.time() < deadline:
                start_val = 2.0 * n * np.max(_full_convolution(best_a if best_n == n else a0))
                best_a_loc, best_val_loc = _subgradient_descent(
                    best_a if best_n == n else a0,
                    start_val,
                    deadline,
                    rng)
                if best_val_loc < best_val:
                    best_val = best_val_loc
                    best_a = best_a_loc.copy()
                    best_n = n
                    print(f"[global best after subgrad] new best = {best_val:.6f} (n={n})")

    # ----------------------------------------------------------------------
    # Fallback – in the extremely unlikely case that nothing was found.
    # ----------------------------------------------------------------------
    if best_a is None:
        best_n = 4000
        best_a = np.full(best_n, 1.0 / best_n)
        best_val = 2.0 * best_n * np.max(_full_convolution(best_a))

    # ----------------------------------------------------------------------
    # Final check via the provided evaluator (ensures all guards are passed)
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
