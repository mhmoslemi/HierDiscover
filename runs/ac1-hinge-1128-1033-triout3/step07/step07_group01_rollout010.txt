def propose_candidate(seed=123, budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine n
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        n = old.shape[0]
    except Exception:
        n = 5000  # choose larger by default

    rng = np.random.default_rng(seed)

    # generate a few initial vectors (list)
    candidates = []
    # Uniform
    candidates.append(np.full(n, 1.0/n))
    # Dirichlet alpha 0.5 (more dispersion)
    candidates.append(rng.gamma(shape=0.5, scale=1.0, size=n))
    # Dirichlet alpha 1
    candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    # Dirichlet alpha 2
    candidates.append(rng.gamma(shape=2.0, scale=1.0, size=n))
    # if old exists, include normalized old
    if 'old' in locals():
        candidates.append(old / old.sum())
    # Simple antisymmetric reorder version of uniform
    candidates.append(_anti_symmetric_reorder(np.full(n, 1.0/n)))

    # normalize candidates to sum 1
    candidates = [c / c.sum() for c in candidates]

    # pick best based on fast evaluate
    best = None
    best_val = float('inf')
    for i,c in enumerate(candidates):
        v = _fast_evaluate(c)
        # print
        print(f"[init {i}] value={v:.6f}")
        if v < best_val:
            best_val = v
            best = c.copy()
    print(f"[init] best value={best_val:.6f}")

    # Phase A: mirror descent (subgradient) to reduce max conv
    a = best.copy()
    it = 0
    # step size schedule: start high then decay
    alpha0 = 0.5
    decay = 0.995
    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        # compute gradient for max term
        grad = np.zeros_like(a)
        # i range: max(0, max_idx-(n-1)) to min(n-1, max_idx)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx = np.arange(i_start, i_end+1)
        jdx = max_idx - idx
        grad[idx] = a[jdx]

        # step size adapt
        alpha = alpha0 * (decay ** (it // 200))
        # exponential update
        a = a * np.exp(-alpha * grad)

        # project onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 1e-12:
            a = np.full(n, 1.0/n)
            s = a.sum()
        a /= s

        # evaluate
        cur_val = _fast_evaluate(a)
        if cur_val < best_val - 1e-8:
            best_val = cur_val
            best = a.copy()
            print(f"[mirror {it}] improved val={best_val:.6f}")

        # occasional reordering
        if it % 500 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best = a_ord.copy()
                a = a_ord.copy()
                print(f"[reorder {it}] improved val={best_val:.6f}")

        it += 1
        if time.time() > deadline - 2.0:
            break

    # Phase B: permutation simulated annealing
    cur = best.copy()
    cur_val = best_val
    # Precompute reversed indices for fast product? We'll just recompute conv.
    T = 0.02
    cooling = 0.9999
    while time.time() < deadline - 0.5:
        # propose a swap of two positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        # evaluate
        new_val = _fast_evaluate(new)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))
        if accept:
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
                print(f"[perm SA] improved to {best_val:.6f}")
        T = max(T * cooling, 1e-6)

    # final evaluation via official evaluate_sequence
    try:
        final_score = evaluate_sequence(best.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence error:", e)

    return best.tolist()

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    The result has length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` for vectors that already sum
    to 1.  Returns 2·n·max(conv) where conv = a*a.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    # a already sums to 1 → denominator = 1
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values on the left side and the smallest on the
    right side – a cheap heuristic that tends to flatten the central
    convolution peak.

    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending

        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:

    return new_a
