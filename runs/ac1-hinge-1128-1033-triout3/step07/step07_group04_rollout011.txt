def propose_candidate(seed=123, budget_s=1000):

          if len(a) != n: a = np.interp(np.linspace(0,1,n), np.linspace(0,1,len(a)), a) 
    except: a = np.full(n, 1.0/n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _eval_fast(best_a)
    print('[init] evaluation', best_val)

    # Define move functions
    def swap_move(seq):
        i,j = rng.integers(0,n, size=2)
        if i==j: return seq
        new = seq.copy()
        new[i], new[j] = new[j], new[i]
        return new
    def transfer_move(seq):
        i,j = rng.integers(0,n, size=2)
        if i==j or seq[i]==0: return seq
        frac = rng.random() * 0.1    # up to 10% mass transfer
        delta = seq[i] * frac
        new = seq.copy()
        new[i] -= delta
        new[j] += delta
        return new
    def block_shuffle_move(seq):
        L = rng.integers(2, min(50, n//10))
        start_idx = rng.integers(0, n-L+1)
        block = seq[start_idx:start_idx+L].copy()
        rng.shuffle(block)
        new = seq.copy()
        new[start_idx:start_idx+L] = block
        return new
    def targeted_move(seq, conv):
        d = int(np.argmax(conv))
        # find i maximizing a_i * a_{d-i}
        best_prod = -1.0
        best_i = None
        for i in range(max(0, d-(n-1)), min(n-1, d)+1):
            j = d - i
            prod = seq[i] * seq[j]
            if prod > best_prod:
                best_prod = prod
                best_i = i
                best_j = j
        if best_i is None:
            return seq
        # choose source = index with larger value
        source = best_i if seq[best_i] >= seq[best_j] else best_j
        target = rng.integers(0, n)
        if source == target:
            target = (target+1)%n
        frac = rng.random() * 0.1
        delta = seq[source] * frac
        new = seq.copy()
        new[source] -= delta
        new[target] += delta
        return new

    # Simulated annealing schedule
    T0 = best_val * 0.05 + 1e-6
    T = T0
    decay = 0.99995

    while time.time() < deadline:
        move_type = rng.random()
        if move_type < 0.25:
            cand = swap_move(best_a)
        elif move_type < 0.5:
            cand = transfer_move(best_a)
        elif move_type < 0.75:
            cand = block_shuffle_move(best_a)
        else:

        val = _eval_fast(cand)
        delta = val - best_val
        if delta < 0 or rng.random() < np.exp(-delta/T):
            best_a = cand
            best_val = val

"""
Hybrid optimiser for the step‑function convolution problem (improved).

Main ideas
-----------
1. **Diversified seeding** – geometric, power‑law and random Dirichlet
   families, each followed by the “anti‑symmetric’’ reorder that pairs
   large values with small ones (the optimal ordering for a given set of
   magnitudes).

2. **Heavy‑weight L‑BFGS** – a long smooth‑surrogate optimisation.  The
   surrogate replaces the max‑operator by a log‑sum‑exp with a decreasing
   smoothing parameter ε.  For every ε we run a full L‑BFGS optimisation
   (max 500 iterations) from several random restarts and try a few
   deterministic reorderings after each pass.

3. **Powerful simulated‑annealing** – temperature‑controlled search that
   mixes four move types:
      * swap two positions,
      * transfer a small fraction of mass,
      * block‑shuffle a short contiguous segment,
      * a targeted transfer that reduces the currently maximal convolution
        entry.
   The schedule is long and the temperature decays slowly.

4. **Final Adam‑style gradient descent** – after annealing we keep the
   best sequence and run a few passes of gradient descent on the
   smoothed objective with a geometric schedule of ε down to 1e‑14.

All heavy work is performed with FFT‑based convolution (`_fft_convolve_self`),
which evaluates the true objective (`_eval_fast`) in ~O(n log n) time.
The whole routine respects the 1000 s budget; the best sequence found is
returned as a plain ``list`` of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global parameters used by the smooth surrogate (updated on the fly)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT of size power‑of‑2."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # tiny negative round‑off artefacts are clamped
    np.maximum(conv, 0.0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` for non‑negative vectors."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε = 2 n max_ε(conv) and its gradient w.r.t. the
    soft‑max parameters w (where a_i = softmax(w)_i).
    """
    # soft‑max (ensures a ≥ 0, sum a = 1)
    shift = w - np.max(w)
    a = np.exp(shift)
    a /= np.sum(a)

    # autocorrelation a ★ a
    conv = _fft_convolve_self(a)
    max_c = np.max(conv)

    # log‑sum‑exp approximation of the max
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form, see original code)
    weights = np.exp(shifted)                     # proportional to exp((c_i−max)/ε)
    p = weights / np.sum(weights)               # a probability distribution over offsets
    p_rev = p[::-1]                              # needed for the convolution
    conv_ap = np.fft.irfft(np.fft.rfft(a) * np.fft.rfft(p_rev, _next_pow2(len(a) + len(p_rev) - 1)),
                           _next_pow2(len(a) + len(p_rev) - 1))[:2 * n - 1]
    seg = conv_ap[n - 1: 2 * n - 1]               # central n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper needed by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large ↔ small pairing (the classic anti‑symmetric pattern)."""
    n = a.shape[0]
    s = np.sort(a)
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[n - 1 - i]
        out[n - 1 - i] = s[i]
    if n % 2:
        out[half] = s[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    s = np.sort(a)
    left, right = 0, len(s) - 1
    out = np.empty_like(a)
    idx = 0
    while left <= right:
        out[idx] = s[right]
        idx += 1
        right -= 1
        if left <= right:
            out[idx] = s[left]
            idx += 1
            left += 1
    return out


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Simple descending sort."""
    return np.sort(a)[::-1]


def _bitrev_reorder(a: np.ndarray) -> np.ndarray:
    """Bit‑reversal ordering (helps to spread large values)."""
    n = a.shape[0]
    bits = (n - 1).bit_length()
    order = np.arange(n)
    rev = np.array([int('{:0{b}b}'.format(i, b=bits)[::-1], 2) for i in order])
    # some indices may exceed n‑1 after reversal when n is not a power of two;
    # we wrap them back.
    rev = rev % n
    return a[rev]


_REORDER_FUNCS = (
    _anti_symmetric_reorder,
    _alternating_reorder,
    _descending_reorder,
    _bitrev_reorder,
)


# ----------------------------------------------------------------------
# Parametric families for seeding
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """a_i ∝ q**i, 0 < q < 1, normalised to sum = 1."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q ** n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)**p, p < 0, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

def _simulated_annealing(a: np.ndarray, cur_val: float, deadline: float):
    """Temperature‑controlled search mixing swaps, transfers, block shuffles
    and a targeted reduction of the current maximal convolution entry."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # start temperature: a modest fraction of the current value
    T = max(cur_val * 0.05, 1e-7)
    decay = 0.99992
    iteration = 0

    while time.time() < deadline:
        move_type = rng.random()
        if move_type < 0.3:                       # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif move_type < 0.55:                    # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * rng.random() * 0.07
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        elif move_type < 0.8:                     # block shuffle (size 5‑30)
            L = int(rng.integers(3, min(30, n // 10) + 1))
            start = rng.integers(0, n - L + 1)
            new_a = a.copy()
            block = new_a[start:start + L]
            rng.shuffle(block)
            new_a[start:start + L] = block

        else:                                      # targeted transfer
            conv = _fft_convolve_self(a)
            d = int(np.argmax(conv))
            # find the pair (i,j) contributing most to this peak
            i_start = max(0, d - (n - 1))
            i_end = min(d, n - 1)
            best_prod = -1.0
            best_i = best_j = None
            for i in range(i_start, i_end + 1):
                j = d - i
                prod = a[i] * a[j]
                if prod > best_prod:
                    best_prod = prod
                    best_i, best_j = i, j
            if best_i is None:
                # fallback to a regular transfer
                i, j = rng.integers(0, n, size=2)
                if i == j or a[i] == 0.0:
                    continue
                delta = a[i] * rng.random() * 0.07
                new_a = a.copy()
                new_a[i] -= delta
                new_a[j] += delta
            else:
                # move a fraction from the larger partner to a random site
                src = best_i if a[best_i] >= a[best_j] else best_j
                tgt = rng.integers(0, n)
                if src == tgt:
                    tgt = (tgt + 1) % n
                delta = a[src] * rng.random() * 0.07
                new_a = a.copy()
                new_a[src] -= delta
                new_a[tgt] += delta

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0.0 or rng.random() < np.exp(-delta_val / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (it {iteration})")
        T = max(T * decay, 1e-12)
        iteration += 1

    return best_seq, best_val

# Simple Adam‑style gradient descent on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_descent(a: np.ndarray, eps_schedule, steps_per_eps, deadline):
    """Run a few Adam‑style steps for each ε in the schedule."""
    n = a.shape[0]
    w = np.log(a + 1e-16)          # initialise logits
    best_a = a.copy()
    best_val = _eval_fast(a)

    # Adam hyper‑parameters
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    for eps in eps_schedule:
        global _GLOBAL_EPS
        _GLOBAL_EPS = eps
        for _ in range(steps_per_eps):
            if time.time() >= deadline:
                return best_a, best_val
            t += 1
            L, grad = _smooth_obj_and_grad(w, eps, n)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # re‑project to the probability simplex via soft‑max
            a = np.exp(w - np.max(w))
            a /= a.sum()

            cur = _eval_fast(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[Adam] eps {eps:.1e} step {t} best = {best_val:.6f}")

        # after finishing a given ε, re‑initialise the optimizer state from the best point
        w = np.log(best_a + 1e-16)
        m.fill(0.0)
        v.fill(0.0)
        t = 0

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size (feel free to adjust – 4000 worked well historically)
    # --------------------------------------------------------------
    n = 4000
    global _GLOBAL_N
    _GLOBAL_N = n

    # --------------------------------------------------------------
    # Load previous construction if available
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] eval = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Hand‑crafted seeds (geometric, power‑law, random Dirichlet)
    # --------------------------------------------------------------
    seed_candidates = []

    for q in np.linspace(0.7, 0.999, 25):
        seed_candidates.append(_geometric_sequence(q, n))

    for p in np.linspace(-3.5, -0.2, 25):
        seed_candidates.append(_power_sequence(p, n))

    for _ in range(40):
        rnd = np.random.rand(n)
        rnd /= rnd.sum()
        seed_candidates.append(rnd)

    for cand in seed_candidates:
        val = _eval_fast(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            print(f"[seed] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Heavy L‑BFGS optimisation on the smoothed surrogate
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    # allocate roughly 30 % of the budget to this stage
    stage_deadline = time.time() + min(0.30 * budget_s, time_left * 0.4)

    # a wide ε‑schedule – start coarse, finish extremely fine
    eps_schedule = np.logspace(np.log10(5e-2), np.log10(5e-13), num=20)

    while time.time() < stage_deadline:
        # random restart
        w = np.random.randn(n) * 2.0
        for eps in eps_schedule:
            if time.time() >= stage_deadline:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps
            # L‑BFGS – give it enough budget
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': 500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            a = np.exp(w - np.max(w))
            a /= a.sum()

            cur = _eval_fast(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[L‑BFGS] eps {eps:.1e} new best = {best_val:.6f}")

            # try deterministic reorderings (they sometimes break symmetry)
            for reorder in _REORDER_FUNCS:
                a_r = reorder(a)
                v_r = _eval_fast(a_r)
                if v_r < best_val:
                    best_val = v_r
                    best_a = a_r.copy()
                    print(f"[reorder] {reorder.__name__} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Extended simulated annealing (the bulk of remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 250.0:
        sa_deadline = time.time() + remaining - 120.0   # keep ~2 min for final polish
    else:
        sa_deadline = deadline - 15.0
    print(f"[SA] running until {sa_deadline:.2f}")
    best_a, best_val = _simulated_annealing(best_a, best_val, sa_deadline)

    # --------------------------------------------------------------
    # 4. Final Adam‑style descent on the smoothed surrogate
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        eps_fine = np.logspace(np.log10(5e-7), np.log10(5e-14), num=12)
        steps_per = 45
        print("[Adam] final refinement")
        best_a, best_val = _adam_descent(best_a, eps_fine, steps_per, deadline)

    # --------------------------------------------------------------
    # 5. Report the exact evaluation via the provided function
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] exact evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
