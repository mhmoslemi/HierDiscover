def _project_onto_simplex(v):
    """Projection onto the probability simplex."""
    # Implementation of Duchi et al.
    # Ensure v is 1-d numpy array.
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative list of heights) that minimises
``evaluate_sequence``.  The search combines:

* a diverse set of handcrafted seeds (geometric, power‑law,
  two‑level, random Dirichlet);
* a fast projected sub‑gradient descent that directly reduces the
  maximal convolution entry; the iterate is kept on the probability
  simplex and sorted in descending order (large values on the left,
  small on the right) – this exploits the rearrangement inequality;
* a short simulated‑annealing phase (the same moves as the reference
  code) to escape possible local minima;
* a final hill‑climbing polish.

The whole routine respects the 1000 s budget and returns the best
sequence found (as a plain ``list[float]``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution helpers (identical to the reference implementation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using a real FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clamp tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast replica of the supplied ``evaluate_sequence`` (non‑negative a).
    Returns 2·n·max(conv)/sum(a)^2.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:          # this should never happen (sum is 1)
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (Duchi et al.)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a real vector onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑d arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return a uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """a_i ∝ q^i, 0<q<1, normalised to sum = 1."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q ** n) / (1.0 - q)
    a = (q ** np.arange(n, dtype=float)) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^p, p<0, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _two_level_sequence(t: float, m: int, n: int) -> np.ndarray:
    """
    First m entries carry total mass t, the remaining n‑m entries carry 1‑t.
    Both blocks are constant, resulting in a decreasing two‑step shape.
    """
    if m <= 0 or m >= n:
        return np.full(n, 1.0 / n)
    x = t / m
    y = (1.0 - t) / (n - m)
    a = np.empty(n, dtype=float)
    a[:m] = x
    a[m:] = y
    return a


# ----------------------------------------------------------------------
# Re‑ordering heuristics (kept for occasional use)
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Largest left, smallest right (pairwise)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = sorted_vals[n - 1 - i]
        out[n - 1 - i] = sorted_vals[i]
    if n % 2:
        out[half] = sorted_vals[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    sorted_vals = np.sort(a)
    left, right = 0, len(sorted_vals) - 1
    out = np.empty_like(a)
    idx = 0
    while left <= right:
        out[idx] = sorted_vals[right]
        idx += 1
        right -= 1
        if left <= right:
            out[idx] = sorted_vals[left]
            idx += 1
            left += 1
    return out


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Simple descending sort."""
    return np.sort(a)[::-1]

# Simulated annealing (copied from the reference code – works well)
# ----------------------------------------------------------------------
def _simulated_annealing_extended(a: np.ndarray, cur_val: float,
                                 deadline: float) -> (np.ndarray, float):
    """SA mixing swaps, small transfers and a targeted transfer."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.05, 1e-6)   # start temperature
    decay = 0.9999
    it = 0

    while time.time() < deadline:
        move = rng.random()
        if move < 0.4:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif move < 0.8:                   # random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * 0.05 * rng.random()
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        else:                               # targeted transfer
            conv = _fft_convolve_self(a)
            k_max = int(np.argmax(conv))
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_best, prod_best = None, -1.0
            for i in range(i_start, i_end + 1):
                j = k_max - i
                prod = a[i] * a[j]
                if prod > prod_best:
                    prod_best = prod
                    i_best, j_best = i, j
            if i_best is None:
                # fallback
                i, j = rng.integers(0, n, size=2)
                if i == j or a[i] == 0.0:
                    continue
                delta = a[i] * 0.05 * rng.random()
                new_a = a.copy()
                new_a[i] -= delta
                new_a[j] += delta
            else:
                source = i_best if a[i_best] >= a[j_best] else j_best
                target = rng.integers(0, n)
                if source == target:
                    continue
                delta = a[source] * 0.05 * rng.random()
                new_a = a.copy()
                new_a[source] -= delta
                new_a[target] += delta

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        T = max(T * decay, 1e-12)
        it += 1

    return best_seq, best_val

# Projected sub‑gradient step that directly minimises max‑convolution
# ----------------------------------------------------------------------
def _gd_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One step of sub‑gradient descent on max(conv) under the simplex.
    a is assumed to be a probability vector (sum = 1, a_i ≥ 0).
    The iterate is projected back onto the simplex and sorted descending.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    k = int(np.argmax(conv))                 # index of the current worst peak
    idx = k - np.arange(n)                   # want a_{k-i}
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]             # gradient of conv_k wrt a

    # simple additive update then projection
    a_new = a - eta * g
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_onto_simplex(a_new)
    # keep the decreasing order – this is the optimal arrangement for a fixed multiset
    a_new = np.sort(a_new)[::-1]
    return a_new
