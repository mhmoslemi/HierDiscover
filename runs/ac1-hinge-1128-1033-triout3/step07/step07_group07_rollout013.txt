def propose_candidate(seed=42,budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8000

    # initialize w from near-uniform distribution
    a = np.full(n, 1.0/n)
    a = a + rng.normal(scale=0.01, size=n)
    a = np.clip(a, 0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= a_sum
    w = np.log(a + 1e-12)

    best_a = a.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))

    # Adam parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # schedule of smoothing parameters
    eps_vals = np.logspace(-2, -12, num=24)  # 24 values

    total_iters = 0
    for eps in eps_vals:
        # compute learning rate scaling: maybe lr = 0.5 * eps
        lr = 0.5 * eps
        # run for e.g. max_iter = int(5000 * (1 + np.log10(1/eps)))  # more for smaller eps
        max_iter = 2000
        for it in range(max_iter):
            if time.time() >= deadline:
                break
            # compute gradient
            _, grad_w = _smooth_obj_and_grad(w, eps, n)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_w
            v = beta2 * v + (1 - beta2) * (grad_w ** 2)
            t = total_iters + 1
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # evaluate current a
            a = np.exp(w - np.max(w))
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum

            # compute objective
            cur_val = 2.0 * n * np.max(_full_convolution(a))
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e} iter {total_iters}] new best = {best_val:.6f}")

            total_iters += 1

        if time.time() >= deadline:
            break

    # Fine subgradient descent
    if time.time() < deadline:
        best_a, best_val = _subgradient_descent(best_a, best_val, deadline, rng)

    # final evaluate
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()

"""
Propose a low‑upper‑bound step function.

Algorithm
---------
1. **Initialization** – start from a near‑uniform probability vector of
   length ``n`` (≈ 8000).  Small random perturbations break symmetry.
   The vector is represented in log‑space ``w`` so that the soft‑max
   automatically enforces non‑negativity and Σa = 1.

2. **Smoothed optimisation** – minimise a smooth “log‑sum‑exp”
   surrogate of the true objective

      L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε),

   where ``conv = a * a`` (full convolution).  The gradient w.r.t. the
   log‑parameters ``w`` is obtained analytically.  An Adam optimiser
   (first‑order, adaptive step size) is used.  The smoothing parameter
   ``ε`` is decreased geometrically (1e‑2 … 1e‑12) while performing many
   cheap gradient steps at each level.

3. **Fine sub‑gradient polishing** – after the smoothing phase a
   classic sub‑gradient method with projection onto the probability
   simplex is applied to the exact nonsmooth objective
   ``2·n·max(conv) / (∑a)^2``.  This often yields a final improvement.

All work respects the supplied time budget (default 1000 s).  The
best sequence found is returned as a plain Python list (the evaluation
framework will call ``evaluate_sequence`` on it)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negative values – clip.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient wrt log‑parameters w
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns the surrogate value and its gradient with respect to ``w``:

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε),

    where a = softmax(w) (Σa = 1, a_i ≥ 0) and conv = a * a.
    """
    # ----- a = softmax(w) (guarantees feasibility) -----
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = a.sum()
    a = a / a_sum

    # ----- convolution and max -----
    conv = _full_convolution(a)
    max_conv = conv.max()

    # ----- log‑sum‑exp smoothing -----
    shifted = (conv - max_conv) / eps
    weights = np.exp(shifted)                     # may underflow to 0 – fine
    p = weights / weights.sum()                   # probabilities over lags

    # ----- gradient wrt a (closed form) -----
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                # length n + n - 1 + n - 1 = 3n‑2
    seg = conv_ap[n - 1: 2 * n - 1]               # central n entries
    grad_a = 4.0 * n * seg[::-1]                  # derivative of L_ε w.r.t a

    # ----- chain rule through soft‑max (w -> a) -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    # ----- value of the surrogate (not used for optimisation) -----
    log_sum = np.logaddexp.reduce(shifted)        # log Σ exp(...)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum
    return L, grad_w


# ----------------------------------------------------------------------
# Projection onto the probability simplex (used by sub‑gradient descent)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fine sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         best_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Sub‑gradient descent on 2·n·max(conv) / (sum(a)**2) with projection.
    Returns the best vector and its corresponding value.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    best_a = cur_a.copy()
    eta = 0.1  # initial step size

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag k
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # gradient step + simplex projection
        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            eta = min(eta * 1.05, 0.5)
        else:
            eta = max(eta * 0.7, 1e-5)

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Optimises a non‑negative step function to minimise
    ``evaluate_sequence``.  The routine respects the given time budget
    and returns the best list of floats found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Problem size – a larger n gives a tighter theoretical bound.
    n = 8000

    # ------------------------------------------------------------------
    # Initialise sequence: try to reuse a previously found construction,
    # otherwise start from a near‑uniform distribution with tiny noise.
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        else:
            a0 = a0.copy()
    except Exception:
        a0 = np.full(n, 1.0 / n) + rng.normal(scale=0.01, size=n)
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        if a0_sum == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / a0_sum

    # Parameterise by logits to keep the simplex constraints automatic.
    w = np.log(a0 + 1e-12)

    # Keep track of the best solution found so far.
    conv0 = _full_convolution(a0)
    best_val = 2.0 * n * np.max(conv0) / (a0.sum() ** 2)
    best_a = a0.copy()
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser state (first‑ and second‑moment estimates)
    # ------------------------------------------------------------------
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    total_steps = 0

    # ------------------------------------------------------------------
    # Smoothed optimisation – geometric schedule for ε, many gradient steps
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-2, -12, num=20)  # 1e‑2 … 1e‑12
    for eps in eps_schedule:
        # Number of inner iterations grows as ε shrinks (more precision needed).
        inner_iters = int(1500 * (1 + -np.log10(eps)))
        for _ in range(inner_iters):
            if time.time() >= deadline:
                break

            # Gradient of the smooth surrogate
            _, grad_w = _smooth_obj_and_grad(w, eps, n)

            # Adam update
            total_steps += 1
            m = beta1 * m + (1.0 - beta1) * grad_w
            v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
            m_hat = m / (1.0 - beta1 ** total_steps)
            v_hat = v / (1.0 - beta2 ** total_steps)

            # Learning rate – modest and constant; Adam’s scaling handles magnitude.
            lr = 0.1
            w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # Recover the probability vector and evaluate the exact objective.
            a = np.exp(w - np.max(w))
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum
            cur_val = 2.0 * n * np.max(_full_convolution(a)) / (a_sum ** 2)

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[smooth eps {eps:.1e} step {total_steps}] new best = {best_val:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Optional random‑restart phase (few short re‑optimisations)
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0 and restart < 4:
        restart += 1
        # Perturb the current best logits and re‑run a short optimisation.
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.5, size=n)
        m = np.zeros_like(w)
        v = np.zeros_like(w)
        total_steps = 0
        for eps in eps_schedule[-5:]:   # only the finer ε values
            for _ in range(400):       # modest number of steps per ε
                if time.time() >= deadline:
                    break
                _, grad_w = _smooth_obj_and_grad(w, eps, n)
                total_steps += 1
                m = beta1 * m + (1.0 - beta1) * grad_w
                v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
                m_hat = m / (1.0 - beta1 ** total_steps)
                v_hat = v / (1.0 - beta2 ** total_steps)
                w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                a = np.exp(w - np.max(w))
                a_sum = a.sum()
                if a_sum == 0:
                    continue
                a /= a_sum
                cur_val = 2.0 * n * np.max(_full_convolution(a)) / (a_sum ** 2)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[restart {restart}] new best = {best_val:.6f}")

        # end inner loops
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fine sub‑gradient polishing (non‑smooth exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _subgradient_descent(best_a,
                                                best_val,
                                                deadline,
                                                rng)

    # ------------------------------------------------------------------
    # Final reporting (uses the provided evaluator)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
