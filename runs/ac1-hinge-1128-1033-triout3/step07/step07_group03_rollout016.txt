def _anneal(a, cur_val, n, deadline):
    best_a = a.copy()
    best_val = cur_val
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    # Parameters
    T = cur_val * 0.01   # or 0.1 maybe
    it = 0
    while time.time() < deadline - 5:
        it += 1
        # schedule temperature
        T = max(T * 0.9999, 1e-12)
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Choose delta: fraction of cur_a[i]
        if cur_a[i] == 0:
            continue
        frac = rng.random() * 0.5   # up to 50%
        delta = cur_a[i] * frac
        # For stability ensure delta not too huge
        # new_a
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # compute objective
        new_val = _objective_fast(new_a, n)
        diff = new_val - cur_val
        if diff < -1e-12 or rng.random() < np.exp(-diff / (T + 1e-12)):
            cur_a = new_a
            cur_val = new_val
            # update best
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[anneal iter {it}] new best = {best_val:.6f}")
    return best_a, best_val

"""
Search for better step function minimizing evaluate_sequence.
Algorithm:
1. Load previous best or initialise uniformly.
2. Run several L‑BFGS optimisations on a smooth max surrogate,
   using a fine ε‑schedule (down to 1e‑12) and many iterations.
3. Perform two mirror‑descent phases:
   a) multi‑peak exponential gradient descent (k=3) as before;
   b) single‑peak descent (k=1) with a smaller step size to polish.
4. Run a simple simulated‑annealing / mass‑transfer local search
   that focuses on the index of the current maximal convolution.
   The objective is computed with a fast FFT‑based convolution.
5. Return the best sequence discovered.
"""

import time, math, sys
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ---------- helper functions ----------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·len−1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv

def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective used in evaluate_sequence (sum(a)=1 is assumed)."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Based on algorithm by Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return uniform vector.
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place large values on the left half and small values on the right half."""
    n = a.shape[0]
    s = np.sort(a)               # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[-1 - i]               # largest → left side
        out[n - 1 - i] = s[i]            # smallest → right side
    if n % 2:
        out[half] = s[half]
    return out

# ---------- smoothed surrogate for L‑BFGS ----------
_GLOBAL_EPS = None
_GLOBAL_N = None

def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """Smooth approximation of 2·n·max(conv) and its gradient w.r.t. w."""
    # soft‑max to enforce non‑negativity and unit sum.
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = a.sum()
    a /= a_sum

    conv = _full_convolution(a)
    max_c = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # derivative w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()          # shape = len(conv)

    # convolution a * p_rev (equivalent to Σ_i a_i p_{k-i})
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]      # length n, derivative contribution
    grad_a = 4.0 * n * seg[::-1]        # factor 4 comes from derivative of a*a

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

def _smooth_obj_and_grad_wrapper(w):
    """Wrapper required by scipy.minimize – uses global eps, n."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

def _l_bfgs_optimize(w_init: np.ndarray, deadline: float, max_restarts: int = 6):
    """Run several L‑BFGS optimisations with a fine ε‑schedule."""
    n = w_init.shape[0]
    best_a = None
    best_val = float('inf')
    restart = 0
    while time.time() < deadline - 30 and restart < max_restarts:
        restart += 1
        # start from current best with small Gaussian noise
        if best_a is not None:
            w0 = np.log(best_a + 1e-14)
        else:
            w0 = np.log(w_init + 1e-14)
        rng = np.random.default_rng()
        w0 += rng.normal(scale=0.15, size=n)

        # ε‑schedule – from 5·10⁻³ down to 10⁻¹² (geometric)
        eps_schedule = np.logspace(np.log10(5e-3), -12, num=10)
        w = w0.copy()
        for eps in eps_schedule:
            if time.time() >= deadline - 15:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x

            # map back to a (simplex)
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()

            cur_val = _objective(a, n)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[L‑BFGS restart {restart} eps {eps:.1e}] new best = {best_val:.6f}")

            # optional anti‑symmetric heuristic after each ε
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _objective(a_ord, n)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

        # early exit if time is low
    # fallback
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift); a /= a.sum()
        best_a = a
        best_val = _objective(a, n)
    return best_a, best_val

# ---------- exponential mirror descent phases ----------
def _exponential_descent(a_start: np.ndarray, deadline: float,
                         top_k: int = 3, alpha0: float = 0.5,
                         decay: float = 0.98):
    """Exponential mirror‑descent using the k largest convolution peaks."""
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(a, n)
    it = 0
    while time.time() < deadline - 5:
        conv = _full_convolution(a)
        # choose top_k peaks
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]
        else:
            idxs = np.arange(conv.size)

        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            partner = k - idx_range
            g[idx_range] += a[partner]

        alpha = alpha0 * (decay ** (it // 2000))
        a = a * np.exp(-alpha * g)

        # projection onto simplex
        a = _project_onto_simplex(a)

        if it % 200 == 0:
            cur_val = _objective(a, n)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[exp‑descent iter {it}] new best = {best_val:.6f}")
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _objective(a_ord, n)
                if val_ord < best_val - 1e-12:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[exp‑descent reorder] improved to {best_val:.6f}")
        it += 1
    return best_a, best_val

def _single_peak_descent(a_start: np.ndarray, deadline: float,
                         alpha0: float = 0.3, decay: float = 0.995):
    """Focused descent on the single (currently maximal) convolution peak."""
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline - 5:
        conv = _full_convolution(a)
        k_max = int(conv.argmax())
        max_val = conv.max()
        # compute partner indices for this shift
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idxs = np.arange(i_start, i_end + 1, dtype=int)
        partner = k_max - idxs
        # gradient contribution for this peak only
        g = np.zeros_like(a)
        g[idxs] = 2.0 * a[partner]   # factor 2 is exact derivative of a*a

        alpha = alpha0 * (decay ** (it // 1000))
        a = a * np.exp(-alpha * g)

        a = _project_onto_simplex(a)

        if it % 100 == 0:
            cur_val = _objective(a, n)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[single‑peak iter {it}] new best = {best_val:.6f}")
            if it % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _objective(a_ord, n)
                if val_ord < best_val - 1e-12:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[single‑peak reorder] improved to {best_val:.6f}")
        it += 1
    return best_a, best_val

# ---------- simple simulated‑annealing local search ----------
def _simulated_annealing(a_start: np.ndarray, deadline: float):
    """Mass‑transfer annealing that focuses on the index of the current maximum."""
    n = a_start.shape[0]
    rng = np.random.default_rng()
    a = a_start.copy()
    cur_val = _objective(a, n)
    best_a = a.copy()
    best_val = cur_val
    # initial temperature relative to current value
    T = cur_val * 0.02
    iteration = 0
    while time.time() < deadline - 2:
        iteration += 1
        # decay temperature slowly
        T = max(T * 0.9995, 1e-12)

        # current maximal convolution index
        conv = _full_convolution(a)
        k_max = int(conv.argmax())

        # locate index contributing most to this peak
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idxs = np.arange(i_start, i_end + 1, dtype=int)
        partners = k_max - idxs
        products = a[idxs] * a[partners]
        i_high = idxs[np.argmax(products)]

        # propose a move: transfer a fraction of a[i_high] to a random j
        j = rng.integers(0, n)
        if j == i_high:
            continue
        if a[i_high] <= 0:
            continue
        frac = rng.random() * 0.25   # up to 25 % of the mass at i_high
        delta = a[i_high] * frac
        new_a = a.copy()
        new_a[i_high] -= delta
        new_a[j] += delta

        # evaluate
        new_val = _objective(new_a, n)
        diff = new_val - cur_val
        if diff < -1e-14 or rng.random() < math.exp(-diff / (T + 1e-18)):
            a = new_a
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[anneal {iteration}] new best = {best_val:.6f}")

        # occasional re‑ordering heuristic (every 2000 accepted moves)
        if iteration % 2000 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _objective(a_ord, n)
            if val_ord < best_val - 1e-12:
                best_val = val_ord
                best_a = a_ord.copy()
                a = a_ord.copy()
                cur_val = val_ord
                print(f"[anneal reorder] improved to {best_val:.6f}")

    return best_a, best_val

"""
Search for a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Load the previously discovered construction (``height_sequence_1``) if it
   exists, otherwise start from the uniform distribution.
2. Run several L‑BFGS optimisations on a smooth surrogate of
   ``2·n·max(conv)``.  The ε‑schedule goes down to 1e‑12 and the optimiser
   is allowed many iterations, giving a high‑quality starting point.
3. Apply two mirror‑descent polishing phases:
   * a multi‑peak exponential update (k=3) – the same idea as in the
     reference implementation,
   * a single‑peak focused update that only attacks the currently maximal
     convolution index.
4. Finish with a cheap simulated‑annealing / mass‑transfer local search.
   Each move transfers a fraction of the mass from the index that
   contributes most to the current maximal peak to a random destination.
   The objective is evaluated with a fast FFT‑based convolution, and a
   temperature schedule allows occasional uphill moves.
5. Return the best sequence found.  The final evaluation is performed with
   the provided ``evaluate_sequence`` function (which also clips values).

All steps respect the 1000 s budget and keep the sequence on the
probability simplex (non‑negative, sum = 1) – the objective is
scale‑invariant, so this does not affect the result.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)            # remove tiny negatives
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective used by ``evaluate_sequence`` when sum(a)=1."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Wang & Carreira‑Perpiñán (2013) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place large values on the left half and small values on the right."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[-1 - i]             # largest → left side
        out[n - 1 - i] = s[i]          # smallest → right side
    if n % 2:
        out[half] = s[half]
    return out


# ----------------------------------------------------------------------
# Smoothed surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None
_GLOBAL_N = None


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate:
        L = 2·n·max(conv) + 2·n·ε·log‑sum‑exp((conv‑max)/ε)

    ``w`` are soft‑max parameters (a_i = exp(w_i)/∑exp(w)).
    Returns (L, ∇_w L).
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)
    max_c = conv.max()

    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (derived from the smoothed max term)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # probability over lag indices

    # convolution a * p_rev
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to derivative ∂L/∂a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                    # factor 4 from d(a*a)/da

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w):
    """Wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize(w_init: np.ndarray, deadline: float, max_restarts: int = 6):
    """
    Several L‑BFGS runs on the smoothed surrogate.
    Uses an ε‑schedule down to 1e‑12 and many inner iterations.
    Returns the best (a, objective) pair found.
    """
    n = w_init.shape[0]
    best_a = None
    best_val = float('inf')
    restart = 0

    while time.time() < deadline - 30 and restart < max_restarts:
        restart += 1
        # start from the current best with a small Gaussian perturbation
        if best_a is not None:
            w0 = np.log(best_a + 1e-14)
        else:
            w0 = np.log(w_init + 1e-14)
        rng = np.random.default_rng()
        w0 += rng.normal(scale=0.15, size=n)

        # ε‑schedule – geometric from 5·10⁻³ to 10⁻¹²
        eps_schedule = np.logspace(np.log10(5e-3), -12, num=10)

        w = w0.copy()
        for eps in eps_schedule:
            if time.time() >= deadline - 15:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x

            # map back to the simplex
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()

            cur_val = _objective(a, n)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[L‑BFGS restart {restart} eps {eps:.1e}] new best = {best_val:.6f}")

            # occasional anti‑symmetric re‑ordering
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _objective(a_ord, n)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = _objective(a, n)

    return best_a, best_val

# ----------------------------------------------------------------------
def _exponential_descent(a_start: np.ndarray, deadline: float,
                         top_k: int = 3, alpha0: float = 0.5,
                         decay: float = 0.98):
    """Mirror‑descent using the ``top_k`` largest convolution peaks."""
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline - 5:
        conv = _full_convolution(a)

        # indices of the ``top_k`` largest convolution entries
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]
        else:
            idxs = np.arange(conv.size)

        # gradient contribution from the selected peaks
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            partner = k - idx_range
            g[idx_range] += a[partner]

        alpha = alpha0 * (decay ** (it // 2000))
        a = a * np.exp(-alpha * g)
        a = _project_onto_simplex(a)

        if it % 200 == 0:
            cur_val = _objective(a, n)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[exp‑descent iter {it}] new best = {best_val:.6f}")
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _objective(a_ord, n)
                if val_ord < best_val - 1e-12:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[exp‑descent reorder] improved to {best_val:.6f}")

        it += 1
    return best_a, best_val


def _single_peak_descent(a_start: np.ndarray, deadline: float,
                         alpha0: float = 0.3, decay: float = 0.995):
    """Mirror‑descent that only attacks the currently maximal convolution peak."""
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline - 5:
        conv = _full_convolution(a)
        k_max = int(conv.argmax())

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idxs = np.arange(i_start, i_end + 1, dtype=int)
        partner = k_max - idxs

        g = np.zeros_like(a)
        g[idxs] = 2.0 * a[partner]          # exact derivative of a*a w.r.t. this peak

        alpha = alpha0 * (decay ** (it // 1000))
        a = a * np.exp(-alpha * g)
        a = _project_onto_simplex(a)

        if it % 100 == 0:
            cur_val = _objective(a, n)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[single‑peak iter {it}] new best = {best_val:.6f}")
            if it % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _objective(a_ord, n)
                if val_ord < best_val - 1e-12:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[single‑peak reorder] improved to {best_val:.6f}")

        it += 1
    return best_a, best_val

# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray, deadline: float):
    """
    Mass‑transfer annealing that concentrates on the index of the current
    maximal convolution peak.  Moves shift a fraction of the mass from the
    point that contributes most to that peak to a random destination.
    """
    n = a_start.shape[0]
    rng = np.random.default_rng()
    a = a_start.copy()
    cur_val = _objective(a, n)
    best_a = a.copy()
    best_val = cur_val

    T = cur_val * 0.02                     # initial temperature
    it = 0
    while time.time() < deadline - 2:
        it += 1
        T = max(T * 0.9995, 1e-12)        # slow exponential cooling

        conv = _full_convolution(a)
        k_max = int(conv.argmax())

        # Find the index that contributes most to this peak
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idxs = np.arange(i_start, i_end + 1, dtype=int)
        partners = k_max - idxs
        products = a[idxs] * a[partners]
        i_high = idxs[np.argmax(products)]

        # Propose a transfer of a fraction of a[i_high] to a random location
        j = rng.integers(0, n)
        if j == i_high or a[i_high] <= 0:
            continue
        frac = rng.random() * 0.25            # up to 25 % of the mass
        delta = a[i_high] * frac
        new_a = a.copy()
        new_a[i_high] -= delta
        new_a[j] += delta

        new_val = _objective(new_a, n)
        diff = new_val - cur_val
        if diff < -1e-14 or rng.random() < math.exp(-diff / (T + 1e-18)):
            a = new_a
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[anneal {it}] new best = {best_val:.6f}")

        # Every few thousand accepted moves try the anti‑symmetric heuristic
        if it % 2000 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _objective(a_ord, n)
            if val_ord < best_val - 1e-12:
                best_val = val_ord
                best_a = a_ord.copy()
                a = a_ord.copy()
                cur_val = val_ord
                print(f"[anneal reorder] improved to {best_val:.6f}")

    return best_a, best_val
