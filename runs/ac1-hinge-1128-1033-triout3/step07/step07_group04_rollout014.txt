    return g0

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm (inspired by the LP‑based scheme mentioned in the statement)

1. Initialise a non‑negative vector a of length n (n = 4000) with
   sum(a) = sqrt(2n).  Several random seeds are tried (uniform, Dirichlet).

2. Iteratively improve a:
   * Compute the full linear convolution a*a and its maximum M.
   * Solve the linear programme
         maximise   1ᵀg
         subject to A(g) ≤ M ,  g ≥ 0
     where A(g) = f * g is the convolution with the current vector f = a.
     A(g) ≤ M is a set of (2n‑1) linear inequalities, which we
     implement with a sparse Toeplitz matrix (pre‑computed index pattern).
   * Rescale the LP solution g so that sum(g) = sqrt(2n) and mix
         a ← (1‑t)·a + t·g   (t = 0.3, later reduced if needed).
   * Project onto the simplex (non‑negative, prescribed sum) to avoid
     numerical drift and enforce a small floor (1e‑9) – this keeps the
     LP bounded.
   * Keep the best vector (lowest evaluation) seen so far.

   The loop terminates when the time budget (≈1000 s) is exhausted.

3. A cheap local hill‑climbing phase finishes the optimisation:
   random swaps and small mass‑transfer moves are accepted if they
   improve the evaluation.

All heavy work uses an FFT‑based fast evaluator that reproduces the
behaviour of ``evaluate_sequence``.  The final sequence is returned as a
plain Python list of floats.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution utilities (identical to those used in the previous code)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # remove tiny negatives caused by numerical noise
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` for a non‑negative vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Pre‑computed sparse matrix pattern for the convolution constraints
# ----------------------------------------------------------------------
def _precompute_conv_pattern(n: int):
    """
    Return (row_idx, col_idx) for the sparse matrix that represents
    the linear map g ↦ f * g (where f will be supplied later).
    The pattern is independent of f; only the data values change.
    """
    # rows = offset + column, where offset = 0..n-1
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    rows = rows.astype(np.int32)
    cols = cols.astype(np.int32)
    return rows, cols


# ----------------------------------------------------------------------
# Linear programme: maximise sum(g) under f * g ≤ M
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float, rows: np.ndarray,
                      cols: np.ndarray, n: int):
    """
    Solve
        maximise   Σ g_i
        subject to (f * g)[k] ≤ M   for all k,
                  g_i ≥ 0.
    Returns the optimal vector g (may be all zeros if the LP fails).
    """
    # Build sparse matrix A where A @ g = f * g
    data = np.repeat(f, n)               # length n * n
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)    # linprog minimises, so negate
    bounds = [(0, None)] * n

    # HiGHS (default in recent scipy) handles sparse matrices efficiently
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        # a fallback – return a zero vector (the LP is always feasible)
        return np.zeros_like(f)
    return res.x


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (swap / transfer)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, cur_val: float, deadline: float) -> (np.ndarray,
                                                                    float):
    """Very cheap local improvement using swaps and small transfers."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = cur_val

    while time.time() < deadline:
        # decide move type (20 % swap, otherwise transfer)
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-12:
                continue
            delta = best_a[i] * 0.05 * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < best_val:
            best_a = new_a
            best_val = val
            # keep the sequence normalised (tiny drift may appear)
            best_a = np.maximum(best_a, 0.0)
            best_a = best_a / best_a.sum() * np.sqrt(2 * n)

    return best_a, best_val

    """Search for a step‑function minimising ``evaluate_sequence``."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    sum_target = np.sqrt(2.0 * n)      # normalisation used by the theory
    rows, cols = _precompute_conv_pattern(n)

    # --------------------------------------------------------------
    # Helper to project a vector onto the simplex with a tiny floor
    # --------------------------------------------------------------
    def _project(x):

        x = x / x.sum() * sum_target
        return x

    # --------------------------------------------------------------
    # Create a few diverse initial seeds and keep the best one
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")
