import numpy as np
from scipy.sparse import diags

def build_A(a):
    n = len(a)
    rows = 2*n-1
    diags_data = []
    offsets = []
    for j in range(n):
        diag_len = min(rows - j, n)
        # vector of length diag_len with constant a[j]
        diags_data.append(np.full(diag_len, a[j]))
        offsets.append(j)
    A = diags(diags_data, offsets, shape=(rows, n), format='csr')
    return A

"""
Hybrid optimizer for the step‑function evaluation problem.

The algorithm runs five consecutive phases within the 1000 s budget:

1. **Smooth surrogate L‑BFGS** – minimise a log‑sum‑exp surrogate of the
   ℓ∞‑norm of the self‑convolution while the vector stays on the
   ℓ₁‑simplex (soft‑max parametrisation).  Several restarts and a
   decreasing smoothing schedule are used.

2. **Deterministic re‑ordering** – after each L‑BFGS run the multiset of
   heights is reordered by two heuristics:
   * “ends‑large” (large, small, large, …) and
   * “alternating‑extremes” (largest ↦ left, second‑largest ↦ right,
     third‑largest ↦ next‑left, …).
   Both often lower the peak of the self‑convolution.

3. **Linear‑programming descent** – for the current vector `a` compute
   `M = max(a*a)`.  Solve the LP  

        maximise 1ᵀg
        subject to a ⊛ g ≤ M   (component‑wise)
                    g ≥ 0

   (the convolution matrix is built as a sparse Toeplitz matrix).  The
   optimiser `g` is normalised to unit ℓ₁‑norm and mixed with `a`
   (`a ← (1‑t)a + t g`) with a modest step size `t`.  This step is a
   provably descent direction for the original problem.

4. **Projected sub‑gradient descent** – improve the vector by moving
   against the sub‑gradient of the active (maximum) convolution entry
   while projecting back onto the simplex.

5. **Final local search** – a short simulated‑annealing on permutations
   (swap moves) followed by a cheap hill‑climbing that also performs
   small mass transfers.

The best vector found is returned as a plain Python list; the official
`evaluate_sequence` is called only once at the very end.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate wrapper (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of ``a`` with itself using FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip numerical noise
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` (sum is forced to 1)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smoothed surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (objective, gradient) for the smooth surrogate

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε),

    where ``conv = a*a`` and ``a`` is the soft‑max of ``w``.
    """
    # soft‑max (guarantees a_i ≥ 0 and Σa = 1)
    w_shift = w - np.max(w)          # improve numerical stability
    a = np.exp(w_shift)
    a /= np.sum(a)

    # convolution a*a
    conv = _fft_convolve_self(a)     # length 2·n‑1
    max_conv = conv.max()

    # smooth approximation of the maximum
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)     # log∑exp((c−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (see derivation in the original code)
    weights = np.exp(shifted)
    p = weights / np.sum(weights)          # probability distribution over k
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]           # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# bind helper convolution for speed
_smooth_obj_and_grad._conv_fft = lambda x, y: np.fft.irfft(
    np.fft.rfft(x, _next_pow2(x.shape[0] + y.shape[0] - 1))
    * np.fft.rfft(y, _next_pow2(x.shape[0] + y.shape[0] - 1)),
    _next_pow2(x.shape[0] + y.shape[0] - 1)
)[:x.shape[0] + y.shape[0] - 1]


# ----------------------------------------------------------------------
# Simple deterministic reorderings
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place large, small, large, … (largest at the very start)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left_ptr = 0
    right_ptr = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right_ptr]
        right_ptr -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left_ptr]
        left_ptr += 1
        i += 1
    return new_a


def _alternating_extremes_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest value at the leftmost position, the second largest at the
    rightmost, the third largest next to the leftmost, etc.
    This tends to separate large entries.
    """
    n = a.shape[0]
    desc = np.sort(a)[::-1]          # descending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    for idx, val in enumerate(desc):
        if idx % 2 == 0:
            new_a[left] = val
            left += 1
        else:
            new_a[right] = val
            right -= 1
    return new_a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Sub‑gradient method that directly minimises  max(a*a) while keeping
    a on the simplex (non‑negative, ℓ₁ = 1).
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    step0 = 0.12                      # initial step size
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur_a)
        k_max = int(np.argmax(conv))

        # gradient of the selected max‑entry w.r.t. a
        idx_arr = k_max - np.arange(n)
        mask = (idx_arr >= 0) & (idx_arr < n)
        grad = np.zeros_like(cur_a)
        grad[mask] = 2.0 * cur_a[idx_arr[mask]]

        alpha = step0 / np.sqrt(it + 1.0)

        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)
        s = np.sum(new_a)
        if s < 1e-12:                     # avoid stagnation
            new_a = np.full(n, 1.0 / n)
            s = 1.0
        new_a /= s                         # project onto the simplex

        new_val = _eval_fast(new_a)

        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it} new best = {best_val:.6f}")

        if new_val < cur_val:             # accept only improvements
            cur_a = new_a
            cur_val = new_val

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap moves only)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """Fast SA that swaps two positions; works directly on the fast evaluator."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99994
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        a[i], a[j] = a[j], a[i]            # propose swap
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            a[i], a[j] = a[j], a[i]        # revert

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing (swap + small mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:          # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                           # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP
        maximise 1ᵀg
        subject to (a ⊛ g)_k ≤ M  (k = 0…2n‑2)
                    g ≥ 0
    Returns the optimal g (not yet normalised).
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build the Toeplitz‑like convolution matrix A (sparse)
    # A_{k,i} = a_{k-i}  if 0 ≤ k-i < n, else 0.
    diag_data = [np.full(n, a[off]) for off in range(n)]
    A = diags(diag_data, offsets=np.arange(n), shape=(rows, n), format='csr')

    # Right‑hand side and objective
    b = np.full(rows, M)
    c = -np.ones(n)                     # linprog minimises → maximise Σg

    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError("LP step failed")
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the challenge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step‑function minimising ``evaluate_sequence``."""
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 1.5
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 1200                     # feel free to adjust; 1200 keeps LP cheap
    global _GLOBAL_N
    _GLOBAL_N = n

    # --------------------------------------------------------------
    # Initialise – an “ends heavy” shape (larger values near the boundaries)
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=float)
    a0 = 1.0 / (idx + 1) + 1.0 / (n - idx)        # two gentle tails
    a0 = np.maximum(a0, 0.0)
    a0 /= np.sum(a0)                               # ℓ₁ = 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1) Smooth surrogate L‑BFGS (multiple restarts, decreasing ε)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=18)

    restart = 0
    while time.time() < deadline - 300.0:          # reserve ~5 min for later phases
        restart += 1
        print(f"[restart {restart}]")
        # initialise soft‑max parameters – log of current a plus a bit of noise
        w = np.log(best_a + 1e-12) + 0.5 * np.random.randn(n)

        for eps in eps_schedule:
            if time.time() >= deadline - 280.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate the current soft‑max vector
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= np.sum(a)
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ------------------------------------------------------------
        # 2) Deterministic reorderings after each restart
        # ------------------------------------------------------------
        for reorder in (_ends_large_reorder, _alternating_extremes_reorder):
            ar = reorder(best_a)
            ar /= np.sum(ar)
            val_r = _eval_fast(ar)
            if val_r < best_val:
                best_val = val_r
                best_a = ar.copy()
                print(f"[reorder] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Linear‑programming descent (convex direction)
    # --------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print("[LP] starting convex LP descent")
        lp_deadline = deadline - 60.0
        while time.time() < lp_deadline:
            conv = _fft_convolve_self(best_a)
            M = np.max(conv)

            try:
                g = _lp_improve(best_a, M)
            except Exception as e:
                print("[LP] failed:", e)
                break

            g_sum = np.sum(g)
            if g_sum == 0:
                break
            g_norm = g / g_sum                         # ℓ₁ = 1

            # mix with a moderate step size; if no improvement, shrink t
            improved = False
            for t in (0.3, 0.15, 0.05):
                a_new = (1 - t) * best_a + t * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new /= np.sum(a_new)
                val_new = _eval_fast(a_new)
                if val_new < best_val:
                    best_val = val_new
                    best_a = a_new.copy()
                    print(f"[LP] new best (t={t}) = {best_val:.6f}")
                    improved = True
                    break
            if not improved:
                # no progress – exit the LP loop
                break

            # occasional extra re‑ordering to shake the multiset
            if np.random.rand() < 0.3:
                ar = _alternating_extremes_reorder(best_a)
                ar /= np.sum(ar)
                val_r = _eval_fast(ar)
                if val_r < best_val:
                    best_val = val_r
                    best_a = ar.copy()
                    print(f"[LP reorder] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Projected sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print("[subgrad] starting projected sub‑gradient descent")
        sub_deadline = deadline - 30.0
        best_a, best_val = _subgradient_descent(best_a, sub_deadline)

    # --------------------------------------------------------------
    # 5) Short simulated‑annealing on permutations (swap moves)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[SA] final simulated annealing (swap only)")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 15.0)

    # --------------------------------------------------------------
    # 6) Final cheap hill‑climbing refinement (swap + mass transfer)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ----------------------------------------------------------------------
    # Report final value using the official evaluator
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required by the framework
    return best_a.tolist()
