def _alternate_large_ends(a):
    sorted_vals = np.sort(a)[::-1]  # descending
    n = len(a)
    new_a = np.empty_like(a)
    left = 0
    right = n-1
    i = 0
    while left <= right:
        new_a[left] = sorted_vals[i]; i+=1
        if left == right:
            break
        new_a[right] = sorted_vals[i]; i+=1
        left += 1; right -= 1
    return new_a
```

def _subgradient_descent_with_momentum(a, deadline):
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_val = _eval_fast(cur)
    v = np.zeros_like(a)
    beta = 0.9
    step0 = 0.1
    it=0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur)
        k_max = int(np.argmax(conv))
        idx_arr = k_max - np.arange(n)
        mask = (idx_arr >=0) & (idx_arr < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[idx_arr[mask]]
        alpha = step0 / np.sqrt(it+1.0)
        v = beta * v - alpha * grad
        new = cur + v
        new = np.maximum(new, 0.0)
        s = np.sum(new)
        if s < 1e-12:
            new = np.full(n, 1.0/n)
            s = 1.0
        new /= s
        new_val = _eval_fast(new)
        if new_val < best_val:
            best_val = new_val
            best = new.copy()
            print(f"[subgrad momentum] iter {it} best {best_val:.6f}")
        # accept if improved
        if new_val <= _eval_fast(cur):
            cur = new
        it += 1
    return best, best_val

"""
Hybrid optimizer for the step‑function max‑convolution problem.

The algorithm improves on the previous version in three ways:

1. **More aggressive L‑BFGS smoothing** – a finer ε‑schedule (down to 1e‑15)
   and larger iteration limits give a tighter approximation of the true max.

2. **Two re‑ordering heuristics** – after each L‑BFGS run we try both the
   “ends‑large” interleaving and a new “alternating‑large‑ends” layout.
   The better ordering is kept.

3. **Longer refinement** – a momentum‑based sub‑gradient descent,
   followed by a longer simulated‑annealing phase that mixes swaps and
   small mass‑transfer moves, and a final hill‑climbing pass.

All phases respect the global 1000 s wall‑time budget.  The best
sequence found is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate wrapper (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip rounding artefacts
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` (identical numerical result
    but without safety checks).  Assumes ``a`` already satisfies the
    non‑negative and sum‑normalisation constraints.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Convolution used inside the gradient of the smooth surrogate
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where L is the smoothed surrogate

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and grad_w is the gradient w.r.t. the soft‑max parameters w.
    """
    # --- soft‑max : positivity & ℓ₁ = 1 ---
    w_shift = w - np.max(w)                # for numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                          # Σa = 1

    # --- convolution a * a ---
    conv = _fft_convolve_self(a)           # length 2·n‑1
    max_conv = conv.max()

    # --- smooth approximation of the max ---
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log∑exp((c−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # --- gradient w.r.t. a (see derivation in the description) ---
    weights = np.exp(shifted)             # exp((c−max)/ε)
    p = weights / np.sum(weights)         # probability distribution over k

    # g_i = 4·n·∑_k p_k·a_{k−i}
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    seg = conv_ap[n - 1: 2 * n - 1]        # central n entries
    grad_a = 4.0 * n * seg[::-1]          # ∂L/∂a

    # --- chain rule through the soft‑max ---
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# bind the helper convolution to the function object (so the wrapper can see it)
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values at the two ends, then interleave small ones."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left_ptr = 0
    right_ptr = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right_ptr]
        right_ptr -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left_ptr]
        left_ptr += 1
        i += 1
    return new_a


def _alternate_large_ends(a: np.ndarray) -> np.ndarray:
    """Largest -> leftmost, second largest -> rightmost, alternating."""
    n = a.shape[0]
    sorted_vals = np.sort(a)[::-1]   # descending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    idx = 0
    while left <= right:
        new_a[left] = sorted_vals[idx]
        idx += 1
        if left == right:
            break
        new_a[right] = sorted_vals[idx]
        idx += 1
        left += 1
        right -= 1
    return new_a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with momentum (exact max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_descent_with_momentum(a: np.ndarray, deadline: float):
    """
    Momentum based sub‑gradient descent on the true max‑convolution.
    Keeps the vector on the simplex (non‑negative, ℓ₁ = 1).
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_val = _eval_fast(cur)

    v = np.zeros_like(a)          # momentum term
    beta = 0.90                    # momentum coefficient
    step0 = 0.15                   # initial step size
    it = 0

    while time.time() < deadline:
        conv = _fft_convolve_self(cur)
        k_max = int(np.argmax(conv))

        # gradient of the max entry w.r.t. a : 2*a_{k_max-i} (if within range)
        idx_arr = k_max - np.arange(n)
        mask = (idx_arr >= 0) & (idx_arr < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[idx_arr[mask]]

        # diminishing step size
        alpha = step0 / np.sqrt(it + 1.0)

        # Nesterov‑like momentum update
        v = beta * v - alpha * grad
        new = cur + v

        # projection onto the simplex
        new = np.maximum(new, 0.0)
        s = np.sum(new)
        if s < 1e-12:                     # fallback to uniform if we collapse
            new = np.full(n, 1.0 / n)
            s = 1.0
        new /= s

        new_val = _eval_fast(new)

        if new_val < best_val:
            best_val = new_val
            best = new.copy()
            print(f"[subgrad‑mom] iter {it} new best = {best_val:.6f}")

        # accept only improving steps – guarantees monotonic descent
        if new_val < _eval_fast(cur):
            cur = new

        it += 1

    return best, best_val

    best_val = cur_val

    # initial temperature as a fraction of the current value
    T = max(cur_val * 0.02, 1e-6)

            new_a = a.copy()

        else:                                            # tiny mass transfer

            if i == j or a[i] == 0.0:

            delta = a[i] * rng.uniform(0.001, 0.02)      # 0.1‑2 % of the source mass

                best_seq = a.copy()

        # temperature schedule
        T *= decay
        if T < 1e-12:
