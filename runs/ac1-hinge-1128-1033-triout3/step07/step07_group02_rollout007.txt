    def project_simplex(v):
        """Project onto simplex {x >= 0, sum(x)=1}."""
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        # find rho = max i such that cond[i] is true
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

        # we can vectorize: indices = k_max - np.arange(n)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = a[idx[mask]]
        # gradient of objective = 4 * n * g
        grad = 4.0 * n * g

"""
Propose a step‑function (non‑negative list) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Initialisation**
   * If a previous solution exists (``height_sequence_1``) it is
     interpolated to the working length ``n``.
   * Sweep a small collection of parametric families (Beta‑shaped,
     sine‑powered, random Dirichlet draws) and keep the best seed.

2. **Projected sub‑gradient descent**
   * Optimises directly the hard ``max(conv(a,a))`` using the exact
     sub‑gradient (the shift of the current maximal‑overlap entry).
   * After each sub‑gradient step the vector is projected onto the
     probability simplex (non‑negative, sum = 1).
   * Several short restarts (with random perturbations of the current
     best) are performed while time permits.

3. **Fine‑grained hill‑climbing (simulated annealing)**
   * The same routine as in the previous state‑of‑the‑art is applied
     to the best vector obtained from step 2.

4. **Return the best vector found**.

All heavy work (FFT‑based convolutions, simplex projection) uses NumPy
and respects the 1000 s budget (a 0.5 s safety margin is kept for the
final ``evaluate_sequence`` call).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    # reference: Duchi et al., ICML 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):          # all entries are non‑positive
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families – generate a few deterministic candidates
# ----------------------------------------------------------------------
def _beta_candidate(p: float, q: float, n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def _sine_candidate(exp: float, n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1)) ** exp
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def _random_dirichlet(alpha: float, n: int) -> np.ndarray:
    """Draw a random point on the simplex (Gamma/Dirichlet)."""
    draws = np.random.gamma(alpha, 1.0, size=n)
    return draws / draws.sum()


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‐convolution objective
# ----------------------------------------------------------------------
def _subgradient_optimize(a_start: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that directly minimises 2·n·max(conv(a,a))
    while enforcing a ≥ 0 and Σa = 1 via projection onto the simplex.
    Returns the best vector and value encountered before *deadline*.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    idle = 0
    # step schedule (linear decay)
    STEP_START = 0.08
    STEP_END = 0.0006

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current convolution and value
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()

        # record improvement
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            idle = 0
            print(f"[SGD {it:7d}] new best = {best_val:.6f}")
        else:
            idle += 1

        # sub‑gradient w.r.t. the max‑entry (k_max)
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)                # vector of k_max - i
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = a[idx[mask]]                     # g_i = a_{k_max-i}

        # linear step‑size schedule
        prog = (now - start_time) / (deadline - start_time)
        step = STEP_START * (1.0 - prog) + STEP_END * prog

        # descent step and projection onto simplex
        a = a - step * g
        a = _project_simplex(a)

        # restart when stuck for a long time
        if idle > 2500:
            a = best_a + np.random.normal(scale=0.015, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            idle = 0
            print(f"[SGD {it:7d}] restart after idle")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (same as the previous state‑of‑the‑art)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from highly
    contributing indices to barely contributing ones, with Metropolis
    acceptance.
    """
    a = a0.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule for maximal transfer fraction and temperature
    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.012, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (probability ∝ contribution)
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index (biased towards low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val

        run_deadline = min(time.time() + 25.0, sg_deadline)

        # start point: perturb the current global best a little
        a_start = best_a + np.random.normal(scale=0.012, size=n)
        a_start = np.maximum(a_start, 0.0)
        a_start = _project_simplex(a_start)

        local_a, local_val = _subgradient_optimize(
            a_start, n, run_deadline, start_time)

        if local_val < best_val:
            best_val, best_a = local_val, local_a.copy()
            print(f"[overall] improved by sub‑grad to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(
        best_a, n,
        deadline=deadline,
        start_time=time.time(),
        best_a=best_a,
        best_val=best_val,
        target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
