def _spread_reorder(a):
    n = len(a)
    sorted_vals = np.sort(a)[::-1]
    new = np.empty_like(a)
    left = 0
    right = n-1
    toggle = True
    for val in sorted_vals:
        if toggle:
            new[left] = val
            left += 1
        else:
            new[right] = val
            right -= 1
        toggle = not toggle
    return new

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise several diverse candidates (uniform, previous best,
   random Dirichlet, sinusoid, spread‑reordered uniform) and keep the
   best one (according to the fast surrogate ``2·n·max(conv)``).
2. Run a *directed mass‑transfer* local optimiser:
   repeatedly move a fraction of probability mass from the index that
   contributes most to the current maximal convolution entry to an index
   that contributes the least.  The move size is adapted and occasional
   random perturbations are injected to escape plateaux.
   This step is performed on the current best vector and on a few random
   restarts while respecting the time budget.
3. Apply a cheap hill‑climbing refinement (random swaps / small
   transfers) for polishing.
4. Return the best sequence found (as a plain Python list) after a final
   call to the official ``evaluate_sequence`` for safety.

All operations keep the vector non‑negative and normalised
(``Σ a_i = 1``), which makes the objective equivalent to minimising the
peak of the self‑convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using FFT of a power‑of‑2 size.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` for vectors with
    non‑negative entries that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1 simplifies the
    original formula).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _spread_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that spreads large values to opposite ends of the vector.
    This often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)[::-1]          # descending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for val in sorted_vals:
        if toggle:
            new_a[left] = val
            left += 1
        else:
            new_a[right] = val
            right -= 1
        toggle = not toggle
    return new_a


def _refine_hill_climb(
    a: np.ndarray,
    cur_val: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).
    The vector is kept on the simplex after each move.
    """
    cur_a = a.copy()
    cur_best = cur_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_best - 1e-12:
            cur_a = new_a
            cur_best = new_val
            print(f"[refine] new best = {cur_best:.6f}")

    return cur_a, cur_best


def _local_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Directed mass‑transfer optimiser.
    Repeatedly moves a fraction of probability mass from the index that
    contributes most to the current maximal convolution entry to an index
    that contributes the least.  Step size adapts; occasional random
    perturbations help escape plateaux.
    """
    a = a_start.copy()
    best_val = _fast_evaluate(a)
    best_a = a.copy()
    move_frac = 0.2               # initial fraction of source mass to move
    no_improve = 0
    rng = np.random.default_rng()
    # relative tolerance for “almost maximal” convolution entries
    tol_rel = 5e-4

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_val = conv.max()
        # indices whose convolution value is within tol_rel of the max
        top_idxs = np.where(conv >= max_val * (1.0 - tol_rel))[0]

        # gradient‑like vector: contribution of each a_i to the selected
        # convolution entries
        g = np.zeros(n)
        for k in top_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            g[idx_range] += a[k - idx_range]

        src = int(np.argmax(g))
        # among the minima of g pick the one with smallest a‑mass
        dst_candidates = np.where(g == g.min())[0]
        if dst_candidates.size == 0:
            dst = int(np.argmin(a))
        else:
            dst = int(dst_candidates[np.argmin(a[dst_candidates])])

        if a[src] <= 0.0:
            break

        delta = min(move_frac * a[src], move_frac * (1.0 - a[dst]))
        if delta < 1e-14:
            break

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        # re‑normalise (protect against tiny drift)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s == 0.0:
            break
        new_a /= s

        new_val = _fast_evaluate(new_a)

        if new_val < best_val - 1e-12:
            # accept improvement
            a = new_a
            best_val = new_val
            best_a = new_a.copy()
            no_improve = 0
            move_frac = min(move_frac * 1.05, 0.5)   # become a bit bolder
            print(f"[local] improvement → {best_val:.6f}")
        else:
            # reject – shrink step size
            no_improve += 1
            move_frac = max(move_frac * 0.8, 0.01)
            if no_improve >= 20:
                # random jitter to shake out of a plateau
                perturb = rng.normal(scale=0.01, size=n)
                a = a + perturb
                a = np.maximum(a, 0.0)
                a /= a.sum()
                no_improve = 0

    return best_a, best_val
