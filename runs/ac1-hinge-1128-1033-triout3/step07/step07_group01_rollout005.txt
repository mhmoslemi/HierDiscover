"""
Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm overview
------------------
1. **Initial seeding** – generate a diverse set of candidates:
   uniform, a copy of the previous best (if available), a few random Dirichlet
   draws, simple sinusoid, and a grid of parametric families (exponential,
   symmetric exponential and power‑law).  Keep the best by the cheap fast
   evaluator.

2. **Smooth surrogate optimisation (L‑BFGS)** – minimise the
   log‑sum‑exp smoothed version of the max‑convolution.
   The schedule starts at ε≈5·10⁻³ and ends at ε≈10⁻¹³.
   Up to 15 restarts are performed, each consisting of several ε‑stages.
   After every restart we try the *anti‑symmetric* reorder heuristic.

3. **Exact mirror‑descent** – use the true sub‑gradient of the
   max‑convolution (the “worst” index) and perform an exponential
   update on the simplex.  Tens of thousands of cheap iterations are
   executed (the iteration is O(n) and therefore cheap).

4. **Simulated annealing with block moves** – random mass‑transfer moves
   (both single‑index and contiguous‑block swaps) are accepted
   according to a geometric cooling schedule.  This helps escaping
   shallow local minima.

5. **Final hill‑climbing** – a very cheap local search that only accepts
   strict improvements (random swaps / mass transfers).

All steps respect the 1000 s budget; the best vector found is finally
checked with the official ``evaluate_sequence`` function and returned as
a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (set before each
# optimisation call).
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    The result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate   L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    with the soft‑max parametrisation a_i = exp(w_i)/Σ_j exp(w_j).

    Returns the scalar value and the gradient w.r.t. w.
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max -----
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- convolution -----
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max -----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    weights = np.exp(shifted)
    p = weights / weights.sum()      # soft‑max over lag indices

    # convolution of a with reversed p (needed for gradient)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # select the segment that aligns with the original a‑vector
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]     # reverse to match ordering of a_i

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return float(L), grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses global ε and n)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the O(n) algorithm from
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already
    satisfy a_i ≥ 0 and Σ a_i = 1.
    Returns 2·n·max(conv) (the denominator is 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places the largest values to the left side
    and the smallest to the right side, often flattening the peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector is kept
    on the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                      # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

    # if it is available, otherwise fall back to 4000).
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # --------------------------------------------------------------
    # 1) Build a diverse set of cheap initial candidates.
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # two random Dirichlet‑like draws
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoid (positive shift to avoid zeros)
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # ---- parametric families -------------------------------------------------
    # exponential decay (left‑leaning)
    for lam in np.logspace(-4, 0, 9):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        init_candidates.append(a)

        # symmetric exponential about the centre
        a_sym = np.exp(-lam * np.abs(np.arange(n) - (n - 1) / 2))
        a_sym /= a_sym.sum()
        init_candidates.append(a_sym)

    # power‑law (negative exponents give heavier tails)
    for p in np.linspace(-2.0, 0.0, 5):
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        init_candidates.append(a)

    # normalise everything onto the simplex (some generators may have zeros)
    init_candidates = [c / c.sum() for c in init_candidates]

    # --------------------------------------------------------------
    # 2) Pick the best cheap candidate.
    # --------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – L‑BFGS on a smooth surrogate of the max‑convolution.
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-13), num=15)
    # start from the soft‑max parameters of the best candidate
    w = np.log(best_a + 1e-12)

    max_restarts = 15
    restart = 0
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        # occasional full random re‑initialisation of w
        if restart % 5 == 0:
            w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n

            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # convert back to a probability vector
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # try anti‑symmetric reorder after each restart
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 2 – exact mirror‑descent on the true max‑convolution.
    # --------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.35          # initial step size (larger than baseline)
    decay = 0.9999
    min_alpha = 0.002
    it = 0
    while time.time() < deadline - 20.0:
        # compute full convolution once per iteration
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]

        # exponential (mirror) update
        alpha = max(alpha0 * (decay ** (it // 500)), min_alpha)
        a = a * np.exp(-alpha * g)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        if it % 200 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[mirror {it}] new best = {best_val:.6f}")

            # occasional anti‑symmetric reorder to shake the solution
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[mirror reorder] improved to {best_val:.6f}")

        it += 1

    # --------------------------------------------------------------
    # 5) Phase 3 – simulated‑annealing style random mass transfers.
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.025                       # initial temperature
    cooling = 0.99995
    rng = np.random.default_rng(seed + 123456)

    while time.time() < deadline - 5.0:
        # Decide between a single‑index move and a block swap
        if rng.random() < 0.65:
            # single‑index mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.12 * rng.random()   # up to 12 % of source mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # swap two short contiguous blocks (length up to 40)
            L = rng.integers(1, min(40, n // 8))
            i = rng.integers(0, n - L)
            j = rng.integers(0, n - L)
            new_a = cur_a.copy()
            # Use temporary copy to avoid overlapping swaps
            tmp = new_a[i:i + L].copy()
            new_a[i:i + L] = new_a[j:j + L]
            new_a[j:j + L] = tmp

        # re‑normalise
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # --------------------------------------------------------------
    # 6) Final cheap hill‑climbing refinement.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # 7) Final safety check with the official evaluator and return.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
