"""
Multi‑phase optimisation for the “evaluate_sequence” objective.

The algorithm proceeds inside the allotted 1000 s budget:

1. **Initialisation** – use a previously‑found construction if present,
   otherwise start from a uniform distribution.

2. **Random restarts** – try a few random Dirichlet vectors and keep the
   best one.

3. **Smooth surrogate minimisation** – minimise a smoothed version of the
   objective (log‑sum‑exp of the self‑convolution) with L‑BFGS‑B.  The
   smoothing parameter ε is decreased on a geometric schedule; a few
   restarts are performed.

4. **Multi‑peak exponential mirror descent** – a multiplicative update
   that reduces the *k* largest convolution peaks (k≈7).  The step‑size
   decays slowly; periodic anti‑symmetric re‑ordering is applied.

5. **Hill‑climbing** – random swaps and small mass‑transfer moves keep the
   solution on the probability simplex and accept only strict improvements.

6. **Simulated annealing** – a final temperature‑controlled search over
   permutations and mass‑transfer moves helps escape any remaining local
   minima.

All intermediate vectors are projected back onto the probability simplex
( Σ aᵢ = 1, aᵢ ≥ 0 ).  The code uses FFT‑based convolutions for speed and
stops automatically when the time budget is exhausted.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global handles for the smooth surrogate optimiser
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself (length 2·len‑1).
    Implemented via FFT for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negative numbers caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest values with the smallest ones when the sequence is
    reversed (a simple heuristic that often flattens the self‑convolution).
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution a*a
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (derived analytically)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability over lag indices

    # convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.minimize (uses the global ε and n)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule: from 5·10⁻³ down to 10⁻⁹ (geometric)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=9)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to a (probability simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric re‑ordering heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val

def _gradient_descent_multi(a_start: np.ndarray,
                            deadline: float,
                            top_k: int = 7,
                            alpha0: float = 0.6,
                            decay_rate: float = 0.98) -> tuple[np.ndarray, float]:
    """
    Exponential mirror‑descent that reduces the *top_k* largest
    convolution peaks.
    Returns the best (a, value) pair found before ``deadline``.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[gd init] value = {best_val:.6f}")

    it = 0
    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)

        # indices of the ``top_k`` largest convolution entries
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]   # sorted descending
        else:
            idxs = np.arange(conv.size)

        # gradient contribution from the selected peaks
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += a[j_range]

        # scheduled step‑size
        alpha = alpha0 * (decay_rate ** (it // 2000))

        # multiplicative update (mirror‑descent)
        a = a * np.exp(-alpha * g)

        # projection onto the probability simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:                     # guard against under‑flow
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        # occasional logging / re‑ordering
        if it % 50 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd iter {it}] new best = {best_val:.6f}")
            if it % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val

def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Random local search: swaps (20 %) and mass transfers (80 %).
    Keeps only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
    return cur_a, cur_val

# Simulated annealing over permutations & mass moves
# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         n: int) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk (swap + mass‑transfer moves).
    Accepts worsening moves with probability exp(–Δ/T) and cools slowly.
    """
    cur_a = a_start.copy()
    best_a = cur_a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    # Initial temperature – a few percent of the typical objective value
    T = max(1e-4, 0.01 * cur_val)
    it = 0

    while time.time() < deadline - 1.0:
        # ----- propose a move -------------------------------------------------
        if rng.random() < 0.3:                     # swap two positions
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # small mass transfer
            i = rng.integers(0, n)
            if cur_a[i] <= 1e-15:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cur_a[i] * 0.10 * rng.random()   # up to 10 % of a_i
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # ----- projection onto the simplex -----------------------------------
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        # ----- evaluate -------------------------------------------------------
        try:
            new_val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        delta_val = new_val - cur_val
        if delta_val < -1e-12:
            # improvement – always accept
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val:.6f}")
        else:
            # occasional uphill move
            prob = np.exp(-delta_val / max(T, 1e-12))
            if rng.random() < prob:
                cur_a = new_a
                cur_val = new_val

        # ----- cooling --------------------------------------------------------
        it += 1
        if it % 1000 == 0:
            T *= 0.95
            if T < 1e-6:
                T = 1e-6

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Keep a safety margin so we can return before the hard deadline.
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – more degrees of freedom give a better chance to
    # flatten the convolution; 8000 is a good compromise.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from a known construction if possible
    # ------------------------------------------------------------------
    old = globals().get('height_sequence_1')
    if old is not None:
        try:
            old = np.asarray(old, dtype=float)
            if old.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, old.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                a = np.interp(new_x, old_x, old)
            else:
                a = old.copy()
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)          # uniform start

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
        s = a.sum()
    a /= s                                      # enforce Σaᵢ = 1

    # ------------------------------------------------------------------
    # Evaluate the initial candidate
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – a few random initial vectors (helps avoid a poor start)
    # ------------------------------------------------------------------
    for tr in range(6):
        if time.time() >= deadline - 200:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate optimisation (L‑BFGS) with several restarts
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 150 and restart < 8:
        restart += 1
        # start from current best plus a modest random perturbation in w‑space
        w_start = np.log(best_a + 1e-12)
        w_start += rng.normal(scale=0.2, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w_start, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[restart {restart}] new best after L‑BFGS = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multi‑peak exponential gradient descent (more peaks)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        a_gd, val_gd = _gradient_descent_multi(best_a,
                                               deadline,
                                               top_k=7,
                                               alpha0=0.6,
                                               decay_rate=0.98)
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[gradient descent] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing (swap + mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2:
        best_a, best_val = _simulated_annealing(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
