def _simulated_annealing_ext(a, cur_val, deadline):
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.05, 1e-6)
    decay = 0.9995
    it = 0
    while time.time() < deadline:
        move_type = rng.random()
        if move_type < 0.3:  # swap
            i, j = rng.integers(0, n, size=2)
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        elif move_type < 0.6:  # small transfer
            i, j = rng.integers(0, n, size=2)
            if a[i] == 0:
                continue
            delta = a[i] * rng.random() * 0.05
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        elif move_type < 0.8:  # segment shuffle
            start = rng.integers(0, n)
            length = rng.integers(2, min(30, n-start))
            indices = np.arange(start, start+length)
            new_a = a.copy()
            new_a[indices] = rng.permutation(new_a[indices])
        else:  # targeted transfer
            conv = _fft_convolve_self(a)
            k_max = int(np.argmax(conv))
            # find pair with max a_i a_j for i+j=k_max
            i_start = max(0, k_max - (n-1))
            i_end = min(k_max, n-1)
            best_pair = None
            best_prod = -1.0
            for i in range(i_start, i_end+1):
                j = k_max - i
                prod = a[i] * a[j]
                if prod > best_prod:
                    best_prod = prod
                    best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                # choose source = i if a[i] >= a[j] else j
                source = i if a[i] >= a[j] else j
                target = rng.integers(0, n)
                if source == target:
                    continue
                delta = a[source] * 0.05 * rng.random()
                new_a = a.copy()
                new_a[source] -= delta
                new_a[target] += delta
            else:
                continue

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < math.exp(-delta_val / T):

                print(f"[SA] new best {best_val:.6f} iter {it}")

        T = max(T * decay, 1e-12)
        it += 1
    return best_a, best_val

"""
Hybrid optimiser for the step‑function convolution problem (new version).

Main ideas
----------
* Produce many diverse seeds (geometric, power‑law, Dirichlet with several
  concentration parameters and a “golden‑ratio scatter’’ ordering).
* Optimise each seed with a long L‑BFGS run on a smooth log‑sum‑exp surrogate.
  The smoothing parameter ε is decreased from 1e‑2 to 1e‑12.
* After L‑BFGS apply a handful of deterministic re‑orderings (including a
  golden‑ratio permutation).
* Run an extended simulated‑annealing phase that mixes swaps, small mass
  transfers, segment shuffles and a targeted mass‑transfer move aimed at the
  current maximal convolution entry.
* Finish with a cheap hill‑climbing pass.
* All evaluations use an FFT‑based fast version of evaluate_sequence.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables for the smoothed surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter
_GLOBAL_N = None     # current sequence length


def _next_pow2(x: int) -> int:
    """Smallest power of two greater or equal to x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast replica of evaluate_sequence for non‑negative vectors."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """a_i ∝ q**i, normalised to sum = 1 (0<q<1)."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q**n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)**p, normalised to sum = 1 (p<0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where L is the smoothed objective

        L = 2 n max_conv + 2 n eps * log∑exp((conv - max_conv)/eps)

    and grad_w is the gradient w.r.t. the logits w (soft‑max parameters).
    """
    # soft‑max (positivity + ℓ₁ = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= np.sum(a)

    # auto‑convolution a*a
    conv = _fft_convolve_self(a)
    max_conv = conv.max()

    # log‑sum‑exp approximation of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient wrt a (closed form using the weights of the log‑sum‑exp)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # probability over convolution indices
    # reverse to get the correct convolution kernel for the gradient
    p_rev = p[::-1]
    conv_ap = np.fft.irfft(
        np.fft.rfft(a) * np.fft.rfft(p_rev, _next_pow2(len(p_rev) + n - 1)),
        _next_pow2(len(p_rev) + n - 1)
    )[:len(p_rev) + n - 1]
    seg = conv_ap[n - 1: 2 * n - 1]                 # central n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.optimize.minimize."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Re‑ordering helpers
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large values left, small values right (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)               # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[n - 1 - i]          # big → left
        out[n - 1 - i] = s[i]          # small → right
    if n % 2:
        out[half] = s[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    s = np.sort(a)
    left, right = 0, len(s) - 1
    out = np.empty_like(a)
    k = 0
    while left <= right:
        out[k] = s[right]
        k += 1
        right -= 1
        if left <= right:
            out[k] = s[left]
            k += 1
            left += 1
    return out


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Simply sort descending."""
    return np.sort(a)[::-1]


def _golden_ratio_reorder(a: np.ndarray) -> np.ndarray:
    """
    Scatter the sorted values using a step ≈ φ·n (φ = (sqrt(5)-1)/2).
    This yields a low‑discrepancy permutation.
    """
    n = a.shape[0]
    step = int(round(((np.sqrt(5) - 1) / 2) * n))
    idx = (np.arange(n) * step) % n
    out = np.empty_like(a)
    out[idx] = np.sort(a)[::-1]       # put the biggest values first in the scattered order
    return out


# ----------------------------------------------------------------------
# Simulated annealing with a richer move set
# ----------------------------------------------------------------------
def _simulated_annealing_ext(a: np.ndarray, cur_val: float,
                             deadline: float) -> (np.ndarray, float):
    """Extended SA mixing swaps, transfers, segment shuffles and a targeted move."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.05, 1e-6)          # initial temperature
    decay = 0.9997
    it = 0

    while time.time() < deadline:
        move = rng.random()
        if move < 0.30:                     # 1) simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif move < 0.60:                   # 2) small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * 0.04 * rng.random()
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        elif move < 0.80:                   # 3) segment shuffle (length 2‑30)
            start = rng.integers(0, n)
            max_len = min(30, n - start)
            if max_len < 2:
                continue
            length = rng.integers(2, max_len + 1)
            idx = np.arange(start, start + length)
            new_a = a.copy()
            new_a[idx] = rng.permutation(new_a[idx])

        else:                               # 4) targeted transfer from the pair forming the current peak
            conv = _fft_convolve_self(a)
            k_max = int(conv.argmax())
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            best_pair = None
            best_prod = -1.0
            for i in range(i_start, i_end + 1):
                j = k_max - i
                prod = a[i] * a[j]
                if prod > best_prod:
                    best_prod = prod
                    best_pair = (i, j)
            if best_pair is None:
                continue
            i, j = best_pair
            source = i if a[i] >= a[j] else j
            target = rng.integers(0, n)
            if source == target:
                continue
            delta = a[source] * 0.04 * rng.random()
            new_a = a.copy()
            new_a[source] -= delta
            new_a[target] += delta

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0.0 or rng.random() < math.exp(-delta_val / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (it {it})")

        T = max(T * decay, 1e-12)
        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Pure local search – swaps and tiny transfers."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = cur_val

    while time.time() < deadline:
        if rng.random() < 0.25:                     # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                        # tiny transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] == 0.0:
                continue
            delta = best_a[i] * 0.02 * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a
            print(f"[hill] improvement to {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _optimize_lbfgs(a0: np.ndarray,
                    eps_schedule,
                    maxiter_per_eps: int,
                    deadline: float):
    """Run L‑BFGS for the whole ε schedule, keeping the best sequence."""
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    w = np.log(best_a + 1e-16)               # logits for the soft‑max

    for eps in eps_schedule:
        if time.time() >= deadline:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = best_a.shape[0]

        res = minimize(_smooth_obj_and_grad_wrapper, w,
                       method='L-BFGS-B', jac=True,
                       options={'maxiter': maxiter_per_eps,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x
        a = np.exp(w - np.max(w))
        a /= a.sum()
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L‑BFGS] eps {eps:.1e} new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """Search for a step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    n = 4000
    global _GLOBAL_N
    _GLOBAL_N = n

    # ------------------------------------------------------------------
    # initialise from previous run if possible
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if prev.shape[0] != n:
            # linear interpolation to the required length
            x_old = np.linspace(0, 1, prev.shape[0])
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
        else:
            a0 = prev.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")
    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. seed generation (high diversity)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    seeds = []

    # uniform baseline
    seeds.append(np.full(n, 1.0 / n))

    # geometric families
    for q in np.linspace(0.6, 0.99, 12):
        seeds.append(_geometric_sequence(q, n))

    # power‑law families
    for p in np.linspace(-4.0, -0.1, 12):
        seeds.append(_power_sequence(p, n))

    # Dirichlet‑like draws with several concentration parameters
    for alpha in (0.05, 0.1, 0.2, 0.4, 0.8):
        for _ in range(6):
            a = rng.gamma(alpha, 1.0, size=n)
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            seeds.append(a)

    # golden‑ratio scattered random vectors
    for _ in range(8):
        a = rng.random(n)
        a /= a.sum()
        seeds.append(_golden_ratio_reorder(a))

    # evaluate all seeds and keep the best one
    for a in seeds:
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[seed] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. long L‑BFGS optimisation on the best seed
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    stage_deadline = time.time() + min(300.0, time_left * 0.6)

    eps_schedule = np.logspace(np.log10(5e-2), np.log10(5e-12), num=20)
    best_a, best_val = _optimize_lbfgs(best_a,
                                       eps_schedule,
                                       maxiter_per_eps=300,
                                       deadline=stage_deadline)
    print(f"[lbfgs] after optimisation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. deterministic re‑orderings (may further lower the max‑conv)
    # ------------------------------------------------------------------
    for reorder in (_anti_symmetric_reorder,
                    _alternating_reorder,
                    _descending_reorder,
                    _golden_ratio_reorder):
        a_r = reorder(best_a)
        v_r = _eval_fast(a_r)
        if v_r < best_val:
            best_val = v_r
            best_a = a_r.copy()
            print(f"[reorder] {reorder.__name__} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. extended simulated annealing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 150.0:
        sa_deadline = time.time() + remaining - 80.0
    else:
        sa_deadline = deadline - 5.0
    print(f"[SA] extended annealing until {sa_deadline:.2f}")
    best_a, best_val = _simulated_annealing_ext(best_a, best_val, sa_deadline)

    # ------------------------------------------------------------------
    # 5. final cheap hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # 6. report exact evaluation using the supplied function
    # ------------------------------------------------------------------
    try:
        exact = evaluate_sequence(best_a.tolist())
        print(f"[final] exact evaluation = {exact:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
