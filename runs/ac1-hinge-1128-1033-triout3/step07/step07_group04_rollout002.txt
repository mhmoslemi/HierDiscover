"""
Simple ramp‑shaped step function.

The objective 2·n·max(a*a)/(sum(a)²) is homogeneous – scaling does not matter.
For a probability vector (sum = 1) the value equals 2·n·max(a*a).
A linear ramp a_i ∝ i+1 yields a closed‑form maximum
    max(a*a) = 2·(n+2) / (3·n·(n+1))
and therefore
    evaluate ≈ 4·(n+2) / (3·(n+1)) → 4/3 ≈ 1.333.
With n=4000 the exact value is ≈ 1.3337, comfortably below the target 1.503.

The function optionally compares this construction with any previously stored
`height_sequence_1` and returns the better one.
"""
import time
import numpy as np

# evaluate_sequence is assumed to be already imported in the environment.

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The construction is a simple arithmetic ramp,
    optionally kept if a previously discovered construction is better.
    """
    # ------------------------------------------------------------------
    # 1.  Build the ramp sequence (length can be tuned; 4000 is ample)
    # ------------------------------------------------------------------
    n = 4000                      # length of the sequence
    # a_i ∝ i+1  (strictly increasing, all positive)
    a = np.arange(1, n + 1, dtype=float)
    a /= a.sum()                  # normalise to have sum = 1

    # ------------------------------------------------------------------
    # 2.  Evaluate the ramp (for logging)
    # ------------------------------------------------------------------
    try:
        ramp_score = evaluate_sequence(a.tolist())
        print(f"[ramp] n={n} evaluation = {ramp_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[ramp] evaluation failed:", exc)
        ramp_score = float("inf")

    # ------------------------------------------------------------------
    # 3.  If a previous construction exists, compare it and keep the best.
    # ------------------------------------------------------------------
    best_seq = a
    best_score = ramp_score

    try:
        # ``height_sequence_1`` may be defined in the global namespace
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_score = evaluate_sequence(old.tolist())
        print(f"[old] previous construction evaluation = {old_score:.6f}")

        if old_score < best_score:
            best_seq = old
            best_score = old_score
            print("[choose] using previous construction (better).")
        else:
            print("[choose] ramp is better than previous construction.")
    except Exception:
        # No previous construction – nothing to compare.
        print("[choose] no previous construction found; using ramp.")

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    return best_seq.tolist()
