    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2 * n * b.max()
    max_frac_start = 0.5
    max_frac_end = 0.001
    no_improve = 0
    IMPROVE_LIMIT = 200_000
    it=0
    while time.time() < deadline:
        progress = (time.time() - start)/ (deadline - start)
        max_frac = max_frac_start * (1 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))
        # compute rev
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        rev = np.zeros(n)
        rev[mask] = a[idx[mask]]
        contrib = a * rev
        i = np.argmax(contrib)  # source
        # compute target candidate: minimize rev (partner) plus maybe a small weight on a
        # compute weight = rev + a   (or rev + 1e-6 * a)
        weight = rev + a
        weight[i] = np.inf
        j = np.argmin(weight)

        # compute a_rev_i and a_rev_j
        if 0 <= k_max - i < n:
            a_rev_i = a[k_max - i]
        else: a_rev_i = 0.0
        if 0 <= k_max - j < n:
            a_rev_j = a[k_max - j]
        else: a_rev_j = 0.0

        # compute second maximum of b (excluding k_max)
        b2 = b.copy()
        b2[k_max] = -np.inf
        second_max = b2.max()
        # bound delta:
        # change in b_max = delta*(a_rev_j - a_rev_i)
        if a_rev_i > a_rev_j:
            # compute max delta to not go below second_max
            max_allowed = (b[k_max] - second_max) / (a_rev_i - a_rev_j)
            # apply safety factor 0.95
            max_allowed = max_allowed * 0.95
        else:
            max_allowed = a[i]  # any, but won't reduce max
        # choose delta = min(a[i] * max_frac, max_allowed)
        delta = min(a[i] * max_frac, max_allowed) * np.random.rand()
        if delta <= 1e-16:
            # possibly pick new random i-j using random scheme fallback
            # but continue
            continue

        # incremental update for b:
        b_new = b.copy()
        # subtract contributions from a[i] decreasing:
        b_new[i:i+n] -= 2 * delta * a
        # add contributions from a[j] increasing:
        b_new[j:j+n] += 2 * delta * a
        # adjust self contributions
        b_new[2*i] += delta*delta
        b_new[2*j] += delta*delta
        b_new[i+j] -= 2 * delta * delta

            a[j] += delta

"""
Optimiser for the step‑function evaluation problem.

The algorithm proceeds in three stages within the allotted time budget:

1️⃣  **Initialisation** – a rich library of parametric families and a few random
    Dirichlet draws provides a strong starting vector on the simplex.

2️⃣  **Smooth‑max Adam** – we minimise a log‑sum‑exp approximation to
    `max(conv(a,a))`.  A log‑softmax parametrisation guarantees non‑negativity
    and unit sum.  The temperature (ε) is annealed from 1e‑1 to 1e‑12 and the
    learning rate from 0.12 down to 0.002.  Restarts keep the search from
    stalling.

3️⃣  **Deterministic hill‑climber** – On the exact objective we repeatedly
    relocate a small amount of mass from the index that contributes most to
    the current maximal convolution entry to the index that contributes the
    least (both in terms of the partner value `a_{k_max‑i}` and its own size).
    The transferred fraction is annealed (0.5 → 0.001) and the step size is
    capped so that the new maximal entry never exceeds the previous second‑
    highest entry.  After stagnation we re‑inject a small amount of noise.

4️⃣  **Uniform mixing** – a final convex combination with the uniform vector
    often reduces the peak a little further.

All stages respect the total budget (default 1000 s).  The returned vector
has non‑negative entries and unit sum, which makes the evaluation
scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) assumed = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Parameteric initialisation – a rich set of shapes
# ----------------------------------------------------------------------
def _initial_candidate(n: int) -> tuple[np.ndarray, float]:
    """Return a promising initial vector (unit‑sum) and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ----- deterministic families ------------------------------------------------
    # linear decreasing
    try_candidate(np.arange(n, 0, -1, dtype=float))

    # symmetric U‑shape
    idx = np.arange(n)
    lin_u = np.maximum(idx + 1, n - idx)
    try_candidate(lin_u)

    # quadratic decreasing (left side)
    try_candidate((np.arange(n, 0, -1, dtype=float) ** 2))

    # quadratic U‑shape
    quad_u = (np.arange(n) + 1) ** -2 + (np.arange(n, 0, -1) + 0) ** -2
    try_candidate(quad_u)

    # half‑wave sine (symmetric)
    try_candidate(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))

    # cosine with peaks at the ends
    try_candidate(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # power‑law symmetric (U‑shape)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_candidate(a)

    # exponential decay (left‑hand side)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997,
              0.996, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        try_candidate(a)

    # symmetric exponential (U‑shape)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997,
              0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_candidate(a)

    # truncated exponential (mass only on a prefix)
    for frac in (0.10, 0.15, 0.20, 0.25, 0.30):
        K = max(1, int(frac * n))
        for r in (0.99, 0.98, 0.95, 0.90):
            a = np.zeros(n, dtype=float)
            a[:K] = r ** np.arange(K, dtype=float)
            try_candidate(a)

    # ----- few random Dirichlet draws -----------------------------------------
    rng = np.random.default_rng(12345)
    for _ in range(12):
        a = rng.dirichlet(np.ones(n))
        try_candidate(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of max(conv(a,a)).
    Log‑softmax parametrisation guarantees non‑negativity and unit sum.
    """
    # logits (add a tiny constant to avoid log(0))
    w = np.log(a0 + 1e-12)

    # hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 60000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- obtain a on the simplex -----
        w_shift = w - np.max(w)                 # for numerical stability
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- schedules -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max via log‑sum‑exp -----
        b_full = _full_convolution(a)          # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()            # soft‑max over positions

        # ----- gradient of smooth objective w.r.t. a -----
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]    # n entries corresponding to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]

        # ----- chain rule through the log‑softmax (log‑softmax) -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter of logits
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.008, size=n)

        # restart if stagnated
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – deterministic hill‑climber on the exact objective
# ----------------------------------------------------------------------
def _deterministic_hillclimb(a_start: np.ndarray,
                             n: int,
                             start: float,
                             deadline: float,
                             best_a: np.ndarray,
                             best_val: float,
                             target_val: float = 1.503) -> np.ndarray:
    """
    Aggressive local search that moves mass from the index contributing most
    to the current maximal convolution entry to the index contributing the
    least.  The transfer amount is capped so that the new peak never exceeds
    the previous second‑largest entry.
    """
    a = a_start.copy()
    b = _full_convolution(a)                     # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # schedule for the transferred fraction (0.5 → 0.001)
    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # index where convolution attains its maximum
        k_max = int(np.argmax(b))

        # partner values a_{k_max‑i}
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        rev = np.zeros_like(a)
        rev[mask] = a[idx[mask]]

        # contribution of each index to the maximal entry
        contrib = a * rev
        i = int(np.argmax(contrib))               # source (largest contribution)

        # target – prefer tiny partner and tiny own value
        weight = rev + a                           # smaller is better
        weight[i] = np.inf                         # avoid i == j
        j = int(np.argmin(weight))                 # target index

        # partner values for i and j (needed for delta bound)
        a_rev_i = rev[i]
        a_rev_j = rev[j]

        # second largest convolution entry (excluding the current maximum)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()

        # bound on delta so that the new maximum does not surpass `second_max`
        if a_rev_i > a_rev_j:
            # change in the maximal entry = delta * (a_rev_j - a_rev_i)
            # we want b_max + delta*(a_rev_j - a_rev_i) ≤ second_max
            allowed = (b[k_max] - second_max) / (a_rev_i - a_rev_j)
            allowed = max(allowed * 0.95, 0.0)   # safety factor
        else:
            allowed = a[i]                       # no useful reduction possible

        # pick a random amount up to the computed bound (and the fraction schedule)
        delta = min(a[i] * max_frac, allowed) * np.random.rand()
        if delta <= 1e-16 or a[i] <= 0.0:
            # fall back to a tiny random move if the deterministic choice stalls
            delta = a[i] * max_frac * np.random.rand()
            if delta <= 0.0:
                it += 1
                continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        # decrease contribution of the source index
        b_new[i:i + n] -= 2.0 * delta * a
        # increase contribution of the target index
        b_new[j:j + n] += 2.0 * delta * a
        # self‑interaction corrections
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:          # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill2 {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – perturb and restart from the best seen
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill2 {it:7d}] perturbing after stagnation")

        it += 1

    return best_a

# Phase C – short mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a' = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
