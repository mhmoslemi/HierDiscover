"""
Propose a step‑function (list of non‑negative heights) that minimises

    evaluate_sequence(sequence) = 2 * n * max(conv(sequence,sequence)) / sum^2

The search consists of three stages:

1. Projected sub‑gradient descent on the exact objective.
2. Aggressive mass‑transfer hill‑climbing (as in the reference code).
3. Simulated‑annealing style random moves to escape any remaining local
   minima.  The phases respect the global time budget (default 1000 s) and
   keep the best vector found.

All vectors are kept on the probability simplex (non‑negative, sum = 1);
the final sequence is returned unchanged – `evaluate_sequence` will apply
its own clipping and normalisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for convolution and objective
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # eliminate tiny negatives caused by numerical noise
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the convolution entry conv[k] w.r.t. a."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad


# ----------------------------------------------------------------------
# Phase 1 – Projected sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    step0 = 0.5  # initial step‑size scale
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        grad = _gradient_of_lag(a, k)

        # back‑tracking line search
        lr = step0 / np.sqrt(t)
        improved = False
        for _ in range(10):
            a_trial = a - lr * grad
            a_trial = np.clip(a_trial, 0.0, None)
            s = a_trial.sum()
            if s == 0.0:
                lr *= 0.5
                continue
            a_trial /= s                     # projection onto simplex
            val = _objective(a_trial)
            if val < best_val:
                a = a_trial
                best_a = a_trial.copy()
                best_val = val
                improved = True
                step0 = min(step0 * 1.02, 2.0)   # modest increase after success
                if t % 200 == 0:
                    print(f"[subgrad] iter {t:5d}  val = {best_val:.6f}")
                break
            else:
                lr *= 0.5

        if not improved:
            step0 = max(step0 * 0.95, 1e-4)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Aggressive mass‑transfer hill‑climbing
# ----------------------------------------------------------------------
def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 5000) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a fraction of the most damaging mass to two low‑mass
    positions. Stops after `max_no_improve` consecutive failures or when
    time runs out.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2                # initial fraction of mass to relocate
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that make up the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass to shift
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # pick two distinct low‑mass destinations, avoiding the donors
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        # sum is already 1 (mass moved), no renormalisation needed
        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Simulated annealing (random mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Randomly moves a small amount of mass between two positions.
    Accepts worsening moves with a temperature that decays linearly.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    # Initial temperature – 5 % of the current objective (tunable)
    T0 = max(cur_val * 0.05, 1e-4)
    start = time.time()
    total_time = max(deadline - start, 1e-6)

    while time.time() < deadline:
        # linear cooling schedule
        elapsed = time.time() - start
        frac = elapsed / total_time
        T = T0 * (1.0 - frac) + 1e-9

        # propose a move: pick source i (must have some mass) and target j
        i = rng.integers(n)
        if a[i] == 0.0:
            continue
        # pick j ≠ i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 10 %) of the source mass
        max_delta = a[i] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # (sum stays 1, non‑negativity guaranteed by construction)

        new_val = _objective(new_a)
        dE = new_val - cur_val

        if dE <= 0.0 or rng.random() < np.exp(-dE / T):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val:.6f}")

    return best_a, best_val
