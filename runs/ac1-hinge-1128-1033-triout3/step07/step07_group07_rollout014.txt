"""
Minimal‐max‑convolution search.

The algorithm follows several complementary phases:

1. **Smooth surrogate optimisation** – minimise a log‑sum‑exp
   approximation of the max‑convolution using L‑BFGS‑B on the
   soft‑max parametrisation `a = softmax(w)`.  The smoothing
   parameter ε is decreased on a logarithmic schedule.

2. **Random restarts** – the best solution so far is perturbed in the
   log‑space and re‑optimised with a finer ε‑schedule.  This helps to
   escape local minima.

3. **Aggressive mass‑transfer hill‑climbing** – repeatedly move a
   fraction of mass away from the lag that attains the maximal
   convolution, accepting only improving moves.

4. **Simple mixing moves** – convex mixtures of the current solution
   with the uniform distribution and with its mirror image often reduce
   the central dot‑product `a·reverse(a)`.

5. **Sub‑gradient descent** – a projected sub‑gradient step on the
   exact objective `2·n·max(conv)` (the L∞‑norm of the convolution).

6. **Simulated annealing** – a temperature‑controlled random walk that
   transfers a small amount of mass between two coordinates; occasional
   uphill moves help to explore the neighbourhood.

All phases respect the total budget (default 1000 s).  The best
sequence found is returned as a plain Python list of floats (non‑negative,
summed to 1, so that `evaluate_sequence` returns `2·n·max(conv)`).

"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using a real FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # remove numerical artefacts
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (value, gradient) of the smoothed objective

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε),

    where a = softmax(w) (∑a = 1, a_i ≥ 0).
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)          # improve numerical stability
    a = np.exp(w_shift)
    a = a / a.sum()

    # full autocorrelation (a * a)
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing term
    shifted = (conv - max_conv) / eps
    log_sum = np.logaddexp.reduce(shifted)          # = log Σ exp(...)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # probability over lags

    # convolution a * reverse(p)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                   # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]                 # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill‑climbing
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              best_val: float,
                              deadline: float,
                              rng: np.random.Generator,
                              step_start: float = 0.2):
    """
    Aggressive hill‑climbing: repeatedly move a fraction of the mass
    away from the lag that attains the maximal convolution.
    Returns the best vector encountered.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    best_a = cur_a.copy()
    step_frac = step_start

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # indices that contribute to lag k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]

        if contrib.size == 0:
            continue

        # pick a few strongest contributors
        top_n = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        # destination indices – preferably low‑mass locations
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue

        # renormalise to sum 1
        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-6)

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max‑convolution objective)
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         best_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Sub‑gradient descent on 2·n·max(conv) with Euclidean projection
    onto the probability simplex.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    best_a = cur_a.copy()
    eta = 0.1  # step size

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # gradient step + projection
        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            eta = min(eta * 1.05, 0.5)
        else:
            eta = max(eta * 0.7, 1e-5)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         rng: np.random.Generator,
                         n: int,
                         start_temp: float = 1e-2,
                         end_temp: float = 1e-6,
                         max_iter: int = 150_000):
    """
    Temperature‑controlled random walk that transfers a small amount of
    mass between two coordinates.  Occasionally accepts uphill moves.
    """
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_v = cur_val
    cur_v = cur_val

    temp = start_temp
    # geometric cooling
    decay = (end_temp / start_temp) ** (1.0 / max_iter)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # propose a move
        i = rng.integers(0, n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move a fraction of i's mass to j
        frac = rng.uniform(0.01, 0.2)
        delta = cur_a[i] * frac
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_v = 2.0 * n * np.max(_full_convolution(new_a))

        if new_v < cur_v or rng.random() < math.exp(-(new_v - cur_v) / max(temp, 1e-12)):
            cur_a = new_a
            cur_v = new_v
            if new_v < best_v:
                best_v = new_v
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_v:.6f}")

        temp *= decay

    return best_a, best_v

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Optimises a non‑negative step‑function to minimise
    ``evaluate_sequence``.  The routine respects a time budget and returns
    the best list of floats it found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – a few thousand entries give a good trade‑off
    n = 4000

    # ------------------------------------------------------------------
    # Initialise – reuse a previously discovered construction if available
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        else:
            a0 = a0.copy()
    except Exception:
        # fall back to a uniform start (simple and feasible)
        a0 = np.full(n, 1.0 / n)

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    best_a = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))
    print(f"[init] n={n}, objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Smooth optimisation with a decreasing ε‑schedule
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-1, -14, num=35)  # start a bit smoother than before
    w = np.log(best_a + 1e-12)  # soft‑max parameters

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        res = minimize(fun_and_grad, w,
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": 200,
                                "ftol": 1e-12,
                                "gtol": 1e-12,
                                "disp": False})
        w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[smooth eps={eps:.1e}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Random restarts with a finer ε‑tail
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0 and restart < 30:
        restart += 1
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.8, size=n)

        # only the most aggressive part of the schedule is needed here
        for eps in eps_schedule[-12:]:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad, w,
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": 150,
                                    "ftol": 1e-12,
                                    "gtol": 1e-12,
                                    "disp": False})
            w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart {restart}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Aggressive mass‑transfer hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _mass_transfer_hill_climb(best_a,
                                                    best_val,
                                                    deadline,
                                                    rng,
                                                    step_start=0.2)

    # ------------------------------------------------------------------
    # 4. Simple mixing with uniform and with the mirror image
    # ------------------------------------------------------------------
    if time.time() < deadline:
        uniform = np.full_like(best_a, 1.0 / n)
        rev = best_a[::-1]
        for t in (0.05, 0.10, 0.15):
            cand = (1 - t) * best_a + t * uniform
            cand /= cand.sum()
            cand_val = 2.0 * n * np.max(_full_convolution(cand))
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
                print(f"[mix uniform t={t:.2f}] new best = {best_val:.6f}")

        for t in (0.05, 0.10):
            cand = (1 - t) * best_a + t * rev
            cand /= cand.sum()
            cand_val = 2.0 * n * np.max(_full_convolution(cand))
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
                print(f"[mix reverse t={t:.2f}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Fine sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _subgradient_descent(best_a,
                                                best_val,
                                                deadline,
                                                rng)

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        best_a, best_val = _simulated_annealing(best_a,
                                                best_val,
                                                deadline,
                                                rng,
                                                n,
                                                start_temp=1e-2,
                                                end_temp=1e-6,
                                                max_iter=120_000)

    # ------------------------------------------------------------------
    # Final reporting using the supplied evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
