def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

Assure sum(w) == 1.

def _grad_max_conv(a, k):

Algorithm
---------

    n = a.size
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a vector with Σa = 1  (objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    n = a.size
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # fallback (should not happen)
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the *active* convolution entry (k_max) w.r.t. a
# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """∇_a (a*a)[k]  (linear convolution of a with itself)."""
    n = a.size
    # map any k ≥ n to its symmetric counterpart (convolution is symmetric)
    if k >= n:
        k = 2 * n - 2 - k
    idx = k - np.arange(n)
    grad = np.zeros_like(a)
    mask = (idx >= 0) & (idx < n)
    grad[mask] += a[idx[mask]]
    idx2 = np.arange(n) - k
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] += a[idx2[mask2]]
    return grad


# ----------------------------------------------------------------------
# Phase‑0: deterministic families (the same collection as the reference)
# ----------------------------------------------------------------------
def _initial_candidates(n: int):
    """Return the best vector among a handful of hand‑crafted families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.size != n or np.any(a < 0):
            return
        a = a / a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # --------------------------------------------------------------
    # Simple deterministic families (exactly the ones from the reference)
    # --------------------------------------------------------------
    try_candidate(np.arange(n, 0, -1, dtype=float))                  # linear decreasing
    idx = np.arange(n)
    try_candidate(np.maximum(idx + 1, n - idx))                     # linear U‑shape
    try_candidate((np.arange(n, 0, -1, dtype=float) ** 2))          # quadratic decreasing
    quad_u = (np.arange(n) + 1) ** -2 + (np.arange(n, 0, -1) + 0) ** -2
    try_candidate(quad_u)                                            # quadratic U‑shape
    try_candidate(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))   # half‑wave sine
    try_candidate(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine peaks at ends

    # power‑law symmetric (U‑shape)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_candidate(a)

    # exponential left‑hand side
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        try_candidate(a)

    # symmetric (U‑shaped) exponential
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_candidate(a)

    # truncated exponential (mass only on a prefix)
    for frac in (0.10, 0.15, 0.20, 0.25, 0.30):
        K = max(1, int(frac * n))
        for r in (0.99, 0.98, 0.95, 0.90):
            a = np.zeros(n, dtype=float)
            a[:K] = r ** np.arange(K, dtype=float)
            try_candidate(a)

    # a few random Dirichlet draws (α=1)
    rng = np.random.default_rng(12345)
    for _ in range(20):
        a = rng.dirichlet(np.ones(n))
        try_candidate(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑1: massive random Dirichlet exploration with several alphas
# ----------------------------------------------------------------------
def _random_dirichlet_search(a_start: np.ndarray,
                             n: int,
                             start: float,
                             deadline: float,
                             best_a: np.ndarray,
                             best_val: float) -> tuple[np.ndarray, float]:
    """Sample many Dirichlet vectors with varying concentrations."""
    rng = np.random.default_rng(424242)
    alphas = [0.1, 0.2, 0.5, 1.0, 2.0, 5.0]   # span from very spiky to almost uniform
    it = 0
    while time.time() < deadline:
        # rotate through alphas to keep a diverse mixture
        alpha = alphas[it % len(alphas)]
        # Dirichlet via Gamma(α,1) draws
        g = rng.gamma(shape=alpha, scale=1.0, size=n)
        a = g / g.sum()
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Rand  {it:6d}] α={alpha:.2f}  new best = {best_val:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑2: sub‑gradient descent on the exact L∞ objective
# ----------------------------------------------------------------------
def _subgradient_descent(a_start: np.ndarray,
                         n: int,
                         start: float,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float) -> tuple[np.ndarray, float]:
    """Iteratively reduce the maximal convolution entry with a projected sub‑gradient."""
    a = a_start.copy()
    rng = np.random.default_rng(20231027)
    no_improve = 0
    it = 0
    # schedule: start with relatively large step, decay to 1e‑5
    while time.time() < deadline:
        conv = _full_convolution(a)
        max_b = conv.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD  {it:6d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # linear decay of the learning rate
        progress = (time.time() - start) / (deadline - start)
        lr = 0.3 * (0.001 / 0.3) ** progress   # 0.3 → 0.001

        # gradient of the *active* entry
        k_max = int(np.argmax(conv))
        grad = _grad_max_conv(a, k_max)

        # sub‑gradient step and simplex projection
        a = a - lr * grad
        a = _project_simplex(a)

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.004, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        # restart after long stagnation
        if no_improve > 120_000:
            print(f"[SGD  {it:6d}] stagnation – re‑initialising from best")
            a = best_a.copy()
            jitter = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            no_improve = 0
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑3: aggressive hill‑climber (exact objective)
# ----------------------------------------------------------------------
def _hillclimb_exact(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> tuple[np.ndarray, float]:
    """Move a small fraction of mass from heavily‑contributing indices to lighter ones."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    max_frac_start, max_frac_end = 0.5, 0.001
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777777)

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        # pick a target index that is *light* for the reversed contribution
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0:
            probs_t = w / w.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill  {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # perturb after prolonged stagnation
        if no_improve > 250_000:
            print(f"[Hill  {it:7d}] perturbing after stagnation")
            a = best_a.copy()
            noise = rng.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑4 – tiny mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test convex combinations a' = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020, 0.050):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the allotted time budget.
    Returns a list of non‑negative floats (the sum is irrelevant;
    the objective is scale‑invariant).  The algorithm combines a
    deterministic library, massive random Dirichlet sampling,
    sub‑gradient descent on the exact L∞ norm and a fine‑grained
    hill‑climber, finishing with a short uniform‑mixing sweep.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries gives a good compromise.
    # ------------------------------------------------------------------
    n = 9000

    # ------------------------------------------------------------------
    # Phase 0 – deterministic families (quick cheap baseline)
    # ------------------------------------------------------------------
    try:
        # If a previous solution is available, interpolate it.
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidates(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)

    # ------------------------------------------------------------------
    # Phase 1 – massive random Dirichlet sampling (≈ 30 % of time)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * (deadline - start_time)
    best_a, best_val = _random_dirichlet_search(
        best_a, n, start_time, phase1_deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent on the exact max‑convolution (≈ 30 %)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.60 * (deadline - start_time)
    best_a, best_val = _subgradient_descent(
        best_a, n, start_time, phase2_deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Phase 3 – aggressive hill‑climber (≈ 35 % of remaining time)
    # ------------------------------------------------------------------
    phase3_deadline = deadline
    best_a, best_val = _hillclimb_exact(
        best_a, n, phase2_deadline, phase3_deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Phase 4 – final uniform mixing sweep
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final secure evaluation (the supplied wrapper will clamp values)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
