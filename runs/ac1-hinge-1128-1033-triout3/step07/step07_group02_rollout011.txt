"""
Improved step–function search for the `evaluate_sequence` objective.

Strategy
--------
1. Initialise a candidate sequence
   * Try to reuse the previous global `height_sequence_1`.
   * Sweep a rich 2‑parameter Beta family (more extreme exponents).
   * Add a few random starts for diversity.

2. Adam optimisation on a smooth‐max surrogate (as in the reference code).

3. Simple sub‑gradient descent on the *exact* max‑convolution.
   The gradient of the active convolution entry is known analytically
   (∂(a*a)_k / ∂a_i = 2·a_{k‑i} when the index is valid).  After each
   step we project back onto the probability simplex.

4. Simulated‑annealing hill‑climbing (identical to the reference routine)
   – this clean‑up step removes remaining spikes.

5. Return the best vector found (its sum is 1, which does not affect the
   objective because the evaluation function is scale‑invariant).

The three optimisation phases are each given a slice of the total budget
(≈30 % / 30 % / 40 %).  All heavy work is performed with NumPy/FFT, and the
code respects the 1000 s global time limit by checking the deadline after
each iteration.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    # reference: https://arxiv.org/abs/1309.1541
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return a uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and gradient w.r.t. a (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and its gradient.
    The gradient corresponds to the exact objective 2·n·max,
    approximated by a soft‑max with temperature `eps`.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with the reversed soft‑max weights
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 4·n from 2·n·max
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimiser that minimises a smooth approximation of the objective."""
    # initialise logits for the soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.001
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 30_000                     # restart after too many idle steps

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedule for temperature (eps) and learning rate (lr)
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max and gradient
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)

        # chain rule through the soft‑max (w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation to avoid lock‑in
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        # restart on stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restarting after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (projected onto simplex)
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          best_a: np.ndarray,
                          best_val: float,
                          target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent that directly follows the
    gradient of the *active* convolution entry.
    """
    a = a0.copy()
    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 60_000   # after this many idle steps we randomise a little

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Subgrad {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # locate a max‑convolution entry
        k_max = int(np.argmax(b))

        # gradient of the active entry: g_i = 2 * a_{k_max - i}  (if index valid)
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k_max - idx]

        # step size schedule – diminishing step
        step = 0.4 / np.sqrt(iteration + 1)

        a_new = a - step * grad
        a = _simplex_projection(a_new)

        # occasional small random jitter to escape plateaus
        if iteration % 7_000 == 0 and np.random.rand() < 0.25:
            a += np.random.normal(scale=0.001, size=n)
            a = _simplex_projection(a)

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # re‑initialise around current best with a small perturbation
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = _simplex_projection(a)
            no_improve = 0
            print(f"[Subgrad {iteration:7d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (identical to reference version)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Local search that transfers a small amount of mass from heavy to light indices."""
    a = a0.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule parameters
    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.012, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – probability proportional to contribution
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – biased towards *low* contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# Parametric sweep of a Beta‑shaped family (extended grid)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ (i+1)^{p‑1}(n‑i)^{q‑1}."""
    best_a = None
    best_val = np.inf

    # richer grid – includes very small and larger exponents
    p_vals = np.concatenate([
        np.geomspace(0.01, 0.5, 12),
        np.linspace(0.5, 3.0, 15),
        np.geomspace(3.0, 15.0, 8)[1:]
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        # pre‑compute i^{p‑1} once per p
        ipow = i ** (p - 1.0)
        for q in q_vals:
            a = ipow * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final call

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previous construction if available.
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – parametric sweep of a Beta‑shaped family.
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    try:
        shape_a, shape_val = _beta_family_best(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] beta‑family sweep improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # fallback random start if needed
    if best_a is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_a, best_val = a, _objective(a)
        print("[init] random start, value=", best_val)

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # Phase B – Projected sub‑gradient descent (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.60 * budget_s
    best_a, best_val = _subgradient_optimize(best_a, n,
                                             deadline=phase_b_deadline,
                                             start_time=phase_a_deadline,
                                             best_a=best_a,
                                             best_val=best_val,
                                             target_val=1.503)

    # ------------------------------------------------------------------
    # Phase C – Simulated‑annealing hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=phase_b_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
