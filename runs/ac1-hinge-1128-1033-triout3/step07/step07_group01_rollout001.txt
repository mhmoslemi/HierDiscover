"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

The algorithm consists of the following stages (all respecting a 1000 s
budget, the last few seconds are kept for a safety check):

1. **Initial pool** – a handful of diverse start vectors (uniform,
   previous best, Dirichlet, exponential, sorted random).  The best
   of these is kept.

2. **Smoothed L‑BFGS** – minimise a smooth log‑sum‑exp surrogate of the
   max‑convolution.  The schedule of smoothing parameters ε is fine‑grained
   and the optimiser is restarted many times.  After each restart both a
   descending‑order and an “alternating‑ends” reorder are tested.

3. **Exact mirror‑descent** – sub‑gradient descent on the true
   ``max(conv)`` using the full gradient
   `g_i = a_i + a_{k*-i}` (where *k* is a maximising index of the
   convolution).  The update is exponential (`a ← a·exp(−η·g)`) which
   preserves non‑negativity; after each step the vector is normalised to
   the simplex.  The step size η decays geometrically.

4. **Simulated‑annealing style perturbations** – random mass transfers,
   swaps and block reversals are attempted; downhill moves are always
   accepted while uphill moves are accepted with a temperature‑controlled
   probability.

5. **Final cheap hill‑climbing** – deterministic swaps / mass transfers
   that strictly improve the objective.

All convolutions are performed with an FFT (power‑of‑2 length) for speed.
The fast evaluator `_fast_evaluate` exploits the fact that the sum of the
vector is kept equal to 1, so the objective reduces to `2·n·max(conv)`.
The best sequence found is finally checked with the official
`evaluate_sequence` before being returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate used by L‑BFGS
# ----------------------------------------------------------------------
_GLOBAL_EPS = None  # smoothing parameter ε
_GLOBAL_N = None    # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical artefacts may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate of ``2·n·max(conv)`` and its gradient w.r.t.
    the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max ----------------------------------------------------
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- convolution -------------------------------------------------
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max --------------------------------------------------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a (see derivation in docstring) -------------
    weights = np.exp(shifted)
    p = weights / weights.sum()      # soft distribution over lag indices

    # Convolution of a with reversed p gives ∂L/∂a (up to constant factor)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # Extract the part that aligns with a (see original implementation)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # ----- chain rule through the soft‑max -----------------------------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that are
    non‑negative and sum to 1.
    Returns 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Return a copy of ``a`` sorted in descending order."""
    return np.sort(a)[::-1]


def _alternating_ends_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values alternately at the leftmost and rightmost
    positions: 0, n‑1, 1, n‑2, …
    """
    vals = np.sort(a)[::-1]  # descending
    n = len(vals)
    out = np.empty_like(vals)
    left = 0
    right = n - 1
    idx = 0
    while left <= right:
        out[left] = vals[idx]
        idx += 1
        left += 1
        if left <= right:
            out[right] = vals[idx]
            idx += 1
            right -= 1
    return out


def _compute_max_gradient(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max(conv)`` at the current vector a.
    For each maximising index k we have
        ∂conv_k/∂a_i = a_i + a_{k-i}   (if k-i is a valid index)
    The returned gradient is the average over all maximising k.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_val = conv.max()
    # Use a tolerance to capture multiple maxima caused by round‑off.
    tol = 1e-12
    max_idxs = np.flatnonzero(conv >= max_val - tol)
    g = np.zeros_like(a)

    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx
        # Both idx and jdx are within [0, n‑1]
        g[idx] += a[idx] + a[jdx]

    g /= max_idxs.size
    return g


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine follows the multi‑phase outline described in the module
    docstring and respects the allotted time budget.
    Returns the best sequence found as a plain Python list.

    rng = np.random.default_rng(seed)

    start_time = time.time()

    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine problem size (length of the known construction if present).
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]

        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # 1️⃣  Initialise a diverse pool of candidates.
    # ------------------------------------------------------------------
    init_candidates = []

    # (a) Uniform distribution
    init_candidates.append(np.full(n, 1.0 / n))

    # (b) Previous best if available

        init_candidates.append(old / old.sum())

    # (c) Random Dirichlet (α = 1 — sparse)
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))

    # (d) Random Dirichlet (α = 5 — smoother)
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # (e) Exponential‑like randomness (α = 0.5)
    init_candidates.append(rng.gamma(shape=0.5, scale=1.0, size=n))

    rand_vec = rng.random(n)
    init_candidates.append(_descending_reorder(rand_vec))

    # Normalise all candidates onto the simplex.
    init_candidates = [c / c.sum() for c in init_candidates]

    # ------------------------------------------------------------------
    # Pick the best cheap candidate.
    # ------------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Phase 1 – L‑BFGS on a smooth surrogate.
    # ------------------------------------------------------------------
    # Schedule: start relatively smooth and finish extremely sharp.
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=15)
    # Initialise soft‑max parameters.
    w = np.log(best_a + 1e-12)

    max_restarts = 18
    restart = 0
    while time.time() < deadline - 35.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        if restart % 5 == 0:

            w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 35.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(
                _smooth_obj_and_grad_wrapper,
                w,
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': 1200,
                         'ftol': 1e-12,
                         'gtol': 1e-12,
                         'disp': False}
            )
            w = res.x
            # Back‑transform to probabilities.
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ------------------------------------------------------------------
        # After each restart, try a couple of deterministic reorders.
        # ------------------------------------------------------------------
        for reorder_fn, name in [( _descending_reorder, "desc"),
                                 ( _alternating_ends_reorder, "alt_ends")]:
            a_ord = reorder_fn(best_a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[reorder {name}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Phase 2 – Exact mirror‑descent (sub‑gradient on the true max).
    # ------------------------------------------------------------------

    # Step‑size schedule – start fairly aggressive and decay.

    decay = 0.9996
    min_eta = 0.005
    it = 0
    while time.time() < deadline - 25.0:

                new_a[i], new_a[j] = new_a[j], new_a[i]
            else:                                       # mass transfer
