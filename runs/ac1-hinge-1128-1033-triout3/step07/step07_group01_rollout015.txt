"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The search consists of several complementary phases that run for up to
1000 s:

1. **Diverse initialisations** – uniform, several deterministic families
   (exponential, power‑law) and the previously known construction.  The
   best of these seeds is kept.

2. **Smooth L‑BFGS** – minimise a log‑sum‑exp surrogate of the
   max‑convolution.  The smoothing parameter ε is decreased gradually.
   After each L‑BFGS run the vector is forced to be monotone decreasing
   (the “anti‑symmetric’’ shape found to be beneficial).

3. **Exact mirror‑descent on the true max‐convolution** – the full
   sub‑gradient (all active shifts) is used, with an exponential update
   and an adaptive step‑size.  The vector is kept monotone decreasing
   after every step.

4. **Simulated‑annealing on the monotone simplex** – random log‑space
   perturbations of the current best vector are generated, re‑sorted
   (to stay monotone) and accepted with a Metropolis criterion.
   Temperature and perturbation magnitude are annealed.

5. **Cheap hill‑climbing** – random swaps / small mass transfers
   (projected back onto the monotone simplex) to polish the result.

All phases respect a global time budget; the best sequence found is
returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS wrapper (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None  # smoothing parameter ε
_GLOBAL_N = None    # length of the sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (useful for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of power‑of‑2 size.
    All entries are non‑negative, so tiny negative rounding artefacts are
    clipped to zero.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient w.r.t. soft‑max
# parameters w_i (so that a_i = exp(w_i)/∑exp(w)).
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # --- soft‑max (a lives on the probability simplex) ---
    w_shift = w - np.max(w)                # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # --- convolution ---
    conv = _full_convolution(a)            # length 2·n‑1
    max_c = conv.max()

    # --- smooth max:
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # --- gradient w.r.t. a (see derivation in the original code) ---
    weights = np.exp(shifted)              # proportional to exp((c_k-max)/ε)
    p = weights / weights.sum()            # “soft‑max” over shift indices

    # correlation of a with the reversed weight vector gives ∂max/∂a
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]        # the part that aligns with a
    grad_a = 4.0 * n * seg[::-1]          # reverse to match a_i order

    # --- chain rule through the soft‑max ---
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _make_monotone(v: np.ndarray) -> np.ndarray:
    """
    Return a monotone *non‑increasing* version of v, normalised to sum = 1.
    The operation consists of sorting in descending order and projecting
    onto the simplex (the scaling does not break monotonicity).
    """
    v = np.sort(v)[::-1]               # non‑increasing order
    v = _project_simplex(v)            # ensures Σv = 1 and v≥0
    return v


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Efficient version of ``evaluate_sequence`` for vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of the map a ↦ max_k (a*a)_k.
    For all shift indices k that attain the maximum (within a tiny tolerance)
    we accumulate the contribution a_{k-i}.
    """
    conv = _full_convolution(a)
    max_c = conv.max()
    # tolerance – we treat any shift within 1e‑12 of the max as active.
    tol = 1e-12
    active = np.where(conv >= max_c - tol)[0]

    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx
        g[idx] += a[jdx]                # contribution of this active shift
    return g


# ----------------------------------------------------------------------
# Core search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The function respects a total running time of ``budget_s`` seconds and
    returns the best sequence found as a plain Python list.
    """
    # ------------------------------------------------------------------
    # Basic set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Keep a safety margin (5 s) for the final call to evaluate_sequence().
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – reuse the length of the previous best construction
    # if it exists, otherwise fall back to a reasonable default.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate quickly and keep track of the best.
    # ------------------------------------------------------------------
    def _maybe_update_best(a: np.ndarray, cur_val: float | None = None):
        nonlocal best_a, best_val
        if cur_val is None:
            cur_val = _fast_evaluate(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[update] new best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Generate a handful of diverse start points.
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # exponential families
    for lam_mul in (0.5, 1.0, 2.0, 5.0):
        lam = lam_mul / n                     # λ = mul / n
        i = np.arange(n, dtype=float)
        a = np.exp(-lam * i)
        a = _make_monotone(a)                 # ensure monotone & sum=1
        init_candidates.append(a)

    # power‑law families
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        a = 1.0 / (i ** p)
        a = _make_monotone(a)
        init_candidates.append(a)

    # random decreasing vectors
    for _ in range(3):
        rnd = rng.random(n)
        a = np.sort(rnd)[::-1]
        a = _make_monotone(a)
        init_candidates.append(a)

    # ------------------------------------------------------------------
    # Pick the best of the cheap initial candidates.
    # ------------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] value = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Phase – L‑BFGS on a smooth surrogate (log‑sum‑exp).
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-14), num=15)
    w = np.log(best_a + 1e-12)                     # initialise soft‑max params

    max_restarts = 15
    restart = 0
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        if restart % 5 == 0:
            # Occasionally inject a fresh random start.
            w = np.log(np.sort(rng.random(n))[::-1] + 1e-12)

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            # back‑transform to probabilities
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            # enforce monotone decreasing – this is the “anti‑symmetric’’ shape
            a_candidate = _make_monotone(a_candidate)

            cur_val = _fast_evaluate(a_candidate)
            _maybe_update_best(a_candidate, cur_val)

    # ------------------------------------------------------------------
    # 3. Phase – exact mirror‑descent on the true max‑convolution.
    # ------------------------------------------------------------------
    a = best_a.copy()
    step = 0.5                                   # initial step size
    it = 0
    while time.time() < deadline - 20.0:
        g = _subgradient_maxconv(a)               # exact sub‑gradient
        a_new = a * np.exp(-step * g)             # exponential mirror step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = np.full(n, 1.0 / n)
        else:
            a_new /= s
        a_new = _make_monotone(a_new)             # keep monotone shape

        new_val = _fast_evaluate(a_new)

        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[mirror {it}] new best = {best_val:.6f}")

        # accept move if it improves, otherwise keep the old vector
        if new_val < _fast_evaluate(a) - 1e-12:
            a = a_new
            # slowly increase the step if we are making progress
            step = min(step * 1.02, 1.0)
        else:
            # otherwise shrink the step to become more cautious
            step *= 0.92
            step = max(step, 1e-4)

        if it % 200 == 0:
            cur_val = _fast_evaluate(a)
            print(f"[mirror it {it}] current value = {cur_val:.6f}")
        it += 1

    # ------------------------------------------------------------------
    # 4. Phase – simulated‑annealing style random log‑space perturbations.
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.02                     # initial temperature
    cooling = 0.99995
    sigma = 0.03                 # initial perturbation magnitude
    sigma_decay = 0.99993

    while time.time() < deadline - 10.0:
        # propose a candidate by adding Gaussian noise in log‑space
        log_a = np.log(cur_a + 1e-12)
        pert = sigma * rng.normal(size=n)
        cand_log = log_a + pert
        cand = np.exp(cand_log)
        cand = _make_monotone(cand)          # keep monotone & normalised

        cand_val = _fast_evaluate(cand)

        if cand_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(cand_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = cand
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = cand.copy()
                print(f"[SA] new best = {best_val:.6f}")

        # anneal temperature and perturbation size
        T = max(T * cooling, 1e-6)
        sigma = max(sigma * sigma_decay, 5e-4)

    # ------------------------------------------------------------------
    # 5. Phase – cheap hill‑climbing (strict improvements only).
    # ------------------------------------------------------------------
    def _refine_hill_climb(a_vec: np.ndarray, cur_best: float,
                           limit: float) -> tuple[np.ndarray, float]:
        """Very cheap hill‑climb that tries swaps and mass‑transfer moves."""
        cur = a_vec.copy()
        cur_val = cur_best
        rng_local = np.random.default_rng(seed + 9999)

        while time.time() < limit:
            if rng_local.random() < 0.2:     # pure swap of two entries
                i, j = rng_local.integers(0, n, size=2)
                if i == j:
                    continue
                new = cur.copy()
                new[i], new[j] = new[j], new[i]
            else:                           # mass transfer
                i, j = rng_local.integers(0, n, size=2)
                if i == j or cur[i] <= 1e-15:
                    continue
                delta = cur[i] * 0.07 * rng_local.random()
                new = cur.copy()
                new[i] -= delta
                new[j] += delta

            # project back onto monotone simplex
            new = np.maximum(new, 0.0)
            new = _make_monotone(new)

            val = _fast_evaluate(new)
            if val < cur_val - 1e-12:
                cur = new
                cur_val = val
                print(f"[refine] improved to {cur_val:.6f}")
        return cur, cur_val

    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final verification with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
