def hillclimb_deterministic(a_start, n, start, deadline, best_a, best_val):
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))
        # compute gradient g for this k_max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros(n, dtype=np.float64)
        g[mask] = 2.0 * a[idx[mask]]
        # source selection
        source_scores = g * a
        i = int(np.argmax(source_scores))
        if a[i] <= 1e-15:
            # fallback: choose any positive index
            pos = np.flatnonzero(a > 1e-15)
            if pos.size == 0:
                break
            i = int(np.random.choice(pos))
        # target selection: inverse weighting of g
        w = np.zeros(n, dtype=np.float64)
        mask_target = np.arange(n) != i
        w[mask_target] = 1.0 / (g[mask_target] + 1e-12)
        if w.sum() > 0:
            probs = w / w.sum()
            j = int(np.random.choice(n, p=probs))
        else:
            # uniform fallback
            j = int(np.random.randint(n-1))
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of b
        b_new = b.copy()
        # shift add/sub
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # delta^2 contributions
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        # clip small negatives
        # (optional)
        # b_new = np.maximum(b_new, 0.0)

        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[HillDet {it:7d}] new best = {best_val:.6f}")
                if best_val <= 1.503:  # early stop

            a = np.maximum(a + noise, 0.0)

            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillDet {it:7d}] perturbing after stagnation")

"""
Improved search for a non‑negative step function minimizing
`evaluate_sequence`.  The optimiser proceeds in three stages:

1️⃣  A quick parametric sweep (linear, quadratic, sinus, exponential,
    power‑law, random Dirichlet) to obtain a good initial vector.

2️⃣  Adam optimisation of a smooth approximation of the max‑convolution.
    The soft‑max temperature and learning rate are annealed to very
    small values, giving a fine‑grained descent on the simplex.

3️⃣  Deterministic hill‑climbing on the exact objective.
    At each iteration we locate the current maximal convolution entry,
    compute its sub‑gradient (which tells which coefficients contribute
    most), move a fraction of mass from the most damaging index to a
    low‑gradient index, and update the convolution incrementally.
    The step‑size fraction decays from 0.5 → 0.001 and we reset from the
    best‑seen solution when progress stalls.

Finally we try a tiny convex combination with the uniform vector and
return the best sequence found.  All operations respect the 1000 s
budget and keep the vector non‑negative with unit sum (scale‐invariant for
the objective).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based for speed)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full, length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1} (Duchi et al.)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric initialisation – a rich library of shapes
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial probability vector and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ----- simple deterministic families -----
    try_candidate(np.arange(n, 0, -1, dtype=float))                       # decreasing
    idx = np.arange(n)
    lin_u = np.maximum(idx + 1, n - idx)                                 # U‑shape linear
    try_candidate(lin_u)

    try_candidate((np.arange(n, 0, -1, dtype=float) ** 2))                # quadratic decreasing
    quad_u = (np.arange(n) + 1) ** -2 + (np.arange(n, 0, -1) + 0) ** -2 # U‑shape quadratic
    try_candidate(quad_u)

    try_candidate(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))        # half‑wave sine
    try_candidate(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))    # cosine at ends

    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_candidate(a)

    # ----- exponential families -----
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        try_candidate(a)

    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_candidate(a)

    # ----- truncated exponential on a prefix -----
    for frac in (0.10, 0.15, 0.20, 0.25, 0.30):
        K = max(1, int(frac * n))
        for r in (0.99, 0.98, 0.95, 0.90):
            a = np.zeros(n, dtype=float)
            a[:K] = r ** np.arange(K, dtype=float)
            try_candidate(a)

    # ----- random Dirichlet draws -----
    rng = np.random.default_rng(12345)
    for _ in range(20):
        a = rng.dirichlet(np.ones(n))
        try_candidate(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam optimisation of a smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam on the log‑softmax parametrisation, smooth‑max of the convolution."""
    w = np.log(a0 + 1e-12)                     # logits (unconstrained)

    # hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 80_000

    it = 0
    while time.time() < deadline:
        # logits → simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max of the convolution
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()                     # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a 2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                  # length 3·n‑2
        segment = conv_ab[n - 1:2 * n - 1]              # n entries
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.008, size=n)

        if no_improve > IMPROVE_LIMIT:
            # restart from a fresh random point
            print(f"[Adam {it:7d}] restart after stagnation")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Deterministic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_deterministic(a_start: np.ndarray,
                             n: int,
                             start: float,
                             deadline: float,
                             best_a: np.ndarray,
                             best_val: float,
                             target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Move mass from the most damaging coefficient (by sub‑gradient) to a low‑gradient one."""
    a = a_start.copy()
    b = _full_convolution(a)                     # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # sub‑gradient for the entry k_max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros(n, dtype=np.float64)
        g[mask] = 2.0 * a[idx[mask]]

        # source – index with largest contribution to the maximal entry
        src_scores = g * a
        i = int(np.argmax(src_scores))
        if a[i] <= 1e-15:                               # fallback if numerical issue
            pos = np.flatnonzero(a > 1e-15)
            if pos.size == 0:
                break
            i = int(np.random.choice(pos))

        # target – low‑gradient indices, sampled inversely to g
        w = np.zeros(n, dtype=np.float64)
        mask_target = np.arange(n) != i
        w[mask_target] = 1.0 / (g[mask_target] + 1e-12)
        if w.sum() > 0:
            probs = w / w.sum()
            j = int(np.random.choice(n, p=probs))
        else:
            # uniform fallback
            j = int(np.random.randint(n - 1))
            if j >= i:
                j += 1

        # amount of mass to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[HillDet {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturbation around the best‑seen vector
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillDet {it:7d}] perturbing after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Fine‑grained projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _subgradient_fine_tune(a_start: np.ndarray,
                           n: int,
                           start: float,
                           deadline: float,
                           best_a: np.ndarray,
                           best_val: float,
                           target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = a_start.copy()
    cur_val = _objective_value(a)

    base_lr = 0.25
    it = 0
    while time.time() < deadline:
        # compute the current maximal convolution entry
        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # sub‑gradient for that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros(n, dtype=np.float64)
        g[mask] = 2.0 * a[idx[mask]]

        # diminishing step size
        lr = base_lr / np.sqrt(it + 1.0)

        # projected update
        a_candidate = _project_simplex(a - lr * g)
        cand_val = _objective_value(a_candidate)

        if cand_val < cur_val:
            a = a_candidate
            cur_val = cand_val
            if cand_val < best_val:
                best_val, best_a = cand_val, a.copy()
                print(f"[SubGrad {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Small post‑processing: convex mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few convex combinations (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative, Σ=1) that minimises
    `evaluate_sequence`.  The routine respects the time budget and returns
    the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good granularity.
    # ------------------------------------------------------------------
    n = 9000

    # ------------------------------------------------------------------
    # Initialise ``a`` – try to interpolate a previously discovered vector
    # (if the global ``height_sequence_1`` exists) else fall back to a sweep.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam on smooth max (≈ 10 % of the budget)
    # ------------------------------------------------------------------
    phase_a_dead = start_time + 0.10 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        a0, n,
        start=start_time,
        deadline=phase_a_dead,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase B – Deterministic hill‑climbing on the exact objective
    # (≈ 70 % of the budget)
    # ------------------------------------------------------------------
    phase_b_dead = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _hillclimb_deterministic(
        best_a, n,
        start=phase_a_dead,
        deadline=phase_b_dead,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase C – Projected sub‑gradient fine‑tuning (≈ 15 % of the budget)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_fine_tune(
        best_a, n,
        start=phase_b_dead,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final secure evaluation (the provided wrapper will clip & check)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
