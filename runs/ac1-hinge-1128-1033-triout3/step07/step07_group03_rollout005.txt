def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself (length 2·len‑1).
    Implemented via FFT for speed.

    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip any tiny negatives that may appear due to round‑off
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest values with the smallest ones when the sequence is
    reversed – a simple heuristic that often lowers the peak of the
    self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Smoothed max surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution a*a
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (derived analytically)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # prob. over lag indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to ∂L/∂a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # derivative w.r.t. a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by `scipy.minimize` (uses the globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule: from 5·10⁻³ down to 10⁻⁹ (geometric)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=9)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to a (probability simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # optional anti‑symmetric reorder
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # Fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val


# ----------------------------------------------------------------------
# Balanced mirror‑descent (uniformity + peak reduction)
# ----------------------------------------------------------------------
def _gradient_descent_balanced(a_start: np.ndarray, deadline: float):
    """
    Exponential mirror‑descent that reduces the *k* biggest convolution peaks
    while simultaneously flattening the whole convolution towards the uniform
    target 1/(2·n‑1).  Returns the best (a, value) pair found before
    ``deadline``.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[gd_bal init] value = {best_val:.6f}")

    it = 0
    eta0 = 0.5          # initial step‑size
    top_k = 3           # how many peaks to use for the max‑gradient
    lam_decay_iters = 500_000  # iterations after which λ→0

    while time.time() < deadline - 5.0:
        # ---------- convolution ----------
        conv = _full_convolution(a)          # length 2·n‑1
        max_conv = conv.max()

        # ---------- uniformity gradient ----------
        target = 1.0 / (2 * n - 1)           # ideal value for each entry
        d = conv - target                    # deviation from uniformity
        size = _next_pow2(len(d) + n - 1)
        fd = np.fft.rfft(d, size)
        fa_rev = np.fft.rfft(a[::-1], size)
        corr = np.fft.irfft(fd * fa_rev, size)
        grad_u = 4.0 * corr[n - 1:n - 1 + n]   # gradient of  ||conv‑target||²

        # ---------- max‑peak gradient ----------
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
        else:
            idxs = np.arange(conv.size)
        grad_max = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            jdx = k - idx_range
            grad_max[idx_range] += a[jdx]
        grad_max *= 2.0                       # derivative of conv_k w.r.t. a_i

        # ---------- combine ----------
        lam = max(0.0, 1.0 - it / lam_decay_iters)  # linearly decay λ → 0
        g = lam * grad_max + (1.0 - lam) * grad_u

        # ---------- step‑size ----------
        eta = eta0 * (0.98 ** (it // 2000))

        # ---------- mirror‑descent update (exponential) ----------
        a = a * np.exp(-eta * g)

        # ---------- project onto simplex ----------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-12:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        # ---------- occasional logging ----------
        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd_bal it {it}] new best = {best_val:.6f} "
                      f"(max={max_conv:.8f})")
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd_bal reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (random swaps / mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random local search: swaps (20 %) and mass transfers (80 %).  Keep only
    strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a few seconds safety margin before returning
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more freedom, but we keep it modest
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from a known construction if possible
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the desired length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = rng.exponential(scale=1.0, size=n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
        s = a.sum()
    a /= s

    # ------------------------------------------------------------------
    # Evaluate the initial candidate
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – a few random vectors (to avoid a poor start)
    # ------------------------------------------------------------------
    for tr in range(5):
        if time.time() >= deadline - 200:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – balanced mirror‑descent (uniformity + peak reduction)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        a_bal, val_bal = _gradient_descent_balanced(best_a, deadline)
        if val_bal < best_val:
            best_val = val_bal
            best_a = a_bal.copy()
            print(f"[balanced GD] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth surrogate optimisation (L‑BFGS) with a couple of restarts
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        restart = 0
        while time.time() < deadline - 30 and restart < 3:
            restart += 1
            w_start = np.log(best_a + 1e-12)
            w_start += rng.normal(scale=0.2, size=n)
            a_opt, val_opt = _l_bfgs_optimize_from_w(w_start, deadline)
            if val_opt < best_val:
                best_val = val_opt
                best_a = a_opt.copy()
                print(f"[restart {restart}] post‑L‑BFGS best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
