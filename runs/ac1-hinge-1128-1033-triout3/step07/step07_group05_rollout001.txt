def run_sa(a, init_val, time_limit):
    rng = np.random.default_rng()
    n = len(a)
    cur_a = a.copy()
    cur_val = init_val
    best_a = cur_a.copy()
    best_val = cur_val
    start = time.time()
    it = 0
    T = cur_val * 0.02 + 1e-12
    while time.time() < time_limit:
        it += 1
        move_type = rng.random()
        new_a = cur_a.copy()
        if move_type < 0.5:
            # mass transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or cur_a[i] <= 0:
                continue
            # choose delta fraction
            delta = cur_a[i] * rng.random() * (0.2 if rng.random() < 0.1 else 0.02) # bigger early
            new_a[i] -= delta
            new_a[j] += delta
        elif move_type < 0.8:
            # swap
            i, j = rng.integers(0, n, size=2)
            if i == j: continue
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # small random perturbation using Dirichlet noise
            # select k random indices to add noise
            idx = rng.choice(n, size=3, replace=False)
            # generate random vector, adjust sum to zero
            perturb = rng.normal(scale=0.01, size=3)
            # adjust to preserve sum
            perturb = perturb - perturb.mean()
            new_a[idx] += perturb
            # ensure non-negative
            new_a = np.clip(new_a, 0, None)
        # renormalize to sum=1
        sum_new = new_a.sum()
        if sum_new < 1e-12:
            continue
        new_a = new_a / sum_new
        new_val = _eval_fast(new_a)  # custom fast eval
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_a = cur_a.copy()
                best_val = cur_val
                print(f"[SA] new best {best_val:.6f} at iter {it}")
        # update temperature gradually
        T *= 0.99999
        if T < 1e-12:
            T = cur_val * 0.001
    return best_a, best_val
```

"""
Propose a step‑function sequence that minimises
    2·n·max(a*a) / (sum(a)^2)

The algorithm uses several complementary stages:

1.  Initialise from the previously found best construction
    (or fall back to a uniform distribution).  The vector is
    interpolated to a working size n = 8000 and normalised to
    ℓ₁‑norm = 1.

2.  Smooth surrogate optimisation (L‑BFGS) on a log‑sum‑exp
    approximation of the max‑convolution.  A few restarts with a
    decreasing smoothing parameter ε quickly shape the distribution.

3.  Deterministic “ends‑large” re‑ordering, which puts the largest
    values at the two extremes – a cheap heuristic that often reduces
    the peak of the self‑convolution.

4.  Projected sub‑gradient descent on the exact objective; this
    quickly exploits the current sub‑gradient while keeping the vector
    on the probability simplex.

5.  A long simulated‑annealing phase that performs three kinds of
    moves:
        * mass transfer (mainly from the centre to the ends),
        * swaps of two positions,
        * small random perturbations of a few entries.
    The acceptance probability follows the Metropolis rule with a
    slowly decaying temperature.  Multiple SA runs are launched while
    time permits, each starting from the best‑so‑far vector with a
    light random jitter.

6.  A final cheap hill‑climbing refinement (extra swaps / tiny mass
    transfers).

All stages respect a global wall‑time budget (default 1000 s).  The
best sequence found is finally evaluated with the official
`evaluate_sequence` function and returned as a plain Python list.

The implementation is fully self‑contained, uses only NumPy/ SciPy,
and avoids closures or lambda‑expressions as required.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate wrapper (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Clip tiny negative values caused by floating‑point round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast evaluator that mirrors ``evaluate_sequence`` but skips all
    safety checks.  Assumes ``a`` is a non‑negative vector.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate of the max‑convolution.

    Parameters
    ----------
    w   – soft‑max parameters (real numbers)
    eps – smoothing constant (ε > 0)
    n   – length of the sequence

    Returns
    -------
    L, grad_w : tuple
        L is the surrogate objective
        grad_w is the gradient w.r.t. ``w``
    """
    # ----- soft‑max (positivity, ℓ₁ = 1) -----
    w_shift = w - np.max(w)               # improve numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                         # Σ a = 1

    # ----- convolution a * a -----
    conv = _fft_convolve_self(a)          # length 2·n‑1
    max_conv = conv.max()

    # ----- smooth approximation of the max -----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log Σ exp((c−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    weights = np.exp(shifted)             # exp((c−max)/ε)
    p = weights / np.sum(weights)         # probability distribution over offsets

    # derivative of Σ_k p_k * conv_k w.r.t. a is the convolution a * p_rev
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    # central n entries correspond to indices [n‑1, 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # ∂L/∂a

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# Bind the helper convolution to the function object so the wrapper can see it.
_smooth_obj_and_grad._conv_fft = _conv_fft


def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic re‑ordering that puts the largest values at the two
    ends and the smallest near the centre.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left_ptr = 0
    right_ptr = n - 1
    i = 0
    while i < n:
        # place a large value
        new_a[i] = sorted_vals[right_ptr]
        right_ptr -= 1
        i += 1
        if i >= n:
            break
        # place a small value
        new_a[i] = sorted_vals[left_ptr]
        left_ptr += 1
        i += 1
    return new_a


def _subgradient_descent(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    The vector is kept on the ℓ₁‑simplex (non‑negative, Σa = 1).
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    step0 = 0.12           # initial step size
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur_a)
        k_max = int(np.argmax(conv))

        # gradient of the max entry w.r.t. a
        idx_arr = k_max - np.arange(n)
        mask = (idx_arr >= 0) & (idx_arr < n)
        grad = np.zeros_like(cur_a)
        grad[mask] = 2.0 * cur_a[idx_arr[mask]]

        alpha = step0 / np.sqrt(it + 1.0)
        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)
        sum_new = np.sum(new_a)
        if sum_new < 1e-12:          # fallback to uniform if degenerated
            new_a = np.full(n, 1.0 / n)
            sum_new = 1.0
        new_a /= sum_new               # project onto the simplex

        new_val = _eval_fast(new_a)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it} new best = {best_val:.6f}")

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val

        it += 1
    return best_a, best_val


def _refine_hill_climb(a0: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """
    Simple hill‑climbing that mixes swaps and tiny mass transfers.
    """
    cur_a = a0.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:          # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                           # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def _sa_mass_transfer(a_start: np.ndarray, cur_val: float,
                     deadline: float) -> (np.ndarray, float):
    """
    Simulated annealing that moves mass (mainly from the centre to the ends),
    swaps elements and performs tiny random perturbations.
    """
    n = a_start.shape[0]
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    best_a = cur_a.copy()
    best_val = cur_val

    # Initial temperature: a small fraction of the current value.
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99995

    # Region limits for biased moves.
    centre_start = n // 3
    centre_end = n - n // 3
    left_end = n // 6
    right_start = n - n // 6

    while time.time() < deadline:
        move_type = rng.random()
        new_a = cur_a.copy()

        if move_type < 0.5:                     # mass transfer (biased)
            if rng.random() < 0.7:
                # source mainly from centre, target from an end
                i = rng.integers(centre_start, centre_end)
                if rng.random() < 0.5:
                    j = rng.integers(0, left_end)
                else:
                    j = rng.integers(right_start, n)
            else:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)

            if new_a[i] <= 0.0:
                continue
            # Occasionally allow a larger chunk.
            frac = rng.random() * (0.2 if rng.random() < 0.1 else 0.02)
            delta = new_a[i] * frac
            new_a[i] -= delta
            new_a[j] += delta

        elif move_type < 0.8:                  # swap two positions
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a[i], new_a[j] = new_a[j], new_a[i]

        else:                                   # small random perturbation
            idx = rng.choice(n, size=4, replace=False)
            perturb = rng.normal(scale=0.01, size=4)
            perturb -= perturb.mean()                       # keep sum unchanged
            new_a[idx] += perturb
            new_a = np.clip(new_a, 0, None)

        # Re‑normalize to keep Σa = 1.
        sum_new = new_a.sum()
        if sum_new < 1e-12:
            continue
        new_a /= sum_new

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val

        if delta_val < 0.0 or rng.random() < np.exp(-delta_val / T):
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

    return best_a, best_val

        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()          # normalise to Σa = 1

    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Smooth surrogate L‑BFGS (few restarts, decreasing ε).
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=15)
    restart = 0
    while time.time() < deadline - 180.0:      # keep ~3 min for later phases
        restart += 1
        print(f"[restart {restart}]")
        # soft‑max parameters – log of current a plus noise
        w = np.log(best_a + 1e-12) + 0.3 * rng.standard_normal(n)

        for eps in eps_schedule:
            if time.time() >= deadline - 150.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 400,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate the current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ------------------------------------------------------------------
        # Re‑order: place large heights at the two ends.
        # ------------------------------------------------------------------
        a_ord = _ends_large_reorder(best_a)
        a_ord /= a_ord.sum()
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent on the exact max‑convolution.
    # ------------------------------------------------------------------
    if time.time() < deadline - 90.0:
        print("[subgrad] starting projected sub‑gradient descent")
        sub_deadline = deadline - 60.0
        best_a, best_val = _subgradient_descent(best_a, sub_deadline)

    # ------------------------------------------------------------------
    # 3. Long simulated‑annealing phase (mass transfers + swaps).
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[SA] entering long simulated‑annealing phase")
        # Keep running independent SA runs while we have time.
        while time.time() < deadline - 30.0:
            run_deadline = min(deadline - 2.0, time.time() + 40.0)

            # Start from the current best with a tiny random jitter.
            start_a = best_a.copy()
            if rng.random() < 0.6:
                jitter = rng.normal(scale=0.01, size=n)
                start_a = start_a + jitter
                start_a = np.clip(start_a, 0, None)
                start_a /= start_a.sum()

            cur_val = _eval_fast(start_a)
            cand_a, cand_val = _sa_mass_transfer(start_a, cur_val, run_deadline)

            if cand_val < best_val:
                best_val = cand_val
                best_a = cand_a.copy()
                print(f"[SA] overall new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final cheap hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report final value using the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    # Return a plain Python list as required by the framework.
    return best_a.tolist()
