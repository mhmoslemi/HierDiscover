            new[i] -= delta
            new[j] += delta

def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    weights = np.exp(shifted)
    p = weights / weights.sum()           # distribution over lag indices

    # Convolution of a with reversed p (using FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # derivative w.r.t. a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

# ----------------------------------------------------------------------
def _gradient_descent_multi(a_start: np.ndarray, deadline: float,
                           top_k: int = 5,
                           alpha0: float = 0.7,
                           decay_rate: float = 0.98) -> tuple[np.ndarray, float]:
    """
    Exponential mirror‑descent that pushes down the *top_k* largest convolution peaks.
    Returns (best_a, best_val) before the deadline.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[gd init] value = {best_val:.6f}")

    it = 0
    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_val = conv.max()

        # indices of the ``top_k`` largest convolution entries
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]   # sorted descending
        else:
            idxs = np.arange(conv.size)

        # build gradient estimate from selected peaks
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            jdx = k - idx_range
            g[idx_range] += a[jdx]

        # step size schedule
        alpha = alpha0 * (decay_rate ** (it // 2000))

        # multiplicative mirror‑descent update
        a = a * np.exp(-alpha * g)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:                     # guard against under‑flow
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        # occasional logging and extra re‑ordering
        if it % 50 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd iter {it}] new best = {best_val:.6f} (max conv {max_val:.8f})")
            if it % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] improved to {best_val:.6f}")

        it += 1
    return best_a, best_val

def _simulated_annealing(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Classic simulated‑annealing on the true objective.
    Random mass transfers (and occasional swaps) are accepted with a
    Metropolis probability that slowly cools down.
    """
    n = a_start.shape[0]
    cur = a_start.copy()
    cur_val = evaluate_sequence(cur.tolist())
    best = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    # initial temperature – a fraction of the current score
    T0 = max(1e-4, 0.05 * cur_val)
    Tmin = 1e-6

    while time.time() < deadline - 2.0:
        # linear cooling
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = (1 - frac) * T0 + frac * Tmin

        # propose a move
        if rng.random() < 0.7:   # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] <= 1e-15:
                continue
            # move a random fraction (up to 80%) of the mass at i
            frac_move = rng.random() * 0.8
            delta = cur[i] * frac_move
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        else:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]

        # re‑project onto simplex (mainly to correct rounding)
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s < 1e-12:
            continue
        new /= s

        new_val = evaluate_sequence(new.tolist())
        delta_val = new_val - cur_val

        accept = False
        if delta_val < 0:
            accept = True
        else:
            # Metropolis acceptance
            if T > 0 and rng.random() < np.exp(-delta_val / T):
                accept = True

        if accept:
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
                print(f"[SA] new best = {best_val:.6f}")

        # every few thousand steps try a cheap re‑ordering heuristic
        if rng.random() < 0.001:
            cand = _anti_symmetric_reorder(cur)
            val = evaluate_sequence(cand.tolist())
            if val < best_val:
                best_val = val
                best = cand.copy()
                cur = cand.copy()
                cur_val = val
                print(f"[SA reorder] improved to {best_val:.6f}")

    return best, best_val

# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int) -> tuple[np.ndarray, float]:
    """
    Local random search that keeps only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:

            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val

# Multi‑peak exponential mirror descent (core of the search)
# ----------------------------------------------------------------------
def _gradient_descent_multi(a_start: np.ndarray, deadline: float,
                           top_k: int = 5,
                           alpha0: float = 0.8,
                           decay_rate: float = 0.97):
    """
    Exponential mirror‑descent that reduces the *top_k* largest convolution peaks.
    Returns the best (a, value) pair found before ``deadline``.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[gd init] value = {best_val:.6f}")

    it = 0
    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_val = conv.max()

        # indices of the top_k largest convolution entries
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]   # sorted descending
        else:
            idxs = np.arange(conv.size)

        # gradient estimate from selected peaks
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            jdx = k - idx_range
            g[idx_range] += a[jdx]

        # scheduled step‑size
        alpha = alpha0 * (decay_rate ** (it // 2000))

        # multiplicative update (mirror‑descent)
        a = a * np.exp(-alpha * g)

        # projection onto the probability simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:                     # guard against under‑flow
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        # occasional logging / re‑ordering
        if it % 50 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd iter {it}] new best = {best_val:.6f} (max conv {max_val:.8f})")
            if it % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] improved to {best_val:.6f}")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (global search)
# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray, deadline: float):
    """
    Classic simulated‑annealing on the true objective.
    Random mass transfers (and occasional swaps) are accepted with a
    Metropolis probability that slowly cools down.
    """
    n = a_start.shape[0]
    cur = a_start.copy()
    cur_val = evaluate_sequence(cur.tolist())
    best = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    T0 = max(1e-4, 0.05 * cur_val)   # initial temperature
    Tmin = 1e-6

    while time.time() < deadline - 2.0:
        # linear cooling
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = (1 - frac) * T0 + frac * Tmin

        # propose a move
        if rng.random() < 0.7:   # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] <= 1e-15:
                continue
            frac_move = rng.random() * 0.8
            delta = cur[i] * frac_move
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        else:                     # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]

        # re‑project onto the simplex
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s < 1e-12:
            continue
        new /= s

        new_val = evaluate_sequence(new.tolist())
        delta_val = new_val - cur_val

        accept = False
        if delta_val < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta_val / T):
                accept = True

        if accept:
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
                print(f"[SA] new best = {best_val:.6f}")

        # occasional cheap re‑ordering attempt
        if rng.random() < 0.001:
            cand = _anti_symmetric_reorder(cur)
            val = evaluate_sequence(cand.tolist())
            if val < best_val:
                best_val = val
                best = cand.copy()
                cur = cand.copy()
                cur_val = val
                print(f"[SA reorder] improved to {best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (random swaps / mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random local search: swaps (20 %) and mass transfers (80 %).
    Keeps only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * (0.05 * rng.random())
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse a previous construction if it exists.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        a = old.copy()
    except Exception:
        n = 8000
        a = np.full(n, 1.0 / n)

    # ensure a is a probability distribution
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
        s = a.sum()
    a /= s

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – a few random Dirichlet starts to diversify the pool.
    # ------------------------------------------------------------------
    for tr in range(4):
        if time.time() >= deadline - 200:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate optimisation (few L‑BFGS restarts).
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 150 and restart < 5:
        restart += 1
        w_start = np.log(best_a + 1e-12)
        w_start += rng.normal(scale=0.2, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w_start, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[restart {restart}] new best after L‑BFGS = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multi‑peak exponential gradient descent (more aggressive).
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        a_gd, val_gd = _gradient_descent_multi(best_a, deadline,
                                               top_k=5,
                                               alpha0=0.8,
                                               decay_rate=0.97)
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[gradient descent] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing for a global escape.
    # ------------------------------------------------------------------
    if time.time() < deadline - 15:
        a_sa, val_sa = _simulated_annealing(best_a, deadline)
        if val_sa < best_val:
            best_val = val_sa
            best_a = a_sa.copy()
            print(f"[sim‑anneal] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
