def _annealing_search(a, deadline):
    cur_val = _objective(a)
    best = a.copy()
    best_val = cur_val
    T0 = 0.01
    iter = 0
    while time.time() < deadline:
        # schedule temperature
        T = T0 * math.exp(-iter * 0.0005) # adjust to drop after maybe 200k steps
        # choose donor and receiver
        donor = rng.integers(n)
        if a[donor] <= 1e-12:
            continue
        # choose amount delta = a[donor] * step_frac where step_frac random in [0.001,0.1]
        delta = a[donor] * (0.01 + 0.09 * rng.random())
        receiver = rng.integers(n)
        if donor == receiver:
            continue
        new_a = a.copy()
        new_a[donor] -= delta
        new_a[receiver] += delta
        # ensure non-neg
        # direct sum preserved, no need to renormalize
        val = _objective(new_a)
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / max(T, 1e-12)):
            a = new_a
            cur_val = val
            if val < best_val:
                best_val = val
                best = new_a.copy()
        iter += 1
    return best, best_val
```

def _project_simplex(v):
    # v is np.ndarray
    if v.ndim != 1:
        raise
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
propose_candidate – heuristic search for a step‑function with a small
evaluate_sequence value.

The algorithm consists of three stages:

1. **Parameter sweep** – a short grid search over a two‑parameter
   family a_i ∝ (i+1)^p (n-i)^p (with an optional mixture with the
   uniform distribution).  This quickly yields a decent starting point
   that is often already better than the plain sub‑gradient method.

2. **Projected Adam (first‑order) optimisation** – a stochastic Adam
   optimiser is run on the exact objective f(a)=2·n·max(conv(a,a)).
   After each step the vector is projected onto the probability simplex
   (non‑negative, sum = 1).  The optimiser is stopped when the time
   budget for this phase is exhausted.

3. **Directed mass‑transfer hill‑climbing** – repeatedly identify the
   pair of indices (i,j) that contributes most to the currently maximal
   convolution lag k_max and move a fraction of their mass to two
   destination positions where the product with the partner of k_max is
   smallest.  The move is kept only if the true objective improves.
   An adaptive step‑size (fraction of transferred mass) speeds up
   convergence.

The whole routine respects a user supplied time budget (default 1000 s)
and returns the best list of heights found before the deadline.  The
result can be passed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # tiny negative values from numerical noise are eliminated
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    w_i ≥ 0 and Σ w_i = 1.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry conv[k] w.r.t. a.
    For non‑negative a the derivative is 2·a_{k‑i} on the feasible range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad


# ----------------------------------------------------------------------
# Stage 1 – parametric families (quick grid search)
# ----------------------------------------------------------------------


def _make_parametric_sequence(n: int, p: float, mix: float) -> np.ndarray:
    """
    Build a candidate sequence of length n.
    The base shape is a_i ∝ (i+1)^p * (n-i)^p   (p may be negative).
    Afterwards a convex combination with the uniform distribution is taken
    using ``mix`` (0 → pure shape, 1 → pure uniform).
    The result is normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64)
    # avoid division by zero for very negative p
    if p == 0.0:
        shape = np.ones_like(idx)
    else:
        # (i+1)^p * (n-i)^p  is symmetric and well behaved for p<0
        shape = np.power(idx + 1.0, p) * np.power(np.maximum(n - idx, 1.0), p)
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.ones_like(idx)
        shape_sum = shape.sum()
    shape /= shape_sum

    if mix > 0.0:
        uniform = np.full_like(shape, 1.0 / n)
        seq = mix * uniform + (1.0 - mix) * shape
    else:
        seq = shape

    # final normalisation (protect against tiny rounding errors)
    seq = _project_simplex(seq)
    return seq


def _parameter_sweep(n: int, rng: np.random.Generator,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap grid search over (p, mix) to obtain a decent seed.
    The grid is small (≈150 points) and finishes well within a second.
    """
    best_seq = None
    best_val = math.inf

    # p range: from -0.9 to -0.1 (spacing 0.1) – negative exponents place
    # more mass near the edges.
    p_vals = np.linspace(-0.9, -0.1, 9)
    # mix range: convex combination with the uniform distribution.
    mix_vals = np.linspace(0.0, 0.5, 6)

    # In addition try a few random Dirichlet vectors for extra diversity.
    random_trials = 10

    for p in p_vals:
        for mix in mix_vals:
            if time.time() >= deadline:
                return best_seq, best_val
            a = _make_parametric_sequence(n, p, mix)
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a

    for _ in range(random_trials):
        if time.time() >= deadline:
            break
        a = rng.random(n)
        a = _project_simplex(a)
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stage 2 – projected Adam (sub‑gradient) optimisation
# ----------------------------------------------------------------------


def _adam_subgradient(a0: np.ndarray, deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimiser on f(a)=2·n·max(conv(a,a)).
    After each step the iterate is projected onto the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.6

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _objective(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # current convolution and its maximal lag
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag
        grad = _gradient_of_lag(a, k)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate decays like 1/√t (standard for sub‑gradients)
        lr = lr0 / math.sqrt(t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        val = _objective(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()

        a = a_new

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – directed mass‑transfer hill‑climbing
# ----------------------------------------------------------------------


def _directed_mass_transfer_hillclimb(a0: np.ndarray, deadline: float,
                                     rng: np.random.Generator,
                                     max_no_improve: int = 4000) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climbing that always attacks the most damaging
    lag.  At each step the pair (i,j) with the largest product
    a_i·a_j (among i+j = k_max) is selected as a donor.  Two
    destination indices are chosen where the partner values a_{k_max-p}
    are smallest, i.e. they are least likely to increase the maximal
    convolution.  A fraction of the donors' mass is transferred.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    # start with a modest transfer fraction, adapt based on success
    step_frac = 0.25
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that contribute to lag k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        # donor pair with the largest contribution
        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we are prepared to relocate
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac

        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-7)
            continue

        # ----- pick two promising destinations -----------------
        # For a destination p the new product added to lag k is
        # a[p] * a[k-p] (a[k-p] is zero outside the range).
        # We therefore minimise the partner weight a[k-p].
        partner = np.zeros(n, dtype=np.float64)
        valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
        partner[valid] = a[k - np.arange(n)[valid]]

        # Prohibit donors
        forbidden = {i, j}
        cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)

        # Choose the two candidates with smallest partner value
        if cand.size >= 2:
            partner_cand = partner[cand]
            smallest = np.argpartition(partner_cand, 2)[:2]
            dest = cand[smallest]
        else:
            # fallback – choose any two distinct positions
            dest = rng.choice(np.arange(n), size=2, replace=False)

        p, q = int(dest[0]), int(dest[1])

        # ----- perform the transfer ---------------------------
        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        new_a[p] += delta
        new_a[q] += delta

        # numerical safety (renormalise onto simplex)
        new_a = _project_simplex(new_a)

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.6)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[hillclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.90, 1e-7)
            no_improve += 1

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.

    The function follows the three‑stage strategy described in the module
    docstring.  It re‑uses a previously discovered construction if the
    global variable ``height_sequence_1`` exists; otherwise a fresh random
    start is used.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin

    # --------------------------------------------------------------
    # Initialise – reuse a previously found sequence if available.
    # --------------------------------------------------------------
    default_n = 4000
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = int(old.shape[0])
        a0 = np.maximum(old, 0.0)
        if a0.sum() == 0.0:
            a0 = rng.random(n)
        a0 = _project_simplex(a0)
        print(f"[init] loaded previous sequence (n={n})")
    except Exception:
        n = default_n
        a0 = rng.random(n)
        a0 = _project_simplex(a0)
        print(f"[init] random start (n={n})")

    # --------------------------------------------------------------
    # 1. Fast parametric sweep – gives a strong initial point.
    # --------------------------------------------------------------
    a_param, val_param = _parameter_sweep(n, rng, deadline)
    if a_param is not None and val_param < _objective(a0):
        a0 = a_param
        print(f"[sweep] best after sweep = {val_param:.6f}")

    # --------------------------------------------------------------
    # 2. Projected Adam optimisation.
    # --------------------------------------------------------------
    a_adam, val_adam = _adam_subgradient(a0, deadline, rng)
    print(f"[adam] best after Adam = {val_adam:.6f}")

    # --------------------------------------------------------------
    # 3. Directed mass‑transfer hill‑climbing (the most effective fine‑tuning).
    # --------------------------------------------------------------
    a_final, val_final = _directed_mass_transfer_hillclimb(a_adam, deadline, rng)
    print(f"[final] best value before evaluate_sequence = {val_final:.6f}")

    # --------------------------------------------------------------
    # 4. Report the score using the provided evaluator (optional).
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_final.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a_final.tolist()
