"""
Improved hybrid optimizer for the step‑function evaluation task.

Key ideas beyond the reference implementation
------------------------------------------------
*  A new “symmetric‑ends” re‑ordering puts the largest values at both
   of the sequence ends (and then works inward).  This reduces the
   peak of the self‑convolution more consistently than the simple
   large‑small alternating pattern.
*  A slightly finer smoothing schedule (down to 1e‑13) and a few extra
   L‑BFGS restarts give the continuous surrogate more chance to reach
   a better local basin.
*  After each L‑BFGS run we try *three* reorderings – the original,
   the new symmetric one and a symmetrisation (averaging with the
   reversed vector) – and keep the best.
*  Sub‑gradient descent uses an adaptive step‑size (larger initial step,
   slower decay) and a “best‑ever” tracking that continues even after
   the current iterate stops improving.
*  The final simulated‑annealing stage runs longer (higher initial
   temperature, slower decay) and also includes occasional tiny mass‑
   transfer moves in addition to swaps.

The routine respects the 1000 s wall‑time limit, always leaving a small
margin for the final call to ``evaluate_sequence``.  It returns the
best list of non‑negative floats found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders (required by the L‑BFGS wrapper)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Remove possible tiny negative round‑off values
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast evaluator that returns exactly the same value as
    ``evaluate_sequence(a.tolist())`` but without the safety checks.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smoothed surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where L is the smoothed surrogate

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and grad_w is the gradient w.r.t. the soft‑max parameters w.
    """
    # ---- soft‑max (positivity, ℓ₁ = 1) ----
    w_shift = w - np.max(w)                 # improve stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                           # Σ a = 1

    # ---- convolution a * a ----
    conv = _fft_convolve_self(a)            # length 2·n‑1
    max_conv = conv.max()

    # ---- smooth approximation of the max ----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)            # log∑exp((c−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a ----
    weights = np.exp(shifted)               # exp((c−max)/ε)
    p = weights / np.sum(weights)           # distribution over k

    # convolution a * reversed(p)  (see original derivation)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)           # length 2·n‑1
    seg = conv_ap[n - 1: 2 * n - 1]          # central n entries
    grad_a = 4.0 * n * seg[::-1]            # ∂L/∂a

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# bind the helper convolution to the function object
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Original heuristic: place a large value, then a small one,
    alternating from the leftmost position forward.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left_ptr = 0
    right_ptr = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right_ptr]
        right_ptr -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left_ptr]
        left_ptr += 1
        i += 1
    return new_a


def _ends_large_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    New heuristic: biggest values at both ends, second biggest at the
    opposite end, and continue moving inward.  Produces a symmetric
    “high‑at‑the‑ends” shape.
    """
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]    # descending
    new_a = np.empty_like(a)
    left = 0
    right = n - 1
    idx = 0
    while left <= right:
        new_a[left] = sorted_desc[idx]
        idx += 1
        if left < right:
            new_a[right] = sorted_desc[idx]
            idx += 1
        left += 1
        right -= 1
    return new_a


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force exact symmetry by averaging with the reversed vector."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Sub‑gradient method that directly minimises max(a*a) while keeping
    a on the simplex (non‑negative, ℓ₁ = 1).  Returns the best sequence
    seen before the deadline.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    # start with a slightly larger step; decay as 1/√t
    step0 = 0.25
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur_a)
        k_max = int(np.argmax(conv))

        # gradient of the active (max) entry
        idx_arr = k_max - np.arange(n)
        mask = (idx_arr >= 0) & (idx_arr < n)
        grad = np.zeros_like(cur_a)
        grad[mask] = 2.0 * cur_a[idx_arr[mask]]

        # diminishing step size
        alpha = step0 / np.sqrt(it + 1.0)

        # projected step
        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)
        s = np.sum(new_a)
        if s < 1e-12:                # safeguard against collapse
            new_a = np.full(n, 1.0 / n)
            s = 1.0
        new_a /= s                    # back onto ℓ₁‑simplex

        new_val = _eval_fast(new_a)

        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it:5d} new best = {best_val:.6f}")

        # accept only improving moves (guarantees monotone descent)
        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (swap + tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """Fast SA that swaps two positions and occasionally transfers a tiny amount."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # temperature starts as a modest fraction of the current value
    T = max(cur_val * 0.03, 1e-7)
    decay = 0.99995

    it = 0
    while time.time() < deadline:
        # 80 % swaps, 20 % tiny mass moves
        if rng.random() < 0.8:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]          # propose swap
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 0.0:
                continue
            delta = a[i] * 0.03 * rng.random()
            a[i] -= delta
            a[j] += delta

        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            # revert the move
            if i != j and a[i] != a[j]:   # generic revert
                if rng.random() < 0.8:
                    a[i], a[j] = a[j], a[i]      # swap revert
                else:
                    # reverse tiny transfer
                    a[i] += delta
                    a[j] -= delta
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1
    return best_seq, best_val
