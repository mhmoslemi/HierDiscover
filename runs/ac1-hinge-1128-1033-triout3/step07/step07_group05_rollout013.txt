"""
Propose a step‑function that minimises ``evaluate_sequence``.

The search proceeds in four stages:

1. **Smooth surrogate optimisation (L‑BFGS)** – we minimise a
   log‑sum‑exp approximation of the ℓ∞‑norm of the self‑convolution,
   keeping the ℓ₁‑norm fixed by a soft‑max parameterisation.
   Multiple restarts with a fine ε‑schedule are performed.

2. **Aggressive sub‑gradient descent** – instead of using only the
   single most‑violating shift we take the average sub‑gradient over
   all shifts whose convolution value is within 0.1 % of the current
   maximum.  This yields a more balanced descent direction.

3. **Permutation simulated‑annealing** – a fast SA that swaps and
   block‑swaps positions, allowing occasional uphill moves.

4. **Final hill‑climb** – cheap mass‑transfer + swap moves to polish
   the best solution found.

The code respects a total wall‑time of ``budget_s`` seconds and returns
a plain ``list[float]`` that can be handed to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders required by the smooth surrogate
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (needed for FFT‑based convolutions)."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``a`` with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip rounding artefacts
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` (avoids safety checks)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    s = np.sum(a)
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth surrogate objective (for L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where L is the smoothed surrogate

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and grad_w is the gradient w.r.t. the soft‑max parameters w.
    """
    # soft‑max representation (positivity + ℓ₁ = 1)
    w_shift = w - np.max(w)          # numeric stability
    a = np.exp(w_shift)
    a /= np.sum(a)                    # Σ a = 1

    # convolution a*a
    conv = _fft_convolve_self(a)      # length 2·n‑1
    max_conv = np.max(conv)

    # smoothed max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)                     # exp((c−max)/ε)
    p = weights / np.sum(weights)                 # distribution over k

    # g_i = 4·n·∑_k p_k·a_{k−i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                 # length 2·n‑1
    seg = conv_ap[n - 1: 2 * n - 1]                # central n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain‑rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Helper convolution (used inside the gradient of the smooth surrogate)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# Bind the helper convolution to the gradient function object (no closure)
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristic – interleave big / small values
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic interleaving: largest, smallest, second‑largest,
    second‑smallest, …
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left_ptr = 0
    right_ptr = n - 1
    i = 0
    while i < n:
        # place a large value
        new_a[i] = sorted_vals[right_ptr]
        right_ptr -= 1
        i += 1
        if i >= n:
            break
        # place a small value
        new_a[i] = sorted_vals[left_ptr]
        left_ptr += 1
        i += 1
    return new_a

# Simulated annealing on permutations (swap + block‑swap moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float, deadline: float):
    """Fast SA that mixes swaps and block‑swaps."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # initial temperature ≈ 2 % of current objective
    T = max(cur_val * 0.02, 1e-8)
    decay = 0.99996

    it = 0
    while time.time() < deadline:
        if rng.random() < 0.3:        # block‑swap move
            # choose block length 2‑10
            length = rng.integers(2, 11)
            i = rng.integers(0, n - length + 1)
            j = rng.integers(0, n - length + 1)
            if i == j:
                continue
            # swap the two blocks
            tmp = a[i:i + length].copy()
            a[i:i + length] = a[j:j + length]
            a[j:j + length] = tmp
        else:                         # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]

        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            # revert move
            if rng.random() < 0.3:
                # revert block‑swap
                tmp = a[i:i + length].copy()
                a[i:i + length] = a[j:j + length]
                a[j:j + length] = tmp
            else:
                a[i], a[j] = a[j], a[i]

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float, deadline: float):
    """Cheap local search – moves mass and swaps positions."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.25:          # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                           # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Aggressive sub‑gradient descent (averaged over near‑max shifts)
# ----------------------------------------------------------------------
def _subgradient_multi(a: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent where the descent direction is the
    average of gradients of all shifts whose convolution value lies
    within 0.1 % of the current maximum.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    step0 = 0.25                      # initial step size (aggressive)
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur_a)
        max_b = np.max(conv)
        # indices whose convolution is close to the peak
        tol = max_b * 1e-3
        near_max = np.where(conv >= max_b - tol)[0]

        # compute averaged sub‑gradient
        grad = np.zeros_like(cur_a)
        for k in near_max:
            # indices i that contribute to shift k
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            i_range = np.arange(start, end + 1)
            j_range = k - i_range               # = a_{k-i}
            grad[i_range] += cur_a[j_range]

        grad *= 2.0                               # ∂(a*a)_k / ∂a_i = 2·a_{k-i}
        grad_norm = np.linalg.norm(grad) + 1e-12
        grad /= grad_norm                         # normalise to keep steps stable

        alpha = step0 / np.sqrt(it + 1.0)

        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)
        s = np.sum(new_a)
        if s < 1e-12:
            new_a = np.full(n, 1.0 / n)
            s = 1.0
        new_a /= s                               # project onto ℓ₁‑simplex

        new_val = _eval_fast(new_a)

        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad‑multi] iter {it} new best = {best_val:.6f}")

        # accept only improving steps (ensures monotonic descent)
        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val

        it += 1
    return best_a, best_val

            old_x = np.linspace(0, 1, a0.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, a0)
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= np.sum(a0)               # enforce ℓ₁ = 1

    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Smooth surrogate L‑BFGS (multiple restarts, fine ε‑schedule)
    # --------------------------------------------------------------
    # allocate ~45 % of the total budget to this phase
    phase1_end = start_time + 0.45 * budget_s
    eps_schedule = np.logspace(-2, -12, num=14)  # more points than before
    restart = 0

    while time.time() < phase1_end - 30.0:      # keep ~30 s buffer for later phases
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        # initialise soft‑max parameters from current best_a (+‑small noise)
        w = np.log(best_a + 1e-12) + 0.5 * np.random.randn(n)

        for eps in eps_schedule:
            if time.time() >= phase1_end - 30.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 600,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate the current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= np.sum(a)
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ----------------------------------------------------------
        # Re‑ordering heuristic (interleaving) after each restart
        # ----------------------------------------------------------
        a_ord = _ends_large_reorder(best_a)
        a_ord /= np.sum(a_ord)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

        # ----------------------------------------------------------
        # Extra random permutation trials (cheap swaps) – 2 sec per restart
        # ----------------------------------------------------------
        perm_deadline = time.time() + 2.0
        a_tmp = best_a.copy()
        cur_tmp = best_val
        while time.time() < perm_deadline:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            a_tmp[i], a_tmp[j] = a_tmp[j], a_tmp[i]
            val = _eval_fast(a_tmp)
            if val < cur_tmp:
                cur_tmp = val
                if cur_tmp < best_val:
                    best_val = cur_tmp
                    best_a = a_tmp.copy()
                    print(f"[perm] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Aggressive projected sub‑gradient descent (averaged over near‑max)
    # --------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print("[subgrad‑multi] starting aggressive sub‑gradient descent")
        sub_deadline = deadline - 45.0
        best_a, best_val = _subgradient_multi(best_a, sub_deadline)

    # --------------------------------------------------------------
    # 3. Simulated annealing on permutations (swap + block‑swap)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[SA] final simulated annealing on permutations")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 15.0)

    # --------------------------------------------------------------
    # 4. Final cheap hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report final value using the official evaluator (includes safety checks)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required by the evaluation framework
    return best_a.tolist()
