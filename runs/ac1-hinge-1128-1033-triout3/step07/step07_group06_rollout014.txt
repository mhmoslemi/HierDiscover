   grad = gradient_of_lag(a, k)
   # gradient of objective = 2 * n * grad (since objective = 2*n * maxb)
   # we can just use grad for moving along maxb direction
   m = beta1 * m + (1-beta1) * grad

    cur_val = objective(cur_a)
    best_a = cur_a.copy()

        T = max(Tf, T0 * (Tf/T0)**fraction)  # geometric cooling

        new_a[p] += delta

"""
Implementation of an iterative optimizer for the step‑function
attack.  The algorithm consists of three phases:

1. Adam‑based projected sub‑gradient descent on the exact
   objective `2·n·max(conv(a,a))`.
2. Aggressive mass‑transfer hill climbing that moves a fraction of the
   mass from the pair contributing most to the current maximal lag
   towards two low‑mass locations.
3. Simulated annealing (single‑donor / single‑recipient moves) that
   explores the neighbourhood of the best solution found so far,
   accepting occasional uphill moves according to an exponential
   schedule.

Multiple random restarts are performed until the time budget is
exhausted; the best sequence encountered is returned.
All numbers are kept non‑negative and are normalised to sum = 1,
so the returned list can be passed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using an FFT (real input)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # Numerical noise may produce tiny negative numbers; clip them.
    np.maximum(conv, 0, out=conv)
    return conv

def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))

def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry conv[k] w.r.t. a.
    For non‑negative a the derivative is 2·a_{k−i} on the feasible range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    # derivative of a_i * a_{k-i} with respect to a_i is a_{k-i}
    grad[i_range] = 2.0 * a[j_range]
    return grad

def _adam_subgradient_descent(a0: np.ndarray,
                              deadline: float,
                              rng: np.random.Generator,
                              max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimizer on f(a)=2·n·max(conv(a,a)).
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _objective(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))
        grad = _gradient_of_lag(a, k)  # derivative of maxb
        # Scale step size by iteration count to guarantee convergence
        lr = lr0 / np.sqrt(t)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_trial = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Projection onto the probability simplex (non‑negative + sum = 1)
        a_trial = np.clip(a_trial, 0.0, None)
        s = a_trial.sum()
        if s > 0.0:
            a_trial /= s
        else:
            # fallback to a uniform vector if everything collapsed
            a_trial = np.full_like(a, 1.0 / n)

        val = _objective(a_trial)
        if val < best_val:
            best_val = val
            best_a = a_trial.copy()
        a = a_trial

        if t % 500 == 0:
            print(f"[adam] iter {t:5d}  best = {best_val:.6f}")

    return best_a, best_val

def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climbing that repeatedly relocates a fraction of the
    mass from the pair contributing most to the current maximal lag.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount to relocate
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac

        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # Choose two low‑mass destinations
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        # Renormalise to keep sum = 1
        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_moves: int = 20000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing fine tuner.
    Single‑donor / single‑recipient moves are proposed; uphill moves
    are accepted with probability exp(−Δ/T).
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    # Temperature schedule (geometric cooling)
    T_start = 0.05
    T_end = 1e-5

    # Pre‑compute log of cooling factor
    if max_moves > 0:
        log_factor = np.log(T_end / T_start) / max_moves
    else:
        log_factor = 0.0

    for move in range(1, max_moves + 1):
        if time.time() >= deadline:
            break

        T = T_start * np.exp(log_factor * move)

        # propose move: transfer a random fraction (up to 20 %) of mass
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        p = rng.integers(n)
        if i == p:
            continue
        # delta = fraction * a[i] , fraction drawn uniformly from (0, step_max)
        step_max = 0.2
        delta = a[i] * rng.random() * step_max
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[p] += delta

        # ensure non‑negative (should hold)
        if new_a[i] < -1e-15:
            continue

        new_val = _objective(new_a)
        delta_E = new_val - cur_val
        if delta_E < 0 or rng.random() < np.exp(-delta_E / (T + 1e-12)):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SA] move {move:5d} new best = {best_val:.6f}")

    return best_a, best_val

        if time.time() >= deadline:
            break

        T = T_start * np.exp(log_factor * move)

        # Choose a donor with non‑trivial mass
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        p = rng.integers(n)
        if i == p:
            continue

        # Transfer up to 20 % of the donor's mass
        step_max = 0.2
        delta = a[i] * rng.random() * step_max
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[p] += delta

        # Safety check for negativity (should never happen)
        if new_a[i] < -1e-15:
            continue

        new_val = _objective(new_a)
        dE = new_val - cur_val
        if dE < 0 or rng.random() < np.exp(-dE / (T + 1e-12)):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SA] move {move:5d} new best = {best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    ``evaluate_sequence`` score.  The function respects the supplied time
    budget, performs several restarts, and returns the best sequence it
    discovers (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Initialise – reuse a previous construction if it exists.
    # --------------------------------------------------------------
    default_n = 4000
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = rng.random(n)
        a0 /= a0.sum()
    except Exception:
        n = default_n
        a0 = rng.random(n)
        a0 /= a0.sum()

    print(f"[init] n={n}, initial objective = {_objective(a0):.6f}")

    best_overall_a = a0.copy()
    best_overall_val = _objective(a0)

    restart = 0
    while time.time() < deadline:
        # allocate a slice of the remaining time to this restart
        remaining = deadline - time.time()
        restart_budget = max(30.0, remaining * 0.3)
        sub_deadline = time.time() + restart_budget

        # occasional reuse of the current best with a small jitter
        if rng.random() < 0.2 and best_overall_val < np.inf:
            a_start = best_overall_a.copy()
            jitter = rng.normal(scale=0.01, size=n)
            a_start = np.maximum(a_start + jitter, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = rng.random(n)
            a_start /= a_start.sum()

        # --------- 1. Adam sub‑gradient descent ----------
        a_sub, _ = _adam_subgradient_descent(a_start, sub_deadline, rng,
                                             max_iters=3000)

        # --------- 2. Aggressive mass‑transfer hill‑climb ----------
        a_hc, _ = _mass_transfer_hillclimb(a_sub, sub_deadline, rng,
                                          max_no_improve=2000)

        # --------- 3. Simulated annealing fine‑tuner ----------
        a_fin, val_fin = _simulated_annealing(a_hc, sub_deadline, rng,
                                              max_moves=15000)

        if val_fin < best_overall_val:
            best_overall_val = val_fin
            best_overall_a = a_fin.copy()
            print(f"[global] restart {restart} new best overall = {best_overall_val:.6f}")

        restart += 1

    # --------------------------------------------------------------
    # Final evaluation using the provided evaluator.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_overall_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_overall_a.tolist()
