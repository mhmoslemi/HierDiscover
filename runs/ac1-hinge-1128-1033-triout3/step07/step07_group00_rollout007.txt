Projection algorithm:

def simplex_projection(v):
    # ensure sum(v) = 1 and v >= 0
    # Implementation based on sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n_small = 1500
    # generate candidate shapes
    best_a = np.full(n_small, 1.0/n_small)
    best_val = _objective_value(best_a)

    # function to test a candidate shape list
    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n_small:
            return
        # ensure nonneg
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a /= s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # families enumeration loops:
    # 1. power law
    for p in np.linspace(0.1, 5.0, 20):
        a = (np.arange(1, n_small+1, dtype=float) ** (-p))
        try_candidate(a)

        a_sym = a + a[::-1]  # symmetric U-shape
        try_candidate(a_sym)

    # 2. exponential
    for r in np.linspace(0.94, 0.9999, 30):
        a = r ** np.arange(n_small, dtype=float)
        try_candidate(a)

        a_sym = r ** np.arange(n_small, dtype=float) + r ** np.arange(n_small-1, -1, -1)
        try_candidate(a_sym)

    # 3. Beta distributions
    for a_param in np.linspace(0.2, 5.0, 15):
        for b_param in np.linspace(0.2, 5.0, 15):
            x = (np.arange(n_small)+0.5)/n_small
            a = (x**(a_param-1) * (1-x)**(b_param-1))
            try_candidate(a)

    # 4. Random Dirichlet
    rng = np.random.default_rng(seed+1)
    for _ in range(200):
        a = rng.dirichlet(np.ones(n_small))
        try_candidate(a)

    # 5. Linear families
    a = np.arange(n_small, 0, -1, dtype=float)
    try_candidate(a)
    a = np.maximum(np.arange(n_small)+1, n_small-(np.arange(n_small)))  # U shape
    try_candidate(a)

    # upscaling to large n
    n_large = 9000
    # we may upscale best_a by interpolation
    x_small = np.linspace(0,1, n_small)
    x_large = np.linspace(0,1, n_large)
    a_large = np.interp(x_large, x_small, best_a)
    a_large = np.maximum(a_large, 0)
    a_large /= a_large.sum()
    # Optionally refine with hillclimb for few seconds
    cur_val = _objective_value(a_large)
    best_val_large = cur_val
    best_a_large = a_large.copy()
    # hillclimb refined for limited time
    # define function for hillclimb iteration using incremental update
    def hillclimb(a, elapsed_limit):
        a = a.copy()
        n = a.shape[0]
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        best_val_local = cur_val
        best_a_local = a.copy()
        max_frac_start, max_frac_end = 0.5, 0.001
        # we will use while time left
        while time.time() < elapsed_limit:
            # schedule fraction based on time
            progress = (time.time() - start) / (deadline - start)
            max_frac = max_frac_start * (1 - progress) + max_frac_end * progress
            k_max = int(np.argmax(b))
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[idx[mask]]
            total_contrib = contrib.sum()
            if total_contrib > 0:
                probs = contrib / total_contrib
                i = np.random.choice(n, p=probs)
            else:
                i = np.argmax(a)

            if a[i] <= 1e-15:
                continue

            j = np.random.randint(n-1)
            if j >= i:
                j += 1

            delta = a[i] * np.random.rand() * max_frac
            if delta <= 0:
                continue

            # update b via incremental formula O(n)
            b_new = b.copy()
            b_new[j:j+n] += 2.0 * delta * a
            b_new[i:i+n] -= 2.0 * delta * a
            # handle diagonals
            if j*2 < len(b_new):
                b_new[2*j] += delta*delta
            if i*2 < len(b_new):
                b_new[2*i] += delta*delta
            if i + j < len(b_new):
                b_new[i+j] -= 2.0 * delta * delta

            new_val = 2.0 * n * b_new.max()
            if new_val < cur_val - 1e-15:
                a[i] -= delta
                a[j] += delta
                b = b_new
                cur_val = new_val
                if new_val < best_val_local:
                    best_val_local = new_val
                    best_a_local = a.copy()
        return best_a_local, best_val_local

    # run hillclimb until deadline
    best_a_large, best_val_large = hillclimb(a_large, deadline)
    # final polishing: mix with uniform
    best_a_large, best_val_large = _mix_with_uniform(best_a_large, n_large)

    # final evaluation
    try:
        final_score = evaluate_sequence(best_a_large.tolist())
        print("[final] score =", final_score)
    except Exception as e:
        print("eval error:", e)

    return best_a_large.tolist()

"""
Propose a step‑function minimising the evaluation function.

Algorithm
---------
1. **Exploratory parametric sweep (small n)**  
   Generate a large collection of candidate shapes on a modest size
   (`n_small ≈ 1500`).  Families include power‑laws, exponentials,
   symmetric “U” shapes, and Beta‑distributions (α,β).  Each candidate
   is normalised to Σa = 1 and its exact objective
   `2·n·max(conv(a,a))` is evaluated via an FFT‑based convolution.
   The best candidate is retained.

2. **Upscaling**  
   The winning small‑scale vector is interpolated to the target length
   (`n_large = 9000`) and re‑normalised.

3. **Local refinement (hill‑climbing)**  
   A fast, exact hill‑climber moves a tiny fraction of mass from indices
   that contribute most to the current maximal convolution entry to
   indices that contribute little.  The maximal transferred fraction
   anneals from 0.5 → 0.001.  The process runs until the time budget is
   exhausted.

4. **Final polishing** – convex combination with the uniform vector
   (very small mixing coefficients) may lower the maximum a little
   further.

All steps respect the 1000 s budget and keep the sequence non‑negative,
Σa = 1, which makes the objective scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two for FFT speed
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    # implementation of the algorithm by Wang & Carreira‑Perpiñán (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small post‑processing: mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a' = (1‑α)a + α·uniform for a few α."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing on the exact objective (fast incremental update)
# ----------------------------------------------------------------------
def _hillclimb(a_start: np.ndarray,
               n: int,
               start_time: float,
               deadline: float,
               best_val: float) -> tuple[np.ndarray, float]:
    """
    Local search moving a small amount of mass from a source index that is
    heavily involved in the current maximal convolution entry to a target
    index that contributes little.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val_local = cur_val

    # mass‑transfer schedule – from 0.5 → 0.001
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear annealing of the transferred fraction
        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – bias towards large contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – usually a location with little reversed contribution
        if np.random.rand() < 0.35:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                # fallback: uniform draw (excluding i)
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            # simple uniform target
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move (fraction of source)
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # diagonal corrections
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < b_new.size:
            b_new[two_j] += delta * delta
        if two_i < b_new.size:
            b_new[two_i] += delta * delta
        if ij < b_new.size:
            b_new[ij] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:          # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val

            if new_val < best_val_local:
                best_val_local = new_val
                best_a = a.copy()
        # occasional random perturbation to escape plateaus
        if np.random.rand() < 0.001:
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            if cur_val < best_val_local:
                best_val_local = cur_val
                best_a = a.copy()

    return best_a, best_val_local
