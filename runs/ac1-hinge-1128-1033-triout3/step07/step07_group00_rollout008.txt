def hillclimb_deterministic(a_start, n, start, deadline, best_a, best_val, target_val=1.503):
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    # schedule for max_frac: start at 0.8 maybe.
    max_frac_start = 0.8
    max_frac_end = 0.001
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 300_000
    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1 - progress) + max_frac_end * progress

        # find max conv index
        k_max = int(np.argmax(b))

        # compute contributions
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        # source index i: maximum contrib
        i = int(np.argmax(contrib))

        # compute rev contr for target: a_rev = a[idx], i.e., same as contributions but unsquared?
        rev_contrib = np.zeros_like(a)
        rev_contrib[mask] = a[idx[mask]]
        # choose target index j with minimal rev_contrib (i.e., small a[k_max - j])
        # Exclude i
        rev_contrib[i] = np.inf  # avoid same index
        j = int(np.argmin(rev_contrib))

        if a[i] <= 1e-16:
            # fallback to random
            i = np.random.choice(n, p=a/np.sum(a))
            if a[i] <= 1e-16:
                continue

        # compute feasible delta: at most a[i]*max_frac
        delta_max = a[i] * max_frac
        if delta_max < 1e-12:
            continue

        # try delta as delta_max; if not improve reduce factor
        delta = delta_max
        # compute new b after move
        for _ in range(6):  # up to 6 reduction attempts
            # compute b_new incrementally
            b_new = b.copy()
            b_new[i:i+n] -= 2.0 * delta * a
            b_new[j:j+n] += 2.0 * delta * a
            # extra terms
            b_new[2*i] += delta * delta
            b_new[2*j] += delta * delta
            b_new[i + j] -= 2.0 * delta * delta

            new_val = 2.0 * n * b_new.max()
            if new_val < cur_val - 1e-15:
                # accept
                a[i] -= delta
                a[j] += delta
                b = b_new
                cur_val = new_val
                no_improve = 0
                if new_val < best_val:
                    best_val = new_val
                    best_a = a.copy()
                    print(f"[HillDet {it:7d}] new best = {best_val:.6f}")
                break

        if no_improve > IMPROVE_LIMIT:

"""
Propose a non‑negative step‑function that (approximately) minimises
the bound

    2·n·max(conv(a,a)) / (sum(a)**2)

used by `evaluate_sequence`.  The algorithm consists of three stages:

1. **Diverse initializer** – a quick sweep over several parametric families
   (uniform, linear, exponential, sine, power‑law, random Dirichlet) and
   keeps the best candidate (measured by the exact objective).

2. **Smooth‑max Adam** – the max‑convolution is replaced by a log‑sum‑exp
   (smooth max) and a Adam optimiser works on the log‑softmax
   parametrisation.  This quickly finds a low‑energy region of the
   feasible simplex.

3. **Deterministic hill‑climber** – repeatedly moves a small amount of
   mass from the index that contributes most to the current maximal
   convolution entry to the index that contributes least.
   The step size decays from 0.8 to 1e‑3, and a line‑search guarantees
   strict improvement.  If progress stalls a tiny random perturbation
   (Gaussian shake) is applied.

A short “mix‑with‑uniform’’ sweep is performed at the end.
All phases respect the overall time budget (default 1000 s) and the
returned vector is normalised to unit sum, non‑negative and therefore
valid for `evaluate_sequence`.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities – real FFT based O(n log n) full linear convolution
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) is assumed to be 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric initialisation – a rich set of low‑dimensional families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a good starting vector (normalised) and its objective value."""
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_val = _objective_value(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ----- deterministic families -------------------------------------------------
    # linear decreasing
    try_candidate(np.arange(n, 0, -1, dtype=float))

    # linear U‑shape
    idx = np.arange(n)
    try_candidate(np.maximum(idx + 1, n - idx))

    # quadratic decreasing
    try_candidate((np.arange(n, 0, -1, dtype=float) ** 2))

    # quadratic U‑shape (inverse square on both ends)
    i = np.arange(n, dtype=float) + 1.0
    try_candidate(i ** -2 + (n - i + 1) ** -2)

    # half‑wave sine (smooth bump)
    try_candidate(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))

    # cosine with peaks at the ends
    try_candidate(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # power‑law symmetric U‑shape
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        try_candidate(i ** (-p) + (n - i + 1) ** (-p))

    # geometric‑decay (left‑hand side)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        try_candidate(r ** np.arange(n, dtype=float))

    # symmetric geometric (U‑shape)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        try_candidate(r ** i + r ** (n - 1 - i))

    # truncated geometric on a prefix
    for frac in (0.10, 0.15, 0.20, 0.25, 0.30):
        K = max(1, int(frac * n))
        for r in (0.99, 0.98, 0.95, 0.90):
            a = np.zeros(n, dtype=float)
            a[:K] = r ** np.arange(K, dtype=float)
            try_candidate(a)

    # random Dirichlet draws (a few quick samples)
    rng = np.random.default_rng(12345)
    for _ in range(20):
        try_candidate(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation (log‑softmax on the simplex)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of max(conv(a,a)).
    Uses log‑softmax re‑parametrisation to keep a on the simplex.
    """
    # initialise logits (avoid log(0))
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.003          # a bit more aggressive than before
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000                 # patience

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- reconstruct a from logits (stable softmax) -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a_sum = a.sum()
        a /= a_sum

        # ----- exact objective for bookkeeping -----
        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- schedule for smoothing and learning‑rate -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max (log‑sum‑exp) -----
        b_full = _full_convolution(a)          # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()            # soft‑max over convolution entries

        # ----- gradient of smooth objective w.r.t. a -----
        # ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]     # the part that corresponds to ∂/∂a
        grad_a = 4.0 * n * segment[::-1]

        # ----- chain rule through log‑softmax -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to avoid exact plateaus
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.006, size=n)

        # restart if stuck for a long while
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – deterministic hill‑climber on the exact objective
# ----------------------------------------------------------------------
def _deterministic_hillclimb(a_start: np.ndarray,
                             n: int,
                             start: float,
                             deadline: float,
                             best_a: np.ndarray,
                             best_val: float,
                             target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Move mass from the index that contributes most to the current maximal
    convolution entry to the index that contributes least.
    The step size (fraction of source mass) decays from 0.8 → 0.001.
    If no progress is observed for a while a small Gaussian shake is applied.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # schedule for the maximal transferable fraction
    max_frac_start, max_frac_end = 0.80, 0.001
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # fraction that shrinks linearly with time
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # ----- locate the currently maximal convolution entry -----
        k_max = int(np.argmax(b))

        # ----- compute contribution of each index to that entry -----
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]

        # source: index with largest contribution
        i = int(np.argmax(contrib))

        if a[i] <= 1e-15:          # safety – choose a random non‑zero source if needed
            nonzero = np.where(a > 1e-15)[0]
            if nonzero.size == 0:
                break
            i = int(np.random.choice(nonzero))

        # ----- target selection – want small a[k_max - j] (little impact on max) -----
        rev_contrib = np.full(n, np.inf, dtype=np.float64)  # default = far away -> no impact
        rev_contrib[mask] = a[idx[mask]]                     # a in the reflected position
        rev_contrib[i] = np.inf                               # avoid picking the source itself
        j = int(np.argmin(rev_contrib))

        # if for some reason j == i (should not happen), fall back to a random other index
        if j == i:
            cand = np.arange(n)
            cand = cand[cand != i]
            j = int(np.random.choice(cand))

        # ----- decide how much mass to move (line‑search for improvement) -----
        delta = a[i] * max_frac
        if delta <= 0.0:
            no_improve += 1
            continue

        # try decreasing delta by a factor of two until we get a strict improvement
        for _ in range(8):
            # incremental convolution update (O(n))
            b_new = b.copy()
            b_new[i:i + n] -= 2.0 * delta * a
            b_new[j:j + n] += 2.0 * delta * a
            b_new[2 * i] += delta * delta
            b_new[2 * j] += delta * delta
            b_new[i + j] -= 2.0 * delta * delta

            new_val = 2.0 * n * b_new.max()
            if new_val < cur_val - 1e-15:          # strict improvement required
                a[i] -= delta
                a[j] += delta
                b = b_new
                cur_val = new_val
                no_improve = 0
                if new_val < best_val:
                    best_val, best_a = new_val, a.copy()
                    print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                    if best_val <= target_val:
                        return best_a, best_val
                break
            else:
                delta *= 0.5          # halve and try again
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            # small Gaussian perturbation around the best seen so far
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturb after stagnation")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – short uniform mixing sweep
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few convex combinations with the uniform vector."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
