def init_sequence(n, method, rng):
    if method == 'random':
        a = rng.random(n)
        a /= a.sum()
    elif method == 'triangular':
        x = np.arange(n)
        a = (x+1)*(n-x)  # shape symmetric bell
        a = a.astype(float)
        a = np.maximum(a, 0)
        a /= a.sum()
    elif method == 'sine':
        x = np.arange(n)
        a = np.sin(np.pi * (x + 0.5) / n)
        a = np.maximum(a, 0)
        a /= a.sum()
    else:
        a = rng.random(n)
        a /= a.sum()
    return a

"""
Multi‑restart stochastic optimizer for the step‑function attack.

The evaluator
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

is homogeneous → we may restrict the search to probability vectors
(sum = 1).  The algorithm proceeds as follows:

1.  Generate many start vectors (random Dirichlet, triangular, sinusoid,
    and optionally the previous best if available).  Lengths around 4000
    are used – they give a good trade‑off between search cost and
    attack surface.

2.  On each start run a *projected sub‑gradient descent* on
    f(a)=2·n·max(conv(a,a)).  The sub‑gradient is taken from a lag that
    attains the current maximum, a step size  ~ step/√t  is used and the
    iterate is projected onto the probability simplex.

3.  Refine the result with an *aggressive mass‑transfer hill‑climb*.
    The lag with maximal convolution is located; the pair of indices
    that contributes most to it is identified and a fraction of their
    mass is moved to two low‑mass positions.  The move is kept only if
    it improves the objective.  The transferred‑mass fraction is
    adapted dynamically.

4.  The whole process is repeated while the global time budget (default
    1000 s) allows.  The best sequence found is returned.

The search explores many restarts, uses a few simple parametric
initialisations and a final refinement of the global best.  All numbers
are non‑negative, the sum is normalised to 1, and the final list is
returned as plain Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (size set to a power of two)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # eliminate tiny negative artefacts from floating point errors
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of the objective for a probability vector (sum(a)=1)."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry conv[k] with respect to a.
    For non‑negative a the derivative is 2·a_{k−i} on the active indices.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 10000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate is always kept on the simplex (non‑negative, sum = 1).
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    step0 = 0.5          # initial multiplier for the step size
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        grad = _gradient_of_lag(a, k)

        lr = step0 / np.sqrt(t)   # simple diminishing step
        improved = False
        for _ in range(10):       # back‑tracking line search
            a_trial = a - lr * grad
            a_trial = np.clip(a_trial, 0.0, None)
            s = a_trial.sum()
            if s == 0.0:           # avoid division by zero
                lr *= 0.5
                continue
            a_trial /= s            # projection onto the simplex
            val = _objective(a_trial)
            if val < best_val:
                a = a_trial
                best_a = a_trial.copy()
                best_val = val
                improved = True
                # slowly increase the base step after a success
                step0 = min(step0 * 1.02, 2.0)
                break
            else:
                lr *= 0.5
        if not improved:
            # gently shrink the step size when the line search fails
            step0 = max(step0 * 0.95, 1e-4)

    return best_a, best_val


def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climbing that moves a fraction of the most damaging
    mass to two low‑mass positions.  The algorithm stops after the time
    budget expires or after ``max_no_improve`` consecutive failed attempts.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2            # initial fraction of mass to relocate
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that make up the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we are willing to shift
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # pick two low‑mass destinations, avoiding the donors
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue
        new_a /= s   # keep the vector on the simplex

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new global best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _init_sequence(n: int, method: str, rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial probability vector of length ``n``.
    ``method`` can be:
        - 'random'      : Dirichlet (uniform) draw
        - 'triangular'  : (i+1)*(n-i) symmetric bell
        - 'sine'        : sin‑shaped half‑wave
        - 'prev'        : a previously known good sequence (handled outside)
    """
    if method == 'random':
        a = rng.random(n)
    elif method == 'triangular':
        x = np.arange(n, dtype=float)
        a = (x + 1) * (n - x)
    elif method == 'sine':
        x = np.arange(n, dtype=float)
        a = np.sin(np.pi * (x + 0.5) / n)   # positive on (0,π)
    else:   # fallback
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a
