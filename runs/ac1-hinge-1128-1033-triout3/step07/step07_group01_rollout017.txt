def _mirror_descent(a, best_val, deadline):
    n = len(a)
    step = 0.5   # initial
    decay = 0.9999
    min_step = 1e-5
    while time.time() < deadline and step > min_step:
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # compute gradient vector g = a_{k - i} for valid indices
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end+1)
        jdx = k - idx
        g[idx] = a[jdx]
        # compute multiplicative update factor
        factor = np.exp(-step * g)
        a = a * factor
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0/n)
        cur_val = _fast_evaluate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # decay step
        step = max(step * decay, min_step)
    return a, best_val

def _permute_hill_climb(a, cur_val, deadline):
    n=len(a)
    cur = a.copy()
    best = cur.copy()
    best_val = cur_val
    T = 0.01
    cooling = 0.9999
    while time.time() < deadline:
        i, j = np.random.randint(0,n, size=2)
        if i==j: continue
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        val = _fast_evaluate(new)
        delta = val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best = new.copy()

        T = max(T * cooling, 1e-8)

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm (≈ 2 min per restart, total < 1000 s):
1. Pick a random Dirichlet‑type distribution of masses (shape α chosen
   log‑uniformly).  The values are sorted and then paired anti‑diagonally:
   for each mirrored pair (i, n‑1‑i) one entry receives a “large’’ value,
   the other a “small’’ value.  The side (left/right) is chosen randomly
   and the order inside the large/small groups is shuffled, which spreads
   heavy values over the whole domain while keeping each large entry
   paired with a tiny one (the central convolution term is thus tiny).
2. Refine the candidate with a few hundred steps of exponential mirror‑
   descent on the exact max‑convolution (the same update used in the
   reference solution, but with a slower decay and a smaller minimum
   step‑size, giving a finer search).
3. Run a short simulated‑annealing / hill‑climbing phase that swaps two
   positions (and occasionally performs a tiny mass‑transfer) and accepts
   uphill moves with a decreasing temperature.  This phase optimises the
   permutation of the values, i.e. the exact placement of large vs. small
   entries.
4. Keep the best sequence seen over all restarts and finally evaluate it
   with the official ``evaluate_sequence``.  The whole search respects the
   1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT convolution, fast evaluation, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the formula simplifies).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _random_antidiagonal_assign(sorted_vals: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Build a sequence where each mirrored pair (i, n‑1‑i) contains one
    “large’’ element and one “small’’ element.  The side is chosen at random
    and the order of large/small elements is shuffled, which spreads heavy
    masses over the whole line while preserving the optimal (large–small)
    pairing for the central convolution index.
    """
    n = len(sorted_vals)
    half = n // 2

    if n % 2 == 0:
        small = sorted_vals[:half].copy()
        large = sorted_vals[half:].copy()
        middle = None
    else:
        small = sorted_vals[:half].copy()
        large = sorted_vals[half + 1:].copy()
        middle = sorted_vals[half]

    rng.shuffle(small)
    rng.shuffle(large)

    a = np.zeros(n, dtype=float)

    for i in range(half):
        # random side choice for the large element
        if rng.random() < 0.5:
            a[i] = large[i]
            a[n - 1 - i] = small[i]
        else:
            a[i] = small[i]
            a[n - 1 - i] = large[i]

    if middle is not None:
        a[half] = middle

    # sanity – sum should already be one (up to rounding)
    a /= a.sum()
    return a


def _mirror_descent(a0: np.ndarray, cur_val: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponential mirror‑descent on the exact max‑convolution.
    Uses a slower decay and a much smaller minimum step‑size than the
    reference implementation, allowing a finer search.
    """
    a = a0.copy()
    n = a.shape[0]
    step = 0.5                # initial multiplicative step
    decay = 0.9995            # geometric decay per iteration
    min_step = 5e-6           # stop when step is tiny
    while time.time() < deadline and step > min_step:
        conv = _full_convolution(a)
        k = int(conv.argmax())

        # gradient of the active convolution term (ignoring the factor 2)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        jdx = k - idx
        g[idx] = a[jdx]

        # exponential update (mirror descent on the simplex)
        a = a * np.exp(-step * g)

        # keep on the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        cur_val = _fast_evaluate(a)
        step = max(step * decay, min_step)

    return a, cur_val


def _permute_hill_climb(a0: np.ndarray, cur_val: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing / simulated‑annealing on the permutation
    (and occasional tiny mass‑transfer).  Only swaps are guaranteed to keep
    the multiset of values unchanged; a small mass‑transfer move allows the
    algorithm to escape shallow plateaus.
    """
    a = a0.copy()
    best = a.copy()
    best_val = cur_val
    n = a.shape[0]
    rng = np.random.default_rng()
    T = 0.02                # initial temperature
    cooling = 0.99997

    while time.time() < deadline:
        if rng.random() < 0.7:                # pure swap (most frequent)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                 # tiny mass‑transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-15:
                continue
            delta = a[i] * 0.05 * rng.random()   # up to 5 % of a_i
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑normalise (protect against drift)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_a.copy()
                print(f"[perm hill] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-8)

    return best, best_val

    deadline = start_time + budget_s - 5.0          # safety margin for final eval

    # ------------------------------------------------------------------
    # Problem size – try to reuse an existing construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # Initialise with a few cheap baselines.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')

    if has_old:
        a0 = old / old.sum()
        v = _fast_evaluate(a0)
        best_seq = a0.copy()
        best_val = v
        print(f"[init] previous construction value = {v:.6f}")

    uniform = np.full(n, 1.0 / n)
    v = _fast_evaluate(uniform)
    if v < best_val:
        best_val = v
        best_seq = uniform.copy()
    print(f"[init] uniform value = {v:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – many short restarts until time runs out.
    # ------------------------------------------------------------------
    restart = 0
    # a small list of α‑values to draw from (log‑uniform sampling will be used)
    while time.time() < deadline:
        restart += 1

        # ---- sample a Dirichlet‑type mass distribution -----------------
        log_alpha = np.random.uniform(np.log(0.15), np.log(5.0))
        alpha = np.exp(log_alpha)
        vals = rng.gamma(shape=alpha, scale=1.0, size=n)
        vals /= vals.sum()                      # now sum = 1
        sorted_vals = np.sort(vals)            # ascending

        # ---- create a few random anti‑diagonal permutations ------------
        local_best = None
        local_best_val = float('inf')
        for _ in range(3):                     # try a handful of random pairings
            a = _random_antidiagonal_assign(sorted_vals, rng)
            v = _fast_evaluate(a)
            if v < local_best_val:
                local_best_val = v
                local_best = a.copy()

        # ---- mirror‑descent refinement ---------------------------------
        md_deadline = min(deadline, time.time() + 8.0)   # ~8 s per restart
        a_md, v_md = _mirror_descent(local_best, local_best_val, md_deadline)
        if v_md < best_val:
            best_val = v_md
            best_seq = a_md.copy()
            print(f"[restart {restart}] after mirror descent new best = {best_val:.6f}")

        # ---- permutation hill‑climb ------------------------------------
        hc_deadline = min(deadline, time.time() + 4.0)   # ~4 s for hill‑climb
        a_hc, v_hc = _permute_hill_climb(a_md, v_md, hc_deadline)
        if v_hc < best_val:
            best_val = v_hc
            best_seq = a_hc.copy()
            print(f"[restart {restart}] after hill‑climb new best = {best_val:.6f}")

        # occasional aggressive reshuffle to avoid being stuck
        if restart % 15 == 0:
            # random permutation without the anti‑diagonal constraint
            a_rand = vals.copy()
            rng.shuffle(a_rand)
            v_rand = _fast_evaluate(a_rand)
            if v_rand < best_val:
                best_val = v_rand
                best_seq = a_rand.copy()
                print(f"[restart {restart}] random shuffle beat best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator and return.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_seq.tolist()
