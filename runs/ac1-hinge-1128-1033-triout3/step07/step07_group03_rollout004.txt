"""
search_candidate.py

Search for a non‑negative step function that minimises
    evaluate_sequence(sequence)

The algorithm consists of several stages within the allotted
time budget (default 1000 s).

Stage 0 –  initialise several diverse candidates (uniform,
exponential, power‑law) and keep the best.

Stage 1 –  sub‑gradient descent on the true L∞‑objective.
          At each iteration the active set of convolution indices
          whose value is within a relative tolerance of the current
          maximum is identified.  The sub‑gradient is obtained by a
          single FFT‑based convolution and a multiplicative (exponentiated)
          update is performed.  The step‑size decays as 1/√t and the
          active‑set tolerance is tightened over time.  Periodically the
          anti‑symmetric re‑ordering heuristic is applied.

Stage 2 –  smooth surrogate optimisation (log‑sum‑exp) with L‑BFGS‑B.
          Several random restarts from the current best point are used.
          The ε‑schedule spans 5·10⁻³ … 10⁻¹² and the optimiser is allowed
          up to 800 iterations per run.

Stage 3 –  final local refinement using simulated‑annealing‑style
          hill‑climbing (random swaps and mass transfers).  Worse moves
          are occasionally accepted to escape shallow local minima.

All updates keep the vector on the probability simplex (non‑negative,
unit sum).  The result is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative floating‑point artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values on the left and the smallest on the right,
    i.e. pair a large coefficient with a small one when the sequence is reversed.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # large → left side
        res[n - 1 - i] = s[i]     # small → right side
    if n % 2:
        res[half] = s[half]
    return res


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)     (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # self‑convolution
    conv = _full_convolution(a)
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t a
    weights = np.exp(shifted)
    p = weights / weights.sum()                       # p_k over convolution indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries give derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                      # ∂L/∂a_i

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.

    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=11)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 800,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to a (probability simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric re‑ordering heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} a_{k‑i} using a single FFT.
    ``active_idx`` contains convolution indices (0 … 2n‑2) belonging
    to the active set.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n-2) - i] for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent_multi(a_start: np.ndarray, deadline: float):
    """
    Exponentiated mirror‑descent that reduces the *active* convolution peaks.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.8                     # initial step‑size factor
    rel_tol = 0.05                    # start with a generous active‑set tolerance

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten the tolerance gradually
        if it > 0 and it % 20000 == 0:
            rel_tol = max(rel_tol * 0.5, 1e-5)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]   # indices of near‑max peaks

        # gradient via FFT (sub‑gradient of the L∞‑norm)
        g = _active_set_gradient(a, active)

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative update
        a = a * np.exp(-alpha * g)

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional logging & re‑ordering
        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")
            if it % 5000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[sgd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing based hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    """
    Random local moves (swap or mass transfer) with an annealing schedule.
    Only strictly better solutions are kept for the reported best,
    but occasional uphill moves are accepted to escape plateaus.
    """
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.02                     # initial “temperature”
    it = 0

    while time.time() < deadline:
        it += 1
        # propose a move
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        # evaluate
        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            accept = True
        else:
            # annealing acceptance probability
            T = T0 * (0.9995 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val_local:.6f}")

    return best_a, best_val_local


# ----------------------------------------------------------------------
# Helper to generate power‑law shaped candidates (stage 0)
# ----------------------------------------------------------------------
def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (dist_to_end+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – larger n gives more freedom but costs more time.
    # n = 8000 was used for the previous best; we keep it for compatibility.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialisation – try to reuse a known construction if present,
    # otherwise generate a few diverse starting points.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        # fall back to a uniform vector
        a = np.full(n, 1.0 / n)

    # ensure simplex
    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 1e-12 else np.full(n, 1.0 / n)

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – a few cheap random / structured candidates
    # ------------------------------------------------------------------
    # (i) exponential random vectors
    for tr in range(3):
        if time.time() >= deadline - 250:
            break
        cand = rng.exponential(scale=1.0, size=n)
        cand /= cand.sum()
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # (ii) symmetric power‑law shapes
    for p in (0.3, 0.5, 0.7, 1.0):
        if time.time() >= deadline - 250:
            break
        cand = _powerlaw_candidate(p, n)
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[power‑law p={p:.1f}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – sub‑gradient descent on the true max‑convolution objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:
        a_sgd, val_sgd = _subgradient_descent_multi(best_a, deadline)
        if val_sgd < best_val:
            best_val, best_a = val_sgd, a_sgd.copy()
            print(f"[sub‑gradient] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth surrogate optimisation (L‑BFGS) with several restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 120 and _GLOBAL_N is None:
        # dummy initialisation of globals for the first call
        pass

    restarts = 0
    while time.time() < deadline - 120 and restarts < 8:
        restarts += 1
        # perturb current best for a fresh start
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.25, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[restart {restarts}] L‑BFGS improved to {best_val:.6f}")

        # one more anti‑symmetric sweep
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[restart {restarts}] reorder improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final refinement via simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_simulated_anneal(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
