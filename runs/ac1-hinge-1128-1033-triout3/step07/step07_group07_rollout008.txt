"""
Custom optimizer for `evaluate_sequence`.

The optimizer consists of three stages, each respecting the global
time‑budget (default 1000 s).

1️⃣  **Smooth surrogate optimisation** –  The non‑smooth “max‑convolution”
    in the original objective is replaced by a log‑sum‑exp surrogate.
    The smoothing parameter ε is driven from 1e‑2 down to 1e‑16
    (log‑space).  For each ε we run a fairly long L‑BFGS‑B optimisation
    on the re‑parameterised variables *w* (soft‑max of *w* yields a
    feasible probability vector).  This phase replaces the original
    L‑BFGS run (300 iterations) by up to 1200 iterations per ε.

2️⃣  **Randomised restarts** –  The best vector from stage 1 is perturbed
    and re‑optimised on the finer tail of the ε‑schedule.  As many
    restarts as the remaining budget permits (≈ 30) are performed.

3️⃣  **Fine polishing** –  Two inexpensive, exact‑objective refinements
    are applied:
        • aggressive mass‑transfer hill‑climbing,
        • projected sub‑gradient descent with an adaptive step‑size.

The best vector found is finally evaluated with the provided
`evaluate_sequence` and returned as a list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # tiny numerical negatives may appear; clip to zero
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (FFT based)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (value, gradient) of the smoothed surrogate

        Lε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    where a = softmax(w) ( Σ a = 1, a_i ≥ 0 ).
    """
    # soft‑max (feasible a)
    w_shift = w - np.max(w)          # improve numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # full convolution a*a
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing term
    shifted = (conv - max_conv) / eps
    # log‑sum‑exp safely
    # np.logaddexp.reduce works element‑wise, we just need the scalar
    log_sum = np.logaddexp.reduce(shifted)   # = log Σ exp(...)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()               # probability vector over lags

    # derivative of Σ p_k * conv_k w.r.t a: convolution with reversed p
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)             # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]           # centre n entries
    grad_a = 4.0 * n * seg[::-1]              # factor 4 comes from both a‑variables

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index of the last positive entry
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill‑climbing
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              cur_best_val: float,
                              deadline: float,
                              rng: np.random.Generator,
                              step_start: float = 0.15):
    """
    Hill‑climbing that repeatedly moves a fraction of mass away from the
    lag attaining the current maximum convolution.
    Returns the best vector and its objective value.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_val = cur_best_val
    step_frac = step_start

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # identify all pairs (i,j) that contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]

        if contrib.size == 0:
            continue

        # choose a few most influential pairs (the larger the contribution)
        top_n = min(4, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        # choose destinations among low‑mass indices (excluding the donors)
        low_mass = np.where(cur_a < cur_a.mean())[0]
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                # take the same amount from both ends of the pair
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.7, 1e-8)
            continue

        # re‑normalise to stay on the simplex
        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.7, 1e-8)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_best_val:
            cur_a = new_a
            cur_best_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            # slowly enlarge the step to speed up when we are improving
            step_frac = min(step_frac * 1.05, 0.4)
        else:
            # shrink step when not improving
            step_frac = max(step_frac * 0.85, 1e-8)

    return best_a, best_val


# ----------------------------------------------------------------------
# Fine sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         cur_best_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Projected sub‑gradient descent on   J(a)=2·n·max_k Σ_i a_i a_{k‑i}
    (exact objective).  Returns the best vector and its value.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_val = cur_best_val
    cur_val = cur_best_val
    eta = 0.12  # initial step size (on the simplex)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient for the active lag
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # take a step and project back onto the simplex
        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            # increase step size modestly when we are making progress
            eta = min(eta * 1.07, 0.5)
        else:
            # otherwise back‑off
            eta = max(eta * 0.73, 1e-6)

    return best_a, best_val

        # the best vector we have.
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    return best_a.tolist()
