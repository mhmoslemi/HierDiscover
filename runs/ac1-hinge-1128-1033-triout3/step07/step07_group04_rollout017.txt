def project_simplex(v):
    """Project onto the probability simplex {x>=0, sum=1}."""
    # Following algorithm: sort descending, find rho, compute theta = (sum_{i=1}^{rho} v_i - 1)/rho.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    n = a.shape[0]
    sorted_vals = np.sort(a)  # ascend
    out = np.empty_like(a)

def _alternating_reorder(a):
    sorted_vals = np.sort(a)
    left = 0
    right = len(sorted_vals)-1
    out = np.empty_like(a)
    idx = 0
    while left <= right:
        out[idx] = sorted_vals[right]
        idx += 1
        right -= 1
        if left <= right:
            out[idx] = sorted_vals[left]
            idx += 1
            left += 1
    return out

def _descending_reorder(a):
    return np.sort(a)[::-1]

"""
Propose a step‑function sequence that minimizes the supplied
evaluate_sequence() value.

The search consists of three phases:

1. **Initialisation** – load the previous best construction (if
   available), otherwise start from a random Dirichlet vector.
   The vector is normalised to sum = 1.

2. **Sub‑gradient simulated annealing** – repeatedly
   * locate the convolution entry that attains the current maximum,
   * compute the exact gradient of that entry w.r.t. the coefficients,
   * perform a multiplicative update `a ← a·exp(‑η·∇)`,
   * re‑normalise to the simplex,
   * accept the move with a simple temperature schedule.
   This directly reduces the true maximum of the autocorrelation and
   usually improves the evaluation by a few‑per‑cent.

3. **Light post‑processing** – a few deterministic re‑orderings,
   random permutations and a short swap/transfer hill‑climb are applied
   until the time budget expires.

The algorithm respects the 1000 s limit, works with any length (default
4000→8000), and returns a plain list of floats ready for the external
``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (exactly reproduces evaluate_sequence)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a using FFT (non‑negative input)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # tiny negatives can appear due to rounding – clamp them
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast version of the provided evaluate_sequence (sum already non‑negative)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – needed after additive updates
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Re‑ordering heuristics (these only permute the values)
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Largest values to the left, smallest to the right (pairwise)."""
    n = a.shape[0]
    vals = np.sort(a)                     # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = vals[n - 1 - i]          # large on the left
        out[n - 1 - i] = vals[i]          # small on the right
    if n % 2:
        out[half] = vals[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    vals = np.sort(a)
    left, right = 0, len(vals) - 1
    out = np.empty_like(a)
    k = 0
    while left <= right:
        out[k] = vals[right]
        k += 1
        right -= 1
        if left <= right:
            out[k] = vals[left]
            k += 1
            left += 1
    return out


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Simply sort in descending order."""
    return np.sort(a)[::-1]


# ----------------------------------------------------------------------
# Sub‑gradient descent with a very light simulated‑annealing accept rule
# ----------------------------------------------------------------------
def _subgradient_sa(a_init: np.ndarray,
                    time_limit: float,
                    step0: float = 5.0,
                    temp0: float = 0.1) -> (np.ndarray, float):
    """
    Reduce max (convolution) by repeated multiplicative updates.
    Returns the best vector seen and its eval_fast value.
    """
    a = a_init.copy()
    cur_val = _eval_fast(a)
    best_a, best_val = a.copy(), cur_val

    T = cur_val * temp0                 # initial temperature
    decay_T = 0.9995
    decay_step = 0.99995
    step = step0

    while time.time() < time_limit:
        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # ----- gradient of the chosen convolution entry (ordered pairs) -----
        n = a.shape[0]
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k_max - i_idx

        g = np.zeros_like(a)
        mask = i_idx == j_idx                     # diagonal terms
        if np.any(mask):
            g[i_idx[mask]] = a[i_idx[mask]]
        if np.any(~mask):
            g[i_idx[~mask]] = 2.0 * a[j_idx[~mask]]

        # ----- multiplicative update (keeps positivity) -----
        factor = np.exp(-step * g)                # element‑wise
        a_new = a * factor
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / s

        new_val = _eval_fast(a_new)
        delta = new_val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_a, best_val = a.copy(), cur_val
                print(f"[subgrad‑SA] new best = {best_val:.6f}")

        # annealing schedule
        T *= decay_T
        step *= decay_step

        # occasional tiny random perturbation to avoid stagnation
        if np.random.rand() < 0.001:
            perturb = np.random.rand(n) * 0.005
            a = np.maximum(a + perturb, 0.0)
            a = a / a.sum()
            cur_val = _eval_fast(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a step‑function minimising ``evaluate_sequence``."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size – a bit larger than the previous 4000 for extra freedom
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # initialise from a previous construction if possible
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.random.rand(n)
        print("[init] random start")

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()                     # project onto simplex
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. sub‑gradient simulated annealing (core optimisation)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    subgrad_deadline = time.time() + min(remaining * 0.60, 300.0)  # at most 5 min
    best_a, best_val = _subgradient_sa(best_a, subgrad_deadline,
                                       step0=5.0, temp0=0.05)
    print(f"[phase‑1] best after sub‑gradient = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. deterministic re‑orderings (they may reduce the peak further)
    # ------------------------------------------------------------------
    for reorder in (_anti_symmetric_reorder,
                    _alternating_reorder,
                    _descending_reorder):
        a_r = reorder(best_a)
        v_r = _eval_fast(a_r)
        if v_r < best_val:
            best_a, best_val = a_r.copy(), v_r
            print(f"[reorder] {reorder.__name__} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. a few random permutations – cheap, sometimes useful
    # ------------------------------------------------------------------
    for _ in range(5):
        perm = np.random.permutation(n)
        a_r = best_a[perm]
        v_r = _eval_fast(a_r)
        if v_r < best_val:
            best_a, best_val = a_r.copy(), v_r
            print(f"[perm] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. final cheap hill‑climbing (swap / small transfer) until deadline
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = best_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # small transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] == 0.0:
                continue
            delta = best_a[i] * 0.05 * rng.random()
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta

        val = _eval_fast(cand)
        if val < best_val - 1e-12:
            best_a, best_val = cand, val
            print(f"[refine] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. final exact evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    return best_a.tolist()
