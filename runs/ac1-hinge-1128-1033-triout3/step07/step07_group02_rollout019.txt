"""
A fast, multi‑phase optimiser for the “step‑function” problem.

The programme works in three stages within the 1000 s budget:

1️⃣ **Differential‑Evolution (DE) over a flexible 3‑parameter family**  
   a_i = λ·Beta(p,q)_i + (1‑λ)·Uniform_i    
   – p,q ∈ [0.1, 6] shape a (truncated) Beta distribution.  
   – λ ∈ [0,1] mixes it with a flat background.  
   The DE loop (pop‑size 30) repeatedly mutates / recombines the (p,q,λ) triples,
   keeping the best vector (exact objective, full FFT convolution).

2️⃣ **Adam on the simplex** – the best DE vector is refined by minimising a
   smooth‑max approximation of the true objective.  The Adam code is the same
   as in the reference implementation but runs longer (≈30 % of the total
   budget) and with a slightly larger idle‑restart limit.

3️⃣ **Simulated‑annealing hill‑climbing** – a specialised Metropolis walk that
   moves a small amount of mass from indices that heavily contribute to the
   current maximal convolution entry to indices that contribute little.
   Temperature and maximal transfer fraction are annealed to zero.
   This phase uses the remaining time.

All heavy work (FFT convolutions, gradients) is performed with NumPy.
The final vector is returned as a plain Python list; its sum is 1, which is
scale‑invariant for ``evaluate_sequence``.
"""

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and gradient w.r.t. a
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and its gradient.
    The gradient corresponds to the exact objective
    2·n·max(conv) approximated by a soft‑max with temperature eps.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # gradient of Σ p_k * b_k w.r.t a
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 4n from 2·n·max
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimiser that minimises a smooth approximation of the objective."""
    w = np.log(a0 + 1e-12)                       # logits → simplex via soft‑max

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.001
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 30_000           # more patience than the reference code

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        smooth_val, grad_a = _smooth_max_and_grad(a, eps)

        # chain rule through the soft‑max (w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # very occasional jitter
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        # restart on stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Local search that moves a small amount of mass from heavy to light contributors."""
    a = a0.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.012, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – prob. ∝ contribution
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – biased towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to build a Beta‑mixture vector from parameters (p,q,λ)
# ----------------------------------------------------------------------
def _construct_vector(p: float, q: float, lam: float, n: int) -> np.ndarray:
    """Return a normalised vector  λ·Beta(p,q) + (1‑λ)·Uniform."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # use log‑space to avoid overflow for large exponents
    log_beta = (p - 1.0) * np.log(i) + (q - 1.0) * np.log(n - i + 1.0)
    beta = np.exp(log_beta)
    beta_sum = beta.sum()
    if not np.isfinite(beta_sum) or beta_sum == 0.0:
        beta = np.ones_like(beta)
        beta_sum = beta.shape[0]
    beta /= beta_sum
    uniform = np.full(n, 1.0 / n)
    a = lam * beta + (1.0 - lam) * uniform
    return a


# ----------------------------------------------------------------------
# Objective wrapper for a parameter triple (p,q,λ)
# ----------------------------------------------------------------------
def _objective_params(params: np.ndarray, n: int) -> float:
    p, q, lam = params
    a = _construct_vector(p, q, lam, n)
    return _objective(a)


# ----------------------------------------------------------------------
# Differential Evolution search over (p,q,λ)
# ----------------------------------------------------------------------
def _de_search(n: int,
               deadline: float,
               start_time: float,
               popsize: int = 30,
               F: float = 0.8,
               Cr: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Simple DE optimiser for the 3‑parameter family.
    Returns the best raw vector and its exact objective.
    """
    # parameter bounds
    low = np.array([0.1, 0.1, 0.0])
    high = np.array([6.0, 6.0, 1.0])

    # initialise population uniformly in the box
    pop = low + (high - low) * np.random.rand(popsize, 3)
    vals = np.empty(popsize, dtype=np.float64)

    for i in range(popsize):
        vals[i] = _objective_params(pop[i], n)

    best_idx = int(np.argmin(vals))
    best_params = pop[best_idx].copy()
    best_val = vals[best_idx]

    # DE loop
    while time.time() < deadline:
        for i in range(popsize):
            # mutation – pick three distinct individuals different from i
            idxs = np.arange(popsize)
            idxs = idxs[idxs != i]
            a, b, c = np.random.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, low, high)

            # binomial crossover
            cross = np.random.rand(3) < Cr
            if not cross.any():
                cross[np.random.randint(3)] = True
            trial = np.where(cross, mutant, pop[i])

            val = _objective_params(trial, n)
            if val < vals[i]:
                pop[i] = trial
                vals[i] = val
                if val < best_val:
                    best_val = val
                    best_params = trial.copy()

    best_vec = _construct_vector(best_params[0], best_params[1],
                                best_params[2], n)
    return best_vec, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries give a good trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # Phase 0 – differential evolution over the (p,q,λ) family
    # ------------------------------------------------------------------
    de_deadline = start_time + 0.15 * budget_s
    print("[init] starting DE search")
    best_a, best_val = _de_search(n, de_deadline, start_time)
    print(f"[init] DE best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.35 * budget_s
    print("[phase A] starting Adam optimisation")
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)
    print(f"[phase A] best after Adam = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    print("[phase B] starting hill‑climbing")
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=phase_a_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)
    print(f"[phase B] final best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
