def _project_simplex(v):
    # ensure non-negative? Actually algorithm can handle any v.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑value step‑function for the given evaluation routine.

Algorithm
---------
1. Initialise a non‑negative vector `a` (probability simplex) – reuse the
   previously found sequence if it exists, otherwise a random Dirichlet
   start.

2. **Smoothed gradient descent** – minimise a smooth approximation of the
   objective `max(conv(a,a))` by using a soft‑max over the convolution
   entries.  The gradient of the smoothed objective can be computed
   efficiently via one FFT‑based convolution.  After each trial step the
   vector is projected back onto the probability simplex.

3. **Simulated annealing** – perform many tiny mass‑transfer moves.
   Donor indices are sampled proportionally to their current mass,
   recipients are chosen preferentially from low‑mass positions.
   Moves are accepted if they improve the exact objective or with a
   probability that depends on a slowly decreasing temperature.

4. **Aggressive mass‑transfer hill‑climb** – a deterministic refinement
   similar to the one used in the reference code but with adaptive
   step‑size.  It repeatedly relocates a fraction of the most damaging
   mass to two low‑mass slots whenever the exact objective improves.

5. Return the best sequence found (as a list of floats) before the
   time budget runs out.  The final score is printed using the provided
   `evaluate_sequence` function.

All steps respect the 1000 s time budget and use only NumPy and the
standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    size = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    return np.fft.irfft(fx * fy, size)[: len(x) + len(y) - 1]


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real input)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # tiny negative values from floating‑point noise are harmless
    conv[conv < 0] = 0.0
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence when the sum of a equals 1."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is all zero or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1. Smoothed (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Gradient descent on a soft‑max approximation of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    n = a.shape[0]

    # initial smoothing parameter (a fraction of the current spread)
    conv = _full_convolution(a)
    eps = max(0.1 * (conv.max() - conv.min()), 1e-12)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a)
        maxc = conv.max()
        minc = conv.min()

        # gradually tighten the smoothing
        eps = max(eps * 0.995, 1e-12)

        # soft‑max weights (numerically stable)
        w = np.exp((conv - maxc) / eps)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isinf(w_sum):
            # fall back to a hard max (only the maximal entries get weight)
            w = np.zeros_like(conv)
            w[conv == maxc] = 1.0
            w_sum = w.sum()
        w /= w_sum

        # gradient = 2 * (a * w_reversed) evaluated at positions (2n‑2‑j)
        w_rev = w[::-1]
        grad_full = _fft_convolve(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # back‑tracking line search
        lr = 0.5 / math.sqrt(it + 1)
        improved = False
        for _ in range(8):
            a_trial = _project_simplex(a - lr * grad)
            val = _objective(a_trial)
            if val < best_val:
                a = a_trial
                best_a = a_trial.copy()
                best_val = val
                improved = True
                break
            lr *= 0.5
        if not improved:
            # keep the current a (no improvement) and continue
            pass

        if it % 200 == 0:
            print(f"[smooth] iter {it:5d}  best = {best_val:.6f}")

    return best_a, best_val

# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iter: int = 60000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a temperature schedule."""
    a = a0.copy()
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    n = a.shape[0]
    T = cur_val * 0.01 + 1e-12  # initial temperature
    cooling = 0.9999
    step_frac = 0.15

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # donors sampled proportionally to current mass (larger donors moved more often)
        i = int(rng.choice(n, p=a))
        j = int(rng.choice(n, p=a))

        # amount to relocate
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            continue

        # choose two distinct recipients, preferring low‑mass slots
        mask = np.ones(n, dtype=bool)
        mask[i] = False
        mask[j] = False
        recip_candidates = np.where(mask)[0]
        inv_weights = 1.0 / (a[recip_candidates] + 1e-12)
        inv_weights /= inv_weights.sum()
        p, q = rng.choice(recip_candidates, size=2, replace=False, p=inv_weights)

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta
        new_a[p] += delta
        new_a[q] += delta

        new_a = np.maximum(new_a, 0.0)
        # projection onto simplex (sum is preserved up to rounding)
        new_a = _project_simplex(new_a)

        new_val = _objective(new_a)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / T):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[anneal] iter {it:6d}  new best = {best_val:.6f}")

            # adapt step size
            step_frac = min(step_frac * 1.01, 0.5)
        else:
            step_frac = max(step_frac * 0.99, 1e-4)

        # temperature cooling
        T = max(T * cooling, 1e-12)

        if it % 5000 == 0:
            print(f"[anneal] iter {it:6d}  current = {cur_val:.6f}  best = {best_val:.6f}")

    return best_a, best_val

def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 6000) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing: move a fraction of the most damaging pair
    to two low‑mass destinations.  The step fraction is adapted based on
    successes/failures.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that form the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass to relocate
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # two low‑mass destinations, avoiding donors
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        p, q = rng.choice(cand, size=2, replace=False)

        new_a[p] += delta
        new_a[q] += delta

        new_a = _project_simplex(new_a)
        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val

            a0 = rng.random(n)
        a0 = _project_simplex(a0)
        print("[init] Re‑using existing sequence.")
    except Exception:
        n = default_n
        a0 = rng.random(n)
        a0 = _project_simplex(a0)
        print("[init] Random start.")

    print(f"[init] n={n}, initial objective = {_objective(a0):.6f}")

    # ---- stage 1 : smoothed gradient descent --------------------------
    a_smooth, val_smooth = _smooth_descent(a0, deadline, rng, max_iters=3000)
    print(f"[smooth] best after descent = {val_smooth:.6f}")

    # ---- stage 2 : simulated annealing --------------------------------
    a_anneal, val_anneal = _simulated_annealing(a_smooth, deadline, rng,
                                                max_iter=60000)
    print(f"[anneal] best after annealing = {val_anneal:.6f}")

    # ---- stage 3 : aggressive hill‑climb -------------------------------
    a_final, val_final = _mass_transfer_hillclimb(a_anneal, deadline, rng,
                                                 max_no_improve=6000)
    print(f"[final] best before evaluate_sequence = {val_final:.6f}")

    # ---- final evaluation ---------------------------------------------
    try:
        final_score = evaluate_sequence(a_final.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a_final.tolist()
