def _smooth_obj_and_grad_groups(w, eps, n, seg_len):
    # w shape G
    w_shift = w - np.max(w)
    p = np.exp(w_shift)
    p_sum = np.sum(p)
    p /= p_sum   # softmax raw (sum 1)
    g = p / seg_len   # since sum(g) = 1/seg_len
    a = np.repeat(g, seg_len)
    # compute conv and gradient w.r.t a using previously defined code (or replicate).
    conv = _fft_convolve_self(a)
    max_conv = conv.max()
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum
    # gradient w.r.t a:
    weights = np.exp(shifted)
    pw = weights / weights.sum()
    p_rev = pw[::-1]
    conv_ap = _conv_fft(a, p_rev)
    seg = conv_ap[n - 1: 2*n - 1]
    grad_a = 4.0 * n * seg[::-1]
    # sum to groups:
    # group sum of grad_a per block of seg_len
    grad_g = np.add.reduceat(grad_a, np.arange(0, n, seg_len))
    # chain rule:
    dot = np.dot(grad_g, p)  # note p sums to 1
    grad_w = (p / seg_len) * (grad_g - dot)
    return L, grad_w

"""
Hybrid group‑based optimiser for the step‑function convolution problem.

Strategy
--------
1.  Reduce the dimensionality: the 4000‑element vector is constrained to be
    piece‑wise constant on blocks of equal length (default 20).  This yields
    G = 200 optimisation variables (the block heights) while preserving the
    full‑length evaluation via `np.repeat`.

2.  Optimise a smooth surrogate of the original objective:
        L_ε(a) = 2·n·max(a∗a) + 2·n·ε·log∑_k exp((a∗a)_k‑max)/ε .
    The gradient w.r.t. the block variables is obtained analytically
    (soft‑max + FFT‑based convolutions) and fed to L‑BFGS‑B.  A continuation
    scheme gradually reduces ε from 1e‑2 to 1e‑12.

3.  Multiple random restarts (including hand‑crafted geometric / power‑law
    families) are performed; after each restart the solution from the
    previous ε‑stage is used as a warm start.

4.  The best block vector is refined by a fast simulated‑annealing phase
    (swap / mass‑transfer moves on the block level) and a final deterministic
    hill‑climbing pass.

5.  The full‑length sequence is reconstructed, evaluated with the provided
    `evaluate_sequence`, and returned.

All heavy work uses FFT‑based convolutions (O(n log n)).  The routine respects
the 1000 s time budget (the deadline is checked after each major loop).
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (soft‑max + log‑sum‑exp)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None          # smoothing parameter for current L‑BFGS run
_GLOBAL_N = None            # length of the full sequence
_GLOBAL_G = None            # number of blocks
_GLOBAL_SEG_LEN = None      # block size (n = G·SEG_LEN)


# ----------------------------------------------------------------------
# Utility: next power of two (for zero‑padding FFTs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# FFT‑based convolutions (real vectors)
# ----------------------------------------------------------------------
def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Clamp tiny negatives caused by numerical noise
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Fast evaluation (identical to `evaluate_sequence` for non‑negative vectors)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray) -> float:
    """Fast version of `evaluate_sequence` for a non‑negative vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=z}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum z."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all values are zero after projection
        return np.full_like(v, z / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth surrogate objective and gradient for block variables
# ----------------------------------------------------------------------
def _smooth_obj_and_grad_groups(w: np.ndarray, eps: float,
                                n: int, seg_len: int) -> (float, np.ndarray):
    """
    Smoothed objective L_ε and its gradient w.r.t. the block parameters w.

    w -- raw (unconstrained) block parameters, length G.
    eps -- smoothing parameter.
    n -- total length of the full sequence.
    seg_len -- length of each block (n = G * seg_len).
    """
    # ---- soft‑max over blocks (ensuring the block heights sum to 1/seg_len) ----
    w_shift = w - np.max(w)                # improve numerical stability
    exp_w = np.exp(w_shift)
    p = exp_w / np.sum(exp_w)              # p sums to 1  (probability on blocks)

    # block heights and full vector
    g = p / seg_len                         # sum(g) = 1/seg_len → sum(full a)=1
    a = np.repeat(g, seg_len)               # full length‑n vector

    # ---- convolution of the full vector with itself ----
    conv = _fft_convolve_self(a)
    max_conv = np.max(conv)

    # ---- log‑sum‑exp smoothing of the max ----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. the full vector a (same formula as in the reference code) ----
    weights = np.exp(shifted)
    pw = weights / np.sum(weights)          # softmax over convolution indices
    p_rev = pw[::-1]                        # reversed for convolution
    conv_ap = _conv_fft(a, p_rev)
    # central part of length n
    seg = conv_ap[n - 1: 2 * n - 1]          # corresponds to derivative of L wrt a
    grad_a = 4.0 * n * seg[::-1]            # see reference implementation

    # ---- aggregate gradient to block level (sum over each block) ----
    grad_g = np.add.reduceat(grad_a, np.arange(0, n, seg_len))

    # ---- chain rule through the soft‑max (block level) ----
    dot = np.dot(grad_g, p)                 # ⟨grad_g , p⟩
    # ∂L/∂w = (1/seg_len) * p * (grad_g - dot)
    grad_w = (p / seg_len) * (grad_g - dot)

    return L, grad_w


def _smooth_obj_and_grad_groups_wrapper(w: np.ndarray):
    """Wrapper required by `scipy.optimize.minimize`."""
    return _smooth_obj_and_grad_groups(w, _GLOBAL_EPS, _GLOBAL_N, _GLOBAL_SEG_LEN)


# ----------------------------------------------------------------------
# Hand‑crafted seed families (block level)
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, G: int) -> np.ndarray:
    """Block heights ∝ q**i, normalised to sum = 1/SEG_LEN."""
    if not (0.0 < q < 1.0):
        return np.full(G, 1.0 / G)
    denom = (1.0 - q ** G) / (1.0 - q)
    raw = q ** np.arange(G, dtype=float) / denom
    return raw


def _power_sequence(p: float, G: int) -> np.ndarray:
    """Block heights ∝ (i+1)**p, normalised to sum = 1/SEG_LEN."""
    idx = np.arange(1, G + 1, dtype=float)
    raw = idx ** p
    s = raw.sum()
    if s == 0.0:
        return np.full(G, 1.0 / G)
    return raw / s


def _repeat_blocks(g: np.ndarray) -> np.ndarray:
    """Expand block vector g to the full length‑n representation."""
    return np.repeat(g, _GLOBAL_SEG_LEN)


# ----------------------------------------------------------------------
# Simulated annealing on block variables (swap / transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing_blocks(g: np.ndarray, cur_val: float,
                                deadline: float) -> (np.ndarray, float):
    """
    SA on block vector `g`.  Moves:
        * swap of two blocks,
        * small mass transfer between two blocks,
        * random perturbation of a single block.
    """
    G = g.shape[0]
    rng = np.random.default_rng()
    best_g = g.copy()
    best_val = cur_val

    # temperature schedule
    T = max(cur_val * 0.05, 1e-6)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        move_type = rng.random()
        new_g = g.copy()

        if move_type < 0.4:                         # swap
            i, j = rng.integers(0, G, size=2)
            if i != j:
                new_g[i], new_g[j] = new_g[j], new_g[i]

        elif move_type < 0.8:                       # transfer
            i, j = rng.integers(0, G, size=2)
            if i != j and new_g[i] > 0.0:
                delta = new_g[i] * 0.05 * rng.random()
                new_g[i] -= delta
                new_g[j] += delta

        else:                                        # perturb a single block
            i = rng.integers(0, G)
            delta = (rng.random() - 0.5) * 0.02   # ±1 %
            new_g[i] = max(new_g[i] + delta, 0.0)

        # re‑project onto the simplex sum(g)=1/seg_len
        new_g = _simplex_projection(new_g, 1.0 / _GLOBAL_SEG_LEN)

        # fast evaluation
        a = _repeat_blocks(new_g)
        new_val = _eval_fast(a)

        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            g = new_g
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_g = g.copy()
                print(f"[SA‑blk] new best = {best_val:.6f} (iter {it})")

        T = max(T * decay, 1e-12)
        it += 1

    return best_g, best_val


# ----------------------------------------------------------------------
# Final deterministic hill‑climbing on block variables
# ----------------------------------------------------------------------
def _hill_climb_blocks(g: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Very cheap local search (swap / tiny transfer)."""
    G = g.shape[0]
    rng = np.random.default_rng()
    best_g = g.copy()
    best_val = cur_val

    while time.time() < deadline:
        if rng.random() < 0.3:                     # swap
            i, j = rng.integers(0, G, size=2)
            if i == j:
                continue
            cand_g = best_g.copy()
            cand_g[i], cand_g[j] = cand_g[j], cand_g[i]
        else:                                      # tiny transfer
            i, j = rng.integers(0, G, size=2)
            if i == j or best_g[i] == 0.0:
                continue
            delta = best_g[i] * 0.02 * rng.random()
            cand_g = best_g.copy()
            cand_g[i] -= delta
            cand_g[j] += delta

        # re‑project to the simplex (the transfer may have broken the sum)
        cand_g = _simplex_projection(cand_g, 1.0 / _GLOBAL_SEG_LEN)

        a = _repeat_blocks(cand_g)
        val = _eval_fast(a)
        if val < best_val - 1e-12:
            best_g = cand_g
            best_val = val
            print(f"[hill‑blk] new best = {best_val:.6f}")

    return best_g, best_val
