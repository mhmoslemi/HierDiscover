    max‑convolution (ε from 1e‑2 down to 1e‑12);
  * a mirror‑descent that follows the exact sub‑gradient of the
    max‑convolution (multiplicative updates on the simplex);

All convolutions use FFT for speed, the simplex projection is
exact, and the schedule of step‑sizes / ε values is chosen to

# ----------------------------------------------------------------------
# Global variables for the smooth surrogate optimiser
# ----------------------------------------------------------------------
_GLOBAL_EPS = None
_GLOBAL_N = None

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (length 2*n-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise can produce tiny negative values – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values on the left side, smallest on the right."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large → left side
        new_a[n - 1 - i] = sorted_vals[i]      # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smoothed objective
        L_ε(a) = 2 n * max(conv) + 2 n ε * log Σ exp((conv - max)/ε)
    and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)      (guarantees a_i ≥ 0, Σ a_i = 1)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution a*a
    conv = _full_convolution(a)          # length 2*n-1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()          # probability distribution over lag indices

    # derivative of max_conv part:
    # ∂max_conv/∂a_i = 2 * a_{k* - i}; we incorporate it by adding p contribution.
    # More conveniently compute derivative of Σ_k p_k conv_k w.r.t. a.
    # This equals 2 * (a *rev p) convolution (see original code).
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # The relevant slice corresponds to indices (n‑1) … (2n‑2‑1)
    grad_a = 4.0 * n * conv_ap[n - 1: 2 * n - 1][::-1]   # shape (n,)

    # Chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize(w_init: np.ndarray, deadline: float, max_iter: int = 2000):
    """
    Run L‑BFGS on the smoothed objective, decreasing ε.
    Returns the best (a, value) pair obtained before the deadline.
    """
    n = w_init.shape[0]
    w = w_init.copy()
    # ε‑schedule from 1e‑2 to 1e‑12 (log‑space, 9 values)
    eps_schedule = np.logspace(-2, -12, num=9)
    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 15.0:   # keep some room for later stages
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper, w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to a (simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # try anti‑symmetric re‑ordering
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback if nothing succeeded

        w_shift = w - np.max(w)
        a = np.exp(w_shift); a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())
    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution (sub‑gradient method)
# ----------------------------------------------------------------------
def _max_subgradient_descent(a_start: np.ndarray, deadline: float,
                             max_iters: int = 500_000):
    """
    Multiplicative mirror‑descent that follows the sub‑gradient of the
    current maximum convolution entry.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[subgrad init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.4   # initial step size
    decay = 0.995  # multiplicative decay every 1000 iterations

    while time.time() < deadline - 5.0 and it < max_iters:
        conv = _full_convolution(a)
        max_val = conv.max()
        # indices attaining the max (could be a few)
        idxs = np.flatnonzero(conv == max_val)
        # pick one arbitrarily (deterministic) to keep gradient cheap
        k = idxs[0]

        # compute sub‑gradient g_i = 2 * a_{k - i} (where defined)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_range
        g = np.zeros_like(a)
        g[idx_range] = a[rev_idx]          # factor 2 omitted (absorbed in step‑size)

        # schedule step size
        alpha = alpha0 * (decay ** (it // 1000))

        # multiplicative update (mirror‑descent on the simplex)
        a = a * np.exp(-alpha * g)

        # projection onto simplex (simple renormalisation, zeros stay zero)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            # numerical fallback – uniform vector
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # periodic logging and best‑track
        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[subgrad iter {it}] new best = {best_val:.6f} (max conv {max_val:.8f})")
            # occasional anti‑symmetric reordering (may improve further)
            if it % 2000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[subgrad reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (targeted mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random local moves that are biased towards indices that contribute
    heavily to the current maximal convolution entry.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        # identify current maximal shift and the pair contributing most
        conv = _full_convolution(cur_a)
        k = np.argmax(conv)
        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_range
        contrib = cur_a[idx_range] * cur_a[rev_idx]
        # pick a heavy contributing index proportionally
        if contrib.sum() == 0:
            i = rng.integers(i_start, i_end + 1)
        else:
            probs = contrib / contrib.sum()
            i = rng.choice(idx_range, p=probs)

        j = k - i  # the partner index

        # propose moving a small fraction from the larger of the pair
        if cur_a[i] < 1e-15 and cur_a[j] < 1e-15:
            continue
        # choose donor = i if a_i > a_j else j
        donor = i if cur_a[i] > cur_a[j] else j
        receiver = rng.integers(0, n)  # any position
        if donor == receiver:
            continue
        # amount to move (up to 5 % of donor)
        delta = cur_a[donor] * 0.05 * rng.random()
        if delta <= 0:
            continue
        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[receiver] += delta

        # re‑normalize (project onto simplex)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

The search proceeds in four stages (all bounded by a 1000 s budget):

    convolution entry.  Only strict improvements are kept.

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders required by the smooth‑surrogate optimiser
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # current vector length


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a real vector (length 2·n‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives caused by rounding
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values on the left, the smallest on the right.
    This implements the optimal rearrangement for the centre convolution
    (pair large with small).
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective
        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    where a_i = softmax(w_i).  Returns (L, ∇_w L).
    """
    # soft‑max (numerically stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution a * a
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (see original implementation for correctness)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # distribution over lag indices
    p_rev = p[::-1]

    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # slice the central n entries – they correspond to ∂L/∂a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # derivative w.r.t. a

    # chain‑rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper for ``scipy.minimize`` – uses the globals for ε and n."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize(w_init: np.ndarray, deadline: float, max_iter: int = 2000):
    """
    Run a short L‑BFGS schedule (ε from 1e‑2 down to 1e‑12) starting from
    the soft‑max parameters ``w_init``.  Returns the best (a, value) pair
    discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()
    # ε‑schedule (log‑space, 9 values)
    eps_schedule = np.logspace(-2, -12, num=9)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 15.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper, w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to the probability simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # try the anti‑symmetric ordering (often improves the centre peak)
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift); a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution (sub‑gradient method)
# ----------------------------------------------------------------------
def _max_subgradient_descent(a_start: np.ndarray, deadline: float,
                             max_iters: int = 500_000):
    """
    Multiplicative mirror‑descent that follows the sub‑gradient of the
    *currently maximal* convolution entry.  Works directly on the simplex.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[subgrad init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.4          # initial step‑size
    decay = 0.995         # decay factor (applied every 1000 iterations)

    while time.time() < deadline - 5.0 and it < max_iters:
        conv = _full_convolution(a)
        max_val = conv.max()
        # pick one index attaining the maximum (deterministic choice is fine)
        k = np.flatnonzero(conv == max_val)[0]

        # sub‑gradient g_i = a_{k‑i} (factor 2 is absorbed in α)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_range
        g = np.zeros_like(a)
        g[idx_range] = a[rev_idx]

        # scheduled step size
        alpha = alpha0 * (decay ** (it // 1000))

        # multiplicative update (mirror‑descent on the simplex)
        a = a * np.exp(-alpha * g)

        # projection onto the simplex (renormalise)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # periodic logging & best‑track
        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[subgrad iter {it}] new best = {best_val:.6f} (max conv {max_val:.8f})")
            if it % 2000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[subgrad reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (targeted mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random local search biased towards the heaviest contributors to the
    current maximal convolution entry.  Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        k = np.argmax(conv)

        # indices that appear in this maximal shift
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_range
        contrib = cur_a[idx_range] * cur_a[rev_idx]

        # pick a heavy‑contributing index (probability ∝ contribution)
        if contrib.sum() == 0:
            i = rng.integers(i_start, i_end + 1)
        else:
            probs = contrib / contrib.sum()
            i = rng.choice(idx_range, p=probs)

        j = k - i            # the partner index in the max‑shift pair
        donor = i if cur_a[i] > cur_a[j] else j
        receiver = rng.integers(0, n)
        if donor == receiver:
            continue

        delta = cur_a[donor] * 0.05 * rng.random()
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[receiver] += delta

        # re‑normalise onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement required
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin before returning
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size – a longer vector gives more freedom.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------

    # --------------------------------------------------------------

        if old.shape[0] != n:

            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)          # uniform start

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # --------------------------------------------------------------
    # Evaluate the initial candidate
    # --------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – a few random Dirichlet / exponential vectors
    # --------------------------------------------------------------
    for tr in range(6):
        if time.time() >= deadline - 200:
            break
        # Dirichlet with concentration < 1 yields a very non‑uniform shape
        alpha = np.full(n, 0.5)
        a_rand = rng.gamma(alpha, 1.0)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS optimisation of the smooth surrogate
    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 150 and restart < 12:
        restart += 1
        w_start = np.log(best_a + 1e-12)
        w_start += rng.normal(scale=0.4, size=n)   # moderate perturbation

        a_opt, val_opt = _l_bfgs_optimize(w_start, deadline, max_iter=2000)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[restart {restart}] new best after L‑BFGS = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – sub‑gradient mirror descent focusing on the maximal shift
    # --------------------------------------------------------------
    if time.time() < deadline - 40:
        a_sub, val_sub = _max_subgradient_descent(best_a, deadline)
        if val_sub < best_val:
            best_val = val_sub
            best_a = a_sub.copy()
            print(f"[subgradient descent] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # --------------------------------------------------------------
    # Final evaluation & return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
