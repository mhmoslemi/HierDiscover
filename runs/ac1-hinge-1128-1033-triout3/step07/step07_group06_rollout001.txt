"""
Soft‑max gradient descent + occasional hill‑climbing for the
step‑function “attack”.

The objective to minimise is

    F(a) = 2·n·max_k (a * a)[k]          (a is non‑negative, sum(a)=1)

We approximate the max with a smooth soft‑max:
    G_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])

For a fixed temperature β the gradient  ∇G_β can be written in closed form
using one extra convolution, which we evaluate with an FFT.
A decreasing step size (∝ 1/√t) together with a projection onto the
probability simplex yields a fast first‑order method that explores
different mass distributions.  Every few hundred iterations a
“mass‑transfer” hill‑climber (the same routine used in the previous
baseline) is invoked to polish the current best candidate.

The routine respects the supplied time budget (default 1000 s) and
prints intermediate progress.  The best sequence found is returned as a
list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT.
    Length of result: 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # eliminate tiny negative round‑off errors
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of the public evaluator when sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Returns (grad, conv, p) where

        grad = ∇  (1/β)·log Σ_k exp(β·conv[k])
        conv = a*a (full convolution)
        p    = soft‑max probabilities over conv (sum to 1)

    The gradient is obtained by a second convolution (cross‑correlation)
    performed with FFTs.
    """
    n = a.shape[0]

    # 1) convolution a*a
    conv = _full_convolution(a)                     # length 2n‑1

    # 2) soft‑max probabilities
    scaled = conv * beta
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / exp_scaled.sum()               # length 2n‑1, sum to 1

    # 3) gradient: 2 * (p ⋆ a)   (cross‑correlation)
    #    (p ⋆ a)[i] = Σ_j p[i+j]·a[j]  , i = 0…n‑1
    size = _next_pow2(len(p) + n - 1)               # length for FFT
    fp = np.fft.rfft(p, size)
    fa_rev = np.fft.rfft(a[::-1], size)             # a reversed
    conv_full = np.fft.irfft(fp * fa_rev, size)[: len(p) + n - 1]
    grad = 2.0 * conv_full[n - 1: n - 1 + n]        # slice length n
    return grad, conv, p


def _project_to_simplex(x: np.ndarray) -> np.ndarray:
    """Simplex projection: returns y≥0 with sum(y)=1 that is closest to x."""
    # Implements the algorithm from Wang & Carreira‑Perpiñán (2013)
    if x.sum() == 1.0 and np.all(x >= 0):
        return x
    n = x.shape[0]
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform distribution
        return np.full_like(x, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(x - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Hill‑climbing phase (same idea as the baseline code)
# ----------------------------------------------------------------------


def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 3000):
    """
    Aggressive mass‑transfer hill‑climb.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that contribute to the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we try to relocate
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # pick two low‑mass destinations (different from i, j)
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.

    The algorithm consists of:
      1. Initialisation (reuse previous construction if available).
      2. Soft‑max gradient descent with a temperature schedule.
      3. Periodic mass‑transfer hill‑climbing for fine‑grained polishing.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise – reuse a previous construction if it exists.
    # ------------------------------------------------------------------
    default_n = 4000
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        a0 = np.maximum(old, 0.0)
        if a0.sum() == 0.0:
            a0 = rng.random(n)
        a0 /= a0.sum()
    except Exception:
        n = default_n
        a0 = rng.random(n)
        a0 /= a0.sum()

    print(f"[init] n={n}, initial objective = {_objective(a0):.6f}")

    # ------------------------------------------------------------------
    # 2. Soft‑max gradient descent with temperature annealing.
    # ------------------------------------------------------------------
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    # Adam‑style moments (optional – they help stabilise the steps)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    iter_count = 0
    # temperature schedule: start small, grow geometrically up to ~200
    beta = 1.0
    beta_max = 200.0
    beta_mul = 1.001  # slowly increase

    # step‑size base (will be scaled by 1/√t)
    base_lr = 0.5

    # target that we would like to reach – stop early if achieved
    target_val = 1.503

    while time.time() < deadline:
        iter_count += 1

        # ----- temperature annealing -----
        beta = min(beta_max, beta * beta_mul)

        # ----- compute gradient of soft‑max approximation -----
        grad, conv, p = _softmax_gradient(a, beta)

        # ----- Adam‑style adaptive step -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iter_count)
        v_hat = v / (1.0 - beta2 ** iter_count)

        lr = base_lr / np.sqrt(iter_count)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            # fallback – tiny random perturbation
            a_new = np.maximum(a + rng.normal(scale=1e-6, size=n), 0.0)
            a_new /= a_new.sum()
        else:
            a_new /= s

        val = _objective(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[softmax] iter {iter_count:5d}  best = {best_val:.6f}")
            if best_val < target_val:
                print("[softmax] target reached – exiting early")
                a = a_new
                break

        # always move forward (even if not improved)
        a = a_new

        # ------------------------------------------------------------------
        # 3. Every few hundred iterations run the hill‑climber to polish.
        # ------------------------------------------------------------------
        if iter_count % 400 == 0:
            a_hc, val_hc = _mass_transfer_hillclimb(best_a, deadline, rng,
                                                    max_no_improve=800)
            if val_hc < best_val:
                best_val = val_hc
                best_a = a_hc.copy()
                a = a_hc.copy()
                print(f"[mix] after hillclimb iter {iter_count:5d}  best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final evaluation via the public evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:    # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
