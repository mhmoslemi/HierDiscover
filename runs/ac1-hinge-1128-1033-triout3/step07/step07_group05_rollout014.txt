def _continuous_sa(a, cur_val, deadline):
    n = len(a)
    rng = np.random.default_rng()
    best = a.copy()
    best_val = cur_val
    T = max(cur_val * 0.1, 1e-6)
    decay = 0.9999
    while time.time() < deadline:
        # pick move type: 0 - swap, 1 - mass transfer
        if rng.random() < 0.3:
            # swap
            i,j = rng.integers(0,n,size=2)
            if i==j: continue
            a[i], a[j] = a[j], a[i]
            new_val = _eval_fast(a)
            # revert if needed later
            revert = lambda: (a.__setitem__(i, a[i]), a.__setitem__(j, a[j]))
        else:
            i,j = rng.integers(0,n,size=2)
            if i==j: continue
            # transfer
            delta = rng.random() * a[i] * 0.5  # up to 50% of a[i]
            a[i] -= delta
            a[j] += delta
            new_val = _eval_fast(a)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
        else:
            # revert
            if move type swap: a[i],a[j]=a[j],a[i]
            else: a[i] += delta; a[j] -= delta
        T *= decay
        if T < 1e-12: T = max(cur_val * 0.01, 1e-12)
    return best, best_val

"""Improved optimizer for the step‑function evaluation problem.

The method consists of three phases:

1. **Smooth surrogate optimisation** – minimise a log‑sum‑exp
   approximation of the max‑convolution via L‑BFGS, using a finer
   schedule of the smoothing parameter and a larger iteration budget.
   Multiple random restarts are performed.

2. **Projected sub‑gradient descent** – directly minimise the true
   max‑convolution on the simplex using a simple sub‑gradient scheme
   with a step‑size that decays as 1/t.  Several restarts are tried.

3. **Continuous simulated‑annealing** – a Metropolis search that
   transfers small amounts of mass between arbitrary positions and
   also swaps values.  This phase can accept uphill moves, helping the
   algorithm escape local minima.

All phases respect a global time budget (default 1000 s).  The best
candidate found is finally evaluated with the official
``evaluate_sequence`` function and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate wrapper
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter
_GLOBAL_N = None     # length of the sequence

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for efficient FFT)."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of ``a`` with itself using FFT.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Clip tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` – identical value but without
    the safety checks.  The routine works with a normalised vector
    (ℓ₁ = 1) so the score reduces to ``2·n·max(conv)``.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    # The original evaluator uses 2·n·max_b / sum_a².
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) – required by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (obj, grad) for the smoothed objective

        L(w) = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε),

    where ``conv`` is the self‑convolution of
    a = softmax(w) (so Σa = 1).  ``grad`` is the gradient w.r.t.
    the logits w.
    """
    # ---------- soft‑max (positivity & ℓ₁ = 1) ----------
    w_shift = w - np.max(w)      # improve numerical stability
    a = np.exp(w_shift)
    a /= np.sum(a)                # Σa = 1

    # ---------- convolution ----------
    conv = _fft_convolve_self(a)
    max_conv = conv.max()

    # ---------- smoothed approximation ----------
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)           # log ∑ exp((c−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---------- gradient ----------
    weights = np.exp(shifted)
    p = weights / np.sum(weights)          # probability distribution over k

    # g_a = 4·n·∑ₖ pₖ·a_{k−i}   (reversed implementation)
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]           # ∂L/∂a

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# Bind a helper convolution to the function object (so the wrapper can
# see it without closures).
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


_smooth_obj_and_grad._conv_fft = _conv_fft


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize`` (globals supplied)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Simple heuristics for ordering a multiset of heights
# ----------------------------------------------------------------------
def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic re‑ordering that places the sorted values in an
    alternating order: largest, smallest, second largest, second
    smallest, …  This spreads large values throughout the sequence.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)              # ascending
    out = np.empty_like(a)
    left = 0
    right = n - 1
    i = 0
    while i < n:
        out[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        out[i] = sorted_vals[left]
        left += 1
        i += 1
    return out


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Simple sub‑gradient algorithm for ``max(a*a)`` on the ℓ₁‑simplex.
    The step size is ``α_t = η / sqrt(t)`` with a modest initial
    learning‑rate η.  Several restarts are performed implicitly by
    invoking this routine repeatedly from different starting points.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_val = _eval_fast(cur)
    best = cur.copy()
    best_val = cur_val
    eta = 0.6                     # larger than the previous 0.12
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the max entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

            cur_val = nxt_val

"""Improved optimizer for the step‑function evaluation problem.

The method consists of three phases:

1. **Smooth surrogate optimisation** – minimise a log‑sum‑exp
   approximation of the max‑convolution via L‑BFGS, using a finer
   schedule of the smoothing parameter and a larger iteration budget.
   Multiple random restarts are performed.

2. **Projected sub‑gradient descent** – directly minimise the true
   max‑convolution on the ℓ₁‑simplex using a simple sub‑gradient scheme
   with a step‑size that decays as 1/t.  Several restarts are tried.

3. **Continuous simulated‑annealing** – a Metropolis search that
   transfers small amounts of mass between arbitrary positions and
   also swaps values.  This phase can accept uphill moves, helping the
   algorithm escape local minima.

All phases respect a global time budget (default 1000 s).  The best
candidate found is finally evaluated with the official ``evaluate_sequence``
function and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate wrapper
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter
_GLOBAL_N = None     # length of the sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding in FFT)."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Clip rounding artefacts
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast evaluator identical to ``evaluate_sequence`` but without
    safety checks.  The function assumes a non‑negative vector ``a``.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    # Original formula: 2 n max_b / sum_a²
    return float(2.0 * n * max_b / (sum_a * sum_a))

# Smoothed surrogate (log‑sum‑exp) – required by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad) where

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and ``grad`` is the gradient w.r.t. the logit vector ``w``.
    The underlying coefficient vector is ``a = softmax(w)`` (Σa = 1).
    """
    # ----- soft‑max (positivity & ℓ₁ = 1) -----
    w_shift = w - np.max(w)            # numeric stability
    a = np.exp(w_shift)
    a /= np.sum(a)                     # Σa = 1

    # ----- convolution a * a -----
    conv = _fft_convolve_self(a)
    max_conv = conv.max()

    # ----- smooth approximation of the max -----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)       # log ∑ exp((c−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    weights = np.exp(shifted)
    p = weights / np.sum(weights)      # probability distribution over positions
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]       # ∂L/∂a

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv
