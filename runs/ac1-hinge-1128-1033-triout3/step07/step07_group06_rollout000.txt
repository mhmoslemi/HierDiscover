"""
Optimiser for the step‑function attack.

The code consists of three main stages:

1️⃣  **Initialisation** – reuse a previously‑found sequence if it exists,
    otherwise start from a random Dirichlet (probability simplex).

2️⃣  **Smoothed Adam optimisation** – minimise a smooth approximation of
    the objective  f(a)=2·n·max(conv(a,a)) .
    The max is replaced by a soft‑max (log‑sum‑exp) with a temperature that
    is gradually increased.  Gradients are computed analytically via
    FFT‑based convolutions and the parameters are updated with Adam.
    After each step the vector is projected back onto the probability
    simplex (non‑negative, sum = 1).

3️⃣  **Aggressive mass‑transfer hill‑climbing** – a local‑search
    refinement that moves a fraction of the mass from the pair of indices
    that contributes most to the maximal convolution entry to two
    low‑mass positions.  The move is kept only if the true objective
    improves.  This stage is identical to the proven “mass‑transfer”
    heuristic but starts from the (typically) better Adam solution.

Both stages respect the 1000 s time budget.  The best sequence found is
returned as a plain Python list; the surrounding harness will call
``evaluate_sequence`` on it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT based convolutions, objective, simplex projection)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using an FFT (returns length len(x)+len(y)-1)."""
    size = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[: len(x) + len(y) - 1]


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)−1)."""
    return _full_convolution(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * np.max(_self_convolution(a))


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán, 2013.
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (stage 1)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   beta_start: float = 2000.0,
                   beta_target: float = 20000.0,
                   beta_mult: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth approximation of the max‑convolution.
    Returns the best vector encountered and its true (non‑smoothed) objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    beta = beta_start
    no_improve_counter = 0

    while time.time() < deadline:
        it += 1

        # --- exact convolution (for the true objective) -----------------
        conv = _self_convolution(a)
        max_conv = conv.max()

        # --- smooth weights (soft‑max) ---------------------------------
        # Numerical stability: shift by the maximal entry.
        exp_arg = beta * (conv - max_conv)
        # Clip to avoid overflow for extreme negatives (not strictly necessary)
        exp_arg = np.clip(exp_arg, -700, 0)    # np.exp(-700) ≈ 5e-305
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback – should never happen, but protect against numerical issues
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the smoothed objective -------------------------
        # grad = 4·n·(w * a_rev) convolved, slice n entries.
        rev_a = a[::-1]
        conv_grad_full = _full_convolution(w, rev_a)          # length = 3·n‑2
        grad = conv_grad_full[n - 1 : n - 1 + n]               # take the middle n entries
        grad = 4.0 * n * grad                                 # chain rule factor

        # --- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # --- projection onto the simplex --------------------------------
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a == 0.0:
            # re‑initialise if we accidentally zeroed everything
            a = rng.random(n)
            a /= a.sum()
        else:
            a /= sum_a

        # --- evaluate the true (non‑smoothed) objective -----------------
        conv_new = _self_convolution(a)
        val = 2.0 * n * conv_new.max()

        if val < best_val:
            best_val = val
            best_a = a.copy()
            no_improve_counter = 0
            print(f"[adam] iter {it:5d} – new best = {best_val:.6f}")
        else:
            no_improve_counter += 1

        # --- slowly tighten the soft‑max (increase beta) ---------------
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mult)

        # Optional early‑stop if stuck for an extremely long time
        if no_improve_counter > 5000:
            # inject a tiny random jitter to try to escape a flat region
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = _project_to_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Mass‑transfer hill‑climbing (stage 2 – unchanged from the reference)
# ----------------------------------------------------------------------


def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing phase that repeatedly moves a fraction of the most
    damaging mass to low‑mass positions.
    Stops after the time budget expires or after `max_no_improve`
    consecutive unsuccessful attempts.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2                # initial fraction of mass to relocate
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _self_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that make up the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we are willing to shift
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # pick two distinct low‑mass destinations, avoiding the donors
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val
