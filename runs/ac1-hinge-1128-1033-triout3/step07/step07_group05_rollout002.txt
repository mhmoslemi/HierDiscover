"""
Propose a step‑function with a low self‑convolution peak.

Algorithm
---------

1. **LP‑based shape optimisation (small problem).**  
   For a moderate length `n_lp` (≈2000) we repeatedly:
   * compute the current peak `M = max(a*a)`;
   * solve a linear programme  
        maximise   Σ b_i
        s.t.       (a * b)_k ≤ M   for every convolution index k
                   b_i ≥ 0
     The constraint matrix is exactly the Toeplitz matrix that
     encodes the convolution with the current `a`.  The LP seeks a
     distribution `b` that can be placed under the same peak `M` but
     with a larger ℓ₁‑norm.  Moving the current vector towards that
     `b` strictly reduces the peak (standard “majorisation” step).

2. **Upscaling.**  
   The best vector from the LP phase is interpolated to a larger
   size `n_target` (≈4000).  The shape obtained on the small scale
   transfers well to the larger scale.

3. **Monotone ends re‑ordering.**  
   The multiset of heights is placed so that the largest values sit
   at the two ends of the sequence, the second largest at the opposite
   end, etc.  This ordering is known to diminish the worst convolution
   entry.

4. **Projected sub‑gradient descent on the exact max‑convolution.**  
   Starting from the reordered vector we repeatedly locate the index
   `k_max` where `a*a` is maximal and move against the sub‑gradient
   `∇_i max(a*a) = 2·a_{k_max-i}` (zero when the index is out of
   range).  After each step we project back onto the probability simplex
   (non‑negative and ℓ₁‑norm = 1).  A diminishing step size
   `α_t = α₀/√t` guarantees stability.

5. **Final permutation fine‑tuning.**  
   A short simulated‑annealing run that swaps two positions,
   occasionally accepting uphill moves, followed by a cheap hill‑climb
   (swaps + tiny mass transfers) removes remaining local irregularities.

All phases respect the global 1000 s wall‑time budget.  The routine
returns a plain Python list, ready for `evaluate_sequence`.
"""

import time
import numpy as np
import scipy.linalg
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution / objective (identical to evaluate_sequence but without checks)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (real inputs)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Clip tiny negative values caused by floating‑point noise
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence(a.tolist()) without safety checks."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Re‑ordering: largest values at the two ends, then second largest opposite,
# etc. (monotone ends ordering)
# ----------------------------------------------------------------------
def _ends_monotone_reorder(a: np.ndarray) -> np.ndarray:
    """Place values in descending order alternately at the left and right ends."""
    n = a.shape[0]
    # descending sort
    sorted_vals = np.sort(a)[::-1]
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    turn = 0  # 0 → fill left, 1 → fill right
    for v in sorted_vals:
        if turn == 0:
            new_a[left] = v
            left += 1
        else:
            new_a[right] = v
            right -= 1
        turn ^= 1
    return new_a


# ----------------------------------------------------------------------
# Linear‑programming refinement (small problem)
# ----------------------------------------------------------------------
def _solve_lp_max_sum_under_peak(f: np.ndarray, M: float):
    """
    Solve
        maximise   Σ b_i
        s.t.       (f * b)_k ≤ M   for all k
                   b_i ≥ 0
    Returns the optimal vector `b` (may be None if LP fails).
    """
    n = f.shape[0]
    # Build Toeplitz matrix that implements the convolution (f * b)
    # First column: f followed by (n-1) zeros.
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # First row: f[0] followed by zeros (size n)
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    A = scipy.linalg.toeplitz(col, row)          # shape (2n‑1, n) – dense, but modest size
    b_ub = np.full(2 * n - 1, M, dtype=f.dtype)

    # Linear programme: minimise -Σb   ↔   maximise Σb
    c = -np.ones(n, dtype=f.dtype)

    # Bounds: b_i ≥ 0, no explicit upper bound
    bounds = [(0, None)] * n

    # Solve with HiGHS (highly efficient)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        return None


def _lp_refine(a: np.ndarray, deadline: float):
    """
    Repeated LP‑based improvement of vector `a` (which is kept on the simplex).
    Returns the best vector found before the deadline.
    """
    best_a = a.copy()
    best_val = _eval_fast(best_a)
    it = 0
    while time.time() < deadline:
        # Current peak
        conv = _fft_convolve_self(best_a)
        M = np.max(conv)

        # Solve LP
        b = _solve_lp_max_sum_under_peak(best_a, M)
        if b is None:
            print("[LP] solver failed – exiting LP phase")
            break

        sum_b = b.sum()
        if sum_b <= 0:
            break
        # Normalise to keep ℓ₁ = 1 (same normalisation as the objective)
        b_norm = b / sum_b

        # Move partially towards b_norm (step size 0.5 works well)
        t = 0.5
        cand = (1.0 - t) * best_a + t * b_norm
        cand /= cand.sum()                     # ensure ℓ₁ = 1
        cand_val = _eval_fast(cand)

        if cand_val < best_val:
            best_val = cand_val
            best_a = cand
            print(f"[LP] iter {it} new best = {best_val:.6f}")

        # If no improvement after a few tries, stop early
        if it >= 12:
            # limit number of LP solves – they are the most expensive step
            break
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float):
    """Sub‑gradient descent on max(a*a) with simplex projection."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val
    step0 = 0.25
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur_a)
        k_max = int(np.argmax(conv))

        # gradient of the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(cur_a)
        grad[mask] = 2.0 * cur_a[idx[mask]]

        alpha = step0 / np.sqrt(it + 1.0)
        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            new_a = np.full(n, 1.0 / n)
            s = 1.0
        new_a /= s

        new_val = _eval_fast(new_a)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it} new best = {best_val:.6f}")

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing on permutations (swap moves only)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float, deadline: float):
    """Swap‑only simulated annealing that tries to lower the objective."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # Initial temperature as a small fraction of the current value
    T = max(cur_val * 0.02, 1e-8)
    decay = 0.99995

    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]          # propose swap
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] iter {it} new best = {best_val:.6f}")
        else:
            a[i], a[j] = a[j], a[i]      # revert

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)
        it += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (swap + tiny mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float, deadline: float):
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.25:           # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                             # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.04 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best list of floats found within the allotted wall‑time.
    """
    np.random.seed(seed)
    start = time.time()
    # Keep a small safety margin for the final evaluation call
    deadline = start + budget_s - 1.5
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # --------------------------------------------------------------
    # Phase 0 – initialise a reasonable start (previous result if any)
    # --------------------------------------------------------------
    n_lp = 2000                           # length for the LP phase
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n_lp:
            # linear interpolation to required size
            x_old = np.linspace(0, 1, old.shape[0])
            x_new = np.linspace(0, 1, n_lp)
            a0 = np.interp(x_new, x_old, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction for LP phase")
    except Exception:
        a0 = np.full(n_lp, 1.0 / n_lp)
        print("[init] using uniform start for LP phase")

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()                         # normalise to ℓ₁ = 1

    # --------------------------------------------------------------
    # Phase 1 – LP refinement (small scale)
    # --------------------------------------------------------------
    lp_deadline = deadline - 600.0          # reserve ~10 min for later phases
    if time.time() < lp_deadline:
        print("[LP] starting linear‑programming refinement")
        a_lp, val_lp = _lp_refine(a0, lp_deadline)
    else:
        a_lp, val_lp = a0, _eval_fast(a0)
        print("[LP] skipped – insufficient time")

    print(f"[LP] finished with best value {val_lp:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Upsample to a larger problem size
    # --------------------------------------------------------------
    n_target = 4000
    x_src = np.linspace(0, 1, a_lp.shape[0])
    x_tgt = np.linspace(0, 1, n_target)
    a_big = np.interp(x_tgt, x_src, a_lp)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    print(f"[upsample] upsampled to n = {n_target}")

    # --------------------------------------------------------------
    # Phase 3 – monotone ends re‑ordering
    # --------------------------------------------------------------
    a_big = _ends_monotone_reorder(a_big)
    a_big /= a_big.sum()
    best_a = a_big
    best_val = _eval_fast(best_a)
    print(f"[order] after ends‑ordering value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – projected sub‑gradient descent (exact objective)
    # --------------------------------------------------------------
    sub_deadline = deadline - 120.0
    if time.time() < sub_deadline:
        print("[subgrad] starting sub‑gradient descent")
        best_a, best_val = _subgradient_descent(best_a, sub_deadline)
        print(f"[subgrad] finished with best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – short simulated‑annealing on permutations
    # --------------------------------------------------------------
    sa_deadline = deadline - 30.0
    if time.time() < sa_deadline:
        print("[SA] final simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val, sa_deadline)

    # --------------------------------------------------------------
    # Phase 6 – final cheap hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report final score using the official evaluator (safety‑checked)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required
    return best_a.tolist()
