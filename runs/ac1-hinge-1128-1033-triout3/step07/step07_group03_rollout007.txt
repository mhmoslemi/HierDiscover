    n = a.shape[0]
    conv_len = 2*n -1
    p = np.zeros(conv_len, dtype=float)
    p[idxs] = 1.0

def _gd_adaptive(a_start, deadline):
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    it = 0
    alpha0 = 0.5
    decay_factor = 0.98
    rel_tol = 0.001  # 0.1% tolerance
    min_peaks = 5

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()
        # Determine indices near max
        threshold = max_conv * (1 - rel_tol)
        idxs = np.where(conv >= threshold)[0]
        if idxs.size < min_peaks:
            # take top min_peaks by partition
            k = min_peaks
            if k < conv.size:
                idxs = np.argpartition(-conv, k-1)[:k]
            else:
                idxs = np.arange(conv.size)
        # Build mask p
        p = np.zeros_like(conv)
        p[idxs] = 1.0
        # Compute gradient g using fft
        p_rev = p[::-1]
        size = _next_pow2(n + p_rev.shape[0] - 1)
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p_rev, size)
        conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
        seg = conv_ap[n - 1: 2 * n - 1]
        g = seg[::-1]  # gradient contributions

        # step size
        alpha = alpha0 * (decay_factor ** (it // 2000))

        # multiplicative update with clipping
        delta = -alpha * g
        # clip to avoid overflow:
        delta = np.clip(delta, -20, 20)
        a = a * np.exp(delta)  # because delta negative for reduction? We want a * exp(-alpha*g). g positive reduces.
        # Norm onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd_adaptive iter {it}] new best = {best_val:.6f} (max conv {max_conv:.8e})")
            # optionally reorder
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd_adaptive reorder] improved to {best_val:.6f}")
        it += 1
    return best_a, best_val

"""
Propose a non‑negative step function minimizing `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – start from the historic construction
   (`height_sequence_1`) if it exists, otherwise draw a random Dirichlet
   vector and normalise it to sum 1.

2. **Adaptive exponential‑gradient descent** – at each iteration compute the
   full self‑convolution `c = a * a`.  All peaks whose value is within a small
   relative tolerance of the global maximum are collected.  The gradient of the
   sum of those selected peaks w.r.t. the coefficients can be written as a
   convolution; it is evaluated with a single FFT.  The coefficients are
   updated multiplicatively (`a ← a·exp(−α·g)`) and re‑normalised, which keeps
   them non‑negative and on the simplex.  The step size `α` decays slowly.

   The FFT‑based gradient makes it cheap to treat many peaks simultaneously,
   allowing a more aggressive reduction of the maximal convolution value
   than the original ``top‑k`` approach.

3. **Anti‑symmetric re‑ordering** – after a successful improvement the vector
   is re‑ordered by pairing the largest values with the smallest ones (the
   rearrangement inequality guarantees a lower central autocorrelation).

4. **Hill‑climbing refinement** – a lightweight local search (random swaps and
   small mass transfers) finishes the optimisation.

All heavy work (the many gradient iterations) is performed without calling
`evaluate_sequence`; the true objective is queried only sparingly for logging
and to keep the current best solution.  The whole procedure respects the
1000 s time budget and returns the best list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # numerical noise may produce tiny negatives – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Gradient of the sum of selected convolution peaks (FFT‑based)
# ----------------------------------------------------------------------
def _gradient_selected_peaks(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈idxs} a_{k-i}   (indices out of range give zero).

    The computation is done via a single FFT:
        p[k] = 1 if k∈idxs else 0,
        g = reverse( (a * p[::-1])[n-1 : 2n-1] ).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1

    # indicator vector for the selected peaks
    p = np.zeros(conv_len, dtype=float)
    p[idxs] = 1.0

    # convolution a * reverse(p)  (full length = n + conv_len - 1 = 3n-2)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to Σ a_{k-i}
    seg = conv_ap[n - 1: 2 * n - 1]
    g = seg[::-1]       # reverse because of the reversal of p
    return g


# ----------------------------------------------------------------------
# Adaptive exponential‑gradient descent
# ----------------------------------------------------------------------
def _adaptive_exp_gradient(a_start: np.ndarray, deadline: float):
    """
    Exponential mirror‑descent that reduces *all* convolution peaks
    within a relative tolerance of the current maximum.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[gd start] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.5               # initial step size
    decay_factor = 0.98        # halve alpha every ~2000 iterations
    rel_tol = 1e-3             # consider peaks within 0.1 % of the max
    min_peaks = 5              # guarantee at least a few peaks

    while time.time() < deadline - 5.0:
        # full self‑convolution
        conv = _full_convolution(a)
        max_conv = conv.max()

        # indices of peaks close to the maximum
        thresh = max_conv * (1.0 - rel_tol)
        idxs = np.where(conv >= thresh)[0]

        # ensure we have a minimum number of peaks (helps when the max is isolated)
        if idxs.size < min_peaks:
            k = min_peaks
            if k < conv.size:
                idxs = np.argpartition(-conv, k - 1)[:k]
            else:
                idxs = np.arange(conv.size)

        # gradient for the selected peaks
        g = _gradient_selected_peaks(a, idxs)

        # step‑size schedule (decays every 2000 iterations)
        alpha = alpha0 * (decay_factor ** (it // 2000))

        # multiplicative update, clipped to avoid overflow/underflow
        delta = -alpha * g
        delta = np.clip(delta, -20.0, 20.0)      # exp(±20) ≈ 4.8e8 – safe range
        a = a * np.exp(delta)

        # projection onto the probability simplex
        a = np.maximum(a, 1e-15)                # keep positivity
        a /= a.sum()

        # occasional logging and optional anti‑symmetric re‑ordering
        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd iter {it}] new best = {best_val:.6f} (max conv {max_conv:.8e})")
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """Random swaps and tiny mass transfers, keeping only strict improvements."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement required
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand coefficients give a good trade‑off
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise from a known construction if possible
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = rng.exponential(scale=1.0, size=n)

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    # --------------------------------------------------------------
    # Evaluate the initial candidate
    # --------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – a few random restarts (to avoid a poor start)
    # --------------------------------------------------------------
    for tr in range(4):
        if time.time() >= deadline - 200:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – adaptive exponential‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 150:
        a_gd, val_gd = _adaptive_exp_gradient(best_a, deadline)
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[adaptive gradient] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – final hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline, n)

    # --------------------------------------------------------------
    # Final evaluation & return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
