"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Initial pool** – generate a handful of diverse probability vectors
   (uniform, previous best, Dirichlet draws, sinusoid).  Keep the best
   according to a fast surrogate ``2·n·max(conv)``.

2. **Smooth L‑BFGS** – optimise a smooth log‑sum‑exp surrogate of the
   max‑convolution using a soft‑max re‑parameterisation.  The smoothing
   parameter ε is decreased on a log‑scale; many restarts (randomising the
   soft‑max parameters) increase the chance of escaping local minima.
   After each restart an *anti‑symmetric* reordering (large ↔ small)
   attempts to flatten the autocorrelation peak.

3. **Exact mirror‑descent** – sub‑gradient descent on the exact
   ``max(conv)`` using the known gradient (g_i = a_{k*‑i}) and an
   exponential mirror map, with a slowly decaying step‑size.

4. **Simulated‑annealing (mass‑transfer & swaps)** – a long SA phase that
   moves a fraction of the mass from one index to another (preserving the
   simplex) and also performs random swaps.  Accepts uphill moves with
   temperature T that is exponentially cooled.

5. **Permutation annealing** – the values of the current best vector are
   shuffled by swaps while the multiset of values stays unchanged.
   This directly searches for a better ordering (the “shape” of the
   distribution is already good, only the placement matters).  The same
   SA acceptance rule is used.

6. **Cheap hill‑climb** – final polishing using very cheap local moves
   (swap / mass‑transfer) that only accept strict improvements.

All phases respect the total time budget (default 1000 s).  The routine
returns a plain Python list that can be passed to the provided
``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS surrogate (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None
_GLOBAL_N = None

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    The result has length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may create tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast surrogate for ``evaluate_sequence`` when a is non‑negative and
    sums to 1.  Returns 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ---------- soft‑max ----------
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---------- convolution ----------
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ---------- smooth max ----------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    weights = np.exp(shifted)
    p = weights / weights.sum()      # probability distribution over lag indices

    # Convolution of a with reversed p gives ∂max/∂a (up to a factor 2)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]   # the part that aligns with a_i
    grad_a = 4.0 * n * seg[::-1]     # factor 4 arises from the two symmetric terms

    # ---------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place large values on the left and small values on the right.
    This often reduces the central autocorrelation peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector stays on the
    simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def _perm_anneal(a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that only permutes the entries of *a*.
    The multiset of values stays unchanged, so the simplex constraint is
    automatically satisfied.  Accepts uphill moves with temperature *T*.
    """
    start = time.time()
    deadline = start + time_limit
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = _fast_evaluate(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    T = 0.02                # initial temperature
    cooling = 0.99997

    while time.time() < deadline:
        # propose a simple swap
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        new_a = cur_a.copy()
        new_a[i], new_a[j] = new_a[j], new_a[i]

        new_val = _fast_evaluate(new_a)
        d = new_val - cur_val
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[perm‑anneal] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    return best_a, best_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the allotted time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # reserve a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – use the length of the known construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # Generate a few diverse starting points.
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # random Dirichlet (α = 1)
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))

    # random Dirichlet (α = 5)
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoidal shape
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # normalise everything onto the simplex
    init_candidates = [c / c.sum() for c in init_candidates]

    # ------------------------------------------------------------------
    # Pick the best of the cheap initial candidates.
    # ------------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on a smooth surrogate of the max‑convolution.
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=14)
    # start from log‑soft‑max of the current best
    w = np.log(best_a + 1e-12)

    max_restarts = 16
    restart = 0
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        if restart % 4 == 0:
            w = np.random.randn(n) * 2.0   # fresh random start

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 600,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x
            # convert back to probabilities via soft‑max
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # optional anti‑symmetric reorder after each restart
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent on the exact max‑convolution.
    # ------------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.5          # initial step size
    decay = 0.9995        # geometric decay
    min_alpha = 0.005
    it = 0
    while time.time() < deadline - 20.0:
        conv = _full_convolution(a)
        k_star = int(conv.argmax())

        # gradient of max(conv) w.r.t. a:
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        partner = k_star - idx
        g[idx] = a[partner]                     # = a_{k*-i}

        # exponential mirror‑descent update
        alpha = max(alpha0 * (decay ** (it // 200)), min_alpha)
        a = a * np.exp(-alpha * g)

        # re‑project onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 100 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[mirror it {it}] new best = {best_val:.6f}")
            if it % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[mirror reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing with swaps & mass transfers.
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.02                # initial temperature for SA
    cooling = 0.99997
    rng = np.random.default_rng(seed + 12345)

    while time.time() < deadline - 12.0:
        if rng.random() < 0.55:          # mass transfer (slightly favoured)
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.09 * rng.random()   # up to ~9 % of a_i
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:                             # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        # sum is unchanged, only need non‑negativity check
        if np.min(new_a) < -1e-12:
            continue

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # Phase 4 – permutation annealing (only re‑ordering of the current best).
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 5.0:
        perm_best, perm_val = _perm_anneal(best_a, time_limit=remaining_time - 2.0)
        if perm_val < best_val:
            best_val = perm_val
            best_a = perm_best
            print(f"[perm‑anneal final] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final cheap hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (safety check) and return.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
