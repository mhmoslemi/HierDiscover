def _lp_max_sum(f, M):
    n = f.shape[0]
    offsets = np.arange(n)
    data = np.broadcast_to(f[:, None], (n, n)).copy()
    T = scipy.sparse.diags(data, offsets, shape=(2*n-1, n), format='csc')
    c = -np.ones(n)  # minimize -sum
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=T, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        print("[LP] failed:", res.message)
        return None
```

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Initialisation** – start from the best known construction
   (`height_sequence_1` if it exists) or from a uniform vector.
2. **Fast L‑BFGS refinement** – run a few L‑BFGS steps on the smooth
   log‑sum‑exp surrogate of the max‑convolution (the same code that was
   used in the previous baseline).  This gives a decent local optimum.
3. **Linear‑programming “mass‑inflation” step** – for the current vector
   *f* compute *M = max(f * f)*.  Solve the LP

        max   Σ b_i
        s.t.  (f * b)_k ≤ M   for all k
              b_i ≥ 0

   (the convolution matrix is built explicitly; the problem size for the
   default n≈4000 is modest for the HiGHS solver).  The optimal *b* has a
   larger ℓ₁‑norm than *f*.  After normalising
   `g = b / Σb` we either replace *f* by *g* (if it immediately improves
   the objective) or take a convex combination `f ← (1‑t)·f + t·g`
   (default *t* = 0.2).  Re‑evaluate *M* and repeat the LP step while
   time permits.
4. **Cheap hill‑climbing** – final random swaps / mass‑transfers that
   accept only strict improvements.
5. **Return** the best vector found (as a plain Python list).  All
   vectors are kept on the probability simplex (sum = 1), which makes
   the evaluation `2·n·max(conv)` identical to the official
   ``evaluate_sequence`` value (scale‑invariant).

The whole routine respects a 1000 s budget; if the time expires it
simply returns the best vector obtained so far.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (useful for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT – returns length 2·n‑1 array."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast surrogate of evaluate_sequence for vectors that already sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the original
    formula simplifies).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _build_conv_matrix(f: np.ndarray) -> scipy.sparse.csc_matrix:
    """
    Build the (2n‑1)×n convolution matrix T such that
    (T @ b) == (f * b)  (full linear convolution).
    Entry T[k, i] = f[i] if 0 ≤ k-i < n else 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # Build column‑wise: column i contains f[i] on rows i … i+n‑1
    data = np.empty((n, n), dtype=np.float64)
    data[:] = f[:, None]               # each row i is constant f[i]
    offsets = np.arange(n)             # 0 … n‑1
    T = scipy.sparse.diags(data, offsets, shape=(rows, n), format='csc')
    return T


def _lp_maximize_sum(f: np.ndarray, M: float, time_limit: float) -> np.ndarray | None:
    """
    Solve
        max Σ b_i
        s.t. (f * b) ≤ M   (element‑wise)
             b_i ≥ 0
    Returns the optimal b (as a dense numpy array) or None on failure.
    """
    n = f.shape[0]
    # Build the constraint matrix once – it has n² non‑zeros.
    T = _build_conv_matrix(f)               # CSC format, size (2n‑1)×n
    A_ub = T
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)       # minimise –Σb_i  == maximise Σb_i
    bounds = [(0.0, None)] * n

    # HiGHS respects a simple "time_limit" option (seconds).  If the SciPy
    # version does not support it, the argument is ignored – the call will
    # just run to completion.
    opts = {"time_limit": time_limit}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=opts)

    if res.success:
        return res.x
    else:
        print("[LP] failure:", res.message)
        return None


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate of 2·n·max(conv) and its gradient w.r.t. the
    soft‑max parameters w (used by L‑BFGS).  The implementation is the same
    as in the baseline but without globals.
    """
    # soft‑max
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)
    max_c = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_c) / eps
    log_sum = np.logaddexp.reduce(shifted)    # stable log‑sum‑exp
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()

    # convolution of a with reversed p (required for derivative)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.size - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.size - 1]

    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray, eps: float, n: int) -> tuple[float, np.ndarray]:
    """Wrapper required by scipy.optimize.minimize."""
    return _smooth_obj_and_grad(w, eps, n)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Simple heuristic: put the largest values on the left side and the
    smallest on the right.  Often reduces the convolution peak.
    """
    n = a.size
    s = np.sort(a)                 # ascending
    out = np.empty_like(a)
    half = n // 2
    out[:half] = s[:half][::-1]   # small → right side
    out[-half:] = s[-half:][::-1] # large → left side
    if n % 2:
        out[half] = s[half]
    return out


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap (swap / mass‑transfer) hill‑climber that keeps the vector
    on the simplex and accepts only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Re‑project onto simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[hill] new best = {cur_val:.6f}")

    return cur_a, cur_val
