"""
Propose a sequence of non‑negative heights that minimises::

    2 * n * max(a*a) / (sum(a)**2)

The code does **not** rely on the heavyweight L‑BFGS stage of the previous
implementation – it focuses on three fast, complementary phases that fit easily
into the 1000 s budget:

1.  **Initialisation** – load the previous best construction if available;
    otherwise draw a sparse Dirichlet vector and symmetrically place the
    largest values on the two ends (alternating left/right).  This already
    improves the baseline considerably.

2.  **Aggregated sub‑gradient descent** – instead of moving away from a single
    peak of the self‑convolution, we collect all indices whose value is within
    a tiny fraction (default 0.1 %) of the maximum.  The sum of the associated
    sub‑gradients is computed with a single FFT and the iterate is projected
    back onto the ℓ₁‑simplex.  A diminishing step size (≈ 1/√t) yields
    monotone progress.

3.  **Continuous simulated‑annealing** – random *swap* and *mass‑transfer*
    moves explore the neighbourhood of the current best sequence.  The
    temperature starts as a modest fraction of the current objective and
    decays slowly; uphill moves are accepted with the usual Metropolis rule.
    The best sequence encountered is retained.

At the end we evaluate the final list with the official ``evaluate_sequence``
function and return a plain Python list.

All helper functions are top‑level (no closures) and use only NumPy
(and optionally SciPy for the very lightweight convolution).  The total
wall‑time is carefully monitored to stay safely below the 1000 s limit.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Fast convolution helpers (FFT based,  O(n log n) )
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = rfft(a, size)
    conv = irfft(A * A, size)[:out_len]
    # remove tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = rfft(x, size)
    fy = rfft(y, size)
    conv = irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Exact analogue of ``evaluate_sequence`` for an already normalised
    (sum = 1) vector – returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    return float(2.0 * n * max_b)


def _normalise(a: np.ndarray) -> np.ndarray:
    """Project a onto the non‑negative ℓ₁‑simplex (sum = 1)."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s == 0.0:
        # fallback: uniform distribution
        return np.full_like(a, 1.0 / a.shape[0])
    return a / s


# ----------------------------------------------------------------------
# Re‑ordering heuristics – put large values on the two ends alternately
# ----------------------------------------------------------------------
def _ends_large_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic re‑ordering that places the largest values alternately at
    the leftmost and rightmost positions and then works inwards.
    Example (n=8):  L0, L1, L2, L3, R3, R2, R1, R0
    This pattern strongly reduces the peak of the self‑convolution for a
    given multiset of heights.
    """
    n = a.shape[0]
    vals = np.sort(a)[::-1]               # descending
    out = np.empty_like(a)
    left, right = 0, n - 1
    i = 0
    while i < n:
        out[left] = vals[i]
        left += 1
        i += 1
        if i >= n:
            break
        out[right] = vals[i]
        right -= 1
        i += 1
    return out


# ----------------------------------------------------------------------
# Aggregated sub‑gradient descent (works on the exact max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_aggregated(a0: np.ndarray,
                            deadline: float,
                            eps_frac: float = 1e-3,
                            step0: float = 0.15) -> (np.ndarray, float):
    """
    Projected sub‑gradient descent that moves away from *all* near‑maximal
    convolution peaks (those within ``eps_frac`` of the maximal value).
    The step size decays as ``step0 / sqrt(t)``.
    Returns the best sequence seen before ``deadline``.
    """
    n = a0.shape[0]
    a = a0.copy()
    cur_val = _eval_fast(a)
    best_a = a.copy()
    best_val = cur_val
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _fft_convolve_self(a)
        max_conv = conv.max()
        # active set: indices whose value is close to the max
        thresh = max_conv * (1.0 - eps_frac)
        active = conv >= thresh
        # p is the indicator of the active set
        p = active.astype(float)

        # gradient = 2 * Σ_{k∈active} a_{k-i}
        # compute the sum efficiently via convolution with the reversed mask
        # note: _conv_fft expects two real vectors
        conv_ap = _conv_fft(a, p[::-1])
        # segment that corresponds to indices [n‑1, 2n‑2)
        seg = conv_ap[n - 1: 2 * n - 1]
        grad = 2.0 * seg[::-1]                     # shape (n,)

        # step size
        alpha = step0 / np.sqrt(it + 1.0)

        # projected gradient step
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalise(a_new)

        new_val = _eval_fast(a_new)
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[subgrad‑agg] iter {it:5d} new best = {best_val:.6f}")

        if new_val <= cur_val:               # accept improving (or equal) steps
            a = a_new
            cur_val = new_val

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Continuous simulated annealing (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing_continuous(a0: np.ndarray,
                                   cur_val: float,
                                   deadline: float) -> (np.ndarray, float):
    """
    Fast SA that mixes two primitive moves:
        * swap the values of two positions,
        * transfer a small amount of mass from one position to another.
    The temperature starts as a modest fraction of the current objective
    and decays geometrically.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_val = cur_val
    cur_val = cur_val

    rng = np.random.default_rng()
    T = max(cur_val * 0.05, 1e-6)      # initial temperature
    decay = 0.99996

    it = 0
    while time.time() < deadline:
        # decide which move to propose
        if rng.random() < 0.4:       # swap move (40 %)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
        else:                        # mass transfer (60 %)
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            # transfer up to 20 % of the source mass
            delta = a[i] * 0.20 * rng.random()
            a[i] -= delta
            a[j] += delta
            # renormalise – only two entries changed, cheap way
            s = a[i] + a[j]
            if s == 0.0:
                # very unlikely, but protect against division by zero
                a[i] = 0.0
                a[j] = 0.0
            else:
                # keep total sum = 1 (the rest of the vector is untouched)
                scale = 1.0 / np.sum(a)
                a *= scale

        new_val = _eval_fast(a)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            # accept
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SA‑cont] iter {it:6d} new best = {best_val:.6f}")
        else:
            # revert the move
            if rng.random() < 0.4:      # this branch knows which move was tried
                a[i], a[j] = a[j], a[i]
            else:
                # reverse mass transfer
                a[i] += delta
                a[j] -= delta
                # fix possible tiny negativity due to rounding
                a = _normalise(a)

        # schedule temperature
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1

    return best_a, best_val
