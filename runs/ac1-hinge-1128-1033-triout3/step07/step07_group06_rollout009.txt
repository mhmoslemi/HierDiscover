def _adam_optimize(a0, deadline, rng, max_iter=20000):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a) # compute actual objective
    # Adam parameters
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # schedule for alpha: start low and increase log-scale.
    alpha0 = 5.0
    alpha_max = 200.0
    # maybe increment per 1000 iters.
    while time.time() < deadline and (max_iter is None or t < max_iter):
        t += 1
        # compute conv = a*a
        FA = np.fft.rfft(a, _next_pow2(2 * n - 1))
        conv = np.fft.irfft(FA * FA, _next_pow2(2 * n - 1))[: 2 * n - 1]
        # compute smooth max via log-sum-exp
        # increase alpha gradually
        alpha = min(alpha_max, alpha0 + (alpha_max - alpha0) * t / max_iter if max_iter else alpha0 + t * 0.001)
        # compute softmax weights
        # stable: subtract max
        max_conv = conv.max()
        # compute exp(alpha*(conv - max_conv)) for numerical stability
        exp_term = np.exp(alpha * (conv - max_conv))
        sum_exp = exp_term.sum()
        p = exp_term / sum_exp  # integrate safely
        # compute gradient: grad = 2 * correlation(p, a)
        # correlation using FFT: corr = IFFT(FA * conj(FP))
        FP = np.fft.rfft(p, _next_pow2(len(p) + n - 1))
        # but need same size for both transforms.
        size = _next_pow2(len(p) + n - 1)
        FA = np.fft.rfft(a, size)
        FP = np.fft.rfft(p, size)
        corr = np.fft.irfft(FA * np.conj(FP), size)[:n]
        grad = 2.0 * corr
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex: clip negatives, renormalize sum to 1
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            # avoid zero vector: random reinit
            a_new = rng.random(n)
            a_new /= a_new.sum()
        a = a_new
        # compute objective
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # optionally print progress
            if t % 200 == 0:
                print(f"[adam] iter {t} best_val = {best_val:.6f}")
        # adapt learning rate based on progress: optional
    return best_a, best_val

"""
Propose a step‑function sequence that minimizes ``evaluate_sequence``.

Algorithm
---------
1.  Initialise a non‑negative vector `a` on the simplex (sum = 1).  If a
    previous solution is available (`height_sequence_1`) it is reused,
    otherwise a random Dirichlet start is generated.

2.  Optimise the smooth approximation of the objective

        f_α(a) = (1/α)·log Σ_k exp(α·conv_k(a))

    where `conv_k(a)` is the full self‑convolution.
    The gradient of `f_α` is
        ∇f_α = 2·corr(p, a) ,
    with `p_k = exp(α·conv_k)/Σ_j exp(α·conv_j)`.  This is computed
    efficiently by FFT‑based correlation.

    Adam (adaptive moment) updates are applied, followed by projection
    onto the probability simplex (clip to ≥0 and renormalise).

    The smoothing parameter `α` is increased slowly from a small value
    (≈5) to a larger value (≈200) so that the optimisation first sees a
    smooth landscape and later focuses on the true max.

3.  A short aggressive hill‑climbing phase (mass transfer) finalises the
    solution.  The pair of indices that contributes most to the current
    maximal convolution entry is identified and a fraction of its mass
    is moved to two low‑mass locations if this strictly improves the
    exact objective.

4.  The best vector found (still on the simplex) is returned as a plain
    Python list.  Progress and intermediate scores are printed.  The
    whole routine respects the supplied time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT / convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT (real input)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fa * Fa, size)[: 2 * n - 1]
    # numerical noise may produce tiny negatives – clip them away
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective value: 2·n·max(conv(a,a)).  Assumes sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * _full_convolution(a).max()


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max objective
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    Returns the best vector encountered and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment

    best_a = a.copy()
    best_val = _objective(a)

    # smoothing schedule
    alpha_start = 5.0
    alpha_end = 200.0

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1

        # ------------------------------------------------------------------
        # 1.  Compute full convolution (needed for both the smooth loss and the
        #     exact max for logging).
        # ------------------------------------------------------------------
        size = _next_pow2(3 * n - 2)          # size large enough for both ops
        Fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(Fa * Fa, size)[: 2 * n - 1]
        np.maximum(conv, 0, out=conv)       # numerical safety

        # ------------------------------------------------------------------
        # 2.  Smooth approximation of the max (log‑sum‑exp) and its weights.
        # ------------------------------------------------------------------
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)
        max_conv = conv.max()
        # stable softmax
        exp_term = np.exp(alpha * (conv - max_conv))
        sum_exp = exp_term.sum()
        p = exp_term / sum_exp   # shape (2*n-1,)

        # ------------------------------------------------------------------
        # 3.  Gradient of the smooth loss:
        #     ∇f = 2 * corr(p, a)  (linear correlation, computed via FFT)
        # ------------------------------------------------------------------
        Fp = np.fft.rfft(p, size)
        corr = np.fft.irfft(Fa * np.conj(Fp), size)[:n]   # linear correlation a ↔ p
        grad = 2.0 * corr

        # ------------------------------------------------------------------
        # 4.  Adam update + projection onto the simplex.
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection: clip negatives and renormalise to sum = 1
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:
            # re‑initialise if everything collapsed to zero
            a_new = rng.random(n)
            a_new /= a_new.sum()

        a = a_new

        # ------------------------------------------------------------------
        # 5.  Track the best exact objective found so far.
        # ------------------------------------------------------------------
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if t % 200 == 0:
                print(f"[adam] iter {t:5d}  objective = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill climbing (same spirit as the reference)
# ----------------------------------------------------------------------


def _mass_transfer_hillclimb(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_no_improve: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Fine‑tune a vector by repeatedly moving a fraction of the most damaging
    mass to two low‑mass positions, accepting only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that participate in this maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass to relocate
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # pick two low‑mass destinations (distinct and different from donors)
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        # re‑normalise
        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val

    deadline = start + budget_s - 0.5  # safety margin for the caller

    # ------------------------------------------------------------------

    default_n = 4000

        n = int(old.shape[0])

        a0 /= a0.sum()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    a_adam, val_adam = _adam_optimize(a0, deadline, rng, max_iter=20000)
