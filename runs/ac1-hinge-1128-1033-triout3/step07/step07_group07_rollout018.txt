def _simulated_annealing(a0, val0, deadline, rng):
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_val = val0
    best_a = cur_a.copy()

        elapsed = time.time() - start
        frac = elapsed / total
        T = T_initial * (1 - frac)  # linear cooling
        if T < 1e-8:

        # generate a move

"""
Propose a step‑function (non‑negative list of heights) that minimizes
evaluate_sequence.  The routine combines three ideas:

1.  **Smoothed L‑BFGS optimisation** –  maximise a log‑sum‑exp
    surrogate of the max‑convolution.  A continuation on the smoothing
    parameter ε drives the iterates toward a true minimiser.

2.  **Hybrid local search** –  aggressive hill‑climbing (mass transfer)
    followed by a simple simulated‑annealing loop.  Both phases use the
    exact max‑convolution objective and keep the best point found.

3.  **Polishing sub‑gradient descent** –  a projected sub‑gradient step
    on the exact 2·n·max(conv) objective brings the final value a little
    lower.

The algorithm respects the 1000 s budget, prints progress and finally
returns the best list of floats it discovered.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negatives due to round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns the value and gradient of the surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε)

    where a = softmax(w) (∑a = 1, a_i ≥ 0).
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()

    # full autocorrelation
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    # log Σ exp(...)
    log_sum = np.logaddexp.reduce(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # probabilities over lags

    # convolution a * rev(p)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                  # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]                # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Exact objective (used by hill‑climb, annealing, sub‑gradient)
# ----------------------------------------------------------------------
def _objective_val(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a normalised a (∑a = 1)."""
    n = a.shape[0]
    return float(2.0 * n * np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              best_val: float,
                              deadline: float,
                              rng: np.random.Generator,
                              step_start: float = 0.2):
    """Move a fraction of mass away from the lag that attains max(conv)."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    best_a = cur_a.copy()
    step_frac = step_start

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # indices that contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue

        # pick a few dominant contributors
        top_n = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        # destination indices – pick low‑mass places
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue

        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue
        new_a /= s

        new_val = _objective_val(new_a)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-6)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing phase (exact objective)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         val0: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Metropolis‑style annealing that moves a fraction of mass from a random
    source index to a random destination.  Acceptance probability
    decreasing linearly in time.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_val = val0
    best_a = cur_a.copy()
    best_val = cur_val

    start = time.time()
    total = max(deadline - start, 1e-6)
    T0 = max(0.1 * cur_val, 1e-3)          # initial temperature

    while time.time() < deadline:
        # linear cooling schedule
        elapsed = time.time() - start
        frac = elapsed / total
        T = T0 * max(0.0, 1.0 - frac)
        if T < 1e-8:
            T = 1e-8

        # propose a move
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if cur_a[i] < 1e-12:
            continue
        beta = rng.random() * 0.2               # up‑to 20 % of source mass
        delta = cur_a[i] * beta
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_val = _objective_val(new_a)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                cur_a = new_a
                cur_val = new_val

    return best_a, best_val


# ----------------------------------------------------------------------
# Fine sub‑gradient descent (projected)
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         best_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Sub‑gradient descent on 2·n·max(conv) with Euclidean projection onto
    the simplex.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    best_a = cur_a.copy()
    eta = 0.1                     # initial step size

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # gradient step + projection
        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = _objective_val(new_a)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            eta = min(eta * 1.05, 0.5)
        else:
            eta = max(eta * 0.7, 1e-5)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple deterministic initialisations
# ----------------------------------------------------------------------
def _init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _init_linear_desc(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)  (high at the left, low at the right)."""
    a = np.arange(n, 0, -1, dtype=float)
    return a / a.sum()


def _init_linear_asc(n: int) -> np.ndarray:
    """a_i ∝ (i+1)  (mirrored version of _init_linear_desc)."""
    a = np.arange(1, n + 1, dtype=float)
    return a / a.sum()


def _init_u_shape(n: int) -> np.ndarray:
    """U‑shaped distribution: larger mass at the extremes."""
    idx = np.arange(n)
    a = 1.0 / (idx + 1) + 1.0 / (n - idx)
    return a / a.sum()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Optimises a non‑negative step‑function to minimise `evaluate_sequence`.
    The routine respects the given time budget and returns the best list of
    floats found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a bit larger than the reference 4000.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Collect a few plausible starting points.
    # ------------------------------------------------------------------
    init_candidates = []

    # try to reuse the previous construction if present
    try:
        a_prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a_prev.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a_prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a_prev = np.interp(new_x, old_x, a_prev)
        a_prev = np.maximum(a_prev, 0.0)
        if a_prev.sum() > 0:
            a_prev = a_prev / a_prev.sum()
            init_candidates.append(a_prev)
    except Exception:
        pass

    # deterministic seeds
    init_candidates.append(_init_uniform(n))
    init_candidates.append(_init_linear_desc(n))
    init_candidates.append(_init_linear_asc(n))
    init_candidates.append(_init_u_shape(n))
    # a random start (will be re‑normalised)
    init_candidates.append(rng.random(n))

    # ------------------------------------------------------------------
    # Smoothed optimisation for each start
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    eps_schedule = np.logspace(-2, -12, num=20)   # continuation on ε

    for idx, a0 in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-12:
            continue
        a0 = a0 / a0.sum()
        w = np.log(a0 + 1e-12)

        local_best_a = a0.copy()
        local_best_val = _objective_val(a0)

        for eps in eps_schedule:
            if time.time() >= deadline:
                break

            # L‑BFGS on the smooth surrogate
            res = minimize(_smooth_obj_and_grad,
                           w,
                           args=(eps, n),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 150,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate exact objective
            a = np.exp(w - np.max(w))
            a /= a.sum()
            cur_val = _objective_val(a)

            if cur_val < local_best_val:
                local_best_val = cur_val
                local_best_a = a.copy()
                print(f"[start {idx} eps {eps:.1e}] new local best = {local_best_val:.6f}")

        if local_best_val < best_val:
            best_val = local_best_val
            best_a = local_best_a.copy()
            print(f"[start {idx}] improved global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Aggressive hill‑climbing phase
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline:
        best_a, best_val = _mass_transfer_hill_climb(best_a,
                                                    best_val,
                                                    deadline,
                                                    rng,
                                                    step_start=0.15)

    # ------------------------------------------------------------------
    # Simulated annealing to escape possible local minima
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5.0:
        best_a, best_val = _simulated_annealing(best_a,
                                                best_val,
                                                deadline,
                                                rng)

    # ------------------------------------------------------------------
    # Fine sub‑gradient polishing
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline:
        best_a, best_val = _subgradient_descent(best_a,
                                                best_val,
                                                deadline,
                                                rng)

    # ------------------------------------------------------------------
    # Final reporting (uses the provided evaluator)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
