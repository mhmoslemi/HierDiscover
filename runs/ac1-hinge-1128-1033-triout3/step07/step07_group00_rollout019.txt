"""
Propose a step‐function (non‑negative vector) that minimises the
objective 2·n·max(conv(a,a))/sum(a)^2.

Algorithm
---------
1.  Build a lightweight library of parametric families (uniform,
    exponential, power‑law, sine, triangular, etc.) and keep the best
    candidate.

2.  Refine the candidate with three complementary optimisation stages,
    each respecting the global time budget:

    • Adam on a smooth log‑sum‑exp approximation of the max‑convolution.
    • Targeted hill‑climbing that moves a small fraction of mass from
      indices that heavily contribute to the maximal convolution entry to
      indices that contribute little.
    • Exact sub‑gradient descent on the true max‑convolution (projected
      onto the probability simplex).

3.  A short final sweep mixing a tiny amount of uniform mass often reduces
    the peak further.

All stages use only NumPy and are written as plain functions (no closures,
no lambdas) so that they can be called from the evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w ≥ 0, Σ w = 1}."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric initialisation – a diverse set of shapes
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector (sum = 1) and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # uniform
    _try(np.ones(n))

    # exponential left‑hand side
    for r in (0.99999, 0.99997, 0.99995, 0.9999, 0.9998, 0.9995, 0.999):
        a = r ** np.arange(n, dtype=float)
        _try(a)

    # symmetric exponential (U‑shaped)
    for r in (0.99999, 0.99997, 0.99995, 0.9999, 0.9998, 0.9995):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # power‑law symmetric
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p) + (n - i + 1) ** (-p)
        _try(a)

    # half‑wave sine (zero at the ends, peak in the middle)
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _try(a)

    # triangular (peak in the centre, zero at the ends)
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)          # values 0 … n/2 … 0
    _try(a)

    # random Dirichlet draws – a few quick samples
    rng = np.random.default_rng(12345)
    for _ in range(12):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    w = np.log(a0 + 1e-12)          # logits → simplex via soft‑max

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 50000

    it = 0
    while time.time() < deadline:
        # --- get point on simplex ---
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # --- schedules ---
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # --- smooth max via log‑sum‑exp ---
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax (log‑softmax gradient)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.006, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – proportional to contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – prefer places that hardly affect the max entry
        if np.random.rand() < 0.35:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and restart from the best seen so far
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – Exact sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n            # initial step size (roughly 0.5·avg(a))
    eta = eta0
    decay = 0.9995
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step & projection onto the simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # decay step size slowly
        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # shake a little and restart step‑size
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small post‑processing: mixing with uniform
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 9000

    # ------------------------------------------------------------------
    # Initialise a – use a known construction if present, otherwise fall back
    # to a rich parametric sweep.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation on a smooth approximation of max(conv)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.25 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        a0, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase B – targeted hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.70 * (deadline - start_time)
    best_a = _hillclimb_targeted(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )
    best_val = _objective_value(best_a)

    # ------------------------------------------------------------------
    # Phase C – exact sub‑gradient descent (projected onto simplex)
    # ------------------------------------------------------------------
    phase_c_deadline = deadline
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
