import time, numpy as np, math

def full_convolution(x):
    n = x.size
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx*fx, size)[:out_len]
    return conv

def conv_fft(x,y):
    n,m = x.size, y.size
    out_len = n+m-1
    size = 1 << (out_len-1).bit_length()
    fx = np.fft.rfft(x,size)
    fy = np.fft.rfft(y,size)
    conv = np.fft.irfft(fx*fy,size)[:out_len]
    return conv

def objective(a):
    n = a.size
    b = full_convolution(a)

"""
Propose a non‑negative step‑function that (approximately) minimizes

    evaluate_sequence(a)

The algorithm works in three phases:

1️⃣  *Coarse search* – on a modest size (n=2000) we sweep a two‑parameter
    Beta family a_i ∝ (i+1)^{p‑1}(n‑i)^{q‑1} and also try a few random
    mixtures of such shapes.  The best tiny vector is kept.

2️⃣  *Scale up* – the best tiny vector is interpolated to a larger size
    (n≈8000) and refined with an Adam optimiser on the smooth‑max
    approximation of  max(a*a).  The parameters (learning‑rate,
    temperature) are annealed logarithmically.

3️⃣  *Local search* – a simulated‑annealing hill‑climber moves a small
    amount of mass from indices that heavily contribute to the current
    maximal convolution entry to indices that contribute little.
    Metropolis acceptance with a linearly decaying temperature allows
    uphill moves and helps escape local minima.

The whole routine respects the 1000 s budget; the final candidate is
returned as a list whose sum is 1 (the exact value does not matter for the
objective because it is scale‑invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of ``x`` (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (Σa = 1, i.e. 2·n·max(a*a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a normalized to sum = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max approximation and its gradient (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) together with the
    gradient of 2·n·max(conv) (≈ objective).  The gradient is obtained
    via the chain rule applied to the soft‑max weights.
    """
    n = a.shape[0]
    b = _full_convolution(a)            # length 2·n‑1
    max_b = b.max()

    # numerically stable log‑sum‑exp
    shifted = (b - max_b) / eps
    max_shift = shifted.max()
    exps = np.exp(shifted - max_shift)
    sum_exp = exps.sum()
    weights = exps / sum_exp            # soft‑max probabilities, sum to 1

    # gradient of Σ_k weights_k * b_k w.r.t. a
    # derivative of each b_k wrt a_i is a_{k-i}+a_i; the convolution
    # implementation below captures the contribution efficiently.
    p_rev = weights[::-1]
    conv_ab = _conv_fft(a, p_rev)      # length 2·n‑1
    segment = conv_ab[n - 1:2 * n - 1]  # the n entries that affect a_i
    # factor 4·n accounts for the 2·n in the objective and the two
    # symmetric contributions in the derivative of the self‑convolution.
    grad_a = 4.0 * n * segment[::-1]

    # smooth max (log‑sum‑exp) value (without the outer factor 2·n)
    smooth_val = max_b + eps * (np.log(sum_exp) + max_shift)
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimiser that minimises a smooth approximation of the objective."""
    # logits guarantee Σa = 1 via the soft‑max map
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.14, 0.001
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedule
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        smooth_val, grad_a = _smooth_max_and_grad(a, eps)

        # back‑prop through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation to escape plateaus
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.02, size=n)

        # restart if we have been idle for too long
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restarting after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Local search that transfers a small amount of mass from “bad” to “good” indices."""
    a = a0.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule for the maximum transfer fraction and temperature
    max_frac_start, max_frac_end = 0.48, 0.001
    temp_start, temp_end = 0.018, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 150_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ---- choose source index -------------------------------------------------
        # with probability 0.85 pick proportionally to contribution, otherwise pick
        # a random heavy index – this adds exploration.
        if total_contrib > 0 and np.random.rand() < 0.85:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ---- choose target index -------------------------------------------------
        # bias towards indices with little contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0 and np.random.rand() < 0.9:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback: uniform among all indices except i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # ---- amount of mass to move ---------------------------------------------
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        # the linear part 2·Δ·a spreads over the whole convolution
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic self‑interaction corrections
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – occasional global perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric sweep of a two‑parameter Beta family (small n)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a Beta‑shaped family a_i ∝ i^{p‑1}(n‑i)^{q‑1} on a small grid."""
    best_a = None
    best_val = np.inf
    # grid of exponents – a mix of linear and geometric progressions
    p_vals = np.concatenate([
        np.linspace(0.1, 1.0, 9),
        np.geomspace(1.0, 12.0, 9)[1:]
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s                               # Σa = 1
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _random_beta_mix_best(n: int,
                          trials: int = 120,
                          components: int = 3) -> tuple[np.ndarray, float]:
    """Create random convex combinations of a few Beta shapes and keep the best."""
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)

    for _ in range(trials):
        # draw random exponents and weights
        ps = np.random.uniform(0.1, 12.0, size=components)
        qs = np.random.uniform(0.1, 12.0, size=components)
        ws = np.random.dirichlet(np.ones(components))
        a = np.zeros(n, dtype=np.float64)
        for p, q, w in zip(ps, qs, ws):
            shape = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            a += w * shape
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Helper: up‑sample a vector by linear interpolation
# ----------------------------------------------------------------------
def _interpolate(src: np.ndarray, new_len: int) -> np.ndarray:
    """Linearly interpolate ``src`` to length ``new_len`` and renormalise."""
    old_len = src.shape[0]
    if old_len == new_len:
        out = src.copy()
    else:
        new_x = np.linspace(0.0, 1.0, new_len)
        old_x = np.linspace(0.0, 1.0, old_len)
        out = np.interp(new_x, old_x, src)
        out = np.maximum(out, 0.0)
    out /= out.sum()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a short safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Phase 0 – coarse exploration on a modest size
    # ------------------------------------------------------------------
    n_small = 2000
    best_a_small, best_val_small = None, np.inf

    # 0a – try to reuse a previous construction if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)    # type: ignore[name-defined]
        best_a_small = _interpolate(old, n_small)
        best_val_small = _objective(best_a_small)
        print("[init] interpolated previous solution (small n) value =", best_val_small)
    except Exception:
        pass

    # 0b – exhaustive Beta sweep
    try:
        beta_a, beta_val = _beta_family_best(n_small)
        if beta_val < best_val_small:
            best_a_small, best_val_small = beta_a, beta_val
            print("[init] beta sweep improved value =", best_val_small)
    except Exception as exc:
        print("[init] beta sweep failed:", exc)

    # 0c – random mixtures of Beta shapes
    try:
        mix_a, mix_val = _random_beta_mix_best(n_small, trials=150, components=3)
        if mix_val < best_val_small:
            best_a_small, best_val_small = mix_a, mix_val
            print("[init] random mixture improved value =", best_val_small)
    except Exception as exc:
        print("[init] random mixture failed:", exc)

    # 0d – fallback random start
    if best_a_small is None:
        a0 = np.random.rand(n_small)
        a0 /= a0.sum()
        best_a_small, best_val_small = a0, _objective(a0)
        print("[init] random start (small n), value =", best_val_small)

    # ------------------------------------------------------------------
    # Phase 1 – scale up to a larger dimension for finer control
    # ------------------------------------------------------------------
    n_large = 8000
    a = _interpolate(best_a_small, n_large)

    # make sure we have a proper simplex point
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_a, best_val = a.copy(), _objective(a)

    print(f"[scale] up‑sampled to n={n_large}, initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈30 % of the total budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _adam_optimize(best_a, n_large,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n_large,
                                         deadline=deadline,
                                         start_time=phase_a_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
