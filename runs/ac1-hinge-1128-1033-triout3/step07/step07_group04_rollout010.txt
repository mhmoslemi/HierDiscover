"""
Hybrid optimiser for the step‑function convolution problem (new version).

The algorithm consists of three main phases:

1. **Diverse seeding** – generate a handful of hand‑crafted families
   (uniform, geometric, power‑law, Gaussian, random Dirichlet) and keep
   the best one according to the fast evaluation routine.

2. **Multiplicative weight descent** – a deterministic sub‑gradient
   method for the objective  f(a)=max_k (a*a)_k .  At each iteration we
   locate the index of the current maximal convolution entry, compute the
   gradient of that entry (which is simply a reversed copy of the vector)
   and update a_i ← a_i·exp(−η·g_i).  The step‑size η is slowly
   annealed.  This phase rapidly pushes the distribution away from the
   peaks of the autocorrelation.

3. **Targeted pair‑wise simulated‑annealing** – a specialised move that
   removes mass from the pair (i,j) contributing most to the current
   maximum convolution value and redistributes a fraction of that mass to
   random positions.  The move is accepted according to a temperature
   schedule, allowing occasional uphill steps.

Finally a short hill‑climbing phase (swap / tiny transfer) polishes the
solution before the exact ``evaluate_sequence`` is called.

The implementation respects the 1000 s time budget, works for any
reasonable sequence length (default n=4000) and returns a list of
non‑negative floats that minimises the supplied evaluation function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – FFT based convolution and fast evaluation
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for a non‑negative vector.
    The vector is assumed to sum to 1 (the routine works for any sum).
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple parametric families for seeding
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """a_i ∝ q**i  (0<q<1), normalised to sum = 1."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q**n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)**p   (p<0), normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _gaussian_sequence(n: int, sigma_ratio: float = 0.2) -> np.ndarray:
    """Discrete Gaussian centred in the middle, normalised to sum = 1."""
    sigma = sigma_ratio * n
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Multiplicative‑weight descent on the max‑convolution
# ----------------------------------------------------------------------
def _multiplicative_descent(a: np.ndarray,
                           cur_val: float,
                           deadline: float) -> (np.ndarray, float):
    """
    Deterministic sub‑gradient descent on  max_k (a*a)_k .
    Uses the update a_i ← a_i·exp(−η·g_i) where g_i = 2·a_{k*−i}.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = cur_val
    eta = 0.35                     # initial step size

    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # gradient of the active convolution entry
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # multiplicative update
        a = a * np.exp(-eta * g)

        # renormalise to keep sum = 1
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional tiny random perturbation to avoid stagnation
        if rng.random() < 0.03:
            i, jrand = rng.integers(0, n, size=2)
            a[i], a[jrand] = a[jrand], a[i]

        cur_val = _eval_fast(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[MD] iteration {it:5d} new best = {best_val:.6f}")

        # slowly anneal the step size
        eta = max(1e-5, eta * 0.9995)
        it += 1

    return best_a, best_val

# Targeted pair‑wise simulated annealing
# ----------------------------------------------------------------------
def _targeted_pair_sa(a: np.ndarray,
                      cur_val: float,
                      deadline: float) -> (np.ndarray, float):
    """
    Simulated annealing that repeatedly selects the (i,j) pair
    contributing most to the current maximal convolution entry and
    transfers a fraction of its mass to random positions.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.05, 1e-6)   # initial temperature
    decay = 0.9998
    it = 0

    while time.time() < deadline:
        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # compute products a_i * a_{k_max-i}
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        prod = np.zeros_like(a)
        prod[mask] = a[i[mask]] * a[j[mask]]
        # pick the pair with the largest product
        idx = int(np.argmax(prod))
        i_src = idx
        j_src = int(k_max - idx)

        moved = False
        new_a = a.copy()

        if a[i_src] > 0 and a[j_src] > 0:
            # transfer a fraction from both members of the pair
            delta = min(a[i_src], a[j_src]) * 0.08 * rng.random()
            if delta > 0:
                t1 = rng.integers(0, n)
                t2 = rng.integers(0, n)
                new_a[i_src] -= delta
                new_a[j_src] -= delta
                new_a[t1] += delta
                new_a[t2] += delta
                moved = True

        # fallback moves if the pair move is not possible
        if not moved:
            if rng.random() < 0.5:
                # simple swap
                i1, i2 = rng.integers(0, n, size=2)
                if i1 != i2:
                    new_a[i1], new_a[i2] = new_a[i2], new_a[i1]
            else:
                # random mass transfer
                src = rng.integers(0, n)
                dst = rng.integers(0, n)
                if src != dst and new_a[src] > 0.0:
                    delta = new_a[src] * 0.06 * rng.random()
                    new_a[src] -= delta
                    new_a[dst] += delta

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[PAIR-SA] it {it:5d} new best = {best_val:.6f}")

        T = max(T * decay, 1e-12)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (swap / tiny transfer)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Local search with swaps and very small transfers."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = cur_val

    while time.time() < deadline:
        if rng.random() < 0.2:                 # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                   # tiny transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * 0.025 * rng.random()
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a.copy()
            a = new_a
            print(f"[REFINE] new best = {best_val:.6f}")

    return best_a, best_val

        if old.shape[0] != n:

            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 0 – diverse hand‑crafted seeds (brief sweep)
    # --------------------------------------------------------------
    seeds = []

    # uniform (already have)
    seeds.append(best_a)

    # geometric families
    for q in np.linspace(0.7, 0.999, 15):
        seeds.append(_geometric_sequence(q, n))

    # power‑law families
    for p in np.linspace(-3.0, -0.2, 15):
        seeds.append(_power_sequence(p, n))

    # Gaussian‑like families
    for sigma in (0.12, 0.18, 0.25):
        seeds.append(_gaussian_sequence(n, sigma_ratio=sigma))

    # random Dirichlet draws (moderately concentrated)
    rng = np.random.default_rng(seed)
    for _ in range(20):
        a = rng.random(n)
        a /= a.sum()
        seeds.append(a)

    # evaluate all seed candidates
    for a in seeds:
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[seed] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – multiplicative‑weight descent (major improvement)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    stage1_deadline = time.time() + min(0.45 * budget_s, time_left * 0.5)
    best_a, best_val = _multiplicative_descent(best_a, best_val, stage1_deadline)

    # --------------------------------------------------------------
    # Stage 2 – targeted pair‑wise simulated annealing
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    stage2_deadline = time.time() + min(0.35 * budget_s, time_left * 0.6)
    best_a, best_val = _targeted_pair_sa(best_a, best_val, stage2_deadline)

    # --------------------------------------------------------------
    # Stage 3 – final hill‑climbing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # Exact evaluation using the supplied function
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    # Return the best sequence found (as plain Python list)
    return best_a.tolist()
