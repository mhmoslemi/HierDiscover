"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence``.

The algorithm builds on the previous multi‑phase scheme but adds a
*mirror‑descent* stage that optimises a smooth approximation of the
max‑convolution using the full log‑sum‑exp gradient (instead of only the
few largest peaks).  The schedule:

1. Initialise from the previous construction (if any) or from a uniform
   vector.
2. A few random Dirichlet starts – keep the best.
3. Several L‑BFGS runs on a smooth surrogate (soft‑max parametrisation)
   with decreasing smoothing ε.
4. **Mirror‑descent smoothing** – for a decreasing list of ε compute the
   exact gradient of the smooth max‑convolution and perform a
   multiplicative (exponentiated‑gradient) update, re‑projecting onto the
   probability simplex after each step.  This drives all convolution
   peaks toward the global average.
5. A short multi‑peak exponential gradient descent (as in the reference
   code) for fine‑grained local improvement.
6. Simple hill‑climbing (random swaps / mass transfers).

The whole procedure respects the 1000 s budget and stays within a single
CPU.  All helper functions are top‑level to satisfy the execution
environment.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself (length 2·len‑1)
    using an FFT of power‑of‑two size.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative artefacts caused by floating point noise
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair large values with small ones when the sequence is reversed.
    This heuristic often reduces the height of the largest convolution
    peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)                     # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # probability over lags

    # convolution of a with reversed p (FFT based)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                    # ∂L/∂a_i

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule: geometric decay from 5·10⁻³ down to 10⁻⁹
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=9)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to a (probability simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric re‑ordering heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:

            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent on the smooth max‑convolution
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, eps: float):
    """
    Gradient of the smooth surrogate
        L_ε(a) = 2·n·ε·log Σ_k exp(conv_k/ε)
    with respect to the vector ``a`` (already normalised to sum=1).

    Returns (grad, max_conv) where ``grad`` has shape (n,) and
    ``max_conv`` is the (non‑smooth) maximal convolution entry.
    """
    n = a.shape[0]
    conv = _full_convolution(a)               # length 2·n‑1
    max_conv = conv.max()

    # softmax weights over the lag indices
    shifted = (conv - max_conv) / eps          # ≤ 0
    weights = np.exp(shifted)                 # underflow is fine
    sum_w = weights.sum()
    if sum_w == 0.0:
        p = np.full_like(weights, 1.0 / weights.size)
    else:
        p = weights / sum_w

    # convolution of a with reversed p (FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # central n entries give ∂L/∂a (up to the constant 4·n)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad = 4.0 * n * seg[::-1]                # gradient w.r.t. a

    return grad, max_conv


def _mirror_descent_smooth(a_start: np.ndarray, deadline: float):
    """
    Exponentiated‑gradient (mirror) descent on the smooth max‑convolution.
    A decreasing ε‑schedule is used; after each ε‑phase a few
    anti‑symmetric re‑orderings are tried.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[MD init] value = {best_val:.6f}")

    # ε‑schedule (from relatively coarse to very fine)
    eps_schedule = [5e-2, 2e-2, 1e-2, 5e-3, 2e-3, 1e-3,
                    5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
                    5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]

    for eps in eps_schedule:
        if time.time() >= deadline - 5.0:
            break

        eta = 0.5                     # initial step size for this ε
        it = 0
        # limit the total work per ε to a few thousand updates
        max_it = 2000

        while time.time() < deadline - 5.0 and it < max_it:
            grad, max_conv = _smooth_gradient(a, eps)

            # multiplicative update (mirror descent)
            step = -eta * grad
            # avoid overflow/underflow in the exponent
            step = np.clip(step, -30.0, 30.0)
            a = a * np.exp(step)

            # project onto the simplex
            s = a.sum()
            if s <= 1e-12:                     # numerical guard
                a = np.full_like(a, 1.0 / a.shape[0])
                s = 1.0
            a /= s

            it += 1

            # occasional bookkeeping
            if it % 100 == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[MD eps {eps:.1e} iter {it}] new best = {best_val:.6f}")

                # anti‑symmetric re‑ordering helps to break symmetry
                if it % 500 == 0:
                    a_ord = _anti_symmetric_reorder(a)
                    val_ord = evaluate_sequence(a_ord.tolist())
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        print(f"[MD reorder eps {eps:.1e}] improved to {best_val:.6f}")

            # slowly decay the learning rate
            if it % 500 == 0:
                eta *= 0.9

    return best_a, best_val

# Multi‑peak exponential gradient descent (as in the reference code)
# ----------------------------------------------------------------------
def _gradient_descent_multi(a_start: np.ndarray, deadline: float):
    """
    Exponential mirror‑descent that reduces the *k* largest convolution peaks.
    Returns the best (a, value) found before ``deadline``.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[gd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.5          # initial step‑size
    top_k = 3             # number of peaks used for the gradient
    decay_rate = 0.98     # decay of α every 2000 iterations

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_val = conv.max()

        # indices of the ``top_k`` largest convolution entries
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]   # sorted descending
        else:
            idxs = np.arange(conv.size)

        # gradient contribution from the selected peaks
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            jdx = k - idx_range
            g[idx_range] += a[jdx]

        # scheduled step‑size
        alpha = alpha0 * (decay_rate ** (it // 2000))

        # multiplicative update (mirror‑descent)
        a = a * np.exp(-alpha * g)

        # projection onto the probability simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:                     # guard against under‑flow
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        # occasional logging / re‑ordering
        if it % 50 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd iter {it}] new best = {best_val:.6f} (max conv {max_val:.8f})")
            if it % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] improved to {best_val:.6f}")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (random swaps / mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random local search: swaps (20 %) and mass transfers (80 %).
    Keeps only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
