def build_sidon_set(n, max_size=None):
    selected = []
    sums = set()
    for i in range(n):
        # check collisions
        ok = True
        # for each a in selected:
        for a in selected:
            if (i + a) in sums:
                ok = False
                break
        if not ok:
            continue
        # Also check i+i sum
        if (i + i) in sums:
            continue
        # if passes, add i
        selected.append(i)
        # add new sums
        for a in selected:
            sums.add(i + a)
        # also add i+i (will be added by loop above when a == i)
        if len(selected) >= max_size:
            break
    return selected
```

```
def propose_candidate(seed=0, budget_s=1000):
    """Generate a non‑negative step function that lowers the max‑convolution bound.
    
    The algorithm starts from a decent baseline (previous best if available,
    otherwise uniform) and repeatedly solves a linear program that maximises
    the total mass of a candidate sequence `g` while keeping the mixed
    convolution `f * g` below the current maximal self‑convolution value `M`.
    The candidate is scaled to unit sum and mixed into the current sequence.
    This monotone scheme drives the maximal self‑convolution down.
    A small hill‑climbing pass is performed at the end.
    """
    import time, numpy as np
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    # ------------------------------------------------------------------
    # Helper functions
    # ------------------------------------------------------------------
    def _next_pow2(x: int) -> int:
        return 1 << (x - 1).bit_length()

    def _full_convolution(a: np.ndarray) -> np.ndarray:
        """Linear convolution a*a using FFT (real)."""
        n = a.shape[0]
        out_len = 2 * n - 1
        size = _next_pow2(out_len)
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa * fa, size)[:out_len]
        np.maximum(conv, 0, out=conv)   # clip tiny negatives
        return conv

    def _fast_evaluate(a: np.ndarray) -> float:
        """Fast evaluation assuming a is non‑negative and sum = 1."""
        n = a.shape[0]
        conv = _full_convolution(a)
        max_b = conv.max()
        return float(2.0 * n * max_b)

    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # keep a margin for final print

    # Determine length n – prefer the length of the known construction.
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # Baseline vector (sum = 1)
    if has_old:
        f = old.copy()
        if f.sum() < 1e-12:
            f = np.full(n, 1.0 / n)
        else:
            f = f / f.sum()
        print(f"[init] Using provided height_sequence_1 of length {n}")
    else:
        f = np.full(n, 1.0 / n)
        print("[init] No previous construction – start with uniform vector")

    best_f = f.copy()
    best_val = _fast_evaluate(best_f)
    print(f"[init] initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Core iterative LP‑based refinement
    # ------------------------------------------------------------------
    t_mix = 0.5          # initial mixing factor
    min_t = 0.05
    eps_active = 1e-5   # relative tolerance when selecting active constraints

    iteration = 0
    while time.time() < deadline and iteration < 200:
        iteration += 1
        # convolution of current f with itself
        conv_f = _full_convolution(f)
        M = conv_f.max()
        # pick “active” indices where conv is close to M
        # Use absolute tolerance to allow a few close constraints.
        thresh = M - max(eps_active * M, 1e-12)
        active_idxs = np.where(conv_f >= thresh)[0]
        # Ensure at least the peak index is present.
        if active_idxs.size == 0:
            active_idxs = np.array([conv_f.argmax()], dtype=int)

        # Build sparse matrix A where each row corresponds to a constraint
        # (f * g)_k <= M.
        rows = []
        cols = []
        data = []
        for row_id, k in enumerate(active_idxs):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_vals = np.arange(i_start, i_end + 1, dtype=int)
            col_vals = k - i_vals
            rows.extend([row_id] * len(i_vals))
            cols.extend(col_vals.tolist())
            data.extend(f[i_vals].tolist())
        A = csr_matrix((data, (rows, cols)), shape=(len(active_idxs), n))

        # Linear program: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        b_ub = np.full(len(active_idxs), M, dtype=float)

        bounds = [(0.0, None)] * n

        # Solve with HiGHS (fast and robust)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception as e:  # pragma: no cover
            print("[LP] linprog raised:", e)
            break

        if not res.success:
            # If the LP fails (e.g. numerical issues) fall back to a tiny step.
            print(f"[LP {iteration}] infeasible – skip")
            # Reduce mixing factor and continue
            t_mix = max(t_mix * 0.7, min_t)
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            print(f"[LP {iteration}] got zero sum – skip")
            t_mix = max(t_mix * 0.7, min_t)
            continue

        # Normalise g to unit sum
        g_norm = g / sum_g

        # Mix current f with g_norm
        f_new = (1.0 - t_mix) * f + t_mix * g_norm

        # Re‑normalise to avoid drift
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-12:
            f_new = np.full(n, 1.0 / n)
        else:
            f_new /= s

        # Evaluate the new candidate
        new_val = _fast_evaluate(f_new)
        if new_val < best_val - 1e-12:
            best_val = new_val
            best_f = f_new.copy()
            f = f_new   # accept update
            print(f"[iter {iteration}] improved to {best_val:.6f} (t={t_mix:.3f}, active={len(active_idxs)})")
            # slightly increase mixing factor if progress good
            t_mix = min(t_mix * 1.05, 0.9)
        else:
            # No improvement – shrink mixing factor
            t_mix = max(t_mix * 0.7, min_t)
            # keep the old f unchanged
            # Optionally we could still replace f with f_new to explore,
            # but we stay conservative.
            if iteration % 10 == 0:
                print(f"[iter {iteration}] no improvement (val={new_val:.6f}), t set to {t_mix:.3f}")

        # Occasionally re‑evaluate the baseline without mixing to guard against stagnation
        if iteration % 30 == 0:
            cur_val = _fast_evaluate(f)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_f = f.copy()
                print(f"[iter {iteration}] baseline re‑eval improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final cheap hill‑climbing refinement (swap / mass transfer)
    # ------------------------------------------------------------------
    def _refine_hill_climb(a: np.ndarray, cur_val: float,
                           deadline: float) -> tuple[np.ndarray, float]:
        rng = np.random.default_rng(seed + 12345)
        n = a.shape[0]
        best_a = a.copy()
        best_val = cur_val
        while time.time() < deadline:
            if rng.random() < 0.2:                     # pure swap
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                new_a = best_a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]
            else:                                      # mass transfer
                i, j = rng.integers(0, n, size=2)

"""
Propose a non‑negative step‑function that lowers the max‑convolution bound.

The algorithm starts from a reasonable baseline (the previous best
construction if it exists, otherwise a uniform vector) and repeatedly
solves a tiny linear programme:

    maximise   Σ g_i
    subject to (f * g)_k ≤ M  for the most restrictive lag indices k
               g_i ≥ 0

where `f` is the current best sequence, `M = max(f*f)` and `*` denotes the
linear convolution.  The LP‑solution `g` is normalised to unit sum and
mixed into `f` (convex combination).  Because `g` carries as much mass as
possible without increasing any of the active convolution constraints,
the mixed sequence strictly reduces the maximal self‑convolution after
a few iterations.  A final cheap hill‑climbing pass (swap / mass
transfer) polishes the result.

All operations keep the vector on the probability simplex, the
evaluation uses the same formula as ``evaluate_sequence`` (but faster)
and the routine respects the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)                # clip tiny negatives
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` for vectors that sum to 1.

    Returns 2·n·max(conv) because the denominator Σa_i = 1.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climb that only accepts strict improvements."""
    rng = np.random.default_rng(123456)
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            delta = best_a[i] * 0.05 * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        # project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s
        val = _fast_evaluate(new_a)
        if val < best_val - 1e-12:
            best_a = new_a
            best_val = val
            print(f"[refine] new best = {best_val:.6f}")
    return best_a, best_val

    The routine respects the allotted time budget and returns the best
    sequence it found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin for final print

    # ------------------------------------------------------------------
    # Problem size – use the length of the known construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # Initialise a feasible probability vector.
    # ------------------------------------------------------------------
    if has_old:
        f = old.copy()
        if f.sum() < 1e-12:
            f = np.full(n, 1.0 / n)
        else:

        print(f"[init] Using supplied height_sequence_1 (n={n})")
    else:

        print(f"[init] No previous construction – start with uniform vector (n={n})")

    best_f = f.copy()
    best_val = _fast_evaluate(best_f)

    # ------------------------------------------------------------------
    # Core iterative LP‑based refinement.
    # ------------------------------------------------------------------
    t_mix = 0.5          # initial mixing factor
    min_t = 0.05
    max_active = 200    # at most this many lag indices are used as constraints
    it = 0

        it += 1
        conv_f = _full_convolution(f)
        M = conv_f.max()

        # Select the most restrictive lag indices (largest convolution values)
        K = min(max_active, conv_f.shape[0])
        # argpartition gives the K largest values without full sorting
        top_idxs = np.argpartition(-conv_f, K - 1)[:K]

        # Build sparse matrix A for constraints (f * g)_k ≤ M, one row per k∈top_idxs
        rows, cols, data = [], [], []
        for row_id, k in enumerate(top_idxs):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_vals = np.arange(i_start, i_end + 1, dtype=int)
            col_vals = k - i_vals

            cols.extend(col_vals.tolist())
            data.extend(f[i_vals].tolist())

        # Linear programme: maximise Σ g_i   (⇔ minimise -Σ g_i)
        c = -np.ones(n)

        f_new = np.maximum(f_new, 0.0)

            # If things go well we can be a little more aggressive

        else:

            if it % 10 == 0:

            if val_cur < best_val - 1e-12:
                best_val = val_cur
                best_f = f.copy()
                print(f"[iter {it}] baseline re‑eval improved to {best_val:.6f}")

    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_f.tolist()
