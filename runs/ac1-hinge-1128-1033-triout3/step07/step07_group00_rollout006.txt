"""
Propose a non‑negative step‑function (list of floats) that minimises
`evaluate_sequence`.  

The algorithm works in two phases:

1.  **Linear‑programming refinement** – starting from a reasonable
    distribution (uniform or an interpolated historic solution) we
    repeatedly solve a small LP:
        maximize   Σ g_i
        subject to  g_i ≥ 0
                    (f * g)_k ≤ M  for every convolution index k,
    where `f` is the current sequence, `M = max(f*f)` and `*` denotes the
    linear (zero‑padded) convolution.  The LP is implemented with
    `scipy.optimize.linprog` and a **fixed sparsity pattern** that is
    reused each iteration, making each solve very cheap (O(n²) memory,
    O(n) non‑zeros).

    The solution `g` is normalised to unit sum and mixed into the
    current sequence: `f ← (1‑t)·f + t·g` with a step size `t` that is
    adapted if the objective does not improve.  This is the exact
    discrete analogue of the iterative scheme described in the problem
    statement and typically beats the earlier gradient–based approach.

2.  **Local polishing** – after the LP loop a short targeted
    hill‑climber (same idea as in the previous code) moves a tiny
    amount of mass from indices that contribute heavily to the maximal
    autocorrelation entry to indices that contribute little.  This
    cleans up any remaining “spikes”.

Both phases respect the global time budget (default 1000 s).  The final
vector is returned as a plain Python list; `evaluate_sequence` will
automatically normalise it (the sum is kept equal to 1 during the
search, making the objective homogeneous).
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution helpers – full linear convolution using FFT (fast for n≈2000)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (zero‑padded), returned as a 1‑D array."""
    n = a.shape[0]
    out_len = 2 * n - 1
    # Next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may produce tiny negative values – clip them.
    conv = np.clip(conv, 0.0, None)
    return conv


def _objective_val(a: np.ndarray) -> float:
    """Exact objective for a vector with Σa = 1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Targeted hill‑climber (exact objective) – same as in the previous script
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """
    Fast local search moving a small amount of mass from indices that
    contribute most to the current maximal convolution entry to indices
    that contribute little.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – proportional to contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – usually where the reversed contribution is tiny
        if np.random.rand() < 0.35:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:          # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and restart from the best seen so far
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a list suitable for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 2000                     # moderate size to keep LP cheap

    # ------------------------------------------------------------------
    # Initialise a – use historic solution if available, otherwise uniform.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    except Exception:
        a = np.full(n, 1.0 / n)

    best_a = a.copy()
    best_val = _objective_val(best_a)
    TARGET = 1.503

    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute the sparsity pattern of the convolution matrix.
    # For a given f, the matrix (f * g) can be written as A(f) @ g,
    # where A has the same non‑zero pattern for every iteration.
    # ------------------------------------------------------------------
    nnz = n * n
    col = np.arange(nnz, dtype=np.int32) // n          # column index (repeated)
    offset = np.arange(nnz, dtype=np.int32) % n       # which entry of f to use
    row = col + offset                                 # row index
    A_shape = (2 * n - 1, n)                           # (2n‑1) × n
    # Allocate a CSR matrix with dummy data; we will replace .data each loop.
    A_ub = sp.csr_matrix((np.empty(nnz, dtype=np.float64), (row, col)), shape=A_shape)

    # ------------------------------------------------------------------
    # Phase A – linear‑programming driven improvement.
    # ------------------------------------------------------------------
    t = 0.4                    # mixing coefficient
    min_t = 0.02
    no_improve = 0
    IMPROVE_LIMIT = 3000       # after this many stagnating steps we random‑restart

    while time.time() < deadline and t >= min_t:
        # Current maximal convolution value
        conv_a = _full_convolution(a)
        M = conv_a.max()
        cur_obj = 2.0 * n * M
        if cur_obj < best_val - 1e-12:
            best_val, best_a = cur_obj, a.copy()
            print(f"[LP] new best = {best_val:.6f}")

        # ---- build the LP (replace only the data array) ----
        A_ub.data = a[offset]     # f_i = a[i]; offset repeats the pattern
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise Σ g_i  →  minimise −Σ g_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            # In rare cases linprog may fail; break out to fallback.
            print("[LP] linprog failed – exiting LP phase")
            break

        g = res.x
        s = g.sum()
        if s <= 0.0:
            break
        g = g / s                     # renormalise to unit sum

        # ---- convex combination ----
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()           # keep Σ = 1

        new_obj = _objective_val(a_new)

        if new_obj < cur_obj:
            a = a_new
            t = min(t * 1.1, 0.9)     # modestly increase step size after success
            no_improve = 0
        else:
            # no improvement → shrink step size
            t = max(t * 0.5, min_t)
            no_improve += 1

        if new_obj < best_val:
            best_val, best_a = new_obj, a.copy()
            print(f"[LP] improved to {best_val:.6f}")

        # early exit if target reached
        if best_val <= TARGET:
            print("[LP] target achieved")
            break

        if no_improve >= IMPROVE_LIMIT:
            # random restart – draw a fresh Dirichlet vector
            a = np.random.default_rng().dirichlet(np.ones(n))
            t = 0.4
            no_improve = 0
            print("[LP] random restart after stagnation")

    # ------------------------------------------------------------------
    # Phase B – short hill‑climbing refinement (exact objective)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        # allocate about half of the remaining budget to hill‑climbing
        hill_deadline = time.time() + 0.5 * remaining
        best_a = _hillclimb_targeted(best_a, n,
                                     start=time.time(),
                                     deadline=hill_deadline,
                                     best_a=best_a,
                                     best_val=best_val,
                                     target_val=TARGET)
        best_val = _objective_val(best_a)
        print(f"[Hill] post‑LP best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Small uniform mixing sweep (optional final polish)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand = (1.0 - alpha) * best_a + alpha * uniform
        cand /= cand.sum()
        val = _objective_val(cand)
        if val < best_val:
            best_val, best_a = val, cand
            print(f"[Mix] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided secure wrapper.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
