def simplex_projection(v):
    v = np.asarray(v)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def lp_improve(a, M, t=0.1):
    n = len(a)
    # Build A with toeplitz
    c = np.concatenate([a, np.zeros(n-1)])
    r = np.concatenate([a[:1], np.zeros(n-1)])
    A = scipy.linalg.toeplitz(c, r)  # shape (2n-1, n)
    # constraints: A @ g <= M
    # Objective: maximize sum(g)
    # Because linprog minimizes, we set c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(2*n-1, M)
    res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a  # no improvement!
    g = res.x
    if g.sum() <= 0:
        return a
    g = g / g.sum()
    new_a = (1-t) * a + t * g
    new_a = np.maximum(new_a, 0)
    new_a /= new_a.sum()
    return new_a

"""
Propose a step‑function that minimises the given evaluation function.

The algorithm consists of three complementary phases:

1. **Seeding** – a few hand‑crafted families (geometric, power‑law) and
   random Dirichlet draws are normalised and the best among them is kept.

2. **Deterministic sub‑gradient descent** – the objective is
   `max_k (a*a)[k]` (the sum of the vector is fixed to 1).  At the current
   maximal lag `k*` the sub‑gradient can be written in closed form.
   A step is taken opposite to the sub‑gradient and the result is projected
   onto the probability simplex.  A decaying step size yields monotone
   improvements.

3. **LP‑based refinement** – given a current vector `a` and its maximal
   convolution value `M = max(a*a)`, we solve a linear program

       maximise   1ᵀg
       subject to  a * g ≤ M   (elementwise convolution)
                   g ≥ 0

   The solution `g` is scaled to have unit sum and mixed with `a`
   (`a ← (1‑t) a + t g`).  Repeating this step strictly reduces the
   maximal convolution value.

The routine respects the 1000 s budget, prints intermediate scores and
returns the best sequence found (as a plain Python list).  The length of
the sequence is chosen as a power of two (2048) to speed up the FFT‑based
convolution evaluator.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper functions (FFT convolution, fast evaluation, projections …)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clamp tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast replica of the supplied `evaluate_sequence` (non‑negative, sum≈1)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find the largest rho such that u_i - cssv_i / i > 0
    mask = u - cssv / ind > 0
    if not np.any(mask):
        theta = 0.0
    else:
        rho = np.nonzero(mask)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry at lag k.
    Gradient component i: ∂/∂a_i Σ_j a_j a_{k-j}
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=int)
    j = k - i
    grad[i] += a[j]                     # contribution a_i * a_{k-i}
    mask = i != j
    if np.any(mask):
        grad[j[mask]] += a[i[mask]]      # symmetric contribution
    return grad


# ----------------------------------------------------------------------
# Parametric families for seeding
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """q∈(0,1): a_i ∝ q**i, normalised to sum = 1."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q**n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """p<0: a_i ∝ (i+1)**p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# LP refinement (step 3 of the algorithm)
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray, M: float, mix: float = 0.1) -> np.ndarray:
    """
    Solve  maximise 1ᵀg   s.t.   a * g ≤ M   (convolution, elementwise),
    g ≥ 0, then mix the solution with `a` using coefficient `mix`.
    """
    n = a.shape[0]
    # Build the (2n‑1) × n Toeplitz matrix that encodes the convolution a*g.
    c = np.concatenate([a, np.zeros(n - 1, dtype=float)])
    r = np.concatenate([a[:1], np.zeros(n - 1, dtype=float)])
    A = toeplitz(c, r)                   # shape (2n‑1, n)

    # Linear program: minimise –1ᵀg  (i.e. maximise 1ᵀg)
    c_obj = -np.ones(n, dtype=float)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'disp': False})
    if not res.success:
        return a  # nothing changed

    g = res.x
    if g.sum() <= 0.0:
        return a
    g = g / g.sum()                     # normalise to unit sum
    new_a = (1.0 - mix) * a + mix * g
    new_a = np.maximum(new_a, 0.0)
    new_a = new_a / new_a.sum()
    return new_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a power of two makes the FFT cheap.
    # ------------------------------------------------------------------
    n = 2048

    # ------------------------------------------------------------------
    # Initialise from the previously saved construction (if any).
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1 – Diverse seeds (geometric, power‑law, random Dirichlet)
    # ------------------------------------------------------------------
    seeds = []

    for q in np.linspace(0.90, 0.9999, 15):
        seeds.append(_geometric_sequence(q, n))
    for p in np.linspace(-2.5, -0.2, 15):
        seeds.append(_power_sequence(p, n))
    for _ in range(20):
        s = np.random.rand(n)
        s /= s.sum()
        seeds.append(s)

    for s in seeds:
        s = np.maximum(s, 0.0)
        s /= s.sum()
        val = _eval_fast(s)
        if val < best_val:
            best_val, best_a = val, s.copy()
            print(f"[seed] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2 – Sub‑gradient descent on the max‑convolution term.
    # ------------------------------------------------------------------
    step = 0.5                     # initial (scaled) step size
    no_improve = 0
    max_no_improve = 2000
    while time.time() < deadline - 30.0:
        conv = _fft_convolve_self(best_a)
        k_max = int(np.argmax(conv))
        grad = _max_conv_gradient(best_a, k_max)
        grad_norm = np.linalg.norm(grad) + 1e-16

        # Scaled step – keep the update magnitude moderate.
        scaled_step = step / grad_norm
        cand = _simplex_projection(best_a - scaled_step * grad)
        cand_val = _eval_fast(cand)

        if cand_val < best_val:
            best_val, best_a = cand_val, cand
            print(f"[subgrad] new best = {best_val:.6f}")
            step = min(step * 1.05, 2.0)   # a slight step‑size increase
            no_improve = 0
        else:
            step *= 0.90
            no_improve += 1
            if no_improve >= max_no_improve:
                # Random tiny perturbation to escape possible flat region.
                i, j = np.random.randint(0, n, size=2)
                if best_a[i] > 0:
                    delta = best_a[i] * 0.02 * np.random.rand()
                    tmp = best_a.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    best_a = _simplex_projection(tmp)
                    best_val = _eval_fast(best_a)
                    print(f"[subgrad] random perturb, val = {best_val:.6f}")
                no_improve = 0

    # ------------------------------------------------------------------
    # 3 – Linear‑programming refinement (monotone improvement of max‑conv).
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:
        conv = _fft_convolve_self(best_a)
        M = conv.max()
        new_a = _lp_improve(best_a, M, mix=0.1)
        new_val = _eval_fast(new_a)
        if new_val < best_val - 1e-9:
            best_val, best_a = new_val, new_a
            print(f"[LP] improved to {best_val:.6f}")
        else:
            # Try a smaller mixing coefficient if the full 0.1 step fails.
            improved = False
            for mix in (0.05, 0.02, 0.01):
                new_a = _lp_improve(best_a, M, mix=mix)
                new_val = _eval_fast(new_a)
                if new_val < best_val - 1e-9:
                    best_val, best_a = new_val, new_a
                    print(f"[LP‑mix {mix:.2f}] improved to {best_val:.6f}")
                    improved = True
                    break
            if not improved:
                break   # cannot improve any further

    # ------------------------------------------------------------------
    # 4 – Final cheap hill‑climbing (swap / small transfer).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        if np.random.rand() < 0.3:          # swap two positions
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                cand = best_a.copy()
                cand[i], cand[j] = cand[j], cand[i]
        else:                               # small mass transfer
            i, j = np.random.randint(0, n, size=2)
            if best_a[i] > 0:
                delta = best_a[i] * 0.02 * np.random.rand()
                cand = best_a.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        cand = _simplex_projection(cand)
        cand_val = _eval_fast(cand)
        if cand_val < best_val:
            best_val, best_a = cand_val, cand
            print(f"[hill] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation (using the supplied function).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] exact evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    return best_a.tolist()
