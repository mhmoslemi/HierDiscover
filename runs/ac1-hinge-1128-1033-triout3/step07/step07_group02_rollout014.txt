"""
Propose a non‑negative step function that minimises the supplied
`evaluate_sequence` objective.

Main ideas
----------
1. Use a modest problem size (n≈2000) – this makes each iteration cheap,
   allowing many independent optimisation runs within the 1000 s budget.
2. Initialise candidates with a diverse set of seeds:
   * random Dirichlet vectors,
   * Beta‑shaped families with a very wide log‑uniform range for the
     exponents (0.01 … 10),
   * optionally the previously found solution (if `height_sequence_1`
     exists).
3. For every seed run a *short* Adam optimiser (smooth‑max surrogate)
   followed by a *short* simulated‑annealing hill‑climber.
   The best candidate of all short runs is kept.
4. With the remaining time perform a *long* Adam optimisation and a
   final hill‑climb on the global best candidate.
5. The whole routine is time‑aware – it always leaves a 0.5 s safety
   margin for the final call to `evaluate_sequence`.

All helper functions are top‑level, there are no closures or lambdas.
The code uses only `numpy` and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient w.r.t. a
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and its gradient.
    The returned gradient corresponds to d(2·n·max)/da.
    """
    n = a.shape[0]
    b = _full_convolution(a)               # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()            # soft‑max weights (2n‑1,)
    # gradient of Σ p_k * b_k w.r.t a.
    # Using the convolution property: ∂b_k/∂a_i = a_{k-i} + a_{i-k}
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)          # length 3n‑2, we need the central n entries
    segment = conv_ab[n - 1:2 * n - 1]      # n entries
    grad_a = 4.0 * n * segment[::-1]       # factor 4·n from 2·n·max
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parameterisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.48,
                   max_iters: int = None):
    """
    Adam optimiser that minimises the smooth approximation of the objective.
    Parameters
    ----------
    a0 : initial point (already normalised to sum‑1)
    n  : length of the vector
    deadline : absolute time (seconds) after which we must stop
    start_time : time when the whole optimisation started
    best_a, best_val : current global best (updated in‑place)
    target_val : early‑stop target (optional)
    max_iters : stop after this many iterations (useful for short runs)
    Returns
    -------
    best_a, best_val   (the best seen during this call)
    """
    # logits that will be fed to soft‑max → a on the simplex
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.25, 0.001          # a little larger initial LR
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 5000                        # restart after many idle steps
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        if max_iters is not None and iteration >= max_iters:
            break

        # ---- a from logits -------------------------------------------------
        w_shift = w - np.max(w)            # for numerical stability
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam iter={iteration:7d}] new global best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # ---- schedules ----------------------------------------------------
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ---- smooth‑max and gradient --------------------------------------
        _, grad_a = _smooth_max_and_grad(a, eps)

        # chain rule through soft‑max (grad w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---- Adam update --------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter to escape flat regions
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.02, size=n)

        # restart if stagnating for a long time
        if idle > MAX_IDLE:
            print(f"[Adam] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.48,
                      max_iters: int = None):
    """
    Local search that moves a small amount of mass from an index that
    contributes heavily to the current maximal convolution entry to an
    index that contributes little.  Accepts uphill moves with a Metropolis
    probability that cools down linearly.
    """
    a = a0.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules (wide at the start, shrink to zero)
    max_frac_start, max_frac_end = 0.4, 0.0005
    temp_start, temp_end = 0.015, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 80000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        if max_iters is not None and iteration >= max_iters:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # centre of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of every index i to b[k_max] (i contributes a_i * a_{k_max-i})
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ---- source index (pick proportionally to its contribution) -----
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            iteration += 1
            continue

        # ---- target index (biased towards low contribution) -------------
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move (relative to source)
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            iteration += 1
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        # the block a contributes to b at offset i and j respectively
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic terms for the moved mass
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis decision
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill iter={iteration:7d}] new global best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – random perturbation of the whole vector
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill iter={iteration:7d}] stagnation – perturbation")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Beta‑family helper (used for diverse initialisations)
# ----------------------------------------------------------------------
def _beta_random(n: int) -> np.ndarray:
    """
    Return a random Beta‑shaped vector:
        a_i ∝ (i+1)^{p-1} (n-i)^{q-1}
    where p,q are drawn log‑uniformly from [0.01, 10].
    """
    # log‑uniform sampling
    logp = np.random.uniform(np.log10(0.01), np.log10(10.0))
    logq = np.random.uniform(np.log10(0.01), np.log10(10.0))
    p = 10 ** logp
    q = 10 ** logq
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.random.rand(n)
    a /= a.sum()
    return a

# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under the 1000 s budget.
    Returns a list of floats whose sum is 1 (the scale does not affect
    the final score because the objective is scale‑invariant).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 0.5 s safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries strike a good speed/quality
    # trade‑off.  2000 works comfortably within the time budget.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Global best holder (may be initialised from a previous run)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # Try to reuse a previously found construction if it exists.
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        if old_len != n:
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        best_a, best_val = a0, val
        print(f"[init] previous solution imported, value={best_val:.6f}")
    except Exception:
        pass

    # If no previous solution, start from a random vector.
    if best_a is None:
        a0 = np.random.rand(n)
        a0 /= a0.sum()
        best_a, best_val = a0, _objective(a0)
        print(f"[init] random start, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – quick multi‑restart exploration (≈40 % of budget)
    # ------------------------------------------------------------------
    multi_deadline = start_time + 0.40 * (deadline - start_time)
    iteration = 0
    while time.time() < multi_deadline:
        # --- pick a diverse seed ---------------------------------------
        if np.random.rand() < 0.5:
            a_seed = np.random.rand(n)
            a_seed /= a_seed.sum()
        else:
            a_seed = _beta_random(n)

        # short Adam (max 2500 iterations)
        a_opt, _ = _adam_optimize(a_seed, n,
                                   deadline=multi_deadline,
                                   start_time=start_time,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=1.48,
                                   max_iters=2500)

        # short hill‑climb (max 4000 iterations)
        a_opt, _ = _hillclimb_anneal(a_opt, n,
                                      deadline=multi_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.48,
                                      max_iters=4000)

        cur_val = _objective(a_opt)
        if cur_val < best_val:
            best_a, best_val = a_opt.copy(), cur_val
            print(f"[multi iter={iteration}] new global best = {best_val:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Phase A – long Adam optimisation on the current best (≈30 % budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * (deadline - start_time) + start_time
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.48)

    # ------------------------------------------------------------------
    # Phase B – final hill‑climb (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=start_time,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.48)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
