def _mass_transfer_hillclimb(a0, deadline, rng, max_no_improve=5000, cand_trials=5):
   a = a0.copy()
   n = a.shape[0]
   cur_val = _objective(a)
   best_a = a.copy()
   best_val = cur_val
   step_frac = 0.2
   no_improve = 0

   while time.time() < deadline and no_improve < max_no_improve:
       conv = _full_convolution(a)
       maxb = conv.max()
       ks = np.flatnonzero(conv == maxb)
       k = int(rng.choice(ks))

       i_start = max(0, k - (n - 1))
       i_end = min(k, n - 1)
       i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
       j_range = k - i_range
       contrib = a[i_range] * a[j_range]
       idx = int(np.argmax(contrib))
       i = int(i_range[idx])
       j = int(j_range[idx])
       if i == j:
           delta = a[i] * step_frac
       else:
           delta = min(a[i], a[j]) * step_frac
       if delta <= 0.0:
           step_frac = max(step_frac * 0.8, 1e-6)
           continue

       # create base new_a after removing delta
       base_a = a.copy()
       if i == j:
           base_a[i] -= delta
       else:
           base_a[i] -= delta
           base_a[j] -= delta

       # candidate destinations
       possible = np.arange(n)
       forbidden = {i, j}
       cand_pool = np.setdiff1d(possible, list(forbidden), assume_unique=True)
       # optionally restrict to low mass
       low_mask = base_a < np.mean(base_a)
       low_indices = np.where(low_mask)[0]
       cand_pool = np.intersect1d(cand_pool, low_indices, assume_unique=True)
       if cand_pool.size < 2:
           cand_pool = np.setdiff1d(possible, list(forbidden), assume_unique=True)

       best_candidate_val = cur_val
       best_dest = None
       # try cand_trials random pairs
       for _ in range(cand_trials):
           if cand_pool.size < 2:
               break
           dest = rng.choice(cand_pool, size=2, replace=False)
           p, q = int(dest[0]), int(dest[1])
           new_a = base_a.copy()
           new_a[p] += delta
           new_a[q] += delta
           new_a_sum = new_a.sum()
           if new_a_sum <= 0.0:
               continue
           new_a /= new_a_sum
           new_val = _objective(new_a)
           if new_val < best_candidate_val:
               best_candidate_val = new_val
               best_dest = (p, q, new_a)

       if best_dest is not None and best_candidate_val < cur_val:
           a = best_dest[2]
           cur_val = best_candidate_val
           no_improve = 0
           step_frac = min(step_frac * 1.02, 0.5)
           if cur_val < best_val:
               best_val = cur_val
               best_a = a.copy()
               print(f"[massclimb] new best = {best_val:.6f}")
       else:
           step_frac = max(step_frac * 0.9, 1e-6)
           no_improve += 1

   return best_a, best_val

        grad[i_range] = 2.0 * p[j_range]

        eta = eta0 / np.sqrt(t)

        maxlog = np.max(logp)
        p = np.exp(logp - maxlog)
        sum_p = p.sum()
        if sum_p == 0.0:

            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

"""
Optimisation of a step‑function (probability) vector for the
objective 2·n·max_k (p∗p)[k].

The algorithm consists of two stages

1. **Exponentiated‑gradient descent** on the exact max‑convolution.
   At each iteration we locate a lag k attaining the maximum,
   compute the sub‑gradient of (p∗p)[k] w.r.t. the probability vector
   and perform an entropic (multiplicative‑weights) update.
   The update keeps all entries non‑negative and automatically projects
   onto the probability simplex.

2. **Aggressive mass‑transfer hill‑climbing** (an enhanced version of
   the hill‑climb used in the previous state‑of‑the‑art).  In each
   step the pair (i,j) that contributes the most to the current
   maximal lag is identified; a fraction of its mass is moved to two
   new positions chosen from a pool of low‑mass indices.  Several
   random destination pairs are tried and the best improvement is kept.
   The step‑size (fraction of moved mass) is adapted dynamically.

Both stages are run repeatedly until the supplied time budget (default
1000 s) is exhausted, with occasional random restarts to escape
local minima.  The best vector found is finally evaluated with the
provided ``evaluate_sequence`` function and returned as a plain Python
list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # eliminate tiny negative artefacts from floating point noise
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` when sum(a)=1.
    (The evaluator discards the overall scale of the vector.)
    """
    n = a.shape[0]
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (entropic mirror descent)
# ----------------------------------------------------------------------


def _expgradient_descent(p0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 50000,
                         eta0: float = 2.5,
                         patience: int = 5000) -> tuple[np.ndarray, float]:
    """
    Entropic mirror descent on  f(p)=2·n·max(p∗p).

    Returns the best probability vector seen and its objective value.
    """
    p = p0.copy()
    n = p.shape[0]
    best_p = p.copy()
    best_val = _objective(p)

    no_improve = 0
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _full_convolution(p)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # sub‑gradient of the maximal lag
        grad = np.zeros_like(p)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        grad[i_range] = 2.0 * p[j_range]

        # Entropic update: p_i ← p_i * exp(-η·grad_i)
        eta = eta0 / np.sqrt(t)
        logp = np.log(p + 1e-30) - eta * grad
        # re‑normalise in log‑space for numerical stability
        maxlog = np.max(logp)
        p = np.exp(logp - maxlog)
        s = p.sum()
        if s == 0.0:
            p = np.full_like(p, 1.0 / n)
        else:
            p /= s

        cur_val = 2.0 * n * maxb   # reuse the max we already have
        if cur_val < best_val:
            best_val = cur_val
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # occasional progress report
        if t % 5000 == 0:
            print(f"[expgrad] iter {t:5d}  best = {best_val:.6f}")

        if no_improve >= patience:
            # stop early if we have not improved for a while
            break

    return best_p, best_val


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill‑climbing (enhanced version)
# ----------------------------------------------------------------------


def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 5000,
                             cand_trials: int = 6) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that repeatedly moves a fraction of the most damaging
    mass to low‑mass positions.  Several random destination pairs are
    examined and the best improvement is kept.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)

    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2               # initial fraction of mass to relocate
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that constitute the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we are willing to shift
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-7)
            continue

        base_a = a.copy()
        if i == j:
            base_a[i] -= delta
        else:
            base_a[i] -= delta
            base_a[j] -= delta

        # candidate destination pool (prefer low‑mass indices)
        all_idx = np.arange(n)
        forbidden = {i, j}
        cand_pool = np.setdiff1d(all_idx, list(forbidden), assume_unique=True)

        low_mask = base_a < np.mean(base_a)
        low_cands = np.where(low_mask)[0]
        # intersect with the general pool – if too few remain fall back
        cand_pool = np.intersect1d(cand_pool, low_cands, assume_unique=True)
        if cand_pool.size < 2:
            cand_pool = np.setdiff1d(all_idx, list(forbidden), assume_unique=True)

        best_candidate_val = cur_val
        best_candidate_a = None

        # try a handful of random destination pairs
        for _ in range(cand_trials):
            if cand_pool.size < 2:
                break
            dest = rng.choice(cand_pool, size=2, replace=False)
            p, q = int(dest[0]), int(dest[1])
            new_a = base_a.copy()
            new_a[p] += delta
            new_a[q] += delta
            s = new_a.sum()
            if s <= 0.0:
                continue
            new_a /= s
            new_val = _objective(new_a)
            if new_val < best_candidate_val:
                best_candidate_val = new_val
                best_candidate_a = new_a

        if best_candidate_a is not None and best_candidate_val < cur_val:
            a = best_candidate_a
            cur_val = best_candidate_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-7)
            no_improve += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.

    The function follows the two‑stage scheme described above,
    performs a few random restarts when time permits, and finally
    returns the best list of heights found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Initialise – reuse a previously found construction if available.
    # ------------------------------------------------------------------
    default_n = 4000
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        a0 = np.maximum(old, 0.0)
        if a0.sum() == 0.0:
            a0 = rng.random(n)
        a0 /= a0.sum()
    except Exception:
        n = default_n
        a0 = rng.random(n)
        a0 /= a0.sum()

    print(f"[init] n={n}, initial objective = {_objective(a0):.6f}")

    # ------------------------------------------------------------------
    # 1. Exponentiated‑gradient descent from the initial vector.
    # ------------------------------------------------------------------
    a_exp, val_exp = _expgradient_descent(a0, deadline, rng, max_iters=40000)
    print(f"[expgrad] best value after descent = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # 2. Hill‑climbing refinement.
    # ------------------------------------------------------------------
    a_cur, val_cur = _mass_transfer_hillclimb(a_exp, deadline, rng)
    print(f"[hillclimb] best value after hill‑climb = {val_cur:.6f}")

    best_a = a_cur
    best_val = val_cur

    # ------------------------------------------------------------------
    # 3. Random‑restart loop (if time permits).
    # ------------------------------------------------------------------
    restart_id = 0
    while time.time() < deadline - 3.0:  # keep a few seconds for final reporting
        remaining = deadline - time.time()
        # allocate a modest amount of work for each restart
        max_iters = int(20000 * remaining / budget_s) + 1

        # perturb the current best a bit to start a new run
        eps = 0.07
        pert = best_a + eps * (rng.random(n) - 0.5)
        pert = np.clip(pert, 0.0, None)
        if pert.sum() == 0.0:
            pert = rng.random(n)
        pert /= pert.sum()

        a_tmp, val_tmp = _expgradient_descent(pert, deadline, rng,
                                              max_iters=max_iters,
                                              eta0=2.5,
                                              patience=3000)
        a_tmp, val_tmp = _mass_transfer_hillclimb(a_tmp, deadline, rng)

        if val_tmp < best_val:
            best_val = val_tmp
            best_a = a_tmp.copy()
            print(f"[restart {restart_id}] new overall best = {best_val:.6f}")

        restart_id += 1

    # ------------------------------------------------------------------
    # 4. Final evaluation and return.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
