"""
Heuristic optimiser for the ``evaluate_sequence`` objective.

The optimiser works in four independent phases, each respecting the
overall time budget:

1. **Smooth surrogate minimisation** – the non‑smooth max‑convolution
   is replaced by a log‑sum‑exp (smooth‑max) surrogate.  A continuation
   in the smoothing parameter ε together with L‑BFGS‑B yields a very
   good approximate minimiser.

2. **Random restarts** – the best solution from phase 1 is repeatedly
   perturbed and re‑optimised (again on the smooth surrogate) to escape
   possible local minima.

3. **Targeted simulated‑annealing** – many cheap moves are performed.
   Each move transfers a small amount of mass away from the lag that
   currently attains the maximal autocorrelation.  Moves are accepted
   if they improve the exact objective or with a probability that
   decreases with a linear temperature schedule.

4. **Fine sub‑gradient polishing** – a classic sub‑gradient descent
   on the exact objective 2·n·max(conv) (with projection onto the
   probability simplex) is applied as a final polishing step.

The whole routine fits into the 1000 s time limit and returns the best
list of non‑negative heights found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient for L‑BFGS‑B
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Value and gradient of the smooth surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·eps·log Σ exp((conv‑max)/eps)

    where a = softmax(w) (∑a = 1, a_i ≥ 0).
    Returns (L, ∇_w L).
    """
    # soft‑max to obtain a feasible probability vector
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()

    # full autocorrelation
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing term
    shifted = (conv - max_conv) / eps
    log_sum = np.logaddexp.reduce(shifted)          # = log Σ exp(...)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # probabilities over lags

    # convolution a * rev(p)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                  # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]                # middle n entries
    grad_a = 4.0 * n * seg[::-1]                   # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w_vec, eps, n):
    """Thin wrapper required by scipy.minimize (no closure)."""
    return _smooth_obj_and_grad(w_vec, eps, n)


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (2·n·max(conv) for a normalised sequence)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray, n: int) -> float:
    """Exact value of evaluate_sequence for a vector whose sum equals 1."""
    return float(2.0 * n * np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Fine sub‑gradient descent on the exact non‑smooth objective
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Sub‑gradient descent on 2·n·max(conv) with projection onto the simplex.
    Returns the best vector and its value found before the deadline.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_val = cur_val
    eta = 0.1                     # initial learning rate

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient for the active lag k
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # gradient step + projection onto the simplex
        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = _objective_value(new_a, n)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            eta = min(eta * 1.05, 0.5)
        else:
            eta = max(eta * 0.7, 1e-5)

    return best_a, best_val

            j_range = k - i_range
            contrib = cur_a[i_range] * cur_a[j_range]

            # Choose a source index:
            if contrib.sum() > 0 and rng.random() < 0.7:
                probs = contrib / contrib.sum()
                src = rng.choice(i_range, p=probs)
            else:
                src = rng.choice(n, p=cur_a)   # fallback proportional to current mass

            # Choose a destination index (prefer a low‑mass position)
            low = np.where(cur_a < cur_a.mean())[0]
            if low.size > 0:
                cand = np.setdiff1d(low, [src])
                if cand.size > 0:
                    dst = rng.choice(cand)
                else:
                    dst = rng.integers(n - 1)
                    if dst >= src:
                        dst += 1
            else:
                dst = rng.integers(n - 1)
                if dst >= src:
                    dst += 1

            # Transfer a fraction of the source mass
            delta = cur_a[src] * rng.uniform(0.0, 0.25)
            if delta <= 0.0:
                continue

            new_a = cur_a.copy()
            new_a[src] -= delta
            new_a[dst] += delta

            new_val = _objective_value(new_a, n)

            # Accept if better; otherwise accept with probability exp(-(Δ)/T)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                cur_a = new_a
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_a = cur_a.copy()
                    print(f"[SA] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _subgradient_descent(best_a,
                                                best_val,
                                                deadline,
                                                rng)

    # ------------------------------------------------------------------
    # Final reporting (uses the provided evaluator)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
