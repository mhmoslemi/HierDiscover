"""
Propose a step‑function (non‑negative list) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1.  Initialise a candidate on the probability simplex.
    * If a previous solution ``height_sequence_1`` exists, interpolate it
      to the working size ``n``.
    * Otherwise start from a half‑sine shape `a_i ∝ sin(pi*(i+0.5)/n)`,
      which already gives a fairly flat self‑convolution.
    * A short Beta‑family sweep (few (p,q) pairs) is performed and the best
      seed is kept.

2.  Adam optimisation (smooth‑max surrogate of the true objective).
    * We optimise the soft‑max of the convolution using the log‑sum‑exp
      smoothing (temperature ``eps``) and a standard Adam update on the
      logits of a soft‑max parametrisation.
    * The optimiser runs for roughly 40 % of the total time budget.

3.  Exact sub‑gradient descent on the true objective.
    * At each iteration we locate the maximal convolution entry ``k_max``.
    * The gradient of the maximal entry is
          g_i = 2·a_{k_max-i}   (if the index is inside the array, else 0)
      – this is the exact sub‑gradient of ``max(conv)``.
    * We move opposite to the sub‑gradient while staying on the simplex:
          d = -g + mean(g)                # zero‑sum direction
          a ← a + α·d                     # α chosen so that a stays ≥0
      Afterwards we renormalise to keep ``∑a = 1``.
    * The step size decays like ``α = α0 / sqrt(t+1)``; α0 is tuned
      experimentally (0.5 works well).  The loop runs until the 1000 s
      budget is exhausted.

4.  The best vector found during the whole run is returned.
   ``evaluate_sequence`` works with any scaling, but we keep the
   vectors normalised to unit sum for numerical stability.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and gradient w.r.t. a (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and its gradient.
    The gradient corresponds to the exact objective 2·n·max(conv) ≈
    2·n·softmax, needed by Adam.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # gradient of Σ p_k * b_k w.r.t a
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 4n from 2·n·max
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a smooth approximation of the objective.
    Returns the best (a, val) seen before *deadline*.
    """
    # initialise logits for the soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.001
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedules
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max and gradient
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)

        # chain rule through the soft‑max (w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional perturbation to escape shallow minima
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        # restart on stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_a: np.ndarray,
                         best_val: float,
                         target_val: float = 1.45) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective 2·n·max(conv).
    Keeps the iterate on the probability simplex (Σa = 1, a ≥ 0).
    """
    a = a0.copy()
    iteration = 0
    idle = 0
    MAX_IDLE = 80_000          # after many idle steps we do a random restart
    alpha0 = 0.5                # initial step size factor

    while True:
        now = time.time()
        if now >= deadline:
            break

        # full convolution and current objective value
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[SubGrad {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # location of the (first) maximal entry
        k_max = int(np.argmax(b))

        # exact sub‑gradient of this entry
        start = max(0, k_max - (n - 1))
        end = min(k_max, n - 1)
        g = np.zeros_like(a)
        if start <= end:
            idx = np.arange(start, end + 1, dtype=int)
            g[idx] = 2.0 * a[k_max - idx]          # g_i = 2·a_{k_max-i}

        # zero‑sum direction (keeps Σa unchanged)
        d = -g + np.mean(g)

        # step‑size schedule (decays with sqrt of iteration count)
        alpha = alpha0 / np.sqrt(iteration + 1.0)

        # make sure we stay non‑negative:
        neg = d < 0
        if np.any(neg):
            # maximal feasible α so that a + α·d stays ≥ 0
            bound = np.min(a[neg] / (-d[neg]))
            # keep a safe margin
            alpha = min(alpha, 0.9 * bound)

        # update
        a = a + alpha * d

        # numerical safety: clip tiny negatives and renormalise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.random.rand(n)
            a /= a.sum()
        else:
            a /= s

        # random restart after a long stagnation phase
        if idle > MAX_IDLE:
            print(f"[SubGrad {iteration:7d}] stagnation – random restart")
            a = np.random.rand(n)
            a /= a.sum()
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric sweep of a simple Beta‑family (tiny grid – cheap for large n)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a very small Beta‑family for a quick seed."""
    best_a = None
    best_val = np.inf
    # small grid that typically captures useful shapes
    p_vals = np.array([0.3, 0.5, 1.0, 2.0, 4.0])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 124,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 (scale does not affect the final
    ``evaluate_sequence`` score).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final ``evaluate_sequence`` call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few‑ten thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 15000                     # larger than the previous 5000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previous construction if available.
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # If no previous solution, fall back to a half‑sine shape
    if a0 is None:
        idx = np.arange(n, dtype=float)
        a0 = np.sin(np.pi * (idx + 0.5) / n)   # strictly positive on (0,π)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] half‑sine seed created")

    # ------------------------------------------------------------------
    # Phase 0 – cheap Beta‑family sweep (tiny grid) for a possible boost.
    # ------------------------------------------------------------------
    best_a, best_val = a0.copy(), _objective(a0)
    try:
        shape_a, shape_val = _beta_family_best(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] beta‑family sweep improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.40 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.5)

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient descent (the non‑smooth objective).
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_descent(best_a, n,
                                            deadline=deadline,
                                            start_time=phase_a_deadline,
                                            best_a=best_a,
                                            best_val=best_val,
                                            target_val=1.45)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
