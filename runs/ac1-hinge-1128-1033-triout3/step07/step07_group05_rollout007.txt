"""
Hybrid optimiser for the step‑function bound problem.

The optimiser works in four (time‑budgeted) phases.

1. **Smooth‑max surrogate minimisation (L‑BFGS).**
   A log‑sum‑exp surrogate for `max(a*a)` is minimised while keeping
   the ℓ₁‑norm fixed (soft‑max parametrisation).  Several random
   restarts and an ε‑schedule from 5·10⁻³ down to 10⁻¹⁴ are used.
   After each round the current iterate is evaluated with the exact
   fast evaluator; the best iterate is retained.

2. **Deterministic re‑ordering.**
   The multiset of heights is rearranged in two ways – the previous
   “ends‑large” heuristic and a new *alternating‑ends* heuristic that
   places the largest values at the two extremes and proceeds
   inwards.  The better ordering is kept.

3. **Projected sub‑gradient descent on the true max‑convolution.**
   Starting from the best vector found so far, a simple sub‑gradient
   method (step size ∝ 1/√t) is run directly on the exact max‑convolution
   while projecting onto the simplex (non‑negative, ℓ₁=1).

4. **Simulated‑annealing + mass‑transfer hill‑climbing.**
   A final SA stage that mixes two elementary moves:
      * swap two entries,
      * transfer a random amount δ from one entry to another
        (preserving the ℓ₁‑norm).
   The temperature decays geometrically; accepted improvements are
   stored as the global best.  After the annealing a short greedy
   hill‑climb (tiny transfers only) squeezes out any remaining
   improvement.

All phases respect a total wall‑time of 1000 s.  The code uses only
NumPy and SciPy, works on a single CPU and never writes to disk.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the L‑BFGS surrogate
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padded FFT)."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Returns array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)  # clip possible tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast evaluator that computes exactly the same value as
    ``evaluate_sequence(a.tolist())`` (up to the clamping limits that are
    never reached in our searches because we keep the vector normalised).
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)          # = 1 for all our candidates
    if sum_a < 0.01:           # safety – should never happen
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Convolution helper used inside the smooth surrogate gradient
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where L is the smoothed surrogate

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and grad_w is the gradient w.r.t. the soft‑max parameters w.
    """
    # soft‑max → non‑negative vector with ℓ₁ = 1
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= np.sum(a)

    # convolution a*a
    conv = _fft_convolve_self(a)                     # length 2·n‑1
    max_conv = conv.max()

    # smoothed max (log‑sum‑exp trick)
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)                     # log Σ exp((c-max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)                        # ∝ exp((c‑max)/ε)
    p = weights / np.sum(weights)                    # probability distribution over shift index

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]                                 # because of convolution symmetry
    conv_ap = _conv_fft(a, p_rev)                   # length 2·n‑1
    seg = conv_ap[n - 1: 2 * n - 1]                 # central n entries correspond to k‑i
    grad_a = 4.0 * n * seg[::-1]                    # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# Bind the helper convolution to the function object so the wrapper can see it
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Deterministic re‑ordering heuristics
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest value, then the smallest, then second‑largest, …."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)
    left_ptr, right_ptr = 0, n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right_ptr]
        right_ptr -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left_ptr]
        left_ptr += 1
        i += 1
    return new_a


def _alternating_ends_reorder(a: np.ndarray) -> np.ndarray:
    """
    New heuristic: put the largest at the left end, the second‑largest at the
    right end, third‑largest next to the left, fourth‑largest next to the right,
    and so on – a simple “alternating” placement.
    """
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]    # descending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    idx = 0
    while idx < sorted_desc.size:
        new_a[left] = sorted_desc[idx]
        left += 1
        idx += 1
        if idx >= sorted_desc.size:
            break
        new_a[right] = sorted_desc[idx]
        right -= 1
        idx += 1
    return new_a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Simple projected sub‑gradient method that directly minimises
    max(a*a) while keeping a on the simplex (non‑negative, ℓ₁ = 1).
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    step0 = 0.12
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur_a)
        k_max = int(np.argmax(conv))

        # gradient of the max entry w.r.t. a is 2·a_{k_max‑i}
        idx_arr = k_max - np.arange(n)
        mask = (0 <= idx_arr) & (idx_arr < n)
        grad = np.zeros_like(cur_a)
        grad[mask] = 2.0 * cur_a[idx_arr[mask]]

        alpha = step0 / np.sqrt(it + 1.0)
        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)

        sum_new = np.sum(new_a)
        if sum_new < 1e-12:          # safeguard – re‑initialise uniformly
            new_a = np.full(n, 1.0 / n)
            sum_new = 1.0
        new_a /= sum_new

        new_val = _eval_fast(new_a)

        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it:5d} new best = {best_val:.6f}")

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Fast SA that mixes two move types:
       * swap two entries,
       * transfer a random fraction of mass from i to j (i≠j, a_i>0).
    Works directly on the fast evaluator.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    cur_seq = a.copy()
    cur_val = cur_val

    T = max(cur_val * 0.02, 1e-6)      # initial temperature
    decay = 0.99994

    it = 0
    while time.time() < deadline:
        if rng.random() < 0.3:        # mass‑transfer move
            i = rng.integers(0, n)
            if cur_seq[i] == 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            # transfer a random fraction of a_i (up to 30 % of its current mass)
            frac = rng.random() * 0.30
            delta = cur_seq[i] * frac
            if delta <= 0.0:
                continue
            cur_seq[i] -= delta
            cur_seq[j] += delta
        else:                         # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_val = _eval_fast(cur_seq)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[SA] new global best = {best_val:.6f} (iter {it})")
        else:
            # revert move
            if rng.random() < 0.3:   # mass‑transfer – revert analytically
                cur_seq[i] += delta      # actually delta is not mass; redo revert
                cur_seq[j] -= delta
                # simpler: just copy best back (cheap, though slightly less efficient)
                cur_seq = best_seq.copy()
                cur_val = best_val
            else:                     # swap – revert
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        T = max(T * decay, 1e-12)
        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small hill‑climbing refinement (tiny mass moves only)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        # tiny transfer: take up to 2 % of the donor mass
        i = rng.integers(0, n)
        if cur_a[i] == 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur_a[i] * 0.02 * rng.random()
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:      # strict improvement
            cur_a = new_a
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                print(f"[refine] new best = {best_val:.6f}")

    return cur_a, best_val


# ----------------------------------------------------------------------
# Helper to construct a random sparse start (≈ √(2n) non‑zeros)
# ----------------------------------------------------------------------
def _random_sparse_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Returns a random vector with k≈√(2n) non‑zero equal entries.
    This provides a good diverse starting point for the optimiser.
    """
    k = int(np.sqrt(2 * n))
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a
