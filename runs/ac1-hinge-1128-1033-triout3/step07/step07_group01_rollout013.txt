"""
propose_candidate – find a non‑negative step‑function with a low
``evaluate_sequence`` score.

Algorithm outline
-----------------
1. **Initial pool** – a few diverse normalised candidates
   (uniform, previous best, Dirichlet, sinusoid).  Keep the best.

2. **Continuation L‑BFGS on a smooth surrogate** – the max‑convolution
   is replaced by a log‑sum‑exp smooth approximation with a
   decreasing smoothing parameter ε.  For every ε a full L‑BFGS
   optimisation (up to 1500 iterations) is performed; the result of
   one ε is used as the start for the next (warm‑start).

3. **Exact sub‑gradient descent** – after the surrogate phase we
   minimise the true objective using the sub‑gradient of the max‑
   convolution.  An exponential mirror‑descent update with a simple
   back‑tracking line‑search is applied repeatedly.

4. **Simulated‑annealing style random moves** – occasional swaps or
   mass‑transfer moves are accepted according to a slowly cooling
   temperature schedule; any improvement updates the current best.

5. **Cheap hill‑climbing polishing** – random swaps (20 %) and
   mass‑transfers (80 %) are performed while only accepting strict
   improvements.

All operations respect the simplex (non‑negative, Σaᵢ = 1).  The
procedure stops before the 1000 s budget expires and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smoothed surrogate (used by scipy‑minimize)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None  # current smoothing ε
_GLOBAL_N = None    # length of the current problem


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using an FFT of power‑of‑2 size.
    Returns a length‑2·n‑1 array.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical inaccuracies can produce tiny negatives.
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε of the true objective and its gradient
    w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ---------- soft‑max ----------
    w_shift = w - np.max(w)               # numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---------- convolution ----------
    conv = _full_convolution(a)           # length 2·n‑1
    max_c = conv.max()

    # ---------- smooth max ----------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    # p_i = exp((conv_i - max_c)/eps) / Σ_j exp(...)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # distribution over lag indices

    # conv of a with reversed p gives ∂max/∂a (up to a constant factor)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the segment that aligns with a_i (see original implementation)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # ---------- chain rule through soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already
    satisfy Σ a_i = 1 (the denominator is then 1).  Returns
    2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        return float('inf')
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic: put the largest values on the left side and the smallest
    on the right – this often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large → left
        new_a[n - 1 - i] = sorted_vals[i]      # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector is
    re‑normalised onto the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:  # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:  # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a tiny safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine problem size and a reasonable start vector.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = old.shape[0]
        base_a = old / old.sum()
        has_old = True
    except Exception:
        n = 4000
        base_a = np.full(n, 1.0 / n)
        has_old = False

    # --------------------------------------------------------------
    # Build a few diverse initial candidates.
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_cands.append(base_a.copy())

    # random Dirichlet (α = 1 – exponential‑like)
    init_cands.append(rng.gamma(shape=1.0, scale=1.0, size=n))

    # random Dirichlet (α = 5 – smoother)
    init_cands.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoid (shifted to stay non‑negative)
    x = np.linspace(0, np.pi, n)
    init_cands.append(np.sin(x) + 0.1)

    # normalise everything onto the simplex
    init_cands = [c / c.sum() for c in init_cands]

    # --------------------------------------------------------------
    # Choose the best cheap initial candidate.
    # --------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_cands):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS on the smoothed surrogate.
    # --------------------------------------------------------------
    # Smoothing schedule: from 1e‑2 down to 1e‑14 (log‑space, 24 points)
    eps_schedule = np.concatenate([
        np.logspace(np.log10(1e-2), np.log10(1e-8), num=12),
        np.logspace(np.log10(1e-8), np.log10(1e-14), num=12),
    ])
    maxiter_per_eps = 1500

    # initialise soft‑max parameters
    w = np.log(best_a + 1e-12)

    for eps in eps_schedule:
        if time.time() >= deadline:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_eps,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # convert back to probabilities via soft‑max
        w_shift = w - np.max(w)
        a_candidate = np.exp(w_shift)
        a_candidate /= a_candidate.sum()

        cur_val = _fast_evaluate(a_candidate)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a_candidate.copy()
            print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # occasional anti‑symmetric re‑order (helps escaping plateaux)
        if eps <= 1e-9:
            a_ord = _anti_symmetric_reorder(best_a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val - 1e-12:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[reorder] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent on the true max‑convolution.
    # --------------------------------------------------------------
    a = best_a.copy()
    cur_val = best_val
    it = 0
    while time.time() < deadline - 20 and it < 3000:
        conv = _full_convolution(a)
        k = int(conv.argmax())          # lag where the current maximum occurs

        # sub‑gradient of max(conv) w.r.t. a (one‑sided contribution)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx
        # each pair (i,j) contributes a_j to ∂/∂a_i
        g[idx] = a[jdx]
        # and symmetrically a_i to ∂/∂a_j (avoid double‑count on diagonal)
        g[jdx] = g[jdx] + a[idx]

        # exponential mirror‑descent with back‑tracking line‑search
        step = 0.5
        improved = False
        while step > 1e-6:
            a_new = a * np.exp(-step * g)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s <= 0:
                break
            a_new /= s
            new_val = _fast_evaluate(a_new)
            if new_val < cur_val - 1e-12:
                a = a_new
                cur_val = new_val
                improved = True
                break
            step *= 0.5

        if improved and cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[subgrad it {it}] new best = {best_val:.6f}")

        # periodic anti‑symmetric reorder (cheap and often helpful)
        if it % 200 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val - 1e-12:
                best_val = val_ord
                best_a = a_ord.copy()
                a = a_ord.copy()
                cur_val = val_ord
                print(f"[subgrad reorder] improved to {best_val:.6f}")

        it += 1

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing style random mass‑transfer / swaps.
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.04                # initial temperature
    cooling = 0.99997

    while time.time() < deadline - 10:
        if rng.random() < 0.5:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.08 * rng.random()   # up to 8 % of source mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        # re‑normalise onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-7)

    # --------------------------------------------------------------
    # Phase 4 – final cheap hill‑climbing polishing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # final safety check using the official evaluator
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    return best_a.tolist()
