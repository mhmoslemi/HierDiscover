"""
Hybrid search for a low‑value step‑function.
The algorithm proceeds as follows:

1. **Parametric initialization** – we generate a family of symmetric
   Beta(α,α) densities (α≈0.5 yields the arcsine shape) on a discrete
   grid of length *n*.  A fine grid search over α is performed to find
   the best starting point.

2. **Smoothed surrogate minimisation** – using a log‑sum‑exp
   approximation of the max‑convolution we run a few L‑BFGS restarts
   with a decreasing smoothing parameter ε.  The variables are the
   soft‑max parameters w so that the positivity and ℓ¹‑normalisation of
   the heights are enforced automatically.

3. **Projected sub‑gradient descent** – with the exact max‑convolution
   we perform a long sub‑gradient descent (step size ∝ 1/√t) while
   projecting back onto the probability simplex.

4. **Permutation annealing** – a short simulated‑annealing phase that
   swaps two positions or moves a small amount of mass between two
   indices, accepting worsening moves with a temperature schedule.

5. **Fine greedy refinement** – final hill‑climbing with tiny mass
   transfers.

All phases respect a total wall‑time budget of *budget_s* seconds.
The best sequence found is finally evaluated with the provided
``evaluate_sequence`` function and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global configuration for the smooth surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter ε
_GLOBAL_N = None     # length of the sequence (n)

# ----------------------------------------------------------------------
# Small utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)   # eliminate tiny negatives from rounding
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Exact copy of ``evaluate_sequence`` but without the safety checks."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smoothed surrogate objective (log‑sum‑exp approximation of the max)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and grad_w is the gradient w.r.t. the soft‑max parameters w.
    """
    # ---- soft‑max to enforce a_i ≥ 0 and Σa_i = 1 --------------------
    w_shift = w - np.max(w)               # improve numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                         # now a lives on the simplex

    # ---- convolution a * a -----------------------------------------
    conv = _fft_convolve_self(a)          # length 2·n‑1
    max_conv = conv.max()

    # ---- smooth approximation of the max ---------------------------
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log ∑ exp((c−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a -----------------------------------------
    weights = np.exp(shifted)             # exp((c−max)/ε)
    p = weights / np.sum(weights)         # probability distribution over k

    # gradient of L w.r.t. a is   4·n·(p ⋆ a)   (⋆ = convolution, reversed p)
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    # central n entries correspond to indices [n‑1, 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # ∂L/∂a

    # ---- chain rule through the soft‑max ---------------------------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)           # ∂L/∂w (simplex‑projected gradient)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# bind the convolution helper to the function object
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Simple re‑ordering – place the largest values at the two ends
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Deterministic ordering: large‑small‑large‑small … (ends get the
    largest values)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap + tiny mass moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float, deadline: float):
    """Swap‑only simulated annealing (fast, works directly on the fast evaluator)."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # starting temperature – a small fraction of the current value
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99994

    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        a[i], a[j] = a[j], a[i]                # propose swap
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            a[i], a[j] = a[j], a[i]            # revert swap

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Hill‑climbing refinement (small mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float, deadline: float):
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:          # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                           # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float):
    """
    Sub‑gradient descent directly on the objective
        f(a) = 2·n·max(a*a) / (sum a)²
    with the ℓ¹‑simplex constraint (non‑negative, sum = 1).
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    step0 = 0.15          # initial step size – empirically good
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur_a)
        k_max = int(np.argmax(conv))

        # gradient of the *single* max entry w.r.t. a:
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(cur_a)
        grad[mask] = 2.0 * cur_a[idx[mask]]

        alpha = step0 / np.sqrt(it + 1.0)
        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:                     # safe‐guard against collapse
            new_a = np.full(n, 1.0 / n)
            s = 1.0
        new_a /= s                         # project onto the simplex

        new_val = _eval_fast(new_a)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it} new best = {best_val:.6f}")

        # accept only improvements – this guarantees monotone descent
        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val

        it += 1
    return best_a, best_val

        """Return a normalised Beta(α,α) shape on the discrete grid."""
        a = np.power(x, alpha - 1.0)        # proportional to (p(1‑p))^{α‑1}
        # Clip potential NaNs (should not happen for α>0) and normalise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        return a

    # coarse grid
    alphas = np.linspace(0.30, 0.80, 11)    # 0.30 … 0.80 in steps of 0.05
    best_a = None
    best_val = np.inf
    best_alpha = None

    for alpha in alphas:
        a = make_shape(alpha)
        val = _eval_fast(a)
        print(f"[grid] α={alpha:.3f} → val={val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a.copy()
            best_alpha = alpha

    # fine‑grained refinement around the current best α
    if time.time() < deadline - 30.0:
        step = 0.02
        while time.time() < deadline - 30.0 and step > 1e-4:
            improved = False
            cand_alphas = np.arange(best_alpha - 0.1, best_alpha + 0.1001, step)
            cand_alphas = np.clip(cand_alphas, 0.01, 0.99)
            for alpha in cand_alphas:
                a = make_shape(alpha)
                val = _eval_fast(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    best_alpha = alpha
                    improved = True
                    print(f"[refine α] α={alpha:.4f} → val={val:.6f}")
            if not improved:
                step *= 0.5

    print(f"[init] best α={best_alpha:.4f}, value={best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Smoothed surrogate minimisation (L‑BFGS, several restarts)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(5e-12), num=14)
    restart = 0
    while time.time() < deadline - 180.0:          # reserve ~3 min for later phases
        restart += 1
        print(f"[restart {restart}]")
        # initialise soft‑max parameters from the current best a
        w = np.log(best_a + 1e-12) + 0.5 * np.random.randn(n)

        for eps in eps_schedule:
            if time.time() >= deadline - 150.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            # L‑BFGS on the smoothed surrogate
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 600,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate the obtained point
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= np.sum(a)
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                best_alpha = None          # we are now away from the pure Beta shape
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # Re‑order the multiset so that the biggest values are at the ends
        a_ord = _ends_large_reorder(best_a)
        a_ord /= np.sum(a_ord)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Projected sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline - 90.0:
        print("[subgrad] starting projected sub‑gradient descent")
        sub_deadline = deadline - 60.0
        best_a, best_val = _subgradient_descent(best_a, sub_deadline)

    # --------------------------------------------------------------
    # 4. Simulated annealing on permutations (swap moves)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[SA] final simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 15.0)

    # --------------------------------------------------------------
    # 5. Final cheap hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ----------------------------------------------------------------------
    # Report final value using the official evaluator
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required by the framework
    return best_a.tolist()
