"""
Hybrid optimizer for ``evaluate_sequence``

The pipeline consists of three stages:

1. **Smooth surrogate optimisation** –  we minimise a
   log‑sum‑exp approximation of the max‑convolution using L‑BFGS‑B.
   The optimisation runs on the soft‑max parameters `w`
   (so the variables stay on the probability simplex).

2. **Randomised restarts** –  the best solution is repeatedly
   perturbed and re‑optimised with a finer ε‑schedule.  This
   explores many basins of attraction without getting stuck.

3. **Simulated‑annealing refinement** –  a simple SA loop moves a
   fraction of mass from a random source index to a random target
   index, accepting moves that improve the exact objective or that
   are accepted probabilistically with a slowly decreasing temperature.
   The current best sequence is kept and returned.

All operations respect a user supplied time budget (default 1000 s)
and the final candidate is evaluated with the supplied
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns the value and gradient of the surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε)

    where a = softmax(w) (∑a = 1, a_i ≥ 0).
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()

    # full autocorrelation
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    log_sum = np.logaddexp.reduce(shifted)          # = log Σ exp(...)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # probabilities over lags

    # convolution a * rev(p)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                  # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]                # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index of the last positive entry
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (same formula as evaluate_sequence, but without
# the extra safety checks – used internally for speed).
# ----------------------------------------------------------------------
def _exact_obj(seq: np.ndarray) -> float:
    """2·n·max(conv) / (sum(seq)²)  – the true objective to minimise."""
    n = seq.shape[0]
    total = seq.sum()
    if total < 0.01:
        return float('inf')
    max_b = _full_convolution(seq).max()
    return 2.0 * n * max_b / (total * total)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement on the exact objective
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         rng: np.random.Generator,
                         init_temp: float = 0.3,
                         final_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a fraction of probability mass from a random
    source index to a random target index.  The temperature decreases
    geometrically.  Returns the best sequence found and its value.
    """
    n = a.shape[0]
    cur_seq = a.copy()
    best_seq = cur_seq.copy()
    best_val = cur_val
    T = init_temp

    while time.time() < deadline:
        # pick distinct source and target indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of the source mass
        if cur_seq[i] <= 0:
            continue
        step_frac = rng.random() * 0.15          # up to 15 % of source mass
        delta = cur_seq[i] * step_frac
        if delta <= 0:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta      # total sum stays constant

        new_val = _exact_obj(new_seq)
        dv = new_val - cur_val
        if dv < 0 or rng.random() < math.exp(-dv / T):
            cur_seq = new_seq
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[SA] new best = {best_val:.6f}")

        # geometric cooling
        T = max(T * 0.9995, final_temp)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Hybrid optimisation for the step‑function attack.
    Returns the list of heights (non‑negative floats) that yields the
    lowest value of ``evaluate_sequence`` found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # tiny safety margin

    # problem size – a moderate length gives a good trade‑off
    n = 3000

    # ------------------------------------------------------------------
    # Build a small portfolio of initial vectors
    # ------------------------------------------------------------------
    init_vectors = []

    # 1. uniform
    init_vectors.append(np.full(n, 1.0 / n))

    # 2. reuse a previously found construction if available
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        init_vectors.append(a0 / a0.sum())
    except Exception:
        pass

    # 3. random Dirichlet draws with different concentration parameters
    for alpha in (0.5, 1.0, 2.0):
        vec = rng.gamma(shape=alpha, scale=1.0, size=n)
        vec /= vec.sum()
        init_vectors.append(vec)

    # 4. deterministic ramps (linear and quadratic) – often a good
    #    starting point for low‑autocorrelation shapes
    ramp = np.arange(1, n + 1, dtype=float)
    ramp /= ramp.sum()
    init_vectors.append(ramp)

    quad = (np.arange(1, n + 1, dtype=float) ** 2)
    quad /= quad.sum()
    init_vectors.append(quad)

    # ------------------------------------------------------------------
    # Helper: smooth‑surrogate optimisation using L‑BFGS‑B
    # ------------------------------------------------------------------
    def _optimize_smooth(a_start: np.ndarray,
                         eps_schedule: np.ndarray,
                         maxiter_per_eps: int,
                         time_limit: float) -> tuple[np.ndarray, float]:
        """Runs the L‑BFGS‑B loop for a single start vector."""
        w = np.log(np.maximum(a_start, 1e-12))
        for eps in eps_schedule:
            if time.time() >= time_limit:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_eps,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        a_opt = np.exp(w - np.max(w))
        a_opt /= a_opt.sum()
        return a_opt, _exact_obj(a_opt)

    # ------------------------------------------------------------------
    # Smooth optimisation on all initial vectors
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-2, -14, num=30)
    best_seq = None
    best_val = float('inf')

    for idx, a0 in enumerate(init_vectors):
        if time.time() >= deadline:
            break
        a0 = np.maximum(a0, 1e-12)
        a0 /= a0.sum()
        a_opt, val_opt = _optimize_smooth(a0,
                                          eps_schedule,
                                          maxiter_per_eps=200,
                                          time_limit=deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[init {idx}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Randomised restarts around the current best
    # ------------------------------------------------------------------
    restart_cnt = 0
    while time.time() < deadline - 5.0 and restart_cnt < 30:
        restart_cnt += 1
        # perturb the best solution in log‑space
        w = np.log(best_seq + 1e-12) + rng.normal(scale=0.7, size=n)

        # run only the fine part of the ε‑schedule
        for eps in eps_schedule[-10:]:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 150,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        a_restart = np.exp(w - np.max(w))
        a_restart /= a_restart.sum()
        val_restart = _exact_obj(a_restart)
        if val_restart < best_val:
            best_val = val_restart
            best_seq = a_restart.copy()
            print(f"[restart {restart_cnt}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing refinement (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        best_seq, best_val = _simulated_annealing(best_seq,
                                                  best_val,
                                                  deadline,
                                                  rng)

    # ------------------------------------------------------------------
    # Final reporting using the supplied evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return the raw list of floats
    return best_seq.tolist()
