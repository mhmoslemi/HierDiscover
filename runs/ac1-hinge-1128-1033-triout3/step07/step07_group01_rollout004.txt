"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm Overview
------------------
1. **Initialisation** – generate a diverse portfolio of start vectors
   (uniform, Dirichlet, sinusoid, alternating high‑low patterns,
   permutations of the previous best, …) and keep the best.
2. **Random re‑ordering search** – try a few random permutations of the
   current best (very cheap, often reduces the peak of the self‑convolution).
3. **L‑BFGS on a smooth log‑sum‑exp surrogate** – a schedule of decreasing
   smoothing parameters `ε` drives the optimisation towards the true
   max‑convolution.  Several restarts (including randomised starts) are
   performed.  After each restart an “alternating‑reorder” heuristic is
   applied.
4. **Exponential mirror‑descent on the exact max‑convolution** – a true
   sub‑gradient step (factor 2 · a_{k*‑i}) is taken with a geometrically
   decaying step size.
5. **Targeted peak‑reduction moves** – the index `k*` where the convolution
   is maximal is identified; mass is transferred from positions that
   contribute heavily to this peak to random locations, accepting
   occasional uphill moves (simulated‑annealing style).
6. **General simulated‑annealing (random swaps / transfers)** – keeps the
   search stochastic and can escape shallow local minima.
7. **Final cheap hill‑climb** – deterministic swaps / small mass transfers
   until the time budget expires.

All vectors are kept on the probability simplex (non‑negative, sum = 1)
so that the objective simplifies to `2·n·max(conv)`.  The routine respects
the 1000 s time budget and returns the best sequence found as a plain
Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth L‑BFGS surrogate (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None  # smoothing parameter ε
_GLOBAL_N = None    # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)   # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Smooth surrogate (max + log‑sum‑exp) and its gradient w.r.t. soft‑max vars
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # -- soft‑max parameterisation (a lives on the simplex) --
    w_shift = w - np.max(w)           # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # -- convolution of a with itself --
    conv = _full_convolution(a)       # length 2·n‑1
    max_c = conv.max()

    # -- smooth max (log‑sum‑exp) --
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    loss = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # -- gradient w.r.t. a (chain rule through the smooth max) --
    weights = np.exp(shifted)                 # proportional to exp((c_i‑max)/ε)
    p = weights / weights.sum()               # distribution over lag indices

    # convolution a * p_rev (reversed p) gives ∂/∂a of Σ p_i * conv_i
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]          # part that aligns with a
    grad_a = 4.0 * n * seg[::-1]            # factor 2 from max_c + 2 from LSE

    # -- back‑prop through the soft‑max (standard formula) --
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return loss, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` for vectors that sum to 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)    # denominator = 1²


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest value on the leftmost position, the second largest on
    the rightmost, the third on the second‑leftmost, etc.  This spreads
    large masses as far apart as possible.
    """
    n = a.shape[0]
    order = np.empty(n, dtype=int)
    sorted_idx = np.argsort(-a)                 # descending
    left, right = 0, n - 1
    toggle = True
    for idx in sorted_idx:
        if toggle:
            order[left] = idx
            left += 1
        else:
            order[right] = idx
            right -= 1
        toggle = not toggle
    return a[order]


def _random_permutation(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a random permutation of ``a``."""
    perm = rng.permutation(a.shape[0])
    return a[perm]


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  After each move the
    vector is projected back onto the simplex.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def _targeted_peak_reduction(a: np.ndarray, best_val: float,
                             deadline: float, seed: int = 0) -> tuple[np.ndarray, float]:
    """
    Moves mass away from the pairwise contributions that create the current
    maximal convolution entry.  The move is accepted if it improves the
    objective, otherwise with a small annealing probability.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng(seed)
    n = cur_a.shape[0]

    T = 0.02
    decay = 0.9999

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        # pairs (i, j) with i+j = max_k
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1)
        jdx = max_k - idx

        contrib = cur_a[idx] * cur_a[jdx]      # product for each pair
        total = contrib.sum()
        if total == 0.0:
            break

        probs = contrib / total
        src = rng.choice(idx, p=probs)         # source that contributes heavily

        # pick a random destination that is *different* from src
        trg = rng.integers(0, n)
        if trg == src:
            continue

        delta = cur_a[src] * 0.07 * rng.random()   # up to 7 % of source mass
        if delta <= 1e-15:
            continue
        new_a = cur_a.copy()
        new_a[src] -= delta
        new_a[trg] += delta

        if new_a[src] < 0:
            continue

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a, cur_val = new_a, val
            if val < best_val:
                best_val, best_a = val, new_a.copy()
                print(f"[peak] new global best = {best_val:.6f}")
        else:
            # occasional uphill acceptance
            if rng.random() < np.exp(-(val - cur_val) / max(T, 1e-12)):
                cur_a, cur_val = new_a, val

        T = max(T * decay, 1e-7)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Problem size – use the length of the known construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # 1. Diverse initial candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # random Dirichlet (α = 1)
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))

    # random Dirichlet (α = 5)
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoid
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # alternating high‑low sinusoid
    init_candidates.append(0.5 + 0.5 * np.sign(np.sin(3 * x)))

    # alternating reorder of uniform (spreads mass)
    init_candidates.append(_alternating_reorder(np.full(n, 1.0 / n)))

    # random permutation of previous best (if present)
    if has_old:
        init_candidates.append(_random_permutation(old / old.sum(), rng))

    # Normalise everything onto the simplex.
    init_candidates = [c / c.sum() for c in init_candidates]

    # ------------------------------------------------------------------
    # Pick the best cheap candidate
    # ------------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] eval = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Random re‑ordering search (cheap, often helps)
    # ------------------------------------------------------------------
    for _ in range(30):
        if time.time() >= deadline:
            break
        cand = _random_permutation(best_a, rng)
        val = _fast_evaluate(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand reorder] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. L‑BFGS on the smooth surrogate (multiple restarts)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-14), num=15)
    w = np.log(best_a + 1e-12)          # initialise soft‑max parameters

    max_restarts = 25
    restart = 0
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        # occasional fresh random start to diversify search space
        if restart % 5 == 0:
            w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n

            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            # back to probability vector
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val, best_a = cur_val, a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # Apply the alternating‑reorder heuristic after each restart
        a_ord = _alternating_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Mirror‑descent on the *exact* max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    alpha0 = 0.7      # initial (larger) step size for faster early progress
    decay = 0.9996
    min_alpha = 0.01

    while time.time() < deadline - 20.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a:
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        # derivative = 2 * a_{k‑i}
        g[idx] = 2.0 * a[jdx]

        # exponential mirror‑descent update
        alpha = max(alpha0 * (decay ** (it // 150)), min_alpha)
        a = a * np.exp(-alpha * g)

        # simplex projection (numerical drift)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        if it % 100 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[mirror it {it}] new best = {best_val:.6f}")

            # occasional reordering to shake the solution
            if it % 500 == 0:
                a_ord = _alternating_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val, best_a = val_ord, a_ord.copy()
                    a = a_ord.copy()
                    print(f"[mirror reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 5. Targeted peak‑reduction moves (focus on the maximal convolution entry)
    # ------------------------------------------------------------------
    best_a, best_val = _targeted_peak_reduction(best_a, best_val,
                                                deadline - 10.0,
                                                seed=seed + 98765)

    # ------------------------------------------------------------------
    # 6. General simulated‑annealing style random moves (swap / transfer)
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.02
    cooling = 0.99995
    rng = np.random.default_rng(seed + 54321)

    while time.time() < deadline - 5.0:
        if rng.random() < 0.5:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.08 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a, cur_val = new_a, new_val
            if new_val < best_val:
                best_val, best_a = new_val, new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-7)

    # ------------------------------------------------------------------
    # 7. Final cheap hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # 8. Final safety check using the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:    # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
