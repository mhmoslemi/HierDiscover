"""
Propose a step‑function that (hopefully) beats the previous best ~1.512.
The strategy combines several cheap, complementary heuristics:

1. **Parametric scans** – a few simple families (cosine “V‑shape”, symmetric
   power law, etc.) are enumerated to obtain a decent starting point quickly.
2. **Smooth surrogate optimisation** – minimise a log‑sum‑exp surrogate of
   2·n·max(conv) with L‑BFGS‑B on the log‑softmax parametrisation.
   A continuation in the smoothing parameter ε and a handful of random
   restarts (perturbations of the best solution) explore the landscape.
3. **Aggressive hill‑climbing** – mass is repeatedly shifted away from the
   lag that currently attains the maximal convolution.  Moves are accepted
   only when the exact (non‑smoothed) objective improves.
4. **Simulated‑annealing fine‑tuning** – random donor/receiver moves are
   sometimes accepted even if they worsen the objective, allowing the search
   to escape shallow local minima.
5. **Projected sub‑gradient descent** – a final polishing phase using the
   exact sub‑gradient of the active lag, projected onto the probability simplex.

All phases respect the 1000 s time budget; intermediate progress is printed.
The returned list is normalised (sum = 1) and contains only non‑negative
floats, satisfying the security checks of `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT helpers (real data) – keep these tiny and top‑level for speed
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full (non‑reversed) autocorrelation of a real vector."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise can produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) / (sum(a)^2)
    (sum(a) is kept = 1 throughout the optimiser, but the formula is
    written in full for safety).
    """
    n = a.shape[0]
    max_b = np.max(_full_convolution(a))
    s = a.sum()
    if s < 1e-12:          # should never happen, protects against div‑by‑zero
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Surrogate L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv−max)/ε)
    with a = softmax(w) (Σa = 1, a_i ≥ 0).

    Returns (value, grad_w) where grad_w is the gradient w.r.t. the `w`
    parameters (the log‑softmax variables).
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()

    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing term
    shifted = (conv - max_conv) / eps
    # stable log‑sum‑exp
    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))

    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed‑form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # probabilities over lags

    # convolution a * rev(p)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                  # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]                # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


# ----------------------------------------------------------------------
# Projection onto simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of vector v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families – quick “pre‑search”
# ----------------------------------------------------------------------
def _scan_simple_families(n: int):
    """Enumerate a few low‑dimensional families; keep the best found."""
    best_val = np.inf
    best_a = None

    # 1) Cosine V‑shape: a_i ∝ max(0, 1 + A·cos(2π i/(n−1)))
    for A in np.linspace(0.0, 0.99, 30):
        a = np.maximum(0.0, 1.0 + A * np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
        total = a.sum()
        if total == 0:
            continue
        a /= total
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # 2) Symmetric power law: a_i ∝ (i+1)^p + (n-i)^p
    for p in np.linspace(-2.0, 2.0, 41):
        idx = np.arange(n) + 1
        a = (idx ** p) + ((n - idx + 1) ** p)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0:
            continue
        a /= total
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill climbing (as in the reference code)
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              cur_val: float,
                              deadline: float,
                              rng: np.random.Generator,
                              step_start: float = 0.2):
    """Move a fraction of mass away from the lag achieving max(conv)."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_val = cur_val
    step_frac = step_start

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # indices that actually contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue

        # pick up to three most influential pairs
        top_n = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        # destination indices – pick low‑mass positions
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue

        # renormalise (keeps Σa = 1)
        new_a /= new_a.sum()

        new_val = _objective(new_a)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-6)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing fine tuning
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Random donor/receiver moves accepted with Metropolis probability.
    Very cheap – uses the exact objective (no expensive evaluate_sequence call).
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_val = cur_val

    T = 0.02                     # initial temperature
    decay = 0.9999               # geometric cooling
    step_scale = 0.05            # typical fraction of donor mass moved

    while time.time() < deadline:
        # pick a donor with some mass and a different receiver
        donor_idx = int(rng.choice(np.where(cur_a > 1e-12)[0]))
        receiver_idx = int(rng.choice(np.delete(np.arange(n), donor_idx)))
        delta = cur_a[donor_idx] * step_scale * rng.random()
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[donor_idx] -= delta
        new_a[receiver_idx] += delta
        # renormalise (numerically)
        new_a /= new_a.sum()

        new_val = _objective(new_a)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[anneal] new best = {best_val:.6f}")

        T *= decay

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """Standard sub‑gradient method with projection onto the simplex."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_val = cur_val
    eta = 0.1                    # initial step size

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # gradient step + projection
        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = _objective(new_a)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            eta = min(eta * 1.05, 0.5)
        else:
            eta = max(eta * 0.7, 1e-5)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Optimises a non‑negative step‑function to minimise ``evaluate_sequence``.
    Returns the best list of floats found within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate n gives a good trade‑off between speed
    # and expressive power.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Try to reuse a previous construction if the global variable exists.
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        else:
            a0 = a0.copy()
    except Exception:
        a0 = None

    # ------------------------------------------------------------------
    # 1. Quick parametric scan – yields a decent baseline within a few ms.
    # ------------------------------------------------------------------
    if a0 is not None:
        # ensure feasibility
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        best_a = a0.copy()
        best_val = _objective(best_a)
    else:
        best_a, best_val = _scan_simple_families(n)

    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Smooth surrogate optimisation (log‑sum‑exp continuation)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-2, -14, num=30)   # a finer tail than before
    w = np.log(best_a + 1e-12)                   # log‑softmax parameters

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        res = minimize(fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # evaluate exact objective
        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[smooth eps {eps:.1e}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Randomised restarts – perturb the current best and re‑optimise.
    # ------------------------------------------------------------------
    restart = 0
    max_restarts = 30
    while time.time() < deadline - 5.0 and restart < max_restarts:
        restart += 1
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.8, size=n)

        # only the finest part of the ε‑schedule, more iterations
        for eps in eps_schedule[-12:]:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 300,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart {restart}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Aggressive hill‑climbing phase (deterministic moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _mass_transfer_hill_climb(best_a,
                                                    best_val,
                                                    deadline,
                                                    rng,
                                                    step_start=0.25)

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing fine tuning (stochastic)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _simulated_annealing(best_a,
                                                best_val,
                                                deadline,
                                                rng)

    # ------------------------------------------------------------------
    # 6. Projected sub‑gradient descent (exact non‑smooth optimisation)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _subgradient_descent(best_a,
                                                best_val,
                                                deadline,
                                                rng)

    # ------------------------------------------------------------------
    # Final reporting – use the provided evaluator to get the official score.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
