def subgradient_max_adam(a_init, steps, lr, beta1, beta2, epsilon, deadline):
    w = np.log(a_init + 1e-16)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    best_a = a_init.copy()
    best_val = _eval_fast(best_a)
    while time.time() < deadline and t < steps:
        a = np.exp(w - np.max(w))
        a = a / np.sum(a)
        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # gradient w.r.t a
        grad_a = np.zeros_like(a)
        # fill gradient: 2 * a_{k_max - i} if valid
        # compute a shift
        # for i=0..n-1:
        #   j = k_max - i
        #   if 0 <= j < n: grad_a[i] = 2 * a[j]
        # done using vectorized.
        idx = np.arange(a.shape[0])
        j = k_max - idx
        mask = (j >= 0) & (j < a.shape[0])
        grad_a[mask] = 2.0 * a[j[mask]]

        # compute grad_w using softmax chain rule
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w * grad_w)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        w = w - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # compute evaluation
        a_new = np.exp(w - np.max(w))
        a_new = a_new / np.sum(a_new)
        val = _eval_fast(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[adam] new best = {best_val:.6f} at iter {t}")
        # optionally early break if time limited.
    return best_a, best_val
```

"""
Hybrid optimiser for the step‑function self‑convolution problem.
The optimiser works with a longer sequence (n = 8000) which gives more
degrees of freedom to flatten the sum distribution a⊛a.
Stages
1️⃣  Diverse seed generation (geometric, power‑law, arcsine, sinusoid,
    random Dirichlet draws).
2️⃣  Coarse L‑BFGS on a smooth log‑sum‑exp surrogate of the maximum
    convolution entry (large smoothing parameter).
3️⃣  Refined L‑BFGS with a much finer ε‑schedule.
4️⃣  Sub‑gradient Adam optimiser on the *exact* max‑convolution (ε = 0).
5️⃣  Extended simulated annealing (swap / transfer / targeted transfer).
6️⃣  Final hill‑climbing polish.
All heavy work uses FFT‑based convolution for speed.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global state for the smooth surrogate (soft‑max + log‑sum‑exp)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter
_GLOBAL_N = None     # current sequence length


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (real input)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Numerical noise may create tiny negatives – clamp them.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast replica of the provided ``evaluate_sequence`` for non‑negative a."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """Return (smooth objective, gradient w.r.t. w) for a given ε."""
    # soft‑max (guarantees a_i ≥ 0 and Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= np.sum(a)

    # autocorrelation a*a
    conv = _fft_convolve_self(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                 # probability over conv indices
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)              # a * p_rev
    seg = conv_ap[n - 1: 2 * n - 1]             # central part, length n
    grad_a = 4.0 * n * seg[::-1]

    # back‑propagation through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Adapter required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Re‑ordering heuristics (sometimes dramatically improve the objective)
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large values left, small values right – pairwise."""
    n = a.shape[0]
    s = np.sort(a)
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[n - 1 - i]
        out[n - 1 - i] = s[i]
    if n % 2:
        out[half] = s[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    s = np.sort(a)
    left, right = 0, len(s) - 1
    out = np.empty_like(a)
    k = 0
    while left <= right:
        out[k] = s[right]
        k += 1
        right -= 1
        if left <= right:
            out[k] = s[left]
            k += 1
            left += 1
    return out


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Simple descending sort."""
    return np.sort(a)[::-1]


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """Geometric progression a_i ∝ q**i (0<q<1), normalised to sum = 1."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q**n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)**p (p<0), normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _arcsine_sequence(n: int) -> np.ndarray:
    """Discrete approximation of the continuous arcsine density ∝1/√(x(1‑x))."""
    idx = np.arange(n) + 0.5
    a = 1.0 / np.sqrt(idx * (n - idx))
    a /= a.sum()
    return a


def _sin_sequence(n: int) -> np.ndarray:
    """Positive half‑sine shape a_i ∝ sin(π·(i+½)/n)."""
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *exact* max (Adam optimiser)
# ----------------------------------------------------------------------
def _subgradient_adam(a_start: np.ndarray,
                      cur_val: float,
                      deadline: float) -> (np.ndarray, float):
    """Adam on the non‑smooth max‑convolution objective (ε = 0)."""
    n = a_start.shape[0]
    rng = np.random.default_rng()
    w = np.log(a_start + 1e-30)      # soft‑max parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    best_a = a_start.copy()
    best_val = cur_val

    lr = 0.2
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    while time.time() < deadline:
        # ---- current a from w ----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ---- convolution and argmax ----
        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # ---- sub‑gradient w.r.t. a (2·a_{k_max‑i}) ----
        grad_a = np.zeros_like(a)
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad_a[mask] = 2.0 * a[j[mask]]

        # ---- chain rule through soft‑max ----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional random jitter to escape flats
        if t % 500 == 0 and rng.random() < 0.3:
            w += 0.01 * rng.standard_normal(n)

        # ---- evaluate ----
        a_new = np.exp(w - np.max(w))
        a_new /= a_new.sum()
        val = _eval_fast(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[adam] new best = {best_val:.6f} (iter {t})")

        # slowly decay learning rate
        if t % 1000 == 0:
            lr *= 0.9

    return best_a, best_val


# ----------------------------------------------------------------------
# Extended simulated annealing (swap / transfer / targeted)
# ----------------------------------------------------------------------
def _simulated_annealing_extended(a: np.ndarray, cur_val: float,
                                 deadline: float) -> (np.ndarray, float):
    """Annealing that mixes swaps, random mass transfers and targeted moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.05, 1e-6)   # initial temperature
    decay = 0.99995
    it = 0

    while time.time() < deadline:
        move = rng.random()
        if move < 0.4:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                new_a = a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]
            else:
                continue
        elif move < 0.8:                  # random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * 0.07 * rng.random()
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:                               # targeted reduction of the current peak
            conv = _fft_convolve_self(a)
            k_max = int(np.argmax(conv))
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            # find the pair within the peak that carries the largest product
            best_pair = None
            best_prod = -1.0
            for i in range(i_start, i_end + 1):
                j = k_max - i
                prod = a[i] * a[j]
                if prod > best_prod:
                    best_prod = prod
                    best_pair = (i, j)
            if best_pair is None:
                continue
            src = best_pair[0] if a[best_pair[0]] >= a[best_pair[1]] else best_pair[1]
            tgt = rng.integers(0, n)
            if src == tgt:
                continue
            delta = a[src] * 0.07 * rng.random()
            new_a = a.copy()
            new_a[src] -= delta
            new_a[tgt] += delta

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")

        T = max(T * decay, 1e-12)
        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final hill‑climbing (swap + tiny transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:           # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                             # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.04 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
