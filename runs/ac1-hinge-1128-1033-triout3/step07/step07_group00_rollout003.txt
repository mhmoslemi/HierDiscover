Implementation (standard):

def project_simplex(v):
    # v is 1-D np array.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function that (hopefully) beats the previous best
objective ≈ 1.50985.  The search combines three ideas:

1. **Initialisation** – reuse the historic best vector if it is
   available; otherwise fall back to a cheap parametric sweep.

2. **Projected sub‑gradient descent** – the exact objective is
   2·n·max(conv(a,a)) (the denominator cancels because we keep the
   vector normalised).  The sub‑gradient of the max‑convolution at the
   active lag `k_max` is simply `a + a_rev`, where `a_rev[i] = a[k_max-i]`
   (out‑of‑range indices contribute 0).  After a step we project the
   result back onto the probability simplex (non‑negative, unit sum).

   A diminishing step size `η_t = η₀ / sqrt(t)` is used; the method is
   cheap (one FFT per iteration) and can be run for a few seconds.

3. **Targeted hill‑climbing** – after the sub‑gradient phase we run a
   refined mass‑transfer local search (the same routine that was used
   in the reference implementation).  This often squeezes a few more
   10⁻⁴ off the objective.

Finally a tiny convex combination with the uniform vector is tried,
because a sprinkle of uniform mass can sometimes lower the maximal
self‑convolution further.

The whole pipeline respects the 1000 s budget and returns the best
vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself, length 2·len(a)‑1."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) assumed = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Helper: project an arbitrary vector onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w ≥ 0, Σ w = 1}."""
    # Duchi et al. (2008) algorithm, O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parameteric initialisation – a quick sweep of a few families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector (normalised) and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ---- deterministic families ------------------------------------------------
    # linear decreasing
    try_candidate(np.arange(n, 0, -1, dtype=float))

    # symmetric U‑shape (large at the ends)
    idx = np.arange(n)
    lin_u = np.maximum(idx + 1, n - idx)
    try_candidate(lin_u)

    # half‑wave sine (peaked in the middle, we invert it later)
    try_candidate(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))

    # power‑law U‑shapes, explore very small exponents
    for p in (0.05, 0.1, 0.2, 0.3, 0.4):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_candidate(a)

    # exponential tails – very slow decay
    for r in (0.999995, 0.99999, 0.9999):
        a = r ** np.arange(n, dtype=float)
        try_candidate(a)
        a = r ** np.arange(n, dtype=float) + r ** (n - 1 - np.arange(n))
        try_candidate(a)

    # random Dirichlet draws – a few samples
    rng = np.random.default_rng(12345)
    for _ in range(15):
        a = rng.dirichlet(np.ones(n))
        try_candidate(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Fast local search that moves a fraction of mass from “heavy” to “light” indices."""
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # mass‑transfer schedule – from 0.8 → 0.001
    max_frac_start, max_frac_end = 0.8, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while True:
        if time.time() >= deadline:
            break

        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – biased by contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – tend to pick an index that hardly participates
        if np.random.rand() < 0.4:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            # Inverse weighting (small contribution → large probability)
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                # fallback uniform
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n) slice updates)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:          # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – perturb and restart from the best seen
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a


# ----------------------------------------------------------------------
# Small post‑processing sweep – mixing with uniform
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a' = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

                     budget_s: int = 1000,

    """

    dedicated hill‑climber and returns the best vector found.
    """

    start_time = time.time()
    # keep a tiny safety buffer for the final secure call
    deadline = start_time + budget_s - 0.5

        if s0 > 0:

    except Exception:

    best_a = a0.copy()

    # ------------------------------------------------------------------
    # Phase A – projected sub‑gradient descent on the exact objective.
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.35 * (deadline - start_time)

    while time.time() < phase_a_deadline:
        # convolution and the index of the maximal entry
        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        idx = np.arange(n)
        j = k_max - idx

        grad = a + rev

        if val_new < best_val:
