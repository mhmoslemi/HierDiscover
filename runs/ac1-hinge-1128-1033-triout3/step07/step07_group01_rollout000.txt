def propose_candidate(seed=42, budget_s=1000):
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # Determine n
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        n = old.shape[0]
    except NameError:
        n = 4000

    rng = np.random.default_rng(seed)

    # initial candidate
    best_a = np.full(n, 1.0/n)
    if 'old' in locals():
        best_a = old / old.sum()
    # also generate some random Dirichlet variants and pick best
    for _ in range(5):
        cand = rng.gamma(1.0,1.0,size=n)
        cand = cand / cand.sum()
        if _fast_evaluate(cand) < _fast_evaluate(best_a):
            best_a = cand

    best_val = _fast_evaluate(best_a)
    print(f"[init] best_val = {best_val:.6f}")

    # Phase A: aggressive L-BFGS with fine epsilon schedule
    eps_schedule = np.logspace(np.log10(1e-1), np.log10(1e-14), num=20)  # starts 0.1
    w = np.log(best_a + 1e-12)
    max_restarts = 8
    for restart in range(max_restarts):
        if time.time() >= deadline - 150: break
        # optionally random w restart
        if restart % 3 == 0:
            w = rng.normal(size=n)
        for eps in eps_schedule:
            if time.time() >= deadline - 150: break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 2000,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x
            a_candidate = np.exp(w - np.max(w))
            a_candidate = a_candidate / a_candidate.sum()
            val = _fast_evaluate(a_candidate)
            if val < best_val - 1e-9:
                best_val, best_a = val, a_candidate.copy()
                print(f"[LBFGS eps={eps:.1e}] new best {best_val:.6f}")
        # optionally anti-symmetric reorder
        a_re = _anti_symmetric_reorder(best_a)
        val_re = _fast_evaluate(a_re)
        if val_re < best_val - 1e-9:
            best_val, best_a = val_re, a_re.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # Phase B: refined mirror descent on max(conv)
    a = best_a.copy()
    step = 0.3
    decay = 0.9999
    min_step = 0.005
    it = 0
    while time.time() < deadline - 100:
        conv = _full_convolution(a)
        max_idx = int(conv.argmax())
        # gradient of max(conv) w.r.t a: 2 * shift of a reversed at diff
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx = np.arange(i_start, i_end+1)
        jdx = max_idx - idx
        g[idx] = a[jdx] * 2
        # mirror descent (exponential)
        cur_step = max(step * (decay ** (it // 300)), min_step)
        a = a * np.exp(-cur_step * g)
        # reproject onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12: a = np.full(n, 1.0/n); s = a.sum()
        a /= s
        if it % 200 == 0:

            if val < best_val - 1e-9:
                best_val, best_a = val, a.copy()
                print(f"[MD it {it}] new best {best_val:.6f}")
        it += 1

    # Phase C: targeted pairwise mass transfer hill climbing
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.02

        idx = np.arange(i_start, i_end+1)

        new_a[donor] -= delta

        # ensure positivity and project to sum=1

        s = new_a.sum()

        new_val = _fast_evaluate(new_a)
        # accept if improvement or with SA criteria

        else:

            cur_val = new_val
            if cur_val < best_val - 1e-9:
                best_val, best_a = cur_val, cur_a.copy()
                print(f"[pairhill] improved to {best_val:.6f}")

    # Final cheap hill climb (swap/transfer)

    # final check

        print(f"[final] evaluate_sequence = {final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence error:", e)
    return best_a.tolist()

"""
Propose a non‑negative step‑function (a list of heights) that minimises
``evaluate_sequence``.

The search consists of four stages:

1. **Aggressive L‑BFGS on a smooth surrogate** – we minimise a
   log‑sum‑exp approximation of the max‑convolution.  The smoothing
   parameter ε is decreased from 1e‑1 down to 1e‑14 over a fine log‑spaced
   schedule, with several random restarts.  Gradient and objective are
   computed analytically in the soft‑max (log‑probability) parametrisation.

2. **Refined mirror‑descent on the exact max‑convolution** – after the
   surrogate optimisation we perform many iterations of exponential
   updates (mirror descent) using the exact sub‑gradient of
   ``max(conv)``.  The step size is slowly decayed.

3. **Targeted pairwise mass‑transfer hill‑climbing** – we repeatedly locate
   the index k where the self‑convolution attains its maximum and move a
   small fraction of mass away from the pair (i,j) that contributes most
   to that peak.  An annealing schedule accepts occasional uphill moves.

4. **Cheap final polishing** – a lightweight stochastic hill‑climber that
   tries random swaps and small mass transfers, accepting only strict
   improvements.

All intermediate vectors are projected onto the probability simplex
(`Σ a_i = 1, a_i ≥ 0`).  The routine respects the prescribed time budget
(≈1000 s) and finally returns the best sequence found as a plain Python
list.  The official ``evaluate_sequence`` is called once at the end for a
safety check.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None  # smoothing parameter ε
_GLOBAL_N = None    # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Clip tiny negatives caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ---------- soft‑max ----------
    w_shift = w - np.max(w)          # numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---------- convolution ----------
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ---------- smooth max ----------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    weights = np.exp(shifted)
    p = weights / weights.sum()      # distribution over lag indices

    # derivative of smooth max wrt each conv entry = p
    # grad_a = 4·n·(p * derivative of conv wrt a)
    # derivative of conv_k = Σ_i a_i·δ_{i + j = k} + Σ_j a_j·δ_{i + j = k}
    # which yields a convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment that corresponds to ∂conv/∂a
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]    # reverse to align with a_i order

    # ---------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the full formula
    simplifies to this expression).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places large values on one side and small values on the
    opposite side – often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %). The vector is kept on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
