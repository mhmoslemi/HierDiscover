"""
step‑function optimiser
-----------------------

The goal is to minimise the value returned by ``evaluate_sequence``

    f(a) = 2·n·max_k (a * a)[k] / (∑ a)^2 ,      a_i ≥ 0

The expression is homogeneous, therefore we normalise the vector to
∑a = 1 and minimise the simpler quantity 2·n·max(conv).  The algorithm
uses three complementary phases:

1. **Smooth‑max gradient descent** – the non‑smooth objective
   ``max(conv)`` is approximated by a log‑sum‑exp (a smooth max).  The
   gradient of this surrogate can be computed analytically and an
   iterative projected gradient step is performed (projection onto the
   probability simplex).  The “temperature’’ α of the smooth max is
   increased during the run, allowing the optimisation to first move
   towards a low‑energy region and later focus on the true maximum.

2. **Projected sub‑gradient descent** – a classic sub‑gradient method
   (as used in the previous state‑of‑the‑art code) is applied to the
   exact objective.  It refines the result obtained by the smooth phase.

3. **Aggressive mass‑transfer hill‑climb** – the same hill‑climbing
   routine from the reference implementation is executed to exploit
   local structure and obtain a final polish.

All phases respect the given time budget.  The best sequence found
throughout the whole run is returned.  The code prints intermediate
objective values so that progress can be monitored during execution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using an FFT.
    Returns a non‑negative array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


def _convolve_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two arbitrary 1‑D arrays using an FFT.
    Result length = len(x) + len(y) – 1.
    """
    size = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: len(x) + len(y) - 1]
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective value (∑a = 1) = 2·n·max(conv)."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


# ----------------------------------------------------------------------
# 1. Smoothed max gradient descent
# ----------------------------------------------------------------------


def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 30_000,
    alpha_start: float = 5.0,
    alpha_end: float = 300.0,
    lr0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of
    max(conv).  The smoothness parameter α is increased
    linearly from ``alpha_start`` to ``alpha_end``.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    start_time = time.time()
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- compute convolution and smooth max weight vector -----
        conv = _full_convolution(a)                # shape (2n‑1,)
        # linearly increase α (schedule can be tuned)
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        scaled = alpha * conv
        # stable log‑sum‑exp
        mx = np.max(scaled)
        exp_scaled = np.exp(scaled - mx)
        w = exp_scaled / exp_scaled.sum()          # non‑negative, sums to 1

        # ----- gradient of the smooth max -----
        # grad_j = 2 * Σ_i w_i * a_{i‑j}
        conv_wa = _convolve_full(w, a[::-1])       # length = 3n‑2
        start = n - 1                               # slice that gives the required sum
        grad = 2.0 * conv_wa[start : start + n]

        # ----- projected gradient step -----
        lr = lr0 / np.sqrt(it)
        improved = False
        for _ in range(8):
            a_trial = a - lr * grad
            a_trial = np.clip(a_trial, 0.0, None)
            s = a_trial.sum()
            if s == 0.0:
                lr *= 0.5
                continue
            a_trial /= s                               # project onto simplex
            val = _objective(a_trial)
            if val < best_val:
                a = a_trial
                best_a = a_trial.copy()
                best_val = val
                improved = True
                # modest step‑size increase after a success
                lr0 = min(lr0 * 1.03, 1.5)
                break
            else:
                lr *= 0.5
        if not improved:
            lr0 = max(lr0 * 0.95, 1e-4)

        # occasional progress report
        if it % 2000 == 0 or (time.time() - start_time) > (deadline - start_time) / 2:
            print(f"[smooth] iter {it:5d}  α={alpha:.1f}  val={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# 2. Projected sub‑gradient descent (same idea as the reference code)
# ----------------------------------------------------------------------


def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the convolution entry conv[k] w.r.t. a."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 15_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    step0 = 0.5
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        grad = _gradient_of_lag(a, k)

        lr = step0 / np.sqrt(t)
        improved = False
        for _ in range(10):
            a_trial = a - lr * grad
            a_trial = np.clip(a_trial, 0.0, None)
            s = a_trial.sum()
            if s == 0.0:
                lr *= 0.5
                continue
            a_trial /= s
            val = _objective(a_trial)
            if val < best_val:
                a = a_trial
                best_a = a_trial.copy()
                best_val = val
                improved = True
                step0 = min(step0 * 1.02, 2.0)
                if t % 200 == 0:
                    print(f"[subgrad] iter {t:5d}  val = {best_val:.6f}")
                break
            else:
                lr *= 0.5

        if not improved:
            step0 = max(step0 * 0.95, 1e-4)

    return best_a, best_val


# ----------------------------------------------------------------------
# 3. Aggressive mass‑transfer hill‑climbing (as in the previous code)
# ----------------------------------------------------------------------


def _mass_transfer_hillclimb(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_no_improve: int = 3000,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing that repeatedly relocates a fraction of the most‑damaging mass."""
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we are willing to shift
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # pick two distinct low‑mass destinations
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        # keep on the simplex (the move preserves the total mass)
        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # leave a small safety margin

    # ------------------------------------------------------------------
    # Initialise – reuse previous construction if it exists.
    # ------------------------------------------------------------------
    default_n = 4000
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = int(old.shape[0])
        a0 = np.maximum(old, 0.0)
        if a0.sum() == 0.0:
            a0 = rng.random(n)
        a0 /= a0.sum()
    except Exception:
        n = default_n
        a0 = rng.random(n)
        a0 /= a0.sum()

    print(f"[init] n={n}, initial objective = {_objective(a0):.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent.
    # ------------------------------------------------------------------
    a_smooth, val_smooth = _smooth_descent(a0, deadline, rng,
                                            max_iters=20000,
                                            alpha_start=5.0,
                                            alpha_end=300.0,
                                            lr0=0.6)
    print(f"[smooth] best value after smooth descent = {val_smooth:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent for exact refinement.
    # ------------------------------------------------------------------
    a_sub, val_sub = _subgradient_descent(a_smooth, deadline, rng,
                                          max_iters=12000)
    print(f"[subgrad] best value after sub‑gradient descent = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive mass‑transfer hill‑climbing.
    # ------------------------------------------------------------------
    a_final, val_final = _mass_transfer_hillclimb(a_sub, deadline, rng)
    print(f"[final] best value after hill‑climb = {val_final:.6f}")

    # ------------------------------------------------------------------
    # Final reporting using the provided evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_final.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a_final.tolist()
