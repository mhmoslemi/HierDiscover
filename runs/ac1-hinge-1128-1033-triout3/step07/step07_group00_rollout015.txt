def project_simplex(v):
    # v: np.ndarray
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    """Euclidean projection onto simplex sum=1, v may be any real vector."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative? then project onto uniform? not needed
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Fast local optimisation of a non‑negative step function for the
`evaluate_sequence` objective.

Algorithm overview
------------------
1.  Initialise a candidate of length `n` (≈15000) using a rich library of
    deterministic families and a few Dirichlet draws.
2.  Phase‑A – Adam on a smooth log‑sum‑exp approximation of the maximal
    convolution (30 % of the time budget).
3.  Phase‑B – Targeted hill‑climbing that moves a fraction of the mass
    from indices that heavily contribute to the current peak to indices
    that hardly contribute (45 % of the budget).
4.  Phase‑C – Very short simulated‑annealing refinement (20 % of the budget).
5.  Phase‑D – Final refinement by minimising a smooth `L_p` norm of the
    autocorrelation (`p` grows from 2 to 32).  This directly attacks the
    max‑convolution while staying on the probability simplex via Euclidean
    projection.  (≈5 % of the budget).
6.  A tiny convex combination with the uniform vector is tried at the end.

All phases respect the overall 1000 s budget; the final sequence is
returned as a plain Python list of non‑negative floats (summing to 1).
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) assumed = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a real vector onto the simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parameteric initialisation – a rich set of shapes
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ---- deterministic families -------------------------------------------------
    # linear decreasing (left → right)
    try_candidate(np.arange(n, 0, -1, dtype=float))

    # linear U‑shape (large at both ends, small in the middle)
    idx = np.arange(n)
    lin_u = np.maximum(idx + 1, n - idx)
    try_candidate(lin_u)

    # quadratic decreasing (left side)
    try_candidate((np.arange(n, 0, -1, dtype=float) ** 2))

    # quadratic U‑shape
    quad_u = (np.arange(n) + 1) ** -2 + (np.arange(n, 0, -1) + 0) ** -2
    try_candidate(quad_u)

    # half‑wave sine (symmetric)
    try_candidate(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))

    # cosine with peaks at the ends
    try_candidate(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # power‑law symmetric (U‑shape)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_candidate(a)

    # exponential left‑hand side
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        try_candidate(a)

    # symmetric (U‑shaped) exponential
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_candidate(a)

    # truncated exponential (mass only on a prefix)
    for frac in (0.10, 0.15, 0.20, 0.25, 0.30):
        K = max(1, int(frac * n))
        for r in (0.99, 0.98, 0.95, 0.90):
            a = np.zeros(n, dtype=float)
            a[:K] = r ** np.arange(K, dtype=float)
            try_candidate(a)

    # comb‑style sparse patterns (equally spaced spikes)
    for m in (8, 12, 16, 20):
        spacing = max(1, n // m)
        indices = np.arange(0, n, spacing)[:m]
        a = np.zeros(n, dtype=float)
        a[indices] = 1.0 / m
        try_candidate(a)

    # random sparse Dirichlet draws (low concentration → many near‑zeros)
    rng = np.random.default_rng(12345)
    for conc in (0.2, 0.5, 0.8):
        for _ in range(6):
            a = rng.dirichlet(np.full(n, conc))
            try_candidate(a)

    # a few standard Dirichlet (α=1)
    for _ in range(6):
        a = rng.dirichlet(np.ones(n))
        try_candidate(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth approximation of max(conv)."""
    w = np.log(a0 + 1e-12)                 # logits

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 60000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- obtain a on the simplex -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- schedules -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max via log‑sum‑exp -----
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()            # soft‑max over positions

        # ----- gradient of smooth objective w.r.t. a -----
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]    # n entries
        grad_a = 4.0 * n * segment[::-1]

        # ----- chain rule through the log‑softmax -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter of logits
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.008, size=n)

        # restart if stagnated
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Fast local search moving mass from contributors to a peak towards low‑contrib indices."""
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # compute contributions to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – weighted by contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – preferably where reversed contribution is tiny
        if np.random.rand() < 0.35:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of b (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – short simulated‑annealing refinement
# ----------------------------------------------------------------------
def _annealing_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Metropolis‑style search with a tiny temperature schedule."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    T_start, T_end = 5e-4, 5e-8
    IMPROVE_LIMIT = 120_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.004, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – Lp‑norm based refinement (smooth approximation of max)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               n: int,
               start: float,
               deadline: float,
               best_a: np.ndarray,
               best_val: float,
               p_values: list[int] = (2, 4, 8, 16, 32),
               max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth Lp‑norm of the autocorrelation.
    The algorithm proceeds through increasing `p` values, performing
    projected gradient steps with a simple back‑tracking line search.
    """
    a = a0.copy()
    best_seq = best_a.copy()
    best_val_local = best_val
    no_improve = 0

    for p in p_values:
        eta0 = 0.5 / float(p)          # initial step size for this p
        print(f"[Lp] starting p = {p}")

        while True:
            now = time.time()
            if now >= deadline:
                return best_seq, best_val_local

            # ----- compute convolution and scaling -----
            b = _full_convolution(a)
            M = b.max()
            if M <= 0:
                break
            u = b / M                      # u <= 1, u[k_max] = 1

            # ----- Lp‑related scalars -----
            up = u ** p
            S = np.sum(up)                 # >= 1
            # factor = 2 * (S ** ((1-p)/p))  (derived in analysis)
            factor = 2.0 * (S ** ((1.0 - p) / p))

            # ----- gradient (using scaled quantities) -----
            u_pow = u ** (p - 1)           # length 2n‑1
            a_rev = a[::-1]
            conv_rev = _conv_fft(a_rev, u_pow)   # length 3n‑2
            g_u = conv_rev[n - 1: n - 1 + n]     # size n
            grad = factor * g_u

            # ----- step size schedule (simple linear decay) -----
            progress = (now - start) / (deadline - start) if deadline > start else 1.0
            eta = eta0 * (1.0 - progress) + 1e-8

            # ----- back‑tracking line search -----
            improved = False
            for _ in range(5):
                a_candidate = _project_simplex(a - eta * grad)
                cur_val = _objective_value(a_candidate)
                if cur_val < best_val_local - 1e-12:
                    a = a_candidate
                    best_val_local = cur_val
                    best_seq = a.copy()
                    no_improve = 0
                    improved = True
                    print(f"[Lp p={p}] new best = {best_val_local:.6f}")
                    break
                else:
                    eta *= 0.5

            if not improved:
                no_improve += 1
                if no_improve >= max_no_improve:
                    # give up on this p and move to the next one
                    break
            else:
                no_improve = 0

    return best_seq, best_val_local


# ----------------------------------------------------------------------
# Small post‑processing sweep – mixing with uniform
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combos a' = (1‑α)a + α·uniform for a few α values."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final secure call
    deadline = start_time + budget_s - 0.5
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # Problem size – a few‑ten‑thousands entries offers a good trade‑off.
    # --------------------------------------------------------------
    n = 15000

    # --------------------------------------------------------------
    # Initialise a – use a known construction if available,
    # otherwise fall back to a rich parametric sweep.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.503

    # --------------------------------------------------------------
    # Phase A – Adam optimisation on a smooth approximation of max(conv)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.25 * total_time
    best_a, best_val = _adam_phase(
        a0, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # --------------------------------------------------------------
    # Phase B – targeted hill‑climbing on the exact objective
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.70 * total_time
    best_a = _hillclimb_targeted(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    best_val = _objective_value(best_a)

    # --------------------------------------------------------------
    # Phase C – brief simulated annealing
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.90 * total_time
    best_a, best_val = _annealing_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # --------------------------------------------------------------
    # Phase D – Lp‑norm based refinement (smooth max approximation)
    # --------------------------------------------------------------
    lp_deadline = deadline
    best_a, best_val = _lp_refine(
        best_a, n,
        start=phase_c_deadline,
        deadline=lp_deadline,
        best_a=best_a,
        best_val=best_val,
        p_values=(2, 4, 8, 16, 32),
    )

    # --------------------------------------------------------------
    # Final polishing – tiny uniform mixing
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
