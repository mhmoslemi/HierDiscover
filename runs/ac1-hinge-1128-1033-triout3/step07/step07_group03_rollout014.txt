"""
Propose a low‑peak non‑negative step function.

The evaluation function is  2·n·max(conv) / (∑a)², where
   conv = np.convolve(a, a) (full linear convolution).

The dominant term is  max(conv) which, for non‑negative a, equals the
largest value of the a‑reversed inner product.
If the vector is sorted opposite to its reversal the products are
paired “large × small’’ and the peak becomes small.  We therefore
optimise two things:

1.  the *values*  – we minimise a smooth approximation of max(conv)
    (log‑sum‑exp) by mirror‑descent (exponentiated gradient) on the
    probability simplex.
2.  the *ordering* – after each major phase we try a few deterministic
    reorderings (anti‑symmetric, interleaved) and keep the one with the
    lowest evaluation.

The optimiser runs many short restarts (random Dirichlet vectors),
a decreasing ε‑schedule (ε→0 gives the true max), and a final
simple hill‑climbing (swap / mass‑transfer) to polish the best
candidate found before the 1000 s budget expires.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise can create tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Largest values left, smallest right (pairing big with small)."""
    n = a.shape[0]
    s = np.sort(a)                # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[n - 1 - i]          # large on the left
        out[n - 1 - i] = s[i]          # small on the right
    if n % 2:
        out[half] = s[half]
    return out


def _interleave_reorder(a: np.ndarray) -> np.ndarray:
    """
    Alternate large / small entries:
    large, small, second‑large, second‑small, …
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    out = np.empty_like(a)
    left, right = n - 1, 0
    pos = 0
    while pos < n:
        out[pos] = s[left]
        left -= 1
        pos += 1
        if pos < n:
            out[pos] = s[right]
            right += 1
            pos += 1
    return out


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector onto the probability simplex (∑=1)."""
    # classic O(n) algorithm – but here n is moderate, so a simple method works.
    v = np.clip(v, 0, None)
    s = v.sum()
    if s <= 0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Smooth approximation of max(conv) and its gradient w.r.t. a.

    Returns
        loss  –  max(conv) + eps * logsumexp((conv - max)/eps)
        grad  –  ∂loss/∂a   (same shape as a, non‑negative)
    """
    n = a.shape[0]

    # convolution a*a
    conv = _full_convolution(a)               # length 2n-1
    max_c = conv.max()

    # log‑sum‑exp (stable)
    shifted = (conv - max_c) / eps
    # subtract max for stability before exponentiation
    m = shifted.max()
    w = np.exp(shifted - m)
    Z = w.sum()
    p = w / Z                                 # weights, sum to 1

    # gradient: g_i = 2 * Σ_k p_k * a_{k-i}  (zero where index out‑of‑range)
    p_rev = p[::-1]                           # reverse for convolution
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the needed slice is indices n‑1 … 2n‑2 (inclusive) → length n
    seg = conv_ap[n - 1: 2 * n - 1]
    grad = 2.0 * seg

    # loss (smooth max)
    loss = max_c + eps * (np.log(Z) + m)      # add back the subtracted m
    return loss, grad


def _exponential_mirror_descent(a: np.ndarray,
                                eps: float,
                                step: float,
                                max_iters: int,
                                deadline: float):
    """
    One phase of mirror‑descent on the smooth surrogate.
    Returns the best vector encountered in this phase.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())

    for it in range(max_iters):
        if time.time() >= deadline - 2.0:
            break
        loss, grad = _smooth_max_and_grad(a, eps)

        # exponentiated update (mirror‑descent on simplex)
        a = a * np.exp(-step * grad)

        # projection back onto the simplex
        a = _project_to_simplex(a)

        # every 20 iterations we evaluate the true objective
        if (it & 31) == 0:
            cur = evaluate_sequence(a.tolist())
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[eps {eps:.1e} it {it}] new best = {best_val:.6f}")

            # try the two deterministic reorderings
            for reorder in (_anti_symmetric_reorder, _interleave_reorder):
                a_ord = reorder(a)
                cur2 = evaluate_sequence(a_ord.tolist())
                if cur2 < best_val:
                    best_val = cur2
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[reorder after eps {eps:.1e}] improved to {best_val:.6f}")

        # simple learning‑rate decay
        step *= 0.995

    return best_a, best_val


def _hill_climb_refine(a: np.ndarray,
                       cur_val: float,
                       deadline: float):
    """
    Very cheap random local search (swap or small mass transfer).
    Accepts only strict improvements.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = cur_val

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = best_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] < 1e-12:
                continue
            delta = best_a[i] * 0.03 * rng.random()
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta

        cand = _project_to_simplex(cand)
        val = evaluate_sequence(cand.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_a = cand
            print(f"[refine] new best = {best_val:.6f}")

    return best_a, best_val

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine runs several optimisation phases within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 5.0               # few seconds safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initial candidate – try to reuse a previously discovered vector.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linearly interpolate to the target length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)               # uniform start

    a = _project_to_simplex(a)
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, eval={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – a couple of random Dirichlet restarts (to avoid a poor basin).
    # --------------------------------------------------------------
    for r in range(3):
        if time.time() >= deadline - 200:
            break
        a_rand = rng.exponential(scale=1.0, size=n)   # Dirichlet-like
        a_rand = _project_to_simplex(a_rand)
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand {r}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – mirror‑descent on smooth max (several restarts, ε‑schedule).
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=12)
    restart = 0
    while time.time() < deadline - 120 and restart < 4:
        restart += 1
        # start from the current best but perturb a little
        w = np.log(best_a + 1e-12)
        w += rng.normal(scale=0.15, size=n)
        a_start = np.exp(w - np.max(w))
        a_start /= a_start.sum()

        for eps in eps_schedule:
            if time.time() >= deadline - 100:
                break
            # step size adapted to the current ε (larger ε → larger step)
            step = 0.8 * np.sqrt(eps) + 0.02
            max_iters = 500
            a_start, val_start = _exponential_mirror_descent(
                a_start, eps, step, max_iters, deadline)
            if val_start < best_val:
                best_val = val_start
                best_a = a_start.copy()
                print(f"[restart {restart} eps {eps:.1e}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – aggressive hill‑climbing (swap / mass‑transfer).
    # --------------------------------------------------------------
    if time.time() < deadline - 20:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # Final evaluation & return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best reported value = {final_score:.6f}")
    except Exception as exc:    # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
