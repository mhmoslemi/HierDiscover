def _mirror_descent(a, cur_val, deadline, rng):
    n = a.shape[0]
    cur_a = a.copy()
    best_a = a.copy()
    best_val = cur_val
    eta = 0.1
    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))
        
        # gradient of max conv with respect to a:
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] = 2.0 * cur_a[k - i_range]
        
        # Mirror descent step
        new_a = cur_a * np.exp(-eta * grad)
        new_a_sum = new_a.sum()
        if new_a_sum == 0:
            break
        new_a /= new_a_sum

            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
        else:
            eta *= 0.9  # reduce step size
            # maybe continue with same cur_a and smaller step size
        # optionally increase eta if improvement
        # break if eta too small

"""
Hybrid optimizer for the ``evaluate_sequence`` objective.

The routine proceeds in four loosely‑coupled stages, each of which
receives a share of the 1000 s budget.

1. **Smoothed surrogate optimisation** –  The non‑smooth max‑convolution
   is replaced by a log‑sum‑exp surrogate.  A continuation in the
   smoothing parameter ε together with L‑BFGS‑B provides a good
   approximate minimiser on the full‑dimensional simplex.

2. **Random restarts** –  The surrogate optimiser is repeatedly
   re‑initialised from a Gaussian perturbation of the current best
   point; only the tighter side of the ε‑schedule is used, allowing the
   solver to explore new basins of attraction.

3. **Mirror‑descent polishing** –  On the exact objective
   ``2·n·max(conv)`` a projected entropy‑mirror update (multiplicative
   weight‑update) is applied to the active lag.  This step often slips
   below the plateau left by the surrogate optimiser.

4. **Simulated‑annealing mass‑transfer** –  A final stochastic search
   moves small fractions of probability mass from indices that
   contribute heavily to the maximal lag toward low‑mass indices.
   Moves are accepted according to a simple Metropolis rule, which
   lets the algorithm escape shallow local minima.

All phases respect the global time limit and keep track of the best
sequence seen.  The returned list is guaranteed to be non‑negative,
normalised (∑a = 1) and is evaluated once at the end with the official
``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)  # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns value and gradient of the surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε),

    where a = softmax(w) (∑a = 1, a_i ≥ 0).
    """
    # soft‑max, guaranteeing feasibility
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # autocorrelation
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing term
    shifted = (conv - max_conv) / eps
    log_sum = np.logaddexp.reduce(shifted)          # = log Σ exp(...)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient wrt a (closed form, see original code)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # a distribution over lags

    # convolution a * rev(p)
    p_rev = p[::-1]
    conv_ap = np.fft.irfft(np.fft.rfft(a, _next_pow2(3 * n - 2)) *
                           np.fft.rfft(p_rev, _next_pow2(3 * n - 2)),
                           _next_pow2(3 * n - 2))[:3 * n - 2]
    seg = conv_ap[n - 1: 2 * n - 1]                # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate optimisation (L‑BFGS) with random restarts
# ----------------------------------------------------------------------
def _optimize_surrogate(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """Runs the smooth log‑sum‑exp optimiser and a few random restarts."""
    best_a = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))
    print(f"[surrogate] initial objective = {best_val:.6f}")

    # Smoothed continuation schedule
    eps_schedule = np.logspace(-2, -14, num=35)

    w = np.log(best_a + 1e-12)

    # --- main continuation -------------------------------------------------
    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        res = minimize(fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 200,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[surrogate eps {eps:.1e}] new best = {best_val:.6f}")

    # --- random restarts ---------------------------------------------------
    max_restarts = 20
    restarts_done = 0
    while time.time() < deadline - 2.0 and restarts_done < max_restarts:
        restarts_done += 1
        # Gaussian perturbation of the log‑variables
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.4, size=n)

        # only use the finer tail of the schedule
        for eps in eps_schedule[-12:]:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 150,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart {restarts_done}] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent polishing on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _mirror_descent(a0: np.ndarray,
                    cur_val: float,
                    deadline: float,
                    rng: np.random.Generator):
    """Entropy mirror descent (multiplicative update) on the exact objective."""
    n = a0.shape[0]
    cur_a = a0.copy()
    best_a = cur_a.copy()
    best_val = cur_val
    eta = 0.7          # initial step size
    idle = 0           # counter for consecutive non‑improvements

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # multiplicative (mirror) step
        new_a = cur_a * np.exp(-eta * grad)
        s = new_a.sum()
        if s <= 0:
            break
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))
        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            idle = 0
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[mirror] new best = {best_val:.6f}")
            eta = min(eta * 1.04, 2.0)
        else:
            idle += 1
            # shrink step size after a few failures
            if idle >= 8:
                eta = max(eta * 0.6, 5e-4)
                idle = 0

    return best_a, best_val

# Simulated‑annealing mass‑transfer phase
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """Moves small masses away from the active lag, accepting uphill moves."""
    n = a0.shape[0]
    cur_a = a0.copy()
    best_a = cur_a.copy()
    best_val = cur_val

    # initial temperature as a fraction of the current objective
    T = cur_val * 0.3
    decay = 0.9997

    while time.time() < deadline:
        T *= decay
        if T < 1e-12:
            T = 1e-12

        # Identify active lag
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # indices that actually appear in the active lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        if i_range.size == 0:
            continue

        # choose a donor index with probability proportional to its contribution
        contrib = cur_a[i_range] * cur_a[k - i_range]
        if contrib.sum() == 0:
            donor = int(rng.choice(i_range))
        else:
            donor = int(rng.choice(i_range, p=contrib / contrib.sum()))

        # choose a recipient from the low‑mass tail
        low = np.where(cur_a < np.median(cur_a))[0]
        if low.size == 0:
            recip = int(rng.integers(n))
        else:
            recip = int(rng.choice(low))
        if donor == recip:
            continue

        # amount to transfer – a fraction that shrinks with temperature
        frac = min(0.12, T / cur_val)  # at most 12 % of the donor’s mass
        delta = cur_a[donor] * frac
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[recip] += delta

        new_val = 2.0 * n * np.max(_full_convolution(new_a))
        delta_val = new_val - cur_val

        # Metropolis acceptance
        accept = False
        if delta_val < 0:
            accept = True
        else:
            prob = np.exp(-delta_val / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[anneal] new best = {best_val:.6f}")

    return best_a, best_val
