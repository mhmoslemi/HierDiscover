"""
Propose a non‑negative step function (list of heights) that minimises
the ``evaluate_sequence`` objective.

Algorithm
---------
1. **Small‑scale LP refinement**
   * Build a modest‑size problem (n≈1500) where the objective is to
     minimise the supremum of the self‑convolution.
   * Starting from a Beta‑family seed, repeatedly solve a linear programme
     (maximise Σg subject to f∗g ≤ M) and take a convex combination of the
     current vector and the LP solution.
   * This yields a shape with a very flat auto‑correlation.

2. **Upsample to the target size**
   * Interpolate the refined small vector to n=5000 (the size used in the
     reference solution) and normalise the sum to 1.

3. **Adam optimisation on a smooth max**
   * Optimise a soft‑max approximation of ``max(f∗f)`` while keeping the
     vector on the probability simplex (sum = 1).  The implementation
     follows the Adam optimiser from the previous state‑of‑the‑art code.

4. **Simulated‑annealing hill‑climbing**
   * A local search that moves a tiny amount of mass from an index that
     contributes heavily to the current maximal convolution entry to an
     index that contributes little.  Accepts uphill moves with a
     Metropolis probability that cools to zero.

5. **Return the best vector found** (the final list is passed to
   ``evaluate_sequence`` for the official score).

The whole routine respects the 1000 s time budget, keeping a 0.5 s safety
margin for the final call to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a vector that is normalised to sum == 1
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Smooth approximation of max(conv(a,a)) and its gradient.
    Returns (smooth_val, grad_a) where ``smooth_val`` is the smooth max
    (without the outer factor 2·n) and ``grad_a`` is the gradient of
    2·n·smooth_max w.r.t. ``a``.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # gradient of Σ p_k * b_k w.r.t a
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 4n from 2·n·max
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a smooth approximation of the objective.
    Returns the best (a, value) seen before *deadline*.
    """
    # logits for a soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.001
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 25_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restarting after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass from heavy‑contributing
    indices to light‑contributing ones.  Accepts uphill moves with a
    Metropolis probability that cools to zero.
    """
    a = a0.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.012, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (prob. ∝ contribution)
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index (biased towards low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric Beta‑family sweep (used for initial seeds)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # grid for exponents (log‑space mix)
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val

    # 2. Linear‑program refinement (small problem)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception as exc:
        print("[LP] SciPy not available – skipping LP stage:", exc)
        lp_success = False
    else:
        lp_success = True
        max_lp_iter = 12            # modest number to keep within time budget
        step = 0.3                  # convex‑combination step size
        # pre‑compute static index arrays for the Toeplitz‐like matrix
        row_idx = np.repeat(np.arange(n_small), n_small) + np.tile(np.arange(n_small), n_small)
        col_idx = np.repeat(np.arange(n_small), n_small)

        for it in range(max_lp_iter):
            now = time.time()
            if now >= deadline:
                break
            M = _full_convolution(a_small).max()
            data = np.repeat(a_small, n_small)
            A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n_small - 1, n_small))

            c = -np.ones(n_small)   # maximise Σg  (= minimise –Σg)
            b_ub = np.full(2 * n_small - 1, M)
            bounds = [(0, None)] * n_small

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success:
                print(f"[LP {it}] linprog failed: {res.message}")
                break

            g = res.x
            sum_g = g.sum()
            if sum_g <= 0:
                break
            g /= sum_g                # normalise to Σg = 1

            # convex update
            a_small = (1 - step) * a_small + step * g
            a_small = np.maximum(a_small, 0.0)
            a_small /= a_small.sum()

            # shrink step gradually
            step = max(step * 0.93, 0.02)

            obj = _objective(a_small)
            print(f"[LP {it}] objective (small) = {obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Upsample to the large problem size
    # ------------------------------------------------------------------
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(x_large, x_small, a_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()

    # ------------------------------------------------------------------
    # 4. Adam optimisation (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _adam_optimize(a_large, n_large,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=a_large,
                                      best_val=_objective(a_large),
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n_large,
                                         deadline=deadline,
                                         start_time=phase_a_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # 6. Finalisation and report
    # ------------------------------------------------------------------
    best_a = best_a / best_a.sum()    # guard against numerical drift
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
