def _subgradient_descent(a, deadline):
    n = a.shape[0]
    best_val = _fast_evaluate(a)
    cur_a = a.copy()
    cur_val = best_val
    iter = 0
    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_val = conv.max()
        if max_val == 0:
            break
        k_star = int(conv.argmax())
        # compute subgradient g
        g = np.zeros_like(cur_a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        jdx = k_star - idx
        g[idx] = cur_a[jdx]
        # attempt line search
        alpha = 1.0  # start
        improved = False
        while alpha > 1e-8:
            new_a = cur_a - alpha * g
            new_a = np.maximum(new_a, 0.0)
            s = new_a.sum()
            if s <= 1e-12:
                alpha *= 0.5
                continue
            new_a /= s
            new_val = _fast_evaluate(new_a)
            if new_val < cur_val - 1e-12:
                cur_a = new_a
                cur_val = new_val
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            # if not improved, maybe do random move to escape
            # random perturbation
            rng = np.random.default_rng()
            i, j = rng.integers(0, n, size=2)
            delta = cur_a[i] * 0.1 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            new_a = np.maximum(new_a, 0.0)
            new_a /= new_a.sum()
            cur_a = new_a
            cur_val = _fast_evaluate(cur_a)
            continue
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = cur_a.copy()
        iter += 1
        if iter % 100 == 0:
            print(f"[subgrad iter {iter}] cur best = {best_val:.6f}")
    return best_a, best_val

"""
propose_candidate
=================

Search for a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The method combines several ideas that were not
used together before:

* a diverse pool of initialisations (uniform, decreasing linear,
  exponential, Dirichlet, the historic best);
* a long L‑BFGS optimisation of a smooth log‑sum‑exp surrogate of the
  max‑convolution (much finer ε schedule than the reference code);
* a dedicated *sub‑gradient* polishing phase that works on the exact
  max‑convolution.  At each iteration we compute the true index ``k*``
  where the convolution is maximal, form the sub‑gradient
  ``g_i = a_{k*‑i}``, and perform an additive line‑search on the simplex.
  This phase is able to shave the final bound a few hundredths.
* a cheap final hill‑climbing (swap / mass‑transfer) pass.

All operations keep the vector non‑negative and normalised
(``Σ a_i = 1``); the official ``evaluate_sequence`` is called only once,
at the end, to obtain the final score.

The routine respects a 1000 s wall‑clock budget (a small safety margin
is kept) and prints progress information.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smooth surrogate (required by scipy.minimize)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None  # smoothing parameter ε
_GLOBAL_N = None    # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors.
    Returns 2·n·max(conv) / (Σ a_i)².
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max
    parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # --- soft‑max ----------
    w_shift = w - np.max(w)           # numeric stability
    a = np.exp(w_shift)
    a /= a.sum()

    # --- convolution ----------
    conv = _full_convolution(a)       # length 2·n‑1
    max_c = conv.max()

    # --- smooth max ----------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # --- gradient w.r.t. a ----------
    weights = np.exp(shifted)
    p = weights / weights.sum()       # distribution over lag indices

    # g_a = ∂/∂a ( Σ_k p_k * conv_k )
    # conv = a * a (linear convolution).  The Jacobian w.r.t a is a
    # reversed convolution with p.
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the relevant segment (indices n‑1 … 2n‑2) aligns with a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]      # chain rule for the 2·n factor

    # --- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exact max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         log_tag: str = "[subgrad]") -> tuple[np.ndarray, float]:
    """
    Perform additive sub‑gradient steps on the exact objective
    ``2·n·max(conv) / (sum a)^2``.
    The step direction at each iteration is the sub‑gradient of the max
    convolution (i.e. g_i = a_{k*‑i} where k* is the current maximiser).
    A simple back‑tracking line‑search is used; after each trial the
    vector is projected back onto the probability simplex.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_val = _fast_evaluate(a)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_val = conv.max()
        k_star = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k_star - idx
        g[idx] = a[jdx]                     # g_i = a_{k*‑i}

        # line‑search on α
        alpha = 1.0
        improved = False
        while alpha > 1e-9:
            a_new = a - alpha * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-12:
                alpha *= 0.5
                continue
            a_new /= s
            val = _fast_evaluate(a_new)
            if val < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # small random perturbation to escape a flat region
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-15:
                delta = a[i] * 0.05 * rng.random()
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a_tmp = np.maximum(a_tmp, 0.0)
                a_tmp /= a_tmp.sum()
                a = a_tmp
                best_val = _fast_evaluate(a)
                best_a = a.copy()

        it += 1
        if it % 100 == 0:
            print(f"{log_tag} iter {it:5d}  best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap hill‑climbing final pass
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Random swaps (20 %) and mass‑transfer moves (80 %).  Only strict
    improvements are kept.
    """
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.08 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[hill] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size – use the length of the known construction if available.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # --------------------------------------------------------------
    # 1. Build a diverse collection of initial candidates.
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # decreasing linear (triangular) shape
    lin = np.arange(n, 0, -1, dtype=float)      # n, n‑1, …, 1
    init_candidates.append(lin / lin.sum())

    # decreasing exponential (r ≈ 0.999)
    r = 0.999
    exp_vec = np.power(r, np.arange(n, dtype=float))
    init_candidates.append(exp_vec / exp_vec.sum())

    # random Dirichlet – several concentrations
    for alpha in (0.3, 1.0, 5.0, 20.0):
        cand = rng.gamma(shape=alpha, scale=1.0, size=n)
        init_candidates.append(cand)

    # anti‑symmetric reorder of the best uniform candidate
    # (large left, small right) – already covered by the linear case,
    # but we keep it for completeness.
    def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
        sorted_vals = np.sort(a)                # ascending
        new_a = np.empty_like(a)
        half = n // 2
        for i in range(half):
            new_a[i] = sorted_vals[-1 - i]          # large → left side
            new_a[-1 - i] = sorted_vals[i]          # small → right side
        if n % 2:
            new_a[half] = sorted_vals[half]
        return new_a

    init_candidates.append(_anti_symmetric_reorder(init_candidates[0]))

    # normalise everything onto the simplex
    init_candidates = [c / c.sum() for c in init_candidates]

    # --------------------------------------------------------------
    # 2. Quick evaluation of the cheap candidates.
    # --------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] eval = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. L‑BFGS on the smooth surrogate.
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-14), num=18)
    w = np.log(best_a + 1e-12)          # soft‑max parameters
    max_restarts = 20
    restart = 0

    while time.time() < deadline - 200.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        if restart % 5 == 0:
            w = np.random.randn(n) * 2.0   # fresh random start

        for eps in eps_schedule:
            if time.time() >= deadline - 200.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 600,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x
            # convert back to probabilities via soft‑max
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # after each restart, enforce a monotone decreasing shape
        # (this often reduces the max‑convolution)
        a_mon = np.sort(best_a)[::-1]     # descending
        a_mon /= a_mon.sum()
        val_mon = _fast_evaluate(a_mon)
        if val_mon < best_val:
            best_val = val_mon
            best_a = a_mon.copy()
            print(f"[monotone reorder] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Exact sub‑gradient polishing (adds a few hundred iterations).
    # --------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print("[subgrad] starting exact sub‑gradient polishing")
        sub_deadline = deadline - 120.0
        best_a, best_val = _subgradient_descent(best_a, sub_deadline,
                                                log_tag="[subgrad]")

    # --------------------------------------------------------------
    # 5. Final cheap hill‑climbing refinement.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[hill] final cheap hill‑climb")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # 6. Final safety check with the official evaluator.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence => {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
