def project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _subgradient_phase(a_start, n, start_time, deadline, best_a, best_val, target_val=1.503):
    a = a_start.copy()
    rng = np.random.default_rng()
    no_improve = 0
    IMPROVE_LIMIT = 50000
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # compute gradient
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            g[idx] = 4.0 * n * a[k_max - idx]

        # step size schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        # from eta_start=0.2 to eta_end=0.001
        eta = 0.2 * (1 - progress) + 0.001 * progress

        # multiplicative update
        a = a * np.exp(-eta * g)

        # renormalize and avoid zeros
        total = a.sum()
        if total <= 0:
            a = best_a.copy()
            total = a.sum()
        a = a / total

        # stagnation
        if no_improve > IMPROVE_LIMIT:
            # perturb best
            a = best_a.copy()
            noise = rng.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0)
            a_sum = a.sum()
            if a_sum > 0:
                a = a / a_sum
            else:
                a = best_a.copy()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Subgrad {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val

"""
propose_candidate
-----------------
A multi‑phase optimiser for the “step‑function” objective used by
`evaluate_sequence`.  The code works on a fixed length `n` (≈9 k) and
combines:

1. An Adam optimiser on a smooth‐max approximation of the maximal
   self‑convolution (phase A).
2. A fast hill‑climber that transfers a small amount of mass from
   indices that heavily contribute to the current maximal convolution
   entry to “quiet’’ indices (phase B).
3. A simple exponentiated‑gradient / sub‑gradient descent that works on
   the exact max‑convolution (phase C).  It follows the sub‑gradient
   `∂/∂a_k 2·n·max_i (a*a)_i = 4·n·a_{i_max‑k}` and updates the vector
   by a multiplicative step, keeping it on the simplex.
4. A short uniform‑mixing sweep (phase D).

The three optimisation phases share a common time budget (≈1000 s)
and keep the best vector seen throughout.  The routine starts from a
hand‑crafted deterministic family or, if available, from the previously
found construction `height_sequence_1`.  All numbers are kept
non‑negative and renormalised to sum to 1, so the final call to
`evaluate_sequence` returns the true objective value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) is assumed to be 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple deterministic families for an initial guess
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # linear decreasing
    try_candidate(np.arange(n, 0, -1, dtype=float))

    # U‑shaped linear
    idx = np.arange(n)
    lin_u = np.maximum(idx + 1, n - idx)
    try_candidate(lin_u)

    # quadratic decreasing
    try_candidate((np.arange(n, 0, -1, dtype=float) ** 2))

    # quadratic U‑shape
    quad_u = (np.arange(n) + 1) ** -2 + (np.arange(n, 0, -1) + 0) ** -2
    try_candidate(quad_u)

    # half‑wave sine (symmetric)
    try_candidate(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))

    # cosine with peaks at the ends
    try_candidate(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # power‑law U‑shape
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_candidate(a)

    # exponential left‑hand side
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        try_candidate(a)

    # symmetric exponential
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_candidate(a)

    # truncated exponential (prefix only)
    for frac in (0.10, 0.15, 0.20, 0.25, 0.30):
        K = max(1, int(frac * n))
        for r in (0.99, 0.98, 0.95, 0.90):
            a = np.zeros(n, dtype=float)
            a[:K] = r ** np.arange(K, dtype=float)
            try_candidate(a)

    # a few random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(12):
        a = rng.dirichlet(np.ones(n))
        try_candidate(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam optimisation on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth max of the self‑convolution."""
    w = np.log(a0 + 1e-12)                 # logits, Σa = 1

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 60000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- a on the simplex ----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ---- exact objective (for bookkeeping) ----
        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedules
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)          # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()            # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        # ∂/∂a  2·n·∑_k p_k (a*a)_k  = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]    # n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.008, size=n)

        # restart if stagnated
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Mass‑transfer hill‑climber that works on the exact objective."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0
    rng = np.random.default_rng()

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – proportional to contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – preferably where the reverse contribution is tiny
        if rng.rand() < 0.35:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = rng.choice(n, p=probs_j)
            else:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
        else:
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        # amount to move
        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:          # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = rng.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – exponentiated‑gradient / sub‑gradient descent on the exact max
# ----------------------------------------------------------------------
def _subgradient_phase(a_start: np.ndarray,
                       n: int,
                       start: float,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simple exponentiated‑gradient descent that directly minimises
    `2·n·max(conv(a,a))`.  Uses the exact sub‑gradient
    `g_i = 4·n·a_{k_max-i}` and updates `a_i ← a_i·exp(-η·g_i)`,
    followed by a renormalisation onto the simplex.
    """
    a = a_start.copy()
    rng = np.random.default_rng()
    no_improve = 0
    IMPROVE_LIMIT = 50000
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient (vectorised)
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            g[idx] = 4.0 * n * a[k_max - idx]

        # step‑size schedule (linear annealing)
        progress = (now - start) / (deadline - start)
        eta = 0.20 * (1.0 - progress) + 0.001 * progress

        # multiplicative update, keeps positivity automatically
        a = a * np.exp(-eta * g)

        # renormalise onto the simplex
        total = a.sum()
        if total <= 0:
            a = best_a.copy()
            total = a.sum()
        a = a / total

        # stagnation handling – random shake around the best seen so far
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = rng.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s
            no_improve = 0
            print(f"[Subgrad {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Small post‑processing sweep – mixing with uniform
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a' = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function inside the given time budget.
    Returns a list of floats (Σ≈1) that (hopefully) yields a low value for
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # Problem size – a few thousand entries gives a good trade‑off.
    n = 9000

    # ------------------------------------------------------------------
    # Initialise a – either from a previously found construction or via
    # the parametric sweep defined above.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.503

    total_time = deadline - start_time

    # --------------------------------------------------------------
    # Phase A – Adam optimiser on a smooth approximation (≈25 % of time)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.25 * total_time
    best_a, best_val = _adam_phase(
        a0, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # --------------------------------------------------------------
    # Phase B – targeted hill‑climber (≈55 % of time)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.80 * total_time
    best_a = _hillclimb_targeted(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )
    best_val = _objective_value(best_a)

    # --------------------------------------------------------------
    # Phase C – exponentiated‑gradient / sub‑gradient fine‑tuning
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.95 * total_time

        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,

    best_a, best_val = _mix_with_uniform(best_a, n)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    try:

        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
