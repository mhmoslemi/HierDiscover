"""
Propose a non‑negative step function (a probability vector) that
minimises 2·n·max(conv(a,a)).  The algorithm combines:

1. A rich family of parametric initial guesses.
2. A smooth‑max Adam optimiser (log‑softmax parametrisation).
3. A continuation‑L‑BFGS optimiser on the same smoothed objective,
   gradually decreasing the smoothing temperature.
4. A refined exact‑objective hill‑climber that moves tiny amounts of
   mass from the most harmful indices to the least harmful ones.
5. A short uniform‑mix sweep for final polishing.

All phases respect a 1000 s time budget, the vector is kept on the
simplex (non‑negative, sum = 1), and the final candidate is returned as
a plain Python list.
"""
import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a)=1)  →  2·n·max(conv)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits w
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (value, grad_w) for the smoothed objective
        2·n·smooth_max(conv(a,a)),
    where a = softmax(w) and smooth_max(b) = eps*log(sum(exp(b/eps))).
    """
    # soft‑max a on the simplex
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = a.sum()
    a /= a_sum

    # convolution a*a (full length 2n-1)
    b = _full_convolution(a)
    max_b = b.max()

    # smooth max via log‑sum‑exp with temperature eps
    shifted = (b - max_b) / eps
    # clip to avoid overflow when eps is extremely small
    shifted = np.minimum(0.0, shifted)   # all entries ≤ 0
    weights = np.exp(shifted)
    sum_weights = weights.sum()
    p = weights / sum_weights            # soft‑max probabilities over positions

    smooth_max = max_b + eps * np.log(sum_weights)
    val = 2.0 * n * smooth_max

    # gradient w.r.t. a: ∂/∂a 2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)        # length 3n‑2
    # central segment corresponds to ∂/∂a_i
    segment = conv_ab[n - 1: 2 * n - 1]  # n entries
    grad_a = 4.0 * n * segment[::-1]     # reverse to align with a_i indexing

    # chain rule through the soft‑max (log‑softmax) parametrisation
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return val, grad_w


# ----------------------------------------------------------------------
# A quick library of low‑dimensional parametric families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising normalised vector and its exact objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ---- simple deterministic families ---------------------------------
    # decreasing linear
    try_candidate(np.arange(n, 0, -1, dtype=float))

    # U‑shape linear
    idx = np.arange(n)
    try_candidate(np.maximum(idx + 1, n - idx))

    # quadratic decreasing
    try_candidate((np.arange(n, 0, -1, dtype=float) ** 2))

    # quadratic U‑shape
    quad_u = (np.arange(n) + 1) ** -2 + (np.arange(n, 0, -1) + 0) ** -2
    try_candidate(quad_u)

    # half‑wave sine (symmetric)
    try_candidate(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))

    # cosine with peaks at the ends
    try_candidate(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # power‑law symmetric (U‑shape)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        try_candidate(i ** (-p) + (n - i + 1) ** (-p))

    # exponential left‑hand side
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        try_candidate(r ** np.arange(n, dtype=float))

    # symmetric exponential (U‑shaped)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        try_candidate(r ** i + r ** (n - 1 - i))

    # truncated exponential (mass only on a prefix)
    for frac in (0.10, 0.15, 0.20, 0.25, 0.30):
        K = max(1, int(frac * n))
        for r in (0.99, 0.98, 0.95, 0.90):
            a = np.zeros(n, dtype=float)
            a[:K] = r ** np.arange(K, dtype=float)
            try_candidate(a)

    # a few random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(12):
        try_candidate(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam optimisation on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam on a smoothed max‑objective (log‑sum‑exp)."""
    w = np.log(a0 + 1e-12)                 # logits, Σa = 1

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 60000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- obtain a on the simplex -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ----- exact objective (for bookkeeping) -----
        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- schedules -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max via log‑sum‑exp -----
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        shifted = np.minimum(0.0, shifted)      # protect against overflow
        weights = np.exp(shifted)
        p = weights / weights.sum()            # soft‑max over positions

        # ----- gradient of smooth objective w.r.t. a -----
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]    # central n entries
        grad_a = 4.0 * n * segment[::-1]

        # ----- chain rule through log‑softmax -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter of logits
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.008, size=n)

        # restart if stagnated
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Continuation L‑BFGS on the smoothed objective
# ----------------------------------------------------------------------
def _lbfgs_phase(a_start: np.ndarray, n: int,
                 start: float, deadline: float,
                 best_a: np.ndarray, best_val: float,
                 target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    L‑BFGS optimisation on the smoothed objective.
    The temperature `eps` is reduced geometrically (continuation).
    """
    w = np.log(a_start + 1e-12)   # initial logits

    # temperature schedule (geometric)
    eps_schedule = (1e-2, 1e-4, 1e-6, 1e-8)
    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        # wrapper returning (value, grad) for scipy
        def fun(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        # run limited‑memory BFGS
        res = scipy.optimize.minimize(fun,
                                      x0=w,
                                      method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': 250,
                                               'disp': False})
        w = res.x

        # evaluate the exact objective
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        cur_val = _objective_value(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[L‑BFGS eps={eps:.0e}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """
    Greedy mass‑transfer hill‑climber.
    Moves a fraction of mass from indices that heavily contribute to the
    current maximal convolution entry towards indices that contribute
    little.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # schedule for transferred fraction (start large → end tiny)
    max_frac_start, max_frac_end = 0.25, 0.0005
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # pick source index proportional to its contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            # nothing to move from this index – try again
            continue

        # pick a target index where the reversed contribution is tiny
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        # favour locations with small rev_contrib
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        w_sum = w.sum()
        if w_sum > 0 and np.random.rand() < 0.7:
            probs_j = w / w_sum
            j = np.random.choice(n, p=probs_j)
        else:
            # fallback uniform choice (excluding i)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move (small fraction of source)
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # Incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – small random perturbation + re‑normalisation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a


# ----------------------------------------------------------------------
# Small post‑processing sweep – mixing with uniform
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a' = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 9000

    # --------------------------------------------------------------
    # Initialise from a previous run if available, else from the library.
    # --------------------------------------------------------------
    try:
        # height_sequence_1 is injected by the surrounding harness.
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.ndim == 1 and old.shape[0] != n:
            # Linear interpolation to the required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    print(f"[Init] objective = {best_val:.6f}")

    TARGET = 1.503

    # --------------------------------------------------------------
    # Phase A – Adam optimisation on a smooth approximation of max(conv)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.25 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        a0, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # --------------------------------------------------------------
    # Phase B – Continuation L‑BFGS with decreasing temperature
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * (deadline - start_time)
    best_a, best_val = _lbfgs_phase(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # --------------------------------------------------------------
    # Phase C – Precise hill‑climbing on the exact objective
    # --------------------------------------------------------------
    best_a = _hillclimb_targeted(
        best_a, n,
        start=phase_b_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    best_val = _objective_value(best_a)

    # --------------------------------------------------------------
    # Final tiny uniform mixing sweep
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # --------------------------------------------------------------
    # Final secure evaluation (the wrapper will clamp/truncate)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
