"""
Propose a non‑negative step function that (approximately) minimises
``evaluate_sequence``.  

The algorithm consists of several stages:

1. **Initialisation** – start from the previously discovered construction
   (``height_sequence_1``) if it exists, otherwise from a uniform or random
   Dirichlet vector.

2. **Random seeding** – try a handful of random vectors to escape a poor
   start.

3. **Smooth L‑BFGS optimisation** – minimise a log‑sum‑exp smooth surrogate
   of the max‑convolution using L‑BFGS‑B.  The surrogate is gradually
   sharpened (ε→0).

4. **Sub‑gradient Adam descent** – directly optimise the true objective.
   The gradient of the max‐convolution is computed by a linear correlation
   between the current sequence and a mask that selects all (near‑)maximal
   convolution entries.  Adam‑style moment estimates give stable steps.
   The sequence is represented in log‑softmax coordinates to keep it
   non‑negative and on the simplex.

5. **Hill‑climbing refinement** – occasional random swaps / mass transfers
   keep the search from stagnating.

All convolutions are performed with FFTs (O(n log n)) so the method works
well for n≈8000 within the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives; clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _linear_correlation(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Compute the linear correlation (a ⋆ mask) for all integer shifts.
    The result has length len(a).  Implemented with FFTs.
    """
    n = a.shape[0]
    m = mask.shape[0]          # = 2*n-1
    size = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask, size)
    # Correlation corresponds to A * conj(M) in the frequency domain.
    corr = np.fft.irfft(fa * np.conj(fm), size)[: n + m - 1]
    start = m - 1               # shift 0 is at index (len(mask)-1)
    return corr[start:start + n]


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Pair large values with small ones by reversing one half of the sorted list."""
    n = a.shape[0]
    sorted_vals = np.sort(a)       # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large → left side
        new_a[n - 1 - i] = sorted_vals[i]      # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε (set by the optimiser)
_GLOBAL_N = None     # length of the current problem vector


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # self‑convolution (FFT)
    conv = _full_convolution(a)
    max_conv = conv.max()

    # smoothed max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient of the smooth max w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()           # distribution over convolution indices

    # derivative of max part:
    #   ∂/∂a_i Σ_k p_k * Σ_j a_j a_{k-j}
    # can be obtained efficiently via a correlation of a with reversed p.
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the middle n entries correspond to the derivative for each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # derivative w.r.t. a (chain rule from a*a)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule: from 5·10⁻³ down to 10⁻⁹ (geometric)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=12)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 250,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric re‑ordering heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:

            best_a = a_ord.copy()

    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val

def _subgradient_adam(a_start: np.ndarray, deadline: float):
    """
    Adam‑style optimiser on the exact objective.
    Works with log‑softmax parameters to stay on the simplex.
    Returns the best (a, value) pair found before ``deadline``.
    """
    n = a_start.shape[0]
    w = np.log(a_start + 1e-12)           # log‑softmax parameters
    best_a = a_start.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[subgrad init] value = {best_val:.6f}")

    # Adam hyper‑parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_opt = 1e-8
    lr0 = 0.12               # base learning rate
    t = 0

    while time.time() < deadline - 5.0:
        t += 1

        # ---- current point on the simplex ----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ---- convolution and mask of (near) maximal entries ----
        conv = _full_convolution(a)
        max_c = conv.max()
        # keep all entries whose value is within a tiny relative tolerance
        rel_tol = 1e-9
        thr = max_c - max(max_c * rel_tol, 1e-12)
        mask = (conv >= thr).astype(float)   # shape 2*n-1

        # ---- sub‑gradient of the max‑convolution (linear correlation) ----
        g = _linear_correlation(a, mask)     # shape n

        # gradient of the objective O = 2*n*max_c   (since sum(a)=1)
        grad_O = 2.0 * n * g
        dot = np.dot(grad_O, a)
        grad_w = a * (grad_O - dot)          # chain rule through the soft‑max

        # ---- Adam update on w ----
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # simple learning‑rate decay (once every 500 iterations)
        lr = lr0 * (0.995 ** (t // 500))

        w = w - lr * m_hat / (np.sqrt(v_hat) + eps_opt)

        # ---- occasional true evaluations ----
        if t % 200 == 0:
            w_shift = w - np.max(w)
            a_eval = np.exp(w_shift)
            a_eval /= a_eval.sum()
            val = evaluate_sequence(a_eval.tolist())
            if val < best_val:
                best_val = val
                best_a = a_eval.copy()
                print(f"[subgrad iter {t}] new best = {best_val:.6f}")

        # ---- periodic anti‑symmetric re‑ordering (helps escape local minima) ----
        if t % 4000 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = evaluate_sequence(a_ord.tolist())
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                # re‑initialise the optimiser at the improved point
                w = np.log(best_a + 1e-12)
                m.fill(0.0)
                v.fill(0.0)
                print(f"[subgrad reorder {t}] improved to {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (random swaps / mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random local search: swaps (20 %) and mass transfers (80 %).
    Keeps only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.07 * rng.random()   # up to 7 % transfer
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a few seconds safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 works well with the FFT based convolutions.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from a known construction if possible
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the desired length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)          # uniform start

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.full(n, 1.0 / n)
    a /= a.sum()

    # ------------------------------------------------------------------
    # Evaluate the initial candidate
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – a few random restarts (helps avoid a poor basin)
    # ------------------------------------------------------------------
    for tr in range(5):
        if time.time() >= deadline - 200:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate optimisation (L‑BFGS) with a few restarts
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 150 and restarts < 4:
        restarts += 1
        # perturb current best before feeding to L‑BFGS
        w_start = np.log(best_a + 1e-12)
        w_start += rng.normal(scale=0.25, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w_start, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[restart {restarts}] new best after L‑BFGS = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient Adam descent on the true objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        a_sub, val_sub = _subgradient_adam(best_a, deadline)
        if val_sub < best_val:
            best_val = val_sub
            best_a = a_sub.copy()
            print(f"[sub‑gradient] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Ensure the return type complies with the harness
    return best_a.tolist()
