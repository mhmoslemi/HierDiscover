"""
Hybrid optimiser for the step‑function convolution problem (new version).

The algorithm combines several complementary ideas that were not
exhaustively explored in the previous implementation:

  1. **Richer seed pool** – in addition to geometric and power‑law
     families we also add classic window functions (Hann, cosine‑square,
     triangular) and a few random “block” patterns.  All seeds are
     normalised to ℓ₁‑norm = 1.

  2. **Adam optimisation of a smooth log‑sum‑exp surrogate** – the same
     surrogate used before (soft‑max parametrisation + log‑sum‑exp
     smoothing) but optimised with the Adam optimiser instead of L‑BFGS.
     Adam’s adaptive learning‑rate and momentum allow us to traverse
     plateaus that L‑BFGS can stall on.

  3. **Aggressive re‑ordering** – after each Adam stage we test three
     deterministic permutations (anti‑symmetric, alternating,
     descending) that often uncover hidden symmetries.

  4. **Extended simulated annealing** – a temperature‑controlled search
     that mixes three move types (swap, mass‑transfer, targeted transfer
     from the pair contributing to the current peak).  The schedule is
     longer than before, giving the Markov chain more time to wander.

  5. **Final hill‑climbing** – cheap local improvements (swap / small
     transfer) until the time budget expires.

All heavy work is performed with the FFT‑based evaluator `_eval_fast`,
which reproduces the behaviour of the provided `evaluate_sequence`
while being much faster.  The function returns a plain Python list of
floats and respects the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global values used by the smooth surrogate (soft‑max + log‑sum‑exp)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current Adam run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clamp tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast surrogate for `evaluate_sequence` (uses FFT, no clamping)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε (log‑sum‑exp approximation of the max)
    and its gradient w.r.t. the soft‑max parameters w.
    """
    # ---- soft‑max (positivity + ℓ₁ = 1) ----
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= np.sum(a)

    # ---- autocorrelation a*a ----
    conv = _fft_convolve_self(a)
    max_conv = conv.max()

    # ---- log‑sum‑exp smoothing of the max ----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (closed‑form) ----
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # probability over convolution indices
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                  # a * p_rev
    seg = conv_ap[n - 1: 2 * n - 1]                 # central part, length n
    grad_a = 4.0 * n * seg[::-1]

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large values left, small values right (pairwise)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = sorted_vals[n - 1 - i]
        out[n - 1 - i] = sorted_vals[i]
    if n % 2:
        out[half] = sorted_vals[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    sorted_vals = np.sort(a)
    left, right = 0, len(sorted_vals) - 1
    out = np.empty_like(a)
    idx = 0
    while left <= right:
        out[idx] = sorted_vals[right]
        idx += 1
        right -= 1
        if left <= right:
            out[idx] = sorted_vals[left]
            idx += 1
            left += 1
    return out


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Simply sort descending."""
    return np.sort(a)[::-1]


# ----------------------------------------------------------------------
# Simple parametric families for seeding
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """a_i ∝ q**i, normalised to sum = 1 (0 < q < 1)."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q ** n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)**p, normalised to sum = 1 (p < 0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Simulated annealing & hill‑climbing helpers
# ----------------------------------------------------------------------
def _simulated_annealing_extended(a: np.ndarray, cur_val: float,
                                 deadline: float) -> (np.ndarray, float):
    """SA mixing three move types and a targeted reduction of the max convolution."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.05, 1e-8)   # start temperature
    decay = 0.99995
    it = 0

    while time.time() < deadline:
        move = rng.random()
        if move < 0.4:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif move < 0.8:                   # random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * 0.1 * rng.random()
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        else:                               # targeted transfer
            conv = _fft_convolve_self(a)
            k_max = int(np.argmax(conv))
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_best, prod_best = None, -1.0
            for i in range(i_start, i_end + 1):
                j = k_max - i
                prod = a[i] * a[j]
                if prod > prod_best:
                    prod_best = prod
                    i_best = i
                    j_best = j
            if i_best is None:
                continue
            source = i_best if a[i_best] >= a[j_best] else j_best
            target = rng.integers(0, n)
            if source == target or a[source] == 0.0:
                continue
            delta = a[source] * 0.1 * rng.random()
            new_a = a.copy()
            new_a[source] -= delta
            new_a[target] += delta

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        T = max(T * decay, 1e-12)
        it += 1
    return best_seq, best_val


def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Cheap local improvements (swap / small transfer)."""
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.15:           # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                             # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-14:
            cur_a = new_a
            cur_val = val
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Adam optimiser for the smooth surrogate
# ----------------------------------------------------------------------
def _adam_optimize(w0: np.ndarray,
                   eps_schedule,
                   max_iter_per_eps: int,
                   deadline: float) -> (np.ndarray, float):
    """
    Adam optimisation of the smoothed log‑sum‑exp objective.
    Returns the best vector (soft‑max of w) and its fast evaluation.
    """
    w = w0.copy()
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t_iter = 0
    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = w.shape[0]
        for _ in range(max_iter_per_eps):
            if time.time() >= deadline:
                break
            L, grad = _smooth_obj_and_grad(w, eps, w.shape[0])
            t_iter += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t_iter)
            v_hat = v / (1 - beta2 ** t_iter)
            # per‑coordinate step size
            step = 0.2 / (np.sqrt(v_hat) + eps_adam)
            w = w - step * m_hat

            a = np.exp(w - np.max(w))
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a step‑function minimising ``evaluate_sequence``."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 4000                      # keep the same size as the reference run
    global _GLOBAL_N
    _GLOBAL_N = n

    # --------------------------------------------------------------
    # helper for normalisation
    # --------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            v = np.full_like(v, 1.0 / v.shape[0])
            s = v.sum()
        return v / s

    # --------------------------------------------------------------
    # 1. seed generation (wide variety)
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n))

    # geometric families
    for q in np.linspace(0.6, 0.99, 12):
        seeds.append(_geometric_sequence(q, n))

    # power‑law families
    for p in np.linspace(-5.0, -0.2, 12):
        seeds.append(_power_sequence(p, n))

    # random Dirichlet‑like draws
    for _ in range(40):
        a = rng.random(n)
        seeds.append(_normalise(a))

    # classic window functions
    i = np.arange(n, dtype=float)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    seeds.append(_normalise(hann))

    cos2 = np.cos(np.pi * i / (n - 1)) ** 2
    seeds.append(_normalise(cos2))

    tri = np.concatenate([np.linspace(0, 1, n // 2 + 1),
                          np.linspace(1, 0, n - n // 2 - 1)])
    seeds.append(_normalise(tri))

    # a few random block patterns
    for _ in range(5):
        block = np.zeros(n)
        nb = rng.integers(5, 15)
        for __ in range(nb):
            pos = rng.integers(0, n)
            length = min(rng.integers(1, n // nb + 1), n - pos)
            value = rng.random()
            block[pos:pos + length] = value
        seeds.append(_normalise(block))

    # --------------------------------------------------------------
    # evaluate seeds, keep the best
    # --------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for a in seeds:
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    print(f"[seed] best evaluation after seeding = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Adam optimisation on the smoothed surrogate (several restarts)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-2), np.log10(1e-12), num=15)
    max_iter_per_eps = 150   # a modest inner loop that fits the budget

    # we only spend a fraction of the time on Adam (≈ 40 % of total)
    adam_deadline = time.time() + 0.4 * (deadline - time.time())

    for seed_idx, a0 in enumerate(seeds[:min(20, len(seeds))]):
        if time.time() >= adam_deadline:
            break
        w0 = np.log(a0 + 1e-16)          # initialise soft‑max parameters
        a_opt, val_opt = _adam_optimize(w0, eps_schedule,
                                        max_iter_per_eps,
                                        adam_deadline)
        if a_opt is not None and val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[adam] seed {seed_idx} new best = {best_val:.6f}")

        # deterministic re‑orderings after each Adam run
        for reorder in (_anti_symmetric_reorder,
                        _alternating_reorder,
                        _descending_reorder):
            a_r = reorder(best_a)
            v_r = _eval_fast(a_r)
            if v_r < best_val:
                best_val = v_r
                best_a = a_r.copy()
                print(f"[reorder‑adam] {reorder.__name__} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. extended simulated annealing (the bulk of the remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sa_deadline = deadline - 30.0
        print("[SA] starting extended simulated annealing")
        best_a, best_val = _simulated_annealing_extended(best_a, best_val,
                                                         sa_deadline)
        print(f"[SA] best after annealing = {best_val:.6f}")

    # 4. final hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)
        print(f"[refine] final hill‑climb best = {best_val:.6f}")

    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    return best_a.tolist()
