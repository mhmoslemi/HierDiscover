def _alternating_ends_reorder(a):
    n = len(a)
    sorted_vals = np.sort(a)[::-1]  # descending
    new_a = np.empty_like(a)
    left = 0
    right = n-1
    idx = 0
    while left <= right:
        new_a[left] = sorted_vals[idx]; idx+=1; left+=1
        if left <= right:
            new_a[right] = sorted_vals[idx]; idx+=1; right-=1
    return new_a

def _subgradient_descent(a, deadline):
    n = len(a)
    cur = a.copy()
    cur_val = _eval_fast(cur)
    best = cur.copy()
    best_val = cur_val
    it = 0
    step = 0.5  # initial step size
    while time.time() < deadline:
        # compute conv and index of max
        conv = _fft_convolve_self(cur)
        k_max = int(np.argmax(conv))
        # compute gradient: for each i, derivative = 2*a_{k_max - i} if k_max - i in range else 0
        idx = k_max - np.arange(n)
        mask = (idx >=0) & (idx < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2 * cur[idx[mask]]
        # step size: step * 1/(sqrt(it+1))
        alpha = step / (np.sqrt(it+1))
        # gradient descent direction: -grad
        new = cur - alpha * grad
        new = np.maximum(new, 0)
        sum_new = new.sum()
        if sum_new == 0:
            new = np.full(n, 1.0/n)
        else:
            new /= sum_new
        new_val = _eval_fast(new)
        if new_val < best_val:
            best_val = new_val
            best = new.copy()
            print(f"[subgrad] iter {it} new best = {best_val:.6f}")
        if new_val < cur_val:
            cur = new
            cur_val = new_val
        it += 1
        # optional step size adaptation: if iter%1000==0 reduce step
        if it % 1000 == 0:
            step *= 0.9
    return best, best_val

def _random_transfer_annealing(a, cur_val, deadline):
    n = len(a)
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    cur_seq = a.copy()
    cur_val = cur_val
    start = time.time()
    # initial temperature relative to cur_val
    T = cur_val * 0.1
    it = 0
    while time.time() < deadline:
        # choose move type
        if rng.random() < 0.3:  # swap
            i, j = rng.integers(0,n,2)
            if i == j: continue
            new_seq = cur_seq.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:
            i = rng.integers(0,n)
            if cur_seq[i] < 1e-12: continue
            j = rng.integers(0,n)
            if i == j: continue
            delta = cur_seq[i] * rng.random() * 0.2  # step factor
            new_seq = cur_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        new_val = _eval_fast(new_seq)
        delta_val = new_val - cur_val
        accept = False
        if delta_val < 0:
            accept = True
        else:
            if T > 0:
                if rng.random() < np.exp(-delta_val / T):
                    accept = True
        if accept:
            cur_seq = new_seq
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[TA] iter {it} new best = {best_val:.6f}")
        # temperature schedule
        it += 1
        if it % 2000 == 0:
            T *= 0.95
            if T < 1e-12:
                T = 1e-12
    return best_seq, best_val

"""
Hybrid optimizer for the step‑function evaluation problem (v2).

The optimizer works in several phases, each respecting the global
time‑budget (default 1000 s).  The goal is to minimise

    2·n·max(a*a) / (sum(a)²)

where *a* is a non‑negative vector with fixed ℓ₁‑norm (here we keep
sum(a)=1, which does not affect the objective).

Phases
------
1. **Exploratory seeding** – a small collection of hand‑crafted shapes
   (uniform, exponential tails) and a burst of random Dirichlet draws.
   The best of these seeds becomes the starting point for the later
   optimisation.

2. **Smooth surrogate optimisation** – we minimise a smooth
   log‑sum‑exp approximation of the max‑convolution with L‑BFGS.  The
   optimisation is performed for a decreasing schedule of the smoothing
   parameter ε and a few random restarts.

3. **Heuristic re‑ordering** – the multiset of heights is reordered
   twice (largest‑at‑ends and alternating‑ends) which often reduces the
   peak of the self‑convolution for a fixed multiset.

4. **Projected sub‑gradient descent** – directly minimises the true
   max‑convolution while staying on the simplex (non‑negative, sum = 1).

5. **Random transfer annealing / hill‑climbing** – a simulated‑annealing
   walk that transfers a small amount of mass between two positions or
   swaps two entries.  The temperature decays slowly; accepted moves are
   kept if they improve the objective, otherwise they may be accepted
   with a Metropolis probability.

At the end the best sequence found is evaluated with the official
`evaluate_sequence` function and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter
_GLOBAL_N = None     # length of the problem


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using FFT.
    Returns array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip rounding artefacts
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast evaluation identical to evaluate_sequence(a.tolist())
    but without the safety checks.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (soft‑max parametrisation, L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where L is the smoothed surrogate

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and grad_w is the gradient w.r.t. the soft‑max parameters w.
    """
    # soft‑max for non‑negativity and ℓ₁‑normalisation
    w_shift = w - np.max(w)               # improve numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                         # Σa = 1

    # convolution a * a
    conv = _fft_convolve_self(a)          # length 2·n‑1
    max_conv = conv.max()

    # smoothed approximation of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log∑exp((c−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)             # exp((c−max)/ε)
    p = weights / np.sum(weights)         # probability vector over k

    # g_i = 4·n·∑_k p_k·a_{k−i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)         # length 2·n‑1
    seg = conv_ap[n - 1: 2 * n - 1]        # central n entries
    grad_a = 4.0 * n * seg[::-1]          # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values at the two ends, then alternating small‑large."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left_ptr = 0
    right_ptr = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right_ptr]
        right_ptr -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left_ptr]
        left_ptr += 1
        i += 1
    return new_a


def _alternating_ends_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic re‑ordering that alternates placing the largest remaining
    value at the leftmost free slot, then the rightmost free slot, and so on.
    """
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]  # descending
    new_a = np.empty_like(a)
    left = 0
    right = n - 1
    idx = 0
    while left <= right:
        new_a[left] = sorted_desc[idx]
        idx += 1
        left += 1
        if left <= right:
            new_a[right] = sorted_desc[idx]
            idx += 1
            right -= 1
    return new_a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Sub‑gradient descent for the true max‑convolution objective.
    Keeps a on the simplex (non‑negative, ℓ₁ = 1).
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    step0 = 0.6                      # initial step magnitude
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur_a)
        k_max = int(np.argmax(conv))

        # gradient of the max entry w.r.t. a : 2·a_{k_max‑i} for valid i
        idx_arr = k_max - np.arange(n)
        mask = (idx_arr >= 0) & (idx_arr < n)
        grad = np.zeros_like(cur_a)
        grad[mask] = 2.0 * cur_a[idx_arr[mask]]

        alpha = step0 / np.sqrt(it + 1.0)
        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            new_a = np.full(n, 1.0 / n)
            s = 1.0
        new_a /= s

        new_val = _eval_fast(new_a)

        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it} new best = {best_val:.6f}")

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val

        it += 1
        if it % 500 == 0:
            step0 *= 0.9   # gentle decay of the step size
    return best_a, best_val


# ----------------------------------------------------------------------
# Random transfer / swap annealing (mass moves)
# ----------------------------------------------------------------------
def _random_transfer_annealing(a: np.ndarray, cur_val: float,
                               deadline: float) -> (np.ndarray, float):
    """Simple simulated‑annealing walk that moves mass between entries."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    cur_seq = a.copy()
    cur_val = cur_val

    # initial temperature – a small fraction of the current value
    T = max(cur_val * 0.08, 1e-6)
    it = 0
    while time.time() < deadline:
        if rng.random() < 0.30:          # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur_seq.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                            # small mass transfer
            i = rng.integers(0, n)
            if cur_seq[i] < 1e-14:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            # transfer a random fraction of a[i] (up to 20 %)
            delta = cur_seq[i] * rng.random() * 0.20
            cand = cur_seq.copy()
            cand[i] -= delta
            cand[j] += delta

        cand_val = _eval_fast(cand)
        delta_val = cand_val - cur_val

        accept = False
        if delta_val < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta_val / T):
                accept = True

        if accept:
            cur_seq = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[anneal] iter {it} new best = {best_val:.6f}")

        # temperature schedule – very slow decay
        it += 1
        if it % 3000 == 0:
            T *= 0.94
            T = max(T, 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper to generate a few hand‑crafted shapes
# ----------------------------------------------------------------------
def _exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential tails: a_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-c * idx) + np.exp(-c * (n - 1 - idx))
    return vals / vals.sum()


def _random_dirichlet(alpha: float, n: int, rng) -> np.ndarray:
    """Draw a random point on the simplex using Gamma(alpha) variables."""
    samples = rng.gamma(shape=alpha, scale=1.0, size=n)
    if samples.sum() == 0:
        return np.full(n, 1.0 / n)
    return samples / samples.sum()
