"""
Hybrid optimisation for the step‑function autocorrelation minimisation.

Key ideas
---------
* Start from the previous best construction (if available) and up‑sample it
  to a larger length (default 8000).  A larger n gives more flexibility to
  spread the mass and usually yields a smaller max‑convolution.
* Enrich the seed pool with many classic window functions (Kaiser, Chebyshev,
  Hann, Hamming, Blackman, Tukey, Gaussian) together with geometric,
  power‑law and random Dirichlet draws.
* Perform coarse then refined L‑BFGS on a smooth log‑sum‑exp surrogate of the
  max‑convolution (the same surrogate used in the reference solution).
* Apply a dedicated sub‑gradient descent on the *exact* max­‑convolution.
  The sub‑gradient of  max_k (a∗a)_k is 2·a reflected around the index of the
  current peak.  After each step we project back onto the probability simplex
  (∑a_i = 1, a_i ≥ 0).  This often yields a quick deterministic improvement.
* Finish with an extended simulated‑annealing phase (swap / mass transfer /
  targeted transfer) and a final hill‑climbing polish.
* All stages respect the 1000 s time budget and the best sequence found is
  returned as a plain Python list.

The implementation is fully self‑contained, uses only NumPy / SciPy and
does not rely on any external files or network access.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
from scipy import signal

# ----------------------------------------------------------------------
# Global parameters for the smoothed surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter
_GLOBAL_N = None     # current sequence length


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to ``x``."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Numerical noise may produce tiny negatives; clamp them.
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` for a non‑negative vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:                     # should never happen for our vectors
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """Return (smooth objective, gradient) for the soft‑max parametrisation."""
    # soft‑max yields a non‑negative vector that sums to 1
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= np.sum(a)

    # convolution a*a
    conv = _fft_convolve_self(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()          # probability distribution over convolution indices
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)       # a * p_rev
    seg = conv_ap[n - 1: 2 * n - 1]      # central part of length n
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.optimize.minimize."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

# Re‑ordering heuristics (helpful after a numerical optimisation step)
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large values left, small values right (pairwise)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = sorted_vals[n - 1 - i]      # large to the left
        out[n - 1 - i] = sorted_vals[i]      # small to the right
    if n % 2:
        out[half] = sorted_vals[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    sorted_vals = np.sort(a)
    left, right = 0, len(sorted_vals) - 1
    out = np.empty_like(a)
    idx = 0
    while left <= right:
        out[idx] = sorted_vals[right]
        idx += 1
        right -= 1
        if left <= right:
            out[idx] = sorted_vals[left]
            idx += 1
            left += 1
    return out


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Simply sort descending."""
    return np.sort(a)[::-1]

# Simple parametric families (used for seeding)
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """a_i ∝ q**i, normalised to sum = 1 (0 < q < 1)."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q ** n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)**p, normalised to sum = 1 (p < 0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Projection onto the probability simplex (∑a = 1, a ≥ 0)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, cur_val: float, deadline: float):
    """Deterministic sub‑gradient optimisation of max(a*a)."""
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    step = 0.1                     # initial step size (on the simplex scale)
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _fft_convolve_self(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # gradient of the active max‑constraint:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        grad = np.zeros_like(a)
        grad[i_idx] = 2.0 * a[j_idx]

        # move opposite to the gradient
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new)

        new_val = _eval_fast(a_new)
        # Accept if improvement, otherwise maybe accept a tiny random move
        if new_val < cur_val or rng.random() < 1e-4:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")

        # simple step‑size adaptation
        if new_val < cur_val:
            step = min(step * 1.05, 1.0)
        else:
            step = step * 0.92
    return best_a, best_val


# ----------------------------------------------------------------------
# Extended simulated annealing (as in the reference implementation)
# ----------------------------------------------------------------------
def _simulated_annealing_extended(a: np.ndarray, cur_val: float,
                                 deadline: float) -> (np.ndarray, float):
    """SA mixing swaps, random transfers and a targeted reduction of the peak."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.05, 1e-6)
    decay = 0.9999
    it = 0

    while time.time() < deadline:
        move = rng.random()
        if move < 0.4:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif move < 0.8:                   # random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * 0.05 * rng.random()
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        else:                               # targeted transfer from the strongest pair
            conv = _fft_convolve_self(a)
            k_max = int(np.argmax(conv))
            low = max(0, k_max - (n - 1))
            high = min(k_max, n - 1)
            i_idx = np.arange(low, high + 1, dtype=int)
            j_idx = k_max - i_idx
            prod = a[i_idx] * a[j_idx]
            max_pair = int(np.argmax(prod))
            i_best = i_idx[max_pair]
            j_best = j_idx[max_pair]
            source = i_best if a[i_best] >= a[j_best] else j_best
            target = rng.integers(0, n)
            if source == target:
                continue
            delta = a[source] * 0.05 * rng.random()
            new_a = a.copy()
            new_a[source] -= delta
            new_a[target] += delta

        new_val = _eval_fast(new_a)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")

        T = max(T * decay, 1e-12)
        it += 1

    return best_seq, best_val

def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:           # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                             # small mass transfer

            cur_val = val

    # ------------------------------------------------------------------
    # Problem size – a larger n usually gives a better bound.
    # ------------------------------------------------------------------
    n = 8000
    global _GLOBAL_N
    _GLOBAL_N = n

    # ------------------------------------------------------------------
    # Initialise from a previous construction if available, otherwise uniform.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()                     # work on the probability simplex
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1.  Diverse seed generation (geometric, power‑law, windows, random)
    # ------------------------------------------------------------------
    seed_candidates = []

    # geometric progression
    for q in np.linspace(0.7, 0.9995, 30):
        seed_candidates.append(_geometric_sequence(q, n))

    # power‑law progression
    for p in np.linspace(-3.0, -0.2, 30):
        seed_candidates.append(_power_sequence(p, n))

    # classic windows (positive by construction)
    # Kaiser
    for beta in np.linspace(0, 14, 15):
        w = signal.kaiser(n, beta)
        w = np.maximum(w, 0.0)
        w /= w.sum()
        seed_candidates.append(w)

    # Chebyshev
    for at in (30, 50, 70, 90, 110):
        w = signal.chebwin(n, at)
        w = np.maximum(w, 0.0)
        w /= w.sum()
        seed_candidates.append(w)

    # Hann / Hamming / Blackman
    for win in (signal.hann, signal.hamming, signal.blackman):
        w = win(n)
        w = np.maximum(w, 0.0)
        w /= w.sum()
        seed_candidates.append(w)

    # Tukey with varying alpha
    for a in np.linspace(0, 1, 6):
        w = signal.tukey(n, a)
        w = np.maximum(w, 0.0)
        w /= w.sum()
        seed_candidates.append(w)

    # Gaussian (std as fraction of n)
    for std_frac in (0.2, 0.3, 0.4, 0.5):
        w = signal.gaussian(n, std_frac * n)
        w = np.maximum(w, 0.0)
        w /= w.sum()
        seed_candidates.append(w)

    # Random Dirichlet‑like draws
    for _ in range(80):
        r = np.random.rand(n)
        r /= r.sum()
        seed_candidates.append(r)

    # Evaluate all seeds
    for a in seed_candidates:
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[seed] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Coarse L‑BFGS (many cheap restarts, relatively large smoothing)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    stage_a_deadline = time.time() + min(0.20 * budget_s, time_left * 0.45)

    eps_coarse = np.logspace(-2, -6, num=10)      # moderate smoothing
    while time.time() < stage_a_deadline:
        w = np.random.randn(n) * 2.0              # random start in log‑softmax space
        for eps in eps_coarse:
            if time.time() >= stage_a_deadline:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': 120,
                                    'ftol': 1e-8,
                                    'gtol': 1e-8,
                                    'disp': False})
            w = res.x
            a = np.exp(w - np.max(w))
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[coarse] eps {eps:.1e} new best = {best_val:.6f}")

        # Deterministic re‑orderings after each restart
        for reorder in (_anti_symmetric_reorder,
                        _alternating_reorder,
                        _descending_reorder):
            a_r = reorder(best_a)
            v_r = _eval_fast(a_r)
            if v_r < best_val:
                best_val = v_r
                best_a = a_r.copy()
                print(f"[reorder] {reorder.__name__} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Refined L‑BFGS (finer smoothing schedule)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    stage_b_deadline = time.time() + min(200.0, max(0.0, time_left - 30.0))

    eps_fine = np.logspace(-5, -12, num=12)
    w = np.log(best_a + 1e-16)          # initialise from the current best

    while time.time() < stage_b_deadline:
        for eps in eps_fine:
            if time.time() >= stage_b_deadline:
                break
            _GLOBAL_EPS = eps
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': 300,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            a = np.exp(w - np.max(w))
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[fine] eps {eps:.1e} new best = {best_val:.6f}")

        # Re‑orderings after a full pass
        for reorder in (_anti_symmetric_reorder, _alternating_reorder):
            a_r = reorder(best_a)
            v_r = _eval_fast(a_r)
            if v_r < best_val:
                best_val = v_r
                best_a = a_r.copy()
                print(f"[reorder‑fine] {reorder.__name__} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Deterministic sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 80.0:
        subgrad_deadline = time.time() + remaining - 40.0    # keep ~40 s for SA + final
        print(f"[subgrad] starting deterministic sub‑gradient descent (until {subgrad_deadline:.2f})")
        best_a, best_val = _subgradient_descent(best_a, best_val, subgrad_deadline)

    # ------------------------------------------------------------------
    # 5. Extended simulated annealing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sa_deadline = time.time() + remaining - 15.0
        print(f"[SA] extended simulated annealing until {sa_deadline:.2f}")
        best_a, best_val = _simulated_annealing_extended(best_a, best_val, sa_deadline)

    # ------------------------------------------------------------------
    # 6. Final hill‑climbing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # 7. Exact evaluation with the provided function
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
