def _smooth_descent(a0, deadline, rng, max_iters=5000):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    base_lr = 0.5
    tau = 0.5
    max_tau = 8.0
    iter_count = 0
    while time.time() < deadline and iter_count < max_iters:
        iter_count += 1
        # compute smoothed gradient
        grad_f, conv_max = _smooth_gradient(a, tau)
        # step size adapt
        lr = base_lr / np.sqrt(iter_count)

        a_trial = a - lr * grad_f
        a_trial = np.clip(a_trial, 0.0, None)
        s = a_trial.sum()
        if s == 0.0:
            continue
        a_trial /= s

        # evaluate exact objective
        val = _objective(a_trial)
        if val < best_val:
            best_val = val
            best_a = a_trial.copy()
            a = a_trial
            # occasionally increase step
            base_lr = min(base_lr * 1.05, 5.0)
        else:
            # shrink step size
            base_lr = max(base_lr * 0.97, 1e-4)

        # increase tau gradually
        tau = min(tau * 1.02, max_tau)

        # optionally log progress
        if iter_count % 200 == 0:
            print(f"[smooth] iter {iter_count:4d}, obj = {best_val:.6f}, tau={tau:.3f}")

    return best_a, best_val

``evaluate_sequence``.
The algorithm combines a smooth‑approximation gradient descent (to flatten
the self‑convolution) with an aggressive mass‑transfer hill‑climber.
Multiple random restarts are performed within the time budget.

Key ideas
---------
* The true objective is  2·n·max_k (a*a)[k] / (sum a)^2 .
  For a normalised vector (sum a = 1) this is just 2·n·max_k (a*a)[k].
* Direct sub‑gradient descent on the max is prone to stall.  We replace
  the max by a smooth log‑sum‑exp, whose gradient can be computed efficiently
  via two FFT‑based convolutions.
* The smoothed gradient is used with a decreasing step size while the
  smoothing parameter τ is slowly increased, annealing the problem towards
  the true max.
* After the descent phase a specialised hill‑climber moves a fraction of the
  most damaging mass to low‑mass positions; this often yields a final
  polishing improvement.
* Several independent restarts are run (each ~20 s) and the best overall
  sequence is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # eliminate tiny numerical negatives
    np.maximum(conv, 0, out=conv)
    return conv


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x*y using FFT (both real)."""
    n, m = x.shape[0], y.shape[0]
    size = _next_pow2(n + m - 1)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: n + m - 1]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Objective and smooth gradient
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (any sum)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _smooth_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the smoothed objective

        fτ(a) = 2·n/τ * log( Σ_k exp( τ·conv_k(a) ) ),

    where conv = a*a (full convolution).  Returns the gradient vector and
    the current maximal convolution value (needed for the true objective).
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _full_convolution(a)                     # length 2n‑1
    conv_max = conv.max()

    # 2) soft‑max weights (numerically stable)
    w = np.exp(tau * (conv - conv_max))
    sum_w = w.sum()
    if sum_w == 0.0:
        # should never happen, but protect against div‑by‑zero
        w = np.ones_like(w)
        sum_w = w.sum()

    # 3) weighted sum of the per‑lag sub‑gradients.
    #    grad_i = Σ_k w_k * 2·a_{k‑i}
    a_rev = a[::-1]
    conv_rev_w = _full_convolution_fft(a_rev, w)    # length 3n‑2
    # slice that corresponds to i = 0 … n‑1
    part = conv_rev_w[n - 1 : n - 1 + n]            # length n
    grad = 2.0 * part                               # Σ w_k ∂b_k/∂a_i

    # 4) chain rule for the smoothed objective
    grad_f = (2.0 * n) * grad / sum_w
    return grad_f, conv_max


# ----------------------------------------------------------------------
# Smoothed gradient descent
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iters: int = 8000):
    """
    Projected gradient descent on the smoothed objective.
    The iterate is kept on the probability simplex (non‑negative, sum = 1).
    Returns the best vector encountered and its true objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _objective(a)

    base_lr = 0.6            # initial learning‑rate multiplier
    tau = 0.5                # start with a fairly smooth approximation
    max_tau = 8.0            # never let τ grow too large (avoid overflow)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # --- compute smooth gradient ---
        grad_f, conv_max = _smooth_gradient(a, tau)

        # learning‑rate schedule (inverse‑sqrt decay)
        lr = base_lr / np.sqrt(it)

        # trial step
        a_trial = a - lr * grad_f
        a_trial = np.clip(a_trial, 0.0, None)
        s = a_trial.sum()
        if s == 0.0:
            # numerical disaster – restart from a fresh random point
            a_trial = rng.random(n)
            a_trial /= a_trial.sum()
        else:
            a_trial /= s

        # exact objective of the trial point
        trial_val = _objective(a_trial)

        # accept if we improve the true objective
        if trial_val < best_val:
            best_val = trial_val
            best_a = a_trial.copy()
            a = a_trial
            # encourage slightly larger steps after a success
            base_lr = min(base_lr * 1.03, 5.0)
        else:
            # otherwise back‑off
            base_lr = max(base_lr * 0.97, 1e-4)

        # slowly anneal τ towards the true max
        tau = min(tau * 1.02, max_tau)

        if it % 200 == 0:
            print(f"[smooth] iter {it:5d}  best = {best_val:.6f}  τ={tau:.3f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill‑climbing (re‑used from the previous code)
# ----------------------------------------------------------------------


def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing phase that repeatedly moves a fraction of the most
    damaging mass to low‑mass positions.
    Stops after the time budget expires or after ``max_no_improve``
    consecutive unsuccessful attempts.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2                # initial fraction of mass to relocate
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices contributing to the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we are willing to shift
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # pick two distinct low‑mass destinations, avoiding the donors
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    The routine performs several random restarts, each consisting of a
    smoothed gradient descent followed by the aggressive hill‑climber.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin for the evaluator

            a0 = rng.random(n)
        a0 /= a0.sum()
    except Exception:
        n = default_n
        a0 = rng.random(n)
        a0 /= a0.sum()

    print(f"[init] n={n}, initial objective = {_objective(a0):.6f}")

    # --------------------------------------------------------------
    # Global best tracking
    # --------------------------------------------------------------
    best_a = a0.copy()
    best_val = _objective(best_a)

    # --------------------------------------------------------------
    # Main optimisation loop – multiple short runs inside the budget.
    # --------------------------------------------------------------
    while time.time() < deadline:
        # allocate a modest time slice for a single run (≈20 s)
        run_deadline = min(time.time() + 20.0, deadline)

        # --- random start for this run ---------------------------------
        a_start = rng.random(n)
        a_start /= a_start.sum()

        # --- smooth gradient descent ------------------------------------
        a_desc, val_desc = _smooth_descent(a_start, run_deadline, rng)
        if val_desc < best_val:
            best_val = val_desc
            best_a = a_desc.copy()
            print(f"[run] new best after smooth descent = {best_val:.6f}")

        # --- hill‑climb polishing --------------------------------------
        a_hc, val_hc = _mass_transfer_hillclimb(a_desc, deadline, rng, max_no_improve=2000)
        if val_hc < best_val:
            best_val = val_hc
            best_a = a_hc.copy()
            print(f"[run] new best after hill‑climb = {best_val:.6f}")

        # Occasionally try to improve the current global best directly.
        if time.time() < deadline - 5.0:
            a_hc2, val_hc2 = _mass_transfer_hillclimb(best_a, deadline, rng, max_no_improve=2000)
            if val_hc2 < best_val:
                best_val = val_hc2
                best_a = a_hc2.copy()
                print(f"[global] new best after extra hill‑climb = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided evaluator.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
