def _full_convolution(x):
    n = x.shape[0]
    m = 2*n -1
    size = 1 << ((m-1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:m]
    return conv

def _exp_grad_optimize(a0, n, deadline, start_time, best_a, best_val, target_val=1.503):

    a = a0.copy()
    cur_val = _objective(a)
    if cur_val < best_val: best_val, best_a = cur_val, a.copy()
    # schedule
    eta_start = 5.0
    eta_end = 1e-5

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # compute convolution and max
        b = _full_convolution(a)
        max_b = b.max()
        K = int(np.argmax(b))

        # compute gradient g
        low = max(0, K - (n - 1))
        high = min(n - 1, K)
        if low <= high:
            idx = np.arange(low, high + 1)
            g = np.zeros_like(a)
            g[idx] = 2.0 * a[K - idx]
        else:
            g = np.zeros_like(a)  # shouldn't happen

        # schedule eta
        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1 - progress) + eta_end * progress

        # multiplicative update with epsilon to avoid log of zero
        # Use mask for non-zero entries
        mask = a > 0
        # compute new values
        # using np.exp directly may underflow; just compute a * np.exp(-eta * g)
        # For zero entries, stay zero.
        a_new = a.copy()
        a_new[mask] = a[mask] * np.exp(-eta * g[mask])
        sum_a = a_new.sum()
        if sum_a == 0:
            a_new = np.full(n, 1.0 / n)
        else:
            a_new /= sum_a

        new_val = _objective(a_new)

        # update best
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[ExpGrad {iteration}] new best = {best_val:.6f}")

        # accept if new_val <= cur_val or with small probability
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # small probability to escape (temperature decays)
            T_start, T_end = 0.005, 1e-6
            T = T_start * (T_end / T_start) ** progress
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a = a_new
            cur_val = new_val
        else:
            # maybe random perturb occasionally
            if np.random.rand() < 0.001:
                # small random perturbation
                noise = np.random.normal(scale=0.01, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                cur_val = _objective(a)

        if best_val <= target_val:
            break

        iteration += 1

    return best_a, best_val

"""
Propose a non‑negative step‑function that minimises the provided
evaluate_sequence objective.

Algorithm
---------
1. Initialise a candidate.
   * If a previous solution `height_sequence_1` exists, interpolate it to the
     working length `n`.
   * Sweep a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} over a
     fine grid (including very small exponents) and keep the best.
   * Add a random start as a safety net.

2. Adam optimisation (≈20 % of the time budget).
   * Optimises a smooth approximation of  max(conv(a,a)) on the simplex
     via Adam.  The smoothness temperature is annealed to zero.

3. Exponentiated‑gradient descent on the exact max‑convolution
   (≈15 % of the budget).
   * Uses the true sub‑gradient of the ∞‑norm: for the current
     max‑index K,   g_i = 2·a_{K‑i} (if the mirrored index exists).
   * Performs a multiplicative update a_i ← a_i·exp(‑η·g_i) and renormalises.
   * The step size η is exponentially decayed.

4. Simulated‑annealing hill‑climbing (remaining time).
   * Repeatedly moves a fraction of mass from an index that contributes
     heavily to the current maximal convolution entry to an index that
     contributes little.
   * Accepts all improving moves and uphill moves with a Metropolis rule.
   * Random perturbations are performed after long periods without
     improvement.

5. Return the best vector found (converted to a plain list).  The final
   vector is evaluated with `evaluate_sequence` for logging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two for speed
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Parametric sweep of a Beta‑shaped family
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a refined two‑parameter Beta family for a good seed."""
    best_a = None
    best_val = np.inf

    # a rich grid covering very small and fairly large exponents
    p_vals = np.unique(
        np.concatenate((
            np.geomspace(0.01, 0.5, 8),          # heavy weight at the ends
            np.linspace(0.5, 5.0, 12),           # medium shapes
            np.geomspace(5.0, 30.0, 6)           # sharply peaked
        ))
    )
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max approximation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation of a smooth max‑approximation."""
    # initialise logits for a soft‑max parametrisation (guarantees Σa=1)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.001
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 25000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedule for temperature (eps) and learning rate (lr)
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max and its gradient
        n_full = 2 * n - 1
        # compute convolution
        b = _full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()                     # (2n‑1,) soft‑max weights

        # gradient of Σ p_k * b_k w.r.t. a:
        # conv(a, reversed(p)) restricted to n entries
        p_rev = p[::-1]
        conv_ab = np.fft.irfft(np.fft.rfft(a) * np.fft.rfft(p_rev, 1 << ((2 * n - 1) - 1).bit_length()))
        conv_ab = conv_ab[: 2 * n - 1]                 # length 2n‑1
        segment = conv_ab[n - 1:2 * n - 1]             # n entries
        grad_a = 4.0 * n * segment[::-1]               # factor 4n from the outer 2·n

        # chain rule through the soft‑max (w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update of logits
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation to avoid lock‑in
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        # restart after prolonged stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restarting after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_grad_optimize(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       best_a: np.ndarray,
                       best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the exact sub‑gradient of
    max(conv(a,a)).  Uses multiplicative updates a_i ← a_i·exp(‑η·g_i)
    where g_i = 2·a_{K‑i} for the current maximal index K.
    """
    a = a0.copy()
    cur_val = _objective(a)
    if cur_val < best_val:
        best_val, best_a = cur_val, a.copy()

    # step‑size schedule (large → small)
    eta_start, eta_end = 5.0, 1e-5
    # temperature for occasional uphill acceptance
    T_start, T_end = 0.005, 1e-6

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # current convolution and its maximiser
        b = _full_convolution(a)
        max_b = b.max()
        K = int(np.argmax(b))

        # sub‑gradient of the max‑entry (size n, zeros where mirror absent)
        g = np.zeros_like(a)
        low = max(0, K - (n - 1))
        high = min(n - 1, K)
        if low <= high:
            idx = np.arange(low, high + 1)
            g[idx] = 2.0 * a[K - idx]

        # schedule step size and temperature
        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1 - progress) + eta_end * progress
        T = T_start * (T_end / T_start) ** progress

        # multiplicative (exponential) update, protecting zeros
        mask = a > 0
        a_new = a.copy()
        a_new[mask] = a[mask] * np.exp(-eta * g[mask])

        # renormalise to stay on the simplex
        sum_a = a_new.sum()
        if sum_a == 0.0:
            a_new = np.full(n, 1.0 / n)
        else:
            a_new /= sum_a

        new_val = _objective(a_new)

        if new_val < best_val:
            best_val, best_a = new_val, a_new.copy()
            print(f"[ExpGrad {iteration:7d}] new best = {best_val:.6f}")

        # simple Metropolis acceptance (rarely needed)
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
            accept = True

        if accept:
            a, cur_val = a_new, new_val

        if best_val <= target_val:
            break

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (identical to the previous version)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local mass‑transfer search.  Moves a fraction of mass from an index that
    heavily contributes to the current maximal convolution entry to an index
    that contributes little.  Accepts improving moves and uphill moves with
    a decaying Metropolis temperature.
    """
    a = a0.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule for maximal transfer fraction and temperature
    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.012, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (probability ∝ contribution)
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index (biased towards low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback uniform among all except i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

    """

    start_time = time.time()

    deadline = start_time + budget_s - 0.5

    # Problem size – 5 000 entries give a good speed / quality trade‑off.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previous construction if it exists.
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)

        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:

    # ------------------------------------------------------------------

    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # fallback random start if nothing else succeeded
    if best_a is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_a, best_val = a, _objective(a)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.35 * budget_s
    best_a, best_val = _exp_grad_optimize(best_a, n,
                                          deadline=phase_b_deadline,
                                          start_time=start_time,
                                          best_a=best_a,
                                          best_val=best_val,
                                          target_val=1.503)

    # ------------------------------------------------------------------
    # Phase C – Simulated‑annealing hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=phase_b_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
