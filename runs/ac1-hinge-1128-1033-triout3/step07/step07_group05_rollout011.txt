"""
Hybrid optimiser for the step‑function bound.

Goal
----
Find a non‑negative sequence a (ℓ₁‑norm = 1) that minimises
    2·n·max(a*a)   (the same quantity returned by evaluate_sequence).

Method
------
1. Initialise a either from the previously saved construction
   (`height_sequence_1`) up‑sampled to the chosen length, or from a uniform start.
2. Repeatedly minimise a smooth approximation of the max‑convolution
   using L‑BFGS‑B.  The smooth surrogate is a log‑sum‑exp of the convolution
   with a decreasing smoothing parameter ε (continuation).  Several random
   restarts are performed.
3. After every L‑BFGS run the sequence is reordered by the “ends‑large”
   heuristic (largest, smallest, second‑largest, …) which is cheap and
   often reduces the max‑convolution.
4. A projected sub‑gradient descent on the exact max‑convolution refines the
   result further while keeping a on the simplex.
5. A long simulated‑annealing phase explores the permutation space.
   Three move types are used:
     * random swap,
     * guided swap – swaps a position that participates heavily in the current
       maximal convolution index with a low‑impact position,
     * small mass transfer between two positions.
   Acceptance follows a standard Metropolis rule with an exponentially
   decaying temperature.
6. Finally a pure hill‑climbing phase (only improving moves) cleans up any
   remaining irregularities.
7. The best sequence encountered is normalised (∑a = 1) and evaluated with the
   official `evaluate_sequence` before being returned.

All phases respect the total wall‑time budget (`budget_s` seconds) and print
progress information.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate wrapper (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Clip tiny negative values caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast_full(a: np.ndarray) -> (float, np.ndarray):
    """Exact objective value together with the full convolution.
    The sequence a is assumed to have ℓ₁‑norm = 1 (no scaling needed).
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    val = float(2.0 * n * max_b)          # since sum(a)=1
    return val, conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """Smoothed surrogate of the max‑convolution.

    Returns (L, grad_w) where

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and grad_w is the gradient w.r.t. the soft‑max parameters w.
    """
    # ---- soft‑max (positivity, ℓ₁ = 1) ----
    w_shift = w - np.max(w)                     # for numerical stability
    a = np.exp(w_shift)
    a /= np.sum(a)                               # now Σa = 1

    # ---- convolution a*a ----
    conv = _fft_convolve_self(a)                 # length 2·n‑1
    max_conv = conv.max()

    # ---- smooth approximation of the max ----
    shifted = (conv - max_conv) / eps
    # log‑sum‑exp of negative numbers is safe
    from scipy.special import logsumexp
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (see derivation in the description) ----
    weights = np.exp(shifted)                    # exp((c−max)/ε)
    p = weights / np.sum(weights)                # probability distribution over k

    # Convolution of a with reversed p (equivalent to the derivative)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                # length 2·n‑1
    seg = conv_ap[n - 1: 2 * n - 1]              # central n entries
    grad_a = 4.0 * n * seg[::-1]                 # ∂L/∂a

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# Bind the helper convolution to the function object so the wrapper can see it
_smooth_obj_and_grad._conv_fft = _conv_fft


def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values at the two ends, then the smallest near the centre.
    Pattern: large, small, second‑large, second‑small, ...
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left_ptr = 0
    right_ptr = n - 1
    i = 0
    while i < n:
        # large
        new_a[i] = sorted_vals[right_ptr]
        right_ptr -= 1
        i += 1
        if i >= n:
            break
        # small
        new_a[i] = sorted_vals[left_ptr]
        left_ptr += 1
        i += 1
    return new_a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a non‑negative vector onto the ℓ₁‑simplex (sum=1)."""
    # standard algorithm for projection onto the simplex
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _subgradient_descent(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """Projected sub‑gradient descent on the exact max‑convolution."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val, cur_conv = _eval_fast_full(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    step0 = 0.12
    it = 0
    while time.time() < deadline:
        # identify index of current maximum
        k_max = int(np.argmax(cur_conv))

        # gradient of the max entry w.r.t. a:
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(cur_a)
        grad[mask] = 2.0 * cur_a[idx[mask]]

        # diminishing step size
        alpha = step0 / np.sqrt(it + 1.0)

        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)
        new_a = _project_simplex(new_a)

        new_val, new_conv = _eval_fast_full(new_a)

        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it:5d} new best = {best_val:.6f}")

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            cur_conv = new_conv

        it += 1
    return best_a, best_val


def _simulated_annealing(a: np.ndarray, cur_val: float, cur_conv: np.ndarray,
                         deadline: float) -> (np.ndarray, float, np.ndarray):
    """Extensive SA using random swaps, guided swaps and small mass transfers."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    best_conv = cur_conv.copy()

    # temperature schedule
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99994
    it = 0

    while time.time() < deadline:
        move_type = rng.random()
        new_a = a.copy()
        if move_type < 0.45:                         # pure random swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                new_a[i], new_a[j] = new_a[j], new_a[i]
        elif move_type < 0.80:                       # guided swap targeting max index
            k_max = int(np.argmax(cur_conv))
            # compute contribution a[i] * a[k_max-i] where valid
            partner = k_max - np.arange(n)
            valid = (partner >= 0) & (partner < n)
            prod = np.zeros(n)
            prod[valid] = a[valid] * a[partner[valid]]
            # pick a heavy index (top‑5 %)
            heavy_candidates = np.where(prod > np.percentile(prod, 95))[0]
            light_candidates = np.where(prod < np.percentile(prod, 20))[0]
            if heavy_candidates.size and light_candidates.size:
                i = rng.choice(heavy_candidates)
                j = rng.choice(light_candidates)
                new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                         # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and new_a[i] > 1e-12:
                delta = new_a[i] * 0.03 * rng.random()   # move up to 3 % of source
                new_a[i] -= delta
                new_a[j] += delta

        # evaluate candidate (exact objective)
        new_val, new_conv = _eval_fast_full(new_a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = new_a
            cur_val = new_val
            cur_conv = new_conv
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                best_conv = cur_conv.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        # cool down
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)
        it += 1

    return best_seq, best_val, best_conv


def _hill_climb(a: np.ndarray, cur_val: float, cur_conv: np.ndarray,
                deadline: float) -> (np.ndarray, float):
    """Simple hill‑climbing (accept only improving swaps or mass moves)."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = cur_val
    cur_a = a.copy()
    cur_conv = cur_conv.copy()
    cur_val = cur_val

    while time.time() < deadline:
        if rng.random() < 0.3:                     # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] < 1e-12:
                continue
            delta = cur_a[i] * 0.01 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_val, new_conv = _eval_fast_full(new_a)
        if new_val + 1e-14 < cur_val:
            cur_a = new_a
            cur_val = new_val
            cur_conv = new_conv
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[refine] new best = {best_val:.6f}")

    return best_a, best_val

        if old.shape[0] != n:
            # linear interpolation to requested size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= np.sum(a0)               # enforce ℓ₁ = 1

    best_a = a0.copy()
    best_val, best_conv = _eval_fast_full(best_a)
    print(f"[init] initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Smooth surrogate L‑BFGS (multiple restarts, decreasing ε)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-2, -12, num=22)   # from 1e‑2 down to 1e‑12
    restart = 0
    while time.time() < deadline - 250.0:         # reserve ~4 min for later phases
        restart += 1
        print(f"[restart {restart}]")
        # initialise soft‑max parameters from current best + a little noise
        w = np.log(best_a + 1e-12) + 0.3 * np.random.randn(n)

        for eps in eps_schedule:
            if time.time() >= deadline - 230.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            # L‑BFGS‑B on the smoothed surrogate
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 1200,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate the current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= np.sum(a)
            cur_val, _ = _eval_fast_full(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # after each restart apply the deterministic ends‑large re‑ordering
        a_ord = _ends_large_reorder(best_a)
        a_ord /= np.sum(a_ord)
        val_ord, _ = _eval_fast_full(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("[subgrad] starting projected sub‑gradient descent")
        sub_deadline = deadline - 100.0
        best_a, best_val = _subgradient_descent(best_a, sub_deadline)
        best_a /= np.sum(best_a)               # re‑normalise (protect against drift)

    # ------------------------------------------------------------------
    # 3. Long simulated‑annealing on permutations + mass moves
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print("[SA] starting extensive simulated annealing")
        sa_deadline = deadline - 50.0
        best_a, best_val, best_conv = _simulated_annealing(best_a, best_val,
                                                            best_conv,
                                                            sa_deadline)
        best_a /= np.sum(best_a)

    # ------------------------------------------------------------------
    # 4. Final pure hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[refine] final hill‑climbing")
        hill_deadline = deadline - 5.0
        best_a, best_val = _hill_climb(best_a, best_val, best_conv,
                                       hill_deadline)
        best_a /= np.sum(best_a)

    # ------------------------------------------------------------------
    # Report final score using the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required by the framework
    return best_a.tolist()
