"""
Optimiser for ``evaluate_sequence`` using a large‑scale simulated annealing
followed by a short projected sub‑gradient polishing stage.

Key ideas
---------
* The objective is  2·n·max(conv)  (the sum of the sequence is kept equal to 1).
* A long simulated‑annealing (SA) loop perturbs the vector by moving a small
  fraction of mass from one index to another.  Acceptance follows the
  Metropolis criterion with a temperature that decays exponentially.
* The initial vector is deliberately *anti‑correlated*: a random list is
  sorted, then the largest, smallest, second‑largest, second‑smallest … are
  interleaved.  This arrangement dramatically reduces the inner product of
  the vector with its reverse, giving a much lower starting bound than the
  uniform sequence.
* After the SA phase a few iterations of projected sub‑gradient descent
  (the exact non‑smooth objective) are performed to polish the best
  candidate found.
* All operations respect the supplied time budget (default 1000 s).  The
  best sequence discovered is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of a real vector with itself.
    Returns an array of length ``2*len(x)-1``.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Remove tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # All entries non‑positive – return the uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[cond][-1] - 1           # zero‑based index of the last positive entry
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant – sum of the vector is 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)).  The sum of ``a`` should be 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact (non‑smooth) objective.
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         best_val: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of sub‑gradient descent with simplex projection.
    Returns the best vector encountered and its objective value.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    best_a = cur_a.copy()
    eta = 0.1                         # initial step size

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag (see analysis in the docstring)
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # gradient step + projection onto the simplex
        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = _objective(new_a)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            eta = min(eta * 1.05, 0.5)
        else:
            eta = max(eta * 0.7, 1e-5)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing – mass‑transfer moves
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         budget_until: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Large‑scale SA that repeatedly moves a fraction of mass from a random
    source index to a random target index.
    """
    n = a0.shape[0]
    cur = a0.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T = 0.12                 # initial temperature
    move_frac = 0.18         # initial fraction of source mass to move

    # exponential cooling parameters (tuned empirically)
    temp_decay = 0.9996
    frac_decay = 0.9997

    while time.time() < budget_until:
        # pick source and destination indices
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # amount of mass to transfer
        delta = cur[i] * move_frac
        if delta <= 0.0:
            continue

        prop = cur.copy()
        prop[i] -= delta
        prop[j] += delta      # sum stays exactly 1, no renormalisation needed

        # evaluate
        prop_val = _objective(prop)

        # Metropolis acceptance
        accept = False
        if prop_val < cur_val:
            accept = True
        else:
            # probability exp(-(Δ)/T) – protect against overflow
            prob = math.exp(-(prop_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = prop
            cur_val = prop_val
            if prop_val < best_val:
                best_val = prop_val
                best = cur.copy()
                print(f"[SA] new best = {best_val:.6f}")

        # cooling schedule
        T = max(T * temp_decay, 1e-8)
        move_frac = max(move_frac * frac_decay, 1e-5)

    return best, best_val


# ----------------------------------------------------------------------

    The routine respects the given time budget (default 1000 s) and returns the
    best list of floats found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical bound.
    # --------------------------------------------------------------
    n = 6000   # fits comfortably in memory and still fast enough for many iterations

    # --------------------------------------------------------------
    # Initial construction
    # --------------------------------------------------------------
    try:
        # Re‑use a previously discovered construction if available.
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            # linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        else:
            a0 = a0.copy()
    except Exception:
        # ---- Build an “anti‑correlated’’ start ----
        # 1. draw random numbers
        raw = rng.random(n)
        # 2. sort descending
        s = np.sort(raw)[::-1]
        # 3. interleave largest, smallest, 2nd largest, 2nd smallest, …
        high = s[: (n + 1) // 2]                # larger half (in descending order)
        low = s[(n + 1) // 2 :] [::-1]          # smaller half (ascending order)
        a0 = np.empty_like(s)
        a0[0::2] = high[: (n + 1) // 2]        # place the big values on even positions
        a0[1::2] = low[: n // 2]               # place the small values on odd positions
        # 4. normalise to unit sum
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

    # Initial objective
    best_a = a0.copy()
    best_val = _objective(best_a)
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # Simulated annealing phase (most of the time budget)
    # --------------------------------------------------------------
    # Reserve ~30 s for the polishing stage at the end.
    sa_deadline = deadline - 30.0
    if time.time() < sa_deadline:

    # --------------------------------------------------------------
    # Short sub‑gradient polishing (exact non‑smooth objective)
    # --------------------------------------------------------------

                                                best_val,
                                                deadline,
                                                rng)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:           # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
