"""
Propose a non‑negative step function with a minimal value of
``evaluate_sequence``.

The algorithm works in three stages (adjusted from the previous
implementation):

1. **Initialisation** – generate a large random vector (default length
   20 000).  The values are sorted and then rearranged with the
   *anti‑symmetric* heuristic so that large coefficients are paired with
   small ones.  The vector is normalised to sum‑to‑one.

2. **Exponential mirror‑descent** – iteratively reduce the largest
   entries of the self‑convolution.  Instead of the previous three‑peak
   rule we use up to ten peaks and an adaptive step‑size.
   The update is multiplicative (mirror‑descent with KL divergence) and
   each iteration ends with a projection onto the probability simplex.

3. **Permutation hill‑climbing** – once a good shape is found we keep the
   multiset of coefficients and search for a better ordering.
   Random swaps are accepted only if they lower the objective; a tiny
   simulated‑annealing component is added to escape shallow local minima.
   The convex‑relaxation is evaluated with an FFT‑based convolution which
   is fast enough for many thousands of swaps.

All heavy work uses ``numpy`` and FFTs, so the whole search comfortably
fits into the 1000 s budget.  The final candidate is returned as a plain
list of floats (non‑negative, sum ≈ 1).  The official ``evaluate_sequence``
is called only once at the end for the official score.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = rfft(x, size)
    conv = irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (sum(a) = 1)."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Heuristic ordering – pair large with small values
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values on the left, the smallest on the right."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[n - 1 - i]          # large → left side
        out[n - 1 - i] = s[i]          # small → right side
    if n % 2:
        out[half] = s[half]
    return out


# ----------------------------------------------------------------------
# Exponential mirror‑descent on the original (non‑smooth) objective
# ----------------------------------------------------------------------
def _exp_mirror_descent(a: np.ndarray,
                        deadline: float,
                        top_k: int = 10,
                        alpha0: float = 0.5,
                        decay: float = 0.985,
                        eval_every: int = 100) -> tuple[np.ndarray, float]:
    """
    Reduce the largest convolution peaks via a multiplicative update.
    Returns the best vector found before ``deadline``.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(best_a)

    it = 0
    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        # pick the k largest peaks (indices)
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]   # descending order
        else:
            idxs = np.arange(conv.size)

        # gradient for the selected peaks
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += a[j_range]               # linear contribution

        # scheduled step‑size
        alpha = alpha0 * (decay ** (it // 2000))

        # multiplicative mirror‑descent
        a = a * np.exp(-alpha * g)

        # projection onto the simplex
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0:
            a /= s
        else:                                   # numerical safeguard
            a = np.full_like(a, 1.0 / n)

        # occasional bookkeeping
        if it % eval_every == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[mirror descent iter {it}] new best = {best_val:.6f}")
            # re‑apply anti‑symmetric ordering every few thousand steps
            if it % 2000 == 0:
                a = _anti_symmetric_reorder(a)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation hill‑climbing (random swaps + tiny annealing)
# ----------------------------------------------------------------------
def _perm_hill_climb(a: np.ndarray,
                     deadline: float,
                     max_no_improve: int = 20000) -> tuple[np.ndarray, float]:
    """
    Keep the current multiset of coefficients and search for a better
    ordering by random swaps.  Accepts strictly improving swaps and,
    with a very low probability, non‑improving swaps (simulated annealing).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(best_a)
    rng = np.random.default_rng()

    no_improve = 0
    while time.time() < deadline - 5.0 and no_improve < max_no_improve:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # perform swap
        a[i], a[j] = a[j], a[i]
        cur_val = _objective(a)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[swap] improved to {best_val:.6f}")
        else:
            # tiny annealing: accept with probability exp(-Δ/T)
            # Temperature decays linearly with remaining time.
            remaining = deadline - time.time()
            T = remaining / (deadline - start_time) * 1e-4 + 1e-6
            if rng.random() < np.exp(-(cur_val - best_val) / T):
                # keep the worse state (rare)
                no_improve = 0
            else:
                # revert the swap
                a[i], a[j] = a[j], a[i]
                no_improve += 1

        # occasionally re‑apply the anti‑symmetric heuristic to stay near a good region
        if rng.random() < 0.001:
            a = _anti_symmetric_reorder(a)

    return best_a, best_val
