"""
Propose a step‑function (list of non‑negative heights) with a low value of
``evaluate_sequence``.  

The algorithm works in three phases:

1. **Multi‑scale Adam optimisation**  
   * The vector is represented in soft‑max (logit) form `w` – this guarantees
     non‑negativity and the sum‑to‑1 normalisation required for the objective
     (the original objective is homogeneous, the normalisation does not change
     its value).  
   * A smooth surrogate of the objective  
     `L_ε(a) = 2·n·max(a*a) + 2·n·ε·log Σ_k exp((a*a)_k‑max)/ε`  
     is minimised.  For a fixed smoothing parameter `ε` the gradient can be
     computed analytically via two FFT‑based convolutions.  
   * An Adam optimiser (first‑order adaptive method) is run for many
     iterations, while `ε` decays geometrically from `5·10⁻³` to `10⁻⁹`.  
   * The optimisation is performed on a sequence of increasing lengths
     `[2000, 4000, 8000]`.  After each stage the current best vector is
     interpolated to the next size, allowing the optimiser to exploit the
     additional degrees of freedom.

2. **Optional anti‑symmetric re‑ordering** – after each evaluation the vector
   is optionally reordered so that large values appear on the left side and
   small values on the right side (`_anti_symmetric_reorder`).  This heuristic
   often reduces the maximum convolution peak.

3. **Local hill‑climbing refinement** – a fast simulated‑annealing style
   search (random swaps and small mass transfers) is applied to the best
   vector found in phase 1.  Only strict improvements are kept.

All heavy work uses FFT‑based convolutions (`O(n log n)`) and the whole
procedure respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures, no lambdas)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2·len(x)‑1.  Uses zero‑padding to the next power of two.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negative values – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic re‑ordering: place the largest values on the left half,
    the smallest values on the symmetric right half.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = s[n - 1 - i]        # large → left side
        new_a[n - 1 - i] = s[i]        # small → right side
    if n % 2:
        new_a[half] = s[half]
    return new_a


def _grad_w_from_a(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Compute the gradient of the smooth surrogate `L_ε` w.r.t. the log‑soft‑max
    parameters `w` (such that a = softmax(w)).
    The formula is taken from the reference implementation – it is a
    closed‑form expression involving two FFT convolutions and a chain rule.
    """
    n = a.shape[0]

    # 1) self‑convolution and its maximum
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # 2) soft‑max over the convolution values (log‑sum‑exp trick)
    shifted = (conv - max_conv) / eps      # ≤ 0, safe for exp
    weights = np.exp(shifted)
    p = weights / weights.sum()            # size 2·n‑1, non‑negative, sum = 1

    # 3) gradient w.r.t. `a`  (see derivation in the reference code)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the segment that corresponds to each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]         # derivative of L_ε w.r.t. a

    # 4) chain rule to the log‑soft‑max parameters w
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return grad_w


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a vector to a new length.
    Used when moving to a larger problem size.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate (phase 1)
# ----------------------------------------------------------------------
def _adam_optimize(a_start: np.ndarray,
                   deadline: float,
                   global_start: float) -> tuple[np.ndarray, float]:
    """
    Run Adam on the smooth surrogate `L_ε` for the current problem size.
    Returns the best vector (on the probability simplex) found before the
    deadline together with its raw evaluation value.
    """
    n = a_start.shape[0]
    # log‑soft‑max parameters
    w = np.log(a_start + 1e-12)

    # Adam bookkeeping
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.5               # base learning rate
    t = 0

    # smoothing schedule (geometric decay)
    eps_start = 5e-3
    eps_end = 1e-9

    # best solution tracking
    best_a = a_start.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[adam init] n={n}  value={best_val:.6f}")

    # main loop
    while time.time() < deadline - 4.0:
        # fraction of total allotted time (0 → start, 1 → end)
        prog = (time.time() - global_start) / (deadline - global_start)
        eps = max(eps_end, eps_start * (0.1 ** prog))

        # 1) current a from w (soft‑max)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # 2) gradient w.r.t. w
        grad_w = _grad_w_from_a(a, eps)

        # 3) Adam parameter update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        w -= step

        # occasional evaluation (every ~200 iters, cheap because a is already known)
        if t % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[adam iter {t:5d}] new best = {best_val:.6f}")

            # try the anti‑symmetric heuristic every few hundred evaluations
            if t % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    # re‑initialise w to match the reordered vector
                    w = np.log(a_ord + 1e-12)
                    m.fill(0.0)
                    v.fill(0.0)
                    t = 0
                    print(f"[adam reorder] improved to {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (phase 3)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Random local search: swaps (20 %) and mass‑transfer moves (80 %).
    Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
