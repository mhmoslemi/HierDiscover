def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.5

    # problem size:
    n = 4000
    # Try to read prior solution
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        if old.shape[0] != n:
            # fallback to uniform
            old = None
    except Exception:
        old = None

    # Stage 1: Build Sidon set
    best_S = None
    best_m = 0
    # theoretical upper bound for m:
    max_m = int((-1 + np.sqrt(16*n - 7)) // 2)
    attempts = 0
    while time.time() < deadline - 30:  # leave time for refinement
        attempts += 1
        S = _greedy_sidonic_set(n, max_m, rng=np.random.default_rng())
        if len(S) > best_m:
            best_m = len(S)
            best_S = S
            print(f"[sidon] attempt {attempts} new best size {best_m}")
            if best_m >= max_m:
                print("[sidon] reached theoretical maximum")
                break

    # Build sequence from best Sidon set
    if best_S is None:
        # fallback uniform sequence
        a = np.full(n, 1.0 / n)
        best_val = _eval_fast(a)
    else:
        m = len(best_S)
        w = 1.0 / m
        a = np.zeros(n)
        a[best_S] = w
        best_val = _eval_fast(a)
        print(f"[sidon] best set size {m}, evaluation = {best_val:.6f}")

    # Stage 2: refinement with subgradient descent
    if time.time() < deadline - 10:
        print("[refine] projected subgradient descent")
        a, best_val = _subgradient_descent(a, deadline - 5)

    # Stage 3: final simulated annealing (swap)
    if time.time() < deadline - 2:
        print("[sa] final simulated annealing")
        a, best_val = _simulated_annealing(a, best_val, deadline - 1)

    # final evaluation (official)
    try:
        final_score = evaluate_sequence(a.tolist())
        print(f"[final] evaluation = {final_score:.6f}")
    except Exception as e:
        print("[final] evaluation error:", e)
        final_score = None

    return a.tolist()

def _greedy_sidonic_set(n, target_size, rng):
    # Random order of indices
    candidates = np.arange(n)
    rng.shuffle(candidates)
    S = []
    sumset = set()
    for idx in candidates:
        # early stop if we already exceed target by size? we aim for up to target.
        # compute needed sums
        coll = False
        # self sum
        s_self = idx + idx
        if s_self in sumset:
            continue
        for j in S:
            if idx + j in sumset:
                coll = True
                break
        if coll:
            continue
        # we can add idx
        sumset.add(s_self)
        for j in S:
            sumset.add(idx + j)
        S.append(idx)
        if len(S) == target_size:
            break
    return S

"""
Hybrid optimizer (improved).

The new version tightens the search in three ways:

1. **More aggressive L‑BFGS** – we use a finer ε‑schedule,
   a larger max‑iteration limit and *many* random restarts.
2. **Two re‑ordering heuristics** – after each L‑BFGS run we try
   (a) the original “ends‑large” ordering and (b) a
   “center‑large” ordering that puts the biggest heights in the middle.
3. **Longer projected sub‑gradient phase** – we spend more time
   refining the best candidate with a sub‑gradient method on the exact
   max‑convolution, then finish with a short simulated‑annealing /
   hill‑climbing pass.

All phases respect the overall 1000 s wall‑time budget.  The program
returns the best list of non‑negative floats it has found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth‐surrogate wrapper (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using FFT.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip rounding artefacts
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` (identical result, no safety checks).
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Convolution used inside the gradient of the smooth surrogate
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where L is the smoothed surrogate

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and grad_w is the gradient w.r.t. the soft‑max parameters w.
    """
    # ---------- soft‑max (positivity, ℓ₁ = 1) ----------
    w_shift = w - np.max(w)                     # numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                               # Σa = 1

    # ---------- convolution a * a ----------
    conv = _fft_convolve_self(a)                 # length 2·n‑1
    max_conv = conv.max()

    # ---------- smooth approximation of the max ----------
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)                # log∑exp((c−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    weights = np.exp(shifted)                   # exp((c−max)/ε)
    p = weights / np.sum(weights)               # probability distribution over k

    # g_i = 4·n·∑_k p_k·a_{k−i}
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    # central n entries correspond to indices [n‑1 , 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                # ∂L/∂a

    # ---------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# bind the helper convolution to the function object so the wrapper can see it
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Put the largest values at the two ends and interleave the smallest.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


def _center_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Put the largest values near the centre and the smallest near the ends.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    # indices from the centre outwards
    centre = (n - 1) // 2
    left = centre
    right = centre + 1
    ptr_low = 0
    ptr_high = n - 1
    i = 0
    while i < n:
        # large value
        new_a[i] = sorted_vals[ptr_high]
        ptr_high -= 1
        i += 1
        if i >= n:
            break
        # small value
        new_a[i] = sorted_vals[ptr_low]
        ptr_low += 1
        i += 1
    return new_a


# ----------------------------------------------------------------------
# Simulated annealing (swap moves) – fast
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """Fast SA that swaps two positions; works directly on the fast evaluator."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.02, 1e-6)      # initial temperature
    decay = 0.99993

    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        a[i], a[j] = a[j], a[i]            # propose swap
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            a[i], a[j] = a[j], a[i]        # revert

        T *= decay
        it += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Sub‑gradient method that directly minimises max(a*a) while
    keeping a on the simplex (non‑negative, ℓ₁ = 1).
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    step0 = 0.13                      # larger initial step (empirically good)
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur_a)
        k_max = int(np.argmax(conv))

        # gradient of the max entry w.r.t. a : 2·a_{k_max-i}  (if inside range)
        idx_arr = k_max - np.arange(n)
        mask = (idx_arr >= 0) & (idx_arr < n)
        grad = np.zeros_like(cur_a)
        grad[mask] = 2.0 * cur_a[idx_arr[mask]]

        alpha = step0 / np.sqrt(it + 1.0)

        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)
        s = np.sum(new_a)
        if s < 1e-12:
            new_a = np.full(n, 1.0 / n)
            s = 1.0
        new_a /= s                     # projection onto ℓ₁ = 1 simplex

        new_val = _eval_fast(new_a)

        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it} new best = {best_val:.6f}")

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val

        it += 1
    return best_a, best_val
