"""
improved_step_function_search.py

Search for a non‑negative step‑function (list of heights) that minimises the
objective used in ``evaluate_sequence``.  The algorithm builds on the previous
implementation but adds several refinements:

* more aggressive random initialisation,
* more L‑BFGS restarts with a finer ε‑schedule,
* a multi‑peak exponential mirror‑descent that uses a larger set of peaks,
* a *smooth* mirror‑descent that uses a soft‑max over **all** convolution
  entries (exponential weighting) – this supplies a richer gradient,
* a lightweight simulated‑annealing hill‑climber that moves mass between
  indices,
* periodic anti‑symmetric re‑ordering (large values ↔ small values) which
  empirically spreads the large products in the self‑convolution.

All heavy convolution work uses FFTs (O(n log n)).  The code respects the
1000 s time budget and returns the best sequence found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS smooth‑max optimiser
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Remove tiny negative artefacts caused by floating point noise
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest coefficients with the smallest ones (mirror ordering).
    This tends to spread large * large products across many convolution indices.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # stable soft‑max
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution a*a
    conv = _full_convolution(a)                # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient of the smooth max term
    weights = np.exp(shifted)
    p = weights / weights.sum()                # distribution over lag indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]               # derivative w.r.t. a

    # chain‑rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule (fine‑grained)
    eps_schedule = np.logspace(np.log10(1e-2), np.log10(1e-9), num=12)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to a (probability simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = 2.0 * n * _full_convolution(a).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best fast = {best_val:.6f}")

        # anti‑symmetric re‑ordering heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = 2.0 * n * _full_convolution(a_ord).max()
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved fast = {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = 2.0 * n * _full_convolution(a).max()

    return best_a, best_val

# Multi‑peak exponential mirror‑descent (uses a handful of largest peaks)
# ----------------------------------------------------------------------
def _gradient_descent_topk(a_start: np.ndarray,
                           deadline: float,
                           top_k: int = 10):
    """
    Mirror‑descent that reduces the *top_k* largest convolution peaks.
    Returns the best (a, value) pair found before ``deadline``.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a).max()
    print(f"[top‑k GD init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.5          # initial step‑size
    decay_rate = 0.98     # decay of α every 2000 iterations

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_val = conv.max()

        # indices of the ``top_k`` largest convolution entries
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]   # sorted descending
        else:
            idxs = np.arange(conv.size)

        # accumulate gradient contributions from the selected peaks
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            jdx = k - idx_range
            g[idx_range] += a[jdx]

        # scheduled step‑size
        alpha = alpha0 * (decay_rate ** (it // 2000))

        # mirror‑descent update (multiplicative)
        a = a * np.exp(-alpha * g)

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:                     # guard against under‑flow
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        # occasional logging / re‑ordering
        if it % 50 == 0:
            cur_val = 2.0 * n * _full_convolution(a).max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[top‑k GD iter {it}] new best fast = {best_val:.6f}")
            if it % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = 2.0 * n * _full_convolution(a_ord).max()
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[top‑k GD reorder] improved fast = {best_val:.6f}")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth (full‑conv) mirror‑descent using exponential weighting of all peaks
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of Σ_k p_k·conv_k w.r.t. a.
    ``p`` must have length 2·n‑1 (the same as the full convolution).
    Returns an array of length n.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    seg = conv_ap[n - 1: 2 * n - 1]            # middle n entries
    return seg


def _gradient_descent_smooth(a_start: np.ndarray,
                             deadline: float,
                             eps_start: float = 1e-3,
                             eps_end: float = 1e-9,
                             eps_decay: float = 0.95,
                             alpha0: float = 0.5,
                             alpha_decay: float = 0.995):
    """
    Mirror‑descent that uses a soft‑max weighting over *all* convolution entries.
    This provides a richer gradient than the pure top‑k approach.
    Returns the best (a, value) pair found before ``deadline``.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a).max()
    print(f"[smooth GD init] value = {best_val:.6f}")

    it = 0
    eps = eps_start
    while time.time() < deadline - 5.0 and eps > eps_end:
        conv = _full_convolution(a)
        max_val = conv.max()

        # exponential weighting around the current maximum
        shifted = (conv - max_val) / eps
        # all entries ≤ 0 → safe exponential
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum == 0.0:
            # extreme case (should not happen) – fall back to a delta at the max
            p = np.zeros_like(conv)
            p[np.argmax(conv)] = 1.0
        else:
            p /= p_sum

        # gradient of the weighted max
        seg = _smooth_gradient(a, p)           # length n
        g = 2.0 * seg                          # factor 2 from d(conv_k)/da_i

        # step‑size schedule
        alpha = alpha0 * (alpha_decay ** it)

        # mirror‑descent update
        a = a * np.exp(-alpha * g)

        # simplex projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        # bookkeeping
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[smooth GD iter {it}] new best fast = {best_val:.6f}")

        # occasional anti‑symmetric re‑ordering
        if it % 300 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = 2.0 * n * _full_convolution(a_ord).max()
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                a = a_ord.copy()
                print(f"[smooth GD reorder] improved fast = {best_val:.6f}")

        # decay ε
        eps = max(eps_end, eps * eps_decay)
        it += 1

    return best_a, best_val

def _simulated_annealing(a_start: np.ndarray,
                         start_val: float,
                         deadline: float,
                         n: int):
    """
    Very lightweight simulated‑annealing that transfers a small fraction of mass
    from one index to another.  Accepts worsening moves with probability
    exp(−Δ/temperature) to escape shallow local minima.
    Returns the best (a, value) pair discovered.
    """
    cur_a = a_start.copy()
    cur_val = start_val
    best_a = cur_a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    temperature = 1e-4
    cooling = 0.9995

    it = 0
    while time.time() < deadline - 5.0:
        # propose a mass transfer
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_a[i] < 1e-16:
            continue
        delta = cur_a[i] * 0.05 * rng.random()   # up to 5 % of the source mass
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # simplex projection (should already sum to 1, but guard against tiny negative)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = 2.0 * n * _full_convolution(new_a).max()
        delta_val = new_val - cur_val
        accept = False
        if delta_val < -1e-12:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-delta_val / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[SA iter {it}] new best fast = {best_val:.6f}")

        # temperature schedule
        temperature = max(temperature * cooling, 1e-12)
        it += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` while respecting the 1000 s time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin before returning
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – 8000 entries have worked well in the past
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from the previously found construction if available
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)          # uniform start

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
        s = a.sum()
    a /= s

    # ------------------------------------------------------------------
    # Fast evaluation of the current candidate
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a).max()
    print(f"[init] n={n}, fast value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – a few random initial vectors to diversify the start
    # ------------------------------------------------------------------
    for tr in range(6):
        if time.time() >= deadline - 200:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        cur_val = 2.0 * n * _full_convolution(a_rand).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] improved fast = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate optimisation (L‑BFGS) with several restarts
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        max_restarts = 6
        restarts = 0
        while time.time() < deadline - 140 and restarts < max_restarts:
            restarts += 1
            # start from current best but add a modest random perturbation
            w_start = np.log(best_a + 1e-12) + rng.normal(scale=0.3, size=n)
            a_opt, val_opt = _l_bfgs_optimize_from_w(w_start, deadline)
            if val_opt < best_val:
                best_val = val_opt
                best_a = a_opt.copy()
                print(f"[L‑BFGS restart {restarts}] new best fast = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multi‑peak exponential mirror‑descent (top‑k)
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        a_gd, val_gd = _gradient_descent_topk(best_a, deadline, top_k=10)
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[top‑k GD] new best fast = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – smooth full‑conv mirror‑descent (exponential weighting)
    # ------------------------------------------------------------------
    if time.time() < deadline - 50:
        a_smooth, val_smooth = _gradient_descent_smooth(best_a, deadline)
        if val_smooth < best_val:
            best_val = val_smooth
            best_a = a_smooth.copy()
            print(f"[smooth GD] new best fast = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climber
    # ------------------------------------------------------------------
    if time.time() < deadline - 20:
        a_sa, val_sa = _simulated_annealing(best_a, best_val, deadline, n)
        if val_sa < best_val:
            best_val = val_sa
            best_a = a_sa.copy()
            print(f"[SA] new best fast = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final lightweight hill‑climbing (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_a = best_a.copy()
        cur_val = best_val
        while time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] < 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            new_a = np.maximum(new_a, 0.0)
            s_new = new_a.sum()
            if s_new < 1e-12:
                continue
            new_a /= s_new
            new_val = 2.0 * n * _full_convolution(new_a).max()
            if new_val < cur_val - 1e-12:
                cur_a = new_a
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = cur_a.copy()
                    print(f"[final climb] new best fast = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided ``evaluate_sequence`` function
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
