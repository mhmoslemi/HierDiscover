def _simplex_projection(v):
    """Projects a vector v onto the probability simplex (non-negative, sum=1)."""
    n = len(v)
    # algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative list) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Initialisation**
   * If a previous solution (`height_sequence_1`) exists, linearly
     interpolate it to the working length `n`.
   * Sweep a two‑parameter Beta family
        a_i ∝ (i+1)^{p‑1}·(n‑i)^{q‑1}
     over a dense grid of (p,q) that also explores very small exponents
     (p,q∈[0.01,0.9]) – these give “U‑shaped’’ profiles that are known to
     flatten the self‑convolution.
   * Keep the best seed (lowest exact objective).

2. **Projected sub‑gradient descent**
   * The objective for a normalised vector (∑a=1) is
        f(a)=2·n·max_k (a*a)_k .
     The sub‑gradient of the max‑convolution term at the current maximal
     index `k*` is   g_i = 2·a_{k*‑i} (if the index exists, otherwise 0).
   * Perform many iterations of a projected gradient step
        a ← Π_{simplex}(a – η·g) ,
     where Π projects onto the probability simplex.
   * Learning‑rate η decays like 0.5/√t; occasional random
     perturbations restart the walk if no improvement is seen for a
     while.

3. **Fine‑tuning with simulated annealing**
   * A short hill‑climbing phase (the same code as the previous
     submission) moves small amounts of mass from “hot’’ indices to
     “cold’’ ones, accepting uphill moves with a decaying temperature.
   * This polishing step extracts the last bits of improvement.

4. **Return the best vector found** (its sum is 1, which does not change
   the final score because the objective is scale‑invariant).

All heavy work (FFT‑based convolutions, simplex projection) uses NumPy
and comfortably fits into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric sweep of a two‑parameter Beta family (including very small exponents)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1} (n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf

    # exponents: very small (U‑shaped), moderate, and larger values
    small = np.geomspace(0.01, 0.09, 9)          # 0.01 … 0.09
    medium = np.linspace(0.1, 1.0, 10)          # 0.1 … 1.0
    large = np.geomspace(1.0, 6.0, 7)[1:]      # 1 … 6 (skip 1 duplicated)
    p_vals = np.concatenate([small, medium, large])
    q_vals = p_vals.copy()

    i = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        for q in q_vals:
            # Beta shape
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_a: np.ndarray,
                  best_val: float) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of projected gradient descent on the exact
    max‑convolution objective.
    Returns the best (a, value) observed before *deadline*.
    """
    a = a0.copy()
    cur_val = _objective(a)
    if cur_val < best_val:
        best_val, best_a = cur_val, a.copy()
        print(f"[PGD init] new best = {best_val:.6f}")

    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 30000  # after this many stagnant steps we add noise

    while True:
        now = time.time()
        if now >= deadline:
            break

        t += 1

        # full self‑convolution and the index of the current maximum
        b = _full_convolution(a)
        max_b = b.max()
        k = int(np.argmax(b))

        # sub‑gradient of the max‑entry w.r.t a
        shift = k - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[shift[mask]]          # 2 * a_{k‑i}

        # step‑size schedule (decays like 1/√t)
        eta = 0.5 / (np.sqrt(t) + 1.0)

        # projected gradient step
        a_candidate = a - eta * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _simplex_projection(a_candidate)

        new_val = _objective(a_candidate)

        # accept if we improve; otherwise keep the old a
        if new_val < cur_val - 1e-15:
            a, cur_val = a_candidate, new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a_candidate.copy()
                print(f"[PGD {t:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1
            # occasionally accept a non‑improving move to escape plateaus
            if np.random.rand() < 0.001:
                a, cur_val = a_candidate, new_val

        # stagnation handling – perturb the current vector a little
        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _simplex_projection(a)
            cur_val = _objective(a)
            no_improve = 0
            print(f"[PGD {t:7d}] stagnation – random perturbation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (same as previous implementation)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from indices that
    contribute heavily to the current maximal convolution entry to indices
    that contribute little.  Accepts uphill moves with a decaying temperature.
    """
    a = a0.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule for maximal transfer fraction and temperature
    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.012, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (probability ∝ contribution)
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index (biased towards low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

                                      best_a=best_a,
                                      best_val=best_val)

    # --------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (remaining time).
    # --------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=phase_a_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
