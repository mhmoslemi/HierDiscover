def hillclimb(a, n, start_time, end_time):
    a = a.copy()
    # ensure sum 1
    a = a / a.sum()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val
    max_frac_start, max_frac_end = 0.5, 0.001
    no_improve = 0
    IMPROVE_LIMIT = 200000
    while True:
        now = time.time()
        if now >= end_time:
            break
        # schedule max_frac
        progress = (now - start_time) / (end_time - start_time)
        max_frac = max_frac_start * (1-progress) + max_frac_end * progress
        # find k_max
        k_max = int(np.argmax(b))
        # compute contrib vector
        idx = k_max - np.arange(n)
        mask = (idx >=0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))
        if a[i] <= 1e-15:
            continue
        # choose target index
        # compute reverse contribution (low contributions)
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        # Inverse weighting
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        totw = w.sum()
        if totw > 0 and np.random.rand() < 0.7:
            probs_j = w / totw
            j = np.random.choice(n, p=probs_j)
        else:
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0:
            continue
        # incremental conv update
        # compute new conv using original a (before move)
        b_new = b.copy()
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a

        b_new[2*i] += delta*delta
        b_new[i+j] -= 2.0 * delta*delta
        new_val = 2.0*n*b_new.max()
        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            no_improve += 1
        if no_improve > IMPROVE_LIMIT:

            no_improve = 0

"""
Propose a step‑function with a low value of the provided evaluation function.

The optimizer works in three stages:

  1. **Exploratory search on a modest size (≈1500).**
     Many simple parametric families (uniform, linear, exponential, power‑law)
     and random Dirichlet draws are generated.  Each candidate is refined by a
     fast, exact hill‑climber that repeatedly moves a small amount of mass
     from an index that contributes heavily to the current maximal
     self‑convolution entry to an index that contributes little.  The best
     vector found on the small grid is stored.

  2. **Progressive up‑sampling.**
     The best small vector is interpolated to a larger grid (≈9000 entries)
     and the hill‑climber is run again.  This up‑sampling step is repeated
     (1500 → 3000 → 6000 → 9000) so that the shape can be refined gradually
     while keeping the cost affordable.

  3. **Final polishing.**
     A short mixing with the uniform vector is tried; finally the
     `evaluate_sequence` wrapper is called to report the score.

All operations respect the 1000 s budget, use only NumPy and SciPy (for the
optimizer we stay in pure NumPy), and guarantee a non‑negative, normalised
output.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft  # fast convolution helpers


# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Return the linear convolution of a with itself (length 2·len(a)‑1)
    using a real FFT.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    # power‑of‑two size for speed
    size = 1 << (out_len - 1).bit_length()
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) assumed = 1)
# ----------------------------------------------------------------------
def objective_value(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * full_convolution(a).max())


# ----------------------------------------------------------------------
# Hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def hillclimb(a0: np.ndarray,
              n: int,
              start: float,
              deadline: float,
              max_frac_start: float = 0.5,
              max_frac_end: float = 0.001,
              improve_limit: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Fast local search that moves a small amount of mass from indices that
    contribute most to the maximal convolution entry to indices that
    contribute little.

    Returns the best vector seen (still normalised to sum 1) and its exact
    objective value.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    b = full_convolution(a)          # length 2·n‑1
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    no_improve = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear schedule for the transferred fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – proportional to its contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – tend to have a small reversed contribution
        if np.random.rand() < 0.7:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            # inverse weighting (favor small contributions)
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                # fallback: uniform choice different from i
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            # plain uniform target
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move (fraction of source mass)
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # Incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:          # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                # print progress only on a real improvement
                print(f"[Hillclimb] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling – shake and restart from the current best
        if no_improve > improve_limit:
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            s = a.sum()
            if s == 0:
                a = best_a.copy()
            else:
                a /= s
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print("[Hillclimb] perturbing after stagnation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Small post‑processing sweep – mixing with uniform
# ----------------------------------------------------------------------
def mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a' = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function (vector) within a given time budget.
    The strategy is:
        * cheap exploration on a small grid,
        * progressive up‑sampling,
        * final hill‑climbing and a tiny uniform mix.
    Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_global = time.time()
    # a small safety margin for the final “evaluate_sequence” call
    deadline_global = start_global + budget_s - 0.5

    # --------------------------------------------------------------
    # Parameters
    # --------------------------------------------------------------
    n_small = 1500                     # cheap grid size
    n_target = 9000                    # final resolution used earlier
    # allocate about 35 % of the budget to the exploratory stage
    deadline_stage1 = start_global + 0.35 * (deadline_global - start_global)

    # --------------------------------------------------------------
    # 1) Exploratory phase on the small grid
    # --------------------------------------------------------------
    best_a_small = np.full(n_small, 1.0 / n_small)
    best_val_small = objective_value(best_a_small)

    # helper for evaluating a candidate (normalising it first)
    def eval_and_update(a_candidate: np.ndarray):
        nonlocal best_a_small, best_val_small
        a = np.maximum(a_candidate, 0.0)
        s = a.sum()
        if s == 0:
            return
        a /= s
        val = objective_value(a)
        if val < best_val_small:
            best_val_small, best_a_small = val, a.copy()
            print(f"[Stage 1] new best (n={n_small}) = {best_val_small:.6f}")

    # a handful of deterministic families
    i = np.arange(n_small, dtype=float) + 1.0
    eval_and_update(np.ones(n_small))                      # uniform
    eval_and_update(np.arange(n_small, 0, -1, dtype=float)) # linear decreasing
    eval_and_update(np.linspace(0.0, 1.0, n_small) ** 0.5)  # sqrt‑shaped
    for r in np.linspace(0.95, 0.9995, 8):
        eval_and_update(r ** np.arange(n_small))                # one‑sided exponential
        eval_and_update(r ** np.arange(n_small) +
                        r ** (n_small - 1 - np.arange(n_small)))  # symmetric exponential
    for p in np.linspace(0.2, 3.0, 10):
        eval_and_update(i ** (-p) + (n_small - i + 1) ** (-p))   # power‑law U‑shape

    rng = np.random.default_rng(12345)
    for _ in range(15):
        eval_and_update(rng.dirichlet(np.ones(n_small)))

    # ------------------------------------------------------------------
    # Run a short hill‑climb on each of the above candidates.
    # ------------------------------------------------------------------
    for idx, cand in enumerate([best_a_small] + [rng.dirichlet(np.ones(n_small))
                                                for _ in range(5)]):
        now = time.time()
        if now >= deadline_stage1:
            break
        # give each run a modest amount of time (≈2 % of total budget)
        stage_deadline = now + 0.02 * (deadline_stage1 - start_global)
        a_opt, val_opt = hillclimb(cand, n_small,
                                   start=now,
                                   deadline=stage_deadline,
                                   improve_limit=120_000)
        eval_and_update(a_opt)

    # --------------------------------------------------------------
    # If we have a previously computed solution, try to import it.
    # --------------------------------------------------------------
    try:
        # height_sequence_1 may be defined in the surrounding notebook; if so,
        # interpolate it to the small grid and use it as another start.
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n_small)
        cand = np.interp(new_x, old_x, old)
        eval_and_update(cand)
    except Exception:
        pass  # ignore if the global does not exist

    # --------------------------------------------------------------
    # 2) Progressive up‑sampling to the target size
    # --------------------------------------------------------------
    def upsample(a_small: np.ndarray, n_new: int) -> np.ndarray:
        """Linear interpolation from a_small (len m) to length n_new."""
        x_old = np.linspace(0.0, 1.0, a_small.shape[0])
        x_new = np.linspace(0.0, 1.0, n_new)
        a_new = np.interp(x_new, x_old, a_small)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0:
            a_new = np.full(n_new, 1.0 / n_new)
        else:
            a_new /= s
        return a_new

    # up‑sample through a cascade of sizes to keep the shape stable
    sizes = [n_small, 3000, 6000, n_target]
    a_current = best_a_small
    for sz in sizes[1:]:
        a_current = upsample(a_current, sz)
        # allocate a small slice of the remaining budget to each stage
        now = time.time()
        remaining = deadline_global - now
        if remaining < 5.0:
            break
        # give each up‑sampling stage at most 5 % of the total remaining time
        stage_deadline = now + 0.05 * (deadline_global - start_global)
        a_current, _ = hillclimb(a_current, sz,
                                 start=now,
                                 deadline=stage_deadline,
                                 improve_limit=150_000)

    # --------------------------------------------------------------
    # 3) Final polishing on the full size
    # --------------------------------------------------------------
    now = time.time()
    if now < deadline_global - 5.0:
        a_current, final_val = hillclimb(a_current, n_target,
                                         start=now,

                                         improve_limit=200_000)
    else:
        final_val = objective_value(a_current)

    # a tiny uniform mixing sometimes helps a little more
    a_current, final_val = mix_with_uniform(a_current, n_target)

    # --------------------------------------------------------------
    # Final evaluation using the provided secure wrapper (just for logging)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_current.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
