def _smoothed_descent(a, deadline, rng, max_iters=20000):
    n = len(a)
    lr = 0.1
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_val = _objective(a)
    alpha = 1.0   # temperature for softmax
    increase_alpha_every = 500  # increase after some iterations
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a)
        maxb = conv.max()
        # if objective improved
        cur_val = 2.0 * n * maxb
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # compute softmax weights
        # Using scaling: w = exp(alpha * (conv - maxb))  (so max weight = 1)
        # Or use w = exp(alpha * conv) with subtract max for stability
        scaled = alpha * (conv - maxb)   # max = 0
        # To avoid underflow for large negative values, clip:
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # compute gradient via correlation of w and a
        # Use np.correlate for efficiency for moderate lengths
        grad = 2.0 * np.correlate(w, a, mode='full')[:n]  # shape n
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # enforce nonnegativity and simplex
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = rng.random(len(a))
            s = a.sum()
        a /= s

        # adapt alpha (increase gradually)
        if it % increase_alpha_every == 0:
            alpha = min(alpha * 1.2, 30.0)   # cap to avoid overflow
    return best_a, best_val

Optimisation of the step‑function attack.

The algorithm is a multi‑restart, continuation‑style optimiser:

1.  Randomly pick a length n (from a small set of candidates) and
    initialise a non‑negative vector a with sum = 1 (Dirichlet draw
    or a resized copy of the previously best construction).

2.  **Smoothed Adam descent** – minimise a smooth approximation of
    ``max_k (a*a)[k]``.
    For a temperature α we form soft‑max weights
        w_k ∝ exp(α·(conv[k]–max(conv))).
    The gradient of the weighted sum is
        ∇ = 2·corr(w, a)                # correlation, not convolution
    (computed with NumPy's correlate, i.e. O(n·2n) which is fast for
    n≤4000).  An Adam update with projection onto the probability simplex
    is applied.  α is increased every few hundred iterations (continuation
    method) so that the optimisation first follows a smooth landscape
    and gradually focuses on the true L∞‑objective.

3.  **Targeted mass‑transfer hill‑climb** – starting from the point
    returned by (2) we repeatedly move a fraction of the mass that
    contributes most to the current maximal lag to two new positions that
    have the smallest *risk* of increasing that lag (the risk for a
    destination p is a[k−p] if the partner index exists, otherwise 0).
    The move is accepted only if the exact objective improves.
    Step‑size (the transferred fraction) is adapted automatically.

4.  The whole process is repeated until the global time budget (default
    1000 s) is exhausted.  The best sequence seen over all restarts is
    returned.  The final score is printed using the provided
    ``evaluate_sequence`` function.

All operations respect the non‑negativity and simplex constraints,
and the code runs on a single CPU without any external I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (FFT‑based convolution, objective, gradient on a lag)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using an FFT (real input)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives; clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the convolution entry conv[k] w.r.t. a."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad


# ----------------------------------------------------------------------
# Smoothed Adam descent (continuation on the soft‑max temperature)
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator,
                  max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of max(conv).
    Returns the best vector encountered and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    lr = 0.06
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 1.0               # start with a very smooth approximation
    alpha_mult = 1.2
    alpha_inc_every = 400     # iterations

    best_a = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a)
        maxb = conv.max()
        cur_val = 2.0 * n * maxb
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # --- soft‑max over convolution entries ---------------------------------
        # shift by the max to avoid overflow; the maximal entry gets weight 1.
        shifted = alpha * (conv - maxb)
        # guard against under‑/overflow
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the weighted sum (correlation) -----------------------
        # grad_i = 2 * sum_j w_{i+j} * a_j   (i = 0..n‑1)
        # This is exactly the (non‑circular) correlation of w and a.
        grad = 2.0 * np.correlate(w, a, mode='full')[:n]

        # --- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the probability simplex
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = rng.random(n)
            s = a.sum()
        a /= s

        # increase temperature slowly – the objective becomes sharper
        if it % alpha_inc_every == 0:
            alpha = min(alpha * alpha_mult, 30.0)   # avoid overflow

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill‑climb (with risk‑aware destination choice)
# ----------------------------------------------------------------------


def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 5000) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that repeatedly moves a fraction of the most damaging
    mass to two low‑risk positions.  Stops after the time budget expires
    or after ``max_no_improve`` consecutive unsuccessful attempts.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.20          # initial fraction of mass to relocate
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that contribute to the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        # pick the pair (i,j) with largest product
        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we are willing to shift
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-8)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # ----- risk‑aware choice of destinations ---------------------------
        # Adding δ to position p increases the maximal lag k by 2·a[k−p]·δ
        # (if k−p is a valid index).  We therefore prefer indices with a
        # minimal partner value.
        partner = k - np.arange(n)          # partner index for each destination
        valid = (partner >= 0) & (partner < n)
        risk = np.zeros_like(new_a)
        risk[valid] = new_a[partner[valid]]
        # mask out the donors
        mask = np.ones(n, dtype=bool)
        mask[i] = False
        mask[j] = False
        risk[~mask] = np.inf               # donors cannot be destinations

        # select two distinct low‑risk positions
        cand = np.argpartition(risk, 2)[:2]
        # In the unlikely case of ties with Inf, fall back to random choice
        if np.isinf(risk[cand]).any():
            low = np.where(mask)[0]
            if low.size < 2:
                cand = np.random.choice(np.arange(n), size=2, replace=False)
            else:
                cand = rng.choice(low, size=2, replace=False)
        p, q = int(cand[0]), int(cand[1])

        new_a[p] += delta
        new_a[q] += delta

        # renormalise (projection onto simplex)
        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-8)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.04, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.90, 1e-8)
            no_improve += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# One‑restart optimisation (descent + hill‑climb)
# ----------------------------------------------------------------------


def _optimise_one(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Runs the Adam descent followed by the targeted hill‑climb.
    Returns the best sequence and its exact objective value.
    """
    # 1. smooth Adam descent
    a_desc, val_desc = _adam_descent(a0, deadline, rng, max_iters=12000)

    # 2. hill‑climb refinement
    a_final, val_final = _mass_transfer_hillclimb(a_desc, deadline, rng,
                                                  max_no_improve=4000)

    return a_final, val_final


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    # Candidate lengths – smaller n gives faster inner loops, larger n

    length_choices = [500, 1000, 1500, 2000, 3000, 4000]

    best_val = np.inf
    best_seq = None
    restart = 0

        old = np.maximum(old, 0.0)
        if old.sum() == 0.0:

    except Exception:
        old = None

    while time.time() < deadline:
        restart += 1
        n = int(rng.choice(length_choices))

        # Initialise a new candidate ------------------------------------------------
        if old is not None:
            # Resize the known good construction to the desired length
            if old.size > n:
                a0 = old[:n].copy()
            else:
                pad = rng.random(n - old.size)
                a0 = np.concatenate([old, pad])
        else:

            a0 = rng.random(n)
            s0 = a0.sum()
        a0 /= s0

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[overall] new best after restart {restart} = {best_val:.6f}")

    # Final reporting via the supplied evaluator
