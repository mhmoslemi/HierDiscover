"""
Hybrid optimizer using the Frank‑Wolfe style linear‑programming step
described in the challenge statement.

Idea
----
For a fixed probability vector f (∑ f_i = 1, f_i ≥ 0) the convolution
(f * g) is linear in g.  Hence we can (approximately) solve the
sub‑problem

    maximise   1ᵀg
    subject to  (f * g)_k ≤ M   for all k,
                g ≥ 0,

where M = max_k (f * f)_k.  The solution g tries to put as much mass
as possible into positions where f is small, thereby flattening the
self‑convolution when we replace f by a convex combination
f←(1−t)f + t g.  Repeating this step (a Frank‑Wolfe / conditional
gradient scheme) drives the peak of the convolution down.

The implementation uses:
* a fast FFT based evaluator (identical to ``evaluate_sequence`` but
  without the safety checks);
* ``scipy.optimize.linprog`` with the HiGHS interior‑point solver to
  solve the linear program;
* a short projected sub‑gradient refinement on the true max‑convolution
  (the final “polish” stage used in the previous code);
* a tiny “ends‑large” re‑ordering heuristic applied at the end.

The whole search respects the 1000 s wall‑time budget and returns the
best sequence discovered.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Fast convolution utilities (identical to the original evaluator)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)    # eliminate tiny negatives from rounding
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Exact copy of evaluate_sequence (no security checks)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    # the original function returns inf if sum_a is too small – we guard against this
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Convolution matrix (Toeplitz) required for the LP step
# ----------------------------------------------------------------------
def _make_conv_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix A with entries
        A[k, j] = a_{k‑j}   if 0 ≤ k‑j < n, else 0 .
    For a fixed a this matrix satisfies (A @ g)_k = (a * g)_k .
    """
    n = a.shape[0]
    # Each diagonal offset d (0 … n‑1) is constant = a[d]
    diag_data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = scipy.sparse.diags(diag_data, offsets,
                           shape=(2 * n - 1, n), format='csr')
    return A


# ----------------------------------------------------------------------
# Linear‑programming step (maximise ∑ g under a * g ≤ M)
# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   1ᵀg
        subject to  (a * g)_k ≤ M   (k = 0 … 2n‑2)
                    g ≥ 0
    Returns the optimal g (not normalised) or None if the solver fails.
    """
    n = a.shape[0]
    A = _make_conv_matrix(a)                     # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64) # right‑hand side
    c = -np.ones(n, dtype=np.float64)            # maximise ∑g  → minimise −∑g
    bounds = [(0.0, None)] * n

    # HiGHS is fast for large, sparse LPs
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True,
                                          'time_limit': 30.0})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Simple projected sub‑gradient descent (polishing)
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Keeps a on the simplex (non‑negative, L₁ = 1).
    """
    n = a.shape[0]
    cur = a.copy()
    cur_val = _eval_fast(cur)
    best = cur.copy()
    best_val = cur_val
    step0 = 0.12
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur)
        k = int(np.argmax(conv))

        # sub‑gradient of max conv w.r.t. a : 2·a_{k−i} (zero outside range)
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[idx[mask]]

        # diminishing step size
        alpha = step0 / np.sqrt(it + 1.0)
        nxt = cur - alpha * grad
        nxt = np.maximum(nxt, 0.0)
        s = np.sum(nxt)
        if s < 1e-12:               # protect against collapse
            nxt = np.full(n, 1.0 / n)
            s = 1.0
        nxt /= s                      # project onto the simplex

        val = _eval_fast(nxt)
        if val < best_val:
            best_val = val
            best = nxt.copy()
            print(f"[subgrad] iter {it:5d}  new best = {best_val:.6f}")

        if val < cur_val:
            cur = nxt
            cur_val = val
        it += 1
    return best, best_val


# ----------------------------------------------------------------------
# Small heuristic – put the largest values at the two ends
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Deterministic re‑ordering that alternates largest / smallest values."""
    n = a.shape[0]
    vals = np.sort(a)               # ascending
    out = np.empty_like(a)
    l, r = 0, n - 1
    i = 0
    while i < n:
        out[i] = vals[r]            # large
        r -= 1
        i += 1
        if i >= n:
            break
        out[i] = vals[l]            # small
        l += 1
        i += 1
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found within the allotted wall‑time.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # 1 s safety margin
    print(f"[init] time budget = {budget_s}s, deadline @ {deadline:.2f}")

    # ------------------------------------------------------------------
    # problem size – we use 2000 entries (good compromise between quality
    # and LP size).  The final evaluation works for any length.
    # ------------------------------------------------------------------
    n = 2000
    # ------------------------------------------------------------------
    # initialise from a previously found construction if available
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the required size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a = np.full(n, 1.0 / n)    # uniform start
        print("[init] using uniform start")

    a = np.maximum(a, 0.0)
    a /= np.sum(a)                # enforce L₁ = 1

    best_a = a.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Frank‑Wolfe style LP iterations
    # ------------------------------------------------------------------
    iter_lp = 0
    # we keep ~30 s for the final polishing stages
    while time.time() < deadline - 40.0:
        iter_lp += 1
        conv = _fft_convolve_self(best_a)
        M = float(np.max(conv))

        g = _lp_step(best_a, M)
        if g is None:
            print("[LP] solver failed – exiting LP loop")
            break

        s = np.sum(g)
        if s <= 0.0:
            print("[LP] got zero‑mass solution – abort")
            break
        g = g / s                         # renormalise to L₁ = 1

        # convex combination – a modest step keeps the iteration stable
        t = 0.3
        cand = (1.0 - t) * best_a + t * g
        cand /= np.sum(cand)              # re‑project onto the simplex

        cand_val = _eval_fast(cand)
        if cand_val < best_val:
            best_val = cand_val
            best_a = cand.copy()
            print(f"[LP] iter {iter_lp:3d} → new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – ends‑large re‑ordering (cheap, sometimes helpful)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        reordered = _ends_large_reorder(best_a)
        reordered /= np.sum(reordered)
        val_r = _eval_fast(reordered)
        if val_r < best_val:
            best_val = val_r
            best_a = reordered.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient refinement (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[refine] starting projected sub‑gradient descent")
        best_a, best_val = _subgradient_descent(best_a, deadline)

    # ------------------------------------------------------------------
    # Final official evaluation (the wrapper does safety checks)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return plain Python list as required by the framework
    return best_a.tolist()
