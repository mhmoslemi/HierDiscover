"""
Hybrid optimiser (extended) for the step‑function convolution problem.

Key ideas
---------
* Use a larger problem size (n = 8000) – the theoretical lower bound (≈1.0)
  improves with n, giving the optimiser more freedom.
* Two smooth surrogates for the max‑convolution:
    - Log‑sum‑exp (the classic surrogate).
    - p‑norm surrogate  ‖conv‖_p (p gradually increased, which
      approximates the ∞‑norm).
* L‑BFGS is applied to the soft‑max parametrisation (ensures
  non‑negativity and ℓ₁‑normalisation).
* After each optimisation stage cheap deterministic re‑orderings are
  tried (anti‑symmetric, alternating, descending).
* A specialised “max‑reduction” gradient step focuses on the current
  maximal convolution entry and moves mass away from the offending
  indices while keeping the vector on the simplex.
* An extended simulated‑annealing phase (swaps, mass transfers,
  max‑reduction moves) finishes the search.
* Finally a short hill‑climbing polish.

The whole routine respects the 1000 s wall‑clock budget and returns the
best sequence found as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global parameters that are modified by the optimiser
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for log‑sum‑exp surrogate
_GLOBAL_P = None     # p‑norm exponent for p‑norm surrogate
_GLOBAL_N = None     # current problem size (length of the sequence)


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Numerical noise may create tiny negatives – clamp them.
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast replica of the provided `evaluate_sequence` (non‑negative a)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# 1. Smooth surrogates and their gradients
# ----------------------------------------------------------------------
def _logsumexp_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """Surrogate using log‑sum‑exp (smooth max)."""
    w_shift = w - np.max(w)                # numerical stability
    a = np.exp(w_shift)
    a /= np.sum(a)                         # soft‑max → simplex

    conv = _fft_convolve_self(a)
    max_conv = np.max(conv)

    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / np.sum(weights)          # softmax over convolution indices
    # p_rev for the transpose‑convolution
    p_rev = p[::-1]
    # a * p_rev  (full linear convolution)
    size = _next_pow2((2 * n - 1) + n - 1)
    Fa = np.fft.rfft(a, size)
    Fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(Fa * Fp, size)[: (2 * n - 1 + n - 1)]
    seg = conv_ap[n - 1: 2 * n - 1]        # central part of length n
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _logsumexp_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by `scipy.optimize.minimize`."""
    return _logsumexp_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _pnorm_obj_and_grad(w: np.ndarray, p: float, n: int):
    """Surrogate using the p‑norm ‖conv‖_p (p → ∞ gives the max)."""
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= np.sum(a)

    conv = _fft_convolve_self(a)                     # length 2n‑1

    # ‖conv‖_p computed safely via log‑sum‑exp on log‑values
    log_conv = np.log(conv + 1e-300)                 # avoid log(0)
    log_sum = logsumexp(p * log_conv)               # = log(∑ conv^p)
    J = np.exp(log_sum / p)                         # p‑norm value

    # weight vector w_i = conv_i^{p‑1} / J^{p‑1}
    # computed in log‑space for stability
    log_w = (p - 1) * log_conv - (p - 1) * np.log(J + 1e-300)
    w_vec = np.exp(log_w)                           # zeros are produced correctly

    # gradient dJ/da = 2 * (C^T w_vec)  (C is convolution matrix of a)
    # compute C^T w_vec via cross‑correlation (FFT)
    size = _next_pow2((2 * n - 1) + n - 1)          # length >= 3n‑2
    Fw = np.fft.rfft(w_vec, size)
    Fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(Fw * np.conj(Fa), size)     # full cross‑correlation
    # First n entries (lags –(n‑1)…0) – we need them reversed
    grad_a = 2.0 * corr[:n][::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return J, grad_w


def _pnorm_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper for the p‑norm surrogate."""
    return _pnorm_obj_and_grad(w, _GLOBAL_P, _GLOBAL_N)


# ----------------------------------------------------------------------
# 2. Re‑ordering heuristics (cheap, sometimes improve the max‑conv)
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large values left, small values right (pairwise)."""
    sorted_vals = np.sort(a)
    n = a.shape[0]
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = sorted_vals[n - 1 - i]
        out[n - 1 - i] = sorted_vals[i]
    if n % 2:
        out[half] = sorted_vals[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    sorted_vals = np.sort(a)
    left, right = 0, len(sorted_vals) - 1
    out = np.empty_like(a)
    idx = 0
    while left <= right:
        out[idx] = sorted_vals[right]
        idx += 1
        right -= 1
        if left <= right:
            out[idx] = sorted_vals[left]
            idx += 1
            left += 1
    return out


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Simple descending sort."""
    return np.sort(a)[::-1]


# ----------------------------------------------------------------------
# 3. Simple parametric families for seeding
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """a_i ∝ q**i, normalised to sum = 1 (0 < q < 1)."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q ** n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)**p, normalised to sum = 1 (p < 0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# 4. Targeted max‑reduction move (gradient step on the current peak)
# ----------------------------------------------------------------------
def _max_reduction_step(a: np.ndarray, step_frac: float = 0.02) -> np.ndarray:
    """
    One deterministic step that reduces the current maximal convolution entry.

    The direction is –a shifted by the index of the peak (i.e. the gradient of
    conv[max] w.r.t. a).  The step size is limited so that all components stay
    non‑negative; after the move the vector is re‑normalised to sum = 1.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    k_max = int(np.argmax(conv))

    # partner[i] = a[k_max - i] if inside the array, else 0
    idx = np.arange(n)
    partner = np.where((k_max - idx >= 0) & (k_max - idx < n),
                       a[k_max - idx], 0.0)

    # direction = -partner  (steepest descent on the peak)
    # maximal admissible step to keep a >= 0:
    mask = partner > 0
    if np.any(mask):
        max_eta = np.min(a[mask] / partner[mask])
        eta = min(step_frac * max_eta, step_frac)  # keep it modest
    else:
        eta = 0.0

    new_a = a - eta * partner
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        return a  # fallback – should never happen
    new_a /= s
    return new_a


# ----------------------------------------------------------------------
# 5. Extended simulated annealing (swaps, transfers, max‑reduction moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """SA that mixes swaps, random mass transfers and the targeted max‑reduction."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.05, 1e-7)   # initial temperature
    decay = 0.99995
    iteration = 0

    while time.time() < deadline:
        move_type = rng.random()
        if move_type < 0.35:                         # swap two positions
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif move_type < 0.70:                       # random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * 0.06 * rng.random()
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        else:                                        # max‑reduction move
            new_a = _max_reduction_step(a, step_frac=0.03)

        # re‑normalise (numerical drift may appear)
        s = new_a.sum()
        if s == 0.0:
            continue
        new_a /= s

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {iteration})")

        T = max(T * decay, 1e-12)
        iteration += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# 6. Final hill‑climbing (swap + tiny transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.25:               # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                 # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.03 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑normalise
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s == 0.0:
            continue
        new_a /= s

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# 7. Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.

    The routine follows the stages described in the module docstring.
    It respects a wall‑clock budget (default 1000 s) and finally returns a
    plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # problem size – larger sequences give a tighter theoretical bound
    # ------------------------------------------------------------------
    n = 8000
    global _GLOBAL_N
    _GLOBAL_N = n

    # ------------------------------------------------------------------
    # 0. initialise from uniform or a previously saved construction
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # interpolate to the new size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Hand‑crafted seed families
    # ------------------------------------------------------------------
    seeds = []

    for q in np.linspace(0.6, 0.999, 30):
        seeds.append(_geometric_sequence(q, n))

    for p in np.linspace(-3.5, -0.2, 30):
        seeds.append(_power_sequence(p, n))

    for _ in range(50):
        a = np.random.rand(n)
        a /= a.sum()
        seeds.append(a)

    for a in seeds:
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[seed] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Coarse L‑BFGS with log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    stage_a_deadline = time.time() + min(0.22 * budget_s,
                                        time_left * 0.45)

    eps_schedule = np.logspace(np.log10(5e-2), np.log10(5e-5), num=6)

    while time.time() < stage_a_deadline:
        # a fresh random start in the soft‑max space
        w = np.random.randn(n) * 2.0
        for eps in eps_schedule:
            if time.time() >= stage_a_deadline:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps
            res = minimize(_logsumexp_obj_and_grad_wrapper, w,
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': 250,
                                    'ftol': 1e-9,
                                    'gtol': 1e-9,
                                    'disp': False})
            w = res.x
            a = np.exp(w - np.max(w))
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[coarse] eps {eps:.1e} new best = {best_val:.6f}")

        # deterministic reorderings after each restart
        for reorder in (_anti_symmetric_reorder,
                        _alternating_reorder,
                        _descending_reorder):
            a_r = reorder(best_a)
            v_r = _eval_fast(a_r)
            if v_r < best_val:
                best_val = v_r
                best_a = a_r.copy()
                print(f"[reorder] {reorder.__name__} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Refined optimisation with p‑norm surrogate (p ↑)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    stage_b_deadline = time.time() + min(300.0,
                                         max(0.0, time_left - 30.0))

    p_schedule = [2, 4, 8, 16, 32, 64, 128]
    # initialise w from the current best (log‑softmax)
    w = np.log(best_a + 1e-16)

    while time.time() < stage_b_deadline:
        for p in p_schedule:
            if time.time() >= stage_b_deadline:
                break
            global _GLOBAL_P
            _GLOBAL_P = p
            res = minimize(_pnorm_obj_and_grad_wrapper, w,
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': 400,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            a = np.exp(w - np.max(w))
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[p‑norm] p={p} new best = {best_val:.6f}")

        # a couple of extra reorderings (they are cheap)
        for reorder in (_anti_symmetric_reorder, _alternating_reorder):
            a_r = reorder(best_a)
            v_r = _eval_fast(a_r)
            if v_r < best_val:
                best_val = v_r
                best_a = a_r.copy()
                print(f"[reorder‑p] {reorder.__name__} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Targeted max‑reduction gradient steps (deterministic)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        move_deadline = time.time() + min(30.0, time_left * 0.5)
        iteration = 0
        while time.time() < move_deadline:
            new_a = _max_reduction_step(best_a, step_frac=0.025)
            new_val = _eval_fast(new_a)
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[max‑reduce] new best = {best_val:.6f} (iter {iteration})")
            iteration += 1

    # ------------------------------------------------------------------
    # 5. Extended simulated annealing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120.0:
        sa_deadline = time.time() + remaining - 80.0   # keep ~80 s for polishing
    else:
        sa_deadline = deadline - 10.0
    print(f"[SA] running until {sa_deadline:.2f}")
    best_a, best_val = _simulated_annealing(best_a, best_val, sa_deadline)

    # ------------------------------------------------------------------
    # 6. Final hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # 7. Exact evaluation (for the record)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
