"""
Mirror‑descent optimiser for `evaluate_sequence`.

The algorithm works on a probability vector `a` (Σa = 1, aᵢ ≥ 0) and
minimises the exact objective  

    F(a) = 2·n·max(a * a)

by a sequence of phases

1. **Smooth mirror descent** – minimise a smoothed
   surrogate of the max (log‑sum‑exp) with an Adam‑type update
   on the simplex.  The smoothing parameter ε is decreased
   geometrically (continuation).

2. **Random restarts** – after a run the current best vector is
   randomly perturbed and optimised again; the best of all runs
   is kept.

3. **Polishing** – aggressive mass‑transfer hill‑climbing
   (as in the reference code) followed by a fine sub‑gradient
   descent with projection onto the simplex.

The whole routine respects the 1000 s time budget and prints
intermediate best values.
"""

import time
import numpy as np
from scipy.optimize import minimize  # only needed for a fallback import

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (sum of a is 1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _exact_objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


# ----------------------------------------------------------------------
# Gradient of the smoothed surrogate L_ε(a) = 2·n·max + 2·n·ε·log Σexp((c‑max)/ε)
# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """Return ∇_a L_ε(a)."""
    n = a.shape[0]
    conv = _full_convolution(a)                # length 2·n‑1
    max_c = conv.max()
    shifted = (conv - max_c) / eps             # max is 0 after division
    # stable exponentiation (maximum is exactly 0)
    w = np.exp(shifted)                         # ≤ 1
    p = w / w.sum()                             # probabilities on lags
    p_rev = p[::-1]                             # reversed for convolution
    # a * p_rev → length 3·n‑2; central n entries give the needed sum
    conv_ap = _conv_fft(a, p_rev)
    seg = conv_ap[n - 1: 2 * n - 1]             # middle n values
    grad = 4.0 * n * seg[::-1]                 # as derived in the analysis
    return grad


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (Adam on the simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   eps_schedule: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Optimises `a` using Adam on the smoothed surrogate.
    Returns the best vector found and its exact objective.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _exact_objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    for eps in eps_schedule:
        # a modest number of inner iterations; we stop early if the budget ends
        max_iter = 400
        for _ in range(max_iter):
            if time.time() >= deadline:
                return best_a, best_val

            # gradient of smoothed objective
            grad = _smooth_grad(a, eps)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # learning‑rate scales with ε – larger ε allows larger steps
            lr = 0.05 * (eps / (eps + 1e-6)) + 1e-4
            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            a = a - step
            # projection onto the simplex
            a = _project_simplex(a)

            # occasional exact evaluation
            if t % 25 == 0:
                cur_val = _exact_objective(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[adam] eps={eps:.1e} iter={t} new best = {best_val:.6f}")

    # final check
    final_val = _exact_objective(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill climbing (same logic as the reference)
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              cur_best_val: float,
                              deadline: float,
                              rng: np.random.Generator,
                              step_start: float = 0.2) -> tuple[np.ndarray, float]:
    """Hill‑climbing that moves a fraction of mass away from the active lag."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_best_val
    best_a = cur_a.copy()
    step_frac = step_start

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # indices that contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]

        if contrib.size == 0:
            continue
        top_n = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta
        if not ok:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue

        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue
        new_a /= s

        new_val = _exact_objective(new_a)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-6)

    return best_a, best_val


# ----------------------------------------------------------------------
# Fine sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         cur_best_val: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Plain projected sub‑gradient descent on 2·n·max(conv)."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_best_val
    best_a = cur_a.copy()
    best_val = cur_val
    eta = 0.1

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = _exact_objective(new_a)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            eta = min(eta * 1.05, 0.5)
        else:
            eta = max(eta * 0.7, 1e-5)

    return best_a, best_val
