"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Problem size** – we use a moderate length (n≈1024).  A smaller n lets us
   explore many more starting points within the 1000 s budget while still
   allowing a very low value of the objective (the theoretical lower bound
   is ≈1 for any n).

2. **Random‑restart smoothed optimisation** –  The nonsmooth objective
   ``max(conv)`` is approximated by a log‑sum‑exp surrogate.
   For each restart we initialise a log‑softmax vector `w` with Gaussian noise
   and minimise the surrogate by L‑BFGS‑B for a decreasing schedule of
   smoothing parameters ε∈[1,10⁻¹⁴].  Each run yields a feasible probability
   vector `a` (Σa=1, aᵢ≥0).

3. **Aggressive hill‑climbing** –  Starting from the best vector found,
   repeatedly relocate a fraction of the mass that contributes to the
   currently maximal convolution lag.  The move is accepted only if the
   exact objective improves.  This phase quickly reduces the peak.

4. **Fine sub‑gradient polishing** –  A sub‑gradient descent on the true
   objective (2·n·max(conv)) with Euclidean projection onto the simplex
   further refines the solution.

All phases respect the global time limit; intermediate progress is printed.
The final list of heights is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full (non‑reversed) convolution of a vector with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (value, gradient) of the surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv−max)/ε),

    where a = softmax(w) (Σa = 1, aᵢ ≥ 0).
    """
    # soft‑max, numerically stable
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # autocorrelation
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    # use logaddexp.reduce for stability
    log_sum = np.logaddexp.reduce(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()      # probability distribution over lags
    # We need (conv * rev(p))_middle
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)   # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]  # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill‑climbing
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              best_val: float,
                              deadline: float,
                              rng: np.random.Generator,
                              step_start: float = 0.25):
    """
    Move mass away from the lag where the convolution attains its maximum.
    Returns the best vector encountered before the deadline.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    best_a = cur_a.copy()
    step_frac = step_start

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # indices (i,j) that form the active lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]

        if contrib.size == 0:
            continue

        # choose the most "dangerous" pairs (largest product)
        top_n = min(4, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        # destination indices – pick from relatively low‑mass locations
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-8)
            continue

        # re‑normalise (preserves Σa)
        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-8)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.08, 0.6)
        else:
            step_frac = max(step_frac * 0.95, 1e-8)

    return best_a, best_val


# ----------------------------------------------------------------------
# Fine sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         best_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Sub‑gradient descent on 2·n·max(conv) with projection onto the simplex.
    Returns the best vector found before the deadline.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    best_a = cur_a.copy()
    eta = 0.12  # initial step size (relative to the simplex scale)

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient for the active lag
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # step + projection onto simplex
        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            eta = min(eta * 1.07, 0.7)
        else:
            eta = max(eta * 0.73, 5e-6)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Returns a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a safety margin

    # ------------------------------------------------------------------
    # Problem dimension – 1024 offers a good trade‑off between expressivity
    # and speed (FFT of size 2048 fits comfortably in memory).
    # ------------------------------------------------------------------
    n = 1024

    # ------------------------------------------------------------------
    # Helper to compute the exact objective (2·n·max(conv)) for a vector.
    # ------------------------------------------------------------------
    def exact_objective(vec):
        return 2.0 * n * np.max(_full_convolution(vec))

    # ------------------------------------------------------------------
    # Initial candidate – try a simple “beta‑shaped’’ distribution
    # (α<1 puts a little more mass near the ends, which tends to flatten
    # the convolution).  We also keep a uniform start as a baseline.
    # ------------------------------------------------------------------
    def beta_initial(alpha):
        """Return a normalized discretisation of Beta(α,α) on n points."""
        xs = (np.arange(n) + 0.5) / n  # mid‑point rule
        # avoid exact 0/1 when α<1 by adding a tiny epsilon
        eps = 1e-12
        xs = np.clip(xs, eps, 1 - eps)
        pdf = (xs ** (alpha - 1)) * ((1 - xs) ** (alpha - 1))
        pdf /= pdf.sum()
        return pdf

    # Small collection of hand‑crafted starters
    starter_alphas = [0.8, 0.6, 0.4, 0.2, 1.0]  # 1.0 == uniform
    starters = []
    for a in starter_alphas:
        starters.append(beta_initial(a))

    # ------------------------------------------------------------------
    # Random‑restart smoothed optimisation
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(0, -14, num=30)   # start from ε=1 → 1e‑14
    best_a = None
    best_val = np.inf
    restart_counter = 0

    while time.time() < deadline - 5.0:
        # choose a starting point (cycle through deterministic starters,
        # then fall back to a random Gaussian logit vector)
        if restart_counter < len(starters):
            a0 = starters[restart_counter].copy()
        else:
            a0 = rng.random(n)
            a0 /= a0.sum()

        # work in log‑softmax space
        w = np.log(a0 + 1e-12)

        # run the continuation on ε
        for eps in eps_schedule:
            if time.time() >= deadline - 2.0:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            # modest iteration budget – we are doing many restarts
            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 80,
                                    'ftol': 1e-11,
                                    'gtol': 1e-11,
                                    'disp': False})
            w = res.x

        # Back‑transform from logits and normalise
        a = np.exp(w - np.max(w))
        a /= a.sum()

        cur_val = exact_objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart {restart_counter}] new best = {best_val:.6f}")

        restart_counter += 1

    # ------------------------------------------------------------------
    # Aggressive hill‑climbing phase (if time permits)
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback – start from a uniform vector
        best_a = np.full(n, 1.0 / n)
        best_val = exact_objective(best_a)

    if time.time() < deadline:
        best_a, best_val = _mass_transfer_hill_climb(best_a,
                                                    best_val,
                                                    deadline,
                                                    rng,
                                                    step_start=0.25)

    # ------------------------------------------------------------------
    # Fine sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _subgradient_descent(best_a,
                                                best_val,
                                                deadline,
                                                rng)

    # ------------------------------------------------------------------
    # Final reporting – run the provided evaluator once for sanity.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
