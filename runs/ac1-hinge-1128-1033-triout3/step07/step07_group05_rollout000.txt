"""
Improved multi‑stage optimizer for the step‑function evaluation problem.

The algorithm builds on the previous hybrid approach but adds:
  * a finer ε‑schedule and more L‑BFGS restarts,
  * two deterministic re‑orderings (ends‑large and alternating‑extreme),
  * a multi‑peak sub‑gradient descent that averages the gradients from the
    top‑k convolution entries (instead of only the absolute maximum),
  * a richer simulated‑annealing that mixes swaps with small mass‑transfer moves,
  * a final hill‑climbing stage with adaptive step sizes.

All heavy work uses a fast FFT based convolution (`_eval_fast`).  The
function ``propose_candidate`` respects the 1000 s wall‑time budget and
returns the best sequence found (as a plain Python list) before the
deadline expires.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global knobs used by the smooth L‑BFGS stage
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter
_GLOBAL_N = None     # sequence length


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Numerical noise can produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` (no safety checks).
    The sequence is assumed to be non‑negative and already normalised.
    """
    n = a.shape[0]
    max_b = np.max(_fft_convolve_self(a))
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smooth surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective and its gradient w.r.t. the soft‑max parameters w.

    The surrogate is:   L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)
    where conv = a*a, a = softmax(w) (∑a = 1, a ≥ 0).
    """
    # --- soft‑max ---
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= np.sum(a)                     # Σa = 1, a ≥ 0

    # --- convolution ---
    conv = _fft_convolve_self(a)       # length 2·n‑1
    max_c = conv.max()

    # --- smooth max ---
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # --- gradient wrt a ---
    weights = np.exp(shifted)          # exp((c−max)/ε)
    p = weights / np.sum(weights)      # probability distribution over k

    # convolution a * reversed(p) gives ∂L/∂a up to scaling
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)      # length 2·n‑1
    segment = conv_ap[n - 1:2 * n - 1]  # central n entries
    grad_a = 4.0 * n * segment[::-1]   # ∂L/∂a

    # --- chain rule through soft‑max ---
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# Bind convolution for speed
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Deterministic re‑ordering heuristics
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values at the two ends, then descending towards the centre."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        # large
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        # small
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


def _alternating_extreme_reorder(a: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, third → next left, …."""
    n = a.shape[0]
    vals_desc = np.sort(a)[::-1]      # descending
    new_a = np.empty_like(a)
    lo, hi = 0, n - 1
    for idx, v in enumerate(vals_desc):
        if idx % 2 == 0:
            new_a[lo] = v
            lo += 1
        else:
            new_a[hi] = v
            hi -= 1
    return new_a


# ----------------------------------------------------------------------
# Multi‑peak sub‑gradient descent (top‑k peaks)
# ----------------------------------------------------------------------
def _subgradient_topk(a: np.ndarray, deadline: float, top_k: int = 5) -> (np.ndarray, float):
    """
    Sub‑gradient descent that averages the gradients from the top_k
    convolution entries (instead of only the absolute maximum).
    """
    n = a.shape[0]
    cur = a.copy()
    cur_val = _eval_fast(cur)
    best = cur.copy()
    best_val = cur_val

    step0 = 0.12
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur)
        # find indices of the k largest convolution coefficients
        if top_k >= len(conv):
            idxs = np.arange(len(conv))
        else:
            idxs = np.argpartition(-conv, np.arange(top_k))[:top_k]

        # average gradient over the selected peaks
        grad = np.zeros_like(cur)
        for k in idxs:
            # gradient of conv_k w.r.t. a_i is 2 * a_{k-i} (if index in range)
            idx_arr = k - np.arange(n)
            mask = (idx_arr >= 0) & (idx_arr < n)
            grad[mask] += 2.0 * cur[idx_arr[mask]]
        grad /= len(idxs)

        # step size diminishing like 1/sqrt(it)
        alpha = step0 / np.sqrt(it + 1.0)

        new = cur - alpha * grad
        new = np.maximum(new, 0.0)
        s = np.sum(new)
        if s < 1e-12:                         # avoid degenerate zero vector
            new = np.full(n, 1.0 / n)
            s = 1.0
        new /= s                               # project onto ℓ₁‑simplex

        new_val = _eval_fast(new)

        if new_val < best_val:
            best_val = new_val
            best = new.copy()
            print(f"[subgrad‑topk] iter {it} new best = {best_val:.6f}")

        # accept improving steps (monotone descent)
        if new_val < cur_val:
            cur = new
            cur_val = new_val

        it += 1
    return best, best_val


# ----------------------------------------------------------------------
# Simulated annealing (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Fast SA that mixes index swaps with small mass‑transfer moves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    cur = a.copy()
    cur_val = cur_val

    # initial temperature as a modest fraction of the current value
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99995

    it = 0
    while time.time() < deadline:
        if rng.random() < 0.45:                     # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur[i], cur[j] = cur[j], cur[i]
        else:                                        # mass‑transfer move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] <= 0.0:
                continue
            # transfer a random fraction (0–20%) of mass from i to j
            delta = cur[i] * 0.2 * rng.random()
            cur[i] -= delta
            cur[j] += delta

        new_val = _eval_fast(cur)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            # revert the move
            if rng.random() < 0.45:
                cur[i], cur[j] = cur[j], cur[i]
            else:
                cur[i] += delta
                cur[j] -= delta

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Final greedy hill‑climbing (small tweaks)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, cur_val: float,
                deadline: float) -> (np.ndarray, float):
    cur = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        # 70 % swaps, 30 % tiny mass transfers
        if rng.random() < 0.7:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] <= 0.0:
                continue
            delta = cur[i] * 0.05 * rng.random()
            new = cur.copy()
            new[i] -= delta
            new[j] += delta

        new_val = _eval_fast(new)
        if new_val < cur_val - 1e-12:
            cur = new
            cur_val = new_val
            print(f"[hill‑climb] new best = {cur_val:.6f}")

    return cur, cur_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start + budget_s - 2.0
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # ------------------------------------------------------------------
    # Problem size (use 4000 – proven to be a good trade‑off)
    # ------------------------------------------------------------------
    n = 4000
    global _GLOBAL_N
    _GLOBAL_N = n

    # ------------------------------------------------------------------
    # Initialise from a previous construction if available, otherwise uniform
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the required size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= np.sum(a0)                     # ℓ₁‑normalise (sum = 1)

    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Smooth surrogate L‑BFGS with many restarts & a finer ε‑schedule
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=22)

    restart = 0
    while time.time() < deadline - 350.0:         # reserve ~6 min for later phases
        restart += 1
        print(f"[restart {restart}]")

        w = np.log(best_a + 1e-12) + 0.5 * np.random.randn(n)
