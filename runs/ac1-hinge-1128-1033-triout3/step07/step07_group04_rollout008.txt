"""
Hybrid optimiser for the step‑function autocorrelation problem.

Algorithm summary
-----------------
1. **Initial seeding** – a collection of handcrafted families
   (uniform, geometric, power‑law, sinusoidal) together with random
   Dirichlet draws are evaluated; the best is kept.

2. **Differential Evolution (DE)** – a simple population‑based optimiser
   works on the simplex of non‑negative vectors (softmax‑normalised).
   DE explores the high‑dimensional space far more globally than the
   previous L‑BFGS‐only approach.

3. **Fine‑grained L‑BFGS** – the best DE individual is refined using the
   smooth log‑sum‑exp surrogate of the max‑convolution (same surrogate as
   the previous implementation).  A schedule of decreasing smoothing
   parameters drives the solution towards the true objective.

4. **Deterministic reorderings** – after each L‑BFGS pass the vector is
   reordered with a few simple heuristics (alternating, anti‑symmetric,
   descending).  Since the convolution depends on the ordering, this
   yields extra improvements at virtually no cost.

5. **Local hill‑climbing** – cheap swap / mass‑transfer moves are applied
   until the time budget expires.

All heavy work uses an FFT‑based evaluator (`_eval_fast`) that reproduces
the behaviour of the supplied `evaluate_sequence`.  The function respects
the 1000 s time limit and returns the best sequence it found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smooth surrogate (soft‑max + log‑sum‑exp)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter for L‑BFGS
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # numerical noise may produce tiny negatives – clamp them
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast version of `evaluate_sequence` for a non‑negative vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – needed for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max parameters w.
    The true objective (with sum(a)=1) is 2·n·max(a*a).
    Here we use: L = 2·n·max + 2·n·eps·logsumexp((conv-max)/eps)
    """
    # soft‑max → a on the simplex, a_i ≥ 0, Σ a_i = 1
    w_shift = w - np.max(w)                      # for numerical stability
    a = np.exp(w_shift)
    a /= np.sum(a)

    # autocorrelation a*a
    conv = _fft_convolve_self(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (derived analytically)
    weights = np.exp(shifted)                    # same as softmax on conv entries
    p = weights / weights.sum()                  # probabilities over convolution indices
    # reverse p because convolution index k corresponds to i + j = k
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                # a * p_rev
    # central part of length n (indices n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                 # gradient ∂L/∂a_i

    # chain‑rule through the soft‑max parametrisation
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by `scipy.optimize.minimize`."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Simple parametric families for seeding
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """Geometric progression a_i ∝ q**i, normalised to sum = 1 (0<q<1)."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q**n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)**p, normalised to sum = 1 (p<0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _sin_sequence(exp: float, n: int) -> np.ndarray:
    """Sinusoidal shape a_i ∝ sin(π·(i+0.5)/n)**exp, normalised."""
    theta = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(theta) ** exp
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Re‑ordering heuristics (depend only on the multiset of values)
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large values left, small values right (pairwise)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = sorted_vals[n - 1 - i]      # large to the left
        out[n - 1 - i] = sorted_vals[i]      # small to the right
    if n % 2:
        out[half] = sorted_vals[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    sorted_vals = np.sort(a)
    left, right = 0, len(sorted_vals) - 1
    out = np.empty_like(a)
    idx = 0
    while left <= right:
        out[idx] = sorted_vals[right]
        idx += 1
        right -= 1
        if left <= right:
            out[idx] = sorted_vals[left]
            idx += 1
            left += 1
    return out


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Simply sort descending."""
    return np.sort(a)[::-1]


# ----------------------------------------------------------------------
# Differential Evolution optimiser (simple, works on the simplex)
# ----------------------------------------------------------------------
def _run_de(pop, fitness, best_a, best_val, deadline):
    """
    Run a classic DE/rand/1/bin loop on the simplex.
    pop      – list of np.ndarray, each sums to 1.
    fitness  – corresponding list of objective values.
    Returns updated (best_a, best_val, pop, fitness).
    """
    n = pop[0].shape[0]
    rng = np.random.default_rng()
    F_min, F_max = 0.4, 0.9        # mutation factor interval
    CR_min, CR_max = 0.7, 1.0      # crossover probability interval

    while time.time() < deadline:
        for i in range(len(pop)):
            # --- mutation ---
            idxs = rng.choice([j for j in range(len(pop)) if j != i], size=3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            F = rng.uniform(F_min, F_max)
            donor = a + F * (b - c)
            donor = np.maximum(donor, 0.0)
            dsum = donor.sum()
            if dsum == 0.0:
                donor = pop[i].copy()
            else:
                donor /= dsum

            # --- crossover (binomial) ---
            CR = rng.uniform(CR_min, CR_max)
            mask = rng.random(n) < CR
            # guarantee at least one dimension from donor
            jrand = rng.integers(0, n)
            mask[jrand] = True
            trial = np.where(mask, donor, pop[i])
            trial = np.maximum(trial, 0.0)
            tsum = trial.sum()
            if tsum == 0.0:
                trial = pop[i].copy()
            else:
                trial /= tsum

            # --- selection ---
            trial_val = _eval_fast(trial)
            if trial_val <= fitness[i]:
                pop[i] = trial
                fitness[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_a = trial.copy()
                    print(f"[DE] new best = {best_val:.6f}")

        # optional early exit if we already beat the target
        if best_val < 1.503:
            break

    return best_a, best_val, pop, fitness


# ----------------------------------------------------------------------
# Local hill‑climbing (swap + small transfers)
# ----------------------------------------------------------------------
def _hill_climb(a, cur_val, deadline):
    """Simple stochastic hill‑climbing until deadline."""
    rng = np.random.default_rng()
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val

    while time.time() < deadline:
        if rng.random() < 0.3:         # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                           # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] < 1e-12:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[Hill] new best = {cur_val:.6f}")

        # stop early if we have already reached the target
        if cur_val < 1.503:
            break

    return cur_a, cur_val
