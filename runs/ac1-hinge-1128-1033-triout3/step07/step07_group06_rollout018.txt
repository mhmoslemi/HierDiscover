def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolve_fft(x, y):
    size = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx*fy, size)[:len(x)+len(y)-1]
    # negative epsilon may appear, clip zero
    np.maximum(conv, 0, out=conv)
    return conv

def _smooth_objective_and_gradient(a, n, alpha):
    # compute conv = self conv
    c = _full_convolution(a)   # length 2n-1
    # compute smoothed objective (approx max)
    # Compute softmax of alpha * c
    max_c = np.max(c)  # for numerical stability
    exp_c = np.exp(alpha * (c - max_c))   # subtract max_c to avoid overflow
    sum_exp = np.sum(exp_c)
    p = exp_c / sum_exp
    # compute gradient:
    # reverse p
    p_rev = p[::-1]
    conv_full = _full_convolve_fft(a, p_rev)   # length 3n-2
    offset = 2 * n - 2
    grad_part = conv_full[offset : offset + n]  # length n
    grad = grad_part + a   # because sum_i p_i = 1
    # smooth objective value (approx max)
    smooth_val = max_c + np.log(sum_exp) / alpha   # from log-sum-exp
    # actual objective:
    obj = _objective(a, n)   # actual max
    return obj, smooth_val, grad

"""
search for a low‑value step‑function.

The evaluation metric is
    evaluate_sequence(a) = 2·n·max(a*a) / (sum(a)**2)
where n = len(a).  The value is invariant under a global scaling of the
vector, so we work with a probability simplex (sum = 1).  The goal is to
reduce max(a*a) – the peak of the (auto‑)convolution.

The program performs many short, independent runs (random restarts).  Each
run consists of

1.  a smoothed “log‑sum‑exp” gradient descent on the max‑convolution
    (the gradient is obtained analytically via a second FFT).  The
    smoothing parameter α is fixed (α≈10) which gives a good trade‑off
    between smoothness and fidelity to the true max.

2.  an aggressive mass‑transfer hill‑climber that repeatedly moves a
    fraction of the most damaging pair of masses to two low‑mass
    locations, accepting the move only if the exact objective improves.

The best vector found over all restarts is returned.  All helper
functions are defined at top level so that no closures are needed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using an FFT (zero‑padded)."""
    size = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: len(x) + len(y) - 1]
    # tiny negative values caused by round‑off are set to zero
    np.maximum(conv, 0, out=conv)
    return conv


def _full_convolution_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real, non‑negative)."""
    return _full_convolve_fft(a, a)


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of evaluate_sequence when sum(a)=1."""
    conv = _full_convolution_self(a)
    return float(2.0 * n * np.max(conv))


def _smooth_objective_and_grad(a: np.ndarray, n: int, alpha: float):
    """
    Smoothed version of the max‑convolution objective.
    Returns:
        - exact objective (using the true max)
        - smooth approximation (log‑sum‑exp)
        - gradient of the smooth approximation w.r.t. a
    """
    conv = _full_convolution_self(a)               # length 2n‑1
    # ----- smooth (log‑sum‑exp) -----
    c_max = np.max(conv)
    # shift for numerical stability
    exp_c = np.exp(alpha * (conv - c_max))
    sum_exp = np.sum(exp_c)
    p = exp_c / sum_exp                            # p_i ≥ 0, Σp_i = 1
    smooth_val = c_max + np.log(sum_exp) / alpha   # ≈ max(conv)

    # ----- gradient of the smooth objective -----
    # ∂/∂a_j  ( Σ_i p_i a_{i‑j} + a_j )
    # The first term is a convolution of a with the reversed p.
    p_rev = p[::-1]
    conv_full = _full_convolve_fft(a, p_rev)       # length 3n‑2
    offset = 2 * n - 2
    grad_part = conv_full[offset: offset + n]      # Σ_i p_i a_{i‑j}
    grad = grad_part + a                           # add the +a_j term
    # exact objective (for acceptance tests)
    exact_val = _objective(a, n)
    return exact_val, smooth_val, grad


# ----------------------------------------------------------------------
# Simplex projection (non‑negative + unit sum)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # standard algorithm (see Wang & Carreira‑Perpiñán 2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numeric safety – fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iters: int = 2000,
                    alpha: float = 10.0):
    """
    Gradient descent on the smoothed max‑convolution objective.
    Returns the best vector found and its exact objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n)

    lr0 = 0.5
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        exact_val, smooth_val, grad = _smooth_objective_and_grad(a, n, alpha)

        # simple back‑tracking line search
        step = lr0 / np.sqrt(it)
        improved = False
        for _ in range(6):
            a_trial = a - step * grad
            a_trial = np.clip(a_trial, 0.0, None)
            s = a_trial.sum()
            if s == 0.0:
                step *= 0.5
                continue
            a_trial = a_trial / s                 # simplex projection
            val_trial = _objective(a_trial, n)
            if val_trial < best_val:
                a = a_trial
                best_a = a_trial.copy()
                best_val = val_trial
                improved = True
                lr0 = min(lr0 * 1.02, 2.0)
                break
            else:
                step *= 0.5

        if not improved:
            lr0 = max(lr0 * 0.95, 1e-4)

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill‑climber
# ----------------------------------------------------------------------
def _mass_transfer_hillclimb(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            rng: np.random.Generator,
                            max_no_improve: int = 3000):
    """
    Hill climbing that repeatedly moves a fraction of the most damaging
    pair of masses to two low‑mass positions.
    """
    a = a0.copy()
    cur_val = _objective(a, n)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution_self(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices participating in the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we want to relocate
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # two destination positions with low mass
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue
        new_a /= s

        new_val = _objective(new_a, n)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, concentration: float) -> np.ndarray:
    """Draw a probability vector from a Dirichlet with given concentration."""
    alpha = np.full(n, concentration, dtype=float)
    v = rng.dirichlet(alpha)
    # safeguard against pathological zero‑vectors
    if v.sum() == 0.0:
        v = np.full(n, 1.0 / n)
    return v

    """
    Return a list of non‑negative heights that (approximately) minimise

    The routine performs several random restarts, each consisting of a
    smoothed gradient descent followed by an aggressive hill‑climber.
    The best solution found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Determine the problem size (n).  Prefer a previously found vector
    # if available, otherwise fall back to a default length.
    # ------------------------------------------------------------------
    default_n = 8000
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        a_initial = np.maximum(old, 0.0)
        if a_initial.sum() == 0.0:
            a_initial = np.full(n, 1.0 / n)
        else:
            a_initial = a_initial / a_initial.sum()
        print(f"[init] using previously stored sequence, n={n}")
    except Exception:
        n = default_n
        a_initial = np.full(n, 1.0 / n)
        print(f"[init] no previous sequence – using uniform, n={n}")

    # ------------------------------------------------------------------
    # Keep track of the global best solution.
    # ------------------------------------------------------------------
    best_a = a_initial.copy()
    best_val = _objective(best_a, n)
    print(f"[init] initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – a few deterministic starts (uniform + previous)
    # ------------------------------------------------------------------
    a_tmp, val_tmp = _smooth_descent(best_a, n, deadline, rng,
                                     max_iters=2500, alpha=8.0)
    a_tmp, val_tmp = _mass_transfer_hillclimb(a_tmp, n, deadline, rng)
    if val_tmp < best_val:
        best_val, best_a = val_tmp, a_tmp.copy()
        print(f"[phase1] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – random restarts until time runs out.
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 2.0:          # keep a tiny buffer

        conc = rng.uniform(0.15, 2.5)

    # ------------------------------------------------------------------

        print(f"[final] polished best = {best_val:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    print(f"[result] best objective = {best_val:.6f} with n={n}")
    return best_a.tolist()
