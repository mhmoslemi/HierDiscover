import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smoothed objective (soft‑max parameters).
# ----------------------------------------------------------------------
_GLOBAL_EPS = None  # smoothing parameter ε
_GLOBAL_N = None    # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a ∗ a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)        # clip tiny negatives
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # --------- soft‑max ----------
    w_shift = w - np.max(w)          # numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # --------- convolution ----------
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # --------- smooth max ----------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # --------- gradient w.r.t. a ----------
    weights = np.exp(shifted)
    p = weights / weights.sum()      # distribution over lag indices

    # convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment that corresponds to the derivative
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]    # reverse to align with a_i order

    # --------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the formula simplifies).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places large values on one side and small values on the
    opposite side – often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]       # large → left side
        new_a[n - 1 - i] = sorted_vals[i]       # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %). The vector is kept on the
    simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine problem size.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000                     # default size
        has_old = False

    # ------------------------------------------------------------------
    # Generate diverse initial candidates.
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # random Dirichlet‑like draws
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    init_candidates.append(rng.gamma(shape=2.0, scale=1.0, size=n))
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoid
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # normalise onto the simplex
    init_candidates = [c / c.sum() for c in init_candidates]

    # ------------------------------------------------------------------
    # Pick the best cheap initial candidate.
    # ------------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on a smooth surrogate of the max‑convolution.
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=15)
    w = np.log(best_a + 1e-12)           # initialise soft‑max parameters

    max_restarts = 15
    restart = 0
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        if restart % 5 == 0:
            w = np.random.randn(n) * 2.0   # fresh random start

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L‑BFGS‑B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            # convert back to probabilities via soft‑max
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # occasional anti‑symmetric reorder
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent on the exact max‑convolution.
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    alpha0 = 0.7
    min_alpha = 0.005
    while time.time() < deadline - 30.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # gradient of max(conv) w.r.t. a:
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]                     # g_i = a_{k*-i}

        # exponential mirror‑descent update
        alpha = max(alpha0 * (0.9999 ** (it // 250)), min_alpha)
        a = a * np.exp(-alpha * g)

        # project onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 200 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[mirror it {it}] new best = {best_val:.6f}")
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[mirror reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – Linear‑programming (Frank‑Wolfe) improvement.
    # ------------------------------------------------------------------
    lp_iters = 0
    max_lp_iters = 8
    while time.time() < deadline - 20.0 and lp_iters < max_lp_iters:
        f = best_a.copy()
        conv_f = _full_convolution(f)
        M = conv_f.max()
        m = 2 * n - 1

        # Build the constraint matrix A_ub where (A_ub @ g) = f * g.
        # Column j of A_ub is f shifted by j positions.
        A_ub = np.zeros((m, n), dtype=np.float64)
        for j in range(n):
            A_ub[j:j + n, j] = f

        b_ub = np.full(m, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)     # maximise sum(g)  <=> minimise -sum(g)

        # Solve LP (HiGHS interior‑point / simplex)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=(0, None), method='highs',
                          options={'presolve': True})
        except Exception as exc:
            print(f"[LP] linprog exception: {exc}")
            break

        if not res.success:
            print(f"[LP] solver failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print("[LP] zero mass returned")
            break

        # Normalise g to have unit sum, then mix with current f.
        g_norm = g / sum_g
        t_mix = 0.3
        new_f = (1 - t_mix) * f + t_mix * g_norm
        new_f = np.maximum(new_f, 0.0)
        new_f /= new_f.sum()

        new_val = _fast_evaluate(new_f)
        if new_val < best_val:
            best_val = new_val
            best_a = new_f.copy()
            print(f"[LP] new best = {best_val:.6f}")

        lp_iters += 1

    # ------------------------------------------------------------------
    # Phase 4 – Targeted simulated‑annealing (focus on the dominant peak).
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.02
    cooling = 0.99995
    rng = np.random.default_rng(seed + 9999)

    while time.time() < deadline - 5.0:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        # indices contributing to the peak
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        contrib = cur_a[idx] * cur_a[jdx]
        worst_pair = np.argmax(contrib)
        i0 = idx[worst_pair]
        j0 = jdx[worst_pair]

        # move a small fraction of mass away from the larger element of the pair
        move_from = i0 if cur_a[i0] >= cur_a[j0] else j0
        move_to = rng.integers(0, n)
        if move_from == move_to:
            continue
        delta = cur_a[move_from] * 0.1 * rng.random()
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[move_from] -= delta
        new_a[move_to] += delta

        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA target] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # Phase 5 – final cheap hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
