"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm (≈1000 s time budget)
--------------------------------
1. **Initialisation** – start from the previously discovered construction
   (``height_sequence_1``) if it exists, otherwise from a uniform or
   exponential random vector.  The vector is normalised to have unit sum
   (the objective is scale‑invariant).

2. **Smooth surrogate optimisation** – minimise a smooth approximation of
   the objective (log‑sum‑exp of the self‑convolution) via L‑BFGS‑B.
   The smoothing parameter ε is decreased geometrically; a few random
   restarts increase robustness.

3. **Active‑set exponential descent** – repeatedly compute the full
   self‑convolution, select all indices whose value is within a small
   fraction of the current maximum, and perform a mirror‑descent
   update ``a_i ← a_i·exp(−α·g_i)`` where ``g_i`` is the sum of the
   coefficients that contribute to the active peaks.  The gradient is
   obtained efficiently by a single FFT‑based correlation.  The step
   size α decays slowly, and every few hundred iterations an
   anti‑symmetric re‑ordering (large values on the left, small on the
   right) is tried.

4. **Final L‑BFGS refinement** – a short L‑BFGS run from the best
   candidate obtained in step 3 to polish the solution.

5. **Hill‑climbing** – simple local search (random swaps / mass
   transfers) that accepts only strict improvements.

All updates project the vector onto the probability simplex (non‑negative,
unit sum).  The routine respects the 1000 s limit by checking the
deadline before each major phase.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the surrogate
_GLOBAL_N = None     # current problem size (used by the surrogate)


def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length is 2·len(x)‑1.  Uses FFT for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # tiny negative values caused by rounding are forced to zero
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Re‑order the vector so that large entries appear on the left side and
    small entries on the right side (the “anti‑symmetric” heuristic).
    """
    n = a.shape[0]
    s = np.sort(a)               # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[n - 1 - i]          # large → left
        b[n - 1 - i] = s[i]          # small → right
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns the smoothed surrogate ``L_ε`` and its gradient w.r.t. the
    soft‑max parameters ``w``.

    a_i = exp(w_i) / Σ_j exp(w_j)      (guaranteed non‑negative, Σ a = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # stable soft‑max
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # self‑convolution
    conv = _full_convolution(a)
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (derived in the original code)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # distribution over lags

    # derivative of max term:
    #   grad_a = 4·n·(conv * p)_reversed  (see original implementation)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    seg = conv_ap[n - 1: 2 * n - 1]                 # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):

    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair found before ``deadline``.

    w = w_init.copy()
    # ε schedule: from 5·10⁻³ down to 10⁻¹² (more aggressive than before)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        # L‑BFGS‑B with a generous max‑iteration budget
        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 800,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to a (probability simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:

def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random local search: swaps (20 %) and mass transfers (80 %).
    Keeps only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – a few random vectors (exploration)
    # ------------------------------------------------------------------
    for tr in range(4):
        if time.time() >= deadline - 200:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate optimisation (L‑BFGS) with several restarts
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 150 and restart < 8:
        restart += 1
        w_start = np.log(best_a + 1e-12)
        w_start += rng.normal(scale=0.2, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w_start, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[restart {restart}] new best after L‑BFGS = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – active‑set exponential gradient descent (core refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        a_gd, val_gd = _gradient_descent_active_set(best_a, deadline)
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[active‑set GD] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – a final quick L‑BFGS polish (small ε, few iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20:
        w_start = np.log(best_a + 1e-12)
        a_opt, val_opt = _l_bfgs_optimize_from_w(w_start, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[final L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
