def solve_lp(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    # for each lag k in 0..2n-2, fill entries
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # i in [i_start, i_end]; define j = k - i
        # for each i, column j index is j
        # coefficient = f[i]
        # So for each i, add entry at (row=k, col=j)
        # Where j = k - i
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # b = np.full(2*n-1, M)
    # Maximize sum(g) => minimize -sum(g)
    c = np.full(n, -1.0)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback to returning zero vector?
        return np.zeros_like(f)
    return res.x

"""
Propose a step‑function (list of non‑negative heights) that minimises
evaluate_sequence().  The algorithm consists of three phases:

1.  Projected sub‑gradient descent on the exact objective
    f(a)=2·n·max((a*a)) with a projected onto the probability simplex.

2.  Aggressive mass‑transfer hill‑climbing (the original “massclimb”
    routine) that exploits the pair of indices that contribute most to
    the current maximal lag.

3.  A simple simulated‑annealing stage.  Randomly move a fraction of
    mass from a donor index to a recipient index; accept moves that
    improve the objective or, with a temperature‑dependent probability,
    worsen it.  This helps escape the local minima reached by the
    deterministic phases.

The whole search respects a 1000 s wall‑clock budget, continuously
checking the deadline.  The best sequence found before the deadline is
returned as a plain Python list.  All helper functions are top‑level and
do not capture external state.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions for convolution and objective evaluation
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (used for zero‑padded FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # eliminate tiny negative values caused by floating point noise
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry conv[k] w.r.t. a.
    For non‑negative a the derivative is 2·a_{k‑i} on the feasible range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad


# ----------------------------------------------------------------------
# Phase 1 – Projected sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    step0 = 0.5      # initial multiplier for the learning rate
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _full_convolution(a)
        maxb = conv.max()
        # pick a lag attaining the maximum (break ties randomly)
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        grad = _gradient_of_lag(a, k)

        # simple back‑tracking line search
        lr = step0 / np.sqrt(t)
        improved = False
        for _ in range(10):
            a_trial = a - lr * grad
            a_trial = np.clip(a_trial, 0.0, None)
            s = a_trial.sum()
            if s == 0.0:
                lr *= 0.5
                continue
            a_trial /= s            # projection onto the simplex
            val = _objective(a_trial)
            if val < best_val:
                a = a_trial
                best_a = a_trial.copy()
                best_val = val
                improved = True
                step0 = min(step0 * 1.02, 2.0)
                if t % 200 == 0:
                    print(f"[subgrad] iter {t:5d}  val = {best_val:.6f}")
                break
            else:
                lr *= 0.5

        if not improved:
            step0 = max(step0 * 0.95, 1e-4)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Aggressive mass‑transfer hill‑climbing
# ----------------------------------------------------------------------


def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 8000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of the most damaging mass to low‑mass
    positions.  Stops when the time budget expires or when a given
    number of consecutive unsuccessful attempts is reached.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2    # initial fraction of mass to relocate
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that make up the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we are willing to shift
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # pick two distinct low‑mass destinations, avoiding the donors
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Simple simulated annealing
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Randomly relocate a fraction of mass and accept moves according to a
    temperature schedule (exponential cooling).  Keeps the best vector
    seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_obj = _objective(a)
    best_a = a.copy()
    best_obj = cur_obj

    start_temp = 0.1
    end_temp = 1e-6
    start_time = time.time()
    total = max(deadline - start_time, 1e-9)        # guard against zero

    # pre‑allocate FFT size for speed
    size = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        fraction = elapsed / total
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** fraction

        # ----- propose a random move -----
        i = rng.integers(n)
        if a[i] < 1e-14:
            continue
        # choose a distinct recipient
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 50 %) of donor's mass
        frac = rng.random() * 0.5
        delta = a[i] * frac
        if delta <= 1e-14:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # ----- evaluate the new objective -----
        fa = np.fft.rfft(a_new, size)
        conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
        np.maximum(conv, 0, out=conv)
        new_obj = 2.0 * n * np.max(conv)

        d = new_obj - cur_obj
        # Metropolis acceptance
        if d < 0 or rng.random() < math.exp(-d / max(T, 1e-12)):
            a = a_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                print(f"[SA] new best = {best_obj:.6f}")

    return best_a, best_obj

    # 2. Aggressive mass‑transfer hill‑climbing
    # --------------------------------------------------------------
    a_hc, val_hc = _mass_transfer_hillclimb(a_sub, deadline, rng, max_no_improve=8000)
    print(f"[massclimb] best value after hill‑climb = {val_hc:.6f}")

    # --------------------------------------------------------------
    # 3. Simulated annealing (global exploration)
    # --------------------------------------------------------------
    a_final, val_final = _simulated_annealing(a_hc, deadline, rng)
    print(f"[SA] final best value = {val_final:.6f}")

    # --------------------------------------------------------------
    # 4. Report the score using the provided evaluator.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_final.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a_final.tolist()
