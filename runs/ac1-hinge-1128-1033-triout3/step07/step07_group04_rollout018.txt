def _fft_convolve_self(a):
    n = a.shape[0]
    out_len = 2*n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]

    return conv

    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:

def _smooth_obj_and_grad(w):

"""
Hybrid optimiser for the step‑function convolution problem.

The method consists of the following stages:

1. **Diverse seed generation** – a handful of analytically‑motivated families
   (geometric, power‑law, edge‑heavy symmetric shapes) together with random
   Dirichlet draws.  The best seed is kept.

2. **Linear‑programming mass‑expansion** – for the current vector `f` we compute
   its maximal self‑convolution `M`.  Keeping `M` fixed we solve the LP

        maximise   Σ g_j
        subject to A_f·g ≤ M,  g ≥ 0

   where `A_f` is the Toeplitz matrix that maps a vector `g` to the
   convolution `f * g`.  The solution typically has a larger ℓ₁‑norm than `f`;
   after normalising it to unit sum we blend it with `f` (parameter `t`).
   Repeating this step drives the maximal convolution down towards its
   theoretical lower bound 1/(2n).

3. **Refinement with a smooth log‑sum‑exp surrogate** – the max can be
   approximated by a log‑sum‑exp.  We minimise the surrogate using L‑BFGS‑B
   on the soft‑max parametrisation of the vector (the gradient is available in
   closed form).  A schedule of decreasing smoothing parameters `ε` yields a
   high‑quality local optimum.

4. **Final hill‑climbing** – cheap swaps and small mass transfers are applied
   until the deadline is reached.

All stages respect the 1000 s budget; the algorithm works for any length,
here we use `n = 2000` (smaller than the previous 4000 to keep the LP cheap)
but the final evaluation is independent of `n`.  The function returns a plain
Python list, ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global values used by the smooth surrogate (soft‑max + log‑sum‑exp)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)      # eliminate tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast version of `evaluate_sequence` for a non‑negative vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _make_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix `A` such that (A @ g) = f * g  (full linear convolution).
    Shape of the matrix is (2·n‑1) × n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient w.r.t. soft‑max params
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray):
    """
    Returns the smoothed objective L_ε and its gradient w.r.t. the
    soft‑max (log‑probability) parameters `w`.
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # soft‑max parametrisation (positivity + ℓ₁ = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # autocorrelation a*a
    conv = _fft_convolve_self(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed‑form)
    weights = np.exp(shifted)
    p = weights / weights.sum()
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)          # a * p_rev
    seg = conv_ap[n - 1: 2 * n - 1]        # central part, length n
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by `scipy.optimize.minimize`. """
    return _smooth_obj_and_grad(w)


# ----------------------------------------------------------------------
# Parametric families for seeding
# ----------------------------------------------------------------------
def _geometric_seq(q: float, n: int) -> np.ndarray:
    """a_i ∝ q**i, normalised to sum = 1 (0 < q < 1)."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q**n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_seq(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)**p, normalised to sum = 1 (p < 0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _sym_exp(c: float, n: int) -> np.ndarray:
    """Symmetric exponential: larger values toward the edges."""
    d = np.abs(np.arange(n) - (n - 1) / 2.0)
    a = np.exp(c * d)
    a /= a.sum()
    return a


def _sym_power(p: float, n: int) -> np.ndarray:
    """Symmetric power law: a_i ∝ ((i+1)*(n-i))**p, p < 0."""
    idx = np.arange(1, n + 1, dtype=float)
    a = (idx * (n + 1 - idx)) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Simple hill‑climbing for final polishing
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                 # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    The implementation follows the plan outlined in the module docstring.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # problem size (2000 gives a good trade‑off between quality and speed)
    # --------------------------------------------------------------
    n = 2000
    global _GLOBAL_N
    _GLOBAL_N = n

    # --------------------------------------------------------------
    # 1. seed generation
    # --------------------------------------------------------------
    seed_vectors = []

    # uniform baseline
    seed_vectors.append(np.full(n, 1.0 / n))

    # geometric decays
    for q in np.linspace(0.6, 0.98, 10):
        seed_vectors.append(_geometric_seq(q, n))

    # reverse geometric (mass near the right edge)
    for q in np.linspace(0.6, 0.98, 8):
        a = q ** np.arange(n - 1, -1, -1, dtype=float)
        a /= a.sum()
        seed_vectors.append(a)

    # power‑law tails
    for p in np.linspace(-3.0, -0.3, 12):
        seed_vectors.append(_power_seq(p, n))

    # symmetric exponential (edges heavy)
    for c in np.linspace(0.0, 0.025, 8):
        seed_vectors.append(_sym_exp(c, n))

    # symmetric power law (edges heavy)
    for p in np.linspace(-2.5, -0.2, 10):
        seed_vectors.append(_sym_power(p, n))

    # random Dirichlet‐like draws
    for _ in range(30):
        a = np.random.rand(n)
        a /= a.sum()
        seed_vectors.append(a)

    best_a = None
    best_val = float('inf')
    for a in seed_vectors:
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    print(f"[seed] best after seeding = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Linear‑programming mass‑expansion phase
    # --------------------------------------------------------------
    # This phase repeatedly expands the ℓ₁‑mass while keeping the current
    # maximal self‑convolution bound `M` fixed, then blends the result.
    blend_factor = 0.6
    while time.time() < deadline - 30.0:
        conv = _fft_convolve_self(best_a)
        M = conv.max()

        # Build Toeplitz matrix A_f (dense, but modest size for n=2000)
        A = _make_conv_matrix(best_a)

        # Linear program: maximise Σ g_j  ⇔ minimise  -Σ g_j
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print("[LP] failure:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        # If the LP cannot increase the total mass we have reached a fixed point.
        if sum_g <= 1.0 + 1e-12:
            print("[LP] no further mass increase – ending LP phase")
            break

        # Normalise g and blend
        g /= sum_g
        best_a = (1.0 - blend_factor) * best_a + blend_factor * g
        best_a /= best_a.sum()

        cur_val = _eval_fast(best_a)
        if cur_val < best_val:
            best_val = cur_val
            print(f"[LP] new best = {best_val:.6f}")

        blend_factor = max(blend_factor * 0.85, 0.05)   # decay

        # early exit if time is running out
        if time.time() > deadline - 30.0:
            break

    # --------------------------------------------------------------
    # 3. Refined optimisation with smooth surrogate (L‑BFGS‑B)
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        # soft‑max parameters initialise from the current best
        w = np.log(best_a + 1e-16)

        # a fine schedule of smoothing parameters
        eps_schedule = np.logspace(-3, -12, num=12)
        for eps in eps_schedule:
            if time.time() >= deadline - 5.0:
                break
            _GLOBAL_EPS = eps
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': 200,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            a = np.exp(w - np.max(w))
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[smooth] eps {eps:.1e} new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climb")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # 5. Exact evaluation with the provided function
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
