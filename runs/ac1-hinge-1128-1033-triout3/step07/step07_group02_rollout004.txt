def _project_simplex(v):
    # ensure sum(v) = 1 and all entries >=0.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _beta_shape(p,q, n):
    i = np.arange(1, n+1, dtype=np.float64)
    a = (i ** (p-1)) * ((n - i + 1) ** (q-1))
    if a.sum()==0:
        return None
    a /= a.sum()
    return a
```

Now mixture:

```
def _mix_beta(p1,q1, p2,q2, lam, n):
    a1 = _beta_shape(p1,q1,n)
    a2 = _beta_shape(p2,q2,n)
    if a1 is None or a2 is None:
        return None
    a = lam * a1 + (1-lam) * a2
    a_sum = a.sum()
    if a_sum ==0:
        return None
    a /= a_sum
    return a
```

"""
Propose a non‑negative step function that (approximately) minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The objective is scale invariant, therefore we keep sum(a)=1 throughout.
The search consists of four stages fitting into the 1000 s budget:

0. *Random Beta‑mix search* – sample single or two‑component Beta‑shapes
   (a_i ∝ (i+1)^{p‑1} (n‑i)^{q‑1}) on a log‑scale, keep the best.
1. *Adam on a smooth‑max surrogate* – optimisation on the simplex via the
   log‑softmax re‑parameterisation.  The smooth‑max uses a temperature that
   exponentially cools, while the learning‑rate follows a similar schedule.
2. *Exact sub‑gradient descent* – the gradient of the exact max‑term is
   simply a shifted copy of the current vector; we perform a projected
   sub‑gradient step on the simplex with a diminishing step size.
3. *Fine‑grained simulated‑annealing hill‑climbing* – the same routine
   used in the reference solution, started from the current best vector.

Every stage respects its own deadline; the best vector seen so far is
carried forward.  At the end we call the provided ``evaluate_sequence``
once (with a small safety margin) and return the best list found.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = rfft(x, size)
    conv = irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: beta‑shape (single component)
# ----------------------------------------------------------------------
def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta‑shaped vector (or None if degenerate)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # powers may be negative → possible infinities; guard against that
    with np.errstate(divide='ignore', invalid='ignore'):
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    if np.any(np.isnan(a)) or np.all(a == 0):
        return None
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth‑max approximation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation of the smooth‑max surrogate."""
    w = np.log(a0 + 1e-12)                # logits for the soft‑max

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.001
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 25000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedule for temperature (eps) and learning rate (lr)
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max & gradient
        # ---- compute soft‑max approximation of max(conv) ----
        b = _full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()                     # length 2n‑1

        # gradient of weighted sum w.r.t. a (as derived in the reference)
        p_rev = p[::-1]
        conv_ab = np.fft.irfft(rfft(a, (len(p_rev) + len(a) - 1).bit_length()) *
                               rfft(p_rev, (len(p_rev) + len(a) - 1).bit_length()))
        conv_ab = conv_ab[: 2 * n - 1]                   # limited to 2n‑1
        segment = conv_ab[n - 1:2 * n - 1]                # n entries
        grad_a = 4.0 * n * segment[::-1]                # factor 4n from 2·n·max

        # chain rule through the soft‑max (w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        # restart after too many idle steps
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑conv) on the simplex
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_a: np.ndarray,
                         best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    t = 0
    eta0 = 0.12                         # initial step size – empirically good

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[Subgrad {t:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        k_max = int(np.argmax(b))

        # sub‑gradient of 2·n·max(conv) w.r.t. a:
        #  ∂/∂a_i  = 4·n·a_{k_max‑i}  (if index in range)
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] = 4.0 * n * a[k_max - idx]

        # diminishing step size
        eta = eta0 / np.sqrt(t + 1.0)

        # gradient step & projection onto simplex
        a_tmp = a - eta * grad
        a = _project_simplex(a_tmp)

        t += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Local search that moves small amounts of mass between indices."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.012, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose source index (biased by contribution)
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # choose target index (biased towards low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback uniform among all except i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta‑mix search (phase 0)
# ----------------------------------------------------------------------
def _random_beta_mix_search(n: int,
                            deadline: float,
                            best_a: np.ndarray,
                            best_val: float) -> tuple[np.ndarray, float]:
    """Explore single‑ and two‑component Beta families."""
    LOG_MIN = np.log10(0.01)
    LOG_MAX = np.log10(10.0)

    while True:
        now = time.time()
        if now >= deadline:
            break

        if np.random.rand() < 0.55:               # single component
            p = 10 ** np.random.uniform(LOG_MIN, LOG_MAX)
            q = 10 ** np.random.uniform(LOG_MIN, LOG_MAX)
            a = _beta_shape(p, q, n)
        else:                                      # mixture of two components
            p1 = 10 ** np.random.uniform(LOG_MIN, LOG_MAX)
            q1 = 10 ** np.random.uniform(LOG_MIN, LOG_MAX)
            p2 = 10 ** np.random.uniform(LOG_MIN, LOG_MAX)
            q2 = 10 ** np.random.uniform(LOG_MIN, LOG_MAX)
            lam = np.random.rand()
            a1 = _beta_shape(p1, q1, n)
            a2 = _beta_shape(p2, q2, n)
            if a1 is None or a2 is None:
                continue
            a = lam * a1 + (1.0 - lam) * a2
            s = a.sum()
            if s == 0.0:
                continue
            a /= s

        if a is None:
            continue

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[RandomMix] new best = {best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,

    """

    Returns a list (sum≈1) that (approximately) minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    safety = 0.5                                 # keep a margin for final call
    deadline = start_time + budget_s - safety

    # ------------------------------------------------------------------
    # Problem size – 5 000 gives a solid trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Initialise from a previous run if possible, otherwise random.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[Init] interpolated previous solution")
    except Exception:
        a0 = np.random.rand(n)
        a0 /= a0.sum()
        print("[Init] random start")

    best_a = a0.copy()
    best_val = _objective(best_a)
    print(f"[Init] initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – random exploration of Beta‑mix families (~15 % of budget)
    # ------------------------------------------------------------------
    phase0_deadline = start_time + 0.15 * budget_s
    best_a, best_val = _random_beta_mix_search(n,
                                                deadline=phase0_deadline,
                                                best_a=best_a,
                                                best_val=best_val)

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on smooth‑max (~30 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase1_deadline,

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.65 * budget_s

                                            deadline=phase2_deadline,

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
