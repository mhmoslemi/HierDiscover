"""
A fast, multi‑phase optimizer for the step‑function problem.

The algorithm builds a diverse pool of initial candidates (uniform,
random Dirichlet draws, geometric / power‑law decays, sinusoid,
anti‑symmetric reorder).  The best of these seeds drives three
gradient‑based refinements:

1.  Adaptive *mirror descent* (multiplicative KL‑update) with an
    automatic step‑size schedule that expands on acceptance and
    contracts on rejection.

2.  Projected *additive* sub‑gradient descent with a diminishing step
    (classic sub‑gradient method).

3.  A *guided simulated‑annealing* phase that draws mass‑transfer moves
    from the current sub‑gradient (large gradient → small gradient)
    together with occasional random swaps.

Each phase runs until the remaining time budget is exhausted.
The final solution is polished by a cheap hill‑climbing pass.
All operations keep the vector non‑negative and on the probability
simplex, so the objective `evaluate_sequence` is invariant to scaling.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT convolution, simplex projection, fast evaluation)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Eliminate tiny negatives from round‑off.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that sum to 1.
    Returns 2·n·max(conv) because the denominator Σ a_i = 1.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places large values on one side and small values on the
    opposite side – often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %). The vector is kept on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _conv_and_max(a: np.ndarray) -> tuple[np.ndarray, int, float]:
    """Return (full convolution, index of the max element, max value)."""
    conv = _full_convolution(a)
    max_idx = int(conv.argmax())
    max_val = conv[max_idx]
    return conv, max_idx, max_val


def _subgradient_of_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    For the maximum at index k*, the gradient is the shifted copy of a:
        g_i = a_{k* - i}   (if index in range) else 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    idx = np.arange(i_start, i_end + 1)
    jdx = max_idx - idx
    g[idx] = a[jdx]
    return g


def _mirror_update(a: np.ndarray, g: np.ndarray, alpha: float) -> np.ndarray:
    """
    Multiplicative KL‐mirror step:  a_i ← a_i·exp(‑α·g_i),
    followed by projection onto the simplex.
    """
    a_new = a * np.exp(-alpha * g)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= s
    return a_new


def _additive_update(a: np.ndarray, g: np.ndarray, step: float) -> np.ndarray:
    """Projected additive sub‑gradient step."""
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse pool of starting points (always normalised)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # anti‑symmetric reorder of uniform (often a good baseline)
    cand.append(_anti_symmetric_reorder(cand[-1]))

    # random Dirichlet draws (α = 1 and 5)
    cand.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    cand.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # geometric decays (r close to 1)
    for r in (0.95, 0.99, 0.999, 0.9995):
        geom = (1.0 - r) * r ** np.arange(n)
        cand.append(geom)

    # power‑law decays
    for p in (0.3, 0.5, 0.7, 1.0):
        pw = 1.0 / (np.arange(1, n + 1) ** p)
        cand.append(pw)

    # simple sinusoid (positive shift)
    x = np.linspace(0, np.pi, n)
    cand.append(np.sin(x) + 0.1)

    # normalise all candidates onto the simplex
    cand = [c / c.sum() for c in cand]
    return cand


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a time budget (default 1000 s) and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size (try to reuse a previously discovered construction)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
        print("[info] using previous construction length =", n)
    except Exception:
        n = 4000
        has_old = False
        print("[info] no previous construction – using n =", n)

    # ------------------------------------------------------------------
    # Phase 0 – diverse initialisation
    # ------------------------------------------------------------------
    init_cands = _initial_candidates(n, rng)

    # include the previous best if available
    if has_old:
        init_cands.append(old / old.sum())

    best_a = None
    best_val = float("inf")
    for idx, a in enumerate(init_cands):
        val = _fast_evaluate(a)
        print(f"[init {idx}] value = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = a.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – adaptive mirror descent (multiplicative KL‑update)
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_val = best_val
    alpha = 0.5            # initial step size
    min_alpha = 1e-5
    max_alpha = 2.0
    it = 0
    # allocate roughly 30 % of the whole budget to this phase
    phase_deadline = start_time + 0.30 * budget_s
    while time.time() < min(deadline, phase_deadline):
        conv, k_max, _ = _conv_and_max(a)
        g = _subgradient_of_max(a, k_max)

        a_new = _mirror_update(a, g, alpha)
        val_new = _fast_evaluate(a_new)

        if val_new < cur_val - 1e-12:
            a = a_new
            cur_val = val_new
            if val_new < best_val:
                best_val = val_new
                best_a = a.copy()
                print(f"[mirror] new best = {best_val:.6f}")
            # successful step -> relax step size a little
            alpha = min(alpha * 1.05, max_alpha)
        else:
            # reject – shrink the step size
            alpha = max(alpha * 0.5, min_alpha)

        # occasionally try the anti‑symmetric reorder which can drop the peak
        if it % 400 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                a = a_ord.copy()
                cur_val = val_ord
                print(f"[mirror reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected additive sub‑gradient descent (diminishing step)
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_val = best_val
    step0 = 0.2
    it = 0
    # allocate another ~25 % of the total time
    phase_deadline = start_time + 0.55 * budget_s
    while time.time() < min(deadline, phase_deadline):
        conv, k_max, _ = _conv_and_max(a)
        g = _subgradient_of_max(a, k_max)

        step = step0 / np.sqrt(it + 1.0)
        a_new = _additive_update(a, g, step)
        val_new = _fast_evaluate(a_new)

        if val_new < cur_val - 1e-12:
            a = a_new
            cur_val = val_new
            if val_new < best_val:
                best_val = val_new
                best_a = a.copy()
                print(f"[additive] new best = {best_val:.6f}")

        # occasional anti‑symmetric shuffle
        if it % 500 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                a = a_ord.copy()
                cur_val = val_ord
                print(f"[additive reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – guided simulated‑annealing with gradient‑directed moves
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_val = best_val
    T = 0.02
    cooling = 0.99995
    it = 0
    # allocate the remaining bulk of time (leaving ~30 s for polishing)
    phase_deadline = deadline - 30.0
    while time.time() < phase_deadline:
        # recompute gradient of the current max‑convolution
        conv, k_max, _ = _conv_and_max(a)
        g = _subgradient_of_max(a, k_max)

        if rng.random() < 0.5:
            # guided mass transfer: move from a large‑gradient position to a
            # small‑gradient one.
            src = int(np.argmax(g))
            dst = int(np.argmin(g))
            if src != dst and a[src] > 1e-15:
                delta = a[src] * 0.07 * rng.random()   # up to 7 % of source mass
                new_a = a.copy()
                new_a[src] -= delta
                new_a[dst] += delta
            else:
                # fallback to a tiny random perturbation
                i, j = rng.integers(0, n, size=2)
                if i == j or a[i] <= 1e-15:
                    continue
                delta = a[i] * 0.07 * rng.random()
                new_a = a.copy()
                new_a[i] -= delta
                new_a[j] += delta
        else:
            # pure random swap (helps escape plateaus)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        # project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val_new = _fast_evaluate(new_a)
        if val_new < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(val_new - cur_val) / max(T, 1e-12))

        if accept:
            a = new_a
            cur_val = val_new
            if val_new < best_val:
                best_val = val_new
                best_a = new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        # temperature schedule
        T = max(T * cooling, 1e-6)
        it += 1

    # ------------------------------------------------------------------
    # Phase 4 – final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check (official evaluator) and return.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
