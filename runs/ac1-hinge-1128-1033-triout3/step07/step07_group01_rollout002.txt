"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1️⃣  Initialise a diverse pool of candidates (uniform, previous best,
    several Dirichlet draws, sinusoid, sqrt‑shape, exponential‑shape,
    anti‑symmetric reorder).  Keep the best according to a fast
    surrogate ``_fast_evaluate`` (exact same objective for Σa=1).

2️⃣  **Mirror‑descent on the exact max‑convolution**  
    Repeatedly:
        * compute the full self‑convolution (FFT‑based),
        * identify all lag indices attaining the current maximum,
        * build a sub‑gradient by averaging the gradients of these active
          lags,
        * update a via an exponential mirror step  a←a·exp(‑α·g)
          and project back onto the probability simplex.
    The step size α decays geometrically from 0.5 to ~1e‑6.
    Occasionally the anti‑symmetric reorder is tried – it often
    flattens the convolution peak.

3️⃣  **Simulated‑annealing fine‑tuning**  
    Random mass transfers or swaps are proposed.
    Improvements are always accepted; uphill moves are accepted with a
    Boltzmann probability exp(-(Δ)/T) where the temperature T cools
    geometrically.

4️⃣  **Very cheap hill‑climbing** (swap / tiny mass transfer) for the
    last few seconds to polish the solution.

All operations keep the vector non‑negative and normalised (Σa=1),
which makes the objective equal to ``2·n·max(conv)``.
The routine respects a 1000 s budget and returns the best sequence
found (as a plain Python list).  The final score is obtained with the
official ``evaluate_sequence`` to guarantee validity.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Clip tiny negatives caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` for non‑negative vectors
    that sum to 1.  Returns 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that puts large values on one side and small values on the
    opposite side – often reduces the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %). The vector is kept
    on the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.005 * rng.random()   # smaller moves
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

    best_a = None
    best_val = float("inf")
    for i, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {i}] fast value = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Mirror‑descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    alpha0 = 0.5           # initial step size
    alpha_min = 1e-6
    decay = 0.9999

    # reserve ~60 s for SA + final hill‑climb
    mirror_deadline = deadline - 60.0

    while time.time() < mirror_deadline:
        # full self‑convolution (FFT)
        conv = _full_convolution(a)
        max_val = conv.max()
        cur_obj = 2.0 * n * max_val
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[mirror {it}] new best = {best_val:.6f}")

        # --- sub‑gradient of the max‑convolution (average over all active lags) ---
        eps = 1e-14
        active = np.where(conv >= max_val - eps)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idxs = np.arange(i_start, i_end + 1)
            jdxs = k - idxs
            g[idxs] += a[jdxs]
        g /= max(1, len(active))

        # --- exponential mirror‑descent update ---
        alpha = max(alpha_min, alpha0 * (decay ** it))
        a = a * np.exp(-alpha * g)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = 1.0
        a /= s

        # occasional anti‑symmetric reorder (helps to escape plateaus)
        if it % 2000 == 0 and it > 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                a = a_ord.copy()
                print(f"[reorder] new best = {best_val:.6f}")

        # tiny random perturbation every few hundred steps (helps to
        # avoid getting stuck when α becomes very small)
        if it % 500 == 0:
            i = rng.integers(0, n)
            if a[i] > 0:
                delta = a[i] * 0.001 * rng.random()
                j = rng.integers(0, n)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()

        it += 1

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing style random mass‑transfer moves
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.02                     # initial temperature
    cooling = 0.99997
    sa_deadline = deadline - 15.0

    while time.time() < sa_deadline:
        if rng.random() < 0.6:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.03 * rng.random()   # up to 3 % of a_i
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        # re‑normalise
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()

        new_val = _fast_evaluate(new_a)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Boltzmann acceptance
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # 4️⃣  Final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check via the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
