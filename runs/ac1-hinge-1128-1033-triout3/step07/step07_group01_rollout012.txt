def gradient_max_conv(a):
    conv = full_convolution(a)
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-16
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)

        i_start = max(0, k - (len(a) - 1))
        i_end = min(len(a)-1, k)
        idx = np.arange(i_start, i_end+1)
        g[idx] += a[k - idx]
    return g, conv, max_val

def alternating_ends_reorder(a):
    n = len(a)
    sorted_vals = np.sort(a)[::-1]   # descending
    new_a = np.zeros_like(a)
    left = 0
    right = n-1
    toggle = True
    for val in sorted_vals:
        if toggle:
            new_a[left] = val
            left += 1
        else:
            new_a[right] = val
            right -= 1
        toggle = not toggle
    return new_a
```

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Result length = 2·n‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Clip tiny negatives caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv

# Smooth surrogate for L‑BFGS (log‑sum‑exp of the convolution).
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    n = _GLOBAL_N

def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """

    Often flattens the self‑convolution peak.

    """
    Distribute the values alternatingly from the left and right ends.
    Large entries become far apart, which can reduce the convolution peak.
