"""
Hybrid optimizer for the step‑function convolution problem (improved version).

The search consists of five phases:

1. **Seed generation** – hand‑crafted families (geometric, power‑law),
   random Dirichlet draws and *sparse* Dirichlet draws (a few hundred
   non‑zero entries) are evaluated cheaply with an FFT‑based surrogate.

2. **Multi‑restart smoothed L‑BFGS** – a log‑sum‑exp smooth surrogate of the
   max‑convolution is minimized.  A decreasing schedule of the smoothing
   parameter (ε) is used and many random restarts are performed (the
   soft‑max parametrisation guarantees positivity and ℓ₁‑normalisation).

3. **Focused simulated annealing** – temperature‑controlled local moves:
   swaps, small mass transfers, *targeted* transfers from the pair that
   contributes most to the current maximal convolution entry, and a
   *split* move that redistributes a fraction of one large entry onto two
   new positions.

4. **Deterministic hill‑climbing** – cheap pure swaps and mass transfers
   until the time budget expires.

5. **Exact evaluation** – the best candidate is finally scored with the
   provided ``evaluate_sequence`` function.

The implementation is deliberately self‑contained, uses only NumPy /
SciPy, and respects the 1000 s budget (a small safety margin is left at
the end).
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global parameters for the smoothed surrogate (soft‑max + log‑sum‑exp)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter for L‑BFGS
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding in FFT)."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using a real FFT.
    The result has length ``2*len(a)-1``.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # numerical noise may produce tiny negatives; clamp to zero.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for a non‑negative 1‑D array.
    The routine does *not* perform the final clamping to 0/1000 – the
    optimiser works in the interior of that domain anyway.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient.
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns the smoothed objective L_ε and its gradient w.r.t. the
    unconstrained parameters ``w`` (soft‑max representation of the
    non‑negative sequence).
    """
    # Soft‑max (ensures a_i ≥ 0 and Σ a_i = 1)
    w_shift = w - np.max(w)              # for numerical stability
    a = np.exp(w_shift)
    a /= np.sum(a)

    # autocorrelation a*a
    conv = _fft_convolve_self(a)
    max_conv = conv.max()

    # Log‑sum‑exp approximation of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # = ε·log Σ exp((c_i−M)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # Gradient w.r.t. a (closed form, see original code)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability distribution over k
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)        # a * p_rev
    seg = conv_ap[n - 1: 2 * n - 1]       # central part (length n)
    grad_a = 4.0 * n * seg[::-1]

    # Chain‑rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Simple re‑ordering heuristics (used after optimisation phases)
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Largest values left, smallest values right (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)
    out = np.empty_like(a)
    half = n // 2
    out[:half] = s[-half:][::-1]
    out[half:][::-1] = s[:half]
    if n % 2:
        out[half] = s[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    s = np.sort(a)
    left, right = 0, len(s) - 1
    out = np.empty_like(a)
    i = 0
    while left <= right:
        out[i] = s[right]
        i += 1
        right -= 1
        if left <= right:
            out[i] = s[left]
            i += 1
            left += 1
    return out


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Plain descending sort."""
    return np.sort(a)[::-1]


# ----------------------------------------------------------------------
# Parametric seed families
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """Geometric progression a_i ∝ q**i (0 < q < 1), ℓ₁‑normalised."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q ** n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)**p (p < 0), ℓ₁‑normalised."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _sparse_seed(m: int, n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform weight on *m* distinct random positions (others zero)."""
    a = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=m, replace=False)
    a[idx] = 1.0 / m
    return a

            new_a[t2] += delta_total * (1 - frac)

        # evaluate the new candidate
        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # occasional progress report
                if int(best_val * 1e6) % 1000 == 0:
                    print(f"[SA] new best = {best_val:.6f}")

        T = max(T * decay, 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final hill‑climbing (very cheap moves)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, cur_val: float,
                deadline: float) -> (np.ndarray, float):
    """Deterministic refinement until the time budget is exhausted."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = cur_val

    while time.time() < deadline:
        if rng.random() < 0.25:          # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = best_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                             # tiny transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] == 0.0:
                continue
            delta = best_a[i] * 0.02 * rng.random()
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta

        val = _eval_fast(cand)
        if val + 1e-12 < best_val:
            best_val = val
            best_a = cand
            print(f"[Hill] improved to {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # problem size (fixed to 4000 – a good trade‑off between quality & speed)
    # ------------------------------------------------------------------
    n = 4000
    global _GLOBAL_N
    _GLOBAL_N = n

    # ------------------------------------------------------------------
    # initialise from a previously stored construction if available
    # ------------------------------------------------------------------
    try:
        # the sandbox may have a variable ``height_sequence_1`` already defined
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the target length
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()                     # ensure Σ a_i = 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. diverse seed generation (including sparse seeds)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    seed_candidates = []

    # geometric progression
    for q in np.linspace(0.70, 0.999, 15):
        seed_candidates.append(_geometric_sequence(q, n))

    # power‑law progression
    for p in np.linspace(-3.0, -0.2, 15):
        seed_candidates.append(_power_sequence(p, n))

    # dense random draws (Dirichlet‑like)
    for _ in range(25):
        a = rng.random(n)
        a /= a.sum()
        seed_candidates.append(a)

    # sparse seeds (uniform on a small support)
    for m in (70, 90, 110, 130, 150):
        seed_candidates.append(_sparse_seed(m, n, rng))

    for a in seed_candidates:
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[seed] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. multi‑restart smoothed L‑BFGS (coarse → fine ε schedule)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    stage_a_deadline = time.time() + min(0.30 * budget_s,
                                         time_left * 0.45)

    eps_schedule_coarse = np.logspace(-2, -7, 10)  # from 1e‑2 down to 1e‑7
    while time.time() < stage_a_deadline:
        # random starting point in the soft‑max space
        w = np.random.randn(n) * 2.0
        for eps in eps_schedule_coarse:
            if time.time() >= stage_a_deadline:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': 120,
                                    'ftol': 1e-8,
                                    'gtol': 1e-8,
                                    'disp': False})
            w = res.x
            a = np.exp(w - np.max(w))
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[coarse L‑BFGS] eps={eps:.1e}  new best={best_val:.6f}")

        # occasional deterministic re‑orderings
        for reorder in (_anti_symmetric_reorder,
                        _alternating_reorder,
                        _descending_reorder):
            a_r = reorder(best_a)
            v_r = _eval_fast(a_r)
            if v_r < best_val:
                best_val = v_r
                best_a = a_r.copy()
                print(f"[reorder] {reorder.__name__} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. refined L‑BFGS (finer ε schedule, more iterations)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    stage_b_deadline = time.time() + min(200.0,
                                         max(0.0, time_left - 30.0))

    eps_schedule_fine = np.logspace(-3, -12, 12)   # from 1e‑3 down to 1e‑12
    w = np.log(best_a + 1e-16)                     # initialise from current best

    while time.time() < stage_b_deadline:
        for eps in eps_schedule_fine:
            if time.time() >= stage_b_deadline:
                break
            _GLOBAL_EPS = eps
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': 350,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            a = np.exp(w - np.max(w))
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[fine L‑BFGS] eps={eps:.1e}  new best={best_val:.6f}")

        # deterministic re‑ordering pass
        for reorder in (_anti_symmetric_reorder, _alternating_reorder):
            a_r = reorder(best_a)
            v_r = _eval_fast(a_r)
            if v_r < best_val:
                best_val = v_r
                best_a = a_r.copy()
                print(f"[reorder‑fine] {reorder.__name__} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. extended simulated annealing (heavy‑weight local search)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 180.0:
        sa_deadline = time.time() + remaining - 100.0   # reserve ~100 s for final polish
    else:
        sa_deadline = deadline - 10.0
    print(f"[SA] running until {sa_deadline:.2f}")
    best_a, best_val = _simulated_annealing(best_a, best_val, sa_deadline)

    # ------------------------------------------------------------------
    # 5. final hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # 6. exact scoring with the provided evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    return best_a.tolist()
