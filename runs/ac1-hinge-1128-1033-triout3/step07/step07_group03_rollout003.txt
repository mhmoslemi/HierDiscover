"""
Propose a non‑negative step function (list of floats) that minimises
``evaluate_sequence``.

The search runs inside the 1000 s time budget and consists of four
lightweight phases:

1. **Initialisation** – start from a previous construction if
   ``height_sequence_1`` exists, otherwise from a few random Dirichlet
   vectors (exponential samples).

2. **Smooth‑surrogate optimisation** – minimise a smooth approximation
   of the true objective (soft‑max parametrisation + log‑sum‑exp of the
   self‑convolution) with L‑BFGS‑B.  Several ε‑values (decreasing) and a
   handful of random restarts are performed.  After each L‑BFGS run the
   candidate is optionally re‑ordered by the anti‑symmetric heuristic.

3. **Simulated‑annealing fine‑tuning** – starting from the best vector
   found so far, repeatedly transfer a tiny amount of mass from one
   coordinate to another (multiplicative mirror‑descent style).  A
   temperature schedule allows occasional uphill moves, helping the
   search escape shallow local minima.  Every few thousand iterations a
   deterministic anti‑symmetric re‑ordering is tried.

4. **Simple hill‑climbing refinement** – random swaps and small mass
   transfers (no temperature) keep improving the solution until the
   deadline.

All updates keep the vector on the probability simplex
(non‑negative, sum = 1), which does not affect the homogeneous objective.
The algorithm is deliberately fast (FFT‑based convolutions) and fits
comfortably into the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # tiny negative values from floating‑point noise are harmless but clipped
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values on the left, the smallest on the right."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution a*a
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (chain‑rule through the smoothed max)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # distribution over lag indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # derivative w.r.t. a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule – geometric progression from 5·10⁻³ down to 10⁻⁹
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=7)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        # a relatively tight tolerance works well with the L‑BFGS‑B implementation
        res = minimize(
            _smooth_obj_and_grad_wrapper,
            w,
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': 120, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False}
        )
        w = res.x

        # map back to the probability simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric re‑ordering heuristic (often improves the bound)
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray, deadline: float):
    """
    Small‑mass mirror‑descent with a temperature schedule.
    Returns the best (a, value) found before ``deadline``.
    """
    n = a_start.shape[0]
    cur = a_start.copy()
    cur_val = evaluate_sequence(cur.tolist())
    best = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start_time = time.time()
    total_budget = deadline - start_time

    it = 0
    # Parameters – they are decayed linearly with the remaining time
    T0 = 5e-3          # initial temperature for Metropolis acceptance
    move_scale0 = 0.3  # initial relative amount of mass that can be moved

    while time.time() < deadline - 5.0:
        it += 1
        remaining = deadline - time.time()
        frac = remaining / total_budget

        # linearly decaying temperature and move size
        T = T0 * frac
        move_scale = move_scale0 * frac
        if move_scale < 5e-5:
            move_scale = 5e-5

        # ----- choose a move ------------------------------------------------
        if rng.random() < 0.15:                     # deterministic re‑ordering move
            cand = _anti_symmetric_reorder(cur)
            cand_val = evaluate_sequence(cand.tolist())
            # accept only if it improves (no uphill for a pure reorder)
            if cand_val < cur_val:
                cur, cur_val = cand, cand_val
                if cand_val < best_val:
                    best, best_val = cand.copy(), cand_val
                    print(f"[SA reorder it {it}] new best = {best_val:.6f}")
            continue

        # mass transfer move
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 1e-15:
            continue
        delta = cur[i] * move_scale * rng.random()
        if delta > cur[i]:
            delta = cur[i]

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate
        cand_val = evaluate_sequence(cand.tolist())

        # Metropolis acceptance
        if cand_val < cur_val:
            accept = True
        else:
            # probability = exp(-(Δ)/T)   (Δ > 0)
            prob = np.exp(-(cand_val - cur_val) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best, best_val = cand.copy(), cand_val
                print(f"[SA iter {it}] new best = {best_val:.6f}")

        # occasional logging
        if it % 5000 == 0:
            print(f"[SA status it {it}] best = {best_val:.6f}, cur = {cur_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (random swaps / mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random local search without temperature – keep only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.02 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto the simplex (the operations keep the sum, but guard against round‑off)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:          # strict improvement
            cur_a, cur_val = new_a, val
            if val < best_val - 1e-12:
                best_val, a = val, new_a.copy()
                print(f"[refine] new best = {best_val:.6f}")

    return a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

    # --------------------------------------------------------------
    # Problem size – 8000 entries gives a good trade‑off between quality
    # and speed (the objective is homogeneous, so the absolute length
    # does not matter for the bound we target).
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise from a known construction if possible
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)          # uniform start

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
        s = a.sum()
    a /= s

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – a few random vectors to avoid a poor start
    # --------------------------------------------------------------
    for tr in range(5):
        if time.time() >= deadline - 250:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val, best_a = val, a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth surrogate optimisation (L‑BFGS) with restarts
    # --------------------------------------------------------------
    restarts = 8
    while time.time() < deadline - 200 and restarts > 0:
        restarts -= 1
        w_start = np.log(best_a + 1e-12)
        w_start += rng.normal(scale=0.4, size=n)   # modest perturbation
        a_opt, val_opt = _l_bfgs_optimize_from_w(w_start, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[restart] new best after L‑BFGS = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline - 80:
        a_sa, val_sa = _simulated_annealing(best_a, deadline)
        if val_sa < best_val:
            best_val, best_a = val_sa, a_sa.copy()
            print(f"[SA] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # --------------------------------------------------------------
    # Final evaluation & return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
