"""
propose_candidate
-----------------

Search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The algorithm is a blend of:

1. **Diverse initialisations** – uniform, previous best, Dirichlet,
   sinusoid and a family of “far‑apart” reorderings of a sorted vector.
2. **Heavy L‑BFGS** – optimisation of a smooth log‑sum‑exp surrogate of
   the max‑convolution with a fine ε‑schedule and many random restarts.
3. **Mirror‑descent** – exact sub‑gradient steps on the true
   max‑convolution (adds mass to indices that are *not* part of the
   current maximal lag).
4. **Intensive random re‑ordering** – after every few hundred mirror‑
   descent steps the current best vector is permuted in a “large‑small”
   alternating pattern; the best permutation is kept.
5. **Final cheap hill‑climb** – swaps or tiny mass transfers that are
   accepted only if they strictly improve the objective.

All operations keep the vector on the probability simplex
(Σ a_i = 1, a_i ≥ 0).  The routine respects the supplied time budget
(≤ 1000 s) and returns the best list it has found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (required by scipy.minimize)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise can produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


# -------------------------- Fast objective --------------------------------
def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already sum to 1.
    Returns 2·n·max(conv) because the denominator (sum_a)^2 = 1.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


# -------------------- Reordering heuristics --------------------------------
def _alternating_extreme_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values as far apart as possible:
    descending sorted values are alternately put at the leftmost free
    position and the rightmost free position.
    """
    vals = np.sort(a)[::-1]          # descending
    n = len(a)
    out = np.empty_like(a)
    lo, hi = 0, n - 1
    toggle = True
    for v in vals:
        if toggle:
            out[lo] = v
            lo += 1
        else:
            out[hi] = v
            hi -= 1
        toggle = not toggle
    return out


def _interleaved_reorder(a: np.ndarray) -> np.ndarray:
    """
    Similar to the previous function but starts with a large value on the
    left, then a large value on the right, then the second‑largest on the
    left‑most remaining slot, etc.  This spreads large entries evenly.
    """
    return _alternating_extreme_reorder(a)  # identical implementation


def _random_spread_reorder(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Take the sorted values and assign them to a random permutation that
    attempts to keep large values apart.  The permutation is generated
    by shuffling the list of target positions that themselves are
    constructed by an alternating‑extreme pattern; the shuffle therefore
    preserves the “far‑apart” property on average.
    """
    n = len(a)
    # Base pattern of positions: 0, n‑1, 1, n‑2, 2, …
    pos = []
    lo, hi = 0, n - 1
    toggle = True
    while lo <= hi:
        pos.append(lo if toggle else hi)
        if toggle:
            lo += 1
        else:
            hi -= 1
        toggle = not toggle
    # Randomly permute the order of the positions (preserving length)
    rng.shuffle(pos)
    vals = np.sort(a)[::-1]      # descending
    out = np.empty_like(a)
    out[pos] = vals
    return out


# ------------------------ Smooth surrogate (L‑BFGS) -----------------------
def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max -----
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- convolution -----
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max -----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    # p is the soft‑max over the shifted values
    weights = np.exp(shifted)
    p = weights / weights.sum()      # distribution over lag indices

    # We need ∂L/∂a = 4·n·(conv * p)   (see derivation in the original code)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # Extract the segment that aligns with a (indices n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # ----- back‑prop through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Thin wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w)


# ----------------- Sub‑gradient of the exact max‑convolution ---------------
def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (max_conv, g) where g is a sub‑gradient of max(conv) w.r.t. a.
    For each maximising lag k, the derivative ∂conv_k/∂a_i equals a_{k‑i}
    (if the index is in range), otherwise 0.
    If several lags share the same maximal value the gradients are
    averaged.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_val = conv.max()
    ks = np.where(conv == max_val)[0]

    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx
        g[idx] += a[jdx]

    if len(ks) > 1:
        g /= len(ks)
    return max_val, g


# ---------------------- Hill‑climbing refinement -------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector is kept on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

    Search for a non‑negative step‑function with minimal ``evaluate_sequence``.
    The function respects the supplied time budget and returns the best
    sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final sanity check
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – use the length of the previous construction if
    # available, otherwise fall back to a moderate default.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    print(f"[info] problem size n = {n}")

    # ------------------------------------------------------------------
    # 1) Build a pool of cheap initial candidates.
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # random Dirichlet samples (α = 0.5 and α = 3)
    init_candidates.append(rng.gamma(shape=0.5, scale=1.0, size=n))
    init_candidates.append(rng.gamma(shape=3.0, scale=1.0, size=n))

    # simple sinusoidal shape
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # normalise everything onto the simplex
    init_candidates = [c / c.sum() for c in init_candidates]

    # ------------------------------------------------------------------
    # 2) Evaluate the pool and keep the best.
    # ------------------------------------------------------------------
    best_a = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] eval = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Generate a few aggressive re‑orderings of the current best.
    #    (These are cheap but often give a sizable drop.)
    # ------------------------------------------------------------------
    reord_candidates = [best_a]
    reord_candidates.append(_alternating_extreme_reorder(best_a))
    reord_candidates.append(_interleaved_reorder(best_a))
    # a handful of random “spread” permutations
    for _ in range(30):
        reord_candidates.append(_random_spread_reorder(best_a, rng))

    for idx, cand in enumerate(reord_candidates):
        val = _fast_evaluate(cand)
        if val < best_val - 1e-12:
            best_val = val
            best_a = cand.copy()
            print(f"[reorder {idx}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) L‑BFGS on a smooth surrogate of max‑convolution.
    # ------------------------------------------------------------------
    # The surrogate uses a soft‑max over the lag values with a
    # decreasing smoothing parameter ε.  The schedule is fairly fine –
    # we go down to 1e‑13.
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-13), num=15)

    # Decision variables are the log‑softmax parameters w.
    w = np.log(best_a + 1e-12)   # initialise w from the best candidate

    max_restarts = 15
    restart = 0
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        if restart % 5 == 0:
            # occasional full random restart to escape possible plateaus
            w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x
            # Convert back to probabilities via soft‑max
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # After each restart try a cheap re‑ordering again
        a_ord = _alternating_extreme_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val - 1e-12:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Mirror‑descent on the exact max‑convolution.
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    # aggressive schedule – we start with a relatively large step and
    # decay it very slowly.
    alpha0 = 0.7
    decay = 0.9997
    min_alpha = 0.005

    while time.time() < deadline - 20.0:
        max_conv, g = _max_conv_and_subgrad(a)

        # Exponential mirror‑descent update (multiplicative weights)
        alpha = max(alpha0 * (decay ** (it // 250)), min_alpha)
        a = a * np.exp(-alpha * g)

        # Projection onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 200 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[mirror it {it}] new best = {best_val:.6f} (max conv {max_conv:.9f})")
            # every few hundred iters try a fresh spread‑reorder
            if it % 800 == 0:
                a_spread = _random_spread_reorder(a, rng)
                val_spread = _fast_evaluate(a_spread)
                if val_spread < best_val - 1e-12:
                    best_val = val_spread
                    best_a = a_spread.copy()
                    a = a_spread.copy()
                    print(f"[mirror reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 6) Simulated‑annealing style random moves (still cheap, but
    #    acceptable because we are close to a local optimum).
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.02
    cooling = 0.99995
    rng = np.random.default_rng(seed + 123456)

    while time.time() < deadline - 10.0:
        # Choose move type
        if rng.random() < 0.55:          # mass transfer (slightly favoured)
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.10 * rng.random()   # up to 10 % of source mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:                            # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        # Re‑normalise
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")
        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # 7) Final cheap hill‑climbing pass.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # 8) Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
