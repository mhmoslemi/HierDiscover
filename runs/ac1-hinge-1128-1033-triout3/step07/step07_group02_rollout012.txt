"""
Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Rich seeding** – generate many candidate shapes (Beta, Gaussian,
   exponential, cosine) on the chosen length ``n`` and keep the
   best (exact objective is evaluated by a fast FFT‑based convolution).

2. **Adam optimisation** – treat the vector as a soft‑max of logits,
   minimise a smooth log‑sum‑exp approximation of the maximal
   convolution entry with an Adam optimiser.  The temperature
   (smoothness) and learning‑rate are annealed.  Restarts are performed
   when the optimiser stalls.

3. **Hill‑climbing (simulated annealing)** – repeatedly move a small
   amount of mass from an index that contributes heavily to the current
   maximal convolution entry to an index that contributes little.
   Moves are accepted if they improve the exact objective, otherwise
   with Metropolis probability.  Stagnation triggers a brief random
   perturbation of the whole vector.

All work respects the 1000 s budget; a 0.5 s safety margin is kept for the
final call to ``evaluate_sequence``.  The returned list is normalised to
unit sum (scale‑invariant for the objective) and contains only
non‑negative numbers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max approximation (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smoothed version of ``max(conv(a,a))`` and its gradient.
    ``eps`` is the temperature (larger → smoother).  The returned gradient
    already includes the factor ``2·n`` from the final objective.
    """
    n = a.shape[0]
    b = _full_convolution(a)               # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick for numerical stability
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()            # soft‑max weights, shape (2n‑1,)
    # Gradient of Σ p_k * b_k w.r.t. a:
    # ∂b_k/∂a_i = 2·a_{k-i} (if indices are inside the range)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)          # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]      # the n entries that affect the gradient
    grad_a = 4.0 * n * segment[::-1]       # factor 4n = 2·n·2 (∂b/∂a)
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a smooth approximation of the objective.
    Returns the best pair (a, value) observed before *deadline*.
    """
    # initialise logits (soft‑max guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.001          # learning‑rate schedule
    eps_start, eps_end = 0.02, 1e-7        # smooth‑max temperature schedule

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 30_000                       # restart after many idle steps

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # convert logits → a (simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # progress (0 → 1) for schedules
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max and its gradient
        _, grad_a = _smooth_max_and_grad(a, eps)

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape tiny plateaus
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.02, size=n)

        # restart on stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] idle {idle} → restarting")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from indices that
    inflate the current maximal convolution entry to indices that do not.
    The exact objective is used for acceptance.
    """
    a = a0.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules
    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.02, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 80_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to b[k_max] (ordered pairs)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ---- source index (probability ∝ contribution) ----
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ---- target index (biased towards low contribution) ----
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback uniform among remaining indices
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        # add/subtract the cross terms
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # diagonal terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – random perturbation")
        iteration += 1

    return best_a, best_val

# Parametric families (Beta, Gaussian, Exponential, Cosine)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1} (n‑i)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # grid for exponents (log‑scale for large values)
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _gaussian_family_best(n: int) -> tuple[np.ndarray, float]:
    """Gaussian‑shaped candidates a_i ∝ exp(‑(i‑μ)² / (2σ²))."""
    best_a = None
    best_val = np.inf
    mu = (n - 1) / 2.0
    # sigma as a fraction of n
    sigma_factors = np.linspace(0.05, 0.5, 10)
    for sf in sigma_factors:
        sigma = sf * n
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        if a.sum() == 0:
            continue
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


def _exponential_family_best(n: int) -> tuple[np.ndarray, float]:
    """Two‑sided exponential a_i ∝ exp(‑λ·|i‑μ|)."""
    best_a = None
    best_val = np.inf
    mu = (n - 1) / 2.0
    lambdas = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    i = np.arange(n, dtype=np.float64)
    for lam in lambdas:
        a = np.exp(-lam * np.abs(i - mu))
        if a.sum() == 0:
            continue
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


def _cosine_family_best(n: int) -> tuple[np.ndarray, float]:
    """Cosine‑raised candidates a_i ∝ (1 + cos(2π·(i‑μ)/n))^p."""
    best_a = None
    best_val = np.inf
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    p_vals = [0.5, 1, 2, 3, 4, 6]
    for p in p_vals:
        cos_part = 1.0 + np.cos(2.0 * np.pi * (i - mu) / n)
        a = np.power(np.maximum(cos_part, 0.0), p)
        if a.sum() == 0:
            continue
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final call

    # --------------------------------------------------------------
    # Choose a problem size – a larger n gives a finer discretisation.
    # --------------------------------------------------------------
    n = 8000   # increase from 5000 used previously

    # --------------------------------------------------------------
    # Initialise a – try to reuse a previous solution if available.
    # --------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            raise ValueError
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 0 – exploration of parametric families
    # --------------------------------------------------------------
    best_a, best_val = None, np.inf

    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    # try the different families, keeping the best
    families = [_beta_family_best, _gaussian_family_best,
                _exponential_family_best, _cosine_family_best]

    for fam in families:
        try:
            cand_a, cand_val = fam(n)
            if cand_val < best_val:
                best_a, best_val = cand_a, cand_val
                print(f"[init] {fam.__name__} improved to {best_val:.6f}")
        except Exception as exc:
            print(f"[init] {fam.__name__} failed:", exc)

    # random fallback if everything else failed
    if best_a is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_a, best_val = a, _objective(a)
        print("[init] random start, value =", best_val)

    # --------------------------------------------------------------
    # Phase A – Adam optimisation (≈35 % of the time budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.35 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # --------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (remaining budget)
    # --------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=phase_a_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
