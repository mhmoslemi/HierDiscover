def random_move(a):
    n = len(a)
    # Pick donor indices with probabilities proportional to a (or top)
    donors = rng.choice(n, size=2, replace=False, p=a/ a.sum())
    i, j = donors
    # pick delta = min(a[i], a[j]) * step_frac where step_frac = random between 0 and 0.5
    # destination: choose two distinct indices not donors weighted towards low a.
    # say probability proportional to 1/(a+ epsilon)
    low_weights = 1/(a + 1e-6)
    low_weights[donors] = 0
    p, q = rng.choice(n, size=2, replace=False, p=low_weights/low_weights.sum())
    # compute delta
    if i == j:
        delta = a[i] * rng.uniform(0.1, 0.3) # fraction
    else:
        delta = min(a[i], a[j]) * rng.uniform(0.1, 0.3)
    new_a = a.copy()
    new_a[i] -= delta
    new_a[j] -= delta
    new_a[p] += delta
    new_a[q] += delta
    # Ensure non-negative
    new_a = np.clip(new_a, 0, None)
    s = new_a.sum()
    if s == 0:
        return a, None
    new_a = new_a / s
    new_val = objective(new_a)
    return new_a, new_val

def _anneal_hillclimb(a0, deadline, rng):
    a = a0.copy()
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val
    start_time = time.time()
    total = deadline - start_time
    # Temperature initial as fraction of cur_val
    init_T = cur_val * 0.05  # adjust
    while time.time() < deadline:
        # proportion of time elapsed
        elapsed = time.time() - start_time
        frac = elapsed / total
        T = init_T * (1 - frac) + 1e-12  # linear cooling
        n = len(a)
        # Choose donors with probability proportional to a (mass)
        prob = a / a.sum()
        i, j = rng.choice(n, size=2, replace=False, p=prob)
        # delta fraction random
        delta_frac = rng.uniform(0.05, 0.3)
        delta = min(a[i], a[j]) * delta_frac
        if delta <= 0:
            continue
        # Choose recipients preferring low a
        inv_prob = 1.0 / (a + 1e-12)
        inv_prob[i] = inv_prob[j] = 0.0
        # Normalize
        inv_prob_sum = inv_prob.sum()
        if inv_prob_sum == 0:
            continue
        inv_prob = inv_prob / inv_prob_sum
        p, q = rng.choice(n, size=2, replace=False, p=inv_prob)

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] -= delta
        new_a[p] += delta
        new_a[q] += delta
        new_a = np.clip(new_a, 0, None)

"""
Propose a step‑function (list of non‑negative heights) that minimizes
``evaluate_sequence``.  The algorithm combines three ideas:

1.  Initialise from the previously found construction (if any) and from a
    few classic window functions (Hann, Hamming, Blackman) as well as from
    simple parametric families (U‑shaped and power‑law profiles).  All of
    these are normalised to sum = 1 and evaluated directly – this often
    already beats the baseline.

2.  A fast projected sub‑gradient descent on the exact objective
    ``f(a)=2·n·max(conv(a,a))``.  The sub‑gradient is taken only for the
    lag that currently attains the maximum.  After each trial step the
    vector is clipped to non‑negative values and projected onto the
    probability simplex.

3.  A simulated‑annealing hill‑climbing phase that repeatedly moves a
    small fraction of mass from high‑mass positions to low‑mass positions.
    Moves are accepted if they improve the objective or with a
    temperature‑controlled probability.  The temperature is linearly
    cooled to zero.  This phase can escape shallow local minima and
    often yields a modest but decisive improvement.

The code respects the supplied time budget (default 1000 s), prints
intermediate scores and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for convolution and objective
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # Tiny negative values can appear because of round‑off; clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` when sum(a) = 1.
    f(a) = 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return float(2.0 * n * np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Projected sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    step0 = 0.5  # initial step‑size factor

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and the lag attaining the maximum
        conv = _full_convolution(a)
        maxb = conv.max()
        # if several lags tie, pick one uniformly at random
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # sub‑gradient for the selected lag
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        grad[i_range] = 2.0 * a[j_range]

        # step size decreasing with √t (standard sub‑gradient schedule)
        lr = step0 / np.sqrt(t)

        # single back‑tracking line‑search
        improved = False
        for _ in range(6):
            a_trial = a - lr * grad
            a_trial = np.clip(a_trial, 0.0, None)
            s = a_trial.sum()
            if s == 0.0:
                lr *= 0.5
                continue
            a_trial /= s  # projection onto the simplex
            val = _objective(a_trial)
            if val < best_val:
                a = a_trial
                best_a = a_trial.copy()
                best_val = val
                improved = True
                step0 = min(step0 * 1.02, 2.0)  # slightly increase step on success
                break
            else:
                lr *= 0.5

        if not improved:
            step0 = max(step0 * 0.95, 1e-4)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing
# ----------------------------------------------------------------------


def _anneal_hillclimb(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing with a simple linear cooling schedule.
    Randomly moves a fraction of mass from high‑mass positions to low‑mass
    positions.  Moves are accepted if they improve the objective or with a
    probability exp((old‑new)/T) where T is the current temperature.
    """
    a = a0.copy()
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    # Initial temperature – a small fraction of the current objective
    init_T = cur_val * 0.05

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = init_T * (1.0 - frac) + 1e-12  # linear cooling to near zero

        n = a.shape[0]

        # ----- pick donors -----
        # With probability 0.7 use a double‑donor move, otherwise a single‑donor.
        if rng.random() < 0.7:
            # two donors, probabilities proportional to current mass
            donor_probs = a / a.sum()
            i, j = rng.choice(n, size=2, replace=False, p=donor_probs)
            delta = min(a[i], a[j]) * rng.uniform(0.05, 0.30)
        else:
            # single donor
            i = int(rng.choice(n, p=a / a.sum()))
            j = -1  # marker for "no second donor"
            delta = a[i] * rng.uniform(0.05, 0.30)

        if delta <= 0.0:
            continue

        # ----- pick recipients (low‑mass) -----
        inv_weights = 1.0 / (a + 1e-12)
        inv_weights[i] = 0.0
        if j >= 0:
            inv_weights[j] = 0.0
        inv_sum = inv_weights.sum()
        if inv_sum == 0.0:
            continue
        inv_weights /= inv_sum
        p, q = rng.choice(n, size=2, replace=False, p=inv_weights)

        # ----- perform the move -----
        new_a = a.copy()
        new_a[i] -= delta
        if j >= 0:
            new_a[j] -= delta
        new_a[p] += delta
        new_a[q] += delta
        new_a = np.clip(new_a, 0.0, None)
        s = new_a.sum()
        if s == 0.0:
            continue
        new_a /= s

        new_val = _objective(new_a)

        # ----- acceptance test -----
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / T):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    The routine combines a small deterministic scan of useful parametric
    families with stochastic sub‑gradient descent and annealed hill‑climbing.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Determine problem size (use previous construction if available)
    # ------------------------------------------------------------------
    default_n = 4000
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = int(old.shape[0])
        a_initial = np.maximum(old, 0.0)
        if a_initial.sum() == 0.0:
            a_initial = rng.random(n)
        a_initial /= a_initial.sum()
    except Exception:
        n = default_n
        a_initial = rng.random(n)
        a_initial /= a_initial.sum()

    print(f"[init] n = {n}")

    # ------------------------------------------------------------------
    # Keep track of the best candidate seen so far
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    def _try_candidate(a: np.ndarray):
        """Normalise a, evaluate, and update the global best if improved."""
        nonlocal best_a, best_val
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            return
        a = a / s
        val = _objective(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[candidate] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Seed with a few deterministic families
    # ------------------------------------------------------------------
    _try_candidate(a_initial)

    # uniform
    _try_candidate(np.ones(n) / n)

    # classic windows (if they exist for this length)
    try:
        _try_candidate(np.hanning(n) / np.hanning(n).sum())
    except Exception:
        pass
    try:
        _try_candidate(np.hamming(n) / np.hamming(n).sum())
    except Exception:
        pass
    try:
        _try_candidate(np.blackman(n) / np.blackman(n).sum())
    except Exception:
        pass

    # symmetric U‑shaped profiles a_i ∝ ((i+1)*(n-i))^{-alpha}
    alphas = [0.02, 0.05, 0.08, 0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40]
    idx = np.arange(n, dtype=float) + 1.0
    b = idx * (n - idx + 1.0)            # (i+1)*(n-i)
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = b ** (-alpha)
        _try_candidate(a)

    # asymmetric power‑law profiles a_i ∝ (i+1)^{-beta}
    betas = [0.02, 0.05, 0.08, 0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40, 0.50]
    for beta in betas:
        if time.time() >= deadline:
            break
        a = idx ** (-beta)
        _try_candidate(a)

    # ------------------------------------------------------------------
    # 2. Local optimisation: sub‑gradient descent + annealed hill‑climb
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 0.2:
        restart += 1
        # Random initial point for this restart
        a0 = rng.random(n)
        a0 /= a0.sum()

        # Sub‑gradient descent (quick, limited iterations)
        a_sub, _ = _subgradient_descent(a0, deadline, rng, max_iters=1500)
        _try_candidate(a_sub)

        # Annealed hill‑climbing from the sub‑gradient result
        a_hc, _ = _anneal_hillclimb(a_sub, deadline, rng)
        _try_candidate(a_hc)

        # Early exit if time is almost exhausted
        if time.time() >= deadline - 0.2:
            break

    # ------------------------------------------------------------------
    # 3. Return the best sequence (as plain Python list) and report score
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else a_initial.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return final_seq
