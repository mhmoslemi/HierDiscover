def project_onto_simplex(v):
    # Implements algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1

"""
Heuristic optimizer for ``evaluate_sequence``.

The objective (for a non‑negative vector *a* with sum = 1) is

    f(a) = 2·n·max_k (a * a)[k]

where ``*`` denotes the full (linear) convolution.  The code
* (i) builds an initial random probability vector,
* (ii) runs an Adam‑style projected sub‑gradient descent on the exact
    max‑convolution objective,
* (iii) applies an aggressive mass‑transfer hill‑climb (the same idea as
    the previous baseline but with a few tuned parameters),
* (iv) finishes with a short simulated‑annealing perturbation loop that
    explores the neighbourhood of the current best solution.

All steps are stopped as soon as the global time budget (default 1000 s)
is exhausted.  The best vector found is returned as a plain Python list,
which can be passed directly to the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using an FFT (real input).
    Returns an array of length 2·len(a)−1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # Numerical noise may generate tiny negative values – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` when the vector is already
    non‑negative and its L1‑norm is (practically) 1.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * np.max(conv))


def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry conv[k] w.r.t. a.
    For non‑negative a the derivative is 2·a_{k‑i} on the feasible range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.  Implements the algorithm of
    Duchi et al. (2008) in O(n log n) time.
    """
    if v.ndim != 1:
        raise ValueError("project_onto_simplex expects a 1‑D array")
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise once more
    w_sum = w.sum()
    if w_sum <= 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum

# Optimisation stages
# ----------------------------------------------------------------------


def _adam_subgradient(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """
    Projected Adam–style sub‑gradient descent on
    f(a) = 2·n·max(conv(a,a)).
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # 1. compute the lag that attains the current maximum
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # 2. sub‑gradient of that lag
        grad = _gradient_of_lag(a, k)

        # 3. Adam update (the learning rate decays like 1/√t)
        lr = 0.5 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_trial = a - step
        a_trial = np.clip(a_trial, 0.0, None)
        a_trial = _project_onto_simplex(a_trial)

        trial_val = _objective(a_trial)

        # Accept only improving steps (standard projected sub‑gradient behaviour)
        if trial_val < cur_val:
            a = a_trial
            cur_val = trial_val

        # Keep a record of the best ever seen
        if trial_val < best_val:
            best_val = trial_val
            best_a = a_trial.copy()
            if t % 200 == 0:
                print(f"[adam] iter {t:5d}  best = {best_val:.6f}")

    return best_a, best_val


def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 4000) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climbing that repeatedly moves a fraction of the most
    damaging mass to two low‑mass destinations.
    Very similar to the baseline but with a slightly larger initial transfer
    fraction and a more permissive adaptation schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)

    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.3          # initial fraction of the donor mass to relocate
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that compose the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass to shift
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-8)
            continue

        new_a = a.copy()
        new_a[i] -= delta
        if i != j:
            new_a[j] -= delta

        # choose two low‑mass destinations, avoiding the donors
        low_mask = new_a < np.mean(new_a)
        cand = np.where(low_mask)[0]
        forbidden = {i, j}
        cand = np.setdiff1d(cand, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        # re‑normalise (should stay ≈1, but we keep the safety net)
        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-8)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.6)
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-8)
            no_improve += 1

    return best_a, best_val


def _annealing_hillclimb(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing walk that perturbs the current vector.
    Two move types are used:
      1) transfer a small amount of mass between a random pair (i, j);
      2) the “high‑pair → low‑pair” move from the mass‑transfer phase,
         but accepted probabilistically.
    The temperature decays geometrically.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    # temperature is a small fraction of the current objective
    T = max(1e-4, cur_val * 0.01)
    decay = 0.9995

    while time.time() < deadline:
        move_type = rng.integers(2)

        if move_type == 0:
            # Simple i→j transfer
            i, j = rng.choice(n, size=2, replace=False)
            # transfer a random fraction of the donor's mass
            max_delta = a[i] * 0.2
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # High‑pair mass‑transfer (same pattern used in the hill‑climb)
            conv = _full_convolution(a)
            maxb = conv.max()
            ks = np.flatnonzero(conv == maxb)
            k = int(rng.choice(ks))

            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
            j_range = k - i_range
            contrib = a[i_range] * a[j_range]
            idx = int(np.argmax(contrib))
            i = int(i_range[idx])
            j = int(j_range[idx])

            delta = min(a[i], a[j]) * 0.1
            if delta <= 0.0:
                continue

            new_a = a.copy()
            new_a[i] -= delta
            if i != j:
                new_a[j] -= delta

            low_mask = new_a < np.mean(new_a)
            cand = np.where(low_mask)[0]
            forbidden = {i, j}
            cand = np.setdiff1d(cand, list(forbidden), assume_unique=True)
            if cand.size < 2:
                cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
            dest = rng.choice(cand, size=2, replace=False)
            p, q = int(dest[0]), int(dest[1])
            new_a[p] += delta
            new_a[q] += delta

        # No need to renormalise – the total mass is preserved by construction.
        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp((cur_val - new_val) / max(T, 1e-12))
            if rng.random() < prob:
                a = new_a
                cur_val = new_val

        # temperature schedule
        T *= decay
        if T < 1e-12:
            T = 1e-12

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.

    The algorithm combines three phases:
      1. Adam sub‑gradient descent on the exact max‑convolution objective,
      2. Aggressive mass‑transfer hill‑climb,
      3. Simulated‑annealing perturbations.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Determine the problem dimension.
    # ------------------------------------------------------------------
    default_n = 8000                         # a larger support gives more freedom
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n_old = int(old.shape[0])
        n = max(default_n, n_old)
        a0 = np.zeros(n, dtype=float)
        # copy the old construction into the first part of the vector
        a0[:n_old] = np.maximum(old, 0.0)
        # fill the remaining entries with random noise to avoid a pure zero tail
        if n > n_old:
            a0[n_old:] = rng.random(n - n_old)
        if a0.sum() == 0.0:
            a0 = rng.random(n)
    except Exception:
        n = default_n
        a0 = rng.random(n)

    # Normalise onto the probability simplex (sum = 1)
    a0 = _project_onto_simplex(a0)

    print(f"[init] n={n}, initial objective = {_objective(a0):.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam sub‑gradient descent.
    # ------------------------------------------------------------------
    a_adam, val_adam = _adam_subgradient(a0, deadline, rng, max_iters=12000)
    print(f"[adam] best value after descent = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Aggressive mass‑transfer hill‑climb.
    # ------------------------------------------------------------------
    a_hill, val_hill = _mass_transfer_hillclimb(a_adam, deadline, rng)
    print(f"[massclimb] best value after hill‑climb = {val_hill:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing refinement.
    # ------------------------------------------------------------------
    a_final, val_final = _annealing_hillclimb(a_hill, deadline, rng)
    print(f"[anneal] best value after annealing = {val_final:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (the external wrapper will also call evaluate_sequence,
    # but we print it here for debugging).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_final.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a_final.tolist()
