    """Project vector v onto simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

    rho = ind[cond].max()

def _lp_step(a, M):
    n = len(a)
    # Build A matrix
    rows = 2*n - 1
    # Use dense numpy array
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i+n, i] = a
    # set bounds and objective
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    ub = np.full(rows, M)
    # call linprog
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

"""
Propose a step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Coarse initialization** – a short Beta family sweep (including very
   small exponents) is used to get a good non‑negative shape on a modest
   size (≈250).  If a previous best (`height_sequence_1`) exists we
   interpolates it instead.

2. **Linear‑program refinement** – for the current shape `a` we solve the LP

        maximise   Σ g_i
        subject to conv(a, g) ≤ M     (M = max(conv(a, a)))
                  g_i ≥ 0

   The solution `g` tells where extra mass can be added without raising the
   current maximal convolution entry.  We mix `a ← (1‑t) a + t g` with a few
   trial step‑sizes `t` and keep the best improvement.  The LP is tiny
   (n≈250) so many restarts fit easily into the time budget.

3. **Upscaling** – the refined small vector is interpolated to the target
   length (5000) and normalised.

4. **Fine optimisation** – the large vector is polished by two standard
   phases used in the previous state‑of‑the‑art code:
   * Adam optimisation on a smooth‑max surrogate (soft‑max temperature
     annealed logarithmically).
   * Simulated‑annealing hill‑climbing that moves a tiny amount of mass
     from indices that heavily contribute to the current maximal
     convolution entry to indices that contribute little.

The three stages together respect the 1000 s budget (≈15 % LP, ≈30 % Adam,
≈55 % hill‑climb).  All intermediate vectors are kept non‑negative and
renormalised, therefore the final call to ``evaluate_sequence`` is safe.
"""

# Convolution helpers

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself (length 2·len‑1)."""
    # For the small‑n phases a simple np.convolve is fast enough.
    return np.convolve(x, x, mode="full")


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (efficient for large n)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient – used inside Adam.
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Smooth approximation of max(conv(a,a)) and its gradient.
    Returns:
        smooth_val – the smooth max (not multiplied by 2·n)
        grad_a      – gradient of 2·n·smooth_max w.r.t. a
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # gradient of Σ p_k * b_k w.r.t a.
    # ∂b_k/∂a_i = a_{k-i} + a_i (mirrored term)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1 : 2 * n - 1]          # n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 4·n from 2·n·max
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a

# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a smooth approximation of the objective.
    Returns the best (a, value) seen before *deadline*.
    """
    # initialise logits for the soft‑max parametrisation
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.001
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 25_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # annealing schedules
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth‑max and gradient
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)

        # chain rule through the soft‑max (w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        # restart on stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restarting after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass from indices that
    heavily contribute to the current maximal convolution entry to indices
    that contribute little.  Accepts uphill moves with a Metropolis rule.
    """
    a = a0.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.012, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (probability ∝ contribution)
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index (biased towards low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric sweep of a two‑parameter Beta‑family (including very small exponents)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a flexible Beta family a_i ∝ (i+1)^{p‑1}(n‑i)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # exponents – combine a dense linear part near 1 with a log‑space tail
    p_vals = np.concatenate([
        np.linspace(0.05, 0.9, 9),          # fine near 0
        np.geomspace(0.9, 6.0, 7)          # log‑space up to 6
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑program refinement (tiny dimensionality)
# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, M: float) -> np.ndarray | None:
    """Solve the LP maximise Σg subject to conv(a,g) ≤ M, g ≥ 0."""
    if not _HAS_LP:                     # pragma: no cover
        return None
    n = a.shape[0]
    rows = 2 * n - 1
    # Build the convolution matrix A (dense – tiny size)
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = a
    c = -np.ones(n)                     # maximise Σg  → minimise –Σg
    bounds = [(0, None) for _ in range(n)]
    ub = np.full(rows, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x


def _lp_refine(a0: np.ndarray,
               deadline: float,
               start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described above, mixing the solution into the
    current vector.  The loop stops when no improvement is observed or when
    the allotted time expires.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current maximal convolution entry
        M = _full_convolution(a).max()

        g = _lp_step(a, M)
        if g is None:
            # LP could not be solved – give up on this stage
            break

        improved = False
        # try a few mixing weights (larger first)
        for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            a_new = (1.0 - t) * a + t * g
            s = a_new.sum()
            if s <= 0.0:
                continue
            a_new /= s
            val = _objective(a_new)
            if val < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val
                improved = True
                print(f"[LP {iteration:5d}] t={t:.3f} → val={best_val:.6f}")
                break

        iteration += 1
        if not improved:
            # no further descent – stop the LP refinement
            break

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Working sizes
    # ------------------------------------------------------------------
    n_small = 250          # tiny size for the LP stage
    n_large = 5000         # final size (good trade‑off between quality & speed)

    # ------------------------------------------------------------------
    # Initialise a on the small grid
    # ------------------------------------------------------------------
    a_small = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n_small)
        old_x = np.linspace(0.0, 1.0, old_len)
        a_small = np.interp(new_x, old_x, old)
        a_small = np.maximum(a_small, 0.0)
        a_small /= a_small.sum()
        print("[init] interpolated previous solution (small grid)")
    except Exception:
        pass

    if a_small is None:
        # Beta family sweep on the small grid
        a_small, val = _beta_family_best(n_small)
        print(f"[init] beta‑family sweep on {n_small} points → val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Linear‑program refinement (at most 15 % of the total budget)
    # ------------------------------------------------------------------
    lp_deadline = start_time + 0.15 * budget_s
    if _HAS_LP:
        a_small, lp_val = _lp_refine(a_small, lp_deadline, start_time)
        print(f"[LP] finished – best small‑grid value {lp_val:.6f}")
    else:
        print("[LP] skipped – scipy not available")

    # ------------------------------------------------------------------
    # Upscale to the large problem size
    # ------------------------------------------------------------------
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(x_large, x_small, a_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    best_a = a_large.copy()
    best_val = _objective(best_a)
    print(f"[scale] up‑scaled to {n_large} → val={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈30 % of budget)
    # ------------------------------------------------------------------
    adam_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _adam_optimize(best_a, n_large,
                                      deadline=adam_deadline,
                                      start_time=time.time(),
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n_large,
                                         deadline=deadline,
                                         start_time=time.time(),
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
