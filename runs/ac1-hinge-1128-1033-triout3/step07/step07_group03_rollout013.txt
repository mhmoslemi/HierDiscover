"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm consists of several cheap and cheap‑ish phases that
fit comfortably into the 1000 s budget.

1. **Parametric start** – try a family of symmetric Beta‑like shapes
   a_i ∝ x^{p‑1}(1‑x)^{p‑1} (p∈(0,1] gives a U‑shape, p>1 a mound).
   The best shape is kept as the starting point.

2. **Smooth surrogate optimisation** – minimise a smooth approximation of
   ``max(conv)`` (log‑sum‑exp) with L‑BFGS‑B.  Several random restarts
   and a decreasing ε‑schedule are used; after each run the candidate
   is optionally reordered by the anti‑symmetric heuristic.

3. **Multi‑peak mirror‑descent** – a sub‑gradient descent on the true
   objective.  At every iteration the *k* largest convolution peaks are
   located (k≈10) and a multiplicative mirror‑descent step
   ``a ← a·exp(‑α·g)`` is performed, where ``g_i = Σ_{peak} a_{peak‑i}``.
   The step‑size α decays slowly and the vector is projected onto the
   probability simplex after every update.  Every few hundred iterations
   the sequence is forced to be symmetric (a←(a+a[::-1])/2) and the
   anti‑symmetric re‑ordering is tried.

4. **Hill‑climbing refinement** – random swaps or tiny mass transfers
   (simulated‑annealing style) are accepted only if they strictly improve
   the objective.

All heavy convolutions are performed with an FFT based routine
(`_full_convolution`) – this is much faster than ``np.convolve`` used
inside ``evaluate_sequence``.  The final candidate is evaluated with the
provided ``evaluate_sequence`` before being returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global handles used by the smooth‑surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a real vector with itself (full length).
    Implemented via FFT for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Small negative rounding errors are clamped to zero.
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest values with the smallest when the sequence is reversed.
    This heuristic often flattens the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce exact left‑right symmetry."""
    return (a + a[::-1]) * 0.5


def _fast_objective(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence for a vector that already sums to 1.
    Returns 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)      (lies on the probability simplex)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution a*a
    conv = _full_convolution(a)         # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (derived analytically)
    weights = np.exp(shifted)
    p = weights / weights.sum()               # probability over lag indices

    # convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries are the needed derivative w.r.t each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # factor 4 = 2 (from max) * 2 (ordered pairs)

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Thin wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run L‑BFGS on the smooth surrogate starting from the soft‑max
    parameters ``w_init``.  Returns the best (a, value) pair found before
    the deadline.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule: from 5·10⁻³ down to 10⁻⁹ (geometric)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=9)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 30.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _fast_objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = _fast_objective(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback – should never happen
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = _fast_objective(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Multi‑peak exponential mirror‑descent (core of the search)
# ----------------------------------------------------------------------
def _gradient_descent_multi(a_start: np.ndarray, deadline: float):
    """
    Mirror descent that reduces the *k* largest convolution peaks.
    Returns the best (a, value) discovered before ``deadline``.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = _fast_objective(a)
    print(f"[gd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.8          # initial step‑size (multiplicative)
    top_k = 10           # number of peaks used for the gradient
    decay_rate = 0.98    # decay of α every 2000 iterations
    sym_interval = 2000  # how often to enforce symmetry

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_val = conv.max()

        # indices of the ``top_k`` largest convolution entries
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]   # sorted descending
        else:
            idxs = np.arange(conv.size)

        # gradient contribution from the selected peaks
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            jdx = k - idx_range
            g[idx_range] += a[jdx]

        # scheduled step‑size
        alpha = alpha0 * (decay_rate ** (it // 2000))

        # multiplicative update (mirror‑descent)
        a = a * np.exp(-alpha * g)

        # projection onto the probability simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:                     # guard against complete under‑flow
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        # occasional re‑ordering / symmetrising
        if it % 50 == 0:
            cur_val = _fast_objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd iter {it}] new best = {best_val:.6f} (max conv {max_val:.8e})")

            if it % 500 == 0:
                # anti‑symmetric re‑ordering
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_objective(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] improved to {best_val:.6f}")

        if it % sym_interval == 0 and it > 0:
            a = _symmetrize(a)   # force exact left–right symmetry
            a /= a.sum()          # renormalise (tiny numeric drift)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (random swaps / mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Local random search: swaps (20 %) and mass transfers (80 %).
    Accepts only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_objective(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between flexibility
    # and speed (convolutions are FFT‑based and cheap).
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweeps (Beta‑like family)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)   # uniform start
    best_val = _fast_objective(best_a)
    print(f"[init] uniform evaluation = {best_val:.6f}")

    # try a few Beta‑like shapes, p in (0.3, 1.4)
    for p in np.linspace(0.3, 1.4, num=12):
        if time.time() >= deadline - 200:
            break
        # Symmetric Beta density: x^{p-1} (1-x)^{p-1}
        x = (np.arange(n) + 0.5) / n
        # add a tiny epsilon to avoid division by zero when p<1
        eps = 1e-12
        a = np.power(x + eps, p - 1) * np.power(1.0 - x + eps, p - 1)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            continue
        a /= a_sum
        val = _fast_objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[beta p={p:.3f}] new best = {best_val:.6f}")

    # a couple of random Dirichlet starts for diversity
    for _ in range(3):
        if time.time() >= deadline - 200:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = _fast_objective(a_rand)
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate optimisation (L‑BFGS) with several restarts
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 150 and restart < 8:
        restart += 1
        w_start = np.log(best_a + 1e-12)
        w_start += rng.normal(scale=0.3, size=n)   # modest perturbation
        a_opt, val_opt = _l_bfgs_optimize_from_w(w_start, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[restart {restart}] new best after L‑BFGS = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multi‑peak exponential mirror‑descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        a_gd, val_gd = _gradient_descent_multi(best_a, deadline)
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[gradient descent] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation with the official function before returning
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    return best_a.tolist()
