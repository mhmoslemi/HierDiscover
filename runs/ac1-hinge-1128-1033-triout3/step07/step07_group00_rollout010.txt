    w = np.log(a0 + 1e-12)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t=0
    no_improve=0
    while time.time() < deadline:
        # compute a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        # conv
        b = _full_convolution(a)
        k_max = np.argmax(b)
        max_b = b[k_max]
        cur_val = 2.0*n*max_b
        # update best if improved
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve=0
            print(f"[AdamSubgrad {t}] new best = {best_val:.6f}")
            if best_val <= target_val: break
        else:
            no_improve+=1

        # gradient g wrt a

"""
Convolution‑maximising step function optimizer.

The programme searches a probability vector `a` (non‑negative, Σa=1) that
minimises

    2·n·max_k (a * a)_k

where `*` denotes linear convolution.  The objective is exactly the value
returned by `evaluate_sequence` (the security wrapper enforces a sum≈1).

Main ideas
----------
* A fast FFT based convolution helper.
* **Phase A** – Adam optimisation on the *exact* sub‑gradient of the
  non‑smooth objective (gradient of the maximal convolution entry).
  Variables are logits `w` (log‑softmax), guaranteeing the simplex
  constraints.
* **Phase B** – Deterministic hill‑climbing that moves a small amount of
  mass from the index contributing most to the maximal entry towards the
  index contributing the *least*.  Incremental updates of the full
  convolution allow millions of cheap moves.
* **Phase C** – Short simulated annealing to escape possible plateaus.
* Final uniform mixing sweep (convex combination with a tiny uniform
  component) often reduces the peak a little further.

The whole routine respects the 1000 s wall‑clock budget and returns the
best vector discovered.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT convolution helpers
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) assumed = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())

# Rich parametric initialisation (deterministic families + few randoms)
# ----------------------------------------------------------------------
def _initial_candidate(n: int) -> Tuple[np.ndarray, float]:
    """Return a promising starting vector (normalised) and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # --------------------------------------------------------------
    # Simple deterministic families
    # --------------------------------------------------------------
    # linear decreasing (left → right)
    try_candidate(np.arange(n, 0, -1, dtype=float))

    # linear U‑shape (large at both ends, small in the middle)
    idx = np.arange(n)
    lin_u = np.maximum(idx + 1, n - idx)
    try_candidate(lin_u)

    # quadratic decreasing (left side)
    try_candidate((np.arange(n, 0, -1, dtype=float) ** 2))

    # quadratic U‑shape
    quad_u = (np.arange(n) + 1) ** -2 + (np.arange(n, 0, -1) + 0) ** -2
    try_candidate(quad_u)

    # half‑wave sine (symmetric)
    try_candidate(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))

    # cosine with peaks at the ends
    try_candidate(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # power‑law symmetric (U‑shape) – extend to larger exponents
    for p in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_candidate(a)

    # exponential left‑hand side
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995,
              0.99, 0.98, 0.95, 0.9):
        a = r ** np.arange(n, dtype=float)
        try_candidate(a)

    # symmetric (U‑shaped) exponential
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997,
              0.996, 0.995, 0.99, 0.98, 0.95, 0.9):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_candidate(a)

    # truncated exponential (mass only on a prefix)
    for frac in (0.10, 0.15, 0.20, 0.25, 0.30):
        K = max(1, int(frac * n))
        for r in (0.99, 0.98, 0.95, 0.90):
            a = np.zeros(n, dtype=float)
            a[:K] = r ** np.arange(K, dtype=float)
            try_candidate(a)

    # random Dirichlet draws – a few quick samples
    rng = np.random.default_rng(12345)
    for _ in range(20):
        a = rng.dirichlet(np.ones(n))
        try_candidate(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam optimisation on the exact sub‑gradient
# ----------------------------------------------------------------------
def _adam_subgrad_phase(a0: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the *exact* objective using the sub‑gradient of the
    maximal convolution entry.  Parameters are kept on the log‑softmax
    (logits) simplex to guarantee feasibility.
    """
    w = np.log(a0 + 1e-12)          # logits, Σsoftmax(w) = 1

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.005
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- a on the simplex -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ----- exact objective -----
        b = _full_convolution(a)
        k_max = int(np.argmax(b))
        max_b = b[k_max]
        cur_val = 2.0 * n * max_b

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- sub‑gradient of the objective wrt a -----
        # derivative of max(conv) = 2 * a_{k_max - i} (if index in range)
        rev = np.zeros_like(a)
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        inds = np.arange(low, high + 1, dtype=int)
        rev[inds] = a[k_max - inds]               # a_rev[i] = a_{k_max-i}
        g_a = 4.0 * n * rev                        # gradient of 2·n·max(conv)

        # ----- chain rule through soft‑max (log‑softmax) -----
        dot = np.dot(g_a, a)
        g_w = a * (g_a - dot)

        # ----- learning‑rate schedule (exponential) -----
        progress = (now - start) / (deadline - start)
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_w
        v = beta2 * v + (1.0 - beta2) * (g_w * g_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to avoid stagnation
        if np.random.rand() < 0.007:
            w += np.random.normal(scale=0.007, size=n)

        # restart after long stagnation
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            # re‑initialise around the best seen
            w = np.log(best_a + 1e-12) + np.random.normal(scale=0.3, size=n)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – deterministic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """
    Fast local search that moves a small amount of mass from the index that
    contributes most to the current maximal convolution entry to the
    index that contributes the *least*.  Incremental updates keep the
    per‑iteration cost O(n) (vectorised slice updates).
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # shrinking transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # -- source: index with largest contribution to the maximal entry --
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        idx_range = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx_range] * a[k_max - idx_range]
        src = idx_range[np.argmax(contrib)]

        if a[src] <= 1e-15:
            # nothing to move – skip this iteration
            continue

        # -- target: index with smallest contribution (excluding src) --
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        prod = np.full(n, np.inf, dtype=float)
        prod[rev_mask] = a[rev_mask] * a[rev_idx[rev_mask]]
        prod[src] = np.inf                     # do not select the source
        tgt = int(np.argmin(prod))

        # amount of mass to transfer
        delta = a[src] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        a_old = a.copy()
        a_new = a_old.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        # ----- incremental convolution update -----
        b_new = b.copy()
        b_new[tgt:tgt + n] += 2.0 * delta * a_old
        b_new[src:src + n] -= 2.0 * delta * a_old
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:          # strict improvement
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – small random perturbation around the best
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a

# Phase C – brief simulated‑annealing refinement (tiny transfers)

def _annealing_phase(a_start: np.ndarray,

    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    T_start, T_end = 5e-4, 5e-8
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source selection (same bias as hill‑climber)
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[k_max - idx]
        src = idx[np.argmax(contrib)]

        if a[src] <= 1e-15:
            continue

        # uniform random target (different from src)
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        # tiny transfer
        delta = a[src] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        a_old = a.copy()
        a_new = a_old.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        # incremental conv update
        b_new = b.copy()
        b_new[tgt:tgt + n] += 2.0 * delta * a_old
        b_new[src:src + n] -= 2.0 * delta * a_old
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # small shake‑up
            noise = np.random.normal(scale=0.004, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val

# Final uniform mixing sweep

def _mix_with_uniform(a: np.ndarray, n: int) -> Tuple[np.ndarray, float]:
    """Try tiny convex combinations with the uniform vector."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
