def project_onto_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    b = convolve(a, a) (full)
    c = 1.0/(2*n - 1)

def project_simplex(v):
    # v: numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _l2_descent(a_init, n, start, deadline):
    a = a_init.copy()
    best_a = a.copy()
    best_val = _objective_value(a)
    eta0 = 0.5
    c_const = 1.0/(2*n - 1)
    while time.time() < deadline:
        # compute convolution
        b = _full_convolution(a)
        d = b - c_const
        # compute gradient
        a_rev = a[::-1]
        # conv d with a_rev
        out_len = len(d) + n -1
        size = 1 << (out_len -1).bit_length()
        fd = np.fft.rfft(d, size)
        fa_rev = np.fft.rfft(a_rev, size)
        conv = np.fft.irfft(fd * fa_rev, size)[:out_len]
        grad = 4.0 * conv[n-1:2*n-1]   # slice length n
        # step size schedule
        progress = (time.time() - start) / (deadline - start)
        eta = eta0 * (1 - progress) + 0.01 * progress   # decays to 0.01
        a_new = a - eta * grad
        # project onto simplex
        a = _project_simplex(a_new)
        cur_val = _objective_value(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[L2 {int(time.time()-start)}s] new best = {best_val:.6f}")
    return best_a, best_val

"""
search for a non‑negative step‑function minimizing `evaluate_sequence`.

Algorithm
---------
1.  Build a modest library of parametric families and keep the best
    (these give a good starting point).
2.  Perform a *L2‑descent*: minimise the squared deviation of the
    self‑convolution from the flat target `c = 1/(2·n‑1)`.  The gradient
    of `∑(b_k‑c)²` w.r.t. the vector `a` is computed by a single FFT
    (cost  O(n log n)).  After each gradient step the vector is projected
    onto the probability simplex (non‑negative, sums to 1).
3.  Refine the result with the targeted hill‑climber from the previous
    state‑of‑the‑art (exact objective, small mass transfers, stochastic
    restarts).
4.  A short simulated‑annealing phase helps the search escape shallow
    plateaus.
5.  Finally, try a few convex mixtures with the uniform vector – this
    sometimes reduces the maximal convolution a little more.

The whole routine respects the 1000 s budget; the longest part is the
L2‑descent (≈ ½ of the time), the remaining phases use the leftover
budget.  All helper functions are top‑level – no closures are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Projection onto the simplex (Σx = 1, x_i ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric initialisation – a modest library of shapes
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector (normalised) and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n or np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ----- deterministic families ------------------------------------------------
    # linear decreasing (left → right)
    try_candidate(np.arange(n, 0, -1, dtype=float))

    # linear U‑shape (large at both ends, small in the middle)
    idx = np.arange(n)
    lin_u = np.maximum(idx + 1, n - idx)
    try_candidate(lin_u)

    # quadratic decreasing (left side)
    try_candidate((np.arange(n, 0, -1, dtype=float) ** 2))

    # quadratic U‑shape
    quad_u = (np.arange(n) + 1) ** -2 + (np.arange(n, 0, -1) + 0) ** -2
    try_candidate(quad_u)

    # half‑wave sine (symmetric)
    try_candidate(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))

    # cosine with peaks at the ends
    try_candidate(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # power‑law symmetric (U‑shape)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_candidate(a)

    # exponential left‑hand side
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996,
              0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        try_candidate(a)

    # symmetric (U‑shaped) exponential
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996,
              0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_candidate(a)

    # truncated exponential (mass only on a prefix)
    for frac in (0.10, 0.15, 0.20, 0.25, 0.30):
        K = max(1, int(frac * n))
        for r in (0.99, 0.98, 0.95, 0.90):
            a = np.zeros(n, dtype=float)
            a[:K] = r ** np.arange(K, dtype=float)
            try_candidate(a)

    # a few random Dirichlet draws (different concentration levels)
    rng = np.random.default_rng(12345)
    for conc in (0.3, 0.7, 1.0, 2.0):
        for _ in range(4):
            a = rng.dirichlet(np.full(n, conc))
            try_candidate(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – L2‑descent (minimise ∑(conv‑c)²)
# ----------------------------------------------------------------------
def _l2_descent(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the L2 loss
        L(a) = Σ_k ( (a*a)_k – c )² ,   c = 1/(2·n‑1)
    The gradient is 4·conv( d , a_rev ) where d = (a*a) – c.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_value(a)

    c = 1.0 / (2 * n - 1)               # target flat convolution
    eta_start, eta_end = 0.8, 0.02     # learning‑rate schedule
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # full self‑convolution
        b = _full_convolution(a)            # length 2·n‑1
        d = b - c

        # gradient: 4·conv(d , a_rev)  (central slice of length n)
        a_rev = a[::-1]
        out_len = d.shape[0] + n - 1        # = 3·n‑2
        size = 1 << (out_len - 1).bit_length()
        fd = np.fft.rfft(d, size)
        fa_rev = np.fft.rfft(a_rev, size)
        conv = np.fft.irfft(fd * fa_rev, size)[:out_len]
        grad = 4.0 * conv[n - 1: 2 * n - 1]  # slice length n

        # linearly annealed step size
        progress = (now - start) / (deadline - start)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # gradient step + projection onto simplex
        a = _project_simplex(a - eta * grad)

        cur_val = _objective_value(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[L2   {it:6d}] new best = {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer hill‑climber that directly minimises the L∞ objective.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # schedule for the transferred fraction
    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contributions to the active (maximal) entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index: biased by contribution size
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – usually where the reverse contribution is tiny
        if np.random.rand() < 0.35:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                # fallback – uniform draw excluding i
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (exact)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:          # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill  {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and restart from current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill  {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val

# Phase C – short simulated‑annealing refinement
# ----------------------------------------------------------------------
def _annealing_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Metropolis‑style search that occasionally accepts worsening moves.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    T_start, T_end = 5e-4, 5e-8
    IMPROVE_LIMIT = 120_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source – same bias as hill‑climber
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # uniform target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer – ≤ 0.5 % of source mass
        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneel {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # small random shake
            noise = np.random.normal(scale=0.004, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Small post‑processing sweep – mixing with uniform
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combinations a' = (1‑α)·a + α·uniform for several α."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020, 0.050):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
