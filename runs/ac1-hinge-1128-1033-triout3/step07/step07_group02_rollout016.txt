     (linear coefficients from the convolution plus the tiny quadratic
     δ²‑terms are taken into account).  Apply δ (≈90 % of the safe
     bound) in O(n) time using incremental convolution updates.
   * Accept the move, update the best solution, and repeat.
   * If no improvement is observed for many iterations a small
     random perturbation is applied to escape plateaus.
5. Return the best vector found (scaled to sum = 1).

The code respects the 1000 s budget, keeps a 0.5 s safety margin for the
final ``evaluate_sequence`` call, and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and gradient (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Soft‑max approximation of max(conv(a,a)) and its gradient w.r.t. a.
    Returns (smooth_max, grad_a) where grad_a already includes the factor
    2·n of the true objective.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # gradient of Σ p_k * b_k w.r.t. a  (see derivation in the prompt)
    p_rev = p[::-1]
    conv_ab = np.fft.irfft(np.fft.rfft(a, 1 << (2 * n - 1).bit_length()) *
                           np.fft.rfft(p_rev, 1 << (2 * n - 1).bit_length()))[:2 * n - 1]
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 4n from 2·n·max
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a smooth approximation of the objective.
    Returns the best (a, value) seen before *deadline*.
    """
    n = a0.shape[0]
    # logits for soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.001
    eps_start, eps_end = 1.0, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedule for temperature (eps) and learning rate (lr)
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max and gradient
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)

        # chain rule through the soft‑max (w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.02, size=n)

        # restart on stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restarting after {idle} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (large safe mass moves)
# ----------------------------------------------------------------------
def _deterministic_hillclimb(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             best_a: np.ndarray,
                             best_val: float,
                             target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Repeatedly lower the exact maximal convolution entry by moving a
    maximal‑contributor mass chunk to a minimal‑contributor slot.  The
    move size is chosen as the largest safe δ that cannot raise any
    other convolution entry above the current maximum (including the
    ε²‑terms).  Occasional random perturbations escape plateaus.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)               # length 2n‑1
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    no_improve = 0
    IMPROVE_LIMIT = 25000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]

        # source = index with largest contribution, target = smallest
        i = int(np.argmax(contrib))
        if a[i] <= 1e-14:                     # safety fallback
            i = np.random.randint(n)

        # avoid picking the same index for target
        tmp = contrib.copy()
        tmp[i] = np.inf
        j = int(np.argmin(tmp))
        if j == i:
            j = (i + 1) % n

        # ------------------------------------------------------------------
        # compute the largest admissible mass transfer δ
        # linear coefficient c = 2*(shift_j - shift_i)
        # ------------------------------------------------------------------
        c = np.zeros_like(b)
        c[j:j + n] += 2.0 * a
        c[i:i + n] -= 2.0 * a

        pos_mask = c > 1e-15
        if np.any(pos_mask):
            # avoid division by zero; cur_max - b[pos] is always ≥0
            delta_lin = np.min((cur_max - b[pos_mask]) / c[pos_mask])
        else:
            delta_lin = a[i]

        # quadratic constraints (δ² terms at 2*i, 2*j)
        delta_quad = a[i]  # start with a[i] as a trivial upper bound
        if b[2 * j] < cur_max:
            delta_quad = min(delta_quad, np.sqrt(cur_max - b[2 * j]))
        if b[2 * i] < cur_max:
            delta_quad = min(delta_quad, np.sqrt(cur_max - b[2 * i]))

        delta_limit = min(a[i], delta_lin, delta_quad)
        # shrink a tiny bit for numerical safety
        delta = 0.9 * delta_limit

        if delta < 1e-12:
            # stagnation – apply a small global noise and recompute
            a += np.random.normal(scale=1e-4, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            no_improve = 0
            continue

        # ------------------------------------------------------------------
        # Apply the mass transfer (exact convolution update, O(n))
        # ------------------------------------------------------------------
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        b[2 * j] += delta * delta
        b[2 * i] += delta * delta
        b[i + j] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        cur_max = b.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        if no_improve >= IMPROVE_LIMIT:
            # re‑inject a small perturbation around the current best
            a = best_a.copy()
            a += np.random.normal(scale=0.02, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Beta‑family sweep (more extensive than the reference)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # log‑spaced grid for the exponents
    p_vals = np.concatenate([np.linspace(0.05, 0.9, 9),
                             np.geomspace(0.9, 6.0, 9)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    total = len(p_vals) * len(q_vals)
    cnt = 0
    for p in p_vals:
        for q in q_vals:
            cnt += 1
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
            # keep the sweep short enough to stay within the budget
            if cnt % 500 == 0:
                pass
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – choose a few‑thousand points (balance quality / time)
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – reuse a previous solution if present.
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(globals().get('height_sequence_1'), dtype=float)
        if old.ndim == 1 and old.size > 0:
            old_len = old.shape[0]
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – Beta‑family sweep (adds a solid deterministic seed)
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    try:
        shape_a, shape_val = _beta_family_best(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] beta‑family sweep improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # fallback random start
    if best_a is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_a, best_val = a, _objective(a)
        print("[init] random start, value=", best_val)

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈45 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _adam_optimize(best_a,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # Phase B – Deterministic hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _deterministic_hillclimb(best_a,
                                                deadline=deadline,
                                                start_time=phase_a_deadline,
                                                best_a=best_a,
                                                best_val=best_val,
                                                target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
