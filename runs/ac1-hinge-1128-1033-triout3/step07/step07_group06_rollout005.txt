"""
Propose a low‑value step‑function for `evaluate_sequence`.

Algorithm
---------
1. Initialise a probability vector `a` (non‑negative, sum = 1).  If a previous
   construction is present in the global variable `height_sequence_1` it is
   reused, otherwise a random Dirichlet draw is used.

2. Smooth gradient descent (Adam) on a smoothed version of the objective:
   F(a) = 2·n·max_k (a*a)[k].  The max is approximated by a soft‑max with a
   temperature that grows during the run, providing gradients from many
   convolution entries instead of only the current maximum.

   The gradient of the soft‑max objective can be written in closed form as
   g = 2·(w * a) where `w` are the soft‑max weights and `*` denotes the
   (full) convolution.  After each update the vector is projected onto the
   probability simplex.

3. After the time‑limited gradient phase a short aggressive hill‑climbing
   phase (mass‑transfer moves) refines the best vector found.

4. The final vector is returned as a plain Python list; its score is also
   printed using the provided `evaluate_sequence` function.

The whole routine respects the user‑supplied time budget (default 1000 s)
and never exceeds it.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray, y: np.ndarray | None = None) -> np.ndarray:
    """
    Full (linear) convolution of real vectors `x` and `y` using FFT.
    If `y` is omitted, computes the self‑convolution `x*x`.
    Returns a 1‑D array of length len(x)+len(y)-1.
    """
    if y is None:
        y = x
    n = x.shape[0] + y.shape[0] - 1
    size = _next_pow2(n)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    # eliminate tiny negative artefacts caused by floating point errors
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` when `sum(a) == 1`.
    The function is 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * np.max(conv)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0, Σ_i x_i = 1 }.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # Find the rho index (last index where the condition holds)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _mass_transfer_hillclimb(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_no_improve: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climbing that repeatedly shifts a fraction of the mass
    from the most damaging pair of indices to two low‑mass destinations.
    Stops when the time budget expires or after `max_no_improve` successive
    unsuccessful attempts.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2  # initial fraction of mass to move
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices contributing to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # mass we are willing to relocate
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-9)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # Choose two low‑mass destinations distinct from the donors
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            # fallback to any two different positions
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        # Re‑normalise to the simplex
        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-9)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-9)
            no_improve += 1

    return best_a, best_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    `evaluate_sequence` value within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise a probability vector `a`.
    # ------------------------------------------------------------------
    default_n = 4000
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        a = old.copy()
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = rng.random(n)
        a /= a.sum()
        print(f"[init] Re‑using previous sequence, n={n}")
    except Exception:
        n = default_n
        a = rng.random(n)
        a /= a.sum()
        print(f"[init] Random start, n={n}")

    best_a = a.copy()
    best_val = _objective(a)
    print(f"[init] initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Smooth Adam optimisation with a growing soft‑max temperature.
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t_iter = 0
    beta_soft = 1.0          # temperature for soft‑max approximation
    max_beta = 200.0         # final temperature (approx. the true max)

    while time.time() < deadline:
        t_iter += 1

        # current convolution and objective
        conv = _full_convolution(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[adam] iter {t_iter:6d}  new best = {best_val:.6f}")

        # ---------- soft‑max weights ----------
        maxc = conv.max()
        # numerical stability: shift by maxc
        exp_term = np.exp(beta_soft * (conv - maxc))
        w = exp_term / exp_term.sum()   # shape (2n-1,)

        # ---------- gradient ----------
        # gradient = 2 * (w * a)  (full convolution, then slice)
        conv_rev = _full_convolution(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the probability simplex
        a = _project_simplex(a)

        # gradually tighten the soft‑max (approach the true max)
        beta_soft = min(beta_soft * 1.005, max_beta)

        # occasional safety check to avoid stagnation
        if t_iter % 2000 == 0:
            elapsed = time.time() - start
            print(f"[adam] iter {t_iter:6d}  elapsed {elapsed:.1f}s  best {best_val:.6f}")

    print(f"[adam] finished after {t_iter} iterations, best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Fine‑grained hill‑climbing phase.
    # ------------------------------------------------------------------
    a_final, val_final = _mass_transfer_hillclimb(best_a, deadline, rng)
    print(f"[final] after hill‑climb best objective = {val_final:.6f}")

    # ------------------------------------------------------------------
    # 4. Report the final score via the provided evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_final.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = float('inf')

    return a_final.tolist()
