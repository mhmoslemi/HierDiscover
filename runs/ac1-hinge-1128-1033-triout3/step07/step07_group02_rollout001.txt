"""
Propose a step‑function (non‑negative list) that minimises ``evaluate_sequence``.

Algorithm Overview
------------------
1. **Problem size** – Use a relatively large support (n≈8000) to obtain fine
   control of the shape while keeping FFT‑based convolutions cheap.

2. **Initialisation**
   * If a previous solution ``height_sequence_1`` exists, linearly interpolate
     it to the current length.
   * Otherwise sweep a two‑parameter Beta‑family
       a_i ∝ (i+1)^{p‑1} (n‑i)^{q‑1}
     over a modest grid and keep the best seed.
   * Finally add a random start if everything else fails.

3. **Phase A – Adam optimisation**
   * Optimise a smooth approximation of the true objective
     (log‑sum‑exp of the self‑convolution) on the probability simplex.
   * Learning‑rate and temperature decay logarithmically.
   * The best vector found is retained.

4. **Phase B – Projected sub‑gradient descent**
   * Repeatedly move in the direction of a sub‑gradient of the exact
     ``max(conv(a,a))`` (i.e. the worst autocorrelation entry) and project
     back onto the simplex.  The step size decays as 1/√t.

5. **Phase C – Simulated‑annealing hill‑climbing**
   * A refined local search that transfers a small amount of mass from an
     index that contributes strongly to the current maximum to a weak one.
   * Metropolis acceptance with a linearly decreasing temperature.

6. **Final evaluation** – Return the best vector found (its sum is 1, which
   does not affect the wrapper’s scale‑invariant objective).

The three optimisation phases together consume the 1000 s budget
(with a 0.5 s safety margin).  The implementation uses only NumPy and the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (linear convolution)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) (sum(a) is assumed 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and gradient w.r.t. a
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and its gradient.
    The gradient corresponds to the exact objective 2·n·max(conv) approximated
    by a soft‑max with temperature eps.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # gradient of Σ p_k * b_k w.r.t a
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 4n from 2·n·max

    # smooth value (not yet multiplied by 2·n)
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max approximation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a smooth approximation of the objective.
    Returns the best vector encountered before *deadline*.
    """
    # logits for soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.002
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v_vec = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # exact evaluation (cheap enough for n≈8000)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedule for temperature (eps) and learning rate (lr)
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max and gradient
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)

        # chain rule through the soft‑max (w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation to avoid lock‑in
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        # restart on stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restarting after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v_vec.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val

# Projected sub‑gradient descent on the exact max‑convolution norm
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_a: np.ndarray,
                         best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Deterministic sub‑gradient descent on the exact objective
    (2·n·max(conv(a,a))) with projection onto the simplex.
    """
    a = a0.copy()
    cur_val = _objective(a)

    # keep the current best
    if cur_val < best_val:
        best_val, best_a = cur_val, a.copy()

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        t += 1
        # full convolution and locate maximal entry
        b = _full_convolution(a)
        max_val = b.max()
        k_max = int(np.argmax(b))

        # gradient of 2·n·b[k_max] w.r.t a
        idx = np.arange(n)
        idx2 = k_max - idx
        mask = (idx2 >= 0) & (idx2 < n)

        grad = np.zeros(n, dtype=np.float64)
        # a[i] term is always present
        grad += a
        # add a[k_max-i] where it exists
        grad[mask] += a[idx2[mask]]
        grad = 2.0 * n * grad   # derivative of 2·n·b[k_max]

        # step size schedule (1/√t)
        step = 0.018 / np.sqrt(t)

        a_candidate = a - step * grad
        a_candidate = _project_to_simplex(a_candidate)

        new_val = _objective(a_candidate)

        # monotone acceptance
        if new_val <= cur_val:
            a = a_candidate
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Subgrad {t:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        # else keep old a (no move)

    return best_a, best_val

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ---- source index (probability ∝ contribution) ----
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ---- target index (biased towards low contribution) ----
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback uniform among all except i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric sweep of a Beta‑shaped family
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p-1} (n-i)^{q-1}."""
    best_a = None
    best_val = np.inf
    # grid (linear + geometric) for exponents
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluate call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – reuse previous construction if available.
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – parametric sweep of a Beta‑shaped family.
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    try:
        shape_a, shape_val = _beta_family_best(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] beta‑family sweep improved value={best_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[init] beta‑family sweep failed:", exc)

    # fallback random start if needed
    if best_a is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_a, best_val = a, _objective(a)
        print("[init] random start, value=", best_val)

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈45 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # Phase B – Projected sub‑gradient descent (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.60 * budget_s
    best_a, best_val = _subgradient_descent(best_a, n,
                                            deadline=phase_b_deadline,
                                            start_time=phase_a_deadline,
                                            best_a=best_a,
                                            best_val=best_val,
                                            target_val=1.503)

    # ------------------------------------------------------------------
    # Phase C – Simulated‑annealing hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=phase_b_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
