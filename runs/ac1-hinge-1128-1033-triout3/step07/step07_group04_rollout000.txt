"""
Hybrid optimiser for the step‑function convolution problem (new version).

The optimiser works in three phases:

1. **Broad seeding** – a collection of hand‑crafted families
   (geometric, power‑law) together with Dirichlet draws having
   various concentration parameters are evaluated; the best seed is kept.

2. **Projected sub‑gradient descent on the exact max‑convolution
   objective** – the objective  f(a)=max(a*a) (with sum(a)=1) is convex.
   At each iteration the gradient of the currently maximal convolution
   entry is computed analytically, a step is taken, and the iterate is
   projected back onto the probability simplex (non‑negative, sum = 1).
   The step size follows a 1/√t schedule.  Several random restarts are
   performed while time permits.

3. **Refinement** – a cheap hill‑climbing phase (swaps and small mass
   transfers) finishes the search in the remaining seconds.

All heavy work (convolution, projection) is vectorised and uses FFTs.
The routine returns a plain Python list of floats and respects the 1000 s
budget.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Returns an array of length 2*len(a)-1.
    Tiny negative rounding errors are clamped to 0.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)      # clamp
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for a non‑negative vector with sum=1.
    Returns 2*n*max(a*a).
    """
    n = a.shape[0]
    max_b = _fft_convolve_self(a).max()
    return float(2.0 * n * max_b)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, sum x_i = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """a_i ∝ q**i, normalised to sum = 1 (0<q<1)."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q**n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)**p, normalised to sum = 1 (p<0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation on max‑convolution
# ----------------------------------------------------------------------
def _subgradient_optimize(start_a: np.ndarray,
                          time_budget: float,
                          global_best_val: float,
                          global_best_a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Runs projected sub‑gradient descent on  f(a)=max(a*a)
    with the simplex constraint.  Returns the best pair (a, f(a))
    found during the allotted time.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    # First evaluation (max of current a)
    conv = _fft_convolve_self(a)
    cur_val = 2.0 * n * conv.max()
    best_val = cur_val
    if cur_val < global_best_val:
        global_best_val = cur_val
        global_best_a = a.copy()

    start = time.time()
    it = 0
    step0 = 0.5                     # initial step‑size (tuned empirically)

    while time.time() - start < time_budget:
        # compute convolution and the index attaining the maximum
        conv = _fft_convolve_self(a)
        max_b = conv.max()
        k = int(conv.argmax())

        # gradient of the maximal entry (sub‑gradient)
        g = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] = 2.0 * a[k - idx]          # 2*a_{k-i}

        # diminishing step size
        eta = step0 / math.sqrt(it + 1)

        # gradient step + Euclidean projection onto the simplex
        a = _project_simplex(a - eta * g)

        # evaluate new point (re‑use max from the just‑computed convolution)
        cur_val = 2.0 * n * max_b

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        if cur_val < global_best_val:
            global_best_val = cur_val
            global_best_a = a.copy()
            print(f"[subgrad] new global best = {global_best_val:.6f} (it {it})")

        it += 1

    return best_a, best_val, global_best_a, global_best_val


def _hill_climb(a0: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing using swaps and small mass transfers.
    Returns the best sequence found.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_val = _eval_fast(a)
    best_a = a.copy()
    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() - start < time_budget:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                a[i], a[j] = a[j], a[i]
        else:                                      # small transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * 0.05 * rng.random()
                a[i] -= delta
                a[j] += delta

        # re‑normalise (project onto simplex) – cheap because we only changed two entries
        a = _project_simplex(a)

        val = _eval_fast(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[hill] new best = {best_val:.6f}")

        it += 1

    return best_a, best_val

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size (fixed to 4000 as in the reference run; can be changed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous construction if available, else start from uniform.
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            # interpolate to the required length
            old_x = np.linspace(0, 1, a0.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, a0)
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)           # ensures sum = 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Diverse seed generation
    # ------------------------------------------------------------------
    seed_candidates = []

    # geometric families
    for q in np.linspace(0.7, 0.999, 20):
        seed_candidates.append(_geometric_sequence(q, n))

    # power‑law families
    for p in np.linspace(-3.0, -0.2, 20):
        seed_candidates.append(_power_sequence(p, n))

    # Dirichlet draws with several concentration parameters
    for alpha in (0.3, 0.6, 1.0, 2.0, 5.0):
        for _ in range(12):
            a = rng.dirichlet(alpha * np.ones(n))
            seed_candidates.append(a)

    # evaluate all seeds
    for a in seed_candidates:
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[seed] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient optimisation (multiple restarts)
    # ------------------------------------------------------------------
    # Allocate most of the remaining time to this phase; keep ~30 s for
    # final hill‑climbing and the exact evaluation call.
    while time.time() < deadline - 30.0:
        # Choose an initial point:
        if rng.random() < 0.5:
            init_a = best_a.copy()                     # deepen current optimum
        else:
            # a fresh random Dirichlet (moderately sparse)
            alpha = rng.choice([0.2, 0.5, 1.0])
            init_a = rng.dirichlet(alpha * np.ones(n))

        # Limit each sub‑gradient run to at most 1 s (or less if time is low)
        run_budget = min(1.0, deadline - time.time() - 30.0)
        _, _, best_a, best_val = _subgradient_optimize(
            init_a, run_budget, best_val, best_a
        )
        # occasional re‑normalisation to counter numerical drift
        best_a = _project_simplex(best_a)

    print(f"[subgrad] best after sub‑gradient phase = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final hill‑climbing refinement (use whatever time is left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_a, best_val = _hill_climb(best_a, remaining)
        print(f"[final hill] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Report exact evaluation using the provided function
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = float('inf')

    return best_a.tolist()
