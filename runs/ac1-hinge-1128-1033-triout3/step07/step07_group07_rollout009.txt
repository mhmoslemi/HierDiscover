"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The optimiser works in three stages:
1. **Coarse optimisation on a small problem** (n≈256).  A simulated‑annealing
   loop makes many cheap moves (mass transfer between two positions) while
   accepting worsening moves with a temperature that slowly cools.  The best
   vector found is kept.
2. **Upscaling** – the small‑scale solution is interpolated to the target size
   (n≈4000).  This supplies a good starting point for the large problem.
3. **Fine optimisation** – the large vector is refined with the same
   annealing scheme and a few greedy “mass‑transfer” steps that explicitly
   reduce the active (maximal) convolution entry.

All operations respect the non‑negativity and simplex constraints; the
objective is evaluated via fast FFT‑based convolution.  If the optimiser
fails to improve over the previously known construction (available as the
global ``height_sequence_1``), that sequence is returned as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for the objective
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (length 2·len‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical artefacts may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) (the sum of a is assumed to be 1)."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


# ----------------------------------------------------------------------
# Moves that change the distribution
# ----------------------------------------------------------------------
def _random_transfer(a: np.ndarray, step_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Move a fraction of the mass from a random donor to a random receiver."""
    n = a.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    if a[i] <= 0.0:
        return a  # nothing to move
    eps = step_frac * a[i]
    eps = min(eps, a[i])          # never make a[i] negative
    new_a = a.copy()
    new_a[i] -= eps
    new_a[j] += eps
    # re‑normalise to keep the sum exactly 1 (removes tiny rounding drift)
    new_a /= new_a.sum()
    return new_a


def _active_lag_transfer(a: np.ndarray, step_frac: float, rng: np.random.Generator) -> np.ndarray:
    """
    Identify a lag where the convolution is maximal and move a fraction of
    mass away from the pair that contributes most to that lag.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_val = conv.max()
    # pick one of the (possibly many) active lags at random
    active = np.flatnonzero(conv == max_val)
    t = int(rng.choice(active))

    # indices that participate in lag t
    i_start = max(0, t - (n - 1))
    i_end = min(t, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = t - i_range

    # contribution of each ordered pair (i,j)
    contrib = a[i_range] * a[j_range]
    if contrib.size == 0:
        return a

    # pick the pair with the largest product
    idx = int(np.argmax(contrib))
    i_donor = int(i_range[idx])
    j_donor = int(j_range[idx])

    # how much mass to move?
    if i_donor == j_donor:
        eps = step_frac * a[i_donor]
        if eps <= 0.0:
            return a
        # pick a destination distinct from the donor
        cand = np.setdiff1d(np.arange(n), [i_donor])
        if cand.size == 0:
            return a
        dest = int(rng.choice(cand))
        new_a = a.copy()
        new_a[i_donor] -= eps
        new_a[dest] += eps
    else:
        eps = step_frac * min(a[i_donor], a[j_donor])
        if eps <= 0.0:
            return a
        # two distinct destinations, avoiding the donors
        forbidden = np.array([i_donor, j_donor])
        cand = np.setdiff1d(np.arange(n), forbidden)
        if cand.size < 2:
            return a
        dest = rng.choice(cand, size=2, replace=False)
        new_a = a.copy()
        new_a[i_donor] -= eps
        new_a[j_donor] -= eps
        new_a[dest[0]] += eps
        new_a[dest[1]] += eps

    # protect against numerical negatives, then renormalise
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s <= 0.0:
        return a
    new_a /= s
    return new_a


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (used for both small and large problems)
# ----------------------------------------------------------------------
def _anneal(a_init: np.ndarray,
            max_seconds: float,
            rng: np.random.Generator,
            start_temp: float = 0.5,
            step_start: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Perform a simple simulated‑annealing search.
    Returns the best vector found and its objective value.
    """
    a = a_init.copy()
    cur_obj = _objective(a)
    best_a = a.copy()
    best_obj = cur_obj

    n = a.shape[0]
    start = time.time()
    it = 0
    T = start_temp
    step_frac = step_start

    # Main loop – stops when the allocated time expires.
    while time.time() - start < max_seconds:
        # Choose which kind of move to try
        if rng.random() < 0.65:
            cand_a = _active_lag_transfer(a, step_frac, rng)
        else:
            cand_a = _random_transfer(a, step_frac, rng)

        cand_obj = _objective(cand_a)
        delta = cand_obj - cur_obj

        # Metropolis acceptance
        if delta < 0.0 or np.exp(-delta / max(T, 1e-12)) > rng.random():
            a = cand_a
            cur_obj = cand_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[anneal] n={n} iter={it} new best = {best_obj:.6f}")

        # Adaptive cooling and step‑size decay
        it += 1
        if it % 1000 == 0:
            T = max(T * 0.97, 1e-8)
            step_frac = max(step_frac * 0.995, 0.005)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Optimises a non‑negative step function under the ``evaluate_sequence``
    objective.  The routine respects the allotted time budget and returns the
    best list of floats found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Load the previous best construction (if it exists) – used as a fallback.
    # ------------------------------------------------------------------
    prior_seq = None
    prior_obj = float("inf")
    try:
        prior_seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # normalise (the evaluator rescales anyway)
        prior_seq = np.maximum(prior_seq, 0.0)
        if prior_seq.sum() > 0:
            prior_seq = prior_seq / prior_seq.sum()
        prior_obj = evaluate_sequence(prior_seq.tolist())
        print(f"[fallback] prior objective = {prior_obj:.6f}")
    except Exception:
        prior_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – cheap optimisation on a small problem (n≈256).
    # ------------------------------------------------------------------
    n_small = 256
    small_budget = min(0.30 * budget_s, 190.0)   # leave room for the large problem
    a_small = np.full(n_small, 1.0 / n_small)
    best_small, best_small_obj = _anneal(a_small,
                                         max_seconds=small_budget,
                                         rng=rng,
                                         start_temp=0.8,
                                         step_start=0.25)

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to the target size.
    # ------------------------------------------------------------------
    n_large = 4000
    # linear interpolation from the small optimum
    old_x = np.linspace(0.0, 1.0, n_small)
    new_x = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(new_x, old_x, best_small)
    a_large = np.maximum(a_large, 0.0)
    if a_large.sum() > 0:
        a_large /= a_large.sum()
    else:
        a_large = np.full(n_large, 1.0 / n_large)

    # ------------------------------------------------------------------
    # Phase 3 – fine optimisation on the full length.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_large, best_large_obj = _anneal(a_large,
                                             max_seconds=remaining * 0.7,
                                             rng=rng,
                                             start_temp=0.6,
                                             step_start=0.2)
    else:
        best_large, best_large_obj = a_large, _objective(a_large)

    # ------------------------------------------------------------------
    # Optional final greedy polishing (only a few passes, cheap).
    # ------------------------------------------------------------------
    final_budget = deadline - time.time()
    if final_budget > 2.0:
        # Run a short deterministic hill‑climb based on the active‑lag move.
        cur_a = best_large.copy()
        cur_obj = best_large_obj
        best_a = cur_a.copy()
        best_obj = cur_obj
        t_end = time.time() + final_budget
        step_frac = 0.12
        while time.time() < t_end:
            cand_a = _active_lag_transfer(cur_a, step_frac, rng)
            cand_obj = _objective(cand_a)
            if cand_obj < cur_obj:
                cur_a = cand_a
                cur_obj = cand_obj
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = cur_a.copy()
                    print(f"[polish] new best = {best_obj:.6f}")
                # slowly increase step size after a success
                step_frac = min(step_frac * 1.03, 0.25)
            else:
                # on failure shrink the step
                step_frac = max(step_frac * 0.95, 0.005)
        best_large = best_a
        best_large_obj = best_obj

    # ------------------------------------------------------------------
    # Final evaluation with the provided ``evaluate_sequence`` routine.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_large.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = float("inf")

    # ------------------------------------------------------------------
    # Choose the best among the newly found solution and the prior fallback.
    # ------------------------------------------------------------------
    if prior_seq is not None and prior_obj < final_score:
        print("[result] falling back to previous best construction")
        return prior_seq.tolist()
    else:
        return best_large.tolist()
