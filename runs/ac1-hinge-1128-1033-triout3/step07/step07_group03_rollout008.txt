def _targeted_max_descent(a_start, deadline):
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    α0 = 0.3
    iteration = 0
    while time.time() < deadline - 5:
        conv = _full_convolution(a)
        max_val = conv.max()
        k_max = conv.argmax()
        # compute partner indices for each i
        # compute gradient contribution as a_rev = zeros
        g = np.zeros_like(a)
        # g_i = a_{k_max - i} if in range
        low = max(0, k_max - (n-1))
        high = min(n-1, k_max)
        i_range = np.arange(low, high+1, dtype=int)
        j_range = k_max - i_range
        g[i_range] = a[j_range]  # gradient result

        # multiplicative update
        α = α0 / (1 + iteration * 1e-4)  # decaying
        a = a * np.exp(-α * g)

        # renormalize
        a = np.maximum(a, 0)
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1.0/n)
        else:
            a /= sum_a

        if iteration % 20 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[max desc iter {iteration}] new best {best_val:.6f} (max conv {max_val:.8f})")
        iteration += 1

    return best_a, best_val

def _refine_targeted(a, best_val, deadline, n):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute conv and max
        conv = _full_convolution(cur_a)
        max_idx = int(conv.argmax())
        max_val = conv.max()
        # identify candidate i positions that contribute heavily to this max
        # compute contributions for each i
        low = max(0, max_idx - (n-1))
        high = min(n-1, max_idx)
        i_range = np.arange(low, high+1)
        j_range = max_idx - i_range
        contributions = cur_a[i_range] * cur_a[j_range]  # elementwise product
        # pick i from top 10% contributions
        top_idxs = i_range[np.argsort(-contributions)[:max(1, len(i_range)//10)]]
        # pick j from bottom 10% contributions (or low contributions)
        bottom_idxs = i_range[np.argsort(contributions)[:max(1, len(i_range)//10)]]
        if len(top_idxs)==0 or len(bottom_idxs)==0:
            continue
        i = rng.choice(top_idxs)
        j = rng.choice(bottom_idxs)
        # mass transfer
        delta = cur_a[i] * 0.05 * rng.random()
        if delta <= 0:
            continue
        new_a = cur_a.copy()

"""

The algorithm is an extended version of the previous pipeline:

0. **Initialisation** – start from the historic construction (if any) or
   from a few random Dirichlet vectors.

1. **Smooth surrogate optimisation** – minimise a log‑sum‑exp
   approximation of the max‑convolution using L‑BFGS‑B with several
   restarts and an ε‑schedule that reaches 1e‑12.

2. **Multi‑peak exponential mirror‑descent** – same idea as before but
   with a larger active set (k=10) and a slightly different
   decay‑schedule.

3. **Focused max‑reduction descent** – at each iteration compute the
   lag achieving the current maximum of the self‑convolution, then
   apply a multiplicative (exponentiated‑gradient) update that
   specifically reduces the contribution to that lag.

4. **Targeted refinement** – a greedy local search that moves mass from
   indices that strongly contribute to the current maximal lag to
   indices that contribute little.

5. **Final hill‑climbing** – random swaps / mass‑transfers as a safety
   net.

All updates keep the vector on the probability simplex, which does not
affect the homogeneous objective.  The whole search respects the
1000 s budget, printing progress along the way.

"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair large values with small ones when the sequence is reversed.
    Resulting pattern is a left‑heavy descending part and a right‑heavy
    ascending part.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # --- soft‑max ---
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = a.sum()
    a /= a_sum

    # --- convolution a*a ---
    conv = _full_convolution(a)               # length 2·n‑1
    max_conv = conv.max()

    # --- smooth max via log‑sum‑exp ---
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # --- gradient w.r.t. a (see derivation in analysis) ---
    weights = np.exp(shifted)
    p = weights / weights.sum()                # distribution over lag indices

    # convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]            # middle n entries
    grad_a = 4.0 * n * seg[::-1]               # dL/da

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

# L‑BFGS optimiser on the smooth surrogate

# Multi‑peak exponential gradient descent (core of the search)
# ----------------------------------------------------------------------
def _gradient_descent_multi(a_start: np.ndarray, deadline: float, top_k: int = 10):
    """
    Exponential mirror‑descent that reduces the *top_k* largest
    convolution peaks.  Returns the best (a, value) pair found.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[gd_multi init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.6          # initial step‑size
    decay_rate = 0.98     # decay of α every 2000 iterations

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_val = conv.max()

        # indices of the ``top_k`` largest convolution entries
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]   # sorted descending
        else:
            idxs = np.arange(conv.size)

        # build gradient contribution from the selected peaks
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            jdx = k - idx_range
            g[idx_range] += a[jdx]

        # scheduled step‑size
        alpha = alpha0 * (decay_rate ** (it // 2000))

        # multiplicative update (mirror‑descent)
        a = a * np.exp(-alpha * g)

        # projection onto the probability simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-14:                     # guard against under‑flow
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

            if it % 500 == 0:

                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd_multi reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val

def _max_focus_descent(a_start: np.ndarray, deadline: float):
    """
    Mirror‑descent that targets the single lag with maximal convolution.
    Returns the best (a, value) discovered.

        idx_range = np.arange(low, high + 1, dtype=int)

        g[idx_range] = a[jdx]          # derivative ∂conv/∂a_i = a_{k_max-i}

        # step‑size schedule (slow decay)
        alpha = alpha0 / (1.0 + it * 1e-4)

        if s < 1e-14:

# Targeted greedy refinement (mass transfers guided by the current max)
# ----------------------------------------------------------------------
def _refine_targeted(a: np.ndarray, best_val: float, deadline: float, n: int):
    """
    Greedy local search: move a small amount of mass from indices that
    contribute strongly to the current maximal lag to indices that
    contribute weakly.  Accept only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        # current convolution and maximal lag
        conv = _full_convolution(cur_a)
        k_max = int(conv.argmax())
        # indices that affect this lag
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_range = np.arange(lo, hi + 1, dtype=int)
        j_range = k_max - i_range
        contributions = cur_a[i_range] * cur_a[j_range]

        # pick source (large contribution) and destination (small contribution)
        if i_range.size == 0:
            break
        # choose from top 10 % and bottom 10 %
        top_cnt = max(1, i_range.size // 10)
        src_idx = i_range[np.argsort(-contributions)[:top_cnt]]
        dst_idx = i_range[np.argsort(contributions)[:top_cnt]]

        i = int(rng.choice(src_idx))
        j = int(rng.choice(dst_idx))
        if i == j or cur_a[i] <= 1e-15:
            continue

        delta = cur_a[i] * 0.03 * rng.random()   # move up to 3 % of source mass
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # projection onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-14:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[targeted refine] new best = {cur_val:.6f}")
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
    return cur_a, cur_val

# Simple hill‑climbing refinement (random swaps / mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random local search: swaps (20 %) and mass transfers (80 %).
    Keeps only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-14:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[hill‑climb] new best = {cur_val:.6f}")

    return cur_a, cur_val
