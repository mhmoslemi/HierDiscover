"""
A two‑phase optimizer for the “step‑function” attack.

Phase 1 – Exponential‑gradient sub‑gradient descent
---------------------------------------------------
* Works directly on the simplex (log‑soft‑max parametrisation).
* At each iteration the exact maximal convolution entry is located.
* Its exact sub‑gradient (a_i + a_{k−i}) is computed and the
  log‑probabilities are moved opposite to the gradient.
* A simple linear schedule for the learning rate is used and a
  small random perturbation restarts the method after long
  stagnation.

Phase 2 – Fast simulated‑annealing hill‑climbing
------------------------------------------------
* Starts from the best point found by Phase 1.
* Transfers a small amount of mass from indices that contribute
  heavily to the current maximal convolution entry to indices that
  contribute little.
* Accepts uphill moves with a Metropolis probability; the temperature
  decays linearly to zero.
* Periodic global perturbations avoid getting stuck.

Both phases respect the 1000 s budget, keep the vector non‑negative,
and normalise it to sum = 1 (scale‑invariant for the true objective).
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Beta‑family sweep (cheap way to obtain a decent seed)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1} (n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # a modest grid – enough to give a good start without wasting time
    p_vals = np.concatenate(
        [np.linspace(0.1, 1.0, 9), np.geomspace(1.0, 6.0, 7)[1:]]
    )
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – Exponential‑gradient sub‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _expgrad_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Projected exponential‑gradient descent on the exact max‑norm objective.
    Returns the best (a, value) seen before *deadline*.
    """
    a = a0.copy()
    # work in log‑space to keep the vector on the simplex automatically
    log_a = np.log(a + 1e-12)

    stagnation = 0            # counts iterations without improvement
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current exact objective
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            stagnation = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
        else:
            stagnation += 1

        k_max = int(np.argmax(b))
        # grad_i = a_i + a_{k_max-i} (if in range)

        j_idx = k_max - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        shift = np.zeros_like(a)
        shift[mask] = a[j_idx[mask]]
        grad = a + shift                     # always non‑negative

        prog = (now - start_time) / (deadline - start_time)
        eta = 0.45 * (1.0 - prog) + 0.02 * prog   # from 0.45 → 0.02

        # --- exponential‑gradient step -----------------------------------------
        log_a -= eta * grad
        # renormalise safely (log‑sum‑exp trick)
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:                     # safety fall‑back
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        # --- occasional random perturbation (to escape plateaus) --------------
        if stagnation > 25000:
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            log_a = np.log(a + 1e-12)
            stagnation = 0
            print(f"[ExpGrad {it:7d}] perturb after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Simulated‑annealing hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _hillclimb_anneal(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Local stochastic search that transfers a small amount of mass from
    heavily contributing indices to weak ones, accepting uphill moves
    with a linearly decaying temperature.
    """
    a = a0.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules (linear in the remaining fraction of the budget)
    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.012, 1e-7

    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # --- locate the currently most offending convolution entry -----------
        k_max = int(np.argmax(b))

        # contributions of each index to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ---- source index (probability ∝ contribution) ----
        if total_contrib > 0.0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ---- target index (biased towards low contribution) ----
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        # contribution of the moved mass to all affected positions
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # the self‑interaction terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # --- stagnation handling -------------------------------------------------
        if no_improve > 120_000:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturb")

        it += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and (attempts to) minimises
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previous construction if available.
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= a0_sum
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – cheap Beta‑family sweep for a decent seed
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    try:
        shape_a, shape_val = _beta_family_best(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] beta‑family sweep improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # fallback random start if needed
    if best_a is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_a, best_val = a, _objective(a)
        print("[init] random start, value=", best_val)

    # ------------------------------------------------------------------
    # Phase 1 – Exponential‑gradient sub‑gradient descent (≈30 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _expgrad_optimize(
        best_a,
        n,
        deadline=phase1_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )

    # ------------------------------------------------------------------
    # Phase 2 – Simulated‑annealing hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(
        best_a,
        n,
        deadline=deadline,
        start_time=phase1_deadline,
        best_a=best_a,
        best_val=best_val,
    )

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
