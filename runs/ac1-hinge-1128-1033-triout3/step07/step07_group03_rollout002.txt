"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Initialisation** – start from a known construction (``height_sequence_1``)
   if available, otherwise use a uniform vector.

2. **Smooth surrogate optimisation** – minimise a smooth approximation of the
   objective with L‑BFGS‑B (multiple restarts, decreasing smoothing).

3. **Exponential mirror descent** – multiplicative updates that push the
   largest peaks of the self‑convolution down (several variants of the
   hyper‑parameters).

4. **Permutation hill‑climbing** – repeatedly swap two positions; keep the
   swap only when the maximal entry of the self‑convolution (hence the
   objective) strictly decreases.  This explores the combinatorial space of
   orderings beyond the simple “anti‑symmetric’’ heuristic.

5. **Final local refinement** – tiny mass transfers and additional swaps.

All phases respect the time budget (default 1000 s) and keep the vector on
the probability simplex (non‑negative, sum = 1), which does not affect the
objective because it is homogeneous.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables for the smooth surrogate optimiser (used by L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter ε
_GLOBAL_N = None     # current length of the vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise can produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values on the left, the smallest on the right."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """
    Interleave the largest and smallest values:
    [max, min, 2nd‑max, 2nd‑min, …].
    """
    vals = np.sort(a)               # ascending
    n = vals.shape[0]
    new = np.empty_like(a)
    lo, hi = 0, n - 1
    pos = 0
    while lo <= hi:
        new[pos] = vals[hi]
        pos += 1
        hi -= 1
        if lo <= hi:
            new[pos] = vals[lo]
            pos += 1
            lo += 1
    return new


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution a*a
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()           # distribution over lag indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # derivative w.r.t. a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()
    # ε‑schedule: geometric from 5·10⁻³ down to 10⁻⁹ (9 points)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=9)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 12.0:   # a tiny safety margin
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    if best_a is None:   # fallback – should never happen
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val

# Exponential mirror‑descent (multi‑peak)
# ----------------------------------------------------------------------
def _gradient_descent_multi(a_start: np.ndarray,
                            deadline: float,
                            top_k: int = 3,
                            alpha0: float = 0.5,
                            decay_rate: float = 0.98,
                            log_every: int = 50):
    """
    Mirror‑descent that reduces the ``top_k`` largest convolution peaks.
    Returns the best (a, value) found before ``deadline``.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[gd init] value = {best_val:.6f}")

    it = 0
    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_val = conv.max()

        # indices of the ``top_k`` largest convolution entries
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]
        else:
            idxs = np.arange(conv.size)

        # gradient contribution from the selected peaks
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            jdx = k - idx_range
            g[idx_range] += a[jdx]

        # scheduled step‑size
        alpha = alpha0 * (decay_rate ** (it // 2000))

        # multiplicative update (mirror‑descent)
        a = a * np.exp(-alpha * g)

        # projection onto the probability simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:               # guard against under‑flow
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        # logging
        if it % log_every == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd iter {it}] new best = {best_val:.6f} (max conv {max_val:.8f})")
            if it % (log_every * 10) == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] improved to {best_val:.6f}")
        it += 1

    return best_a, best_val

    """

    entry of the self‑convolution (hence the objective) strictly decreases.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = _full_convolution(a)
    best_max = conv.max()
    best_val = 2.0 * n * best_max          # because sum(a)=1
    best_a = a.copy()
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline - 3.0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform the swap
        a[i], a[j] = a[j], a[i]

        # evaluate quickly via convolution
        conv_new = _full_convolution(a)
        max_new = conv_new.max()
        if max_new < best_max - 1e-12:    # strict improvement
            best_max = max_new
            best_val = 2.0 * n * best_max
            best_a = a.copy()
            if it % log_every == 0:
                print(f"[permute iter {it}] new best = {best_val:.6f} (max conv {best_max:.8e})")
        else:
            # revert – the swap was not beneficial
            a[i], a[j] = a[j], a[i]
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (random swaps / mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       n: int):
    """
    Random local search: swaps (20 %) and mass transfers (80 %).
    Keeps only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.

    The pipeline follows the stages described in the module docstring.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a few seconds safety margin before returning
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more freedom, we pick 8000 (as before)
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from a known construction if possible
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the target length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)          # uniform start

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
        s = a.sum()
    a /= s

    # ------------------------------------------------------------------
    # Evaluate the initial candidate
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – a few random initial vectors (to avoid a poor start)
    # ------------------------------------------------------------------
    for tr in range(4):
        if time.time() >= deadline - 180:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate optimisation (L‑BFGS) with a few restarts
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 140 and restart < 5:
        restart += 1
        w_start = np.log(best_a + 1e-12)
        w_start += rng.normal(scale=0.2, size=n)   # small perturbation
        a_opt, val_opt = _l_bfgs_optimize_from_w(w_start, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[restart {restart}] new best after L‑BFGS = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multi‑peak exponential gradient descent (first pass)
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        a_gd, val_gd = _gradient_descent_multi(best_a, deadline,
                                               top_k=3,
                                               alpha0=0.6,
                                               decay_rate=0.97,
                                               log_every=50)
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[gd pass 1] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – focused gradient descent on the single worst peak
    # ------------------------------------------------------------------
    if time.time() < deadline - 60:
        a_gd2, val_gd2 = _gradient_descent_multi(best_a, deadline,
                                                 top_k=1,
                                                 alpha0=1.0,
                                                 decay_rate=0.99,
                                                 log_every=25)
        if val_gd2 < best_val:
            best_val = val_gd2
            best_a = a_gd2.copy()
            print(f"[gd pass 2 (single peak)] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – permutation hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        a_perm, val_perm = _permute_hillclimb(best_a, deadline, log_every=200)
        if val_perm < best_val:
            best_val = val_perm
            best_a = a_perm.copy()
            print(f"[permute] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – apply the alternating heuristic (may improve ordering)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20:
        a_alt = _alternating_reorder(best_a)
        val_alt = evaluate_sequence(a_alt.tolist())
        if val_alt < best_val:
            best_val = val_alt
            best_a = a_alt.copy()
            print(f"[alternating reorder] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
