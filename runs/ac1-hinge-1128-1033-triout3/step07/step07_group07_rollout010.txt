def subgradient_optimize(a0, n, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = exact_obj(a)
    s0 = 0.5
    while time.time() < deadline:
        conv = full_convolution(a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))
        # compute subgradient: grad[i] = 2 * a[k-i] if in range else 0
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end+1, dtype=int)
        grad[i_range] = 2.0 * a[k - i_range]
        # project step
        s = s0
        improved = False
        while s > 1e-6:
            a_new = a - s * grad
            a_new = _project_simplex(a_new)
            val_new = exact_obj(a_new)
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            s *= 0.5
        if not improved:
            # reduce step size for next iteration
            s0 *= 0.9
        else:
            # maybe increase s0 a bit
            s0 = min(s0 * 1.1, 1.0)
    return best_a, best_val
```

    w = w_init.copy()
    best_a = np.exp(w - np.max(w))
    best_a /= best_a.sum()

    m = np.zeros_like(w)
    v = np.zeros_like(w)

    beta2 = 0.999
    eps_adam = 1e-8
    t_iter = 0

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The optimiser works in three stages:

1.  A smooth surrogate of the max‑convolution objective is minimised
    by Adam on the soft‑max (probability‑simplex) parametrisation.
    The smoothing parameter ε is decreased logarithmically, allowing
    the optimiser to first move freely and then focus on the true
    max‑convolution.

2.  The best iterate is refined with a classic sub‑gradient descent
    on the exact non‑smooth objective (2·n·max(conv)).
    A back‑tracking line‑search guarantees monotone improvement.

3.  Finally a fast “mass‑transfer’’ hill‑climber makes aggressive
    local moves that are accepted only when they improve the exact
    objective.

Multiple initialisations (interpolated previous solution,
random, sinusoidal, Gaussian) and a few random perturbations are
tried; the best candidate seen before the time deadline is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# FFT utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # numerical noise may produce tiny negatives – clamp to zero
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) for the smooth surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε),

    where a = softmax(w) (∑a = 1, a_i ≥ 0).
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # full autocorrelation
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing (shifted to avoid overflow)
    shifted = (conv - max_conv) / eps
    # weights = exp(shifted)  – they sum to 1 after normalisation
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # probabilities over lags

    # gradient w.r.t. a (closed form)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                  # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]                # middle n entries
    grad_a = 4.0 * n * seg[::-1]                   # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    # smooth objective value (not needed for optimisation, but returned for logging)
    log_sum = np.logaddexp.reduce(shifted)          # = log Σ exp(...)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum
    return L, grad_w


# ------------------------------------------------------------
# Projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact objective for a probability vector (sum = 1)
# ------------------------------------------------------------
def _exact_obj(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv) for a (assumed ∑a = 1)."""
    return float(2.0 * n * np.max(_full_convolution(a)))


# ------------------------------------------------------------
# Aggressive mass‑transfer hill‑climbing (same as before)
# ------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              best_val: float,
                              deadline: float,
                              rng: np.random.Generator,
                              step_start: float = 0.2):
    """
    Repeatedly move a fraction of mass away from lag(s) attaining the
    maximal convolution. Returns the best vector and its objective.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    best_a = cur_a.copy()
    step_frac = step_start

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # indices that contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]

        if contrib.size == 0:
            continue

        top_n = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue

        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue
        new_a /= s

        new_val = _exact_obj(new_a, n)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-6)

    return best_a, best_val


# ------------------------------------------------------------
# Adam optimisation on the smooth surrogate
# ------------------------------------------------------------
def _adam_optimize(w_init: np.ndarray,
                   eps_schedule: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator):
    """
    Adam on the log‑sum‑exp surrogate.
    Returns the best probability vector found and its exact objective.
    """
    w = w_init.copy()
    best_a = np.exp(w - np.max(w))
    best_a /= best_a.sum()
    best_val = _exact_obj(best_a, n)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t_iter = 0

    for eps in eps_schedule:
        # inner steps – more steps when eps is tiny (higher fidelity)
        inner_steps = 40 if eps > 1e-6 else 120
        for _ in range(inner_steps):
            if time.time() >= deadline:
                return best_a, best_val

            _, grad_w = _smooth_obj_and_grad(w, eps, n)

            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_w
            v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            lr = 0.05  # a moderate learning rate works well for this scale
            w = w - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # evaluate exact objective occasionally (every 5 steps)
            if t_iter % 5 == 0:
                a = np.exp(w - np.max(w))
                a /= a.sum()
                cur_val = _exact_obj(a, n)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[adam] new best = {best_val:.6f}")

    return best_a, best_val


# ------------------------------------------------------------
# Sub‑gradient descent with back‑tracking line‑search
# ------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Exact sub‑gradient descent on 2·n·max(conv) with line‑search.
    Returns the best vector found.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _exact_obj(a, n)
    step = 0.5

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * a[k - i_range]

        # back‑tracking line‑search
        improved = False
        s = step
        while s > 1e-7:
            a_new = a - s * grad
            a_new = _project_simplex(a_new)
            val_new = _exact_obj(a_new, n)
            if val_new < best_val - 1e-12:   # strict improvement
                best_val = val_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            s *= 0.5

        if not improved:
            step = max(step * 0.85, 1e-5)
        else:
            step = min(step * 1.08, 0.9)

    return best_a, best_val


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Optimises a non‑negative step‑function to minimise ``evaluate_sequence``.
    The routine respects the given time budget and returns the best list
    of floats found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # Different problem sizes – we try a couple (larger n gives more freedom)
    n_choices = [4000, 6000]

    best_overall_a = None
    best_overall_score = float("inf")

    # --------------------------------------------------------
    # Helper: compute the public score (calls evaluate_sequence)
    # --------------------------------------------------------
    def _public_score(vec):
        try:
            return float(evaluate_sequence(vec))
        except Exception:
            return float("inf")

    for n in n_choices:
        if time.time() >= deadline:
            break
        print(f"[run] starting optimisation for n = {n}")

        # ----- generate a few diverse starting points ----------
        inits = []

        # 1) previous construction (interpolated if needed)
        try:
            old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            if old.shape[0] != n:
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                old = np.interp(x_new, x_old, old)
            inits.append(old)
        except Exception:
            pass

        # 2) random positive vector
        inits.append(rng.random(n))

        # 3) sinusoidal shape (smooth bump)
        idx = np.arange(n)
        sinus = np.sin(np.pi * (idx + 0.5) / n) ** 2
        inits.append(sinus)

        # 4) Gaussian‑like bump
        sigma = n / 9.0
        gauss = np.exp(-0.5 * ((idx - n/2) / sigma) ** 2)
        inits.append(gauss)

        # ----- optimisation for every initial vector ------------
        for a0 in inits:
            if time.time() >= deadline:
                break

            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0:
                a0 = rng.random(n) + 1e-3
            a0 /= a0.sum()
            w0 = np.log(a0 + 1e-12)

            # ---- smooth Adam optimisation -----------------------
            eps_schedule = np.logspace(-2, -14, num=30)
            a_opt, val_opt = _adam_optimize(w0, eps_schedule, n, deadline, rng)

            # ---- sub‑gradient refinement ------------------------
            if time.time() < deadline:
                a_opt, val_opt = _subgradient_descent(a_opt, n, deadline, rng)

            # ---- aggressive hill‑climb -------------------------
            if time.time() < deadline:
                a_opt, val_opt = _mass_transfer_hill_climb(a_opt,
                                                          val_opt,
                                                          deadline,
                                                          rng,
                                                          step_start=0.1)

            # ---- evaluate with the official evaluator ------------
            score = _public_score(a_opt.tolist())
            print(f"[final n={n}] candidate score = {score:.6f}")

            if score < best_overall_score:
                best_overall_score = score
                best_overall_a = a_opt.copy()
                print(f"[global] new best overall score = {best_overall_score:.6f}")

        # after each n, give a quick sanity check on remaining time
        remaining = deadline - time.time()
        if remaining < 5.0:
            print("[run] low on time – stopping early")
            break

    # fallback – uniform distribution if nothing succeeded
    if best_overall_a is None:
        n = n_choices[0]
        best_overall_a = np.ones(n) / n

    # final conversion to plain Python list
    return best_overall_a.tolist()
