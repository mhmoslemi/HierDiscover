def update_convolution(b, a, i, j, delta):
    # b is length 2n-1
    # need to update b entries for all sums where i or j appears.
    # The effect of decreasing a_i by delta: for each k, contribution a_i * a_{k-i} appears in b_k (as part of sum over index i).
    #Since b = a * a (convolution), linear in a, but a * a is quadratic; but considering small delta, the exact new b can be computed via:
    # b_new = b - 2*delta * conv_shift_i(a) + 2*delta * conv_shift_j(a) + delta**2 * (e_i * e_i + e_j * e_j - 2 * e_i * e_j)
    # where conv_shift_i(a) is array of length 2n-1: value at k = a_{k-i} (i.e., a shifted).
    # Implementation uses slices: b[i:i+n] -= 2*delta * a (since for k = i + s where s runs 0..n-1, term includes a_i * a_s = a_i * a_s)
    # Similarly for j.

    # However need to ensure that we treat self contributions, double counting? Let's verify earlier algorithm used:
    # b_new[j:j + n] += 2.0 * delta * a
    # b_new[i:i + n] -= 2.0 * delta * a
    # b_new[2 * j] += delta * delta
    # b_new[2 * i] += delta * delta
    # b_new[i + j] -= 2.0 * delta * delta

    # Let's verify correctness: Given a new a' where a'[i] = a[i] - delta, a'[j] = a[j] + delta.
    # The convolution b' = a' * a' = (a - delta e_i + delta e_j)*(a - delta e_i + delta e_j)
    # = a*a - 2 delta e_i * a + 2 delta e_j * a + delta^2 e_i*e_i + delta^2 e_j * e_j - 2 delta^2 e_i*e_j

"""
Propose a step‑function (non‑negative list) that minimises
evaluate_sequence(sequence).

The algorithm consists of three stages.

1.  **Initial library** – a set of simple parametric families
    (uniform, linear, power‑law, exponential, arcsine‑like) is scanned.
    The best shape (normalised to sum = 1) is kept.

2.  **Deterministic hill‑climbing** – starting from the best shape we
    repeatedly move a tiny amount of mass from an index that contributes
    heavily to the current maximal convolution entry to an index that
    contributes very little.  The convolution is updated incrementally
    (O(n) per step).  The step size decays linearly with the elapsed
    time; when no improvement is seen for a long while a small random
    perturbation is added to escape plateaus.  Periodically the full
    convolution is recomputed to avoid numerical drift.

3.  **Uniform mixing sweep** – the final candidate is mixed with the
    uniform distribution for a range of mixing coefficients α; the best
    mixture (still respecting the simplex) is returned.

The whole procedure respects the supplied time budget (default 1000 s)
and never returns an empty list.  All vectors are kept on the probability
simplex, therefore the objective 2·n·max(conv) is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (Σa = 1) → 2·n·max(conv)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective for a normalised (sum = 1) vector."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Library of simple parametric families
# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator):
    """Yield a sequence of promising normalised candidates."""
    # Uniform
    yield np.full(n, 1.0 / n)

    # Linear decreasing (left → right)
    yield np.linspace(1.0, 0.0, n)

    # Linear increasing
    yield np.linspace(0.0, 1.0, n)

    # Linear U‑shape (large on both ends)
    idx = np.arange(n)
    yield np.maximum(idx + 1, n - idx).astype(float)

    # Quadratic decreasing
    yield (np.linspace(1.0, 0.0, n) ** 2)

    # Power‑law symmetric (U‑shape)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p) + (n - i + 1) ** (-p)
        yield a

    # Exponential decay (left‑hand side)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        yield a

    # Symmetric exponential (U‑shape)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        yield a

    # Arcsine‑like shape: a_i ∝ 1/√i + 1/√(n‑i)
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(i) + 1.0 / np.sqrt(n - i + 1)
    yield a

    # Random Dirichlet draws (a few quick samples)
    for _ in range(12):
        yield rng.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(a: np.ndarray,
               deadline: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climbing with incremental convolution updates."""
    n = a.shape[0]
    # initialise convolution once
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    no_improve = 0
    IMPROVE_LIMIT = 250_000          # when to perturb
    step = 0

    # helper: recompute full convolution (used periodically)
    def recompute():
        return _full_convolution(a)

    while True:
        now = time.time()
        if now >= deadline:
            break

        # find the entry of the convolution that is maximal
        k_max = int(b.argmax())

        # contributions of each index to this entry (a_i * a_{k_max-i})
        idx = np.arange(n)
        rev = k_max - idx
        mask = (rev >= 0) & (rev < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[rev[mask]]

        # source: index with the largest contribution
        i = int(np.argmax(contrib))
        if contrib[i] <= 0.0:                 # degenerate – pick a random source
            i = int(np.random.randint(n))

        # target: minimize the partner value a_{k_max-j}
        rev_target = np.full(n, np.inf, dtype=float)
        rev_target[mask] = a[rev[mask]]
        rev_target[i] = np.inf                # never move mass back to itself
        j = int(np.argmin(rev_target))
        if j == i:                             # fallback – pick a different index
            j = (i + 1) % n

        # step‑size schedule – linearly decays from 5 % to 0.1 %
        progress = (now - start_time) / (deadline - start_time)
        max_frac = 0.05 * (1.0 - progress) + 0.001 * progress
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        #   b ← b – 2·δ·a shifted by i   + 2·δ·a shifted by j
        #   plus the quadratic correction terms
        b[i:i + n] -= 2.0 * delta * a
        b[j:j + n] += 2.0 * delta * a
        b[2 * i] += delta * delta
        b[2 * j] += delta * delta
        b[i + j] -= 2.0 * delta * delta

        # update the vector itself (mass transfer)
        a[i] -= delta
        a[j] += delta

        # numerical safety
        if a[i] < -1e-12:                     # should never happen
            a[i] = 0.0

        # evaluate the new objective (exact, no smoothing)
        cur_val = 2.0 * n * b.max()

        # ----- keep the best ever seen -----
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Hill {step:8d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            # perturb around the current best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.005 * (1.0 - progress),
                                    size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            b = recompute()
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            # small log line to indicate a shake
            print(f"[Hill {step:8d}] perturbation after stagnation")

        # occasional re‑normalisation of b to avoid drift
        if step % 10_000 == 0:
            b = recompute()
            cur_val = 2.0 * n * b.max()

        step += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Mixing with the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Try convex combinations a' = (1‑α)·a + α·uniform for several α."""
    n = a.shape[0]
    best_a, best_val = a, _objective(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in np.linspace(0.001, 0.020, 20):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
