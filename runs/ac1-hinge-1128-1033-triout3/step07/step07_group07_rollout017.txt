    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    cur_a = a.copy()
    cur_val = cur_val
    # schedule parameters
    total = deadline - time.time()
    # initial temperature as relative to cur_val: maybe 0.01 * cur_val?
    T0 = cur_val * 0.05  # start high
    while time.time() < deadline:
        # compute temperature linear cooling
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / (deadline - start))  # linear decrease to 0
        if T <= 0:
            T = 1e-12
        # pick random donor and receiver
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # propose move
        # Choose step fraction random maybe uniform in [0, step_max] where step_max scaled by temperature:
        step_frac = rng.random() * min(0.1, T / cur_val)   # ensure small step
        delta = cur_a[i] * step_frac
        if delta <= 0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # ensure non-negative
        # Probability sum still 1 because we transferred mass.
        # Evaluate new value
        new_val = 2.0 * n * np.max(_full_convolution(new_a))
        # Accept condition
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):

            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[SA] new best = {best_val:.6f}")
        # else reject
    return best_a, best_val

Optimiser for the ``evaluate_sequence`` objective.

The optimiser works in four stages:

1.  **Smooth surrogate minimisation** –  the max‑convolution
    is approximated by a log‑sum‑exp surrogate.
    A continuation in the smoothing parameter ε together with
    L‑BFGS‑B (with a soft‑max parametrisation) yields a good
    approximate solution.

2.  **Random‑restart refinement** –  the best vector so far is
    perturbed and the same continuation is performed again.
    Several restarts (up to the time budget) increase the chance
    of escaping local minima.

3.  **Projected sub‑gradient descent** –  the exact
    nonsmooth objective 2 n · max(conv) is reduced by
    sub‑gradient steps on the active lag, with a decreasing step‑size
    and Euclidean projection onto the probability simplex.

4.  **Simulated‑annealing polishing** –  a simple SA loop moves a
    small amount of probability mass between randomly chosen
    positions; uphill moves are accepted with a temperature that
    decays linearly to zero.  This often pushes the bound a little
    further.

All stages respect the global 1000 s time budget.
The returned list is ready for ``evaluate_sequence``.

    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives
    return conv

    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1

    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)

def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns the value and gradient of the surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε)

    where a = softmax(w) (∑a = 1, a_i ≥ 0).
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()

    # full autocorrelation
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing (numerically stable)
    shifted = (conv - max_conv) / eps
    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted - max_shifted)

    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                  # length 3·n‑2

    dot = np.dot(grad_a, a)

def _objective(a: np.ndarray) -> float:
    """Exact 2·n·max(conv) for a probability vector a."""
    n = a.shape[0]
    return float(2.0 * n * np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Adaptive sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         best_val: float,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int | None = None):
    """
    Sub‑gradient descent on 2·n·max(conv) with projection onto the simplex.
    Returns the best vector encountered.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    best_a = cur_a.copy()
    eta0 = 0.2
    it = 0

    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break

        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] new best = {best_val:.6f}")

        cur_a = new_a
        cur_val = new_val
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing polishing phase
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Very light‑weight SA that moves a small amount of probability mass
    between two positions.  The temperature decays linearly.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    cur_a = a.copy()
    cur_val = cur_val
    start = time.time()
    total_time = max(deadline - start, 1e-12)
    T0 = cur_val * 0.05   # initial temperature (a few percent of the objective)

    while time.time() < deadline:
        elapsed = time.time() - start
        T = max(T0 * (1 - elapsed / total_time), 1e-12)

        # draw source index proportional to its current mass
        i = rng.choice(n, p=cur_a)
        j = rng.integers(n)
        if j == i:
            continue

        # step size proportional to temperature
        step_frac = min(0.5, T / cur_val) * rng.random()
        delta = cur_a[i] * step_frac
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Optimises a non‑negative step‑function to minimise ``evaluate_sequence``.
    Uses a multi‑stage pipeline (smooth surrogate, random restarts,
    sub‑gradient descent, simulated annealing) and returns the best
    list of floats found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    n = 4000  # moderate size – gives a good trade‑off between quality and speed

    # ------------------------------------------------------------------
    # Initialise sequence (reuse previous construction if present)
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        else:
            a0 = a0.copy()
    except Exception:
        a0 = rng.random(n)

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    w = np.log(a0 + 1e-12)

    best_a = a0.copy()
    best_val = _objective(best_a)
    print(f"[init] n={n}, objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – smooth optimisation with continuation in ε
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-2, -16, num=30)

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def _fun_and_grad(v, eps_local=eps, n_local=n):
            return _smooth_obj_and_grad(v, eps_local, n_local)

        res = minimize(_fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 400,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – random‑restart refinement (focus on the most aggressive ε)
    # ------------------------------------------------------------------
    restart = 0
    max_restarts = 30
    while time.time() < deadline - 5.0 and restart < max_restarts:
        restart += 1
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.5, size=n)

        for eps in eps_schedule[-12:]:
            if time.time() >= deadline:
                break

            def _fun_and_grad(v, eps_local=eps, n_local=n):
                return _smooth_obj_and_grad(v, eps_local, n_local)

            res = minimize(_fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart {restart}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – adaptive projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _subgradient_descent(best_a,
                                                best_val,
                                                deadline,
                                                rng,
                                                max_iters=20000)

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _simulated_annealing(best_a,
                                                best_val,
                                                deadline,
                                                rng)

    # ------------------------------------------------------------------
    # Final reporting via the provided evaluator (optional)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
