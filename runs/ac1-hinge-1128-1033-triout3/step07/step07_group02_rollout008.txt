"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Approach
--------
1. *Initialisation* – try to reuse the previous solution, otherwise
   sweep a two‑parameter Beta family (also an exponential family) and
   keep the best seed.
2. *Sub‑gradient descent* – minimise the exact objective
   ``2·n·max(conv(a,a))`` by repeatedly moving opposite to the sub‑gradient
   of the current maximal convolution entry and projecting back onto the
   simplex.
3. *Adam optimiser* – a smooth‑max (log‑sum‑exp) approximation of the
   same objective is optimised with Adam on the simplex.
4. *Simulated‑annealing hill‑climbing* – fine‑tune the vector by small
   mass‑transfer moves biased away from the currently dominant
   convolution index.
5. *Final evaluation* – the best vector found is returned.

All phases respect a global 1000 s budget (minus a 0.5 s safety margin).
The implementation uses only NumPy and the provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the simplex {x >= 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                     # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_size: int, conv_len: int) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)-1)."""
    fx = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fx * fx, fft_size)[:conv_len]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray, fft_size: int, conv_len: int) -> np.ndarray:
    """Full linear convolution of a and b."""
    fx = np.fft.rfft(a, fft_size)
    fy = np.fft.rfft(b, fft_size)
    conv = np.fft.irfft(fx * fy, fft_size)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_size: int, conv_len: int) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    b = _full_convolution(a, fft_size, conv_len)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float,
                         n: int, fft_size: int, conv_len: int):
    """
    Returns a smooth approximation of max(conv(a,a)) and its gradient.
    """
    b = _full_convolution(a, fft_size, conv_len)                # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    w = np.exp(shifted)
    p = w / w.sum()                                            # (2n‑1,)
    # gradient of Σ p_k * b_k w.r.t. a:
    # ∂b_k/∂a_i = a_{k-i} + a_i   (ordered pair count = 2)
    # The term "a_i" leads to a constant factor once summed over k,
    # therefore we compute the convolution a * rev(p) as in the original code.
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev, fft_size, conv_len)          # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]                         # n entries
    grad_a = 4.0 * n * segment[::-1]                           # factor 4n
    smooth_val = max_b + eps * np.log(w.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (logits parameterisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   start_time: float, deadline: float,
                   best_a: np.ndarray, best_val: float,
                   target_val: float = 1.503):
    """Adam optimiser that minimises a smooth approximation of the objective."""
    # logits – guarantees Σa = 1 after soft‑max
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.10, 0.001               # more conservative schedule
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max -> a on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a, n, _fft_size, _conv_len)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedules (linear in log‑time)
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        smooth_val, grad_a = _smooth_max_and_grad(a, eps, n, _fft_size, _conv_len)

        # chain rule through the soft‑max (gradient w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation (helps escape flat regions)
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        # restart on stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restarting after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray, n: int,
                     start_time: float, deadline: float,
                     best_a: np.ndarray, best_val: float,
                     target_val: float = 1.503):
    """
    Simple projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Uses the exact sub‑gradient of the current maximal convolution entry.
    """
    a = a0.copy()
    it = 0
    step0 = 0.5                     # base step size (will be scaled down)
    beta = 0.9                      # momentum coefficient
    momentum = np.zeros_like(a)

    while True:
        now = time.time()
        if now >= deadline:
            break

        # compute convolution and current objective
        b = _full_convolution(a, _fft_size, _conv_len)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        # index of the maximal convolution entry
        kstar = int(np.argmax(b))

        # sub‑gradient of that entry (ignoring the outer 4·n factor)
        g = np.zeros_like(a)
        i_start = max(0, kstar - (n - 1))
        i_end = min(n - 1, kstar)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = kstar - i_range
        g[i_range] = a[j_range]

        # momentum + step size
        it += 1
        step = step0 / math.sqrt(it)
        momentum = beta * momentum + (1.0 - beta) * g
        a -= step * momentum

        # project back onto the simplex
        a = _proj_simplex(a)

        # occasional tiny random perturbation to avoid getting stuck
        if it % 5000 == 0:
            noise = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (same idea as the reference code)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray, n: int,
                      start_time: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503):
    """Local SA that moves a little mass from “bad’’ to “good’’ indices."""
    a = a0.copy()
    b = _full_convolution(a, _fft_size, _conv_len)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules
    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.012, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # most dominant convolution entry
        k_max = int(np.argmax(b))

        # contributions of each coefficient to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose source index (probability ∝ contributing amount)
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # choose a target index where contribution is small
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to relocate
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # self‑interaction corrections
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a, _fft_size, _conv_len)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# Parametric families – Beta and exponential
# ----------------------------------------------------------------------
def _beta_family_best(n: int):
    """Search a two‑parameter Beta family a_i ∝ i^{p-1}(n‑i+1)^{q-1}."""
    best_a = None
    best_val = np.inf
    # moderate grid (log‑space) for the exponents
    p_vals = np.concatenate([np.linspace(0.2, 2.0, 9),
                             np.geomspace(2.0, 12.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a, n, _fft_size, _conv_len)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _exp_family_best(n: int):
    """Search a simple exponential decay family a_i ∝ exp(-λ·i)."""
    best_a = None
    best_val = np.inf
    lam_vals = np.geomspace(0.5, 30.0, 15)   # λ varies from gentle to steep
    i = np.arange(n, dtype=np.float64)
    for lam in lam_vals:
        a = np.exp(-lam * i)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a, n, _fft_size, _conv_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val

        a /= a.sum()
        best_a, best_val = a, _objective(a, n, _fft_size, _conv_len)
        print("[init] random start, value=", best_val)

    # ------------------------------------------------------------------
    # 2. Sub‑gradient descent (≈10 % of the budget)
    # ------------------------------------------------------------------
    subgrad_deadline = start_time + 0.10 * budget_s
    best_a, best_val = _subgrad_descent(best_a, n,
                                        start_time, subgrad_deadline,
                                        best_a, best_val,
                                        target_val=1.503)

    # ------------------------------------------------------------------
    # 3. Adam optimisation (≈30 % of the budget)
    # ------------------------------------------------------------------
    adam_deadline = start_time + 0.40 * budget_s          # 10 % + 30 %
    best_a, best_val = _adam_optimize(best_a, n,
                                      start_time, adam_deadline,
                                      best_a, best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing hill‑climbing (rest of the budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         start_time, global_deadline,
                                         best_a, best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # 5. Final evaluation (wrap with the provided secure checker)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
