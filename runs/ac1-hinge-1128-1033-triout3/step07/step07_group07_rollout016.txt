The evaluator is invariant under a global scaling of the sequence.
Therefore we work with a probability vector `p` (∑p = 1) and the true
objective is  

```
obj(p) = 2·n·max(conv(p,p))
```

The algorithm consists of four stages, each respecting the global time
budget:

1. **Smoothed surrogate optimisation** –  
   Replace the non‑smooth `max` by a log‑sum‑exp surrogate with a
   continuation on the smoothing parameter ε.  The surrogate is
   differentiable; we minimise it with L‑BFGS‑B and, after the ε‑schedule,
   continue with many Adam steps.

2. **Random restarts** –  
   Perturb the current logits and repeat the ε‑schedule to escape
   possible local minima.

3. **Aggressive mass‑transfer hill climbing** –  
   Identify the lag that attains the current maximum, pick a few
   contributing index‑pairs, move a fraction of their mass towards low‑mass
   positions and accept the move only if the exact objective improves.

4. **Projected sub‑gradient descent** –  
   Perform a classic sub‑gradient step on the exact objective
   `2·n·max(conv)` with projection onto the probability simplex.

At the end we rescale the probability vector to the required
normalisation `√(2·n)` and return it as a plain list of floats.  All
intermediate progress is printed, and the best sequence found at any
time is always kept.

```python
"""
Improved heuristic optimiser for `evaluate_sequence`.

The optimiser works with a probability vector (∑p = 1) because the
objective 2·n·max(conv(p,p)) is invariant under scaling.  It proceeds in
four phases:

1. Smoothed surrogate minimisation (log‑sum‑exp) with an ε‑schedule,
   first using L‑BFGS‑B and then many Adam steps.
2. Random restarts to explore other basins.
3. Aggressive mass‑transfer hill climbing that moves mass away from the
   lag producing the current maximum.
4. Projected sub‑gradient descent on the exact nonsmooth objective.

All phases respect the global time budget (default 1000 s) and the best
sequence found is returned (scaled to √(2·n) as required by the
evaluator).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself (autocorrelation
    without reversal).  Length = 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Eliminate tiny negative round‑off errors
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (value, grad) of the smoothed objective

        L_ε(p) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε)

    where p = softmax(w) (∑p = 1, p ≥ 0).
    """
    # soft‑max (feasible probability vector)
    w_shift = w - np.max(w)
    p = np.exp(w_shift)
    p = p / p.sum()

    # full autocorrelation of p
    conv = _full_convolution(p)
    max_conv = conv.max()

    # log‑sum‑exp part
    shifted = (conv - max_conv) / eps
    # numerically stable log‑sum‑exp
    log_sum = np.logaddexp.reduce(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. p (see derivation in the description)
    weights = np.exp(shifted)
    prob = weights / weights.sum()                     # probabilities over lags

    # convolution p * rev(prob)
    prob_rev = prob[::-1]
    conv_ap = _conv_fft(p, prob_rev)          # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]           # middle n entries
    grad_p = 4.0 * n * seg[::-1]             # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_p, p)
    grad_w = p * (grad_p - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Aggressive mass‑transfer hill climbing
# ----------------------------------------------------------------------


def _mass_transfer_hill_climb(p: np.ndarray,
                              best_val: float,
                              deadline: float,
                              rng: np.random.Generator,
                              step_start: float = 0.2):
    """
    Move a fraction of mass away from the lag attaining the maximal
    convolution.  Accept only if the exact objective improves.
    """
    n = p.shape[0]
    cur_p = p.copy()
    cur_val = best_val
    best_p = cur_p.copy()
    step_frac = step_start

    while time.time() < deadline:
        conv = _full_convolution(cur_p)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # indices that contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_p[i_range] * cur_p[j_range]

        if contrib.size == 0:
            continue

        top_n = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        # destination – indices with relatively low mass
        low_mass = np.where(cur_p < cur_p.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_p = cur_p.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_p[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_p[i] -= delta
                p_idx = int(dest[2 * d])
                new_p[p_idx] += delta
            else:
                delta = min(new_p[i], new_p[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_p[i] -= delta
                new_p[j] -= delta
                p_idx = int(dest[2 * d])
                q_idx = int(dest[2 * d + 1])
                new_p[p_idx] += delta
                new_p[q_idx] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-7)
            continue

        # renormalise (projection onto simplex is cheap here)
        new_p = _project_simplex(new_p)

        new_val = 2.0 * n * np.max(_full_convolution(new_p))

        if new_val < cur_val:
            cur_p = new_p
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_p = cur_p.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-7)

    return best_p, best_val


# ----------------------------------------------------------------------
# Fine sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_descent(p: np.ndarray,
                         best_val: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Sub‑gradient descent on 2·n·max(conv) with Euclidean projection onto
    the simplex.
    """
    n = p.shape[0]
    cur_p = p.copy()
    cur_val = best_val
    best_p = cur_p.copy()
    eta = 0.1  # initial step size

    while time.time() < deadline:
        conv = _full_convolution(cur_p)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag
        grad = np.zeros_like(cur_p)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_p[k - i_range]

        # gradient step + projection
        new_p = cur_p - eta * grad
        new_p = _project_simplex(new_p)

        new_val = 2.0 * n * np.max(_full_convolution(new_p))

        if new_val < cur_val:
            cur_p = new_p
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_p = cur_p.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            eta = min(eta * 1.05, 0.5)
        else:
            eta = max(eta * 0.7, 1e-6)

    return best_p, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Optimises a non‑negative step‑function to minimise ``evaluate_sequence``.
    The routine respects the given time budget and returns the best list of
    floats found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # Problem size – a larger n gives a richer search space.
    n = 5000
    sqrt_2n = np.sqrt(2.0 * n)  # required scaling for the final output

    # ------------------------------------------------------------------
    # Initialise probability vector (sum = 1)
    # ------------------------------------------------------------------
    try:
        # reuse a previously discovered construction if present
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()                # normalise to a probability vector
    except Exception:
        # fallback: random Dirichlet‑like start
        a0 = rng.random(n)
        a0 = a0 / a0.sum()

    # logits for the soft‑max representation
    w = np.log(a0 + 1e-12)

    # exact objective for the initial point
    best_p = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_p))
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Smoothed optimisation – ε‑continuation with L‑BFGS‑B
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-2, -12, num=30)   # from 1e‑2 down to 1e‑12

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        res = minimize(fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # exact evaluation of the current logits
        p = np.exp(w - np.max(w))
        p = p / p.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(p))
        if cur_val < best_val:
            best_val = cur_val
            best_p = p.copy()
            print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Adam optimisation on the smallest ε (fine‑grained descent)
    # ------------------------------------------------------------------
    eps = eps_schedule[-1]          # the tightest smoothing parameter
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    adam_lr = 0.05
    step = 0

    while time.time() < deadline - 5.0:
        L, grad_w = _smooth_obj_and_grad(w, eps, n)
        step += 1
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        w = w - adam_lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # periodic exact check
        if step % 200 == 0:
            p = np.exp(w - np.max(w))
            p = p / p.sum()
            cur_val = 2.0 * n * np.max(_full_convolution(p))
            if cur_val < best_val:
                best_val = cur_val
                best_p = p.copy()
                print(f"[adam] step {step} new best = {best_val:.6f}")

        # slowly decay the learning rate
        if step % 1000 == 0:
            adam_lr *= 0.9

    # ------------------------------------------------------------------
    # 3️⃣ Random restarts (perturb current optimum and redo ε‑schedule)
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 10.0 and restarts < 12:
        restarts += 1
        w = np.log(best_p + 1e-12) + rng.normal(scale=0.6, size=n)

        for eps in eps_schedule[-10:]:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 300,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        p = np.exp(w - np.max(w))
        p = p / p.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(p))
        if cur_val < best_val:
            best_val = cur_val
            best_p = p.copy()
            print(f"[restart {restarts}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Aggressive hill‑climbing phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_p, best_val = _mass_transfer_hill_climb(best_p,
                                                     best_val,
                                                     deadline,
                                                     rng,
                                                     step_start=0.3)

    # ------------------------------------------------------------------
    # 5️⃣ Fine sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_p, best_val = _subgradient_descent(best_p,
                                                best_val,
                                                deadline,
                                                rng)

    # ------------------------------------------------------------------
    # 6️⃣ Return the sequence scaled to √(2·n) (any positive scaling yields the
    #    same evaluate_sequence value, but this scaling matches the original
    #    normalisation used in the reference method).
    # ------------------------------------------------------------------
    result = (best_p * sqrt_2n).tolist()

    # final sanity check (optional)
    try:
        final_score = evaluate_sequence(result)
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return result
```
