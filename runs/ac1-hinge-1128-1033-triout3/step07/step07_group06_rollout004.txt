"""
Step‑function minimiser.

The algorithm works in three phases inside the allotted time budget.

1. **Flattening (L2‑stage).**
   We minimise the variance of the self‑convolution:
       F(a) = Σ_k ( (a*a)[k] – μ )² ,   μ = 1/(2n‑1)
   (with the L1‑norm of `a` fixed to 1).  The exact gradient is
   derived analytically and evaluated by FFTs.  After each gradient
   step the vector is projected onto the probability simplex.

2. **Max‑stage (projected sub‑gradient).**
   Starting from the best point of phase 1 we run a classic
   projected sub‑gradient descent on the true objective
       G(a) = 2·n·max_k (a*a)[k] .
   This sharpens the worst‑lag entry.

3. **Aggressive mass‑transfer hill‑climbing.**
   As in the reference code, mass from the most damaging pair of
   indices is shifted to two low‑mass locations whenever this improves
   the objective.

All phases respect the global time limit (default 1000 s) and constantly
track the best sequence seen.  The final sequence is returned as a list
of non‑negative floats; `evaluate_sequence` is invariant under scaling,
so we keep the L1‑norm equal to 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT‑based convolutions)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: len(x) + len(y) - 1]
    # Clip tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x≥0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal after projection – fall back to uniform.
        return np.full(n, 1.0 / n, dtype=float)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 1 – L2 flattening
# ----------------------------------------------------------------------


def _gradient_l2(a: np.ndarray, target: float) -> np.ndarray:
    """
    Gradient of Σ_k ( (a*a)[k] – target )² with respect to `a`.
    The formula reduces to
        grad = 4 * ( (conv – target) ⋆ a )[i],
    where ⋆ denotes cross‑correlation.
    """
    n = a.shape[0]
    conv = _full_convolution(a)               # length 2n‑1
    B = conv - target                          # deviation from the flat target
    a_rev = a[::-1]                            # reversed a for correlation
    # Full correlation via convolution with the reversed vector
    corr_full = _conv_fft(B, a_rev)            # length 3n‑2
    # Extract the valid part (indices n‑1 … 2n‑2 inclusive)
    grad = 4.0 * corr_full[n - 1 : n - 1 + n]
    return grad


def _flatten_stage(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the L2‑objective that tries to make the
    self‑convolution as flat as possible.
    """
    a = a0.copy()
    n = a.shape[0]
    target = 1.0 / (2 * n - 1)                 # the average value of the convolution
    best_a = a.copy()
    best_val = _objective(a)

    step0 = 0.5                                 # initial learning‑rate multiplier

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # Gradient of the flatness loss
        grad = _gradient_l2(a, target)

        # Simple diminishing step size
        lr = step0 / np.sqrt(it + 1.0)

        a_trial = a - lr * grad
        a_trial = _project_simplex(a_trial)

        val = _objective(a_trial)

        if val < best_val:
            best_val = val
            best_a = a_trial.copy()
            step0 = min(step0 * 1.02, 2.0)      # slightly accelerate after success
            print(f"[flatten] iter {it:5d}  val = {best_val:.6f}")
        else:
            step0 = max(step0 * 0.97, 1e-5)      # back‑off on failure

        a = a_trial

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the max‑objective
# ----------------------------------------------------------------------


def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry (a*a)[k]
    for non‑negative `a`.  It is 2·a_{k‑i} on the feasible range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on f(a)=2·n·max(a*a).
    Returns the best point encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    step0 = 0.5
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        grad = _gradient_of_lag(a, k)

        # Back‑tracking line search (max 10 attempts)
        lr = step0 / np.sqrt(t)
        improved = False
        for _ in range(10):
            a_trial = a - lr * grad
            a_trial = np.clip(a_trial, 0.0, None)
            if a_trial.sum() == 0.0:
                lr *= 0.5
                continue
            a_trial = _project_simplex(a_trial)
            val = _objective(a_trial)
            if val < best_val:
                a = a_trial
                best_a = a_trial.copy()
                best_val = val
                improved = True
                step0 = min(step0 * 1.02, 2.0)
                if t % 200 == 0:
                    print(f"[subgrad] iter {t:5d}  val = {best_val:.6f}")
                break
            else:
                lr *= 0.5

        if not improved:
            step0 = max(step0 * 0.95, 1e-4)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Aggressive mass‑transfer hill‑climbing
# ----------------------------------------------------------------------


def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that repeatedly moves a fraction of the most damaging
    mass to two low‑mass positions.  Stops after the time budget expires
    or after `max_no_improve` unsuccessful attempts.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # Indices that participate in the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # Amount of mass to relocate
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # Choose two distinct low‑mass destinations
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        new_a = _project_simplex(new_a)

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Initialise – reuse previous construction if it exists.
    # ------------------------------------------------------------------
    default_n = 4000
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        a0 = np.maximum(old, 0.0)
        if a0.sum() == 0.0:
            a0 = rng.random(n)
    except Exception:
        n = default_n
        a0 = rng.random(n)

    a0 = _project_simplex(a0)                # guarantees Σa = 1 and a≥0
    print(f"[init] n={n}, initial objective = {_objective(a0):.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – flatten the convolution using an L2 loss.
    # ------------------------------------------------------------------
    a_flat, val_flat = _flatten_stage(a0, deadline, rng,
                                      max_iters=3000)
    print(f"[flatten] best value after flatten = {val_flat:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true max objective.
    # ------------------------------------------------------------------
    a_sub, val_sub = _subgradient_descent(a_flat, deadline, rng,
                                          max_iters=8000)
    print(f"[subgrad] best value after sub‑gradient = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive mass‑transfer hill‑climbing.
    # ------------------------------------------------------------------
    a_final, val_final = _mass_transfer_hillclimb(a_sub, deadline, rng,
                                                  max_no_improve=3000)
    print(f"[final] best value before evaluate_sequence = {val_final:.6f}")

    # ------------------------------------------------------------------
    # Report via the provided evaluator (for logging purposes only).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_final.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a_final.tolist()
