def random_local_search(a, deadline, rng):
    best_a = a.copy()
    best_val = _objective(a)
    cur_a = a.copy()
    cur_val = best_val
    
    temp0 = 0.01  # initial temperature scaling relative to best_val.
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # typical step size reduces over time
        # choose i != j
        i, j = rng.choice(n, size=2, replace=False)
        # choose delta as fraction of a[i]
        max_move = cur_a[i]
        if max_move == 0:
            continue
        delta = max_move * rng.uniform(0.001, 0.05)  # 0.1% to 5% of a[i]
        # new distribution
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # Ensure non-negative
        # (a_i may become negative if delta > a[i], but we used fraction)
        # compute objective
        new_val = _objective(new_a)
        # acceptance: if new_val < cur_val, accept; else maybe accept with prob exp(-(new_val-cur_val)/temp)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (temp0 + iteration * 1e-4)):

                best_a = new_a.copy()
                # optionally print
        # gradually reduce temperature
        temp0 *= 0.9999
        if iteration % 5000 == 0:

    return best_a, best_val

"""
Solution for the step‑function minimisation problem.

The algorithm combines several classic optimisation ideas:

1.  **Multiple random restarts** – we generate a few Dirichlet‑distributed
    probability vectors (sum = 1) and run a short projected sub‑gradient
    descent on each.  The best of these starts is kept.

2.  **Projected sub‑gradient descent** – the exact objective for a normalised
    vector `a` is `2·n·max(conv(a,a))`.  The sub‑gradient of the active
    convolution entry is cheap to compute; after each step we project back
    onto the probability simplex (clip to non‑negative and renormalise).

3.  **Aggressive hill‑climbing** – starting from the best vector we
    repeatedly identify the lag `k` where the convolution attains its
    maximum and the pair `(i,j)` that contributes most to this lag.
    A small fraction of the mass of `i` and `j` is moved to two low‑mass
    positions.  The move is kept only if the true objective improves.
    The transferred‑mass fraction is adapted dynamically.

4.  **Fine‑grained random local search** – after the hill‑climb we perform
    many tiny random mass‑transfer moves (simulated‑annealing style) to
    escape possible plateaux and squeeze the objective a little further.

All steps respect the global time budget (default 1000 s).  The routine
returns the best sequence it has found; `evaluate_sequence` is called only
once at the very end to report the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # Numerical noise can create tiny negatives – clip them away.
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` when ``sum(a) == 1``:
    2·n·max(conv(a,a)).
    The optimiser always works with normalised vectors, therefore this
    is the quantity we minimise.
    """
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry `conv[k]` w.r.t. `a`
    (for non‑negative a).  The derivative w.r.t. a_i is 2·a_{k‑i}
    whenever the pair (i, k‑i) is feasible.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on `f(a) = 2·n·max(conv(a,a))`.
    `a` is kept on the probability simplex (non‑negative, sum = 1).
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    step_factor = 0.5  # initial multiplier for the learning rate
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _full_convolution(a)
        maxb = conv.max()
        # pick a random lag among all maximisers
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        grad = _gradient_of_lag(a, k)

        # simple back‑tracking line search
        lr = step_factor / np.sqrt(t)
        improved = False
        for _ in range(10):
            a_trial = a - lr * grad
            a_trial = np.clip(a_trial, 0.0, None)
            s = a_trial.sum()
            if s == 0.0:
                lr *= 0.5
                continue
            a_trial /= s
            val = _objective(a_trial)
            if val < best_val:
                a = a_trial
                best_a = a_trial.copy()
                best_val = val
                improved = True
                step_factor = min(step_factor * 1.02, 2.0)
                break
            lr *= 0.5

        if not improved:
            step_factor = max(step_factor * 0.95, 1e-4)

    return best_a, best_val


def _aggressive_hillclimb(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_no_improve: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that repeatedly transfers a fraction of the most
    damaging mass to two low‑mass positions.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices contributing to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # mass we try to relocate
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # choose two low‑mass destinations, avoiding the donors
        low = np.argsort(new_a)[:4]  # a few smallest entries
        forbidden = {i, j}
        cand = [idx for idx in low if idx not in forbidden]
        if len(cand) < 2:
            cand = [idx for idx in range(n) if idx not in forbidden]
        p, q = rng.choice(cand, size=2, replace=False)

        new_a[p] += delta
        new_a[q] += delta

        # non‑negativity check (numeric safety)
        if np.any(new_a < -1e-12):
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[hillclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val


def _fine_grained_random_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_steps: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Very small random perturbations (simulated‑annealing style) to
    escape possible plateaux after the deterministic hill‑climb.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)
    cur_val = best_val

    temperature = 0.02  # initial “temperature”

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] == 0.0:
            continue

        # move a tiny fraction of mass from i to j
        frac = rng.uniform(1e-4, 5e-2)          # between 0.01 % and 5 %
        delta = a[i] * frac
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # evaluate
        new_val = _objective(new_a)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / temperature):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[random] step {step} new best = {best_val:.6f}")

        # cool down slowly
        temperature *= 0.9999
        if temperature < 1e-8:
            temperature = 1e-8

    return best_a, best_val
