def simplex_projection(v): 
    # Project onto simplex {x>=0, sum x = 1}
    mu = np.sort(v)[::-1]
    cumulative = np.cumsum(mu)
    rho = np.where(mu - (cumulative - 1) / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = (cumulative[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Will use double precision.

"""
Propose a step‑function (non‑negative vector) that minimizes
`evaluate_sequence`.  

The algorithm consists of three stages:

1. **Initialisation** – a quick sweep of several parametric families
   (uniform, linear, power‑law, exponential, etc.) and a few Dirichlet
   draws.  The best of these is kept as the starting point.

2. **Targeted line‑search hill‑climber** – repeatedly identify the
   convolution index `k_max` where the auto‑correlation is maximal.
   The mass that contributes most to this entry is moved to a location
   that contributes the least, choosing the transfer size so that the
   maximal entry is reduced (approximately) to the second‑largest entry.
   The convolution is updated incrementally, and a simplex projection is
   enforced implicitly by moving mass between two coordinates.

3. **Polishing** – if the search stalls, a small Gaussian perturbation is
   added and the search continues.  The best vector seen is finally
   reported to `evaluate_sequence`.

The whole search respects a 1000 s wall‑clock budget and returns a
list of non‑negative floats whose sum is 1 (scale‑invariant for the
objective).
"""

import time
import numpy as np
from scipy.optimize import linprog  # only used if the LP fallback is needed

# ----------------------------------------------------------------------
# Convolution helpers (FFT based – full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple simplex projection (used only for safety)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x >= 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D vector")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerically degenerate – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families – a fast, deterministic sweep for a good start
# ----------------------------------------------------------------------
def _initial_candidate(n: int) -> np.ndarray:
    """Return a promising starting vector (sum = 1)."""
    best = np.full(n, 1.0 / n)  # uniform baseline
    best_val = 2.0 * n * _full_convolution(best).max()

    def consider(a):
        nonlocal best, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n or np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = 2.0 * n * _full_convolution(a).max()
        if val < best_val:
            best, best_val = a.copy(), val

    # Uniform
    consider(best)

    # Linear decreasing (left → right)
    consider(np.arange(n, 0, -1, dtype=float))

    # Linear U‑shape
    idx = np.arange(n)
    consider(np.maximum(idx + 1, n - idx))

    # Power‑law U‑shape (p from 0.1 to 2.5)
    for p in (0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        consider(a)

    # Symmetric exponential (r close to 1)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        consider(a)

    # Truncated exponential on a prefix
    for frac in (0.10, 0.15, 0.20, 0.25):
        K = max(1, int(frac * n))
        for r in (0.99, 0.98, 0.95):
            a = np.zeros(n, dtype=float)
            a[:K] = r ** np.arange(K, dtype=float)
            consider(a)

    # Random Dirichlet – a few draws
    rng = np.random.default_rng(12345)
    for _ in range(15):
        consider(rng.dirichlet(np.ones(n)))

    return best


# ----------------------------------------------------------------------
# Targeted line‑search hill‑climber
# ----------------------------------------------------------------------
def _targeted_line_search(a: np.ndarray,
                          n: int,
                          start: float,
                          deadline: float,
                          best_a: np.ndarray,
                          best_val: float) -> tuple[np.ndarray, float]:
    """
    Main optimisation loop.
    - At each iteration the index `k_max` where the self‑convolution is maximal
      is identified.
    - The source index `i` with the largest contribution `a_i * a_{k_max-i}` is
      selected, and the target index `j` with the smallest contribution to the
      same entry is selected.
    - The transfer amount `delta` is chosen so that the maximal entry would be
      (approximately) reduced to the second‑largest entry.
    - The convolution is updated incrementally.  If no improvement occurs,
      a smaller `delta` is tried; after many idle steps a small Gaussian shake
      is applied.
    """
    b = _full_convolution(a)                # current convolution
    cur_val = 2.0 * n * b.max()
    no_improve = 0
    stagnation_limit = 200_000
    iter_cnt = 0
    rng = np.random.default_rng(0)

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- locate the peak and the second‑largest entry -----
        k_max = int(np.argmax(b))
        max_b = b[k_max]

        # second largest (simple copy – cheap for n≈9 k)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_b = b_tmp.max()

        # ----- gradient‑like vector a_rev (a_{k_max‑i}) -----
        a_rev = np.zeros_like(a)
        if k_max < n:
            a_rev[:k_max + 1] = a[k_max::-1]
        else:
            start_idx = k_max - n + 1
            a_rev[start_idx:] = a[:2 * n - 1 - k_max][::-1]

        # ----- source (largest contribution) and target (smallest) -----
        contrib = a * a_rev                         # element‑wise product
        i = int(np.argmax(contrib))

        # avoid picking the same index as source
        mask = np.arange(n) != i
        # prefer an index whose a_rev is minimal (zero is ideal)
        j = int(np.argmin(np.where(mask, a_rev, np.inf)))

        # ---- compute a safe transfer amount ----
        diff = a_rev[i] - a_rev[j]                  # reduction per unit mass
        if diff <= 0.0:
            # fallback: random pair
            i_candidates = np.where(contrib > 0)[0]
            if i_candidates.size == 0:
                i = rng.integers(0, n)
            else:
                i = int(rng.choice(i_candidates))
            j = int(rng.choice(np.setdiff1d(np.arange(n), [i])))
            diff = a_rev[i] - a_rev[j] + 1e-12

        # We would like the maximal entry to drop to `second_b`.
        # Linear approximation: reduction ≈ 2 * delta * diff.
        desired = max_b - second_b
        if desired <= 0:
            # already flat enough – random small move
            delta = a[i] * 0.01
        else:
            delta_needed = desired / (2.0 * diff + 1e-12)
            # Restrict delta to a safe fraction of the source mass
            delta = min(a[i] * 0.9, delta_needed, a[i])

        # Guard against pathological tiny deltas
        if delta < 1e-12:
            delta = a[i] * 0.01  # tiny but positive

        # ----- incremental update of the convolution -----
        # The formulas are derived from moving `delta` from i → j.
        b_new = b.copy()
        # slices are always inside the 0 … 2n‑2 range
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        idx_2i = 2 * i
        idx_2j = 2 * j
        idx_ij = i + j
        if idx_2i < b_new.size:
            b_new[idx_2i] += delta * delta
        if idx_2j < b_new.size:
            b_new[idx_2j] += delta * delta
        if idx_ij < b_new.size:
            b_new[idx_ij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < max_b - 1e-15:      # strict improvement
            # accept the move
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = 2.0 * n * new_max
            no_improve = 0
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[LineSearch {iter_cnt:7d}] new best = {best_val:.6f}")
        else:
            # No improvement – try a reduced delta (half) a couple of times
            improved = False
            for _ in range(3):
                delta *= 0.5
                if delta < 1e-15:
                    break
                b_tmp = b.copy()
                b_tmp[j:j + n] += 2.0 * delta * a
                b_tmp[i:i + n] -= 2.0 * delta * a
                if idx_2i < b_tmp.size:
                    b_tmp[idx_2i] += delta * delta
                if idx_2j < b_tmp.size:
                    b_tmp[idx_2j] += delta * delta
                if idx_ij < b_tmp.size:
                    b_tmp[idx_ij] -= 2.0 * delta * delta
                if b_tmp.max() < max_b - 1e-15:
                    a[i] -= delta
                    a[j] += delta
                    b = b_tmp
                    cur_val = 2.0 * n * b.max()
                    improved = True
                    break
            if not improved:
                no_improve += 1

        # ----- stagnation handling -----
        if no_improve > stagnation_limit:
            # small random shake, then re‑normalise
            noise = rng.normal(scale=0.003, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[LineSearch {iter_cnt:7d}] perturb after stagnation")
        iter_cnt += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # problem size – 9000 gives a good trade‑off between quality and speed
    n = 9000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previously found construction if present
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        # interpolate to the required length
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0 = _initial_candidate(n)

    best_a = a0.copy()
    best_val = 2.0 * n * _full_convolution(best_a).max()
    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – targeted line‑search hill‑climber for the bulk of the budget
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.90 * (deadline - start_time)
    best_a, best_val = _targeted_line_search(
        best_a,
        n,
        start=start_time,
        deadline=phase1_deadline,
        best_a=best_a,
        best_val=best_val,
    )

    # ------------------------------------------------------------------
    # Phase 2 – final very short uniform mixing (sometimes helps)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010):
        cand = (1.0 - alpha) * best_a + alpha * uniform
        cand /= cand.sum()
        val = 2.0 * n * _full_convolution(cand).max()
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Mix] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
