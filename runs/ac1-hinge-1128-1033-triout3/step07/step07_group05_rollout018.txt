"""
Hybrid optimizer for the “step‑function” evaluation problem.

The new version improves on the previous baseline by

* trying several random restarts of the smooth‑max surrogate (L‑BFGS) with a
  finer ε‑schedule,
* testing two deterministic re‑orderings (the original “ends‑large” and a
  new alternating‑ends scheme) and keeping the better one,
* running a longer projected sub‑gradient descent with a modest momentum term,
* performing a much more aggressive simulated‑annealing phase that uses three
  move types (swap, insertion, segment‑reversal) and a slower temperature
  decay, and
* keeping the final cheap hill‑climbing refinement.

All phases respect the total wall‑time budget (default 1000 s) and the best
sequence found so far is always reported via ``evaluate_sequence`` at the end.
The code works with a single CPU and uses only ``numpy``, ``scipy`` and the
provided ``evaluate_sequence`` validator.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for L‑BFGS
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT (real‑valued)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # guard against tiny negative round‑off
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast evaluation that matches ``evaluate_sequence`` (no safety checks)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate of the max‑convolution.

    L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    Returns (L, grad_w) where grad_w is the gradient w.r.t. the soft‑max
    parameters w.
    """
    # ---- soft‑max (positivity & ℓ₁ = 1) ----
    w_shift = w - np.max(w)            # stability
    a = np.exp(w_shift)
    a /= np.sum(a)                     # Σa = 1

    # ---- convolution a*a ----
    conv = _fft_convolve_self(a)       # length 2·n‑1
    max_conv = conv.max()

    # ---- smooth max approximation ----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)       # log∑exp((c−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a ----
    weights = np.exp(shifted)
    p = weights / np.sum(weights)      # probabilities over shift indices

    # g_i = 4·n·∑_k p_k·a_{k−i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)      # length 2·n‑1
    seg = conv_ap[n - 1: 2 * n - 1]    # central n entries
    grad_a = 4.0 * n * seg[::-1]       # ∂L/∂a

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# bind the helper convolution to the function object for the wrapper
_smooth_obj_and_grad._conv_fft = _conv_fft


def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic re‑ordering used previously:
    largest, smallest, second‑largest, second‑smallest, …
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left_ptr = 0
    right_ptr = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right_ptr]
        right_ptr -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left_ptr]
        left_ptr += 1
        i += 1
    return new_a


def _alternating_ends_reorder(a: np.ndarray) -> np.ndarray:
    """
    New heuristic: place the largest, second‑largest at opposite ends,
    then third‑largest next to the left end, fourth‑largest next to the right,
    and so on.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    idx = n - 1                       # start from the largest
    while left <= right:
        new_a[left] = sorted_vals[idx]
        idx -= 1
        left += 1
        if left > right:
            break
        new_a[right] = sorted_vals[idx]
        idx -= 1
        right -= 1
    return new_a


def _subgradient_descent(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Uses a simple momentum term to smooth the updates.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    step0 = 0.12                 # initial step size
    momentum = np.zeros_like(a)  # accumulated gradient
    beta = 0.9                    # momentum decay
    it = 0
    while time.time() < deadline:
        conv = _fft_convolve_self(cur_a)
        k_max = int(np.argmax(conv))

        # gradient of the active convolution entry
        idx_arr = k_max - np.arange(n)
        mask = (idx_arr >= 0) & (idx_arr < n)
        grad = np.zeros_like(cur_a)
        grad[mask] = 2.0 * cur_a[idx_arr[mask]]

        # diminishing step size and momentum update
        alpha = step0 / np.sqrt(it + 1.0)
        momentum = beta * momentum + (1.0 - beta) * grad
        new_a = cur_a - alpha * momentum

        # projection onto the simplex (non‑negative, ℓ₁ = 1)
        new_a = np.maximum(new_a, 0.0)
        s = np.sum(new_a)
        if s < 1e-12:                     # avoid division by zero
            new_a = np.full(n, 1.0 / n)
            s = 1.0
        new_a /= s

        new_val = _eval_fast(new_a)

        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it} new best = {best_val:.6f}")

        # accept only improving steps
        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val

        it += 1
    return best_a, best_val


def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Aggressive SA on permutations.
    Three move types:
        0 – swap two positions,
        1 – take element i and insert it at position j,
        2 – reverse a random contiguous segment.
    The temperature decays slowly to keep exploration alive.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.02, 1e-6)   # initialise temperature
    decay = 0.99995                # slow exponential cooling

    while time.time() < deadline:
        move = rng.integers(0, 3)   # 0=swap,1=insert,2=reverse
        new_a = a.copy()

        if move == 0:               # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif move == 1:             # insertion
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            val = new_a[i]
            if i < j:
                new_a[i:j] = new_a[i+1:j+1]
                new_a[j] = val
            else:
                new_a[j+1:i+1] = new_a[j:i]
                new_a[j] = val

        else:                       # reverse a segment
            i, j = sorted(rng.integers(0, n, size=2))
            if i == j:
                continue
            new_a[i:j+1] = new_a[i:j+1][::-1]

        new_val = _eval_fast(new_a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

    return best_seq, best_val


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    """
    Simple hill‑climbing that mixes pure swaps (20 %) with tiny mass
    transfers (80 %).  The transfers move at most 5 % of the source entry.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:          # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                           # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point required by the challenge.  Returns a Python list of
    non‑negative floats with the smallest evaluation value found within the
    allotted wall‑time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the final call to ``evaluate_sequence``
    deadline = start_time + budget_s - 1.5
    print(f"[init] total budget = {budget_s}s, internal deadline @ {deadline:.2f}")

    # --------------------------------------------------------------
    # Problem size – we keep the original 4000 (good trade‑off)
    # --------------------------------------------------------------
    n = 4000
    global _GLOBAL_N
    _GLOBAL_N = n

    # --------------------------------------------------------------
    # Initialise – reuse a previous construction if available
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= np.sum(a0)                # enforce ℓ₁ = 1

    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Smoothed surrogate L‑BFGS (multiple restarts, fine ε‑schedule)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-13), num=22)

    restart = 0
    # leave ~350 s for the later phases
    while time.time() < deadline - 350.0:
        restart += 1
        print(f"[restart {restart}]")
        # Initialise soft‑max parameters from the current best vector
        w = np.log(best_a + 1e-12) + 0.5 * np.random.randn(n)

        for eps in eps_schedule:
            if time.time() >= deadline - 330.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= np.sum(a)
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # --- re‑ordering -------------------------------------------------
        # try both deterministic heuristics and keep the better one
        a1 = _ends_large_reorder(best_a)
        a1 /= np.sum(a1)
        v1 = _eval_fast(a1)

        a2 = _alternating_ends_reorder(best_a)
        a2 /= np.sum(a2)
        v2 = _eval_fast(a2)

        if v1 < best_val:
            best_val = v1
            best_a = a1.copy()
            print(f"[reorder ends‑large] improved to {best_val:.6f}")

        if v2 < best_val:
            best_val = v2
            best_a = a2.copy()
            print(f"[reorder alt‑ends] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Projected sub‑gradient descent (longer, with momentum)
    # --------------------------------------------------------------
    if time.time() < deadline - 250.0:
        print("[subgrad] starting extended projected sub‑gradient descent")
        sub_deadline = deadline - 200.0
        best_a, best_val = _subgradient_descent(best_a, sub_deadline)

    # --------------------------------------------------------------
    # 3. Aggressive simulated‑annealing on the permutation
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("[SA] entering long simulated annealing")
        sa_deadline = deadline - 120.0
        best_a, best_val = _simulated_annealing(best_a, best_val, sa_deadline)

    # --------------------------------------------------------------
    # 4. Final cheap hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report final value using the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required by the framework
    return best_a.tolist()
