def full_convolution(a):
    n = a.shape[0]
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv
```

"""
Hybrid optimizer for the step‑function evaluation problem.

The algorithm proceeds in three stages within the 1000 s budget:

1. **Initialisation** – a rich library of parametric families (uniform,
   power‑law, exponential, U‑shapes, truncated blocks, random Dirichlet)
   is scanned on a modest size (≈2000).  The best shape is interpolated to
   the final length (default 9000) and normalised.

2. **Subgradient mirror descent** – we minimise the exact non‑smooth
   objective `max(conv(a,a))` on the simplex using exponentiated‑gradient
   updates with the true sub‑gradient (the reversed vector at the index of
   the current maximal convolution entry).  This quickly removes large
   irregularities.

3. **Deterministic peak‑equalising hill‑climb** – while time permits we
   repeatedly take the current maximal convolution entry `k_max`,
   identify a *source* index that contributes most to it and a *target*
   index that contributes least, and move the maximal amount of mass that
   can flatten the peak (computed analytically from the linearised effect
   on the convolution).  Stagnation triggers a mild random perturbation.

A short post‑processing step mixes in a tiny amount of the uniform vector,
which often reduces the maximal convolution further.

All updates maintain non‑negativity and unit sum, so the final vector can
be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) == 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric initialisation – a rich collection of families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Explore several parametric families on length `n`, return the best."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ----- deterministic families ------------------------------------
    # linear decreasing (left→right)
    try_candidate(np.arange(n, 0, -1, dtype=float))

    # linear U‑shape (large ends, small centre)
    idx = np.arange(n)
    try_candidate(np.maximum(idx + 1, n - idx))

    # quadratic decreasing (left side)
    try_candidate((np.arange(n, 0, -1, dtype=float) ** 2))

    # quadratic U‑shape
    quad_u = (np.arange(n) + 1) ** -2 + (np.arange(n, 0, -1) + 0) ** -2
    try_candidate(quad_u)

    # half‑wave sine (symmetric)
    try_candidate(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))

    # cosine with peaks at the ends
    try_candidate(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # power‑law symmetric (U‑shape) – sweep p
    for p in np.linspace(0.3, 3.0, 27):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_candidate(a)

    # exponential left‑hand side – sweep r
    for r in np.linspace(0.90, 0.9999, 30):
        a = r ** np.arange(n, dtype=float)
        try_candidate(a)

    # symmetric (U‑shaped) exponential
    for r in np.linspace(0.90, 0.9999, 30):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_candidate(a)

    # truncated exponential (mass on a prefix)
    for frac in np.linspace(0.05, 0.30, 6):
        K = max(1, int(frac * n))
        for r in np.linspace(0.90, 0.99, 5):
            a = np.zeros(n, dtype=float)
            a[:K] = r ** np.arange(K, dtype=float)
            try_candidate(a)

    # two‑block constant on both ends – sweep block size
    for frac in np.linspace(0.01, 0.20, 10):
        K = max(1, int(frac * n))
        a = np.zeros(n, dtype=float)
        a[:K] = 1.0 / (2 * K)
        a[-K:] = 1.0 / (2 * K)
        try_candidate(a)

    # random draws – a few quick Dirichlet samples
    rng = np.random.default_rng(123456)
    for _ in range(20):
        try_candidate(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient mirror‑descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_mirror_descent(a: np.ndarray,
                                start: float,
                                deadline: float,
                                best_a: np.ndarray,
                                best_val: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent using the true sub‑gradient of max(conv)."""
    n = a.shape[0]
    it = 0
    eta0 = 1.0    # initial step‑size scale
    while time.time() < deadline:
        # current convolution and index of the peak
        b = _full_convolution(a)
        k = int(np.argmax(b))

        # sub‑gradient of the peak entry:
        #   ∂c_k/∂a_i = 2 * a_{k-i}  (if the index exists)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]

        # step size schedule (1/√t)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad)
        # protect against collapse to zero
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        cur_val = 2.0 * n * _full_convolution(a).max()
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= 1.503:
                break
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic peak‑equalising hill‑climb (mass‑transfer with exact bound)
# ----------------------------------------------------------------------
def _equaliser_hillclimb(a: np.ndarray,
                         start: float,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly flatten the maximal convolution entry.
    In each iteration the exact amount of mass that can equalise the current
    peak with the second‑largest entry is transferred from a source index
    (large contribution) to a target index (tiny contribution).
    """
    n = a.shape[0]
    # initial convolution
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        b_max = b[k_max]

        # second‑largest entry (without allocating a copy)
        left_max = b[:k_max].max() if k_max > 0 else -np.inf
        right_max = b[k_max + 1:].max() if k_max + 1 < b.size else -np.inf
        second_max = left_max if left_max > right_max else right_max

        # ----- source index (largest contribution to the peak) -----
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k_max - idx]
        i = int(idx[np.argmax(contrib)])

        # ----- target index (tiny contribution to the peak) -----
        # valid j satisfy 0 ≤ j ≤ n‑1 and 0 ≤ k_max - j ≤ n‑1
        j_lo = max(0, k_max - (n - 1))
        j_hi = min(n - 1, k_max)
        j_idx = np.arange(j_lo, j_hi + 1)
        rev_contrib = np.full(n, np.inf)
        rev_contrib[j_idx] = a[k_max - j_idx]
        rev_contrib[i] = np.inf                     # exclude source
        j = int(np.argmin(rev_contrib))

        # Ensure diff > 0; otherwise fall back to a random target
        diff = rev_contrib[i] - rev_contrib[j]
        if diff <= 0.0:
            # pick a random different index
            j = i
            while j == i:
                j = np.random.randint(n)
            diff = rev_contrib[i] - rev_contrib[j]
            if diff <= 0.0:
                diff = 1e-12

        # ----- maximal transferable mass that can flatten the peak -----
        delta_max = (b_max - second_max) / (2.0 * diff) if diff > 0 else 0.0

        # schedule for the allowable fraction of the source mass
        progress = (time.time() - start) / (deadline - start)
        max_frac = 0.5 * (1.0 - progress) + 0.001 * progress
        delta_cap = max_frac * a[i]

        delta = min(delta_max, delta_cap)
        if delta <= 1e-15:
            # if the analytical bound is too small, take a modest random step
            delta = max_frac * a[i] * np.random.rand()

        # safety: never move more than the source contains
        delta = min(delta, a[i] * 0.999)

        # ----- apply the mass transfer (incremental convolution update) -----
        a_old = a.copy()
        b_new = b.copy()
        # linear part
        b_new[j:j + n] += 2.0 * delta * a_old
        b_new[i:i + n] -= 2.0 * delta * a_old
        # quadratic correction terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-15:
            # accept move
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Equal {it:7d}] new best = {best_val:.6f}")
                if best_val <= 1.503:
                    break
        else:
            no_improve += 1

        # stagnation handling – gentle perturbation around the best seen
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Equal {it:7d}] perturb after stagnation")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Small post‑processing sweep – mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a' = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a list of floats (Σ = 1) that (hopefully) minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a thin margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a compromise between granularity and speed.
    # --------------------------------------------------------------
    n = 9000

    # --------------------------------------------------------------
    # Initialise a – use a known construction if available,
    # otherwise fall back to a parametric scan on a smaller grid.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        # cheap scan on a smaller size then upscale
        n_small = 2000
        best_small, _ = _initial_candidate(n_small)
        # interpolate to the target length
        x_small = np.linspace(0.0, 1.0, n_small)
        x_up = np.linspace(0.0, 1.0, n)
        a0 = np.interp(x_up, x_small, best_small)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    print(f"[Init] objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – sub‑gradient mirror descent (≈20 % of the time budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * (deadline - start_time)
    best_a, best_val = _subgradient_mirror_descent(
        best_a,
        start=start_time,
        deadline=phase1_deadline,
        best_a=best_a,
        best_val=best_val,
    )

    # --------------------------------------------------------------
    # Stage 2 – deterministic equaliser hill‑climb (remaining time)
    # --------------------------------------------------------------
    best_a, best_val = _equaliser_hillclimb(
        best_a,
        start=phase1_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
    )

    # --------------------------------------------------------------
    # Final polishing – tiny uniform mixing
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # --------------------------------------------------------------
    # Final evaluation via the provided secure wrapper
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
