"""
Step‑function optimizer for `evaluate_sequence`.

The objective is
        f(a) = 2·n·max(conv(a,a)) / (sum a)^2 .
Because the function is scale‑invariant we keep Σa = 1 and minimise
        2·n·max(conv(a,a)).

The algorithm consists of three fast, complementary phases
(0/ A/ B/ C) that fit into the 1000 s budget:

* **Phase‑0 – quick families** – evaluate a handful of parametric
  families (uniform, powers, exponentials, U‑shapes, random Dirichlet)
  and keep the best one.

* **Phase‑A – Exponential‑gradient descent** – treat the max‑convolution
  entry as a (sub‑)gradient.  For the current maximal index k we have
  the sub‑gradient gᵢ = 2·a_{k‑i} (zero outside the support).  An
  exponentiated‑gradient (multiplicative) update
        a ← a·exp(‑η·g)
  preserves non‑negativity; after each update we renormalise to Σa = 1.
  The step size η decays from 1e‑1 → 1e‑5.

* **Phase‑B – deterministic greedy hill‑climb** – repeatedly move a
  small fraction of the mass from the index that contributes most to
  the current peak (maximal reverse entry) to the index that contributes
  the least.  The transferred amount decays from 50 % → 0.1 % of the
  source weight.  Only moves that strictly lower the objective are kept.
  After a long period without improvement a tiny random shake is added.

* **Phase‑C – final uniform mixing** – test convex combinations with a
  tiny amount of the uniform vector; this occasionally reduces the peak
  further.

All phases respect the time limit, keep the sequence non‑negative,
renormalise after each modification and maintain the best vector found.
The final candidate is returned as a plain list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return the exact value returned by `evaluate_sequence` for a normalized vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Simple parametric families – fast initial sweep
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)                # uniform
    best_val = _objective(best_a, n)

    def try_update(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(12345)

    # a few deterministic families
    idx = np.arange(n)
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # U‑shape linear
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # random Dirichlet draws (quick)
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _expgrad_phase(a0: np.ndarray, n: int,
                   start: float, deadline: float,
                   best_a: np.ndarray, best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent using the sub‑gradient
    of the maximal convolution entry.
    """
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # sub‑gradient for the active peak
        k = int(np.argmax(b))
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]                     # a_{k‑i}
        g = 2.0 * rev                               # sub‑gradient (ignoring the diagonal factor ½)

        # step‑size schedule (linear decay)
        progress = (now - start) / (deadline - start)
        eta = (1.0 - progress) * 0.12 + progress * 0.0005   # from ≈0.12 → 5e‑4

        # exponentiated update (keeps positivity)
        cur = cur * np.exp(-eta * g)

        # renormalise to Σa = 1
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # occasional jitter if stuck
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.008, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – deterministic greedy hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer: move a fraction of weight from the index that
    contributes most to the current peak to the index that contributes
    least.  The transferred fraction decays from 0.5 → 0.001.
    """
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        # evaluate current state
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # annealing of the transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the maximal convolution entry
        k = int(np.argmax(b))

        # reverse‑relevant slice of a (a_{k‑i})
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        # source = index with largest rev (biggest contribution)
        i = int(np.argmax(rev))

        # target = index with smallest rev (excluding the source)
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            # source depleted – try a random jitter
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        # amount to move (random fraction of the allowed max)
        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # apply move temporarily
        cur[i] -= delta
        cur[j] += delta

        # recompute objective (full convolution; cheap enough for modest loops)
        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:          # accept only improvements
            # keep the move
            cur_val = new_val
            # best update already handled at loop start
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – final tiny uniform mixing
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a′ = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
