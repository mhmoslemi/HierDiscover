"""
Improved optimizer for the “evaluate_sequence” objective.

The algorithm consists of three phases:

1. **Linear‑programming guided mass spreading** –  Starting from the best
   known sequence we maximise the total mass of a new candidate `b`
   under the *steady‑state* constraint
   `a * b ≤ M` (point‑wise) where `M` is the current maximal convolution.
   The LP is small because only the `K` largest lags are kept as constraints.
   After solving, `b` is renormalised and mixed with the current sequence.
   This step dramatically lowers the maximal convolution.

2. **Projected sub‑gradient descent** –  The exact nonsmooth objective
   `2·n·max(a*a)` is then polished using a sub‑gradient step with
   projection onto the simplex.

3. **Final reporting** –  The best sequence found is evaluated with the
   provided `evaluate_sequence` function and returned.

All work respects the supplied time budget (default 1000 s).  The code
contains no hidden state, uses only NumPy / SciPy and prints progress
messages.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may produce tiny negative values – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of evaluate_sequence (without clipping)."""
    return float(2.0 * n * np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Build the sparse matrix for the LP constraints a * b ≤ M
# ----------------------------------------------------------------------
def _build_constraint_matrix(a: np.ndarray, ks: np.ndarray) -> csr_matrix:
    """
    Returns a sparse matrix A (|ks| × n) such that
        (A @ b)[i] = (a * b)[ ks[i] ] .
    """
    n = a.shape[0]
    rows, cols, data = [], [], []
    for row_idx, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        rows.extend([row_idx] * i_range.size)
        cols.extend(j_range.tolist())
        data.extend(a[i_range].tolist())
    return csr_matrix((data, (rows, cols)), shape=(len(ks), n))


# ----------------------------------------------------------------------
# Project onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact objective) – polishing phase
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         best_val: float,
                         deadline: float,
                         rng: np.random.Generator,
                         stride: int = 1) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   2·n·max(a*a) .
    Returns the best vector observed.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    best_a = cur_a.copy()
    eta = 0.1                       # initial step size

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # all lags attaining the maximum belong to the active set
        active = np.flatnonzero(conv == max_b)
        # pick a random active lag
        k = int(rng.choice(active))

        # sub‑gradient of the active lag (see derivation in comment)
        grad = np.zeros_like(cur_a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # gradient step + projection onto simplex
        new_a = cur_a - eta * grad
        new_a = _project_simplex(new_a)

        new_val = _objective(new_a, n)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            eta = min(eta * 1.05, 0.5)
        else:
            eta = max(eta * 0.7, 1e-5)

    return best_a, best_val
