   iterations the sequence is optionally reordered by the anti‑symmetric
   heuristic.

4. **Final hill‑climbing (simulated‑annealing style)** – random swaps and
   mass‑transfer moves are applied; worsened solutions are occasionally
   accepted according to a simple temperature schedule, allowing the
   optimiser to escape shallow local minima.

All intermediate candidates are evaluated with ``evaluate_sequence``.
The best sequence found before the deadline is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # numerical noise may produce tiny negatives; clip them
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values on the left, smallest on the right (mirror)."""
    n = a.shape[0]
    s = np.sort(a)
    new = np.empty_like(a)
    half = n // 2
    new[:half] = s[n - 1: n - half - 1: -1]   # descending largest half
    new[-half:] = s[:half]                    # ascending smallest half
    if n % 2:
        new[half] = s[half]
    return new


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
_GLOBAL_EPS = 1.0
_GLOBAL_N = 1


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. the log‑softmax parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k−max)/ε)
    """
    # stable soft‑max
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)   # length 2·n‑1
    max_conv = conv.max()

    # smooth approximation of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # distribution over lag indices

    # convolution of a with reversed p   (conv_ap = a * p_rev)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the central n entries give the derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]   # derivative of L w.r.t a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w):
    """Wrapper required by scipy.optimize (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize(w_start: np.ndarray, deadline: float, n: int):
    """
    Perform an ε‑schedule L‑BFGS optimisation starting from soft‑max params w_start.
    Returns the best (a, value) pair observed.
    """
    best_a = None
    best_val = float('inf')
    w = w_start.copy()

    # ε values: from 1e‑2 down to 1e‑12 (geometric)
    eps_schedule = np.logspace(-2, -12, num=12)

    for eps in eps_schedule:
        if time.time() >= deadline - 15.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper, w,
                       method='L-BFGS-B', jac=True,
                       options={'maxiter': 400,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to probability simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # try anti‑symmetric reordering
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback safety
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())
    return best_a, best_val


# ----------------------------------------------------------------------
# Multi‑peak exponential mirror‑descent
# ----------------------------------------------------------------------
def _gradient_descent_multi_peaks(a_start: np.ndarray,
                                 deadline: float,
                                 top_k: int = 12,
                                 alpha0: float = 0.5,
                                 decay_rate: float = 0.985):
    """
    Exponential mirror‑descent that reduces the `top_k` largest convolution peaks.
    Returns the best (a, value) encountered.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[gd init] value = {best_val:.6f}")

    it = 0
    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # indices of the top_k peaks (sorted descending)
        if top_k < conv.size:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]
        else:
            idxs = np.arange(conv.size)

        # gradient contribution from the selected peaks
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += a[j_range]

        # step size (decays slowly)
        alpha = alpha0 * (decay_rate ** (it // 1500))

        # multiplicative mirror‑descent update
        a = a * np.exp(-alpha * g)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-14:                     # underflow guard
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        # logging every ~100 iterations
        if it % 100 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd iter {it}] new best = {best_val:.6f} (max conv {max_conv:.8f})")
            # occasional anti‑symmetric reordering
            if it % 800 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing refinement with simple simulated annealing
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       n: int):
    """
    Random local moves (swap or small mass transfer) with a temperature that
    slowly cools.  Only strictly better solutions are kept, but occasional
    uphill moves are accepted with probability exp(−Δ / T).
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    T0 = 1e-3
    decay = 0.9995

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time()
        T = T0 * (decay ** (elapsed))

        # decide move type
        if rng.random() < 0.15:                       # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                         # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] < 1e-14:
                continue
            delta = cur_a[i] * 0.03 * rng.random()    # up to 3 % of source mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        # acceptance rule
        if val < cur_val - 1e-12:
            cur_a, cur_val = new_a, val
            if val < best_val - 1e-12:
                best_val, best_a = val, new_a.copy()
                print(f"[refine] NEW GLOBAL BEST = {best_val:.6f}")
        else:
            # Metropolis acceptance
            prob = math.exp(-(val - cur_val) / (T + 1e-12))
            if rng.random() < prob:
                cur_a, cur_val = new_a, val

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) minimising
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep the well‑studied size 8000 (fast FFTs)
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Load historic construction if available, otherwise uniform start
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)

    # Normalise to simplex (sum = 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
        s = a.sum()
    a /= s

    # ------------------------------------------------------------------
    # Keep the best vector seen so far
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – a few random starts (to avoid bad local minima)
    # ------------------------------------------------------------------
    for tr in range(5):
        if time.time() >= deadline - 250:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val, best_a = val, a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate optimisation (L‑BFGS) with several restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 150 and best_val > 1.45:   # stop early if already great
        # log‑softmax parameters for the current best vector
        w_start = np.log(best_a + 1e-12)
        w_start += rng.normal(scale=0.15, size=n)

        a_opt, val_opt = _l_bfgs_optimize(w_start, deadline, n)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[L‑BFGS restart] best improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multi‑peak exponential gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 60:
        a_gd, val_gd = _gradient_descent_multi_peaks(best_a, deadline,
                                                     top_k=12,
                                                     alpha0=0.6,
                                                     decay_rate=0.983)
        if val_gd < best_val:
            best_val, best_a = val_gd, a_gd.copy()
            print(f"[gradient descent] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing / simulated annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation (safety)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = best_val

    return best_a.tolist()
