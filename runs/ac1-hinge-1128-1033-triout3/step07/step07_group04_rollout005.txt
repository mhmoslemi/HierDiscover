"""
Mirror‑descent optimiser for the step‑function convolution problem.

The idea:
* Work with a probability vector ``a`` (non‑negative, sum = 1).  
* The objective is 2·n·max_k (a*a)[k]; the factor 2·n is constant,
  so we minimise the maximal autocorrelation entry.
* For a given maximal index k* the gradient of the max entry is known
  in closed form (only the pairs (i, k*‑i) matter).  Using the
  exponentiated‑gradient (mirror‑descent with the KL‑divergence) we
  repeatedly move probability mass away from indices that contribute the
  most to the current peak and renormalise.
* A simple back‑tracking line‑search on the step‑size η is employed.
* Several random seeds (uniform, geometric, power‑law, Dirichlet) are
  used; each seed is run for a limited time budget with the mirror‑descent
  loop.  The best sequence found is finally polished by a tiny
  simulated‑annealing hill‑climber.

The optimiser works directly with the fast FFT‑based evaluator and
returns a Python list suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution / evaluation (identical to the one used in the
# previous optimiser – kept here to avoid repeated calls to the
# security‑checked ``evaluate_sequence`` during the search).
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (size = power of two)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Negligible negatives caused by rounding are clamped to zero.
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Very fast replica of the supplied ``evaluate_sequence`` for a
    non‑negative vector (sum is assumed > 0)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Gradient of the (non‑smooth) maximal autocorrelation entry.
# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient ∇_a ( (a*a)[k] ) for the index k where the convolution attains its
    maximum.  The formula follows from
        (a*a)[k] = Σ_i a_i a_{k-i}
    and hence ∂/∂a_j = a_{k-j} + a_{j-k} (terms outside the array are zero).
    """
    n = a.shape[0]
    idx = k - np.arange(n)               # gives k-i
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(a)
    grad[mask] = a[idx[mask]]

    idx2 = np.arange(n) - k               # gives i-k
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] += a[idx2[mask2]]
    return grad


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (exponentiated‑gradient on the max‑convolution).
# ----------------------------------------------------------------------
def _mirror_descent_optimize(
    a0: np.ndarray,
    deadline: float,
    *,
    eta_init: float = 0.5,
    eta_max: float = 2.0,
    eta_min: float = 1e-6,
    max_iter: int = 200_000,
    jitter_prob: float = 0.005,
    jitter_scale: float = 0.005,
) -> tuple[np.ndarray, float]:
    """
    Runs a mirror‑descent loop starting from ``a0`` (already normalised).
    Returns the best vector seen and its objective value.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-15                     # avoids log(0)
    best_a = a.copy()
    best_val = _eval_fast(best_a)

    eta = eta_init
    it = 0

    while time.time() < deadline and it < max_iter:
        # current convolution, peak index and value
        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # gradient of the *current* maximal entry
        g = _grad_max_conv(a, k_max)

        # exponentiated‑gradient step (mirror step w.r.t. KL)
        w = np.log(a + eps) - eta * g
        a_new = np.exp(w)
        a_new /= a_new.sum()

        # occasional random jitter – helps escape shallow plateaus
        if rng.random() < jitter_prob:
            w += jitter_scale * rng.standard_normal(n)
            a_new = np.exp(w)
            a_new /= a_new.sum()

        # evaluate proposal
        val_new = _eval_fast(a_new)

        # accept if we improve; otherwise shrink step‑size
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            # increase step size a little (but stay bounded)
            eta = min(eta * 1.1, eta_max)
        else:
            # rejection – shrink η (guarantees convergence of the sub‑gradient method)
            eta = max(eta * 0.5, eta_min)

        # safety guard – if η became tiny we inject a small random perturbation
        if eta <= eta_min:
            # random re‑initialisation of the weights (keeps the current mass distribution)
            w = np.log(a + eps)
            w += 0.01 * (rng.random(n) - 0.5)
            a = np.exp(w)
            a /= a.sum()
            eta = eta_init

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Lightweight simulated annealing for a final polish (swap / transfer).
# ----------------------------------------------------------------------
def _final_hill_climb(
    a0: np.ndarray,
    deadline: float,
    *,
    temperature: float = 1e-4,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Very cheap SA: swaps and small mass transfers, always accepting
    downhill moves and occasionally uphill moves according to a fixed
    temperature schedule.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    cur_val = _eval_fast(a)
    best_a, best_val = a.copy(), cur_val
    T = temperature
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.4:               # pure swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                new_a = a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]
            else:
                continue
        else:                                 # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * 0.04 * rng.random()
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / max(T, 1e-12)):
            a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
        T = max(T * decay, 1e-12)

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper families for seeding (identical to the previous code).
# ----------------------------------------------------------------------
def _geometric_sequence(q: float, n: int) -> np.ndarray:
    """a_i ∝ q**i, normalised to sum = 1 (0 < q < 1)."""
    if not (0.0 < q < 1.0):
        return np.full(n, 1.0 / n)
    denom = (1.0 - q ** n) / (1.0 - q)
    a = q ** np.arange(n, dtype=float) / denom
    return a


def _power_sequence(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)**p, normalised to sum = 1 (p < 0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s
