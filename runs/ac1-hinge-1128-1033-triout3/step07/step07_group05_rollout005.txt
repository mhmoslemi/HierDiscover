    start_time = time.time()

"""
Hybrid search for a low‑value step‑function.

The algorithm works in four logical phases (all respecting the 1000 s wall‑time):
1. **Exploratory seeding** – generate a few analytically‑motivated shapes.
   We use U‑shaped Beta‑type densities (α < 1) which concentrate mass at the
   two ends of the interval.  After normalisation the shapes are evaluated
   with a fast FFT based evaluator.
2. **Projected sub‑gradient descent** – starting from the best seeded shape,
   repeatedly move mass away from the index that currently attains the
   convolution peak.  After each gradient step we project onto the probability
   simplex (non‑negative, ℓ₁ = 1).  This is a cheap way to lower the maximal
   autocorrelation.
3. **Simulated‑annealing on permutations** – once the values are reasonably
   balanced we only permute them (swap moves).  Swaps are accepted according
   to a simple temperature schedule; the temperature decays slowly so the
   search can escape shallow local minima.
4. **Fine hill‑climbing** – final polish using a mixture of swaps and tiny
   mass‑transfer moves.

All heavy lifting (convolution, objective) uses an FFT routine that mimics
the official ``evaluate_sequence`` implementation but avoids its safety
checks for speed.  The final sequence is returned as a plain Python list;
the official evaluator is called once at the end to report the true score.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (exactly the same arithmetic as evaluate_sequence)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Numerical noise can produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence (same result for valid inputs).
    a must be a 1‑D numpy array of non‑negative floats, sum(a) > 0.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:                 # same safeguard as the official evaluator
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic seed shapes – Beta(α,α) with α<1 (U‑shaped densities)
# ----------------------------------------------------------------------
def _beta_u_shape(alpha: float, n: int) -> np.ndarray:
    """
    Returns a normalised U‑shaped density on {0,…,n‑1}.
    For α=1 we obtain the uniform distribution.
    """
    if alpha == 1.0:
        a = np.full(n, 1.0 / n)
        return a

    # points in the open interval (0,1) to avoid the singularities at 0 and 1
    x = (np.arange(n) + 0.5) / n          # shape (n,)
    # Beta(α,α) density up to a constant factor
    with np.errstate(divide='ignore', invalid='ignore'):
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # replace NaNs (which can appear if alpha<1 with x=0 or 1) by zero
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a_sum = a.sum()
    if a_sum == 0.0:                     # fallback – should never happen
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
# Simple deterministic reorder – biggest values at the two ends
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values at the extremes, the smallest near the centre."""
    sorted_vals = np.sort(a)          # ascending
    n = a.shape[0]
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Reduce max(a*a) while keeping a on the probability simplex.
    Simple sub‑gradient method with a 1/√t step‑size schedule.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    step0 = 0.12                     # empirical initial step length
    it = 0
    while time.time() < deadline:
        # 1) compute convolution and the location of its maximum
        conv = _fft_convolve_self(cur_a)
        k_max = int(np.argmax(conv))

        # 2) gradient of the *active* max entry (see analysis)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(cur_a)
        # for every i that participates in the peak, ∂b_k/∂a_i = 2 a_{k-i}
        grad[mask] = 2.0 * cur_a[idx[mask]]

        # 3) gradient step and projection onto ℓ₁ = 1 simplex
        alpha = step0 / np.sqrt(it + 1.0)
        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:                # avoid degenerate zero vector
            new_a = np.full(n, 1.0 / n)
            s = 1.0
        new_a /= s                    # ℓ₁‑normalisation (simplex projection)

        # 4) evaluate
        new_val = _eval_fast(new_a)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it:6d} new best = {best_val:.6f}")

        # 5) accept only improving steps (guaranteed descent)
        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap only)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """Fast SA that swaps two positions; works directly on the fast evaluator."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # Initialise temperature as a small fraction of the current value
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99994

    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        a[i], a[j] = a[j], a[i]        # propose swap
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            a[i], a[j] = a[j], a[i]    # revert

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Final fine‑grained hill‑climbing (swap + tiny mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Neighbourhood search mixing swaps and small mass moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = cur_val

    while time.time() < deadline:
        if rng.random() < 0.2:              # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = best_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] == 0.0:
                continue
            delta = best_a[i] * 0.04 * rng.random()
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta

        cand /= cand.sum()                  # keep on simplex
        val = _eval_fast(cand)
        if val < best_val - 1e-12:
            best_a = cand
            best_val = val
            print(f"[refine] new best = {best_val:.6f}")

    return best_a, best_val
