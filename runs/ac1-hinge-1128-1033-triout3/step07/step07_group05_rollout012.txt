"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence.  The algorithm blends three ideas:

1. **Smooth surrogate optimisation** – we minimise a
   log‑sum‑exp smooth approximation of the max‑convolution using a
   custom Adam optimiser.  The surrogate remains an upper bound on the
   true objective, but its gradients are cheap (FFT based) and
   reliable.

2. **Diverse re‑orderings + uniform mixing** – after each optimisation
   run we try several deterministic permutations of the multiset of
   heights (ascending, descending and the original “large‑ends”
   interleaving).  A small convex blend with the uniform distribution
   is also tried; any permutation that yields a lower fast‑evaluation
   value is kept.

3. **Final hill‑climbing / simulated‑annealing** – a short SA on swaps
   (temperature tuned to the current objective) followed by a
   mass‑transfer hill‑climber refines the best vector found.

All stages respect a 1000 s wall‑time budget and constantly track the
best sequence.  The fast evaluator uses FFT convolution to avoid the
heavy safety‑checks of ``evaluate_sequence``.  The returned list is a
plain Python list ready for the challenge framework.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities (exactly match evaluate_sequence's behaviour)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # remove tiny negatives from round‑off
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast copy of evaluate_sequence (no safety checks)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smoothed surrogate: L = 2·n·max + 2·n·ε·log∑exp((c−max)/ε)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (loss, grad_w) for the smooth surrogate.
    w are unconstrained log‑weights; a = softmax(w) ≥0, Σa=1.
    """
    # soft‑max (guarantees positivity & ℓ₁=1)
    w_shift = w - np.max(w)                 # numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # convolution a*a
    conv = _fft_convolve_self(a)            # length 2·n‑1
    max_conv = conv.max()

    # smooth approximation of max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = np.logaddexp.reduce(shifted)   # log∑exp(...)
    loss = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / np.sum(weights)           # probability distribution over k

    # ∂L/∂a_i = 4·n·∑_k p_k·a_{k-i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)           # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]          # central n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return loss, grad_w


# ----------------------------------------------------------------------
# Simple deterministic re‑orderings
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Original interleaved large‑small ordering."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    l, r = 0, n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[r]
        r -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[l]
        l += 1
        i += 1
    return new_a


def _ascending_reorder(a: np.ndarray) -> np.ndarray:
    """Monotonically increasing (small → large)."""
    return np.sort(a)


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Monotonically decreasing (large → small)."""
    return np.sort(a)[::-1]

                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad * grad)
                m_hat = m / (1 - beta1 ** step)
                v_hat = v / (1 - beta2 ** step)
                lr = base_lr / np.sqrt(step)          # slowly decaying stepsize
                w = w - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                if time.time() >= stage1_deadline:
                    break

            # evaluate the current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= np.sum(a)
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[restart {restart} eps {eps:.1e}] new best = {best_val:.6f}")

    print(f"[stage1] best value after surrogate optimisation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deterministic re‑orderings + mixing with uniform
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 30:
        reorder_fns = [
            (_ascending_reorder, "ascending"),
            (_descending_reorder, "descending"),
            (_ends_large_reorder, "interleaved")
        ]
        for fn, name in reorder_fns:
            a_ord = fn(best_a)
            a_ord = a_ord / np.sum(a_ord)
            val = _eval_fast(a_ord)
            if val < best_val:
                best_val = val
                best_a = a_ord.copy()
                print(f"[reorder {name}] improved to {best_val:.6f}")

        # small convex blend with the uniform distribution can smooth spikes
        uniform = np.full(n, 1.0 / n)
        for alpha in (0.05, 0.10, 0.15):
            mixed = (1 - alpha) * best_a + alpha * uniform
            mixed /= np.sum(mixed)
            val = _eval_fast(mixed)
            if val < best_val:
                best_val = val
                best_a = mixed.copy()
                print(f"[mix α={alpha:.2f}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated annealing on permutations + hill‑climbing
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5:
        cur_seq = best_a.copy()
        cur_val = best_val
        rng = np.random.default_rng()

        # simulated annealing (swap moves only)
        T = max(cur_val * 0.02, 1e-6)
        decay = 0.99995
        while time.time() < deadline - 2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]      # propose swap
            new_val = _eval_fast(cur_seq)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = cur_seq.copy()
                    print(f"[SA] new best = {best_val:.6f}")
            else:
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]  # revert
            T = max(T * decay, 1e-12)

        # final hill‑climbing (swap + mass‑transfer moves)
        while time.time() < deadline:
            if rng.random() < 0.3:           # mass‑transfer move
                i, j = rng.integers(0, n, size=2)
                if i == j or cur_seq[i] == 0.0:
                    continue
                delta = cur_seq[i] * 0.05 * rng.random()
                new_seq = cur_seq.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
            else:                             # pure swap
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                new_seq = cur_seq.copy()
                new_seq[i], new_seq[j] = new_seq[j], new_seq[i]

            val = _eval_fast(new_seq)
            if val < cur_val - 1e-12:
                cur_seq = new_seq
                cur_val = val
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = cur_seq.copy()
                    print(f"[hill] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (safety‑checked)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] official evaluation = {final_score:.6f}")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    # Return a plain Python list as required by the challenge framework
    return best_a.tolist()
