    n = len(v)

    cssv = np.cumsum(u) - 1

    cond = u - cssv / ind > 0

"""
Optimise a non‑negative step function (a list of heights) for the metric

    O = 2·n·max(conv(seq, seq)) / (sum(seq))² .

The algorithm mixes three ideas
  * a cheap grid search over Beta‑shapes and a handful of Dirichlet draws, 
    keeping the best few as starting points,
  * a monotone entropy‑mirror‑descent optimiser with a simple back‑tracking
    line‑search on the exact max‑entry of the self‑convolution,
  * a final polishing phase that spends the remaining budget on a long
    mirror‑descent run starting from the best candidate found.

The optimiser works directly with the true max‑convolution (no smooth
approximation) and enforces symmetry of the sequence, which is
theoretically optimal for the self‑convolution problem.  All updates stay
on the probability simplex, so scaling does not affect the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (full length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Return the evaluation objective  O = 2·n·max(conv(a,a)).
    The vector a is assumed to be non‑negative and already normalised
    to sum to 1 (any other scaling yields the same O).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑entry of conv(a,a) at the index k.
    For each i such that 0 ≤ i < n and 0 ≤ k‑i < n we have
          ∂/∂a_i  conv[k] = 2·a_{k‑i}.
    """
    n = a.shape[0]
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    g = np.zeros_like(a)
    if lo <= hi:
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] = 2.0 * a[k - idx]
    return g


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce v_i = v_{n-1-i} by averaging with its reverse."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (monotone, back‑tracking line‑search)
# ----------------------------------------------------------------------
def _mirror_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Run entropy mirror descent on the exact max‑convolution.
    Stops when `time.time()` reaches *deadline*.
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    # ensure we are on the simplex and symmetric
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    a = _symmetrize(a)

    best_a = a.copy()
    best_obj = _objective(a)

    eta = 0.5          # initial step size
    it = 0

    while time.time() < deadline - 0.3:
        # current convolution and the index of the maximal entry
        conv = _full_convolution(a)
        max_conv = conv.max()
        k = int(conv.argmax())
        cur_obj = 2.0 * n * max_conv

        # sub‑gradient of the max‑entry
        g = _subgradient_max(a, k)

        # try to improve using back‑tracking on η
        improved = False
        eta_try = eta
        while eta_try > 1e-8:
            # entropic mirror‑step
            a_new = a * np.exp(-eta_try * g)
            s_new = a_new.sum()
            if s_new > 0.0:
                a_new /= s_new
            else:
                a_new = np.full(n, 1.0 / n)
            a_new = _symmetrize(a_new)

            obj_new = _objective(a_new)
            if obj_new < best_obj - 1e-13:

                best_a = a_new.copy()
                best_obj = obj_new
                eta = eta_try * 1.02      # gently increase for next iteration
                improved = True
                break
            else:
                eta_try *= 0.5            # shrink and retry

        if not improved:

            eta *= 0.5
            if eta < 1e-9:

                if s > 0.0:

                eta = 0.5

    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.ndim == 1 and old.size > 0:
            old_len = old.shape[0]
            x_new = np.linspace(0.0, 1.0, n)
            x_old = np.linspace(0.0, 1.0, old_len)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            ss = a.sum()
            if ss > 0.0:
                a /= ss
            else:
                a = np.full(n, 1.0 / n)
            init_cands.append((_objective(a), a))
            print("[init] added interpolated previous solution")
    except Exception:
        pass

    # 2) the uniform distribution
    a_uni = np.full(n, 1.0 / n)
    init_cands.append((_objective(a_uni), a_uni))

    # 3) a dense grid of Beta shapes (including U‑shaped)
    alphas = np.concatenate([
        np.linspace(0.2, 0.9, 8),      # < 1   → U‑shaped
        np.linspace(1.0, 5.0, 8)       # ≥ 1   → bell‑shaped / flat
    ])
    idx = np.arange(1, n + 1, dtype=np.float64) / n   # values in (0,1]
    for a_par in alphas:
        for b_par in alphas:
            a = (idx ** (a_par - 1.0)) * ((1.0 - idx) ** (b_par - 1.0))
            ss = a.sum()
            if ss > 0.0:
                a /= ss
            else:
                a = np.full(n, 1.0 / n)
            init_cands.append((_objective(a), a))

    # 4) random Dirichlet draws with a few concentration parameters
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(3):
            a = np.random.dirichlet(np.full(n, conc))
            init_cands.append((_objective(a), a))

    # Keep only the 30 best initial vectors (to stay within time budget)
    init_cands.sort(key=lambda tup: tup[0])
    init_cands = init_cands[:30]
    print(f"[init] selected {len(init_cands)} promising starts (best O≈{init_cands[0][0]:.6f})")

    # ------------------------------------------------------------------
    # Run mirror descent on the selected starts, splitting the remaining time
    # ------------------------------------------------------------------
    best_overall_obj = np.inf
    best_overall_a = None

    # allocate roughly 80 % of the remaining time to the optimisation loops,
    # keep the last 20 % for a final polishing run.
    time_for_runs = (deadline - time.time()) * 0.80

    per_start = time_for_runs / max(1, len(init_cands))
    for idx_start, (obj0, a0) in enumerate(init_cands):
        now = time.time()
        if now >= deadline - 0.2:
            break
        time_limit = now + min(per_start, deadline - now - 0.2)
        print(f"[run] start #{idx_start+1}/{len(init_cands)} (initial O={obj0:.6f})")
        a_opt, obj_opt = _mirror_descent(a0, time_limit)
        print(f"[run] finished start #{idx_start+1} → best O={obj_opt:.6f}")

        if obj_opt < best_overall_obj:
            best_overall_obj = obj_opt
            best_overall_a = a_opt.copy()

    # ------------------------------------------------------------------
    # Final polishing on the best candidate (use up the remaining budget)
    # ------------------------------------------------------------------
    if best_overall_a is not None:
        time_left = deadline - time.time()
        if time_left > 0.5:
            print(f"[polish] final polishing with {time_left:.1f}s left")
            best_overall_a, best_overall_obj = _mirror_descent(best_overall_a, deadline)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_overall_a.tolist() if best_overall_a is not None else a_uni.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence gave {final_score:.6f}  (approx O={best_overall_obj:.6f})")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
