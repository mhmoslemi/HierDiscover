   optimised in the log‑domain (`z = log a`).  The surrogate objective
   is the weighted average of the convolution coefficients with a
   temperature α that grows from 1 to ~2000, making the landscape
   gradually approach the true max.  Adam (β₁=0.9, β₂=0.999) is used
   with a linearly annealed learning‑rate (0.20 → 0.005).  Stagnation

1000 s deadline, finally returning the best sequence found.

import time

# FFT based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two (different) vectors (full)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Mirror‑descent polishing (exact max of the self‑convolution)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 3000, eta_start: float = 0.6) -> np.ndarray:
    """Entropy‑mirror descent on the exact max‑entry of a*a."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of max‑entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]                     # a_j appears twice except when i=j
        eta = eta_start * (1.0 - t / steps)                # linear decay
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)
    return a

    if best_a is not None:
        time_left = deadline - time.time()
        # decide the number of mirror steps (roughly 30 µs per step on typical HW)
        steps = max(8000, int(min(50000, time_left * 17000)))
        print(f"[polish] mirror descent start with {steps} steps")
        a_refined = _max_refine(best_a, steps=steps, eta_start=0.6)
        conv_ref = _full_convolution(a_refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = a_refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation (the function we are trying to minimise)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else init_vectors[0].tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx O={best_obj:.6f})")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
