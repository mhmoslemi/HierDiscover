"""
Propose a step‐function that minimises ``evaluate_sequence``.

Approach
--------
1. **Family search** – generate candidates from several parametric families
   (monotone power, exponential, Beta) on a fine grid and keep the best
   by the cheap surrogate 2·n·max(conv) (the sum is normalised to 1).

2. **Adam‑style refinement** – starting from the best candidate,
   repeatedly
      * compute the full convolution,
      * obtain a sub‑gradient of the max‑convolution,
      * perform an Adam update,
      * project back onto the probability simplex.
   The schedule is time‑aware, allowing many thousands of iterations
   within the 1000 s budget.

3. **Escapes** – occasional jitter (tiny random noise) prevents stagnation
   in flat regions.

The method respects all constraints (non‑negative, normalised) and
prints progress.  The final sequence is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact surrogate objective 2·n·max(conv(a,a)) (assumes Σa = 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_shape(p: float, n: int) -> np.ndarray | None:
    """a_i ∝ (i+1)^p, i=0,…,n‑1  (p>0). Normalised to sum = 1."""
    if p <= 0:
        return None
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, p)
    total = a.sum()
    if total == 0.0 or not np.isfinite(total):
        return None
    a /= total
    return a


def _exp_shape(r: float, n: int) -> np.ndarray | None:
    """a_i ∝ r^i, r>1. Normalised to sum = 1."""
    if r <= 1.0:
        return None
    i = np.arange(n, dtype=np.float64)
    # use logarithms to avoid overflow for large r
    logs = i * np.log(r)
    # shift to avoid extremely large exponentials
    max_log = np.max(logs)
    a = np.exp(logs - max_log)
    total = a.sum()
    if total == 0.0 or not np.isfinite(total):
        return None
    a /= total
    return a


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Beta‑type shape a_i ∝ i^{p‑1} (n‑i+1)^{q‑1}, normalised."""
    if p <= 0 or q <= 0:
        return None
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     max_active: int = 5,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Uses the most active lags (those attaining or near the maximum).
    """
    n = a.shape[0]
    max_val = conv.max()
    # select up to max_active indices with the largest values
    if max_active >= conv.size:
        active = np.arange(conv.size)
    else:
        active = np.argpartition(conv, -max_active)[-max_active:]
        mask = conv[active] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            active = np.array([int(np.argmax(conv))])
        else:
            active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad

# Main entry point
# ----------------------------------------------------------------------

    for _ in range(rnd_trials):

            q = rng.uniform(0.5, 6.0)

            tag = f"rnd_beta(p={p:.3f},q={q:.3f})"
        if cand is not None:

        if time.time() >= deadline - 250.0:

    # 3️⃣  Adam‑style projected gradient descent from the best candidate
    # ------------------------------------------------------------------
    if best_seq is None:

        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

        print("[fallback] using uniform start")

    a = best_seq.copy()
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 1

            a = np.maximum(a + jitter, 0.0)

            # reset moments to avoid stale direction
            m.fill(0.0)
            v.fill(0.0)

        t_iter += 1

    except Exception as exc:   # pragma: no cover
        print(f"[final] evaluate_sequence raised: {exc}")

    return best_seq.tolist()
