"""
Propose a step‑function (non‑negative list) that minimises

        evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))².

The optimisation is performed directly on the exact L∞‑objective
using exponentiated‑gradient (entropy‑mirror) descent.  Several
initialisations (previous solution, Beta shapes, random Dirichlet)
are tried; the best vector is kept and finally polished with a long
mirror‑descent run.  All updates stay on the probability simplex,
so the scale of the sequence is irrelevant (evaluate_sequence
renormalises internally).  The routine respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (zero‑padded, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2*len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max entry of the convolution
# ----------------------------------------------------------------------
def _mirror_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One mirror‑descent step for the max‑convolution objective.
    The step size ``eta`` is a positive scalar.
    Returns the updated (still non‑negative, not yet normalised) vector.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k = int(conv.argmax())                     # index of the max entry
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)

    # sub‑gradient of the max entry w.r.t. a
    g = np.zeros_like(a)
    if lo <= hi:
        idx = np.arange(lo, hi + 1, dtype=int)
        # each contributing pair (i, k-i) adds 2*a_{k-i} to ∂/∂a_i
        g[idx] = 2.0 * a[k - idx]

    # exponentiated (entropy‑mirror) update
    a_new = a * np.exp(-eta * g)
    return a_new


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.sum() == 1.0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]                     # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # renormalise numerically (should already sum to 1)
    s = w.sum()
    return w / s if s > 0 else np.full_like(w, 1.0 / len(w))

# Main entry point expected by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The algorithm runs several mirror‑descent restarts with different
    initialisations and returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    # a tiny safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8000 worked well in previous experiments.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Helper to normalise a vector to a probability distribution.
    # ------------------------------------------------------------------
    def _normalise(v):
        s = v.sum()
        return v / s if s > 0 else np.full_like(v, 1.0 / len(v))

    # ------------------------------------------------------------------
    # Build a collection of initial vectors.
    # ------------------------------------------------------------------
    init_vectors = []

    # 1) previous solution (if it exists and can be interpolated)
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        try:
            old = np.asarray(old_seq, dtype=float)
            if old.shape[0] != n:
                # linear interpolation onto the new grid
                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                old = np.interp(x_new, x_old, old)
            old = np.maximum(old, 0.0)
            old = _normalise(old)
            init_vectors.append(("prev", old))
            print("[init] interpolated previous solution")
        except Exception as exc:
            print("[init] failed to use previous solution:", exc)

    # 2) a smooth centrally‑peaked Beta(2,2) distribution
    i = np.arange(1, n + 1, dtype=np.float64)
    beta22 = (i * (n - i + 1))        # Beta(p=2,q=2) ∝ i·(n‑i+1)
    beta22 = np.maximum(beta22, 0.0)
    beta22 = _normalise(beta22)
    init_vectors.append(("beta22", beta22))

    # 3) a U‑shaped Beta(0.5,0.5) distribution (heavy edges)
    p = q = 0.5
    beta005 = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    beta005 = np.maximum(beta005, 0.0)
    beta005 = _normalise(beta005)
    init_vectors.append(("beta005", beta005))

    # 4‑6) three random Dirichlet starts
    for k in range(3):
        rnd = np.random.rand(n)
        rnd = np.maximum(rnd, 0.0)
        rnd = _normalise(rnd)
        init_vectors.append((f"rand{k}", rnd))

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_a = None
    best_obj = np.inf
    total_steps_per_restart = 20000            # adjustable if time permits

    # ------------------------------------------------------------------
    # Run a mirror‑descent optimisation from every start.
    # ------------------------------------------------------------------
    for name, a0 in init_vectors:
        if time.time() >= deadline:
            break
        a = a0.copy()
        # evaluate the starting point
        conv = _full_convolution(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[init] {name:7s} – objective {obj:.6f}")

        stagn_counter = 0
        last_obj = obj

        for step in range(total_steps_per_restart):
            if time.time() >= deadline:
                break

            # --- mirror‑descent step ---------------------------------
            # schedule a linearly decaying step size
            frac = step / total_steps_per_restart
            eta = 0.5 * (1.0 - frac) + 0.001 * frac   # 0.5 → 0.001

            a = _mirror_step(a, eta)
            a = _project_to_simplex(a)               # keep ∑a = 1

            # --- objective & bookkeeping ----------------------------
            conv = _full_convolution(a)
            obj = 2.0 * n * conv.max()

            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                stagn_counter = 0
                print(f"[impr] {name:7s} – new best O={obj:.6f}  step={step}")
            else:
                stagn_counter += 1

            # --- escape from stagnation ----------------------------
            if stagn_counter >= 2000:
                # small random jitter (additive, then re‑normalise)
                perturb = np.random.normal(scale=0.005, size=n)
                a = np.maximum(a + perturb, 0.0)
                a = _normalise(a)
                stagn_counter = 0
                print(f"[jitt] {name:7s} – jitter after {step} steps")

        # end of this restart
        print(f"[done] restart {name} – best O seen {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the overall best vector.
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback – uniform distribution
        best_a = np.full(n, 1.0 / n)
        print("[fallback] using uniform start")

    polish_steps = 50000                         # longer refinement
    for t in range(polish_steps):
        if time.time() >= deadline:
            break
        # step size decays to a very small value
        frac = t / polish_steps
        eta = 0.3 * (1.0 - frac) + 0.001 * frac

        best_a = _mirror_step(best_a, eta)
        best_a = _project_to_simplex(best_a)

    # ------------------------------------------------------------------
    # Return the result as plain Python list.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(internal O≈{best_obj:.6f})")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
