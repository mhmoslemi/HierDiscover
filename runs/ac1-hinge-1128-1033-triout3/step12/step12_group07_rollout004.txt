"""
Propose a step‑function (non‑negative sequence) that minimizes
the evaluation function 2·n·max(conv(a,a))/sum(a)².

Algorithm
---------
1. **Small‑scale LP refinement**
   * Work with a reduced size `n_small` (≈1500) to keep the linear
     program tractable.
   * Starting from a random Beta‑shaped distribution we solve
     ``max  Σ g_i``  s.t. ``f * g ≤ M`` and ``g ≥ 0`` where
     ``M = max(f * f)``.
   * The LP solution always yields ``Σ g_i > 1`` (unless ``f`` is already
     optimal), so after scaling ``g`` back to a probability vector we can
     convex‑combine it with ``f`` and strictly decrease ``M``.
   * The process is repeated (a few iterations) or stopped when no
     improvement is possible or the time budget is exhausted.

2. **Upsampling**
   * The best small‑scale solution is interpolated to the target length
     `n_big` (=8000).  Normalisation guarantees a valid input for the
     evaluator.

3. **Large‑scale polishing**
   * A short Adam optimisation on a smooth log‑sum‑exp approximation of
     the max‑convolution.
   * Followed by an aggressive hill‑climber (source→target mass moves)
   * Finally a few passes of exact sub‑gradient and exponentiated‑gradient
     polishing.

All steps respect the constraints of ``evaluate_sequence`` (non‑negative
entries, any normalisation) and run within the 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # LP solver (HiGHS backend)


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p (channel‑wise derivative)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    seg = conv_ab[n - 1:2 * n - 1]                # exactly n entries
    grad_a = 4.0 * n * seg[::-1]                 # factor 2·n already included
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families (used for the tiny‑scale initial population)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-7.0, 2.0)
    logq = rng.uniform(-7.0, 2.0)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Small‑scale LP improvement (core of the new idea)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray,
                deadline: float,
                start_time: float,
                max_iters: int = 12) -> np.ndarray:
    """
    Repeatedly solves the linear program
        maximise Σ g_i     s.t.   f * g ≤ M ,   g ≥ 0
    where M = max(f * f).
    The solution g is scaled back to Σ g = 1 and mixed with f.
    The procedure stops when no improvement is possible or the time
    budget is exhausted.
    """
    n = f.shape[0]

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # current worst convolution entry
        M = _full_convolution(f).max()

        # Build the Toeplitz matrix representing the convolution constraints
        # A @ g <= M  (A.shape = (2*n-1, n))
        m_rows = 2 * n - 1
        A = np.zeros((m_rows, n), dtype=np.float64)
        # each column i contributes f[i] on rows i … i+n-1
        for i in range(n):
            A[i:i + n, i] = f[i]

        # Linear program: minimise -sum(g)  →  maximise sum(g)
        c = -np.ones(n, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=np.full(m_rows, M, dtype=np.float64),
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': max(0.0, deadline - now)}
        )
        if not res.success:
            # LP failed – fall back to the current f
            break

        g_raw = res.x
        sum_g = g_raw.sum()

        # If the LP cannot produce a larger total mass, we are stuck
        if sum_g <= 1.0 + 1e-12:
            break

        # Scale back to a probability vector (Σg = 1)
        scale = 1.0 / sum_g
        g = g_raw * scale

        # Convex combination – try a relatively aggressive mix
        mixed = False
        for t in (0.5, 0.2, 0.1):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            new_M = _full_convolution(f_new).max()
            if new_M < M - 1e-15:
                f = f_new
                mixed = True
                # print progress
                print(f"[LP] iter {it+1}, t={t:.2f}, M ↓ {M:.6e} → {new_M:.6e}")
                break
        if not mixed:
            # No mixing step decreased the max‑convolution → stop.
            break
    return f


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex for the smooth max‑convolution."""
    w = np.log(a0 + 1e-12)          # work in log‑softmax space

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.001      # a slightly larger start LR
    eps_start, eps_end = 0.5, 1e-12     # smoother at the very end

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 5000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule learning rate and smoothing
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # grad already includes 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)

        # restart if idle too long
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (source→target moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """Local search with simulated‑annealing acceptance."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.9, 0.001   # more aggressive early moves
    temp_start, temp_end = 0.03, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 40000                     # perturb earlier
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index: proportional to its contribution
        if total > 0.0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index: favour low contributions
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation (it={iteration}), re‑perturb")
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        idxs = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in idxs:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to escape flats
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------
def _expgrad_polish(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient updates for the exact max‑convolution objective."""
    a = a0.copy()
    cur_val = _objective(a)

    eta_start, eta_end = 1.0, 1e-5

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        idxs = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in idxs:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # schedule eta (mirrored gradient step)
        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # EG update, stay strictly positive
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        new_val = _objective(a)

        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[ExpG] new best = {best_val:.6f}")

        # tiny jitter to escape flat spots
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.018, size=n), 0.0)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random large‑move simulated‑annealing (final polishing)
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple annealing with relatively large source→target transfers."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)            # linear cooling
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # occasional jitter
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a ~1000 s budget.
    Returns a list whose sum is 1 and (tries to) minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1️⃣ Small‑scale LP phase
    # --------------------------------------------------------------
    n_small = 1500
    # start from a reasonably peaked distribution so that M > 1/n_small
    f_small = _random_beta_candidate(n_small, rng)
    f_small = f_small / f_small.sum()

    best_f_small = f_small.copy()
    best_val_small = _objective(best_f_small)
    print(f"[Init] small‑scale start, value={best_val_small:.6f}")

    # give the LP stage roughly 15 % of the total budget
    phase_lp_deadline = start_time + 0.15 * budget_s

    # we will do a few random restarts; keep the best result
    while time.time() < phase_lp_deadline:
        f_small = _random_beta_candidate(n_small, rng)
        f_small = f_small / f_small.sum()
        f_small = _lp_improve(f_small,
                              deadline=phase_lp_deadline,
                              start_time=start_time,
                              max_iters=12)
        val = _objective(f_small)
        if val < best_val_small:
            best_val_small, best_f_small = val, f_small.copy()
            print(f"[LP] new best small‑scale = {best_val_small:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Upsample to the target size
    # --------------------------------------------------------------
    n_big = 8000
    x_small = np.linspace(0.0, 1.0, n_small)
    x_big = np.linspace(0.0, 1.0, n_big)
    a_big = np.interp(x_big, x_small, best_f_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    best_a = a_big.copy()
    best_val = _objective(best_a)
    print(f"[Upsample] value after up‑sampling = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Large‑scale polishing (Adam → Hill → Sub‑grad → Exp‑grad → Anneal)
    # --------------------------------------------------------------
    # Phase A – Adam (≈20 % of remaining budget)
    phase_a_deadline = start_time + 0.35 * budget_s
    best_a, best_val = _adam_optimize(best_a, n_big,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val)
    best_a, best_val = _blend_with_uniform(best_a, n_big, best_val)

    # Phase B – Hill‑climbing (≈45 % of remaining budget)
    phase_b_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _hillclimb(best_a, n_big,
                                  deadline=phase_b_deadline,
                                  start_time=start_time,
                                  best_a=best_a,
                                  best_val=best_val)
    best_a, best_val = _blend_with_uniform(best_a, n_big, best_val)

    # Phase C – Sub‑gradient polishing (≈15 % of remaining budget)
    phase_c_deadline = start_time + 0.95 * budget_s
    best_a, best_val = _subgradient_polish(best_a, n_big,
                                           deadline=phase_c_deadline,
                                           start_time=start_time,
                                           best_a=best_a,
                                           best_val=best_val,
                                           max_iters=250_000)
    best_a, best_val = _blend_with_uniform(best_a, n_big, best_val)

    # Phase D – Mirror‑descent polishing (≈5 % of remaining budget)
    phase_d_deadline = start_time + 0.99 * budget_s
    best_a, best_val = _expgrad_polish(best_a, n_big,
                                       deadline=phase_d_deadline,
                                       start_time=start_time,
                                       best_a=best_a,
                                       best_val=best_val,
                                       max_iters=100_000)

    # Phase E – Final simulated annealing (use any leftover time)
    if time.time() < deadline:
        best_a, best_val = _random_anneal(best_a, n_big,
                                          deadline=deadline,
                                          start_time=start_time,
                                          best_a=best_a,
                                          best_val=best_val,
                                          rng=rng)

    # ------------------------------------------------------------------
    # Final secure evaluation (optional, just for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return a plain Python list of floats
    return best_a.tolist()
