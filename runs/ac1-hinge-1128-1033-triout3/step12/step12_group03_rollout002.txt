"""
Hybrid multi‑resolution optimiser for the self‑convolution max problem.

We first optimise a relatively small problem (n≈1024) with many
Adam iterations using a smooth‑max (log‑sum‑exp) objective.  The
solution is interpolated to the target size (n=8000) and refined
there with a shorter Adam run followed by a cheap projected
sub‑gradient sweep.  The schedule for the smooth‑max temperature
λ grows from a modest value to a large one, after which we switch
to the exact sub‑gradient (one‑hot weight).  This hierarchy lets us
spend most of the 1000 s budget on cheap coarse optimisation while
still producing a high‑quality solution for the final evaluation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (FFT)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation needed for the gradient:
        g_i = Σ_k w_k * a_{k-i}
    Returned vector has length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
#  Simplex projection (Euclidean) – used only in the final sub‑gradient phase
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
#  Smooth‑max utilities (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Return the weight vector w = softmax(lam * conv).
    For very large lam we fall back to a one‑hot vector on the max.
    """
    if lam > 2000.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    conv_max = conv.max()
    scaled = lam * (conv - conv_max)          # shift for numerical stability
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    return exp_vals / sum_exp


def _smooth_value(conv: np.ndarray, lam: float) -> float:
    """
    Smooth‑max value: (1/lam) * log Σ exp(lam * conv).
    If lam is huge we simply return the exact max.
    """
    if lam > 2000.0:
        return float(conv.max())
    conv_max = conv.max()
    scaled = lam * (conv - conv_max)
    sum_exp = np.exp(scaled).sum()
    return conv_max + math.log(sum_exp) / lam


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative sequence that (approximately) minimises
    ``evaluate_sequence``.  The method works in two resolutions:
    1) a cheap high‑iteration optimisation on n_coarse ≈ 1024,
    2) up‑sampling to n_fine = 8000 and a short fine‑grained run.
    Finally a few projected sub‑gradient sweeps squeeze out the last
    bits of improvement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem sizes
    # ------------------------------------------------------------------
    n_fine = 8000                     # final length (as used in the reference)
    n_coarse = 1024                   # a power of two – cheap to optimise

    # ------------------------------------------------------------------
    # Initialise a starting distribution (interpolate old solution if possible)
    # ------------------------------------------------------------------
    try:
        old_seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old_seq.size > 1:
            # interpolate to coarse grid
            old_x = np.linspace(0.0, 1.0, old_seq.size)
            new_x = np.linspace(0.0, 1.0, n_coarse)
            a0_coarse = np.interp(new_x, old_x, old_seq)
            a0_coarse = np.clip(a0_coarse, 0.0, None)
            if a0_coarse.sum() < 1e-12:
                a0_coarse = np.full(n_coarse, 1.0 / n_coarse)
            else:
                a0_coarse /= a0_coarse.sum()
            print("[init] interpolated previous solution to coarse grid")
        else:
            raise Exception
    except Exception:
        a0_coarse = np.full(n_coarse, 1.0 / n_coarse)
        print("[init] using uniform coarse start")

    # ------------------------------------------------------------------
    # Global best tracker (kept on the fine grid, but we initialise it
    # from the coarse solution after up‑sampling)
    # ------------------------------------------------------------------
    best_fine = None
    best_val = float("inf")   # value = exact max of self‑convolution

    # ------------------------------------------------------------------
    # Helper: evaluate an array (any length) and update global best
    # ------------------------------------------------------------------
    def _maybe_update_best(a: np.ndarray):
        nonlocal best_fine, best_val
        conv = _full_convolution(a)
        cur_max = float(conv.max())
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_fine = a.copy()
            print(f"[global] new best (max conv) = {best_val:.9f}")

    # ------------------------------------------------------------------
    # 1️⃣  COARSE optimisation (lots of cheap Adam steps)
    # ------------------------------------------------------------------
    coarse_deadline = start_time + (budget_s * 0.55)   # 55 % of the budget
    # Adam hyper‑parameters for the coarse phase
    lr = 0.07
    b1, b2 = 0.9, 0.999
    eps = 1e-12
    lambda_start, lambda_end = 5.0, 5000.0

    # initialise logits (log‑softmax parameters)
    z = np.log(a0_coarse + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    iter_coarse = 0
    stagnation = 0
    IMPROVE_LIMIT = 3000

    while time.time() < coarse_deadline:
        # ------------------------------------------------------------------
        # 1.1  Convert logits to probabilities (soft‑max)
        # ------------------------------------------------------------------
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # ------------------------------------------------------------------
        # 1.2  Full convolution (exact) – needed for both loss and λ schedule
        # ------------------------------------------------------------------
        conv = _full_convolution(a)

        # ------------------------------------------------------------------
        # 1.3  Update global best (uses exact max)
        # ------------------------------------------------------------------
        cur_max = float(conv.max())
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_fine = a.copy()          # temporary – will be up‑sampled later
            stagnation = 0
            print(f"[coarse iter {iter_coarse:06d}] new best = {best_val:.9f}")
        else:
            stagnation += 1

        # ------------------------------------------------------------------
        # 1.4  Smooth‑max temperature (linear in elapsed proportion)
        # ------------------------------------------------------------------
        prog = (time.time() - start_time) / (coarse_deadline - start_time)
        lam = lambda_start * (1.0 - prog) + lambda_end * prog

        # ------------------------------------------------------------------
        # 1.5  Weights for gradient (soft‑max over convolution entries)
        # ------------------------------------------------------------------
        w = _smooth_weights(conv, lam)

        # ------------------------------------------------------------------
        # 1.6  Gradient w.r.t. a (using cross‑correlation)
        # ------------------------------------------------------------------
        g_raw = _cross_corr(w, a)          # shape (n_coarse,)
        g = 2.0 * g_raw                     # derivative of Σ w_k (a*a)_k

        # ------------------------------------------------------------------
        # 1.7  Back‑prop through the soft‑max (logits → a)
        # ------------------------------------------------------------------
        dot_ag = np.dot(a, g)
        grad_z = a * (g - dot_ag)

        # ------------------------------------------------------------------
        # 1.8  Adam update on logits
        # ------------------------------------------------------------------
        iter_coarse += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z ** 2)
        m_hat = m / (1.0 - b1 ** iter_coarse)
        v_hat = v / (1.0 - b2 ** iter_coarse)
        z -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ------------------------------------------------------------------
        # 1.9  Add occasional noise to escape long plateaus
        # ------------------------------------------------------------------
        if stagnation > IMPROVE_LIMIT:
            z += np.random.normal(scale=0.07, size=n_coarse)
            stagnation = 0
            print(f"[coarse] added noise at iteration {iter_coarse}")

        # ------------------------------------------------------------------
        # cheap early exit if we already beat the target bound (≈1.503)
        # ------------------------------------------------------------------
        if best_val * 2.0 * n_coarse < 1.503:   # max conv * 2n = evaluate_sequence when sum=1
            print("[coarse] target already reached – breaking early")
            break

    # ------------------------------------------------------------------
    # 2️⃣  INTERPOLATE to the target size (n_fine)
    # ------------------------------------------------------------------
    if best_fine is None:                     # fallback – use the latest a
        best_fine = a.copy()
    # linear interpolation of the probability vector
    x_coarse = np.linspace(0.0, 1.0, n_coarse)
    x_fine   = np.linspace(0.0, 1.0, n_fine)
    a_fine = np.interp(x_fine, x_coarse, best_fine)
    # clamp and renormalise
    a_fine = np.clip(a_fine, 0.0, None)
    if a_fine.sum() < 1e-12:
        a_fine = np.full(n_fine, 1.0 / n_fine)
    else:
        a_fine /= a_fine.sum()
    _maybe_update_best(a_fine)

    # ------------------------------------------------------------------
    # 3️⃣  FINE optimisation – fewer iterations but on the true size
    # ------------------------------------------------------------------
    fine_deadline = deadline
    # Adam hyper‑parameters for fine phase (more conservative)
    lr_f = 0.04
    b1_f, b2_f = 0.9, 0.999
    lambda_start_f, lambda_end_f = 5.0, 8000.0
    stagnation_f = 0
    IMPROVE_LIMIT_F = 2000

    # initialise logits for the fine grid
    z_f = np.log(a_fine + eps)
    m_f = np.zeros_like(z_f)
    v_f = np.zeros_like(z_f)
    iter_fine = 0

    while time.time() < fine_deadline:
        # --------------------------------------------------------------
        # 3.1 probabilities
        # --------------------------------------------------------------
        max_zf = np.max(z_f)
        exp_zf = np.exp(z_f - max_zf)
        a = exp_zf / exp_zf.sum()

        # --------------------------------------------------------------
        # 3.2 convolution
        # --------------------------------------------------------------
        conv = _full_convolution(a)

        # --------------------------------------------------------------
        # 3.3 track best
        # --------------------------------------------------------------
        cur_max = float(conv.max())
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_fine = a.copy()
            stagnation_f = 0
            print(f"[fine iter {iter_fine:06d}] new best = {best_val:.9f}")
        else:
            stagnation_f += 1

        # --------------------------------------------------------------
        # 3.4 temperature schedule (linear in overall runtime)
        # --------------------------------------------------------------
        prog = (time.time() - start_time) / (fine_deadline - start_time)
        lam = lambda_start_f * (1.0 - prog) + lambda_end_f * prog

        # --------------------------------------------------------------
        # 3.5 compute smooth‑max weights
        # --------------------------------------------------------------
        w = _smooth_weights(conv, lam)

        # --------------------------------------------------------------
        # 3.6 gradient w.r.t. a
        # --------------------------------------------------------------
        g_raw = _cross_corr(w, a)
        g = 2.0 * g_raw

        # --------------------------------------------------------------
        # 3.7 back‑prop through soft‑max
        # --------------------------------------------------------------
        dot_ag = np.dot(a, g)
        grad_z = a * (g - dot_ag)

        # --------------------------------------------------------------
        # 3.8 Adam update
        # --------------------------------------------------------------
        iter_fine += 1
        m_f = b1_f * m_f + (1.0 - b1_f) * grad_z
        v_f = b2_f * v_f + (1.0 - b2_f) * (grad_z ** 2)
        m_hat_f = m_f / (1.0 - b1_f ** iter_fine)
        v_hat_f = v_f / (1.0 - b2_f ** iter_fine)
        z_f -= lr_f * m_hat_f / (np.sqrt(v_hat_f) + 1e-8)

        # --------------------------------------------------------------
        # 3.9 occasional noise to avoid getting stuck
        # --------------------------------------------------------------
        if stagnation_f > IMPROVE_LIMIT_F:
            z_f += np.random.normal(scale=0.05, size=n_fine)
            stagnation_f = 0
            print(f"[fine] added Gaussian noise at iter {iter_fine}")

        # --------------------------------------------------------------
        # 3.10 early exit if already below target
        # --------------------------------------------------------------
        if best_val * 2.0 * n_fine < 1.503:
            print("[fine] target reached – exiting early")
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final projected sub‑gradient polishing (as in the reference)
    # ------------------------------------------------------------------
    a = best_fine.copy()
    obj = best_val
    step = 0.025

    for sweep in range(1300):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the true max (see reference implementation)
        g = np.zeros_like(a)
        n = n_fine
        if kmax < n:
            # overlap region: 0 … kmax
            g[:kmax + 1] = 2.0 * a[kmax::-1]
        else:
            offset = kmax - (n - 1)
            L = n - offset
            g[offset:] = 2.0 * a[:L][::-1]

        a_trial = _project_simplex(a - step * g)
        conv_trial = _full_convolution(a_trial)
        obj_trial = float(conv_trial.max())

        if obj_trial < obj - 1e-13:
            a = a_trial
            obj = obj_trial
            best_fine = a.copy()
            best_val = obj
            step = min(step * 1.12, 0.5)
            print(f"[subgrad] improved to {obj:.9f}")
            if obj * 2.0 * n_fine < 1.503:
                print("[subgrad] reached target – stopping sweep")
                break
        else:
            step *= 0.71

    # ------------------------------------------------------------------
    # 5️⃣  Final secure evaluation (the only place we call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_fine.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_fine.tolist()
