    a = a0.copy()
    # convert to log domain
    eps = 1e-30
    z = np.log(a + eps)
    # Adam set up
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(z); v = np.zeros_like(z)
    for i in range(steps_stage1):
        alpha = 1 + (max_alpha - 1) * ((i+1)/steps_stage1) ** 1.5
        a = np.exp(z)
        a = a / a.sum()
        conv = full_convolution(a)
        maxc = conv.max()
        # compute softmax weights
        exp_scaled = np.exp(alpha * (conv - maxc))
        w = exp_scaled / exp_scaled.sum()
        rev_a = a[::-1]
        conv_rev = full_convolution_two(rev_a, w)
        g_a = 4.0 * n * conv_rev[n-1:n-1+n]
        # Chain rule
        dot = np.dot(g_a, a)
        grad_z = a * (g_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z**2)
        m_hat = m / (1 - beta1 ** (i+1))
        v_hat = v / (1 - beta2 ** (i+1))
        lr = start_lr * (1 - (i+1)/steps_stage1) + end_lr * ((i+1)/steps_stage1)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
    
    # End of stage1: get a
    a = np.exp(z)
    a = a / a.sum()
    # Stage2 mirror descent
    for i in range(steps_stage2):
        conv = full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)

  slightly more aggressive step‑size schedule.

All restarts share the same budget (the total wall‑clock time is limited to
`budget_s`).  The best vector seen (lowest objective) is stored; after the
budget expires it is polished by a long mirror‑descent run and finally returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two different vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max‑entry (polishing)
# ----------------------------------------------------------------------
def _mirror_descent(a: np.ndarray,
                    steps: int = 20000,
                    eta_start: float = 0.9,
                    eta_end: float = 0.01) -> np.ndarray:
    """
    Run entropy‑mirror descent on the exact max‑convolution entry.
    The vector a is assumed to be a probability vector (sum=1, a_i≥0).
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]          # sub‑gradient of the max entry
        eta = eta_start * (1.0 - t / steps) + eta_end * (t / steps)
        # multiplicative (exponential) mirror update
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine respects the 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 works well and matches the previous best.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Helper: objective for a probability vector (sum = 1)
    # ------------------------------------------------------------------
    def obj_value(a_vec: np.ndarray) -> float:
        """Exact objective for a probability vector."""
        conv = _full_convolution(a_vec)
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # Keep the globally available solution as a possible starting point
    # ------------------------------------------------------------------
    global_init = None
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        try:
            old = np.asarray(old_seq, dtype=float)
            if old.shape[0] != n:
                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / a0.sum()
            global_init = a0
            print("[init] using interpolated previous solution")
        except Exception:
            global_init = None

    # ------------------------------------------------------------------
    # Book‑keeping for the best solution seen overall
    # ------------------------------------------------------------------
    best_a = None
    best_obj = np.inf
    restart_counter = 0

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets a modest budget of iterations.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # estimate remaining time, allocate a small chunk (≈4 s) per restart
        now = time.time()
        time_left = deadline - now
        if time_left < 4.0:
            break

        # --------------------------------------------------------------
        # Initialise a probability vector for this restart.
        # --------------------------------------------------------------
        if restart_counter == 0 and global_init is not None:
            a = global_init.copy()
        else:
            # random Dirichlet – concentration parameter chosen stochastically
            conc = rng.uniform(0.3, 4.0)          # controls sparsity / flatness
            a = rng.gamma(conc, 1.0, size=n)
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)

        # work in log‑domain to guarantee positivity
        eps = 1e-30
        z = np.log(a + eps)

        # ------------------------------------------------------------------
        # Hyper‑parameters for this restart (slightly randomised)
        # ------------------------------------------------------------------
        # Adam parameters
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lr_start = 0.30
        lr_end   = 0.008

        # Temperature schedule for the soft‑max surrogate
        max_alpha = rng.uniform(300.0, 600.0)   # final temperature (large)
        temp_power = 2.0                       # growth exponent

        # Number of iterations for each phase – chosen to fit the time slice
        # (very cheap – a few thousand iterations are enough to make progress)
        steps_phase1 = 3000
        steps_phase2 = 2000

        # --------------------------------------------------------------
        # Phase 1 – smoothed Adam
        # --------------------------------------------------------------
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        for it in range(steps_phase1):
            # temperature α (soft‑max smoothness)
            prog = (it + 1) / steps_phase1
            alpha = 1.0 + (max_alpha - 1.0) * (prog ** temp_power)

            # probability vector from logits
            a = np.exp(z)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a_sum

            # full self‑convolution
            conv = _full_convolution(a)

            # soft‑max weights over the convolution entries
            maxc = conv.max()
            scaled = np.exp(alpha * (conv - maxc))
            w = scaled / scaled.sum()                     # Σ w = 1

            # gradient of the smoothed objective w.r.t. a
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, w)
            # slice corresponding to the original length
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]

            # chain rule through the softmax(z) mapping
            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))

            lr = lr_start * (1.0 - prog) + lr_end * prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------------------------------------------------------------
        # Phase 2 – exact entropy mirror descent
        # --------------------------------------------------------------
        a = np.exp(z)
        a_sum = a.sum()
        a = a / a_sum if a_sum > 0 else np.full_like(a, 1.0 / n)

        eta_start, eta_end = 0.9, 0.01
        for it in range(steps_phase2):
            conv = _full_convolution(a)
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)

            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                g[idx] = 2.0 * a[k - idx]

            stage_prog = (it + 1) / steps_phase2
            eta = eta_start * (1.0 - stage_prog) + eta_end * stage_prog

            a = a * np.exp(-eta * g)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

        # --------------------------------------------------------------
        # Evaluate the candidate from this restart
        # --------------------------------------------------------------
        cur_obj = obj_value(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[restart {restart_counter}] new global best O={best_obj:.6f}")

        restart_counter += 1

    # ------------------------------------------------------------------
    # Final polishing – a longer mirror‑descent run on the best vector found
    # ------------------------------------------------------------------
    if best_a is None:
        # fall back to a simple Beta(2,2) shape
        i = np.arange(1, n + 1, dtype=np.float64)
        best_a = (i * (n - i + 1))
        best_a = best_a / best_a.sum()
        best_obj = obj_value(best_a)

    refined = _mirror_descent(best_a, steps=20000, eta_start=0.85, eta_end=0.001)
    refined_obj = obj_value(refined)
    if refined_obj < best_obj:
        best_obj = refined_obj
        best_a = refined
        print(f"[polish] improved after final mirror descent: O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Prepare the result for the wrapper; we return a plain Python list.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
