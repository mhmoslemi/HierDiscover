def _run_optimizer(start_time, deadline, a0, n, rng):
    import time
    z = np.log(a0 + 1e-30)
    # Hyperparameters as defined.

    best_a = None
    best_obj = np.inf
    stagn_counter=0
    it=0
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    mirror_mode=False
    switch_prog=None

    while time.time() < deadline:
        now = time.time()
        prog = (now - start_time) / (deadline - start_time)
        alpha = 1.0 + (max_alpha - 1.0) * (prog ** 1.2) # maybe power 1.2
        
        if not mirror_mode and alpha >= switch_alpha:
            mirror_mode=True
            switch_prog=prog
            # Convert to probability vector a from logits
            a = np.exp(z)
            a = a / a.sum() if a.sum()>0 else np.full_like(a, 1.0 / n)
            print(f"[run] switch to mirror mode at prog={prog:.3f}")
        
        # get a
        if not mirror_mode:
            a = np.exp(z)
            a = a / a.sum()
        # compute convolution
        conv = _full_convolution(a)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv / (a.sum()**2)   # a sum is 1 due to normalization but safe.
        # record best
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagn_counter = 0
            print(f"[run] new best O={best_obj:.6f} at it={it}")
        else:
            stagn_counter +=1

        # gradient update
        if not mirror_mode:
            maxc = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc))
            sumexp = exp_scaled.sum()
            weights = exp_scaled / sumexp
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            g_a = 4.0 * n * conv_rev[n-1:n-1+n]
            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z**2)
            m_hat = m / (1.0 - beta1**it)
            v_hat = v / (1.0 - beta2**it)
            lr = lr_start * (1.0 - prog) + lr_end * prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        else:
            # exact entropy mirror descent on max entry
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi+1)
                g[idx] = 2.0 * a[k - idx]
            # step size schedule
            stage_prog = (prog - switch_prog) / (1.0 - switch_prog) if switch_prog else prog
            eta_start, eta_end = 0.5, 0.005
            eta = eta_start * (1.0 - stage_prog) + eta_end * stage_prog
            a = a * np.exp(-eta * g)
            a = a / a.sum()
            it += 1

        # stagnation escape
        if stagn_counter >= stagn_limit:
            if not mirror_mode:
                z += rng.normal(scale=jitter_adam, size=n)
                print(f"[run] jitter Adam at it={it}")
            else:
                perturb = rng.normal(scale=jitter_mirror, size=n)
                a = np.maximum(a + perturb, 0.0)
                if a.sum() == 0:
                    a = np.full_like(a, 1.0 / n)
                else:
                    a = a / a.sum()
                print(f"[run] jitter Mirror at it={it}")
            stagn_counter=0

    return best_a, best_obj

def _greedy_set(max_c, n, rng):
    counts = np.zeros(2*n - 1, dtype=np.int16)
    S = []
    candidates = np.arange(n, dtype=np.int32)
    rng.shuffle(candidates)
    for i in candidates:
        # Check self-pair
        if counts[2*i] >= max_c:
            continue
        if S:
            idxs = i + np.array(S, dtype=np.int32)
            if (counts[idxs] >= max_c).any():
                continue
        # Accept i
        for j in S:
            counts[i + j] += 1
        counts[2*i] += 1
        S.append(i)
    return np.array(S, dtype=np.int32), counts

"""
Propose a step‑function that beats the previous bound (≈1.508).
----------------------------------------------------------------
Algorithm
---------

* **Greedy support construction** – build a set of indices S such that
  each unordered sum i+j appears at most *C* times (C∈{2,3,4}).  This
  yields a sparse support on which the convolution peak is already
  modest.

* **Positive‑log‑domain optimisation** – initialise a probability vector
  with a small uniform background and weight 1 on the support S.
  Optimise the probability vector a (∑a=1, a≥0) with two stages:

    – *Stage 1* – smooth Adam on a soft‑max surrogate of the maximal
      convolution entry.  The temperature α grows from 1 to a large
      value (≈350) and the Adam learning‑rate is annealed.

    – *Stage 2* – entropy‑mirror descent directly on the exact maximal
      convolution entry (sub‑gradient g_i = 2·a_{k−i}) with a linearly
      decreasing step‑size η.

  Stagnation is broken by adding small Gaussian jitter.
  The best vector seen during the run is remembered.

* **Multiple restarts** – the whole procedure is performed a few
  times (different random seeds, different C) within the global time
  budget; the globally best vector is kept.

* **Final polishing** – a longer entropy‑mirror descent (10 000 steps)
  is run on the best vector to squeeze out the last bits.

The function respects the 1000 s budget, prints progress information
and finally returns the best sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolutions (not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two (different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max‑entry of a self‑convolution
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 10000,
                eta_start: float = 0.45) -> np.ndarray:
    """Entropy‑mirror descent on the exact maximum entry of *a*a."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]                # sub‑gradient

        # linear decay of step size
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Greedy construction of a B2[C] set (bounded pair‑sum multiplicity)
# ----------------------------------------------------------------------
def _greedy_set(max_c: int, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Returns a set of indices S ⊂ {0,…,n‑1} such that each unordered sum
    i+j appears at most ``max_c`` times.  The construction is greedy and
    randomised.
    """
    counts = np.zeros(2 * n - 1, dtype=np.int16)   # multiplicities of sums
    S = []

    cand = np.arange(n, dtype=np.int32)
    rng.shuffle(cand)

    for i in cand:
        # self‑pair (i,i)
        if counts[2 * i] >= max_c:
            continue

        if S:
            idxs = i + np.array(S, dtype=np.int32)
            if (counts[idxs] >= max_c).any():
                continue

        # add i to the set
        for j in S:
            counts[i + j] += 1
        counts[2 * i] += 1
        S.append(i)

    return np.array(S, dtype=np.int32)


# ----------------------------------------------------------------------
# Core optimiser – smoothed Adam + exact mirror descent
# ----------------------------------------------------------------------
def _run_optimizer(start_time: float, deadline: float,
                   a0: np.ndarray, n: int,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Optimises a probability vector a (sum = 1, a ≥ 0) to minimise
    ``evaluate_sequence``.  Returns the best vector seen and its objective.
    """
    # keep everything in the log‑domain for positivity
    eps = 1e-30
    z = np.log(a0 + eps)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    lr_start, lr_end = 0.25, 0.006          # learning‑rate schedule
    max_alpha = 350.0                      # final soft‑max temperature
    switch_alpha = 200.0                   # when to hand over to mirror mode

    # stagnation handling
    stagn_limit = 1500
    jitter_adam = 0.02
    jitter_mirror = 0.004

    best_a = None
    best_obj = np.inf
    stagn_counter = 0
    it = 0
    mirror_mode = False
    switch_prog = None

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)   # 0 → 1
        alpha = 1.0 + (max_alpha - 1.0) * (prog ** 1.2)       # temperature growth

        # ------------------------------------------------------------------
        # Switch to exact mirror descent when the surrogate is sharp enough
        # ------------------------------------------------------------------
        if not mirror_mode and alpha >= switch_alpha:
            mirror_mode = True
            switch_prog = prog
            a = np.exp(z)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)
            print(f"[mode] → mirror descent (prog={prog:.3f})")

        # ------------------------------------------------------------------
        # Current probability vector
        # ------------------------------------------------------------------
        if not mirror_mode:
            a = np.exp(z)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        # ------------------------------------------------------------------
        # Full self‑convolution and objective
        # ------------------------------------------------------------------
        conv = _full_convolution(a)
        obj = 2.0 * n * conv.max()          # exact objective (sum(a) = 1)

        # ------------------------------------------------------------------
        # Record best solution
        # ------------------------------------------------------------------
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagn_counter = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}")
        else:
            stagn_counter += 1

        # ------------------------------------------------------------------
        # Gradient / update step
        # ------------------------------------------------------------------
        if not mirror_mode:
            # ---- Stage 1: smoothed Adam on the soft‑max surrogate ----
            maxc = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc))
            sumexp = exp_scaled.sum()
            weights = exp_scaled / sumexp                     # soft‑max distribution

            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]        # gradient w.r.t a

            # chain rule through softmax(z)
            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = lr_start * (1.0 - prog) + lr_end * prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        else:
            # ---- Stage 2: exact entropy‑mirror descent on the max entry ----
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)

            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                g[idx] = 2.0 * a[k - idx]                     # sub‑gradient

            # step‑size linearly decays inside the mirror phase
            stage_prog = (prog - switch_prog) / (1.0 - switch_prog) if switch_prog else prog
            eta_start, eta_end = 0.5, 0.005
            eta = eta_start * (1.0 - stage_prog) + eta_end * stage_prog

            a = a * np.exp(-eta * g)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

            it += 1

        # ------------------------------------------------------------------
        # Escape stagnation
        # ------------------------------------------------------------------
        if stagn_counter >= stagn_limit:
            if not mirror_mode:
                z += rng.normal(scale=jitter_adam, size=n)
                print(f"[reset] jitter (Adam) at it={it}")
            else:
                perturb = rng.normal(scale=jitter_mirror, size=n)
                a = np.maximum(a + perturb, 0.0)
                s = a.sum()
                a = a / s if s > 0 else np.full_like(a, 1.0 / n)
                print(f"[reset] jitter (Mirror) at it={it}")
            stagn_counter = 0

    return best_a, best_obj

# ----------------------------------------------------------------------

    """

    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # Determine problem size
    # ------------------------------------------------------------------
    old_seq = globals().get("height_sequence_1")

            print("[init] using length from previous solution:", n)
        except Exception:
            n = 8000
    else:
        n = 8000
    print("[init] problem size n =", n)

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_obj = np.inf
    restarts = 3                     # number of independent restarts
    remaining = restarts

    # ------------------------------------------------------------------
    # Main loop over restarts (time‑aware)
    # ------------------------------------------------------------------
    for r in range(restarts):
        now = time.time()
        if now >= deadline:

        # allocate roughly equal remaining time to each future restart
        time_left = deadline - now
        # give a generous slice to this run (leave a little for polishing)

        # pick a bound on the unordered‑sum multiplicity

        print(f"[restart {r}] max_c={max_c}, support size={len(S)}")

        # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    return final_seq
