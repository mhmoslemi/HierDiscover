    best_a = None

    # Pre-alloc some arrays unify.

    it = 0
    while time.time() < deadline:
        # compute conv via FFT
        conv = _full_convolution(a)
        M = conv.max()
        obj = 2 * n * M

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print progress.

        # compute gradient (subgradient) for shift(s) achieving max
        max_indices = np.where(conv >= M - 1e-12)[0]  # handle ties
        # compute subgradient sum over all active shifts
        g = np.zeros_like(a)
        # For each k offset compute gradient for distance d = k - (n - 1)
        for k in max_indices:
            d = k - (n - 1)
            if d >= 0:
                g[:n - d] += a[d:]
                g[d:] += a[:n - d]
            else:
                d = -d
                g[d:] += a[:n - d]
                g[:n - d] += a[d:]

        # average if multiple
        g /= len(max_indices)

        # compute step size
        it += 1
        eta = 0.8 / np.sqrt(it)   # constant can be tuned

        # gradient descent step (note we want to reduce max -> move against gradient)
        a_new = a - eta * g

        # Ensure non-negative: if negative entries, set to 0 and renormalize.
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= s

        a = a_new

        # occasional jitter injection if stagnant
        if it % 5000 == 0:
            jitter = np.random.normal(scale=0.01, size=n)
            a = a + jitter
            a = np.maximum(a, 0)
            a /= a.sum()

    # After loop, polishing using mirror descent with many steps
    if best_a is None:
        best_a = a
    time_left = deadline - time.time()
    steps = int(min(50000, max(5000, time_left * 15000)))  # adapt
    best_a = _max_refine(best_a, steps=steps, eta_start=0.5)

    final_seq = best_a.tolist()
    score = evaluate_sequence(final_seq)
    print final.

    return final_seq

Implementation of projection onto simplex using standard algorithm:

def _project_simplex(v):
    """Project v onto the probability simplex (nonnegative, sum=1)."""
    # using algorithm from Wang & Carreira-Perpinan (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step function with a low objective value.

Algorithm
---------
1. **Initialisation** – start from the previously best sequence if it
   exists (global variable ``height_sequence_1``).  If not, fall back to the
   uniform distribution on the chosen length *n* (default 8000).

2. **Projected sub‑gradient descent** – the objective
   `f(a)=max_k (a∗a)[k]` is convex.  Its sub‑gradient at the active index
   `k*` is simply `g_i = 2·a_{k*‑i}` for all `i` where the index is
   inside the array.  In each iteration we
   * evaluate the full convolution via an FFT,
   * compute the sub‑gradient at the maximal entry,
   * take a step `a ← a – η·g` with a diminishing step size `η = η0 / sqrt(t)`,
   * project the result back onto the probability simplex (non‑negative,
     sum = 1).

   Simple back‑tracking halves `η` until the maximal convolution does not
   increase.  When no improvement has been seen for a while we inject a
   small random jitter to escape plateaus.

3. **Entropy‑mirror polishing** – once the time budget for the main loop
   is exhausted we switch to the highly‑effective mirror‑descent scheme
   from the reference implementation (`_max_refine`).  This works directly
   on the true max‑entry (no smoothing) and can reduce the bound further.

4. **Final evaluation** – the best sequence found is handed to the
   supplied ``evaluate_sequence`` routine and returned.

The whole procedure respects a 1000 s wall‑clock limit, uses only a
single CPU core, and works entirely with NumPy (no external solvers).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full convolution (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {w | w_i ≥ 0, Σ w_i = 1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays.")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Mirror‑descent polishing (exact max, same as reference)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 3000, eta_start: float = 0.5) -> np.ndarray:
    """Entropy‑mirror descent on the exact max‑entry of the self‑convolution."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of max‑entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # linearly decaying step size
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights that (hopefully) yields a lower
    value of ``evaluate_sequence`` than the current best (~1.5081).
    The routine respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # reserve a small safety margin for the final call and polishing
    deadline = start + budget_s - 0.9

    # --------------------------------------------------------------
    # Problem size – use the length of the previously known solution if any
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        a = np.maximum(old, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)
        print("[init] using previous solution with n =", n)
    except Exception:
        n = 8000
        a = np.full(n, 1.0 / n)
        print("[init] no previous solution – start with uniform, n =", n)

    # --------------------------------------------------------------
    # Sub‑gradient descent parameters
    # --------------------------------------------------------------
    eta0 = 0.50                # initial step size (will be divided by sqrt(t))
    stagnation = 0
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    polishing_reserve = 0.15 * budget_s   # time reserved for the final polishing

    # --------------------------------------------------------------
    # Main optimisation loop (projected sub‑gradient descent)
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline - polishing_reserve:
            break

        # Full self‑convolution and objective
        conv = _full_convolution(a)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv

        # Record best solution seen so far
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagnation = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}  t={now-start:.1f}s")
        else:
            stagnation += 1

        # ----------------------------------------------------------
        # Sub‑gradient of the active max‑entry
        # ----------------------------------------------------------
        k = int(conv.argmax())                # active index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]        # 2·a_{k‑i}

        # ----------------------------------------------------------
        # Step size (diminishing) and back‑tracking line search
        # ----------------------------------------------------------
        it += 1
        eta = eta0 / np.sqrt(it)

        for _ in range(5):                    # at most 5 back‑track attempts
            a_candidate = a - eta * g
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

            # evaluate candidate (one extra FFT)
            conv_cand = _full_convolution(a_candidate)
            if conv_cand.max() <= max_conv + 1e-15:
                a = a_candidate
                break
            eta *= 0.5                        # shrink step and retry
        # ----------------------------------------------------------
        # Random jitter to escape plateaus
        # ----------------------------------------------------------
        if stagnation >= 5000:
            jitter = np.random.normal(scale=0.008, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            stagnation = 0
            print(f"[jitter] injected after {it} iterations")

    # --------------------------------------------------------------
    # Polishing phase – entropy‑mirror descent on the exact max entry
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_a is not None and time_left > 0.2:
        # allocate as many mirror steps as the remaining time permits
        steps = max(3000, int(min(60000, time_left * 18000)))
        print(f"[polish] mirror descent start – {steps} steps")
        a_refined = _max_refine(best_a, steps=steps, eta_start=0.45)
        conv_ref = _full_convolution(a_refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = a_refined
            print(f"[polish] improved to O={best_obj:.6f}")
        else:
            print(f"[polish] no improvement, best O remains {best_obj:.6f}")

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx O={best_obj:.6f})")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
