   shape for the full‑size problem while keeping the total number of

   * A tiny regularising term `λ·a` is added to the sub‑gradient so that

import time

# -------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# -------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)−1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two (generally different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# -------------------------------------------------------------
# Entropy‑mirror descent on the exact max entry (polishing)
# -------------------------------------------------------------

    """Exact mirror‑descent refinement on the maximal self‑convolution entry."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            # correct sub‑gradient (2·a_i on the diagonal)
            g[idx] = 2.0 * a[k - idx]

        # linear decay of η
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0:
            a /= s
        else:                      # safeguard against collapse
            a.fill(1.0 / n)
    return a


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Return a non‑negative step‑function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects a 1000 s budget."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5               # small safety margin

    # ---------------------------------------------------------
    # Phase split (fractions of the total budget)
    # ---------------------------------------------------------
    phase1_frac = 0.32   # coarse optimisation on a smaller n
    phase2_frac = 0.58   # fine optimisation on the target n
    t1_end = start + budget_s * phase1_frac
    t2_end = t1_end + budget_s * phase2_frac

    # ---------------------------------------------------------
    # Helper to compute the exact objective (sum is forced to 1)
    # ---------------------------------------------------------
    def _obj(a_vec: np.ndarray) -> float:
        n_loc = a_vec.shape[0]
        conv = _full_convolution(a_vec)
        return 2.0 * n_loc * conv.max()

    # ---------------------------------------------------------
    # Phase 1 – coarse search (n≈2000)
    # ---------------------------------------------------------
    n_coarse = 2000
    best_a = None
    best_obj = np.inf

    while time.time() < t1_end:
        # random Dirichlet start (α=0.5 encourages sparsity)
        init_alpha = 0.5
        a = np.random.gamma(init_alpha, 1.0, size=n_coarse)
        a /= a.sum()
        z = np.log(a + 1e-30)                # logits for positivity

        # Adam parameters
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        it = 0

        while True:
            now = time.time()
            if now >= t1_end:
                break

            # progress inside phase 1 (0 → 1)
            prog = (now - start) / (t1_end - start)

            # soft‑max temperature grows polynomially
            alpha = 1.0 + (200.0 - 1.0) * (prog ** 1.4)

            # probabilities from logits
            a = np.exp(z)
            s = a.sum()
            a = a / s if s > 0 else np.full(n_coarse, 1.0 / n_coarse)

            # convolution and soft‑max surrogate
            conv = _full_convolution(a)
            maxc = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc))
            weights = exp_scaled / exp_scaled.sum()

            # gradient of the surrogate with respect to a
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            g_a = 4.0 * n_coarse * conv_rev[n_coarse - 1:n_coarse - 1 + n_coarse]

            # chain rule through the soft‑max (logits)
            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            # Adam update on the logits
            it += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)

            # learning‑rate schedule (high at start, low at end of phase 1)
            lr = 0.25 * (1.0 - prog) + 0.01 * prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # exact objective check (sum already 1)
            obj = _obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                # print progress sparingly to avoid I/O overhead
                # print(f"[coarse] new best O={best_obj:.6f} time={now-start:.1f}s")

    # ---------------------------------------------------------
    # If the coarse stage produced nothing (unlikely) fall back
    # ---------------------------------------------------------
    if best_a is None:
        # smooth Beta(2,2) shape on the target size
        n_target = 8000
        i = np.arange(1, n_target + 1, dtype=np.float64)
        a = (i * (n_target - i + 1)).astype(np.float64)
        a = a / a.sum()
        best_a = a
        best_obj = _obj(best_a)

    # ---------------------------------------------------------
    # Phase 2 – up‑sample & fine optimisation on the target size
    # ---------------------------------------------------------
    n_target = 8000
    if best_a.shape[0] != n_target:
        x_small = np.linspace(0.0, 1.0, best_a.shape[0])
        x_big = np.linspace(0.0, 1.0, n_target)
        a_fine = np.interp(x_big, x_small, best_a)
        a_fine = np.maximum(a_fine, 0.0)
        a_fine /= a_fine.sum()
    else:
        a_fine = best_a.copy()

    # Initialise Adam on the fine problem (logits)
    n = n_target
    z = np.log(a_fine + 1e-30)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    it = 0

    # hyper‑parameters for the fine stage
    lr_start, lr_end = 0.20, 0.005
    max_alpha = 300.0
    switch_alpha = 140.0

    mirror_mode = False
    stagn_counter = 0
    stagn_limit = 5000
    jitter_adam = 0.02
    jitter_mirror = 0.003

    best_a_fine = a_fine.copy()
    best_obj_fine = _obj(best_a_fine)

    while True:
        now = time.time()
        if now >= t2_end:
            break

        prog = (now - start) / (t2_end - start)   # overall progress into phase 2
        alpha = 1.0 + (max_alpha - 1.0) * (prog ** 1.6)

        # -----------------------------------------------------------------
        # Switch to exact mirror descent once the temperature is large enough
        # -----------------------------------------------------------------
        if not mirror_mode and alpha >= switch_alpha:
            mirror_mode = True
            mirror_start = prog
            # materialise probability vector for the mirror phase
            a = np.exp(z)
            a = a / a.sum()
            # print(f"[fine] entering mirror mode at t={now-start:.1f}s")

        # -----------------------------------------------------------------
        # Current probability vector
        # -----------------------------------------------------------------
        if not mirror_mode:
            a = np.exp(z)
            a = a / a.sum()
        # else: a already holds the updated probabilities from the previous step

        conv = _full_convolution(a)
        obj = 2.0 * n * conv.max()

        # keep the best ever seen in this phase
        if obj < best_obj_fine - 1e-12:
            best_obj_fine = obj
            best_a_fine = a.copy()
            stagn_counter = 0
            # print(f"[fine] new best O={best_obj_fine:.6f}")
        else:
            stagn_counter += 1

        # -----------------------------------------------------------------
        # Update step
        # -----------------------------------------------------------------
        if not mirror_mode:
            # --- Smoothed Adam (soft‑max surrogate) ---
            maxc = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc))
            weights = exp_scaled / exp_scaled.sum()

            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]

            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            it += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)

            lr = lr_start * (1.0 - prog) + lr_end * prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        else:
            # --- Exact entropy‑mirror descent on the max entry ---
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)

            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                # correct sub‑gradient (2·a_i on the diagonal)
                g[idx] = 2.0 * a[k - idx]

            # tiny regulariser that pushes the distribution a little towards uniformity
            stage_prog = (prog - mirror_start) / (1.0 - mirror_start) if prog > mirror_start else 0.0
            lam = 0.02 * (1.0 - stage_prog)               # linearly decaying
            if lam > 0:
                g += lam * a

            # step‑size schedule for the mirror phase
            eta_start, eta_end = 0.45, 0.02
            eta = eta_start * (1.0 - stage_prog) + eta_end * stage_prog

            a = a * np.exp(-eta * g)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

            # keep logits in sync (useful if we ever return to Adam)
            z = np.log(a + 1e-30)

        # -----------------------------------------------------------------
        # Stagnation handling – occasional jitter
        # -----------------------------------------------------------------
        if stagn_counter >= stagn_limit:
            if not mirror_mode:
                z += np.random.normal(scale=jitter_adam, size=n)
                # print(f"[fine] jitter (Adam) at it={it}")
            else:
                pert = np.random.normal(scale=jitter_mirror, size=n)
                a = np.maximum(a + pert, 0.0)
                a /= a.sum()
                z = np.log(a + 1e-30)
                # print(f"[fine] jitter (Mirror) at it={it}")
            stagn_counter = 0

    # -------------------------------------------------------------
    # Phase 3 – long polishing with exact mirror descent
    # -------------------------------------------------------------
    polished = _max_refine(best_a_fine, steps=8000, eta_start=0.45)
    obj_polish = _obj(polished)
    if obj_polish < best_obj_fine:
        best_a_fine = polished
        best_obj_fine = obj_polish
        # print(f"[polish] improved to O={best_obj_fine:.6f}")

    # -------------------------------------------------------------
    # Final evaluation (the wrapper will clamp and normalise again)
    # -------------------------------------------------------------
    final_seq = best_a_fine.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx O={best_obj_fine:.6f})")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
