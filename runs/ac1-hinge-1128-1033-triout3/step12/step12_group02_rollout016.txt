def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start=time.time()
    deadline = start + budget_s - 0.5
    n=20000
    a = np.full(n, 1.0/n)   # uniform
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(_full_convolution(a)) # since sum=1
    stagn_counter=0
    stagn_limit=5000
    it=0
    # Precompute constant for step size schedule
    max_iters_possible = 200000  # arbitrarily large; we'll cap by time
    while time.time() < deadline:
        # compute conv and objective
        conv = _full_convolution(a)
        max_b = conv.max()
        obj = 2.0 * n * max_b  # sum=1
        if obj < best_obj:
            best_obj=obj
            best_a=a.copy()
            stagn_counter=0
            # print update
        else:
            stagn_counter+=1
        # gradient for max entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi+1)
            g[idx] = 2.0 * a[k - idx]
        # step size schedule
        prog = it / max_iters_possible
        eta = 0.5 * (1 - prog) + 0.01 * prog  # from 0.5 to 0.01
        # update multiplicatively
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0/n)
        it+=1
        # stagnation escape
        if stagn_counter >= stagn_limit:
            # add jitter
            jitter = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            stagn_counter=0
        # time check break if time?

    # final polish optional: a few thousand mirror descent steps with smaller eta
    refined = _max_refine(best_a, steps=20000, eta_start=0.3)
    # evaluate refined obj
    conv_ref = _full_convolution(refined)
    obj_ref = 2.0 * n * conv_ref.max()
    if obj_ref < best_obj:
        best_obj=obj_ref
        best_a=refined

    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)

"""
Propose a step‑function that beats the previous 1.508 bound.

Idea
----
* The objective (with a normalised sequence, sum=1) is
      O = 2·n·max(conv(a,a)).
  Hence we only have to make the maximal entry of the self‑convolution
  as small as possible.

* Uniform weights are a stationary point of the exact sub‑gradient
  update (the gradient is everywhere equal), so we start from a **random**
  non‑uniform vector – this breaks the symmetry immediately.

* We run a large number of *entropy‑mirror descent* steps on the exact
  maximal convolution entry:
      a ← a·exp(‑η·g)   ,   a ← a / sum(a)
  where g_i = 2·a_{k‑i} for the index k attaining the current maximum.
  The step size η is decreased linearly from a relatively large value
  (0.5) to a small one (≈0.01).

* Several short random restarts are performed while time permits; the
  best vector seen is kept.

* Finally a longer polishing phase (20 000 mirror‑descent steps) is
  applied to the best candidate.

The whole routine respects the 1000 s budget and returns a plain list of
floats that can be passed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution via FFT (size = next power of two)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power of two for an efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max entry (polishing)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 20000, eta_start: float = 0.4) -> np.ndarray:
    """
    Perform many mirror‑descent updates on the exact maximal convolution
    entry.  ``a`` is assumed to be non‑negative and normalised to sum=1.
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g = np.zeros_like(a)
            g[idx] = 2.0 * a[k - idx]
        else:
            g = np.zeros_like(a)

        # linear decay of the step size
        eta = eta_start * (1.0 - t / steps) + 0.01 * (t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a[:] = 1.0 / n
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative floats (the step‑function heights) that
    minimise ``evaluate_sequence``.  The routine runs for up to
    ``budget_s`` seconds and uses random restarts + entropy‑mirror
    descent.
    """
    np.random.seed(seed)
    start = time.time()
    # keep ~0.5 s margin for the final evaluation call
    deadline = start + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a larger n gives a tighter theoretical bound.
    # --------------------------------------------------------------
    n = 20000

    # --------------------------------------------------------------
    # Helper to compute the objective (sum is normalised → O = 2·n·max)
    # --------------------------------------------------------------
    def objective(a_vec: np.ndarray) -> float:
        conv = _full_convolution(a_vec)
        return 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # Initialise best solution (use the previous global one if available)
    # --------------------------------------------------------------
    best_a = None
    best_obj = np.inf
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        try:
            old = np.asarray(old_seq, dtype=float)
            # normalise to sum=1
            s = old.sum()
            if s > 0.0:
                old = old / s
            else:
                old = np.full_like(old, 1.0 / old.shape[0])
            # if the old length differs, interpolate to our n

                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                old = np.interp(x_new, x_old, old)
            cand_obj = objective(old)
            best_a = old.copy()
            best_obj = cand_obj
            print(f"[init] used previous solution, O={best_obj:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    restart = 0
    # we keep a small safety buffer for the final polishing stage
    polishing_time = 8.0
    while time.time() < deadline - polishing_time:
        restart += 1
        # start from a fresh random Dirichlet vector
        a = np.random.rand(n)
        a = a / a.sum()

        max_iters = 3000
        stagn_counter = 0

        for it in range(max_iters):

                a /= s

                a /= a.sum()
                stagn_counter = 0

        print(f"[restart {restart}] finished (best O so far = {best_obj:.6f})")

    # --------------------------------------------------------------
    # Polishing – many more mirror‑descent steps on the best vector
    # --------------------------------------------------------------
    if best_a is not None:

        refined_obj = objective(refined)

            best_a = refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # Final conversion and safety evaluation
    # --------------------------------------------------------------

    try:

    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
