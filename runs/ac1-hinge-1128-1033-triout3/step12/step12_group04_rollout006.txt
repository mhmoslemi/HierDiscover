"""
Propose a step‑function minimizing the L∞ self‑convolution bound.

The algorithm is a smoothed‐max gradient descent on the objective

    f(a) = (1/p)·log Σ_k exp(p·(a*a)_k)        (p → ∞ gives max)

with projection onto the probability simplex (Σ a_i = 1, a_i ≥ 0).
A small softmax parameter *p* gives a smooth landscape; it is increased
gradually (continuation).  The gradient of f is

    ∇f_i = 2·Σ_k w_k·a_{k-i},
    w_k = exp(p·c_k) / Σ_j exp(p·c_j) ,  c = a*a .

The convolution required for the gradient is computed by FFT (O(n log n)).
An Adam‑style adaptive step is used, and the whole routine is run for
several random restarts (including a warm start from a previously saved
solution if the global name ``height_sequence_1`` exists).  The best
sequence found before the time budget expires is returned and evaluated
once with ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level as required
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    # Algorithm 1 from https://arxiv.org/abs/1101.6081
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of real vectors x and y using FFT of given size."""
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: len(x) + len(y) - 1]
    return conv


def _full_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Return a*a (linear convolution) using the pre‑allocated FFT size."""
    return _fft_convolve(a, a, size_fft)


def _conv_with_weights(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """Return a * w_rev where w_rev = w[::-1] (linear convolution)."""
    w_rev = w[::-1]
    return _fft_convolve(a, w_rev, size_fft)

# Main search routine
# ----------------------------------------------------------------------

    budget_s: int = 1000,
    **kwargs,

    Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.
    The algorithm uses a smoothed‑max (log‑sum‑exp) objective, Adam updates,
    simplex projection, and several random restarts.  The best sequence found
    before the time budget expires is returned.
    """

    start_time = time.time()
    # keep a tiny safety margin for the final evaluation
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem dimension – a compromise between speed and expressive power.
    # ------------------------------------------------------------------
    n = 2000                     # you may change this (must be >0)
    size_fft = _next_pow2(3 * n - 2)   # large enough for a*w_rev

    # ------------------------------------------------------------------
    # Load a warm start if the previous solution is available.
    # ------------------------------------------------------------------
    warm_start = None

        if s < 1e-12:

            m = beta1 * m + (1.0 - beta1) * g

            p = min(p * p_mul, p_max)

    # Final secure evaluation (single call)

        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
