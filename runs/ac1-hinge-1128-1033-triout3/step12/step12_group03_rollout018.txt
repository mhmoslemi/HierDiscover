def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # For j = k - i, range ... compute

            data.append(f[i])  # coefficient f_i for g_j
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n))

"""
Hybrid optimiser for the self‑convolution max problem.

The routine works in three stages:

1️⃣  A low‑dimensional (n≈2000) linear‑programming stage.
    Starting from a random probability vector we repeatedly solve
    the LP   maximise Σ g_i   subject to   (f * g) ≤ M,
    where M = max(f*f).  The LP produces an extreme feasible point g.
    A convex combination f←(1‑t)f+t·g (with a line‑search on t) reduces M.
    A few outer LP iterations quickly move the distribution away from the
    uniform plateau.

2️⃣  Upsample the small‑scale solution to the target size (n≈8000)
    by linear interpolation and renormalisation.

3️⃣  A high‑dimensional smooth‑max Adam optimisation followed by a cheap
    projected sub‑gradient refinement (identical in spirit to the reference
    implementation, but with a longer schedule).  This squeezes out the
    remaining slack and drives the evaluation score below the target.

All steps respect the 1000 s budget and the final sequence is returned as
a plain Python list of non‑negative floats.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution x*x (full length) via FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of weight vector w (length 2n‑1) with a.
    Returns vector of length n: (w ⋆ a)[i] = Σ_k w_k a_{k‑i}.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1          # = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(self‑convolution) for a probability vector a."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that A @ g = f*g (linear convolution).
    A[k, j] = f_{k‑j}  if 0 ≤ k‑j < n, else 0.
    """
    n = f.shape[0]
    offsets = np.arange(n)                     # 0 … n‑1
    data = [np.full(n, f[d]) for d in offsets]  # each diagonal is constant f[d]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise `evaluate_sequence`.  The routine runs a low‑dimensional LP
    refinement, up‑samples to the final size and finishes with a
    smooth‑max Adam optimisation plus a projected sub‑gradient sweep.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin for the final call

    # --------------------------------------------------------------
    # 1️⃣  Low‑dimensional LP stage (n_small ≈ 2000)
    # --------------------------------------------------------------
    n_small = 2000
    best_small = None
    best_val_small = np.inf

    # Use a few random restarts to avoid poor local optima.
    restarts = 3
    for rs in range(restarts):
        if time.time() >= deadline:
            break
        # initialise a random probability vector (Dirichlet) – can be uniform
        a = np.random.dirichlet(alpha=np.ones(n_small))
        # ensure non‑negativity and normalisation (dirichlet already does it)
        a = _project_simplex(a)

        # Compute initial bound
        conv = _full_convolution(a)
        M = conv.max()
        val = 2.0 * n_small * M
        print(f"[LP init | restart {rs+1}] objective = {val:.9f}")

        # Outer LP iteration loop
        for outer in range(7):                     # a handful of LP refinements
            if time.time() >= deadline:
                break

            # 1) build convolution matrix for current a
            A = _build_conv_matrix(a)

            # 2) solve LP: maximise Σ g_i  (i.e. minimise –Σ g_i)
            c = -np.ones(n_small)                  # minimise –sum(g)
            b_ub = np.full(2 * n_small - 1, M)     # (f * g) ≤ M
            bounds = [(0.0, None)] * n_small

            res = opt.linprog(c, A_ub=A, b_ub=b_ub,
                              bounds=bounds, method='highs')
            if not res.success:
                print("[LP] linprog failed, breaking")
                break
            g = res.x
            sum_g = g.sum()
            if sum_g < 1e-12:
                print("[LP] trivial solution, stopping")
                break

            # 3) scale g back to a probability vector
            g = g / sum_g

            # 4) line‑search for mixing weight t that actually reduces M
            t = 0.5
            improved = False
            while t > 1e-4:
                a_new = (1.0 - t) * a + t * g
                a_new = _project_simplex(a_new)   # protect against numerical drift
                M_new = _full_convolution(a_new).max()
                if M_new < M - 1e-14:
                    a = a_new
                    M = M_new
                    val = 2.0 * n_small * M
                    improved = True
                    print(f"[LP outer {outer+1}] mix t={t:.4f} → objective {val:.9f}")
                    break
                t *= 0.5

            if not improved:
                # no improvement even with tiny t → stop this outer loop
                print("[LP] no improvement found, break outer")
                break

        # keep the best small‑scale solution seen so far
        if val < best_val_small - 1e-12:
            best_val_small = val
            best_small = a.copy()
            print(f"[LP] new global best small objective = {best_val_small:.9f}")

    # ------------------------------------------------------------------
    # 2️⃣  Upsample to the target size (n_large)
    # ------------------------------------------------------------------
    if best_small is None:
        # fallback – uniform distribution if something went wrong
        n_large = 8000
        best_a = np.full(n_large, 1.0 / n_large)
        print("[fallback] using uniform large sequence")
    else:
        n_large = 8000
        old_x = np.linspace(0.0, 1.0, best_small.size)
        new_x = np.linspace(0.0, 1.0, n_large)
        best_a = np.interp(new_x, old_x, best_small)
        best_a = np.maximum(best_a, 0.0)
        best_a = _project_simplex(best_a)
        print(f"[upsample] up‑sampled objective = {_objective(best_a, n_large):.9f}")

    # --------------------------------------------------------------
    # 3️⃣  Large‑scale Adam optimisation (smooth‑max)
    # --------------------------------------------------------------
    n = n_large
    # initialise logits from the current best
    eps = 1e-12
    z = np.log(best_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Adam hyper‑parameters
    adam_lr = 0.05
    b1, b2 = 0.9, 0.999
    adam_eps = 1e-8
    lambda_start = 5.0
    lambda_end = 5000.0
    max_steps = 30000

    global_best_a = best_a.copy()
    global_best_val = _objective(global_best_a, n)

    t_iter = 0
    while time.time() < deadline and t_iter < max_steps:
        t_iter += 1

        # soft‑max probabilities
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # evaluate convolution and objective
        conv = _full_convolution(a)
        conv_max = conv.max()
        cur_val = 2.0 * n * conv_max

        if cur_val < global_best_val - 1e-12:
            global_best_val = cur_val
            global_best_a = a.copy()
            # occasional progress report
            if t_iter % 2000 == 0:
                print(f"[Adam] iter {t_iter:06d} new best = {global_best_val:.9f}")

        # temperature schedule (linear)
        prog = t_iter / max_steps
        lam = lambda_start * (1.0 - prog) + lambda_end * prog

        # smooth‑max weights
        if lam > 2500.0:
            idx = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[idx] = 1.0
        else:
            exp_vals = np.exp(lam * (conv - conv_max))
            w = exp_vals / exp_vals.sum()

        # gradient of smooth‑max w.r.t. a  ( 2 * cross‑correlation )
        g_raw = _cross_corr(w, a)
        g = 2.0 * g_raw

        # back‑prop through the soft‑max (logits)
        dot_ag = np.dot(a, g)
        grad_z = a * (g - dot_ag)

        # Adam update
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z ** 2)
        m_hat = m / (1.0 - b1 ** t_iter)
        v_hat = v / (1.0 - b2 ** t_iter)
        z -= adam_lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # very occasional random jitter to escape plateaus
        if t_iter % 8000 == 0:
            z += np.random.normal(scale=0.02, size=n)
    print(f"[Adam] finished after {t_iter} steps, best value = {global_best_val:.9f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    a = global_best_a.copy()
    obj = global_best_val
    step = 0.02

    for sub_it in range(1500):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the true max (vector g_i = 2·a_{kmax‑i})
        g = np.zeros_like(a)
        if kmax < n:
            g[:kmax + 1] = 2.0 * a[kmax::-1]
        else:
            offset = kmax - (n - 1)
            L = n - offset
            g[offset:] = 2.0 * a[:L][::-1]

        a_trial = _project_simplex(a - step * g)
        conv_trial = _full_convolution(a_trial)
        obj_trial = 2.0 * n * conv_trial.max()

        if obj_trial < obj - 1e-13:
            a = a_trial
            obj = obj_trial
            step = min(step * 1.1, 0.5)
            # progress report
            if sub_it % 100 == 0:
                print(f"[subgrad] iter {sub_it:04d} improved to {obj:.9f}")
        else:
            step *= 0.7

    # --------------------------------------------------------------
    # 5️⃣  Final secure evaluation (the only place we call evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return a.tolist()
