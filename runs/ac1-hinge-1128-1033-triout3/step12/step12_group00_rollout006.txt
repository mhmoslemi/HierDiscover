    # compute grad as before
    n = a.shape[0]
    b = conv_full(a)  # length 2n-1
    max_b = b.max()
    k_maxes = np.where(b == max_b)[0]

        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] += 2.0 * a[idx[mask]]
    # multiplicative update
    new_a = a * np.exp(-step * grad)

def _lbfgs_optimize(a0, n, deadline, start_time, best_a, best_val, eps=1e-6, max_iter=1500):
    # use softmax parameterization
    x0 = np.log(a0 + 1e-12)
    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val
    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)
    time_left = deadline - time.time()
    if time_left < 1.0:
        return best_a, best_val
    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val
    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()

"""
Hybrid optimiser for the step‑function attack.

Algorithm
---------
1. Initialise from the previously discovered sequence (global
   ``height_sequence_1``) or fall back to a uniform distribution.

2. L‑BFGS‑B on a smooth approximation of the max‑convolution.
   The smooth function is a log‑sum‑exp with a decreasing ε.
   This quickly drives the score near the best known value (~1.508).

3. Projected sub‑gradient descent on the exact objective
   2·n·max(conv(a,a)).  The sub‑gradient is computed analytically.
   A linear step‑size schedule is used, together with occasional
   random jitter to escape plateaus.

4. Final refinement by random convex mixing with a fresh random
   vector (simple uniform Dirichlet).  The mixing weight decays
   over time.

5. The best sequence found is evaluated with the provided
   ``evaluate_sequence`` function and returned as a plain Python
   list of floats (sum is normalised to 1, all entries ≥0).

The whole routine respects the 1000 s budget and prints progress
information.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (self‑convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of ``x`` with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Convolution of two arbitrary vectors (used for the smooth‑max gradient)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. ``a``.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2·n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2·n‑1,)

    # gradient: convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2·n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries that correspond to forward part
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → probability simplex)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_val: float,
    eps: float = 1e-6,
    max_iter: int = 500,
) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on the smoothed objective."""
    x0 = np.log(a0 + 1e-12)

    # objective for minimise()
    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    # gradient for minimise()
    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    # early exit if no time left
    if deadline - time.time() < 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:    # pragma: no cover
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Project onto the probability simplex (clipping + renormalise)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return a non‑negative vector with Σ=1 (simple clipping + renorm)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # fallback: uniform distribution
    n = v.shape[0]
    return np.full_like(v, 1.0 / n)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    indices = np.arange(n)

    while time.time() < deadline:
        now = time.time()
        progress = (now - start_time) / (deadline - start_time)
        # linear step‑size schedule (more aggressive early on)
        step = 0.5 * (1.0 - progress) + 1e-5 * progress

        # compute sub‑gradient for the current a
        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - indices
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # projected update
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val + 1e-12 < best_val:
            best_val, best_a = new_val, a_new.copy()
            a = a_new
            print(f"[SubG]   new best = {best_val:.6f}")
        else:
            a = a_new

        # occasional random jitter to escape tiny plateaus
        if rng.random() < 0.015:
            jitter = rng.normal(scale=0.012, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Random mixing refinement (convex combination with a fresh random vector)
# ----------------------------------------------------------------------
def _random_mixing_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Try convex combinations  t·a + (1‑t)·u where u is a fresh random vector.
    The mixing weight ``t`` is gradually driven to 1.
    """
    best_a = a0.copy()
    best_val = _objective(best_a)

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        t = max(0.0, 1.0 - prog)          # weight for the current best
        # generate a fresh random vector (uniform Dirichlet)
        u = rng.random(n)
        u /= u.sum()
        cand = t * best_a + (1.0 - t) * u
        cand = _project_simplex(cand)
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Mix]   new best = {best_val:.6f}")

    return best_a, best_val

        print("[Final] evaluate_sequence raised:", exc)

    # Ensure a plain list of Python floats is returned
    return best_a.tolist()
