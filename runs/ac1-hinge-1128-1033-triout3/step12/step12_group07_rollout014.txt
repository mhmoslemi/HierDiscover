def _diff_evolution_mix(n, deadline, start_time, best_val, best_a):
    import scipy.optimize as opt

    def decode(param):
        # param is array length 9
        alphas = np.exp(param[0:6:2]) # logα
        betas = np.exp(param[1:6:2]) # logβ
        raw_weights = param[6:9]
        w = np.exp(raw_weights)
        # compute each component
        i = np.arange(1, n+1, dtype=np.float64)
        comps = []
        for a,b in zip(alphas, betas):
            # avoid overflow from extreme exponents; if a,b produce huge or zero values, handle
            comp = (i ** (a - 1.0)) * ((n - i + 1) ** (b - 1.0))
            comps.append(comp)
        comps = np.stack(comps)  # shape K,n
        # Weighted sum
        weighted = (w[:, None] * comps).sum(axis=0)
        total = weighted.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = weighted / total
        return a

    def obj(param):
        a = decode(param)
        val = _objective(a)

def _de_mix_optimize(best_a, best_val, n, deadline, start_time, rng):
    import scipy.optimize as opt
    K = 3
    dim = 2*K + K  # logα and logβ (2 each) + raw weights (K) = 3K
    # define decode as earlier
    def decode(params):
        alphas = np.exp(params[0:2*K:2])
        betas = np.exp(params[1:2*K:2])
        raw_weights = params[2*K:]
        weights = np.exp(raw_weights)
        # compute components
        i = np.arange(1, n+1, dtype=np.float64)
        # compute each component in log domain to avoid overflow
        comps = []
        for a,b in zip(alphas, betas):
            log_comp = (a-1.0)*np.log(i) + (b-1.0)*np.log(n - i + 1.0)
            # subtract max for stability
            log_comp -= np.max(log_comp)
            comp = np.exp(log_comp)
            comps.append(comp)
        comps = np.stack(comps) # shape K,n
        weighted = (weights[:, None] * comps).sum(axis=0)
        total = weighted.sum()
        if total == 0.0:
            a_vec = np.full(n, 1.0 / n)
        else:
            a_vec = weighted / total
        return a_vec

    # objective function for DE.
    def obj(params):
        a = decode(params)
        val = _objective(a)
        nonlocal best_val, best_a
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[DE] new best = {best_val:.6f}")
        # time check
        if time.time() >= deadline:
            raise RuntimeError("Timeout")
        return val

        pass

"""
Propose a step‑function that minimizes the evaluation function.

Strategy
--------
1. Initialise with a uniform vector and a quick random‑Beta sweep.
2. Refine the shape using a *low‑dimensional* parameterisation:
   a mixture of up to three Beta distributions.  The 9 parameters
   (log‑α, log‑β for each component + log‑weight) are optimised by
   differential evolution (global search).  The best mixture is kept.
3. Fine‑tune the full‑dimensional vector with a series of L‑BFGS‑B
   optimisations on a smooth approximation of the max‑convolution.
   The smoothness parameter ε is decreased successively.
4. Perform an aggressive hill‑climber (source→target transfers) and
   a final simulated‑annealing phase to polish any remaining spikes.
5. Return the best sequence found before the 1000 s wall‑clock budget
   expires.  All intermediate steps respect the constraints of
   ``evaluate_sequence`` (non‑negative entries, sum > 0).
"""

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Convolution primitives (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # stabilised log‑sum‑exp
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # already contains factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Candidate families (used for quick initialisation)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Beta‑shaped vector on a log‑scale."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    """Deterministic Beta candidate for given log‑parameters."""
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray, n: int, cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Differential‑Evolution optimisation of a mixture of up‑to‑3 Betas
# ----------------------------------------------------------------------
def _de_mix_optimize(best_a: np.ndarray,
                     best_val: float,
                     n: int,
                     deadline: float,
                     start_time: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Global search over a 3‑component Beta mixture using DE."""
    K = 3
    dim = 2 * K + K                     # logα, logβ for each component + log‑weight

    # ------------------------------------------------------------------
    # Decode a parameter vector → full‑length probability vector
    # ------------------------------------------------------------------
    def _decode(params: np.ndarray) -> np.ndarray:
        alphas = np.exp(params[0:2 * K:2])
        betas = np.exp(params[1:2 * K:2])
        raw_w = params[2 * K:]
        w = np.exp(raw_w)               # enforce positivity
        i = np.arange(1, n + 1, dtype=np.float64)

        comps = []
        for a, b in zip(alphas, betas):
            # compute in log‑space for stability
            log_comp = (a - 1.0) * np.log(i) + (b - 1.0) * np.log(n - i + 1.0)
            log_comp -= np.max(log_comp)          # shift
            comp = np.exp(log_comp)
            comps.append(comp)
        comps = np.stack(comps)           # shape (K, n)
        weighted = (w[:, None] * comps).sum(axis=0)
        total = weighted.sum()
        if total == 0.0:
            a_vec = np.full(n, 1.0 / n)
        else:
            a_vec = weighted / total
        return a_vec

    # ------------------------------------------------------------------
    # Objective for DE (exact objective, but we also keep track of best)
    # ------------------------------------------------------------------
    def _obj(params: np.ndarray) -> float:
        nonlocal best_a, best_val
        if time.time() >= deadline:
            raise RuntimeError("timeout")
        a_vec = _decode(params)
        val = _objective(a_vec)
        if val < best_val:
            best_val = val
            best_a = a_vec.copy()
            print(f"[DE] new best = {best_val:.6f}")
        return val

    # bounds: logα, logβ ∈ [−7.5, 2]; log‑weight ∈ [−5, 5]
    bounds = [(-7.5, 2.0)] * (2 * K) + [(-5.0, 5.0)] * K

    # reasonable DE settings – cheap enough to run many generations
    max_iter = 30
    popsize = 12
    try:
        opt.differential_evolution(
            _obj,
            bounds,
            seed=int(rng.integers(2 ** 31 - 1)),
            maxiter=max_iter,
            popsize=popsize,
            polish=False,
            updating='deferred',
            init='latinhypercube',
            disp=False,
        )
    except RuntimeError:
        # timeout – just return the current best
        pass
    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    eps: float,
                    max_iter: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float) -> tuple[np.ndarray, float]:
    """L‑BFGS‑B on the smooth max with a fixed ε."""
    w0 = np.log(a0 + 1e-12)               # work in log‑softmax space

    # ------------------------------------------------------------------
    # Objective (smooth) and gradient w.r.t. w (logits)
    # ------------------------------------------------------------------
    def _fun(w):
        nonlocal best_a, best_val
        if time.time() >= deadline:
            raise RuntimeError("timeout")
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)   # grad contains factor 2·n
        L = 2.0 * n * smooth_val

        # update global best using the *exact* objective (no smoothing)
        exact = _objective(a)
        if exact < best_val:
            best_val = exact
            best_a = a.copy()
            print(f"[LBFGS] new best = {best_val:.6f}")
        return L

    def _jac(w):
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        _, grad_a = _smooth_max_and_grad(a, eps)   # already includes 2·n
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)
        return grad_w

    try:
        opt.minimize(
            _fun,
            w0,
            jac=_jac,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'disp': False}
        )
    except RuntimeError:
        # deadline hit – simply return current best
        pass
    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (source→target moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """Local search with simulated‑annealing acceptance."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.9, 0.001
    temp_start, temp_end = 0.03, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 40000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # fast update of the convolution
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Final large‑move simulated annealing
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple annealing with relatively large source→target transfers."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)            # linear cooling
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # occasional jitter to retain diversity
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is positive and (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 8000

    # ------------------- 0️⃣  initialise -------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # try to reuse a previously discovered solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0, val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # ----- random Beta draws (less than before) -----
    for _ in range(2000):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- systematic Beta grid (coarse) -----
    for lp in np.linspace(-7.5, 2.0, 9):
        for lq in np.linspace(-7.5, 2.0, 9):
            cand = _beta_grid_candidate(n, lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ----- Dirichlet draws for a few concentrations -----
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    print(f"[Init] best after quick init = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Global search on a low‑dimensional Beta mixture
    # --------------------------------------------------------------
    best_a, best_val = _de_mix_optimize(
        best_a, best_val, n, deadline, start_time, rng
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase B – L‑BFGS‑B refinement on a smooth max (decreasing ε)
    # --------------------------------------------------------------
    eps_schedule = [0.5, 0.1, 0.02, 0.005, 0.001]
    for eps in eps_schedule:
        if time.time() >= deadline:
            break
        # allocate a modest number of LBFGS iterations per ε
        max_iter = 250
        best_a, best_val = _lbfgs_optimize(
            best_a, n, eps, max_iter, deadline, start_time, best_a, best_val
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase C – Aggressive hill‑climbing
    # --------------------------------------------------------------
    best_a, best_val = _hillclimb(
        best_a, n, deadline, start_time, best_a, best_val
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase D – Final simulated annealing with larger moves
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_anneal(
            best_a, n, deadline, start_time, best_a, best_val, rng
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
