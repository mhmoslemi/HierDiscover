def _optimize_initial(a0, n, start_time, end_time):
    # Convert to logs:
    eps=1e-30
    z = np.log(a0 + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr_start, lr_end = 0.25, 0.005
    max_alpha = 350.0
    switch_alpha = 150.0
    stagn_limit = 2000
    jitter_scale_adam = 0.03
    jitter_scale_mirror = 0.008
    best_a = None
    best_obj = np.inf
    stagn_counter = 0
    it = 0
    mirror_mode = False
    switch_prog = None

    while time.time() < end_time:
        now = time.time()
        prog = (now - start_time) / (end_time - start_time)
        alpha = 1.0 + (max_alpha - 1.0) * (prog ** 1.5)

        if not mirror_mode and alpha >= switch_alpha:
            mirror_mode = True
            switch_prog = prog
            # compute a from z for logs
            a = np.exp(z)
            a = a / a.sum()

        if not mirror_mode:
            a = np.exp(z)
            a = a / a.sum()
        else:
            # keep a from previous iteration; but if mirror_mode, we will have a from earlier iteration after updating a using mirror update; so we keep a variable defined outside loop perhaps.
            pass

        # compute conv
        conv = _full_convolution(a)
        maxb = conv.max()
        obj = 2.0 * n * maxb  # unit-sum objective
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagn_counter = 0
        else:
            stagn_counter += 1

        if not mirror_mode:
            # compute gradient as previous code
            maxc = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc))
            sumexp = exp_scaled.sum()
            weights = exp_scaled / sumexp
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            g_a = 4.0 * n * conv_rev[n-1:n-1+n]  # length n
            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            lr = lr_start * (1 - prog) + lr_end * prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        else:
            # entropy mirror descent step
            k = int(conv.argmax())
            lo = max(0, k - (n-1))
            hi = min(k, n-1)
            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi+1, dtype=int)
                g[idx] = 2.0 * a[k - idx]
            # schedule eta
            stage_prog = (prog - switch_prog) / (1 - switch_prog) if switch_prog else prog
            eta_start, eta_end = 0.5, 0.01
            eta = eta_start * (1 - stage_prog) + eta_end * stage_prog
            a = a * np.exp(-eta * g)
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else np.full_like(a, 1.0 / n)
            it += 1

        # stagnation jitter
        if stagn_counter >= stagn_limit:
            if not mirror_mode:
                z += np.random.normal(scale=jitter_scale_adam, size=n)
                print(f"[restart] Adam jitter at it={it}")
            else:
                perturb = np.random.normal(scale=jitter_scale_mirror, size=n)
                a = np.maximum(a + perturb, 0.0)
                a_sum = a.sum()
                a = a / a_sum if a_sum > 0 else np.full_like(a, 1.0 / n)
                print(f"[restart] Mirror jitter at it={it}")
            stagn_counter = 0

    # After loop, if best_a not None, refine via entropy mirror for many steps
    if best_a is None:
        best_a = a0.copy()
    refined = _entropy_mirror_refine(best_a, steps=20000, eta_start=0.45)
    # compute refined objective
    conv_ref = _full_convolution(refined)
    obj_ref = 2.0 * n * conv_ref.max()
    if obj_ref < best_obj:
        best_obj = obj_ref
        best_a = refined

    return best_a, best_obj

def _euclidean_subgrad_refine(a, steps, step_start):
    n = a.shape[0]
    a = a.copy()
    step = step_start
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi+1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # step schedule linearly decreasing
        cur_step = step_start * (1 - t / steps)
        a = a - cur_step * g
        # project onto simplex
        a = _project_onto_simplex(a)
    return a

def _project_onto_simplex(v):
    # ensure sum to 1, nonnegative
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative? return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Note: we treat a sum as 1. If a gets zero sum after subtracting g, we project again.

"""
Propose a step‑function (non‑negative heights) that minimizes
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))²

Algorithm
---------

* **Multiple restarts** – the total 1000 s budget is split into a few
  independent runs, each starting from a different sensible distribution
  (previous solution, smooth Beta, random Dirichlet).  The best result
  over all restarts is kept.

* **Stage 1 – smoothed Adam** – while the temperature α is small we
  minimise a soft‑max surrogate of the max‑convolution.  Positivity is
  enforced by optimising in the log‑domain and a standard Adam update
  is applied.

* **Stage 2 – entropy‑mirror descent** – once α becomes large the
  surrogate is close to the true max.  We switch to exact sub‑gradient
  descent on the maximal convolution entry using the KL‑mirror update
  ``a ← a·exp(‑η g)`` and re‑normalise to the simplex.

* **Polishing** – after the main loop a long entropy‑mirror descent
  (up to the remaining time) is performed, followed by a Euclidean
  projected sub‑gradient refinement.

* **Book‑keeping** – the best vector (unit‑sum) seen so far is tracked
  and finally scaled to the required sum √(2n) before returning.

The routine respects the 1000 s budget, prints progress, and returns a
plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two different vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max entry (polishing)
# ----------------------------------------------------------------------
def _entropy_mirror_refine(a: np.ndarray,
                           steps: int = 5000,
                           eta_start: float = 0.5,
                           eta_end: float = 0.01) -> np.ndarray:
    """Mirror descent (KL) on the exact max‑entry of a*a."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        eta = eta_start * (1.0 - t / steps) + eta_end * (t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Euclidean projected sub‑gradient descent (polishing)
# ----------------------------------------------------------------------
def _euclidean_subgrad_refine(a: np.ndarray,
                              steps: int = 20000,
                              step_start: float = 0.5) -> np.ndarray:
    """Projected sub‑gradient descent on max‑convolution entry."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        step = step_start * (1.0 - t / steps) + 0.001 * (t / steps)
        a = a - step * g
        a = _project_onto_simplex(a)
    return a


# ----------------------------------------------------------------------
# Core optimiser for a single restart
# ----------------------------------------------------------------------
def _optimize_one_restart(a0: np.ndarray,
                          n: int,
                          start_time: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Run the two‑stage optimiser (Adam + entropy mirror) on a single
    start vector a0 (unit‑sum).  Returns the best unit‑sum vector seen
    and its objective value (2·n·max(conv)).
    """
    eps = 1e-30
    z = np.log(a0 + eps)                     # logits for Adam
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # schedules
    lr_start, lr_end = 0.25, 0.008
    max_alpha = 350.0
    switch_alpha = 150.0

    stagn_limit = 3000
    jitter_scale_adam = 0.03
    jitter_scale_mirror = 0.008

    best_a = None
    best_obj = np.inf
    stagn_counter = 0
    it = 0
    mirror_mode = False
    switch_prog = None

    a = None  # will be materialised when needed

    while time.time() < deadline:
        now = time.time()
        prog = (now - start_time) / (deadline - start_time)   # 0 → 1
        alpha = 1.0 + (max_alpha - 1.0) * (prog ** 1.5)

        # ------------------------------------------------------------------
        # Switch to exact mirror descent when temperature is high enough
        # ------------------------------------------------------------------
        if not mirror_mode and alpha >= switch_alpha:
            mirror_mode = True
            switch_prog = prog
            a = np.exp(z)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)
            print(f"[mode] switch to mirror (t={now - start_time:.1f}s)")

        # ------------------------------------------------------------------
        # Current probability vector
        # ------------------------------------------------------------------
        if not mirror_mode:
            a = np.exp(z)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        # ------------------------------------------------------------------
        # Full self‑convolution and objective on the unit‑sum scale
        # ------------------------------------------------------------------
        conv = _full_convolution(a)
        maxb = conv.max()
        obj = 2.0 * n * maxb               # = evaluate_sequence if sum(a)=1

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagn_counter = 0
            print(f"[iter] new best O={best_obj:.6f} it={it} t={now-start_time:.1f}s")
        else:
            stagn_counter += 1

        # ------------------------------------------------------------------
        # Gradient / update
        # ------------------------------------------------------------------
        if not mirror_mode:
            # ---- Stage 1: smoothed Adam on the soft‑max surrogate ----
            maxc = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc))
            sumexp = exp_scaled.sum()
            weights = exp_scaled / sumexp                     # Σ w = 1

            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]        # length n

            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = lr_start * (1.0 - prog) + lr_end * prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        else:
            # ---- Stage 2: exact entropy‑mirror descent on the max entry ----
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)

            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                g[idx] = 2.0 * a[k - idx]

            stage_prog = (prog - switch_prog) / (1.0 - switch_prog) if switch_prog else prog
            eta_start, eta_end = 0.5, 0.01
            eta = eta_start * (1.0 - stage_prog) + eta_end * stage_prog

            a = a * np.exp(-eta * g)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)
            it += 1

        # ------------------------------------------------------------------
        # Jitter to escape stagnation
        # ------------------------------------------------------------------
        if stagn_counter >= stagn_limit:
            if not mirror_mode:
                z += np.random.normal(scale=jitter_scale_adam, size=n)
                print(f"[reset] Adam jitter at it={it}")
            else:
                perturb = np.random.normal(scale=jitter_scale_mirror, size=n)
                a = np.maximum(a + perturb, 0.0)
                s = a.sum()
                a = a / s if s > 0 else np.full_like(a, 1.0 / n)
                print(f"[reset] Mirror jitter at it={it}")
            stagn_counter = 0

    # ------------------------------------------------------------------
    # Final polishing on the best vector found (long entropy mirror)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = a0.copy()
    # allocate remaining time for polishing (if any)
    polish_deadline = deadline
    remaining = polish_deadline - time.time()
    if remaining > 0.1:
        # Number of polishing steps roughly proportional to remaining time
        est_step = 0.0015                     # rough seconds per iteration
        steps = max(5000, int(remaining / est_step))
        best_a = _entropy_mirror_refine(best_a,
                                         steps=steps,
                                         eta_start=0.45,
                                         eta_end=0.01)

        conv = _full_convolution(best_a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            print(f"[polish] improved to O={best_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects a 1000 s wall‑clock budget, uses several
    random restarts, Adam + mirror descent and a final long polishing.
    Returns a list of floats (the raw heights).
    """
    np.random.seed(seed)
    start = time.time()
    # keep a tiny margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between flexibility
    # and speed.
    # ------------------------------------------------------------------
    n = 8000
    S = np.sqrt(2.0 * n)          # required sum of the raw heights

    # ------------------------------------------------------------------
    # Prepare several restart initialisations
    # ------------------------------------------------------------------
    restart_fns = []

    # 0 – try to reuse the previous best if it exists
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        def init_from_old():
            old = np.asarray(old_seq, dtype=float)
            old_len = old.shape[0]
            if old_len != n:
                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old_len)
                a = np.interp(x_new, x_old, old)
            else:
                a = old.copy()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s
            else:
                a = np.full(n, 1.0 / n)
            return a
        restart_fns.append(init_from_old)
        print("[init] added previous solution as a restart")
    else:
        print("[init] no previous solution available")

    # 1 – smooth Beta(2,2) shape
    def init_beta():
        i = np.arange(1, n + 1, dtype=np.float64)
        p = q = 2.0
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        a = a / a.sum()
        return a
    restart_fns.append(init_beta)

    # 2 … 4 – random Dirichlet‑like starts (different concentrations)
    for _ in range(3):
        def make_random():
            # draw a concentration parameter in [0.3, 5]
            conc = np.random.uniform(0.3, 5.0)
            a = np.random.gamma(shape=conc, scale=1.0, size=n)
            # bias the shape a little (optional)
            beta = np.random.uniform(0.0, 3.0)
            x = np.linspace(0.0, 1.0, n)
            a = a * (x ** beta + (1 - x) ** beta + 0.1)
            a = a / a.sum()
            return a
        restart_fns.append(make_random)

    # ------------------------------------------------------------------
    # Run the restarts, sharing the total time budget
    # ------------------------------------------------------------------
    best_global_a = None
    best_global_obj = np.inf
    num_restarts = len(restart_fns)
    for idx, init_fn in enumerate(restart_fns):
        now = time.time()
        if now >= deadline:
            break
        # Allocate a roughly equal slice of the remaining time
        remaining_total = deadline - now
        remaining_restarts = num_restarts - idx
        deadline_this = now + remaining_total / remaining_restarts

        print(f"[restart {idx}] time slice {deadline_this - now:.1f}s")
        a0 = init_fn()
        best_a, best_obj = _optimize_one_restart(a0, n, now, deadline_this)

        if best_obj < best_global_obj:
            best_global_obj = best_obj
            best_global_a = best_a.copy()
            print(f"[global] new overall best O={best_global_obj:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – long entropy mirror + Euclidean refinement
    # ------------------------------------------------------------------
    if best_global_a is None:
        best_global_a = init_beta()

    # Use any remaining time for a long entropy‑mirror refinement
    time_left = deadline - time.time()
    if time_left > 0.2:
        # estimate steps (≈0.0015 s per iteration)
        est_step = 0.0015
        steps = max(5000, int(time_left / est_step))
        best_global_a = _entropy_mirror_refine(best_global_a,
                                               steps=steps,
                                               eta_start=0.45,
                                               eta_end=0.01)
        conv = _full_convolution(best_global_a)
        obj = 2.0 * n * conv.max()
        if obj < best_global_obj:
            best_global_obj = obj
            print(f"[final polish] entropy improved to O={best_global_obj:.6f}")

    # A short Euclidean sub‑gradient polish (fixed budget ~20 s)
    time_left = deadline - time.time()
    if time_left > 5.0:
        steps = int(time_left / 0.001)  # ~1 ms per iteration
        steps = max(2000, steps)
        best_global_a = _euclidean_subgrad_refine(best_global_a,
                                                  steps=steps,
                                                  step_start=0.5)
        conv = _full_convolution(best_global_a)
        obj = 2.0 * n * conv.max()
        if obj < best_global_obj:
            best_global_obj = obj
            print(f"[final polish] Euclidean improved to O={best_global_obj:.6f}")

    # ------------------------------------------------------------------
    # Scale to the required sum √(2n) and return
    # ------------------------------------------------------------------
    final_seq = (best_global_a * S).tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx O={best_global_obj:.6f})")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
