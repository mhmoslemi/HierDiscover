"""
Propose a candidate step‑function for the L∞ self‑convolution problem.

Algorithm
---------
* Initialise a non‑negative sequence a (length n≈8000).  If a previous
  solution ``height_sequence_1`` is available it is interpolated, otherwise the
  uniform distribution is used.
* The objective depends only on the shape (scaling is irrelevant), so a is
  kept on the probability simplex (non‑negative, sum = 1).
* At each iteration:
    1. Compute the full self‑convolution c = a * a (FFT, O(n log n)).
    2. Identify the K largest convolution entries (K≈10).  Their indices are
       the most responsible for the current maximal value.
    3. Build a weighted sub‑gradient:
           g_i = Σ_{k in topK} w_k·2·a_{k‑i}
       where w_k are normalised positive weights proportional to the
       corresponding convolution values.
    4. Take an (Euclidean‑projected) gradient step
           a ← Π_S (a – η·g)               (Π_S = simplex projection)
       with a geometrically decaying step size η.
    5. Occasionally inject a small Gaussian jitter when progress stalls.
* The best sequence seen so far (lowest value of the true objective) is
  remembered and finally passed to ``evaluate_sequence``.
* The routine respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution x*x using a real FFT (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán, “Projection onto the probability simplex:
    #   An efficient algorithm with a simple proof, and an application”.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(x: np.ndarray) -> float:
    """Return the objective 2·n·max(a*a) / (sum(a)**2)."""
    n = x.shape[0]
    conv = _full_convolution(x)
    max_val = conv.max()
    s = x.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_val / (s * s))


def _topk_weighted_gradient(a: np.ndarray,
                            conv: np.ndarray,
                            top_k: int,
                            weight_power: float = 1.0) -> np.ndarray:
    """
    Compute a weighted sub‑gradient using the top_k largest entries of conv.
    The weight of each selected index k is proportional to (conv[k])**weight_power.
    """
    n = a.shape[0]
    # indices of the K largest convolution entries
    idxs = np.argpartition(conv, -top_k)[-top_k:]
    # raw (positive) weights
    w = conv[idxs] ** weight_power
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / top_k)
    else:
        w = w / w_sum

    g = np.zeros_like(a)
    # accumulate contributions of each selected convolution entry
    for weight, k in zip(w, idxs):
        # the sub‑gradient of (a*a)_k w.r.t. a:
        #   g_i += 2·a_{k-i} for all i where 0 ≤ k-i < n
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        counterpart = k - idx
        g[idx] += weight * 2.0 * a[counterpart]
    return g


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     n: int = 8000,
                     top_k: int = 12,
                     weight_power: float = 2.0,
                     eta_start: float = 0.12,
                     eta_min: float = 1e-4,
                     decay: float = 0.9998,
                     stagnation_limit: int = 2500,
                     jitter_scale: float = 0.015) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Initialise a from a previous solution if present, otherwise uniform
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        x_old = np.linspace(0.0, 1.0, old_len)
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
        print("[init] interpolated previous solution (len:", old_len, "→", n, ")")
    except Exception:
        a = np.full(n, 1.0 / n)
        print("[init] uniform start, n =", n)

    cur_obj = _objective(a)
    best_a = a.copy()
    best_val = cur_obj
    print("[init] objective =", best_val)

    # --------------------------------------------------------------
    # Optimisation state
    # --------------------------------------------------------------
    eta = eta_start
    stagnation = 0
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # 1) current convolution
        conv = _full_convolution(a)

        # 2) weighted top‑K sub‑gradient
        g = _topk_weighted_gradient(a, conv, top_k, weight_power)

        # 3) gradient step + simplex projection
        v = a - eta * g
        a_new = _project_simplex(v)

        # 4) evaluate
        new_obj = _objective(a_new)

        # 5) accept if improvement
        if new_obj < cur_obj:
            a = a_new
            cur_obj = new_obj
            stagnation = 0
            if new_obj < best_val:
                best_val = new_obj
                best_a = a.copy()
                print(f"[iter {iteration:07d}] new best = {best_val:.9f} "
                      f"(time {time.time() - start_time:.1f}s)")
        else:
            stagnation += 1
            # occasional random acceptance to avoid dead‑ends
            if np.random.rand() < 0.0008:
                a = a_new
                cur_obj = new_obj

        # 6) step‑size decay
        eta = max(eta * decay, eta_min)

        # 7) handle stagnation – inject a tiny jitter
        if stagnation > stagnation_limit:
            jitter = np.random.normal(scale=jitter_scale, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            cur_obj = _objective(a)
            stagnation = 0
            print(f"[reset] jitter at iter {iteration} (time {time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # Final secure evaluation (single call to the provided function)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score =", final_score)
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
