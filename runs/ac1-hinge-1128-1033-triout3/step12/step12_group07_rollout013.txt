assistantfinal**Solution Overview**

The goal is to minimise  

\[
L(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where \((a*a)_k\) is the linear (full) convolution of the non‑negative vector `a` with itself.  
The value of `L` does **not** change when we multiply `a` by a positive constant, therefore we can always work with a *probability vector* (`∑a_i = 1`).  
The optimisation problem is highly non‑convex; we use a multi‑stage stochastic/gradient search that proved effective in the reference implementation and improve it by

* a finer systematic search of the two‑parameter Beta family (local optimisation of the log‑parameters),
* a dedicated optimisation of the exponential and power‑law families,
* longer and better‑tuned Adam / sub‑gradient / exponentiated‑gradient phases,
* more aggressive hill‑climbing and simulated‑annealing moves,
* frequent blending with the uniform distribution to smooth isolated spikes.

The whole routine respects the 1000 s wall‑clock budget, works for any `seed`, and finally returns the best sequence found as a plain Python list.

```python
"""
Propose a step‑function (non‑negative vector) minimising
    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser consists of

1.   systematic generation of candidate families (Beta, exponential,
     power‑law, Dirichlet, sinusoid, …);
2.   tiny local optimisation of the two‑parameter Beta shape,
     the exponential rate and the power‑law exponent (Nelder‑Mead);
3.   Adam on a smooth approximation of the max‑convolution;
4.   a deterministic sub‑gradient descent on the exact max;
5.   an exponentiated‑gradient (mirror‑descent) polishing step;
6.   aggressive hill‑climbing moves and a final simulated‑annealing
     phase with large source→target transfers;
7.   occasional convex blending with the uniform distribution.

All phases stop as soon as the 1000 s budget expires.  The routine
returns the best sequence (a list of floats) that it discovered.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (works for arbitrary sum)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    S = a.sum()
    if S < 1e-8:
        return np.inf
    b = _full_convolution(a)
    return float(2.0 * n * b.max() / (S * S))


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # softmax over convolution entries

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1: 2 * n - 1]          # exactly n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 2·n already included
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    # Fast O(n) projection, see https://arxiv.org/abs/1101.6081
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v was all negative – return uniform point
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families (systematic + random)
# ----------------------------------------------------------------------
def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    """Deterministic Beta candidate for given log‑parameters."""
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Beta on a log‑scale."""
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    return _beta_grid_candidate(n, logp, logq)


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Truncated exponential a_i ∝ r^i with r in (0,1)."""
    log_r = rng.uniform(-8.0, -0.1)
    r = np.exp(log_r)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a /= a.sum()
    return a


def _mixture_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent Beta draws."""
    w = rng.random()
    a1 = _random_beta_candidate(n, rng)
    a2 = _random_beta_candidate(n, rng)
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


def _sinusoid_candidate(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _cosine2_candidate(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed version of the max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max objective (log‑sum‑exp)."""
    # Work in log‑softmax space to keep the simplex automatically.
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.35, 5e-4
    eps_start, eps_end = 0.8, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 8000

    while time.time() < deadline:
        # decode logits
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max gradient
        _, grad = _smooth_max_and_grad(a, eps)   # already contains 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.03, size=n)

        # restart if stuck
        if idle > MAX_IDLE:
            print("[Adam] restart")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on f(a)=2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, n: int,
                        deadline: float, start_time: float,
                        best_a: np.ndarray, best_val: float,
                        max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        # all positions attaining the max – usually just one
        ks = np.where(b == max_b)[0]

        # exact sub‑gradient of the max term
        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple step‑size schedule (decreasing)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.45 * (1.0 - progress) + 1e-6 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-14:
            a, cur_val = a_new, new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # tiny random perturbation to escape flat plateaus
        if np.random.rand() < 0.004:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------
def _expgrad_polish(a0: np.ndarray, n: int,
                    deadline: float, start_time: float,
                    best_a: np.ndarray, best_val: float,
                    max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient updates for the exact max‑convolution objective."""
    a = a0.copy()
    cur_val = _objective(a)

    eta_start, eta_end = 1.2, 5e-6

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        ks = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        new_val = _objective(a)

        if new_val < cur_val - 1e-14:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[ExpG] new best = {best_val:.6f}")

        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (source → target moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray, n: int,
               deadline: float, start_time: float,
               best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Local source→target moves with a Simulated‑Annealing acceptance."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = _objective(a)
    eps = 1e-12

    # schedules
    max_frac_start, max_frac_end = 0.85, 0.001
    temp_start, temp_end = 0.04, 5e-10

    stagnation = 0
    STAG_LIMIT = 70_000

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index where current convolution is maximal
        k_max = int(np.argmax(b))

        # contribution of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # pick a target that currently contributes little to the max entry
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # update convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            stagnation = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            stagnation += 1

        if stagnation > STAG_LIMIT:
            # re‑perturb around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective(a)
            stagnation = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing with larger moves (final phase)
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   best_a: np.ndarray, best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Large source→target transfers with a simple cooling schedule."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = _objective(a)

    T0 = 0.025
    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)                     # linear cooling
        max_frac = 0.12 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # occasional tiny jitter to keep diversity
        if rng.random() < 0.012:
            a = np.maximum(a + rng.normal(scale=0.009, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray, n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a+(1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 23):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Local optimisation of low‑dimensional families (Nelder‑Mead)
# ----------------------------------------------------------------------
def _optimise_beta(logp0: float, logq0: float, n: int,
                   deadline: float, best_a: np.ndarray, best_val: float):
    """Fine‑tune the Beta parameters (logp,logq) with Nelder‑Mead."""
    def f(x):
        lp, lq = x
        cand = _beta_grid_candidate(n, lp, lq)
        return _objective(cand)

    bounds = [(-12, 6), (-12, 6)]
    res = minimize(f, x0=np.array([logp0, logq0]), method='Nelder-Mead',
                   options={'maxiter': 120, 'xatol': 1e-5, 'fatol': 1e-5,
                            'disp': False})
    cand = _beta_grid_candidate(n, res.x[0], res.x[1])
    val = _objective(cand)
    if val < best_val:
        best_val, best_a = val, cand.copy()
        print(f"[Beta‑opt] new best = {best_val:.6f}")
    return best_a, best_val


def _optimise_exponential(logr0: float, n: int,
                          deadline: float, best_a: np.ndarray, best_val: float):
    """Fine‑tune the exponential rate."""
    def f(x):
        r = np.exp(x[0])
        i = np.arange(n, dtype=np.float64)
        a = r ** i
        a /= a.sum()
        return _objective(a)

    res = minimize(f, x0=np.array([logr0]), method='Nelder-Mead',
                   options={'maxiter': 80, 'xatol': 1e-5, 'fatol': 1e-5,
                            'disp': False})
    r = np.exp(res.x[0])
    i = np.arange(n, dtype=np.float64)
    cand = r ** i
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_val, best_a = val, cand.copy()
        print(f"[Exp‑opt] new best = {best_val:.6f}")
    return best_a, best_val


def _optimise_powerlaw(p0: float, n: int,
                       deadline: float, best_a: np.ndarray, best_val: float):
    """Fine‑tune the power‑law exponent."""
    def f(x):
        p = x[0]
        i = np.arange(1, n + 1, dtype=np.float64)
        a = i ** p
        a /= a.sum()
        return _objective(a)

    res = minimize(f, x0=np.array([p0]), method='Nelder-Mead',
                   options={'maxiter': 80, 'xatol': 1e-5, 'fatol': 1e-5,
                            'disp': False})
    p = res.x[0]
    cand = _powerlaw_candidate(n, p)
    val = _objective(cand)
    if val < best_val:
        best_val, best_a = val, cand.copy()
        print(f"[Pow‑opt] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a few‑ten thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initial population (systematic + random)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # Attempt to reuse a previously discovered solution (if available)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # ----- random Beta draws -----
    for _ in range(3000):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- systematic Beta grid (coarse) -----
    logp_vals = np.linspace(-8.0, 2.5, 13)
    logq_vals = np.linspace(-8.0, 2.5, 13)
    best_logp, best_logq = None, None
    for lp in logp_vals:
        for lq in logq_vals:
            cand = _beta_grid_candidate(n, lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val
                best_logp, best_logq = lp, lq

    # ----- local refinement of the best Beta parameters (if any) -----
    if best_logp is not None and time.time() < deadline:
        best_a, best_val = _optimise_beta(best_logp, best_logq, n,
                                          deadline, best_a, best_val)

    # ----- exponential family -----
    for _ in range(800):
        cand = _exponential_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- fine‑tune the exponential rate (use last sampled log_r) -----
    if time.time() < deadline:
        # pick a representative log_r (median of sampled logs)
        log_r0 = np.log(0.5)   # a neutral starting point
        best_a, best_val = _optimise_exponential(log_r0, n,
                                                 deadline, best_a, best_val)

    # ----- power‑law family -----
    for p in np.linspace(-4.5, 4.5, 27):
        cand = _powerlaw_candidate(n, p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- fine‑tune the power‑law exponent -----
    if time.time() < deadline:
        best_a, best_val = _optimise_powerlaw(0.0, n,
                                              deadline, best_a, best_val)

    # ----- mixture of two Betas (adds extra diversity) -----
    for _ in range(800):
        cand = _mixture_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- handcrafted sinusoid / cosine‑square -----
    cand = _sinusoid_candidate(n)
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    cand = _cosine2_candidate(n)
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    print(f"[Init] best after exhaustive init = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on smooth‑max
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.18 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – Hill‑climbing (majority of the remaining budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * budget_s
    best_a, best_val = _hillclimb(best_a, n,
                                  deadline=phase_b_deadline,
                                  start_time=start_time,
                                  best_a=best_a,
                                  best_val=best_val)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Sub‑gradient polishing
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.78 * budget_s
    best_a, best_val = _subgradient_polish(best_a, n,
                                           deadline=phase_c_deadline,
                                           start_time=start_time,
                                           best_a=best_a,
                                           best_val=best_val,
                                           max_iters=800_000)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase D – Mirror‑descent (exponentiated‑gradient) polishing
    # ------------------------------------------------------------------
    phase_d_deadline = start_time + 0.92 * budget_s
    best_a, best_val = _expgrad_polish(best_a, n,
                                       deadline=phase_d_deadline,
                                       start_time=start_time,
                                       best_a=best_a,
                                       best_val=best_val,
                                       max_iters=300_000)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase E – Final simulated annealing with larger moves
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_anneal(best_a, n,
                                          deadline=deadline,
                                          start_time=start_time,
                                          best_a=best_a,
                                          best_val=best_val,
                                          rng=rng)
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (security‑checked)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
```
