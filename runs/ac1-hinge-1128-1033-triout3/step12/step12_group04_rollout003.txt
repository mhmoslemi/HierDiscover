Propose a candidate step‑function with a low ‑∞ self‑convolution value.

Algorithm
---------
1. Work on a modest size (default n0 = 2000) to keep each iteration cheap.
2. Initialise a from the previously found solution if it exists,
   otherwise from the uniform distribution; always project onto the
   probability simplex (non‑negative entries summing to 1).
3. Perform many iterations of a *smoothed* gradient descent:
   * compute the full self‑convolution a*a,
   * turn it into a soft‑max weight vector  w_k ∝ exp(conv_k/τ)  (τ
     gradually shrinks, so the method starts smooth and ends
     essentially sub‑gradient on the true maximum),
   * the gradient of   Φ(a)=∑_k w_k (a*a)_k   is
         g_i = 2·∑_k w_k·a_{k‑i} .
     This can be evaluated efficiently with a sliding‑window view of w.
   * take a step a ← a – η·g and project back onto the simplex.
   * adapt step size η (exponential decay) and smoothing τ,
     accept improvements, occasionally accept a worsening move,
     and inject a random jitter after long stagnation.
4. When the time budget for the small problem is exhausted,
   linearly interpolate the best sequence to the target length
   (n_target = 8000) and renormalise it to sum 1.
5. Run a short classic sub‑gradient refinement on the full‑size
   sequence (the same update used in the reference implementation)
   to polish the result.
6. Return the final list; the caller will invoke ``evaluate_sequence``
   once to obtain the official score.

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (no closures, no lambdas – all top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of ``x`` with itself using a real FFT.
    Returns an array of length 2*len(x)-1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    # Implementation of the algorithm from
    #   "Efficient Projections onto the L1‑Ball for Learning in High Dimensions"
    #   (Duchi et al. 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(x: np.ndarray) -> float:
    """
    Compute the exact objective used by ``evaluate_sequence`` for a
    vector normalised to sum 1, i.e.  2·n·max(a*a).
    """
    n = x.shape[0]
    conv = _full_convolution(x)
    return float(2.0 * n * conv.max())

        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        x_old = np.linspace(0.0, 1.0, old_len)
        x_new = np.linspace(0.0, 1.0, n_small)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
    except Exception:
        # Uniform start if nothing else is available.
        a = np.full(n_small, 1.0 / n_small)

    # Normalise to the probability simplex.
    a = a / a.sum()
    best_a = a.copy()
    best_obj = _objective(a)

    print(f"[init] small problem n={n_small}, objective={best_obj:.9f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the smoothed gradient descent.
    # ------------------------------------------------------------------
    step = 0.5               # initial Euclidean step size
    step_min = 0.001
    step_decay = 0.9995

    # smoothing parameter τ – start relatively large and shrink.
    tau_factor = 0.02        # τ ≈ tau_factor * current max(conv)
    tau = None
    tau_min = 1e-9
    tau_decay = 0.9995

    stagnation = 0
    max_stagn = 4000         # after this many non‑improving iterations we jitter

    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop on the small problem.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        conv = _full_convolution(a)
        max_val = conv.max()
        cur_obj = 2.0 * n_small * max_val

        # Initialise / update smoothing parameter τ.
        if tau is None:
            tau = max(tau_factor * max_val, tau_min)
        else:
            tau = max(tau * tau_decay, tau_min)

        # ---------- soft‑max weights ----------
        # stable soft‑max: subtract the max before exponentiation.
        shifted = conv - conv.max()
        w = np.exp(shifted / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            # numeric safeguard – fall back to a hard max
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w /= w_sum

        # ---------- gradient of the smoothed objective ----------
        # For each i: g_i = 2 * Σ_k w_k * a_{k‑i}
        # This is a sliding dot‑product of w with a.
        # With w length 2·n_small‑1 the view w[i : i+n_small] is always valid.
        windows = np.lib.stride_tricks.sliding_window_view(w, n_small)   # shape (n_small, n_small)
        g = 2.0 * (windows @ a)   # vector of length n_small

        # ---------- Euclidean step + projection ----------
        v = a - step * g
        a_new = _project_simplex(v)

        # Evaluate the new point (re‑use convolution if cheap).
        conv_new = _full_convolution(a_new)
        new_obj = 2.0 * n_small * conv_new.max()

        # ---------- acceptance ----------
        if new_obj < cur_obj:
            a = a_new
            cur_obj = new_obj
            stagnation = 0
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:6.1f}s] new best (small) = {best_obj:.9f}")
        else:
            # occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = a_new
                cur_obj = new_obj
            stagnation += 1

        # ---------- step‑size decay ----------
        step = max(step * step_decay, step_min)

        # ---------- stagnation handling ----------
        if stagnation > max_stagn:
            # inject a modest Gaussian jitter and renormalise
            jitter = np.random.normal(scale=0.02, size=n_small)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            stagnation = 0
            cur_obj = _objective(a)
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.1f}s] jitter (restart)")

    # ------------------------------------------------------------------
    # Upscale the best small‑size solution to the required length.
    # ------------------------------------------------------------------
    x_small = np.linspace(0.0, 1.0, n_small)
    x_big = np.linspace(0.0, 1.0, n_target)
    upscaled = np.interp(x_big, x_small, best_a)
    upscaled = np.maximum(upscaled, 0.0)
    upscaled = upscaled / upscaled.sum()    # keep it on the simplex

    # ------------------------------------------------------------------
    # Short sub‑gradient polishing on the full‑size vector.
    # ------------------------------------------------------------------
    a_big = upscaled
    cur_obj_big = _objective(a_big)
    best_big = a_big.copy()
    best_obj_big = cur_obj_big

    # parameters for the polishing phase
    step_big = 0.08
    step_big_min = 0.001
    step_big_decay = 0.9998
    stagn_big = 0
    max_stagn_big = 2000

    while time.time() < deadline:
        conv_big = _full_convolution(a_big)
        k_max = int(conv_big.argmax())

        # sub‑gradient of the hard max (identical to the reference code)
        g_big = np.zeros_like(a_big)
        i_low = max(0, k_max - (n_target - 1))
        i_high = min(n_target - 1, k_max)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        counterpart = k_max - idx
        g_big[idx] = 2.0 * a_big[counterpart]

        v_big = a_big - step_big * g_big
        a_big = _project_simplex(v_big)

        new_obj_big = _objective(a_big)

        if new_obj_big < cur_obj_big:
            cur_obj_big = new_obj_big
            stagn_big = 0
            if new_obj_big < best_obj_big:
                best_obj_big = new_obj_big
                best_big = a_big.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:6.1f}s] new best (full) = {best_obj_big:.9f}")
        else:
            if np.random.rand() < 0.001:
                a_big = a_big  # accept the move
                cur_obj_big = new_obj_big
            stagn_big += 1

        step_big = max(step_big * step_big_decay, step_big_min)

        if stagn_big > max_stagn_big:
            # random jitter to shake out of a plateau
            jitter = np.random.normal(scale=0.01, size=n_target)
            a_big = np.maximum(best_big + jitter, 0.0)
            a_big = a_big / a_big.sum()
            cur_obj_big = _objective(a_big)
            stagn_big = 0
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.1f}s] jitter (full)")

    # ------------------------------------------------------------------
    # Final secure call (single evaluation as required).
    # ------------------------------------------------------------------
    final_seq = best_big.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_score:.9f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
