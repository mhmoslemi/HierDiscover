"""
Propose a step‐function (non‑negative list) that minimizes
``evaluate_sequence``.

The method works in three stages:

1. **Parameterized shape search** – a cheap grid search over a simple
   two‑parameter family
   a_i(p,w) = (1‑w)·uniform + w·(d_i+1)^{-p},
   where d_i = min(i, n‑1‑i) is the distance to the nearest end.
   The best shape from this search is used as a warm start.

2. **Aggregated sub‑gradient descent** – instead of using the sub‑gradient
   of a single maximal convolution entry, we take the (convex) sum of the
   sub‑gradients of *all* entries whose value lies within a small tolerance
   of the current maximum.  The sub‑gradient can be computed efficiently by
   a single FFT‑based convolution.  After each step we project back onto the
   probability simplex (non‑negative entries that sum to 1).

3. **Final evaluation** – the best sequence seen is passed once to the
   supplied ``evaluate_sequence`` function.

The routine respects the supplied time budget (default 1000 s) and prints
progress information.  Only one call to ``evaluate_sequence`` is made.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT‑based convolution utilities (all top‑level as required)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def _convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    out_len = x.shape[0] + y.shape[0] - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Convolution x*x (real FFT, length 2·len(x)-1)."""
    return _convolve(x, x)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    # Algorithm 1 from https://arxiv.org/abs/1101.6081
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_maxconv(a: np.ndarray, n: int) -> float:
    """Return 2·n·max_k (a*a)_k for a vector that already satisfies Σa = 1."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


def _active_subgradient(a: np.ndarray, conv: np.ndarray,
                        tol_rel: float = 0.01) -> np.ndarray:
    """
    Compute an aggregated sub‑gradient of max(conv) using all entries within
    ``tol_rel``·max(conv) of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance (absolute)
    tol = max_val * tol_rel
    active = np.where(conv >= max_val - tol)[0]

    # indicator vector for active k (length 2·n‑1)
    L = 2 * n - 1
    indicator = np.zeros(L, dtype=float)
    indicator[active] = 1.0

    # cross‑correlation indicator * a  (see analysis for the formula)
    # grad_i = 2 * Σ_{k∈active} a_{k‑i}
    corr = _convolve(indicator, a[::-1])          # length 3·n‑2
    grad = 2.0 * corr[n - 1:n - 1 + n]            # slice of length n
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------

                     budget_s: int = 1000,

    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best list of heights discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – work with 8000 coefficients (same as the reference).
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise from a previously saved solution if possible.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_len = old.shape[0]
        x_old = np.linspace(0.0, 1.0, old_len)
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)
        print("[init] interpolated previous solution (len:", old_len, "→", n, ")")
    except Exception:
        a = np.full(n, 1.0 / n)
        print("[init] uniform start, n =", n)

    # --------------------------------------------------------------
    # Stage 1 – cheap parametric shape search
    # --------------------------------------------------------------
    distances = np.minimum(np.arange(n), n - 1 - np.arange(n))
    best_val = _objective_maxconv(a, n)
    best_a = a.copy()
    print("[stage 1] start objective =", best_val)

    # grid of parameters (p controls the decay towards the centre,
    # w mixes with the uniform distribution)
    p_vals = np.linspace(0.0, 3.0, 31)       # 0 → uniform, larger → more U‑shape
    w_vals = np.linspace(0.0, 1.0, 21)       # mixing weight

        base = (distances + 1.0) ** (-p)       # shape that is large at the ends
        for w in w_vals:

            cand = np.maximum(cand, 0.0)

                print(f"[grid] p={p:.2f} w={w:.2f} → new best {best_val:.9f}")
        # early‑exit check

    a = best_a.copy()

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    eta = 0.15                # initial step size
    eta_min = 0.001
    decay = 0.9995            # exponential decay per iteration
    stagnation = 0
    max_stagn = 2500

    while time.time() < deadline:

        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv

        # 2) record best solution
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[iter {iteration:06d}] new best {best_val:.9f} "
                  f"time {time.time() - start_time:.1f}s")

        # 3) aggregated sub‑gradient (sum over near‑max entries)
        g = _active_subgradient(a, conv, tol_rel=rel_tol)

        # 4) gradient step + projection onto simplex
        v = a - eta * g
        a_new = _project_simplex(v)

        # 5) evaluate the candidate
        conv_new = _full_convolution(a_new)
        new_obj = 2.0 * n * conv_new.max()

        # 6) simple acceptance rule with occasional uphill moves

            a = a_new
            stagnation = 0
        else:
            if np.random.rand() < 0.001:   # rare uphill acceptance
                a = a_new
                cur_obj = new_obj
            stagnation += 1

    # ------------------------------------------------------------------

        print("[final] reported score =", final_score)
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
