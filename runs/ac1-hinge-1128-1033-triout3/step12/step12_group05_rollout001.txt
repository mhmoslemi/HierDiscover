def _conv_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv
```

"""
Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm is a smoothed‑max gradient descent on the full auto‑convolution.
Instead of the non‑differentiable objective  max(conv(a,a)) we minimise the
soft‑max

    M_β(a) = (1/β)·log Σ_k exp(β·conv_k(a))

which converges to the true max as β→∞.  For a fixed β the gradient is
available in closed form:

    ∂M_β/∂a_i = 2·Σ_k w_k·a_{k‑i},
    where w_k = exp(β·conv_k) / Σ_j exp(β·conv_j).

The gradient is computed by a single FFT‑based convolution of the weight
vector w with the reversed current sequence.  An exponentiated‑gradient
(update stays on the probability simplex) together with a linear
step‑size decay and an annealing schedule for β yields fast progress.
Occasional Gaussian jitter helps to leave plateaux.  The best sequence
encountered during the run is remembered and finally evaluated with the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a vector with itself (length 2·len(x)‑1).
    Uses FFT for speed.
    """
    n = x.shape[0]
    size = _next_pow2(2 * n - 1)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of x and y, using FFT.
    Length = len(x) + len(y) - 1.
    """
    n = len(x) + len(y) - 1
    size = _next_pow2(n)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


# ----------------------------------------------------------------------
# Cheap objective used for internal comparisons (Σa = 1)
# ----------------------------------------------------------------------


def _cheap_objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed‑max gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Compute the gradient of the soft‑max of the auto‑convolution.

    Returns
    -------
    grad : np.ndarray
        Gradient (same shape as a).
    conv : np.ndarray
        Full auto‑convolution (kept for diagnostics).
    max_c : float
        Maximum entry of conv (the true objective = 2·n·max_c).
    """
    conv = _full_convolution(a)               # length 2·n‑1
    max_c = conv.max()

    # soft‑max weights (stable by subtracting the max)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * (w * a_rev)  (convolution)
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)                 # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * tmp[n - 1 : n - 1 + n]       # slice to length n
    return grad, conv, max_c

# Main entry point required by the competition framework
# ----------------------------------------------------------------------

    # ------------------------------------------------------------------
