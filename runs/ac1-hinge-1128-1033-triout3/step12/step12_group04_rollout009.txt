import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    # For each k, convolution
    for k in range(2*n-1):
        # coefficients
        # i from max(0, k-(n-1)) to min(k, n-1)
        i_low = max(0, k-(n-1))
        i_high = min(k, n-1)
        coeffs = f[i_low:i_high+1][::-1]  # reversed
        # sum_i f[i] * g[k-i] = sum_{i_low..i_high} f[i] * g_{k-i}

        constraints.append( coeffs @ g[i_low:i_high+1] <= M )
    # optional constraint sum(g) <= something? Not needed.
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=10000)
    return g.value
```

"""
Iterated linear‑programming (ALP) optimiser for the L∞ self‑convolution
problem.

The algorithm works with a probability vector a (a_i ≥ 0, Σ a_i = 1).
At each iteration

    1.  Compute the self‑convolution b = a * a  and the current worst
        entry M = max(b).
    2.  Solve a linear program

           maximise   Σ g_j
           subject to  (a * g)_k ≤ M   for all k = 0 … 2n‑2
                       g_j ≥ 0

        The convolution a * g is linear in g, so the constraints are a
        small sparse matrix–vector product.  The LP is solved with
        SciPy’s HiGHS interior‑point optimiser.
    3.  Scale the LP solution g to have the same total mass as a,
        then move a a little towards g:

           a ← (1‑t)·a + t·g     (t ≈ 0.2, slowly decayed)

        The move is projected back onto the probability simplex.
    4.  Keep the best a (lowest objective 2·n·max(a*a) / (Σa)²) seen so
        far.  If no improvement is seen for a while, sprinkle a tiny
        random jitter to escape flat regions.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found.  It can start from the global
`height_sequence_1` if that variable is present, otherwise it uses a
uniform start.  The final score is obtained with a single call to the
provided `evaluate_sequence`.
"""

import time
import math
import numpy as np

# optional SciPy imports – they are available in the execution environment
import scipy.sparse as sp
import scipy.optimize as opt


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution x * x using a real FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Return the exact value of evaluate_sequence on a (sum(a) assumed > 0)."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    s = a.sum()
    return 2.0 * n * max_b / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w ≥ 0, Σ w = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_convolution_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that (a * g) = A @ g.
    For each k, (a * g)_k = Σ_i a_i g_{k-i} with out‑of‑range
    indices treated as 0.  The matrix is Toeplitz‑like and can be
    assembled with `scipy.sparse.diags`.
    """
    n = a.shape[0]
    offsets = np.arange(n)                 # row – col = offset ≥ 0
    A = sp.diags(a, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _solve_lp(a: np.ndarray, M: float, time_limit: float) -> np.ndarray:
    """
    Solve  maximise Σ g_j
           subject to (a * g)_k ≤ M  (k = 0 … 2n‑2)
                       g_j ≥ 0

    Returns the optimal g (may be all‑zeros if the LP fails).
    """
    n = a.shape[0]
    A = _build_convolution_matrix(a)

    c = -np.ones(n)                     # maximise Σ g_j  → minimise −Σ g_j
    b_ub = np.full(2 * n - 1, M)        # RHS vector (M repeated)

    bounds = [(0, None)] * n

    # Use HiGHS; pass a remaining time limit to avoid overrunning the budget
    opts = {"presolve": True, "time_limit": max(time_limit, 0.0)}
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options=opts)

    if not res.success or res.x is None:
        # fallback: return a tiny random vector
        g = np.random.rand(n) * 1e-9
        return g / g.sum()
    else:
        return res.x


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  The function respects the 1000 s budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choice of problem size – smaller n makes each LP cheap; n≈1024 works well.
    # ------------------------------------------------------------------
    n = 1024

    # ------------------------------------------------------------------
    # Initialise a (probability vector) – use the global `height_sequence_1`
    # if available, otherwise fall back to uniform.
    # ------------------------------------------------------------------
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        try:
            old = np.asarray(old_seq, dtype=float)
            old_len = old.shape[0]
            # linear interpolation onto the new grid
            x_old = np.linspace(0.0, 1.0, old_len)
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            if a.sum() < 1e-12:
                a = np.full(n, 1.0 / n)
            else:
                a = a / a.sum()
            print("[init] interpolated from height_sequence_1 (len:", old_len,
                  "→", n, ")")
        except Exception as exc:      # pragma: no cover
            print("[init] interpolation failed:", exc)
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)
        print("[init] uniform start, n =", n)

    # Ensure exact sum == 1 (the evaluation routine works with any sum,
    # but the objective we compute assumes this normalisation).
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a)
    print("[init] objective =", best_val)

    # ------------------------------------------------------------------
    # Parameters for the outer alternating‑LP loop.
    # ------------------------------------------------------------------
    t = 0.2                # mixing weight (will be slowly decayed)
    decay_t = 0.999        # geometric decay for t
    stagnation = 0
    max_stagn = 30        # after this many non‑improving iterations jitter

    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # --------------------------------------------------------------
        # 1) current worst entry of the self‑convolution
        # --------------------------------------------------------------
        conv = _full_convolution(a)
        M = float(conv.max())

        # --------------------------------------------------------------
        # 2) solve the linear program to get a “more massive” partner g
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        g_raw = _solve_lp(a, M, remaining)

        # --------------------------------------------------------------
        # 3) normalise g (same total mass as a) and mix
        # --------------------------------------------------------------
        sum_g = g_raw.sum()
        if sum_g <= 0:
            # numerical mishap – fall back to a tiny random perturbation
            g = np.random.rand(n)
            g = g / g.sum()
        else:
            g = g_raw * (1.0 / sum_g)   # now Σ g = 1

        # convex combination and simplex projection
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new)

        # --------------------------------------------------------------
        # 4) compute new objective and possibly accept the move
        # --------------------------------------------------------------
        new_val = _objective(a_new)

        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[iter {iteration:06d}] new best = {best_val:.9f} "
                  f"(time {time.time() - start_time:.1f}s)")

        # accept the move if it improves the objective, otherwise keep old a
        if new_val < _objective(a):
            a = a_new
            stagnation = 0
        else:
            stagnation += 1
            # occasional acceptance of a worsening move to avoid dead‑ends
            if np.random.rand() < 0.002:
                a = a_new
                stagnation = 0

        # --------------------------------------------------------------
        # 5) schedule update
        # --------------------------------------------------------------
        t = max(t * decay_t, 0.01)

        # --------------------------------------------------------------
        # 6) stagnation handling – tiny Gaussian jitter
        # --------------------------------------------------------------
        if stagnation > max_stagn:
            jitter = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            stagnation = 0
            print(f"[reset] jitter at iter {iteration} "
                  f"(time {time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # Final secure evaluation (single call) and return the best sequence.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score =", final_score)
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Convert to plain Python list of floats as required.
    return best_a.tolist()
