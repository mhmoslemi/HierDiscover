"""
Gradient descent with smooth max (log‑sum‑exp) and projection onto the probability simplex.
The algorithm repeatedly
  * computes a linear self‑convolution,
  * forms a softmax weight over all shift indices,
  * obtains the gradient of the smooth surrogate  τ·log∑exp( conv/τ ),
  * takes a step with decreasing step‑size,
  * projects back to the simplex.
The temperature τ is reduced on a schedule so that the surrogate converges to the
true ‑∞ norm. A small random jitter is injected when progress stalls.
The best sequence encountered is finally evaluated with the supplied
``evaluate_sequence``.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (real FFT). Returns length 2·len(a)-1 array."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv

def _gradient_from_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of τ·log∑exp(conv/τ) w.r.t. a.
    w must be a probability vector of length 2·len(a)-1 (softmax weights).
    Returns an array of the same shape as a.
    """
    n = a.shape[0]
    # reverse w and convolve with a
    w_rev = w[::-1]
    size = _next_pow2(n + w_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[: n + w_rev.shape[0] - 1]
    # indices that correspond to the gradient components
    idx = (w_rev.shape[0] - 1) - np.arange(n)
    g = 2.0 * conv[idx]
    return g

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

"""
Smooth‑max gradient descent on the self‑convolution objective.

The true objective is   2·n·max_k (a*a)_k   with a_i ≥ 0 and Σ a_i = 1.
Direct sub‑gradient descent on the max can get stuck because it focuses on a
single index.  Instead we minimise the smooth surrogate

    F_τ(a) = 2·n·τ·log∑_k exp( (a*a)_k / τ ),

whose gradient is

    ∇F_τ(a)_j = 2·∑_k w_k a_{k-j},
    w_k = softmax_k( (a*a)_k / τ ).

For a given temperature τ the gradient can be obtained by a single
cross‑correlation which we compute with FFTs.  τ is reduced geometrically
(continuation / annealing) so that the surrogate converges to the true
ℓ_∞‑norm.  After each step the vector is projected onto the probability
simplex.  Small random jitter is injected when progress stalls.

The best sequence found before the time budget expires is finally evaluated
with the supplied ``evaluate_sequence`` function.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (real FFT, length 2·len(a)-1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _gradient_from_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of τ·log∑exp(conv/τ) w.r.t. a.
    w – a probability vector of length 2·len(a)-1 (soft‑max weights).
    Returns an array of the same shape as a.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # reverse because we need a⋆w
    size = _next_pow2(n + w_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[: n + w_rev.shape[0] - 1]
    # indices that give the gradient components
    idx = (w_rev.shape[0] - 1) - np.arange(n)
    return 2.0 * conv[idx]


def _project_simplex(v: np.ndarray) -> np.ndarray:

    u = np.sort(v)[::-1]

    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)
