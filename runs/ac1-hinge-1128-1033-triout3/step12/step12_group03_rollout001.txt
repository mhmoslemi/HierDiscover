"""
Hybrid “smooth‑max → exact‑max” optimizer with aggressive annealing and a
simple greedy refinement.  The routine works on the probability simplex
(sum = 1) so the evaluation metric reduces to

        F(a) = 2·n·max(a * a)               ( * is linear convolution )

The optimizer repeatedly:
  • draws a fresh start (uniform, Dirichlet or a sparse random vector);
  • runs a long Adam‑style optimisation where the smooth‑max
    temperature λ is increased linearly with *global* wall‑clock time;
    once λ is large we switch to the true max (one‑hot weight vector);
  • after each Adam run a cheap projected‑gradient (sub‑gradient) sweep
    further tightens the maximum;
  • keeps the best candidate seen so far.

The whole search respects the 1000‑second budget and finally returns the
best sequence found (as a list of floats).  It also prints progress
information and the final score obtained from ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (FFT based, no Python loops over the sequence length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a real vector with itself.
    Length = 2*len(x)-1, computed with a power‑of‑two FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation g_i = Σ_k w_k · a_{k‑i}.
    Returned vector has length len(a).
    Implemented via an FFT‑based convolution.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reverse a
    out_len = w.shape[0] + n - 1        # = (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1                       # keep the middle n entries
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max( a * a ) for a probability vector a."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a tiny safety margin for the final secure call.
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a little larger than the reference (8000) gives
    # the optimizer more degrees of freedom while keeping FFT cheap.
    # --------------------------------------------------------------
    n = 12000

    # --------------------------------------------------------------
    # Initialise from a previous construction if it exists.
    # --------------------------------------------------------------
    try:
        old_seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old_seq.size > 1:
            # Linear interpolation onto the new grid.
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_seq.size)
            a0 = np.interp(new_x, old_x, old_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 1e-12 else np.full(n, 1.0 / n)
            print("[init] interpolated previous solution")
        else:
            raise Exception
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] uniform start")

    # --------------------------------------------------------------
    # Global best tracker
    # --------------------------------------------------------------
    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[global] start objective = {best_val:.9f}")

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters (tuned for the 1000 s budget)
    # ------------------------------------------------------------------
    lambda_start = 5.0          # temperature at the very beginning
    lambda_end   = 8000.0       # final temperature (smooth‑max sharpness)
    adam_lr      = 0.05
    beta1, beta2  = 0.9, 0.999
    adam_eps     = 1e-8

    # ------------------------------------------------------------------
    # Main outer loop – several restarts until the time budget expires
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline:
        restart_counter += 1

        # -----------------------------------------------------------
        # 1️⃣  Choose a fresh starting distribution.
        #    Mix three strategies to keep the search diverse.
        # -----------------------------------------------------------
        r = np.random.rand()
        if r < 0.4:
            # uniform (already good baseline)
            a_start = np.full(n, 1.0 / n)
        elif r < 0.8:
            # Dirichlet with concentration <1 → sparse vectors
            conc = 0.3
            a_start = np.random.gamma(conc, 1.0, size=n)
            a_start = a_start / a_start.sum()
        else:
            # explicit sparsity: pick ~sqrt(n) random locations
            m = max(2, int(np.sqrt(n)))
            idx = np.random.choice(n, m, replace=False)
            a_start = np.zeros(n)
            a_start[idx] = np.random.rand(m)
            s = a_start.sum()
            a_start = a_start / s if s > 1e-12 else np.full(n, 1.0 / n)

        # blend a little with the current global best to keep progress
        a_start = 0.6 * a_start + 0.4 * best_a
        a_start = a_start / a_start.sum()

        # convert to log‑softmax parameters (logits)
        eps = 1e-12
        z = np.log(a_start + eps)
        # a small random jitter helps to escape shallow basins
        z += np.random.normal(scale=0.04, size=n)

        # Adam accumulators
        m = np.zeros_like(z)
        v = np.zeros_like(z)

        # -----------------------------------------------------------
        # 2️⃣  Run a relatively long Adam optimisation.
        # -----------------------------------------------------------
        # Number of inner iterations – we keep it modest enough to stay
        # inside the budget but large enough to make progress.
        max_inner = 60000
        iter_counter = 0
        last_improve_iter = 0

        while time.time() < deadline and iter_counter < max_inner:
            iter_counter += 1

            # ---- 2.1 soft‑max → probability vector
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            a = exp_z / exp_z.sum()

            # ---- 2.2 evaluate objective (exact max)
            conv = _full_convolution(a)
            conv_max = conv.max()
            cur_val = 2.0 * n * conv_max

            # ---- 2.3 record a new global best
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                last_improve_iter = iter_counter
                now = time.time()
                print(f"[restart {restart_counter:02d} iter {iter_counter:06d}] "
                      f"new best = {best_val:.9f} (t={now-start_time:.1f}s)")

            # ---- 2.4 smooth‑max temperature (global linear schedule)
            overall_prog = (time.time() - start_time) / (deadline - start_time)
            lam = lambda_start * (1.0 - overall_prog) + lambda_end * overall_prog

            # ---- 2.5 weight vector for the (smooth) max
            if lam > 2000.0:
                # Exact max → one‑hot weight; gives the true sub‑gradient.
                kmax = int(np.argmax(conv))
                w = np.zeros_like(conv)
                w[kmax] = 1.0
            else:
                # Smooth‑max via log‑sum‑exp.
                # Use a numerically stable formulation.
                exp_vals = np.exp(lam * (conv - conv_max))
                w = exp_vals / exp_vals.sum()

            # ---- 2.6 gradient of (smooth) max w.r.t. a
            # g = 2 * Σ_k w_k a_{k-i}  =  2 * cross‑correlation(w, a)
            g_raw = _cross_corr(w, a)
            g = 2.0 * g_raw

            # ---- 2.7 back‑prop through the soft‑max
            dot_ag = np.dot(a, g)
            grad_z = a * (g - dot_ag)   # ∂f/∂z_i

            # ---- 2.8 Adam update
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** iter_counter)
            v_hat = v / (1.0 - beta2 ** iter_counter)
            z -= adam_lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # ---- 2.9 occasional noise if stuck for a long time
            if iter_counter - last_improve_iter > 8000:
                # simple Gaussian perturbation of logits
                z += np.random.normal(scale=0.06, size=n)
                last_improve_iter = iter_counter
                print(f"[restart {restart_counter}] added noise at iter {iter_counter}")

        # -----------------------------------------------------------
        # 3️⃣  Cheap projected sub‑gradient refinement on the best so far.
        # -----------------------------------------------------------
        a_ref = best_a.copy()
        obj_ref = best_val
        step = 0.02
        for _ in range(2500):
            if time.time() >= deadline:
                break
            conv = _full_convolution(a_ref)
            kmax = int(np.argmax(conv))

            # Sub‑gradient of the exact max (same formula as in the main loop)
            # g_i = 2·a_{kmax‑i} whenever the index is in range.
            g = np.zeros_like(a_ref)
            if kmax < n:
                # overlap region 0 … kmax
                g[:kmax + 1] = 2.0 * a_ref[kmax::-1]
            else:
                # kmax >= n → overlap region kmax-(n‑1) … n‑1
                offset = kmax - (n - 1)
                length = n - offset
                g[offset:] = 2.0 * a_ref[:length][::-1]

            # trial step & projection onto the simplex
            a_trial = _project_simplex(a_ref - step * g)
            conv_trial = _full_convolution(a_trial)
            obj_trial = 2.0 * n * conv_trial.max()

            if obj_trial < obj_ref - 1e-12:
                a_ref = a_trial
                obj_ref = obj_trial
                # keep global best up‑to‑date
                best_a = a_ref.copy()
                best_val = obj_ref
                step = min(step * 1.1, 0.5)
                print(f"[subgrad] improved to {best_val:.9f} (t={time.time()-start_time:.1f}s)")
                if best_val < 1.503:
                    break
            else:
                step *= 0.7   # shrink step when no improvement

        # -----------------------------------------------------------
        # 4️⃣  Early stop if we already beat the target bound.
        # -----------------------------------------------------------
        if best_val < 1.503:
            print("[early] target achieved, stopping early")
            break

    # ------------------------------------------------------------------
    # 5️⃣  Final secure evaluation (the only place we call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
