def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.8 # final evaluation safety

    n = 8000

    # initial candidate
    best_a = None
    best_O = np.inf

    # add previous solution if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        # resample to length n if needed
        if old.shape[0] != n:
            # interpolate
            x_new = np.linspace(0, 1, n)
            x_old = np.linspace(0, 1, old.shape[0])
            old_interp = np.interp(x_new, x_old, old)
        else:
            old_interp = old
        old_interp = np.maximum(old_interp, 0.0)
        if old_interp.sum() > 0:
            old_interp /= old_interp.sum()
        best_a = old_interp
        best_O = 2.0 * n * np.max(_full_convolution(old_interp))
        print(f"[init] starting from previous solution O={best_O:.6f}")
    except Exception:
        pass

    # random search
    rand_end = start + (budget_s * 0.3)  # 30% time for random search
    n = 8000
    while time.time() < rand_end:
        # random generate shape
        family = np.random.choice(['power', 'exp', 'beta'])
        if family == 'power':
            p = np.random.uniform(0.1, 3.0)
            shift = np.random.uniform(0.5, 5.0)
            i = np.arange(n) + shift
            a = i ** (-p)
        elif family == 'exp':
            lam = np.random.uniform(0.0005, 0.01)
            shift = np.random.uniform(0.0, 5.0)
            a = np.exp(-lam * (np.arange(n) + shift))
        else:  # beta
            p_alpha = np.random.uniform(-0.9, 3.0)
            p_beta = np.random.uniform(-0.9, 3.0)
            i = np.arange(n)
            # to prevent log of zero, we use i+1 and n-i
            log_a = p_alpha * np.log(i + 1) + p_beta * np.log(n - i)
            a = np.exp(log_a)

        # ensure positivity
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s

        conv = _full_convolution(a)
        max_b = conv.max()
        O_val = 2.0 * n * max_b   # since sum =1

        if O_val < best_O - 1e-7:
            best_O = O_val
            best_a = a.copy()
            print(f"[rand] new best O={best_O:.6f}  family={family}")

    # Mirror descent refinement
    if best_a is None:
        # fallback: uniform
        best_a = np.full(n, 1.0 / n)
        best_O = 2.0 * n * np.max(_full_convolution(best_a))
        print(f"[fallback] uniform O={best_O:.6f}")

    # Mirror descent (entropy) for the remaining time
    remaining = deadline - time.time()
    steps_est = int(min(2000000, remaining * 18000))  # assume 50 µs per step
    if steps_est < 1000:

    print(f"[refine] mirror descent steps={steps_est}")

    a_refined = _max_refine(best_a, steps=steps_est, eta_start=0.45)

    conv_ref = _full_convolution(a_refined)
    O_ref = 2.0 * n * conv_ref.max()
    if O_ref < best_O:
        best_O = O_ref
        best_a = a_refined
        print(f"[refine] improved to O={best_O:.6f}")

    # final evaluation
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_O:.6f})")
    except Exception as exc:

"""
Candidate generator for the step‑function optimisation challenge.

The algorithm proceeds in three stages:

1. **Initial guess** – reuse the previously stored best sequence (if
   available) and a handful of analytically motivated shapes
   (power‑law, exponential and Beta‑type families).  All candidates are
   normalised to unit ℓ₁‑norm and the one with the smallest
   `2·n·max(conv(a,a))` (which equals the evaluation objective for a
   normalised vector) is kept.

2. **Exploratory random search** – for a configurable fraction of the
   time budget we draw random parameters for the three shape families
   (including simple mixtures) and keep any improvement.

3. **Exact mirror‑descent polishing** – starting from the best candidate,
   we run an entropy mirror‑descent on the *true* max‑entry of the
   self‑convolution.  The update
   `a ← a·exp(-η·g)` (element‑wise) guarantees non‑negativity and
   projection onto the probability simplex after each step.
   The step size η decays linearly; occasional jitter is injected when
   progress stalls.  The best vector seen during this phase is the final
   answer.

All heavy work is performed with FFT‑based convolutions (O(n log n)) and
the routine respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max‑entry of the self‑convolution.
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray,
                steps: int = 20000,
                eta_start: float = 0.45,
                jitter_every: int = 8000,
                jitter_scale: float = 0.015) -> np.ndarray:
    """
    Perform entropy‑mirror descent on the max‑entry of `a*a`.

    Parameters
    ----------
    a : np.ndarray
        Starting probability vector (non‑negative, sum = 1).
    steps : int
        Number of mirror‑descent updates.
    eta_start : float
        Initial step‑size (decays linearly to zero).
    jitter_every : int
        If no improvement has been observed for this many steps,
        a small random jitter is applied.
    jitter_scale : float
        Standard deviation of the Gaussian jitter (in log‑space).

    Returns
    -------
    np.ndarray
        The best vector encountered during the optimisation.
    """
    n = a.shape[0]
    a = a.copy()
    best_obj = np.inf
    best_vec = a.copy()
    stagnation = 0

    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())            # index of the current maximum
        max_b = conv[k]

        # exact objective (sum(a) == 1)
        obj = 2.0 * n * max_b

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_vec = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # sub‑gradient of the max‑entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]

        # linearly decaying step size
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)

        # renormalise to the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)

        # occasional jitter to escape plateaus
        if stagnation >= jitter_every:
            jitter = np.random.normal(scale=jitter_scale, size=n)
            # work in log‑space to keep positivity
            log_a = np.log(a + 1e-30) + jitter
            a = np.exp(log_a)
            a /= a.sum()
            stagnation = 0

    return best_vec
