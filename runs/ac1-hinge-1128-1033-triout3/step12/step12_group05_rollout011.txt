"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.

The search works in three stages:

1️⃣  **Random mask** – pick a support of size ~n/2 and put equal weight on the
    chosen indices.  The mask gives a cheap proxy
    ``obj = 2·n·max(conv(mask,mask)) / |mask|²`` which is exactly the value
    returned by ``evaluate_sequence`` when the weights are uniform.

2️⃣  **Simulated‑annealing on the mask** – starting from the best mask found
    above, repeatedly swap a chosen “on” index with an “off” index.
    The convolution counts of the binary mask are kept incrementally,
    therefore each trial update is O(|mask|).  Swaps that lower the
    maximal convolution count are always accepted; worse swaps are accepted
    with a temperature‑controlled probability.  Restarts are performed when
    no improvement is seen for a while.  This phase drives the objective
    towards the theoretical lower bound (≈1).

3️⃣  **Mirror‑descent refinement** – the binary mask is turned into a
    probability vector (uniform weight on the support) and a few thousand
    exponentiated‑gradient steps are performed, using a sub‑gradient of the
    max‑convolution (active‑set version).  This fine‑tunes the weights and
    often yields a small extra gain.

The whole routine respects the supplied time budget (default 1000 s)
and finally returns the best sequence discovered as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    size = 1 << (2 * n - 2).bit_length()          # next power of two ≥ 2·n‑1
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 7,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """Return a sub‑gradient of ``max(b)`` w.r.t. ``a``."""
    n = a.shape[0]
    max_val = b.max()
    if max_active >= b.size:
        active = np.arange(b.size)
    else:
        active = np.argpartition(b, -max_active)[-max_active:]
        mask = b[active] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            active = np.array([int(np.argmax(b))])
        else:
            active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Binary mask utilities (fast integer convolution)
# ----------------------------------------------------------------------


def _conv_counts_from_mask(mask: np.ndarray) -> np.ndarray:
    """Convolution counts for a 0/1 mask (ordered pairs)."""
    # ``mask`` is a bool array; cast to int for convolution.
    return np.convolve(mask.astype(np.int32), mask.astype(np.int32))


def _mask_objective(mask: np.ndarray, n: int) -> float:
    """
    Cheap objective for a binary mask:
        obj = 2·n·max(counts) / (|mask|²)
    (identical to the official score for uniform weights on the support).
    """
    k = mask.sum()
    if k == 0:
        return np.inf
    cnt = _conv_counts_from_mask(mask)
    max_c = int(cnt.max())
    return float(2.0 * n * max_c / (k * k))


# ----------------------------------------------------------------------
# Simulated‑annealing on the mask
# ----------------------------------------------------------------------


def _anneal_mask(mask: np.ndarray,
                 n: int,
                 deadline: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a simple simulated‑annealing / hill‑climbing on a binary mask.
    Returns the best mask seen and its cheap objective value.
    """
    k = int(mask.sum())
    conv = _conv_counts_from_mask(mask)
    cur_max = int(conv.max())
    best_mask = mask.copy()
    best_max = cur_max

    # list of present indices – will be updated in‑place
    support = np.flatnonzero(mask).astype(np.int32)

    # a few constants for the schedule
    avg_cnt = k * k / (2.0 * n)                     # average count per sum
    init_temp = max(1.0, cur_max - avg_cnt)        # typical gap to the average
    no_improve = 0
    max_no_improve = 25000                         # restart trigger

    while time.time() < deadline - 0.5:
        # --- propose a random swap -------------------------------------------------
        out_pos = rng.integers(len(support))          # index inside ``support``
        i_out = int(support[out_pos])

        # pick a random zero‑position
        while True:
            i_in = int(rng.integers(n))
            if not mask[i_in]:
                break

        # ----- incremental update of the convolution counts -----------------------
        # make a copy that we can modify safely
        new_conv = conv.copy()

        # removal of i_out
        sums_rem = i_out + support                     # includes self‑pair
        new_conv[sums_rem] -= 2
        new_conv[2 * i_out] += 1                       # correct the self‑pair

        # addition of i_in
        # support after removal = support without i_out
        support_wo = np.delete(support, out_pos)
        sums_add = i_in + support_wo
        new_conv[sums_add] += 2
        new_conv[2 * i_in] += 1

        new_max = int(new_conv.max())

        # ----- acceptance test ----------------------------------------------------
        # temperature decays linearly with elapsed time
        elapsed = time.time() - (deadline - 1000.0)    # start time approx.
        progress = min(1.0, elapsed / (deadline - (deadline - 1000.0)))
        temp = init_temp * (1.0 - progress) + 0.5

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # occasional uphill moves
            prob = np.exp(-(new_max - cur_max) / max(1e-6, temp))
            accept = rng.random() < prob

        if accept:
            # commit the swap
            mask[i_out] = False
            mask[i_in] = True
            support[out_pos] = i_in                     # keep list up‑to‑date
            conv = new_conv
            cur_max = new_max
            no_improve = 0

            if cur_max < best_max:
                best_max = cur_max
                best_mask = mask.copy()
                # Print a short progress line
                obj = 2.0 * n * best_max / (k * k)
                print(f"[anneal] new best maxcnt={best_max} obj={obj:.6f}")

        else:
            no_improve += 1

        # ----- restart if we are stuck -------------------------------------------
        if no_improve >= max_no_improve:
            # random re‑initialisation with the same support size
            mask[:] = False
            idx = rng.choice(n, size=k, replace=False)
            mask[idx] = True
            conv = _conv_counts_from_mask(mask)
            cur_max = int(conv.max())
            support = np.flatnonzero(mask).astype(np.int32)
            no_improve = 0
            print("[anneal] restart with fresh mask")

    # return the best mask and its cheap objective value
    best_obj = 2.0 * n * best_max / (k * k)
    return best_mask, best_obj

def propose_candidate(seed: int = 42,

    ``evaluate_sequence``.  The method combines a random binary mask,
    simulated‑annealing on the mask, and a short mirror‑descent fine‑tuning.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # a small safety margin

    # problem size – a good trade‑off between expressivity and speed
    # --------------------------------------------------------------
    n = 8000                     # length of the sequence

    # --------------------------------------------------------------

        nonlocal best_seq, best_val
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[update] new best cheap objective={best_val:.6f}")

    # try to import a historic solution
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the required length
            x_new = np.linspace(0.0, 1.0, n)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            old = np.interp(x_new, x_old, old)
        old = np.clip(old, 0.0, None)
        s = old.sum()
        if s > 0.0:
            old = old / s
            _maybe_update_candidate(old)
            print("[init] imported previous solution")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Random‑mask stage – cheap exploration
    # --------------------------------------------------------------
    # we mainly search for a good *support*; weights will be uniform.
    support_size = n // 2                    # 50 % density (empirically strong)

        mask = np.zeros(n, dtype=bool)
        idx = rng.choice(n, size=support_size, replace=False)
        mask[idx] = True
        a = np.zeros(n, dtype=float)
        a[mask] = 1.0 / support_size
        _maybe_update_candidate(a)

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_val = _objective(best_seq)
        print("[fallback] uniform start, cheap obj=", best_val)

    # --------------------------------------------------------------
    # 2️⃣  Simulated‑annealing on the binary mask
    # --------------------------------------------------------------
    # start from the mask of the currently best cheap candidate
    start_mask = (best_seq > 0).astype(bool)   # uniform weight ⇒ binary support
    if start_mask.sum() == 0:                  # very unlikely, but be safe

        start_mask[:support_size] = True

    a = np.zeros(n, dtype=float)
    k = mask.sum()

    _maybe_update_candidate(a)                 # evaluate after annealing

    # --------------------------------------------------------------
    # 3️⃣  Mirror‑descent refinement (exponentiated gradient)
    # --------------------------------------------------------------
    a = best_seq.copy()                        # start from the best cheap seq
    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    no_improve = 0
    max_no_improve = 12000

        conv_full = _full_convolution(a)
        cur_val = 2.0 * n * conv_full.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[optim] iter {it} new best {best_val:.6f}")
        else:

        # sub‑gradient from the most active lag values

        # occasional jitter to escape plateaux
        if rng.random() < 0.001 or no_improve >= max_no_improve:
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            momentum.fill(0.0)
            no_improve = 0

        it += 1

    # --------------------------------------------------------------
    # 4️⃣  Final secure evaluation (the real score)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
