"""
Improved step‑function optimiser (≈1000 s)

Features
--------
* Larger problem size (n=20000) – richer search space.
* Diverse cheap initialisation:
    – interpolated previous solution (if available)
    – uniform, Beta, Dirichlet, exponential and symmetric‑triangular families
* Two‑stage smooth optimisation:
    – several L‑BFGS‑B runs on the log‑sum‑exp smooth‑max surrogate
    – epsilon is decreased geometrically throughout the stage.
* Aggressive local stochastic search (Gaussian jitter + simplex projection)
  that runs after each L‑BFGS run.
* A final exact sub‑gradient polishing step (same as the reference code)
  and a short high‑precision L‑BFGS run.

All helper functions are defined at module level – no closures or lambdas.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def exact_objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with sum 1."""
    n = a.shape[0]
    b = full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Compute the smooth approximation of max(conv(a,a)) and the gradient
    of 2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = full_convolution(a)                # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick (avoid overflow)
    shifted = (b - max_b) / eps
    w = np.exp(shifted)
    p = w / w.sum()                        # soft weights over the 2n‑1 positions

    # convolution of a with reversed p (the gradient of the smooth max)
    p_rev = p[::-1]
    conv_ab = conv_fft(a, p_rev)           # length 2n‑1
    # only the n entries that line up with the forward part are needed
    segment = conv_ab[n - 1 : 2 * n - 1]    # shape (n,)
    grad_a = 4.0 * n * segment[::-1]       # factor 2·n already included

    smooth_val = max_b + eps * np.log(w.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → probability simplex)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x | x ≥ 0, Σx = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def lbfgs_optimize(a0: np.ndarray, n: int, eps: float,
                   max_iter: int, deadline: float,
                   best_a: np.ndarray, best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a single L‑BFGS‑B optimisation on the smoothed objective.
    The variables are stored in log‑space (soft‑max) so the simplex
    constraints are satisfied automatically.
    """
    # initialise in log‑space, add a tiny constant to avoid log(0)
    x0 = np.log(a0 + 1e-12)

    def fun(x, n, eps):
        a = softmax(x)
        smooth_val, _ = smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    def jac(x, n, eps):
        a = softmax(x)
        _, grad_a = smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    # give SciPy a few seconds to finish if the deadline is near
    time_left = deadline - time.time()
    if time_left < 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, args=(n, eps), jac=jac,
                       method="L-BFGS-B", options=opts)
    except Exception as exc:                 # pragma: no cover
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = softmax(res.x)
    val_opt = exact_objective(a_opt)

    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Small stochastic refinement: Gaussian jitter + simplex projection
    # ------------------------------------------------------------------
    jitter_iters = 40
    for _ in range(jitter_iters):
        if time.time() >= deadline:
            break
        sigma = 0.01 * (0.5 + rng.random())   # modest noise
        cand = a_opt + rng.normal(scale=sigma, size=n)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        val = exact_objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            a_opt = cand
            print(f"[L‑BFGS‑j] improved to {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (true max‑convolution)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray, n: int, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    cur_val = exact_objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum of contributions from every maximal diagonal
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step‑size schedule – aggressive early, quiet later
        progress = (now - start_time) / (deadline - start_time)
        step = 0.45 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = project_simplex(a_new)

        new_val = exact_objective(a_new)

        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional random jitter to escape tiny plateaus
        if it % 500 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.012, size=n), 0.0)
            a = project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps to smooth spikes)
# ----------------------------------------------------------------------
def blend_with_uniform(a: np.ndarray, n: int, cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = exact_objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generators (cheap initialisation)
# ----------------------------------------------------------------------
def random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑family shape: a_i ∝ i^{p‑1} (n‑i+1)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-λ·i)."""
    lam_log = rng.uniform(-4.0, 2.0)          # λ in roughly [1e‑4, 100]
    lam = np.exp(lam_log)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def triangular_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric triangular shape (peak at centre, linearly decreasing)."""
    peak = rng.integers(n // 3, 2 * n // 3)   # random peak location
    left = np.arange(peak + 1)
    right = np.arange(n - peak - 1)
    left_vals = left / (peak + 1)
    right_vals = right[::-1] / (n - peak)
    a = np.concatenate([left_vals, right_vals])
    a /= a.sum()
    return a


def dirichlet_candidate(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Simple Dirichlet draw with concentration `alpha`."""
    a = rng.dirichlet(np.full(n, alpha))
    return a


def interpolate_previous(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a previous solution to a new grid size."""
    old_len = old.shape[0]
    new_x = np.linspace(0.0, 1.0, new_n)
    old_x = np.linspace(0.0, 1.0, old_len)
    a = np.interp(new_x, old_x, old)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and that minimises `evaluate_sequence`.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – a larger n gives a richer optimisation landscape.
    # --------------------------------------------------------------
    n = 20000

    # --------------------------------------------------------------
    # 0️⃣  Coarse initialisation (exact evaluation)
    # --------------------------------------------------------------
    candidates: list[tuple[float, np.ndarray]] = []

    # 0.1  Interpolate a previously discovered solution (if present)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a0 = interpolate_previous(old, n)
        val = exact_objective(a0)
        candidates.append((val, a0))
        print(f"[Init] interpolated previous solution, value={val:.6f}")
    except Exception:
        pass

    # 0.2  Uniform baseline
    uniform = np.full(n, 1.0 / n)
    candidates.append((exact_objective(uniform), uniform))

    # 0.3  Random Beta family
    for _ in range(250):
        cand = random_beta_candidate(n, rng)
        candidates.append((exact_objective(cand), cand))

    # 0.4  Exponential decay family
    for _ in range(80):
        cand = exponential_candidate(n, rng)
        candidates.append((exact_objective(cand), cand))

    # 0.5  Symmetric triangular shapes
    for _ in range(80):
        cand = triangular_candidate(n, rng)
        candidates.append((exact_objective(cand), cand))

    # 0.6  Dirichlet draws with several concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            cand = dirichlet_candidate(n, rng, alpha)
            candidates.append((exact_objective(cand), cand))

    # keep the best few for the next phase
    candidates.sort(key=lambda t: t[0])
    top_k = 7
    best_a, best_val = candidates[0][1], candidates[0][0]
    print(f"[Init] best after sweep: value={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – L‑BFGS on smooth‑max, multiple restarts and jitter
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.40 * budget_s
    eps_start = 5e-1
    eps_final = 1e-12

    while time.time() < phase_a_deadline and time.time() < deadline:
        prog = (time.time() - start_time) / (phase_a_deadline - start_time)
        eps = eps_start * (eps_final / eps_start) ** prog

        for idx in range(min(top_k, len(candidates))):
            if time.time() >= phase_a_deadline:
                break
            a_start = candidates[idx][1]
            best_a, best_val = lbfgs_optimize(
                a_start, n, eps,
                max_iter=1300,
                deadline=deadline,
                best_a=best_a,
                best_val=best_val,
                rng=rng,
            )
            best_a, best_val = blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase B – Stochastic local search (Gaussian jitter + projection)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.75 * budget_s
    while time.time() < phase_b_deadline and time.time() < deadline:
        # decreasing Gaussian noise scale
        elapsed = time.time() - start_time
        total = phase_b_deadline - start_time
        sigma = 0.03 * (1.0 - elapsed / total) + 0.004

        cand = best_a + rng.normal(scale=sigma, size=n)
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            continue
        cand /= cand.sum()
        val = exact_objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Jitter] improved to {best_val:.6f}")

        # occasional larger perturbation to escape plateaus
        if rng.random() < 0.02:
            idx1, idx2 = rng.choice(n, size=2, replace=False)
            delta = 0.02 * (rng.random() - 0.5)
            cand = best_a.copy()
            cand[idx1] = max(0.0, cand[idx1] - delta)
            cand[idx2] = max(0.0, cand[idx2] + delta)
            cand = project_simplex(cand)
            val = exact_objective(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[Swap] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase C – Exact sub‑gradient polishing (≈20 % of remaining budget)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.95 * budget_s
    best_a, best_val = subgradient_polish(
        best_a, n,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        max_iters=350_000,
    )
    best_a, best_val = blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase D – Final high‑precision L‑BFGS run (tiny ε)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = lbfgs_optimize(
            best_a, n, eps=1e-12,
            max_iter=900,
            deadline=deadline,
            best_a=best_a,
            best_val=best_val,
            rng=rng,
        )

    # --------------------------------------------------------------
    # Final secure evaluation and return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Ensure we return a plain Python list of floats
    return best_a.tolist()
