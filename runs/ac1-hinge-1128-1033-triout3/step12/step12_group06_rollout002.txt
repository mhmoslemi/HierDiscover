def _softmax_grad(a, lam, size_g):
    # compute FFT of a
    A = np.fft.rfft(a, size_g)
    conv_full = np.fft.irfft(A * A, size_g)[:2*n-1]

"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

The search is performed in three stages:

1. **Coarse parametric search** –  samples a two‑parameter
   Beta‑family  a_i ∝ i^{p-1}(n-i+1)^{q-1}  on a grid and keeps the best
   (the uniform case corresponds to p = q = 1).  This family generalises the
   monotone power law used before and provides many more shapes.

2. **Smooth‑max gradient descent** –  starting from the best shape,
   the objective  2·n·max(conv)  is approximated by a soft‑max
   (log‑sum‑exp) with an increasing temperature λ.
   The gradient of the smoothed objective is computed analytically using
   FFTs and an exponentiated‑gradient update (which stays inside the
   probability simplex).  λ is ramped from a small value (≈10) to a large
   value (≈500) so that the algorithm initially follows a smooth landscape
   and later focuses on the true maximum.

3. **Final mirror‑descent polishing** –  a few iterations of the original
   sub‑gradient “exponentiated mirror‑descent’’ (identical to the previous
   reference implementation) finish the optimisation.

The whole routine respects a 1000 s time budget, prints progress and
returns the best sequence found (as a plain ``list[float]``).  All numbers
are non‑negative and normalised to sum to one, which does not affect the
final score because ``evaluate_sequence`` is homogeneous of degree 0.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full, length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2·n‑1
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1 … n  (Beta‑family).

    Returns a normalised non‑negative vector with sum = 1,
    or None if the parameters produce an invalid (non‑positive) vector.
    """
    if p <= 0 or q <= 0:
        return None
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, p - 1.0) * np.power(n - i + 1.0, q - 1.0)
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Sub‑gradient for the (hard) max‑convolution (used in final polishing)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 5,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """Sub‑gradient of max(b) w.r.t. a (active‑set version)."""
    n = a.shape[0]
    max_val = b.max()
    if max_active >= b.size:
        active = np.arange(b.size)
    else:
        active = np.argpartition(b, -max_active)[-max_active:]
        mask = b[active] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            active = np.array([int(np.argmax(b))])
        else:
            active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Smooth‑max helper (soft‑max of the convolution)
# ----------------------------------------------------------------------


def _softmax_grad(a: np.ndarray,
                  lam: float,
                  size_fft: int,
                  n: int) -> tuple[float, np.ndarray]:
    """
    Returns (current_max, gradient) of the λ‑soft‑max surrogate:

        f_λ(a) = (1/λ)·log Σ_k exp(λ·conv_k)

    Gradient is ∂f_λ/∂a and already incorporates the factor 2 from the
    convolution derivative.
    """
    # FFT of a (padded to size_fft)
    A = np.fft.rfft(a, size_fft)

    # full convolution a*a  (length 2·n‑1)
    conv_full = np.fft.irfft(A * A, size_fft)[: 2 * n - 1]

    # soft‑max weights
    scaled = lam * conv_full
    max_scaled = scaled.max()
    exp_shift = np.exp(scaled - max_scaled)          # safe exponentiation
    w = exp_shift / exp_shift.sum()                # probabilities, Σ w = 1

    # gradient: 2·(a * w) evaluated on the "valid" part
    W = np.fft.rfft(w, size_fft)
    conv_aw_full = np.fft.irfft(A * W, size_fft)
    # slice indices n‑1 … 2·n‑2  (inclusive) → length n
    grad = 2.0 * conv_aw_full[n - 1: 2 * n - 1]

    # actual max of the (hard) convolution (needed for the true objective)
    cur_max = conv_full.max()
    return cur_max, grad.real


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The algorithm follows the three‑stage plan described in the module
    doc‑string.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size – 8000 points gives a good trade‑off.
    # --------------------------------------------------------------
    n = 8000

    # Pre‑computed FFT size that can accommodate the longest convolution
    # needed in the smooth‑max phase (a * w, length 3·n‑2).
    size_fft = 1 << ((3 * n - 2).bit_length())

    best_seq: np.ndarray | None = None
    best_val = np.inf
    best_p = best_q = None

    # --------------------------------------------------------------
    # Helper to possibly update the global best
    # --------------------------------------------------------------
    def _maybe_update(seq: np.ndarray, src: str = "grid"):
        nonlocal best_seq, best_val, best_p, best_q
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            # keep the parameters that produced the sequence if they exist
            if src.startswith("beta"):
                best_p, best_q = src.split(":")[1].split(",")
                best_p, best_q = float(best_p), float(best_q)
            print(f"[{src}] new best val={best_val:.6f}")

    # --------------------------------------------------------------
    # 0️⃣  Warm‑start from a previously found solution (if available)
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)          # type: ignore[name-defined]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _maybe_update(a0, src="init")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Coarse grid search over the Beta‑family (p,q)
    # --------------------------------------------------------------
    p_grid = np.linspace(0.5, 10.0, 20)          # 20 points → step ≈0.5
    q_grid = np.linspace(0.3, 5.0, 20)           # include q<1 for heavy tails
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is not None:
                _maybe_update(cand, src=f"beta:{p},{q}")
            if time.time() > deadline - 200.0:
                break
        if time.time() > deadline - 200.0:
            break

    # --------------------------------------------------------------
    # 2️⃣  Random local refinement in (p,q) space
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    random_trials = 3000
    for _ in range(random_trials):
        if best_p is None or best_q is None:
            # fall back to a pure random sample
            p = rng.uniform(0.5, 10.0)
            q = rng.uniform(0.3, 5.0)
        else:
            p = rng.normal(best_p, 0.8)
            q = rng.normal(best_q, 0.5)
        if p <= 0 or q <= 0:
            continue
        cand = _beta_shape(p, q, n)
        if cand is not None:
            _maybe_update(cand, src="rand")
        if time.time() > deadline - 200.0:
            break

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max gradient descent (exponentiated‑gradient)
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback to uniform – should never happen
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)

    a = best_seq.copy()
    step_start = 0.5            # initial multiplicative step size
    step_min = 1e-5
    lam_start = 10.0
    lam_end = 500.0

    while time.time() < deadline - 150.0:
        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)

        # temperature λ grows linearly
        lam = lam_start + (lam_end - lam_start) * progress

        # compute smoothed max and its gradient
        cur_max, grad = _softmax_grad(a, lam, size_fft, n)

        # true objective (hard max) for bookkeeping
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[smooth] iter {int(elapsed):5d}s new best {best_val:.6f}")

        # exponentiated‑gradient update (stay in the simplex)
        step = step_start * (1.0 - progress) + step_min
        log_factor = -step * grad
        # avoid overflow/underflow
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)

        # re‑normalise to Σa = 1
        s = a.sum()
        if not np.isfinite(s) or s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

    # --------------------------------------------------------------
    # 4️⃣  Final polishing with the original sub‑gradient mirror‑descent
    # --------------------------------------------------------------
    # (identical to the reference implementation, but we start from the
    #  best sequence discovered so far)
    a = best_seq.copy()
    m = np.zeros_like(a)
    beta_mom = 0.9
    no_improve = 0
    max_no_improve = 2000
    it = 0

    while time.time() < deadline:
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[polish] iter {it:5d} new best {best_val:.6f}")
        else:
            no_improve += 1

        # sub‑gradient of the hard max
        grad = _subgradient_max(a, conv, max_active=5, rel_tol=1e-6)

        # momentum
        m = beta_mom * m + (1.0 - beta_mom) * grad

        # step‑size schedule (slowly decaying)
        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)
        step = 0.2 * (1.0 - progress) + 1e-5

        # exponentiated update
        log_factor = -step * m
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)
        a_sum = a.sum()
        if a_sum == 0.0 or not np.isfinite(a_sum):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        # occasional jitter to jump out of plateaus
        if np.random.rand() < 0.001 or no_improve >= max_no_improve:
            jitter = np.random.normal(scale=1e-3, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            no_improve = 0

        it += 1

    # --------------------------------------------------------------
    # 5️⃣  Final evaluation (only for logging)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
