"""
step‑function optimiser – 1000 s budget

The optimiser works in four stages:

1️⃣  Broad stochastic sweep – generate thousands of candidates from
    several families (Beta, Dirichlet, piece‑wise constant,
    alternating, sinusoidal).  Only the top‑k (default 20) are kept,
    using the exact objective 2·n·max(conv(a,a)).

2️⃣  Smooth‑max L‑BFGS – each elite candidate is refined with a
    log‑sum‑exp surrogate of the max.  The surrogate smoothness
    parameter ε is drawn at random (log‑uniform) for every restart,
    which helps to escape shallow local minima.  The optimiser works
    in the soft‑max (log‑its) parametrisation, automatically enforcing
    the simplex constraints.

3️⃣  Fast hill‑climbing – a few rounds of tiny random mass transfers
    (i ↦ j) are performed on the current best solution; only moves that
    improve the exact objective are kept.  This non‑gradient search
    explores directions that the smooth surrogate may miss.

4️⃣  Exact sub‑gradient polishing – a projected sub‑gradient descent on
    the true non‑smooth objective finishes the run.  The step size is
    scheduled aggressively early and tapers off later; occasional
    jitter helps to leave plateaus.

The best sequence found is finally checked with the provided
`evaluate_sequence` and returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear full convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length = 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) surrogate and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and
    the gradient of 2·n·smooth_max w.r.t. a (already multiplied by 2·n).
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2·n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    # protect against overflow – use np.exp on shifted (negative numbers)
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2·n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2·n‑1
    # keep the n entries that correspond to the “forward” part
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n

    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → probability simplex) & Euclidean simplex projection
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exact, non‑smooth objective)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    cur_val = _objective(a)

    n = a.shape[0]
    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum of contributions from all maximal diagonals
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # diminishing step size – aggressive early, tiny later
        progress = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 400 == 0 and np.random.rand() < 0.03:
            jitter = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Convex blending with the uniform distribution (helps to smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray, cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex mixtures t·a + (1‑t)·u for t∈[0,1] (21 points)."""
    n = a.shape[0]
    u = np.full(n, 1.0 / n)
    best_a, best_val = a, cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float,
                    max_iter: int = 1200) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smoothed objective for a single restart.
    The variables are stored in log‑it space, guaranteeing feasibility.
    """
    n = a0.shape[0]
    # initialise in log‑space (soft‑max guarantees Σa=1)
    x0 = np.log(a0 + 1e-12)

    # ----- objective ----------------------------------------------------
    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    # ----- gradient -----------------------------------------------------
    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        # chain rule through soft‑max
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    # Respect remaining time – abort early if not enough.
    if deadline - time.time() < 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:       # pragma: no cover
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generation helpers
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return a normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ≈1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _alternating_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a mildly alternating pattern a_i = exp(A⋅(-1)^i)."""
    A = rng.uniform(0.0, 1.0)          # contrast control
    a = np.exp(A * ((-1) ** np.arange(n)))
    a /= a.sum()
    return a


def _sine_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sinusoidally modulated density: a_i ∝ exp(A·sin(2π·f·i/n+φ))."""
    A = rng.uniform(0.0, 1.5)          # amplitude
    f = rng.integers(1, 15)           # frequency (integer to keep periodic)
    phi = rng.uniform(0, 2 * math.pi)
    i = np.arange(n, dtype=np.float64)
    raw = np.exp(A * np.sin(2.0 * math.pi * f * i / n + phi))
    raw /= raw.sum()
    return raw


def _piecewise_constant_candidate(n: int, rng: np.random.Generator,
                                  n_segments: int = 200) -> np.ndarray:
    """Random step‑function with a given number of constant segments."""
    # Dirichlet draws for segment heights
    seg_heights = rng.dirichlet(np.full(n_segments, 1.0))
    # assign each position a segment index
    idx = np.floor(np.arange(n) * n_segments / n).astype(int)
    a = seg_heights[idx]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Top‑k maintenance (small memory footprint)
# ----------------------------------------------------------------------
def _maintain_top(candidates: list, val: float, a: np.ndarray, k: int) -> None:
    """Insert (val,a) into the sorted top‑k list in‑place."""
    if len(candidates) < k:
        candidates.append((val, a))
        candidates.sort(key=lambda t: t[0])
    elif val < candidates[-1][0]:
        candidates[-1] = (val, a)
        candidates.sort(key=lambda t: t[0])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The routine respects a hard time
    limit (≈budget_s seconds) and returns the best list of floats it
    discovers.
    """
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # moderate problem size – large enough to give a rich search space
    n = 12000

    # ------------------------------------------------------------------
    # 0️⃣  Broad stochastic sweep (exact objective)
    # ------------------------------------------------------------------
    top_k = 20
    candidates: list[tuple[float, np.ndarray]] = []

    # 0.1  Interpolate an older solution if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val0 = _objective(a0)
        _maintain_top(candidates, val0, a0, top_k)
        print(f"[Init] interpolated prior solution → value={val0:.6f}")
    except Exception:
        pass

    # 0.2  Uniform baseline
    uniform = np.full(n, 1.0 / n)
    _maintain_top(candidates, _objective(uniform), uniform, top_k)

    # 0.3  Random Beta candidates
    for _ in range(2200):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        _maintain_top(candidates, val, cand, top_k)

    # 0.4  Dirichlet draws with several concentration levels
    for alpha in (0.02, 0.05, 0.2, 0.5, 1.0, 3.0, 10.0):
        for _ in range(5):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            _maintain_top(candidates, val, cand, top_k)

    # 0.5  Alternating pattern
    for _ in range(30):
        cand = _alternating_candidate(n, rng)
        val = _objective(cand)
        _maintain_top(candidates, val, cand, top_k)

    # 0.6  Sinusoidal modulation
    for _ in range(40):
        cand = _sine_candidate(n, rng)
        val = _objective(cand)
        _maintain_top(candidates, val, cand, top_k)

    # 0.7  Piece‑wise constant (random steps)
    for _ in range(80):
        cand = _piecewise_constant_candidate(n, rng, n_segments=150)
        val = _objective(cand)
        _maintain_top(candidates, val, cand, top_k)

    # sort final shortlist
    candidates.sort(key=lambda t: t[0])
    best_val = candidates[0][0]
    best_a = candidates[0][1].copy()
    print(f"[Init] best after coarse sweep → value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – L‑BFGS on smooth‑max, many random ε‑restarts
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    while time.time() < phase_a_deadline and time.time() < deadline:
        # draw a random ε for this restart (log‑uniform)
        eps = 10 ** rng.uniform(-12, -2)   # between 1e‑12 and 1e‑2
        for val, a_start in candidates[:top_k]:
            if time.time() >= phase_a_deadline:
                break
            best_a, best_val = _lbfgs_optimize(a_start, deadline, start_time,
                                               best_a, best_val,
                                               eps=eps, max_iter=1000)
            best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ------------------------------------------------------------------
    # Phase B – Fast hill‑climbing on the current best solution
    # ------------------------------------------------------------------
    def _hill_climb_optimize(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             best_a: np.ndarray,
                             best_val: float,
                             max_moves: int = 30_000) -> tuple[np.ndarray, float]:
        """Tiny random mass‑transfer moves, accepting only improvements."""
        a = a0.copy()
        cur_val = _objective(a)
        n = a.shape[0]
        move_mass = 0.004      # about 0.4 % of total mass per move
        for it in range(max_moves):
            if time.time() >= deadline:
                break
            # choose donor i (must have enough mass) and receiver j
            i = rng.integers(n)
            if a[i] < move_mass:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            delta = move_mass * rng.random()
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # no need for projection – convex combination keeps Σ=1 and non‑neg.
            new_val = _objective(a_new)
            if new_val < cur_val - 1e-12:
                a = a_new
                cur_val = new_val
                if new_val < best_val:
                    best_val, best_a = new_val, a_new.copy()
                    print(f"[Hill] new best = {best_val:.6f}")
        return best_a, best_val

    # give hill‑climbing a modest slice of the remaining budget
    phase_b_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _hill_climb_optimize(best_a, phase_b_deadline, start_time,
                                            best_a, best_val, max_moves=25000)

    # ------------------------------------------------------------------
    # Phase C – Exact sub‑gradient polishing (most of the remaining time)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.95 * budget_s
    best_a, best_val = _subgradient_polish(best_a,
                                           deadline=phase_c_deadline,
                                           start_time=start_time,
                                           best_a=best_a,
                                           best_val=best_val,
                                           max_iters=500_000)
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ------------------------------------------------------------------
    # Phase D – Final short L‑BFGS with ultra‑small ε
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(best_a,
                                           deadline,
                                           start_time,
                                           best_a,
                                           best_val,
                                           eps=1e-12,
                                           max_iter=800)

    # ------------------------------------------------------------------
    # Final secure evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
