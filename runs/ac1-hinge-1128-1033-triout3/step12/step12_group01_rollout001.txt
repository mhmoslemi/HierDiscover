def project_simplex(v):
    # v may have negative values
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _full_convolution(x):
    n = len(x)
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def project_simplex(v):
    n = v.shape[0]
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative adjust
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step function that (empirically) improves the
objective O = 2·n·max(conv(seq,seq)) / (sum(seq))².

Algorithm
---------

1. **Initialisation** – start from the previously discovered best sequence
   (global ``height_sequence_1``) if it exists, otherwise from a random
   Dirichlet draw.  The vector is normalised to sum = 1.

2. **Projected sub‑gradient descent** – the objective depends only on the
   maximum entry of the self‑convolution.  For a given index *k* where the
   maximum is attained the exact sub‑gradient is
       g[i] = a[k‑i]   for i ∈ [max(0,k−n+1) , min(k,n‑1)],
   zero elsewhere.  We move a in the negative direction,
       a ← a – η·g,
   then project back onto the probability simplex (non‑negative, sum = 1).
   The step size η is decreased geometrically; a small random *jitter*
   is injected periodically to escape plateaus.

3. **Entropy‑mirror polishing** – after the gradient phase we apply a few
   thousand steps of the mirror‑descent scheme used in the reference code
   (exact max‑entry sub‑gradient, multiplicative update) to polish the
   solution.

4. **Final evaluation** – the best vector found is returned as a plain
   Python list; ``evaluate_sequence`` is called once for reporting.

The whole routine respects a 1000 s wall‑clock budget.  All helper
functions are defined at module level; no closures or file I/O are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full, length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two different vectors (full)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto the probability simplex)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Entropy‑mirror descent polishing (exact max entry of self‑convolution)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 3000, eta_start: float = 0.5) -> np.ndarray:
    """Mirror‑descent polishing on the exact max‑entry of the self‑convolution."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g = np.zeros_like(a)
            g[idx] = 2.0 * a[k - idx]            # sub‑gradient of the max entry
            eta = eta_start * (1.0 - t / steps)
            a = a * np.exp(-eta * g)            # multiplicative update
        # renormalise to the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)
    return a

# Main entry point
# ----------------------------------------------------------------------

    The routine respects the 1000 s time budget.
    """

    start = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start + budget_s - 0.8

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Build the initial sequence (reuse previous best if possible)
    # --------------------------------------------------------------
    try:

        if old.shape[0] != n:
            # linear interpolation to the required length
            x_old = np.linspace(0.0, 1.0, old.shape[0])

            a = np.interp(x_new, x_old, old)
        else:
            a = old.copy()
        print("[init] interpolated previous solution")
    except Exception:
        # fall back to a random Dirichlet draw (moderately sparse)
        a = np.random.dirichlet(np.full(n, 0.1))

    # normalise to the simplex

        # compute sub‑gradient at the current maximum

    # --------------------------------------------------------------
