def _mixture_beta_sequence(params, n):
    logp1, logq1, logp2, logq2, t = params
    p1 = np.exp(logp1)
    q1 = np.exp(logq1)
    p2 = np.exp(logp2)
    q2 = np.exp(logq2)
    w = 1.0 / (1.0 + np.exp(-t))
    i = np.arange(1, n+1, dtype=np.float64)
    a1 = (i ** (p1 - 1.0)) * ((n - i + 1) ** (q1 - 1.0))
    a2 = (i ** (p2 - 1.0)) * ((n - i + 1) ** (q2 - 1.0))
    # Avoid all zeros: ensure positive
    # Clip extremely small values:
    a1 = np.maximum(a1, 0.0)
    a2 = np.maximum(a2, 0.0)
    a = w * a1 + (1 - w) * a2
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

"""
Improved step‑function optimizer.

Key ideas
---------
* Increase problem size (n=10 000) – a larger discrete space lets the
  distribution better approximate the continuous optimum (theoretical
  limit ≈ 1.5).
* Add a cheap but expressive *mixture‑Beta* parametric family and perform
  a focussed random‑search / local refinement on its 5 parameters before
  the main continuous optimisation.  This often discovers a shape that
  the generic random draws miss.
* Re‑balance the time budget: a short “parameter‑search” phase (≈12 %),
  a slightly longer Adam phase, a more aggressive hill‑climber, and a
  final extended simulated‑annealing phase.
* Keep the robust blend‑with‑uniform polishing after every major phase.
  The whole pipeline is deterministic apart from the RNG seed, so the
  final sequence is reproducible.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Beta‑shaped vector on a log‑scale."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    """Deterministic Beta candidate for given log‑parameters."""
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Truncated exponential shape a_i ∝ r^i with r∈(0,1)."""
    log_r = rng.uniform(-8.0, -0.1)   # r between ~0.0003 and 0.9
    r = np.exp(log_r)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i ∝ (i+1)^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a /= a.sum()
    return a


def _mixture_beta_sequence(params: np.ndarray, n: int) -> np.ndarray:
    """
    Mixture of two Beta distributions.
    params = [logp1, logq1, logp2, logq2, t] with
    w = sigmoid(t) ∈ (0,1).
    """
    logp1, logq1, logp2, logq2, t = params
    p1 = np.exp(logp1)
    q1 = np.exp(logq1)
    p2 = np.exp(logp2)
    q2 = np.exp(logq2)
    w = 1.0 / (1.0 + np.exp(-t))          # sigmoid
    i = np.arange(1, n + 1, dtype=np.float64)
    a1 = (i ** (p1 - 1.0)) * ((n - i + 1) ** (q1 - 1.0))
    a2 = (i ** (p2 - 1.0)) * ((n - i + 1) ** (q2 - 1.0))
    # safeguard against all‑zero rows (possible for extreme params)
    a1 = np.maximum(a1, 0.0)
    a2 = np.maximum(a2, 0.0)
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _mix_beta_random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random mixture‑Beta candidate (used in the cheap random phase)."""
    logp1, logq1 = rng.uniform(-9, 2, size=2)
    logp2, logq2 = rng.uniform(-9, 2, size=2)
    t = rng.uniform(-5, 5)
    return _mixture_beta_sequence(np.array([logp1, logq1, logp2, logq2, t]), n)


# ----------------------------------------------------------------------
# Parameter‑search for the mixture‑Beta family
# ----------------------------------------------------------------------
def _optimize_mixture_params(n: int,
                             start_time: float,
                             deadline: float,
                             best_a: np.ndarray,
                             best_val: float) -> tuple[np.ndarray, float]:
    """
    Simple random‑search + local refinement for the 5‑parameter mixture‑Beta
    family.  Returns an improved (a, value) pair if found before ``deadline``.
    """
    rng = np.random.default_rng()
    best_params = None

    # ----- coarse random sweep (fast, no loops) -----
    while time.time() < deadline:
        # generate a batch of candidates at once to reduce overhead
        batch = 64
        logp1 = rng.uniform(-9, 2, size=batch)
        logq1 = rng.uniform(-9, 2, size=batch)
        logp2 = rng.uniform(-9, 2, size=batch)
        logq2 = rng.uniform(-9, 2, size=batch)
        t = rng.uniform(-5, 5, size=batch)
        for k in range(batch):
            a = _mixture_beta_sequence(
                np.array([logp1[k], logq1[k], logp2[k], logq2[k], t[k]]), n)
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                best_params = np.array([logp1[k], logq1[k], logp2[k],
                                        logq2[k], t[k]])
        # break early if we are close to the overall deadline
        if time.time() > deadline - 0.2:
            break

    # ----- local refinement (small Gaussian steps) -----
    if best_params is None:
        # fall back to a reasonable default
        best_params = np.array([np.log(0.5), np.log(0.5),
                                np.log(0.5), np.log(0.5), 0.0])

    cur_params = best_params.copy()
    scale = np.array([0.5, 0.5, 0.5, 0.5, 1.0])   # initial step sizes
    while time.time() < deadline:
        new_params = cur_params + rng.normal(scale=scale)
        # keep parameters within reasonable bounds
        new_params[:4] = np.clip(new_params[:4], -12, 4)
        new_params[4] = np.clip(new_params[4], -8, 8)
        a = _mixture_beta_sequence(new_params, n)
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            cur_params = new_params
            # gradually shrink step size after a success
            scale *= 0.95
        else:
            # occasional step‑size increase to escape flat regions
            if rng.random() < 0.05:
                scale *= 1.05

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex for L(a)=2·n·smooth_max(a)."""
    w = np.log(a0 + 1e-12)          # work in log‑softmax space

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.001
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 5000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits → probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # learning‑rate / smoothing schedule
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and gradient
        _, grad = _smooth_max_and_grad(a, eps)

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape shallow plateaus
        if np.random.rand() < 0.018:
            w += np.random.normal(scale=0.02, size=n)

        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (source→target moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """Local search with simulated‑annealing acceptance."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # more aggressive early moves
    max_frac_start, max_frac_end = 0.97, 0.001
    temp_start, temp_end = 0.04, 1e-9
    IMPROVE_LIMIT = 60000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        # contribution of each a[i] to the current peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # pick source index proportional to its contribution
        if total > 0.0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index: favour indices with small contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling – jump back to best + noise
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation (it={iteration}), re‑perturb")
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        # occasional jitter
        if np.random.rand() < 0.02:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional random jitter
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------
def _expgrad_polish(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    max_iters: int = 100_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient updates for the exact max‑convolution objective."""
    a = a0.copy()
    cur_val = _objective(a)

    eta_start, eta_end = 1.0, 1e-5

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # schedule eta
        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        new_val = _objective(a)

        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[ExpG] new best = {best_val:.6f}")

        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.018, size=n), 0.0)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Larger‑move simulated‑annealing (final phase)
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple annealing with relatively large source→target transfers."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)            # linear cooling
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # tiny jitter to keep diversity
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and (tries to) minimise ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 10_000            # larger than before – gives more flexibility

    # --------------------------------------------------------------
    # 0️⃣  Rich initial population (random + systematic)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # try to reuse a previously discovered solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # ----- random Beta draws (more than before) -----
    for _ in range(2500):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- systematic Beta grid -----
    logp_vals = np.linspace(-7.5, 2.0, 18)   # finer grid
    logq_vals = np.linspace(-7.5, 2.0, 18)
    for lp in logp_vals:
        for lq in logq_vals:
            cand = _beta_grid_candidate(n, lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ----- Dirichlet draws -----
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(25):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ----- exponential family -----
    for _ in range(500):
        cand = _exponential_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- power‑law family -----
    for p in np.linspace(-4.0, 4.0, 30):
        cand = _powerlaw_candidate(n, p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- mixture of two Betas (quick random sweep) -----
    for _ in range(800):
        cand = _mix_beta_random_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- handcrafted sinusoid / cosine‑square -----
    i = np.arange(n, dtype=np.float64)
    cand = np.sin(np.pi * (i + 0.5) / n)
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    cand = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    print(f"[Init] best after exhaustive init = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Parameter‑search on mixture‑Beta (≈12 % of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.12 * budget_s
    best_a, best_val = _optimize_mixture_params(
        n, start_time, phase1_deadline, best_a, best_val)

    # --------------------------------------------------------------
    # 2️⃣ Adam optimisation (≈18 % of the budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase2_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # 3️⃣ Hill‑climbing (≈45 % of the budget)
    # --------------------------------------------------------------
    phase3_deadline = start_time + 0.75 * budget_s
    best_a, best_val = _hillclimb(
        best_a, n,
        deadline=phase3_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # 4️⃣ Sub‑gradient polishing (≈7 % of the budget)
    # --------------------------------------------------------------
    phase4_deadline = start_time + 0.82 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase4_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=300_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # 5️⃣ Mirror‑descent polishing (≈7 % of the budget)
    # --------------------------------------------------------------
    phase5_deadline = start_time + 0.89 * budget_s
    best_a, best_val = _expgrad_polish(
        best_a, n,
        deadline=phase5_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=150_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # 6️⃣ Final simulated‑annealing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_anneal(
            best_a, n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            rng=rng,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure a plain list of floats is returned
    return best_a.tolist()
