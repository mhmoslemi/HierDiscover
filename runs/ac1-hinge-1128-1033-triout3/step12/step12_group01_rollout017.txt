import time
from scipy.linalg import toeplitz
from scipy.optimize import linprog

def full_convolution(a):
    # using np.convolve (fast enough for n <= 2000)
    return np.convolve(a, a)

def lp_step(f, M):
    n = len(f)
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.concatenate([np.array([f[0]]), np.zeros(n-1)])
    A = toeplitz(col, row)   # shape (2n-1, n)
    # impose constraints: A @ b <= M
    c = -np.ones(n)   # maximize sum(b)
    bounds = [(0, None)] * n
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback to uniform
        return np.full(n, 1.0/n)
    b = res.x
    # enforce positivity
    b = np.maximum(b, 0.0)
    if b.sum() <= 0:
        b = np.full(n, 1.0/n)
    return b

def compute_O(a):
    # sum is normalized to 1
    conv = np.convolve(a, a)
    return 2.0 * len(a) * conv.max()
 
def _max_refine(a, steps=3000, eta_start=0.5):
    n = len(a)
    a = a.copy()
    for t in range(steps):
        conv = np.convolve(a, a)
        k = int(conv.argmax())
        # subgradient of max w.r.t a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[idx] = 2.0 * a[k - idx]
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a.fill(1.0 / n)
    return a

# ----------------------------------------------------------------------
# FFT based full convolution (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full, length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

# Objective O = 2·n·max(conv)  (the sum is normalised to 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    conv = _full_convolution(a)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Linear‑programming step: maximise Σ b  subject to  f∗b ≤ M
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Given a normalised vector `f` (sum=1) and the current bound `M`,
    solve the LP that maximises Σ b_i under the linear constraints
    (f∗b)_k ≤ M.  Returns the raw optimum vector `b` (not normalised).
    """
    n = f.shape[0]
    # Toeplitz matrix that implements the full convolution f∗b
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([np.array([f[0]], dtype=f.dtype), np.zeros(n - 1, dtype=f.dtype)])
    A = toeplitz(col, row)                # shape (2n‑1, n)

    # Linear programme: maximise 1ᵀb  ⇔ minimise (−1)ᵀb
    c = -np.ones(n, dtype=np.float64)     # minimise negative sum → maximise sum
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    # Linprog with the HiGHS interior‑point solver (fast for dense problems)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30.0})
    if not res.success:
        # Very unlikely, but fall back to a uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)
    b = np.maximum(res.x, 0.0)            # safety against tiny negatives
    return b

# Entropy‑mirror descent on the exact max‑entry of f∗f
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 3000, eta_start: float = 0.5) -> np.ndarray:
    """Mirror‑descent polishing that directly minimises max(f∗f)."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of the max‑entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        eta = eta_start * (1.0 - t / steps)          # linear decay
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)
    return a
