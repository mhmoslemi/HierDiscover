def _hillclimb_multi(a0, n, deadline, start_time, best_a, best_val, rng):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule parameters
    max_frac_start = 0.8
    max_frac_end = 0.001
    temp_start = 0.03
    temp_end = 1e-9

    no_improve = 0
    stagnation_limit = 60000 maybe.

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        max_val = b.max()
        S = np.where(np.abs(b - max_val) < 1e-15)[0]  # or b == max_val
        # compute contributions
        contrib = np.zeros_like(a)
        for s in S:
            idx = s - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib[mask] += a[mask] * a[idx[mask]]

        total = contrib.sum()
        if total > 0:
            src_probs = contrib / total
            i = rng.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target weighting
        tgt_weights = np.zeros_like(a)
        # compute weights inverse to contributions
        tgt_weights = 1.0 / (contrib + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = rng.choice(n, p=tgt_weights / wsum)
        else:
            # choose random j != i
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0:
            continue

        # update convolution
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling: after too many fails, reinitialize small random perturb.
        if no_improve > 50000:
            print("[Hill] stagnation, re-perturb")
            a = best_a.copy()
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

    return best_a, best_val

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    # choose n
    n = 12000

    # initial candidate
    best_a = np.full(n, 1.0/n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start: {best_val:.6f}")

    # initialize candidate generation within some time slice (e.g., 5% budget)
    init_deadline = start + 0.05*budget_s
    families = ['beta','exp','power']
    while time.time() < init_deadline:
        # pick random family
        choice = rng.integers(3)
        if choice == 0:
            cand = _random_beta_candidate(n, rng)
        elif choice == 1:
            cand = _exponential_candidate(n, rng)
        else:
            p = rng.uniform(-4.0, 4.0)
            cand = _powerlaw_candidate(n, p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] new best {best_val:.6f}")

    # systematic beta grid
    logp_vals = np.linspace(-7.0, 2.0, 11)
    logq_vals = np.linspace(-7.0, 2.0, 11)
    for lp in logp_vals:
        for lq in logq_vals:
            cand = _beta_grid_candidate(n, lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val
    print(f"[Init] after grid: {best_val:.6f}")

    # Phase A: Adam (5% time)
    phase_a_deadline = start + 0.05*budget_s
    best_a, best_val = _adam_optimize(best_a, n, phase_a_deadline, start, best_a, best_val)

    # Phase B: Hillclimb (70% time)
    phase_b_deadline = start + 0.75*budget_s
    best_a, best_val = _hillclimb_multi(best_a, n, phase_b_deadline, start, best_a, best_val, rng)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # Phase C: Subgradient (15% time)
    phase_c_deadline = start + 0.90*budget_s
    best_a, best_val = _subgradient_polish(best_a, n, phase_c_deadline, start, best_a, best_val, max_iters=200_000)

    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # Phase D: Expgrad (5% time)

    best_a, best_val = _expgrad_polish(best_a, n, phase_d_deadline, start, best_a, best_val, max_iters=80_000)

    print(f"[Final] score = {final_score:.6f}")
    return best_a.tolist()

"""
Improved multi‑stage optimizer for the step‑function attack.

Key ideas
---------
* Use a larger problem size (n≈12000) – the theoretical optimum tends
  to 1 as n grows, so a finer discretisation gives more room for improvement.
* Enrich the initial population with systematic Beta‑grid candidates **and**
  fast random draws from several families.
* A dedicated hill‑climber that accounts for **all** maximal convolution
  entries (active set) when choosing source/target indices.  This gives a
  better estimate of the sub‑gradient of max (conv(a,a)).
* Re‑allocate the time budget: a short Adam warm‑up, a long
  hill‑climbing phase (≈70 % of the wall‑clock), followed by light
  polishing steps (sub‑gradient, exponentiated‑gradient) and a final
  simulated‑annealing refinement.
* All heavy convolutions are performed with an FFT‑based routine
  (`_full_convolution`) for speed; the final score is obtained by the
  provided `evaluate_sequence` function.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# --------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may create tiny negative values – clip to 0
    conv = np.maximum(conv, 0.0)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    conv = np.maximum(conv, 0.0)
    return conv


# --------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# --------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# --------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp) – used by Adam
# --------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # the gradient of Σ_i a_i a_{k-i} w.r.t a is
    #   2 * (a convolved with indicator of shift k)
    # the smoothed gradient becomes a weighted sum over k:
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# --------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Candidate families
# --------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑shaped vector on a log‑scale."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    """Deterministic Beta candidate for given log‑parameters."""
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Truncated exponential shape a_i ∝ r^i with r∈(0,1)."""
    log_r = rng.uniform(-8.0, -0.1)   # r between ~0.0003 and 0.9
    r = np.exp(log_r)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i ∝ (i+1)^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a /= a.sum()
    return a


def _sinusoid_candidate(n: int) -> np.ndarray:
    """Simple half‑sine shape."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _cosine_sq_candidate(n: int) -> np.ndarray:
    """Cosine‑square shape."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# --------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution.
# --------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex for L(a)=2·n·smooth_max(a)."""
    # work in log‑softmax space to keep positivity
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.001
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 8000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule learning rate and smoothing
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max gradient
        _, grad = _smooth_max_and_grad(a, eps)   # already includes 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape flat spots
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.02, size=n)

        if idle > MAX_IDLE:
            # restart from random point
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# --------------------------------------------------------------
# Aggressive hill‑climbing that uses *all* maximal entries.
# --------------------------------------------------------------
def _hillclimb_multi(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     best_a: np.ndarray,
                     best_val: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Local search moving mass between indices based on the active set."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.8, 0.001
    temp_start, temp_end = 0.03, 1e-9
    eps = 1e-14
    no_improve = 0
    IMPROVE_LIMIT = 60000  # restart after this many non‑improving attempts

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # active set of maximal convolution entries
        max_b = b.max()
        S = np.where(np.abs(b - max_b) < 1e-15)[0]

        # contribution of each index to the active set (sub‑gradient of max)
        contrib = np.zeros_like(a)
        for s in S:
            idx = s - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib[mask] += a[mask] * a[idx[mask]]

        total = contrib.sum()
        if total > 0:
            src_probs = contrib / total
            i = rng.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index: favour tiny contributions
        tgt_weights = 1.0 / (contrib + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = rng.choice(n, p=tgt_weights / wsum)
        else:
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional jitter to keep diversity
        if rng.random() < 0.015:
            a = np.maximum(a + rng.normal(scale=0.018, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        if no_improve > IMPROVE_LIMIT:
            # re‑perturb around the current best
            print("[Hill] stagnation – re‑perturb")
            a = best_a.copy()
            a = np.maximum(a + rng.normal(scale=0.03, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

    return best_a, best_val


# --------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution objective).
# --------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        S = np.where(np.abs(b - max_b) < 1e-15)[0]

        grad = np.zeros_like(a)
        for s in S:
            idx = s - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple diminishing step‑size
        prog = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - prog) + 1e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # tiny random jitter
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# --------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) polishing
# --------------------------------------------------------------
def _expgrad_polish(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient updates for the exact max‑convolution objective."""
    a = a0.copy()
    cur_val = _objective(a)

    eta_start, eta_end = 1.0, 1e-5

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        S = np.where(np.abs(b - max_b) < 1e-15)[0]

        grad = np.zeros_like(a)
        for s in S:
            idx = s - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        prog = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - prog) + eta_end * prog

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)     # avoid exact zeros
        a /= a.sum()

        new_val = _objective(a)
        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[ExpG] new best = {best_val:.6f}")

        # occasional jitter
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.012, size=n), 0.0)
            a /= a.sum()

    return best_a, best_val


# --------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# --------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for several t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# --------------------------------------------------------------
# Larger‑move simulated‑annealing (final phase)
# --------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple annealing with relatively large source→target transfers."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)            # linear cooling
        max_frac = 0.12 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast conv update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # occasional tiny jitter
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and (tries to) minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical limit.
    # --------------------------------------------------------------
    n = 12000

    # --------------------------------------------------------------
    # 0️⃣  Initial population (fast random + systematic Beta grid)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value = {best_val:.6f}")

    # Fast random generation – use roughly 5 % of total budget
    init_deadline = start_time + 0.05 * budget_s
    while time.time() < init_deadline:
        family = rng.integers(4)
        if family == 0:
            cand = _random_beta_candidate(n, rng)
        elif family == 1:
            cand = _exponential_candidate(n, rng)
        elif family == 2:
            p = rng.uniform(-4.0, 4.0)
            cand = _powerlaw_candidate(n, p)
        else:
            cand = _sinusoid_candidate(n)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] new best (rand) = {best_val:.6f}")

    # Systematic Beta grid (small, cheap)
    logp_vals = np.linspace(-7.0, 2.0, 11)
    logq_vals = np.linspace(-7.0, 2.0, 11)
    for lp in logp_vals:
        for lq in logq_vals:
            cand = _beta_grid_candidate(n, lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val
                print(f"[Init] new best (grid) = {best_val:.6f}")

    # Add a couple of handcrafted shapes
    for cand_fun in (_sinusoid_candidate, _cosine_sq_candidate):
        cand = cand_fun(n)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] new best (hand) = {best_val:.6f}")

    print(f"[Init] best after exhaustive init = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Very short Adam warm‑up (≈5 % of budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.05 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val)

    # --------------------------------------------------------------
    # Phase B – Long hill‑climbing (≈70 % of the budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.75 * budget_s
    best_a, best_val = _hillclimb_multi(best_a, n,
                                        deadline=phase_b_deadline,
                                        start_time=start_time,
                                        best_a=best_a,
                                        best_val=best_val,
                                        rng=rng)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase C – Sub‑gradient polishing (≈15 % of the budget)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.90 * budget_s
    best_a, best_val = _subgradient_polish(best_a, n,
                                           deadline=phase_c_deadline,
                                           start_time=start_time,
                                           best_a=best_a,
                                           best_val=best_val,
                                           max_iters=200_000)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase D – Exponentiated‑gradient polishing (≈5 % of the budget)
    # --------------------------------------------------------------
    phase_d_deadline = start_time + 0.95 * budget_s
    best_a, best_val = _expgrad_polish(best_a, n,
                                       deadline=phase_d_deadline,
                                       start_time=start_time,
                                       best_a=best_a,
                                       best_val=best_val,
                                       max_iters=80_000)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase E – Final simulated‑annealing with relatively large moves
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_anneal(best_a, n,
                                          deadline=deadline,
                                          start_time=start_time,
                                          best_a=best_a,
                                          best_val=best_val,
                                          rng=rng)
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation (only once!)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Ensure a plain list of floats is returned
    return best_a.tolist()
