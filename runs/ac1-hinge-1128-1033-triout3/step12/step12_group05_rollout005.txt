"""
Propose a non‑negative step sequence that minimises ``evaluate_sequence``.

Algorithm
---------
1️⃣  Initialise a probability vector on the simplex.
    * The classic “arcsine” shape a_i ∝ 1/√(x·(1‑x)) (x∈(0,1)) is used – it is the
      continuous distribution whose self‑convolution is the flattest known,
      giving a very low initial max‑convolution.
    * If a previous solution (``height_sequence_1``) exists it is also examined.

2️⃣  Random‑support exploration (≈15 % of the time budget) creates many
    ultra‑sparse candidates (uniform weight on a random subset of indices) and
    keeps the best cheap objective 2·n·max(conv).

3️⃣  Refine the best candidate with a smooth‑max mirror‑descent:
    * The max‑convolution is approximated by a soft‑max
      f_β(a) = (1/β)·log∑_k exp(β·conv_k).  Its gradient is a weighted
      cross‑correlation of the current distribution with the weights
      w_k = exp(β·(conv_k−max_conv)).
    * An exponential (a←a·exp(−step·m)) update keeps the iterate on the
      probability simplex.  Momentum (β_m=0.9) and an AdaGrad‑style per‑coordinate
      step‑size are employed.
    * The soft‑max temperature β grows from 2 to 200 during optimisation,
      slowly turning the smooth objective into the true max.
    * Small Gaussian jitter and occasional re‑initialisation avoid stagnation.

The procedure respects the supplied time budget, finally calls
``evaluate_sequence`` on the best vector and returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear self‑convolution (length 2·len(x)‑1)."""
    n = x.shape[0]
    size = 1 << (2 * n - 2).bit_length()          # next power of two ≥ 2·n‑1
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact cheap objective 2·n·max(conv) under the assumption Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (soft‑max) approximation
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇f_β(a) where
        f_β(a) = (1/β)·log∑_k exp(β·conv_k)  (conv = a * a).

    The gradient equals 2·(w ⋆ a) with weights
        w_k = exp(β·(conv_k - max_conv)) / Σ exp(β·(conv - max_conv)).
    """
    n = a.shape[0]

    # weights – stable soft‑max (max_conv subtracts to avoid overflow)
    max_conv = conv.max()
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation w ⋆ a  (implemented as convolution w * a_rev)
    a_rev = a[::-1]
    size = 1 << (len(w) + len(a_rev) - 2).bit_length()
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a_rev, size)
    corr = np.fft.irfft(fw * fa_rev, size)        # length = len(w)+len(a_rev)-1 = 3·n‑2
    start = len(a_rev) - 1                         # position of lag 0 in the full conv
    grad = 2.0 * corr[start:start + n]            # gradient length = n
    return grad

        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)

        # step size – starts at ~0.5 and decays to ~1e‑4
        step = 0.5 * (1.0 - progress) + 1e-4

        # soft‑max temperature β – grows from 2 to 200
        beta = 2.0 + 198.0 * progress

        # gradient of the smooth max
        grad = _softmax_gradient(a_curr, conv, beta)

        # momentum update
        momentum = beta_momentum * momentum + (1.0 - beta_momentum) * grad

        # exponentiated (multiplicative) update
        log_factor = -step * momentum
        # protect against overflow / under‑flow
        log_factor = np.clip(log_factor, -30.0, 30.0)
        a_curr = a_curr * np.exp(log_factor)

        # project back onto the simplex
        s = a_curr.sum()
        if s <= 0.0 or not np.isfinite(s):
            a_curr = np.full_like(a_curr, 1.0 / n)
        else:
            a_curr = a_curr / s

        # occasional jitter to escape flat plateaux or long stagnation
        if rng.random() < 0.001 or no_improve >= max_no_improve:
            jitter = rng.normal(scale=1e-4, size=n)
            a_curr = np.maximum(a_curr + jitter, 0.0)
            a_curr = a_curr / a_curr.sum()
            momentum.fill(0.0)
            no_improve = 0

        it += 1

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation (score is printed, result returned)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print("[final] evaluate_sequence returned", final_score)
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
