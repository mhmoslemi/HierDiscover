import numpy as np

def _next_pow2(x):
    """Return smallest power of two >= x (for FFT size)."""
    return 1 << (x - 1).bit_length()

def _full_convolution(x):
    """Linear convolution of x with itself using real FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _softmax(w):
    """Numerically stable softmax."""
    w_shift = w - np.max(w)
    e = np.exp(w_shift)
    return e / e.sum()

def _max_convolution_and_grad(a):
    """Return (max_conv, subgradient g, index k_max) for current a.

    Subgradient of max entry of a*a w.r.t. a is:
        g_i = 2 * a_{k_max-i}  for indices i where that term exists,
        else 0.

    counterpart = k_max - idx
    g[idx] = 2.0 * a[counterpart]
    return max_val, g, k_max

that maximal entry is g_i = 2·a_{k_max‑i} (zero when the index is out of
range).  The gradient of the objective w.r.t. w is

    ∂L/∂w_j = a_j·(g_j – ⟨g,a⟩) .

We update w with the Adam optimiser, map back to a with a soft‑max, and
keep the best sequence seen.  If no improvement occurs for a while a
small Gaussian jitter is added to w (and Adam moments are reset) to
escape plateaus.  The routine respects the 1000 s time budget and finally
calls the provided ``evaluate_sequence`` exactly once.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used as FFT size)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself using a real FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    w_shift = w - np.max(w)
    e = np.exp(w_shift)
    return e / e.sum()


def _max_convolution_and_grad(a: np.ndarray):
    """
    Return (max_conv, sub‑gradient g, index k_max) for the current a.

    Sub‑gradient of the maximal convolution entry with respect to a_i is:
        g_i = 2·a_{k_max‑i}   if 0 ≤ k_max‑i < n   else 0.
    """
    conv = _full_convolution(a)
    k_max = int(conv.argmax())
    max_val = conv[k_max]

    n = a.shape[0]
    g = np.zeros_like(a)

    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    counterpart = k_max - idx           # always in [0, n‑1]
    g[idx] = 2.0 * a[counterpart]

    return max_val, g, k_max


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Adam‑based mirror descent on the simplex to minimise
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 8000 matches the reference implementation.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise a: try to reuse a previously discovered construction.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, old)
        else:
            a = old.copy()
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total
        print("[init] start from height_sequence_1 (len:", old.shape[0],
              "→", n, ")")
    except Exception:
        a = np.full(n, 1.0 / n)
        print("[init] uniform start, n =", n)

    # --------------------------------------------------------------
    # Mirror‑descent state (soft‑max parametrisation)
    # --------------------------------------------------------------
    eps = 1e-12
    w = np.log(a + eps)          # initialise log‑probabilities

    # Adam hyper‑parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr = 0.3                     # initial learning rate
    lr_min = 0.001
    decay_lr = 0.9995

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    max_val, _, _ = _max_convolution_and_grad(a)
    best_val = 2.0 * n * max_val
    best_a = a.copy()
    print("[init] objective =", best_val)

    last_improve = 0
    iter_cnt = 0
    stagnation_thr = 3000        # after this many non‑improving iters we jitter
    jitter_scale = 0.02

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iter_cnt += 1

        # 1) current convolution and its sub‑gradient
        max_val, g, _ = _max_convolution_and_grad(a)
        obj = 2.0 * n * max_val

        # 2) gradient w.r.t. w (soft‑max mirror map)
        dot = np.dot(g, a)                 # ⟨g,a⟩
        grad_w = a * (g - dot)             # ∂L/∂w

        # 3) Adam update on w
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        t = iter_cnt
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w = w - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # 4) map back to probability simplex
        a = _softmax(w)

        # 5) evaluate the new candidate
        new_max_val, _, _ = _max_convolution_and_grad(a)
        new_obj = 2.0 * n * new_max_val

        # 6) keep the best sequence ever seen
        if new_obj < best_val - 1e-12:
            best_val = new_obj
            best_a = a.copy()
            last_improve = iter_cnt
            print(f"[iter {iter_cnt:06d}] new best = {best_val:.9f} "
                  f"(time {time.time() - start_time:.1f}s)")

        # 7) stagnation handling – small Gaussian jitter in w‑space
        if iter_cnt - last_improve > stagnation_thr:
            w += rng.normal(scale=jitter_scale, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iter_cnt
            print(f"[reset] jitter at iter {iter_cnt} "
                  f"(time {time.time() - start_time:.1f}s)")

        # 8) learning‑rate decay
        lr = max(lr * decay_lr, lr_min)

    # ------------------------------------------------------------------
    # Final secure evaluation (single call to the provided function)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score =", final_score)
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
