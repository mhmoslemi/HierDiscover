def adam_step(x, grad, m, v, t, lr):
    beta1=0.9; beta2=0.999; eps=1e-8
    m = beta1*m + (1-beta1)*grad
    v = beta2*v + (1-beta2)*(grad*grad)
    m_hat = m/(1-beta1**t)
    v_hat = v/(1-beta2**t)
    step = lr * m_hat / (np.sqrt(v_hat)+eps)
    # multiplicative update: x = x * np.exp(-step)
    x_new = x * np.exp(-step)
    x_new = _project_simplex(x_new)
    return x_new, m, v
```

"""
Propose a non‑negative step function that (approximately) minimises
``evaluate_sequence``.

Approach
--------
1.  Build a set of parametric families (power‑law, Beta‑shaped) and
    evaluate them on a fairly fine grid.  The objective is
    ``2·n·max(conv(a,a))`` (the denominator Σa = 1 is enforced by normalising).

2.  Refine the most promising power‑law exponent by a golden‑section search
    (unimodal behaviour in the neighbourhood of the optimum).

3.  A short Adam‑style mirror‑descent run starting from the best candidate
    further reduces the maximal convolution value.

4.  Finally try a few convex blends with the uniform distribution – on some
    instances a tiny amount of smoothing yields a lower bound.

All work respects the given time budget (default 1000 s) and returns a plain
Python list ready for the outer framework.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT, exact objective)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – keeps the vector a valid
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex Σx = 1,  x ≥ 0."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric families
# ----------------------------------------------------------------------


def _monotone_power_shape(p: float, n: int) -> np.ndarray | None:
    """a_i ∝ (i+1)^p, i = 0 … n‑1  (strictly increasing for p>0)."""
    if p <= 0:
        return None
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Beta‑family a_i ∝ (i+1)^{p‑1}·(n‑i)^{q‑1}.
    Allows skewed unimodal shapes (q<1 gives monotone increase).
    """
    if p <= 0 or q <= 0:
        return None
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, p - 1.0) * np.power(n - i + 1, q - 1.0)
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (used in refinement)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 5,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Sub‑gradient of max(b) w.r.t. a.
    ``max_active`` limits the number of active lags that are considered.
    """
    n = a.shape[0]
    max_val = b.max()
    if max_active >= b.size:
        active = np.arange(b.size)
    else:
        active = np.argpartition(b, -max_active)[-max_active:]
        mask = b[active] >= max_val * (1.0 - rel_tol)
        active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad
