def simplex_projection(v):
    # v: numpy array
    # return w: max(v,0) projected onto simplex.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find rho = max i s.t. u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8000

    # init a
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        try:
            a = np.asarray(old_seq, dtype=float)
            a = np.maximum(a, 0.0)
            if a.shape[0] != n:
                x_old = np.linspace(0, 1, a.shape[0])
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_old, a)
                print("[init] interpolated old sequence to length", n)
            else:
                print("[init] using old sequence")
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # Beta(2,2)
        i = np.arange(1, n + 1, dtype=np.float64)
        a = (i ** 1) * ((n - i + 1) ** 1)   # p=q=2 -> i*(n-i+1)
        # Actually Beta(2,2): a_i ∝ i*(n-i+1)
        a = a.astype(float)
        print("[init] generated Beta(2,2) shape")

    # ensure positivity:
    eps = 1e-12
    a = np.maximum(a, eps)
    a = a / a.sum()
    # best tracking
    best_a = a.copy()
    conv = _full_convolution(a)
    best_obj = 2.0 * n * conv.max()
    best_iter = 0
    it = 0
    stagn_counter = 0
    stagn_limit = 5000
    jitter_scale = 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute conv
        conv = _full_convolution(a)
        maxc = conv.max()
        obj = 2.0 * n * maxc
        # update best
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            best_iter = it
            stagn_counter = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}")
        else:
            stagn_counter += 1

        # subgradient for max index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

        # compute step size eta
        progress = (now - start) / (deadline - start)
        eta_start, eta_end = 2.0, 0.001
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # compute g
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g = np.zeros_like(a)
            g[idx] = 2.0 * a[k - idx]
        else:
            g = np.zeros_like(a)

        a = a * np.exp(-eta * g)
        # renormalize
        s = a.sum()
        if s > 0:

            a = np.full_like(a, 1.0 / n)

        # handle stagnation

            a += np.random.normal(scale=jitter_scale, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stagn_counter = 0
            print(f"[reset] jitter added at it={it}")

        it += 1

    # final polishing (maybe additional steps with tiny eta)
    # Use remaining time (if any)
    remaining = deadline - time.time()
    if remaining > 0.5:

        for t in range(extra_steps):
            conv = _full_convolution(a)
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))

            g = np.zeros_like(a)

    except Exception as e:

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full (linear) self‑convolution
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power‑of‑two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv
