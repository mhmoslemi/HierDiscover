"""
Improved step‑function optimiser (≈1000 s budget)
-------------------------------------------------
* Generate a large, diverse set of seed sequences (uniform, triangular,
  cosine‑wave, Gaussian, power‑law / arcsine, many random Beta/Dirichlet draws).
* Keep the best few seeds and run L‑BFGS‑B on a smooth log‑sum‑exp
  approximation of the max‑convolution.  The smoothing parameter ε is
  decreased continuously.
* Polish the current best solution with a long exponentiated‑gradient
  descent on the *exact* max‑convolution objective (projected onto the
  probability simplex by the KL‑mirror update).  The step‑size decays as
  1/√t and an occasional jitter helps escape plateaus.
* Finish with a short high‑precision L‑BFGS run (ε≈1e‑12) and return the
  sequence after a final secure evaluation.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max(conv(a,a)) and gradient of 2·n·smooth_max
    w.r.t. `a`.  The returned gradient already carries the factor 2·n.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (logits → simplex)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex (used only by the old pipeline)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 1500) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on the smoothed objective starting from `a0`."""
    x0 = np.log(a0 + 1e-12)

    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    time_left = deadline - time.time()
    if time_left < 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:       # pragma: no cover
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple convex blend with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑family draw on a log‑scale."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ~1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _triangle_candidate(n: int) -> np.ndarray:
    """Deterministic symmetric triangle (linear increase then decrease)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n - i + 1)
    a /= a.sum()
    return a


def _cosine_candidate(n: int) -> np.ndarray:
    """Half‑sine shape: a_i ∝ sin(π·(i+0.5)/n)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _gaussian_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Centered Gaussian on the index axis, σ sampled on a log‑scale."""
    sigma_factor = rng.uniform(0.05, 0.30)   # σ as a fraction of n
    sigma = sigma_factor * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _powerlaw_candidate(alpha: float, n: int) -> np.ndarray:
    """Power‑law / arcsine‑type shape: a_i ∝ (i·(n‑i+1))^{-α}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i * (n - i + 1)) ** (-alpha)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (exact max‑convolution objective)
# ----------------------------------------------------------------------
def _exponentiated_gradient_polish(a_start: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   best_a: np.ndarray,
                                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Long EG run on the exact objective 2·n·max(conv(a,a)).
    The update is performed in log‑space (mirror descent w.r.t. KL).
    """
    # initialise log‑probabilities
    log_a = np.log(a_start + 1e-50)
    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # reconstruct a from log_a (stable)
        log_a_shift = log_a - np.max(log_a)
        a = np.exp(log_a_shift)
        a_sum = a.sum()
        a /= a_sum

        # evaluate current objective
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[EG] iter {it:6d} new best = {best_val:.6f}")

        # compute the binary mask of maximal convolution entries
        b = _full_convolution(a)
        max_b = b.max()
        d = (b == max_b).astype(float)            # shape (2n‑1,)

        # gradient of the exact objective (includes factor 2·n)
        d_rev = d[::-1]
        conv_d = _conv_fft(a, d_rev)
        segment = conv_d[n - 1:2 * n - 1]
        grad = 4.0 * n * segment[::-1]            # shape (n,)

        # step‑size (decaying, also normalised by max‑grad to stay stable)
        eta = 0.5 / ((np.sqrt(it + 1)) * (np.max(np.abs(grad)) + 1e-12))

        # mirror‑descent (exponentiated‑gradient) update in log‑space
        log_a = log_a - eta * grad

        # occasional jitter to escape tiny plateaus
        if it % 1000 == 0 and np.random.rand() < 0.05:
            log_a += np.random.normal(scale=0.01, size=n)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and that minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a richer search space.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # 0️⃣  Initial candidate sweep (exact objective)
    # ------------------------------------------------------------------
    candidates: list[tuple[float, np.ndarray]] = []

    # 0.1  Previous solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= a0.sum()
        val = _objective(a0)
        candidates.append((val, a0))
        print(f"[Init] interpolated previous solution, value={val:.6f}")
    except Exception:
        pass

    # 0.2  Uniform baseline
    uniform = np.full(n, 1.0 / n)
    candidates.append((_objective(uniform), uniform))

    # 0.3  Deterministic families
    candidates.append((_objective(_triangle_candidate(n)), _triangle_candidate(n)))
    candidates.append((_objective(_cosine_candidate(n)), _cosine_candidate(n)))
    # arcsine‑like power‑law shapes for several α
    for alpha in (0.25, 0.35, 0.45, 0.5, 0.6):
        cand = _powerlaw_candidate(alpha, n)
        candidates.append((_objective(cand), cand))

    # 0.4  Random Gaussian seeds
    for _ in range(120):
        cand = _gaussian_candidate(n, rng)
        candidates.append((_objective(cand), cand))

    # 0.5  Random Beta draws (more than before)
    for _ in range(800):
        cand = _random_beta_candidate(n, rng)
        candidates.append((_objective(cand), cand))

    # 0.6  Dirichlet draws with various concentrations
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            cand = rng.dirichlet(np.full(n, alpha))
            candidates.append((_objective(cand), cand))

    # keep the best few for the optimisation phases
    candidates.sort(key=lambda t: t[0])
    top_k = min(15, len(candidates))
    top_candidates = [cand for _, cand in candidates[:top_k]]
    best_a, best_val = candidates[0][1], candidates[0][0]
    print(f"[Init] best after sweep: value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – L‑BFGS on smooth‑max (multiple restarts, annealing ε)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    eps_start = 5e-1
    eps_final = 1e-12

    while time.time() < phase_a_deadline and time.time() < deadline:
        prog = (time.time() - start_time) / (phase_a_deadline - start_time)
        eps = eps_start * (eps_final / eps_start) ** prog
        for a0 in top_candidates:
            if time.time() >= phase_a_deadline:
                break
            best_a, best_val = _lbfgs_optimize(
                a0, n, deadline, start_time,
                best_a, best_val,
                eps=eps, max_iter=2000)
            best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – Exponentiated‑gradient polishing on the exact objective
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.90 * budget_s
    best_a, best_val = _exponentiated_gradient_polish(
        best_a, n, deadline=phase_b_deadline,
        best_a=best_a, best_val=best_val)

    # one final uniform blend (helps smooth a few remaining spikes)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Final high‑precision L‑BFGS (tiny ε)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(
            best_a, n, deadline, start_time,
            best_a, best_val,
            eps=1e-12, max_iter=1200)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return plain Python list of floats
    return best_a.tolist()
