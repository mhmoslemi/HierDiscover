def _fine_hillclimb(a0, n, deadline, start_time, best_a, best_val, rng):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12
    max_frac_start, max_frac_end = 0.2, 0.0001
    temp_start, temp_end = 0.01, 1e-12
    no_improve = 0
    iteration = 0
    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog
        # compute contributions as in _hillclimb:
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            src_probs = contrib / total
            i = rng.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))
        if a[i] <= eps: 
            continue
        # target weight inversely proportional to contrib, but also ensure diff:
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = rng.choice(n, p=tgt_weights / wsum)
        else:
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue
        # incremental update:
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta
        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[FineHill] new best = {best_val:.6f}")
        else:
            no_improve += 1
        # occasional jitter:
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.001, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
        iteration += 1
    return best_a, best_val

# convolution helpers
def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

def _smooth_max_and_grad(a: np.ndarray, eps: float):
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()
    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)
    segment = conv_ab[n - 1:2 * n - 1]
    grad_a = 4.0 * n * segment[::-1]
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a

def _project_simplex(v: np.ndarray) -> np.ndarray:
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Candidate families
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    log_r = rng.uniform(-8.0, -0.1)
    r = np.exp(log_r)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a

def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a /= a.sum()
    return a

def _mixture_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    w = rng.random()
    a1 = _random_beta_candidate(n, rng)
    a2 = _random_beta_candidate(n, rng)
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a

def _u_shape_candidate(n: int, alpha: float) -> np.ndarray:
    """U‑shaped candidate: high at ends, low in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-alpha) + (n - i) ** (-alpha)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Optimisation primitives
def _adam_optimize(a0: np.ndarray, n: int, deadline: float,
                   start_time: float, best_a: np.ndarray, best_val: float):
    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.001
    eps_start, eps_end = 0.5, 1e-12
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 6000
    while time.time() < deadline:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1
        progress = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress
        _, grad = _smooth_max_and_grad(a, eps)
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0
    return best_a, best_val

def _hillclimb(a0: np.ndarray, n: int, deadline: float,
               start_time: float, best_a: np.ndarray, best_val: float,
               rng: np.random.Generator):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12
    max_frac_start, max_frac_end = 0.9, 0.001
    temp_start, temp_end = 0.03, 1e-9
    no_improve = 0
    IMPROVE_LIMIT = 50000
    iteration = 0
    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            src_probs = contrib / total
            i = rng.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))
        if a[i] <= eps:
            continue
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = rng.choice(n, p=tgt_weights / wsum)
        else:
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta
        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1
        if no_improve > IMPROVE_LIMIT:
            # re‑perturb around current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
        iteration += 1
    return best_a, best_val

def _subgradient_polish(a0: np.ndarray, n: int, deadline: float,
                        start_time: float, best_a: np.ndarray, best_val: float,
                        max_iters: int = 200_000):
    a = a0.copy()
    cur_val = _objective(a)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        progress = (time.time() - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress
        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        new_val = _objective(a_new)
        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)
    return best_a, best_val

def _expgrad_polish(a0: np.ndarray, n: int, deadline: float,
                    start_time: float, best_a: np.ndarray, best_val: float,
                    max_iters: int = 120_000):
    a = a0.copy()
    cur_val = _objective(a)
    eta_start, eta_end = 1.0, 1e-5
    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        progress = (time.time() - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        new_val = _objective(a)
        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[ExpG] new best = {best_val:.6f}")
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.018, size=n), 0.0)
            a /= a.sum()
    return best_a, best_val

def _random_anneal(a0: np.ndarray, n: int, deadline: float,
                   start_time: float, best_a: np.ndarray, best_val: float,
                   rng: np.random.Generator):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02
    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta
        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
    return best_a, best_val

def _blend_with_uniform(a: np.ndarray, n: int, cur_best_val: float):
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step‑function that minimises the evaluation score.

    The algorithm builds a rich initial population (beta, exponential, power‑law,
    mixed families and a few handcrafted shapes), then runs a cascade of
    optimisation phases:
      * Adam on a smooth max‑convolution surrogate,
      * aggressive hill‑climbing with source→target moves,
      * projected sub‑gradient descent,
      * exponentiated‑gradient polishing,
      * a large‑step simulated‑annealing,
      * a final fine‑grained hill‑climb for tiny adjustments.

    Throughout the run the best seen sequence is kept and periodically blended
    with the uniform distribution to smooth spikes.  The final list is returned
    (its entries sum to 1, all non‑negative) and can be passed directly to
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    n = 8000

    # ---- initial best = uniform ----
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # try to import a previous solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # ---- systematic families ----
    # random beta draws
    for _ in range(4000):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # systematic beta grid
    logp_vals = np.linspace(-7.5, 2.0, 15)
    logq_vals = np.linspace(-7.5, 2.0, 15)
    for lp in logp_vals:
        for lq in logq_vals:
            cand = _beta_grid_candidate(n, lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # Dirichlet draws
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(30):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # exponential family
    for _ in range(800):
        cand = _exponential_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # power‑law family
    for p in np.linspace(-4.0, 4.0, 25):
        cand = _powerlaw_candidate(n, p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # mixture of two Betas
    for _ in range(1200):
        cand = _mixture_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # handcrafted sinusoid & cosine‑square
    i = np.arange(n, dtype=np.float64)
    cand = np.sin(np.pi * (i + 0.5) / n)
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    cand = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    # U‑shape candidates (high at ends)
    for alpha in np.linspace(0.5, 5.0, 12):
        cand = _u_shape_candidate(n, alpha)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] best after exhaustive init = {best_val:.6f}")

    # ---- Phase A: Adam ----
    phase_a_deadline = start_time + 0.20 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ---- Phase B: Hill‑climbing ----
    phase_b_deadline = start_time + 0.55 * budget_s
    best_a, best_val = _hillclimb(best_a, n,
                                  deadline=phase_b_deadline,
                                  start_time=start_time,
                                  best_a=best_a,
                                  best_val=best_val,
                                  rng=rng)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ---- Phase C: Sub‑gradient polish ----
    phase_c_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _subgradient_polish(best_a, n,
                                           deadline=phase_c_deadline,
                                           start_time=start_time,
                                           best_a=best_a,
                                           best_val=best_val,
                                           max_iters=500_000)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ---- Phase D: Exponentiated‑gradient polish ----
    phase_d_deadline = start_time + 0.95 * budget_s
    best_a, best_val = _expgrad_polish(best_a, n,
                                       deadline=phase_d_deadline,
                                       start_time=start_time,
                                       best_a=best_a,
                                       best_val=best_val,
                                       max_iters=200_000)
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ---- Phase E: Simulated annealing (large moves) ----
    if time.time() < deadline:
        best_a, best_val = _random_anneal(best_a, n,
                                          deadline=deadline,
                                          start_time=start_time,
                                          best_a=best_a,
                                          best_val=best_val,
                                          rng=rng)
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ---- Phase F: Fine‑grained hill‑climb ----

"""Optimise a non‑negative step‑function to minimise the given evaluation score.

The search consists of:
  * a diverse deterministic/random initial population (Beta, exponential,
    power‑law, U‑shapes, sinusoid, mixture families);
  * Adam optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution;
  * aggressive hill‑climbing moves that shift mass from indices that largely
    contribute to the current maximal convolution entry towards indices that
    contribute little;
  * projected sub‑gradient descent on the exact objective;
  * exponentiated‑gradient (mirror‑descent) polishing;
  * a simulated‑annealing phase with larger source→target transfers;
  * a final fine‑grained hill‑climb that uses a much smaller step size and a
    very low temperature to polish the last digits.

All intermediate states are blended with the uniform distribution to smooth
spikes.  The best sequence found before the deadline is returned as a plain
list of floats (its entries sum to 1 and are non‑negative)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a given a (sum(a)=1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Smooth approximation of max(conv(a,a)) and gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Beta‑shaped vector on a log‑scale."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    """Deterministic Beta candidate for given log‑parameters."""
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Truncated exponential shape a_i ∝ r^i with r∈(0,1)."""
    log_r = rng.uniform(-8.0, -0.1)
    r = np.exp(log_r)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i ∝ (i+1)^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a /= a.sum()
    return a


def _mixture_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two Beta families."""
    w = rng.random()
    a1 = _random_beta_candidate(n, rng)
    a2 = _random_beta_candidate(n, rng)
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


def _u_shape_candidate(n: int, alpha: float) -> np.ndarray:
    """U‑shaped candidate: mass concentrated near both ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-alpha) + (n - i) ** (-alpha)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex for L(a)=2·n·smooth_max(a)."""
    w = np.log(a0 + 1e-12)          # work in log‑softmax space

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.001      # a slightly larger start LR
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 6000

    while time.time() < deadline:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        progress = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        _, grad = _smooth_max_and_grad(a, eps)   # grad already contains 2·n factor

        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)

        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (source→target moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Local search with simulated‑annealing acceptance."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.9, 0.001   # more aggressive early moves
    temp_start, temp_end = 0.03, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 50000
    iteration = 0

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index: proportional to its contribution
        if total > 0:
            src_probs = contrib / total
            i = rng.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index: favour low contributions
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = rng.choice(n, p=tgt_weights / wsum)
        else:
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation (it={iteration}), re‑perturb")
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        progress = (time.time() - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------
def _expgrad_polish(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient updates for the exact max‑convolution objective."""
    a = a0.copy()
    cur_val = _objective(a)

    eta_start, eta_end = 1.0, 1e-5

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        progress = (time.time() - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        new_val = _objective(a)

        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[ExpG] new best = {best_val:.6f}")

        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.018, size=n), 0.0)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Larger‑move simulated‑annealing (final phase)
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple annealing with relatively large source→target transfers."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)            # linear cooling
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val

# Fine‑grained hill‑climbing (tiny steps) for the final polish

def _fine_hillclimb(a0: np.ndarray,

    """Very small source→target moves with a low temperature."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.2, 0.0001
    temp_start, temp_end = 0.01, 1e-12

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            src_probs = contrib / total
            i = rng.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = rng.choice(n, p=tgt_weights / wsum)
        else:
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[FineHill] new best = {best_val:.6f}")

        if rng.random() < 0.005:
            a = np.maximum(a + rng.normal(scale=0.001, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and (tries to) minimise ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Rich initial population (random + systematic)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # try to reuse a previously discovered solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # ----- random Beta draws (more than before) -----
    for _ in range(4000):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- systematic Beta grid -----
    logp_vals = np.linspace(-7.5, 2.0, 15)
    logq_vals = np.linspace(-7.5, 2.0, 15)
    for lp in logp_vals:
        for lq in logq_vals:
            cand = _beta_grid_candidate(n, lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ----- Dirichlet draws for several concentrations -----
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(30):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ----- exponential family -----
    for _ in range(800):
        cand = _exponential_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- power‑law family -----
    for p in np.linspace(-4.0, 4.0, 25):
        cand = _powerlaw_candidate(n, p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- mixture of two Betas -----
    for _ in range(1200):
        cand = _mixture_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- handcrafted sinusoid / cosine‑square (already in original) -----
    i = np.arange(n, dtype=np.float64)
    cand = np.sin(np.pi * (i + 0.5) / n)
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    cand = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    # ----- U‑shaped candidates (high at both ends) -----
    for alpha in np.linspace(0.5, 5.0, 12):
        cand = _u_shape_candidate(n, alpha)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] best after exhaustive init = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam optimisation (≈20 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.20 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase B – Hill‑climbing (≈35 % of the budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * budget_s
    best_a, best_val = _hillclimb(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        rng=rng,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase C – Sub‑gradient polishing (≈25 % of the budget)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase_c_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=500_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase D – Mirror‑descent (exponentiated‑gradient) polishing
    # --------------------------------------------------------------
    phase_d_deadline = start_time + 0.95 * budget_s
    best_a, best_val = _expgrad_polish(
        best_a, n,
        deadline=phase_d_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=200_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase E – Final simulated‑annealing with larger moves
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_anneal(
            best_a, n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            rng=rng,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase F – Fine‑grained hill‑climb (tiny steps)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _fine_hillclimb(
            best_a, n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            rng=rng,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation (for logging only)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure a plain list of floats is returned
    return best_a.tolist()
