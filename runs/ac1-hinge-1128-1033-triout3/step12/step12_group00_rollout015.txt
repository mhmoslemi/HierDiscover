"""
Optimiser for the “step‑function” challenge.

The aim is to minimise
        2·n·max(conv(a,a)) / (sum(a)**2)
for a non‑negative vector a (the “heights”).  The optimiser proceeds in
four stages within the 1000 s budget:

0️⃣  Very diverse initialisation – random Beta, Dirichlet, sinusoidal and
    piece‑wise families; the best few candidates are kept.

1️⃣  Smooth‑max L‑BFGS – the non‑smooth max is replaced by a log‑sum‑exp
    surrogate (ε‑smooth max).  The variables are re‑parameterised via a
    soft‑max so that the simplex constraints (a≥0, Σa=1) are satisfied
    automatically.  Several restarts with a schedule that drives ε from
    0.5 down to 1e‑12 are performed.

2️⃣  Projected sub‑gradient polishing – the exact (non‑smooth) objective
    is minimised by a simple sub‑gradient descent with a decreasing step
    size and occasional random jitter to escape plateaus.

3️⃣  Fine hill‑climbing – a cheap coordinate‑transfer heuristic refines
    the best solution further.

The code is completely self‑contained, uses only NumPy/SciPy and respects
the 1 CPU limit.  Progress is printed to stdout; the final list of floats
is returned and also evaluated with the supplied ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1, i.e. a lies on the probability simplex)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a simplex‑vector a."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient.
# Returned gradient already contains the factor 2·n.
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """Smooth approximation of max(conv(a,a)) and gradient w.r.t. a."""
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick for numerical stability
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # probability vector over shifts
    # gradient of 2·n·smooth_max
    # conv of a with reversed p gives d/da of sum_k p_k * (a*a)_k
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # keep the n elements that correspond to the “forward” part
    segment = conv_ab[n - 1:2 * n - 1]            # size n
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → simplex) and Euclidean projection onto simplex
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (Σ=1, ≥0)."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x|x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random candidate generators
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑family shape (i^{p-1} (n-i+1)^{q-1}) on the simplex."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_dirichlet_candidate(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    a = rng.dirichlet(np.full(n, alpha))
    return a


def _random_sine_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sin² shape: a_i ∝ sin²(π·i/(n+1)) + tiny noise."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1)) ** 2
    a += rng.uniform(0.0, 1e-6, size=n)          # avoid exact zeros
    a /= a.sum()
    return a


def _random_piecewise_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Three‑segment linear piecewise shape (low‑mid‑high)."""
    # random break points
    cuts = sorted(rng.integers(1, n, size=2))
    left, right = cuts
    # random heights
    h1 = rng.uniform(0.1, 2.0)
    h2 = rng.uniform(0.1, 2.0)
    h3 = rng.uniform(0.1, 2.0)
    a = np.empty(n, dtype=np.float64)
    a[:left] = np.linspace(h1, h2, left, endpoint=False)
    a[left:right] = np.linspace(h2, h3, right - left, endpoint=False)
    a[right:] = np.linspace(h3, h1, n - right)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS on smooth‑max (multiple restarts)
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 1200) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on the smooth surrogate from start point a0."""
    x0 = np.log(a0 + 1e-12)                     # log‑its → unconstrained

    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)               # chain rule through softmax

    # enough time left?
    if deadline - time.time() < 0.8:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:                     # pragma: no cover
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true (non‑smooth) objective."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: contribution of every maximal shift
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)                 # positions of a that overlap
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple linear step‑size schedule (more aggressive early)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to avoid tiny plateaus
        if it % 500 == 0 and np.random.rand() < 0.02:
            a = np.maximum(a + np.random.normal(scale=0.01, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing (coordinate transfers)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """Very cheap local search: transfer a tiny amount between random coordinates."""
    a = a0.copy()
    cur_val = _objective(a)

    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 0.02   # roughly ±1 % of an entry
        if a[i] - delta < 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta                         # sum unchanged
        val = _objective(cand)
        if val < cur_val:
            a = cand
            cur_val = val
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[Hill]   new best = {best_val:.6f}")

        # occasional larger random reshuffle
        if np.random.rand() < 0.001:
            k = rng.integers(0, n)
            cand = a.copy()
            cand[k] = max(0.0, cand[k] + rng.normal(scale=0.02))
            cand = _project_simplex(cand)
            val = _objective(cand)
            if val < cur_val:
                a = cand
                cur_val = val
                if val < best_val:
                    best_val, best_a = val, a.copy()
                    print(f"[Hill]   new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Convex blend with uniform distribution (helps to smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:                      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Ensure we return a plain list of floats
    return best_a.tolist()
