Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().  The optimisation proceeds in three stages:

1.  Adam optimisation of a smooth approximation of the max‑convolution.
    The temperature of the smooth‐max is increased linearly so the
    optimisation first sees a smooth landscape and then focuses on the
    true maximum.  Restarts with small random jitter escape plateaus.

2.  A cheap projected sub‑gradient refinement that uses the exact
    sub‑gradient (i.e. a one‑hot weight at the active convolution index)
    and a simple step‑size adaptation.

3.  A final random local search that moves a tiny amount of mass from
    one coordinate to another; any improvement is kept.

All vectors are kept on the simplex (non‑negative, sum = 1) so the
objective 2·n·max(conv) is exactly the value returned by
evaluate_sequence() (the L1‑normalisation cancels out).  The routine
stops as soon as the target bound 1.503 is reached or the time budget
expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution x * x (full length 2*len(x)-1) using FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of weight vector w (length 2n‑1) with a (length n).
    Returns a vector of length n:
        (w ⋆ a)_i = Σ_k w_k * a_{k-i} .
    Implemented via FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1            # = (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Given a full self‑convolution array, return the evaluation value."""
    return 2.0 * n * conv.max()
