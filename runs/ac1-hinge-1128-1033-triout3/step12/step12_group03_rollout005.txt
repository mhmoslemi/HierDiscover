"""
Hybrid optimiser for the self‑convolution max problem.

The algorithm works in three phases:

1️⃣  Continuation with a smooth‑max objective.  We store the
    probability vector as soft‑max logits and minimise the
    smoothed quantity 2·n·∑_k w_k·(a*a)_k using L‑BFGS‑B.
    λ (the smoothness temperature) is increased in stages,
    ending with the exact max (a one‑hot weight).

2️⃣  A long projected sub‑gradient sweep on the true
    max‑objective (w one‑hot).  The step size is adapted
    automatically and many thousands of iterations are
    performed – far more than in the reference code.

3️⃣  A cheap stochastic local‑search that moves tiny amounts
    of probability mass between random coordinates and keeps any
    improvement.  This helps escape very shallow basins.

All work is bounded by the supplied time budget (default 1000 s)
and the final sequence is evaluated by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2*len(x)-1, computed via FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k-i}
    (cross‑correlation of weight vector w with a).
    Returned vector has length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1                         # extract indices n‑1 … n‑1+(n‑1)
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_weights(conv: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Return the weight vector w used for the (smooth) max of conv,
    together with conv_max = max(conv).
    For large λ the weight collapses to a one‑hot at the maximal index.
    """
    conv_max = conv.max()
    if lam > 2000.0:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        # conv_max is the maximum, so (conv - conv_max) ≤ 0.
        exp_vals = np.exp(lam * (conv - conv_max))
        w = exp_vals / exp_vals.sum()
    return w, conv_max

            mx = np.max(z)
            expz = np.exp(z - mx)
            a = expz / expz.sum()

            conv = _full_convolution(a)
            w, conv_max = _smooth_weights(conv, lam)

            # smoothed objective (exact max when λ is huge)
            if lam > 2000.0:
                obj = 2.0 * n * conv_max
            else:
                obj = 2.0 * n * np.dot(w, conv)

            # gradient wrt a : g = 2 * cross‑correlation(w, a)
            g_a = 2.0 * _cross_corr(w, a)

            # back‑prop through the soft‑max
            dot_ag = np.dot(a, g_a)
            grad_z = a * (g_a - dot_ag)          # ∂obj/∂z

            return obj, grad_z

        # allow only as many L‑BFGS iterations as the remaining budget permits
        time_left = deadline - time.time()
        max_iter = max(5, int(min(200, time_left / 0.02)))   # very cheap per iteration
        res = minimize(fun=fun,
                       x0=logits,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter, 'disp': False})
        logits = res.x
        # obtain the corresponding probability vector
        mx = np.max(logits)
        expz = np.exp(logits - mx)
        a = expz / expz.sum()
        cur_val = 2.0 * n * _full_convolution(a).max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[lbfgs λ={lam:.0f}] new best = {best_val:.9f}")

        # ------------------------------------------------------------------
        # 3.  Projected sub‑gradient refinement for the exact max‑objective
        # ------------------------------------------------------------------
        step = 0.02
        a_ref = best_a.copy()
        obj_ref = best_val
        sub_iters = 0
        while time.time() < deadline and sub_iters < 8000:
            sub_iters += 1
            conv = _full_convolution(a_ref)
            kmax = int(np.argmax(conv))
            # sub‑gradient for the true max (one‑hot weight at kmax)
            w_one = np.zeros_like(conv)
            w_one[kmax] = 1.0
            g = 2.0 * _cross_corr(w_one, a_ref)

            a_trial = _project_simplex(a_ref - step * g)
            obj_trial = 2.0 * n * _full_convolution(a_trial).max()

            if obj_trial < obj_ref - 1e-13:
                a_ref = a_trial
                obj_ref = obj_trial
                best_a = a_ref.copy()
                best_val = obj_ref
                step = min(step * 1.1, 0.5)
                print(f"[subgrad] improved to {best_val:.9f}")
                if best_val < 1.503:
                    break
            else:
                step *= 0.7  # shrink step on failure

        # store the result of this λ‑stage
        logits = np.log(best_a + eps)

        if best_val < 1.503:
            break

    ######################################################################
    # 4.  Final stochastic local search – tiny mass moves
    ######################################################################
    a = best_a.copy()
    cur_val = best_val
    while time.time() < deadline:
        # pick a source index proportionally to its probability
        src = np.random.choice(n, p=a)
        if a[src] < 1e-12:
            continue
        # pick a destination distinct from src
        dst = np.random.randint(0, n - 1)
        if dst >= src:
            dst += 1

        # move a tiny amount of mass (bounded to keep non‑negativity)
        delta = min(a[src] * 0.3, 0.003)          # at most a few‑‰ of the total mass
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        obj_new = 2.0 * n * _full_convolution(a_new).max()
        if obj_new < cur_val - 1e-12:
            a = a_new
            cur_val = obj_new
            best_a = a.copy()
            best_val = cur_val
            print(f"[move] improved to {best_val:.9f}")
            if best_val < 1.503:
                break

    ######################################################################
    # 5.  Final secure evaluation
    ######################################################################
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
    return best_a.tolist()
