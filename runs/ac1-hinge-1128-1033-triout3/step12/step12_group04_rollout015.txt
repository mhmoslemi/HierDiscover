"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The algorithm is a smoothed‐max gradient descent on the objective

    F(a) = 2·n·max_k (a*a)_k / (∑a)^2 ,

with a ∈ ℝ_+^n and ∑a = 1.
A softmax (log‑sum‑exp) with a temperature α approximates the max;
its gradient can be written as a single convolution, which we compute
with FFTs.  An Adam‑style adaptive step and a growing temperature
help to escape local plateaus.  After every update we project back onto
the probability simplex (clip negatives and renormalise).  When the
objective has not improved for a while a small Gaussian jitter is
added to re‑inject diversity.  The routine respects a 1000 s budget,
keeps the best sequence seen and finally evaluates it with the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of ``x`` with itself (real FFT).
    Returns an array of length 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _gradient_from_softmax(a: np.ndarray, w: np.ndarray,
                          size_grad: int) -> np.ndarray:
    """
    Compute ∇_a  ( 1/α·log∑_k exp(α·(a*a)_k) )  using convolution.
    ``w`` must be the softmax weights (sum to 1) over the 2·n‑1
    convolution entries.  The gradient is 2·∑_k w_k·a_{k‑i}.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reverse a once
    fa = np.fft.rfft(a_rev, size_grad)
    fw = np.fft.rfft(w, size_grad)
    # full convolution length = n + (2n‑1) – 1 = 3n‑2
    L = 3 * n - 2
    conv_full = np.fft.irfft(fa * fw, size_grad)[:L]
    # slice indices n‑1 … n‑1+n‑1 give the required sums
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex {w≥0, Σw=1}. """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

    # --------------------------------------------------------------
    # Soft‑max temperature schedule
    # --------------------------------------------------------------
    alpha = 0.2               # start very smooth
    alpha_max = 15.0
    alpha_inc = 0.001         # linear increase

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a).max() / (a.sum() ** 2)
    print("[init] objective =", best_val)

    iteration = 0
    last_improve = 0
    stagnation_limit = 5000   # after this many loops without improvement we jitter

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- 1) self‑convolution and raw objective -----
        fx = np.fft.rfft(a, size_conv)
        conv = np.fft.irfft(fx * fx, size_conv)[: 2 * n - 1]
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv / (a.sum() ** 2)

        # ----- 2) keep the best seen -----
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = iteration
            print(f"[iter {iteration:06d}] new best = {best_val:.9f} "
                  f"(time {time.time() - start_time:.1f}s)")

        # ----- 3) soft‑max (smoothed max) -----
        #   w_k ∝ exp(α·conv_k) , Σ w_k = 1
        scaled = alpha * conv
        scaled -= scaled.max()               # numerical stability
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- 4) gradient of the smoothed objective -----
        grad = _gradient_from_softmax(a, w, size_grad)

        # ----- 5) Adam adaptive step -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ----- 6) enforce feasibility (non‑negative, sum = 1) -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # Optional sparsification – very small entries are set to zero
        tiny = 1e-7
        a[a < tiny] = 0.0
        a /= a.sum()          # renormalise after sparsification

        # ----- 7) schedule updates -----
        lr = max(lr * decay_lr, 0.01)       # keep a modest floor
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        # ----- 8) stagnation handling -----
        if iteration - last_improve > stagnation_limit:
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = iteration
            print(f"[reset] jitter at iter {iteration} "
                  f"(time {time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # Final secure evaluation (single call)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score =", final_score)
    except Exception as exc:                                   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
