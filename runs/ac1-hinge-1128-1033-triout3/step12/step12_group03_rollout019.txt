        c = -np.ones(n, dtype=np.float64)

def build_conv_matrix(f):
    n = f.shape[0]
    m = 2 * n - 1
    A = np.empty((m, n), dtype=np.float64)
    # using for loop assigning slices
    for k in range(m):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # Use slice inclusive: i_start..i_end
        A[k, i_start:i_end+1] = f[i_start:i_end+1]
    return A

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine n based on time and memory. Choose n = 1500 or maybe 2000.
    n = 1500
    target_sum = math.sqrt(2.0 * n)  # S

    # Initialize f:
    try:
        old_seq = np.asarray(height_sequence_1, dtype=float)   # use previous if exists
        # truncate/resize to n via interpolation
        old_seq = np.clip(old_seq, 0, None)
        if old_seq.size > 1:
            # Interpolate to length n
            old_x = np.linspace(0, 1, old_seq.size)
            new_x = np.linspace(0, 1, n)
            f = np.interp(new_x, old_x, old_seq)
            f_sum = f.sum()
            if f_sum > 0:
                f = f * (target_sum / f_sum)
            else:
                f = np.full(n, target_sum / n)
            print("[init] interpolated from previous")
        else:
            raise Exception
    except Exception:
        f = np.full(n, target_sum / n)
        print("[init] uniform start")

    def compute_M(vec):
        conv = np.convolve(vec, vec)
        return conv.max()

    best_f = f.copy()
    best_M = compute_M(f)
    print(f"[init] M = {best_M:.9f}")

    iter_count = 0
    # algorithm parameters
    t_step = 0.2
    # a small step for mixing
    while time.time() < deadline:
        M = compute_M(f)
        # Build constraint matrix
        A = np.empty((2 * n - 1, n), dtype=np.float64)
        for k in range(2 * n - 1):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            A[k, i_start:i_end + 1] = f[i_start:i_end + 1]

        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)  # maximize sum

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            print(f"[LP] iteration {iter_count} failed: {res.message}")
            # possibly break
            break

        b_vec = res.x
        B_sum = b_vec.sum()
        # scaling factor
        alpha = target_sum / B_sum if B_sum > 0 else 0.0
        g = b_vec * alpha

        # Ensure positivity
        # Combine with small mixing factor
        # Adaptive t: reduce if M does not improve
        t = t_step
        improved = False
        # attempt to mix and maybe reduce t
        while t > 1e-4:
            f_candidate = (1 - t) * f + t * g
            # due to mixing sums remain target_sum (since both have same sum). No need to renorm.
            M_candidate = compute_M(f_candidate)
            if M_candidate < M - 1e-12:
                f = f_candidate
                M = M_candidate
                improved = True
                break
            else:
                t *= 0.5

        if improved:
            if M < best_M - 1e-12:
                best_M = M
                best_f = f.copy()
                print(f"[iter {iter_count}] new best M = {best_M:.9f}")
        else:
            # Not improved; optionally add small perturbation to f to escape
            f += np.random.normal(scale=target_sum * 0.001, size=n)
            # project onto simplex of sum = target_sum
            f = np.maximum(f, 0.0)
            f_sum = f.sum()
            if f_sum > 0:
                f = f * (target_sum / f_sum)
            print(f"[iter {iter_count}] no improvement, perturbed")
        iter_count += 1

        # early exit if best_M < target
        if best_M < 1.503:
            print("[target] achieved", best_M)
            break

    # final evaluation
    try:
        final_score = evaluate_sequence(best_f.tolist())
        print(f"[final] evaluate_sequence score = {final_score:.9f}")
    except Exception as e:
        print("[final] evaluate error:", e)

    return best_f.tolist()

"""
Hybrid L‑BFGS / sub‑gradient optimiser for the self‑convolution maximum problem.

*   Work in the probability simplex (∑ a_i = 1).  The evaluation function
    is scale‑invariant, so the final sequence is simply scaled to the
    canonical sum √(2 n) before returning.

*   The objective is approximated by a smooth “soft‑max’’ of the
    convolution values.  A schedule of increasing temperature λ turns
    the smooth approximation into the exact max, allowing a fast
    L‑BFGS optimisation in the early phases and a sub‑gradient finish.

*   A final projected sub‑gradient sweep (as used in the reference code)
    squeezes a little more out of the best candidate.

*   The routine respects a 1000 s budget, falls back to a simple Adam
    optimiser if SciPy is unavailable, and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global parameters used by the loss/gradient function (required for L‑BFGS)
# ----------------------------------------------------------------------
GLOBAL_N = 0          # length of the sequence (set at runtime)
GLOBAL_LAMBDA = 5.0   # current temperature for the smooth‑max

# ----------------------------------------------------------------------
# Helper utilities – all at module level (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(v: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = v.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fv = np.fft.rfft(v, size)
    conv = np.fft.irfft(fv * fv, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of weight vector w (length 2n‑1) with a (length n).
    Result has length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1          # = (2n‑1)+n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Loss and gradient (used by L‑BFGS).  They read GLOBAL_N / GLOBAL_LAMBDA.
# ----------------------------------------------------------------------
def _loss_and_grad(z: np.ndarray):
    """
    Returns (loss, gradient) for the current temperature λ.
    The loss is 2 n · smooth_max(conv) and the gradient is the
    corresponding (approximate) sub‑gradient w.r.t. the logits z.
    """
    n = GLOBAL_N
    lam = GLOBAL_LAMBDA

    # ---- probability vector (soft‑max) ----
    a = _softmax(z)

    # ---- full self‑convolution ----
    conv = _full_convolution(a)

    # ---- smooth‑max weights ----
    if lam > 2000.0:                     # exact max (single active index)
        kmax = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[kmax] = 1.0
        weighted_sum = conv[kmax]       # this is the true max
    else:
        conv_max = conv.max()
        exp_vals = np.exp(lam * (conv - conv_max))
        w = exp_vals / exp_vals.sum()
        weighted_sum = np.dot(w, conv)   # smooth approximation of the max

    # ---- gradient w.r.t. a (ignoring the λ‑dependence of w, as in the
    #      reference implementation) ----
    g_raw = 2.0 * _cross_corr(w, a)     # shape (n,)
    grad_a = 2.0 * n * g_raw            # d(2 n·Σ w·conv)/da

    # ---- back‑prop through the soft‑max (logits → probabilities) ----
    dot_ag = np.dot(a, grad_a)          # Σ a_i g_i
    grad_z = a * (grad_a - dot_ag)      # ∂L/∂z_i = a_i (g_i – Σ a_j g_j)

    # ---- loss (the quantity we finally want to minimise) ----
    loss = 2.0 * n * weighted_sum

    return float(loss), grad_z.astype(np.float64)

        print("[init] using uniform start")

    # Convert to logits (log‑softmax parameters)
    eps = 1e-12
    z = np.log(a0 + eps)

    best_a = a0.copy()
    best_val = 2.0 * n * max(np.convolve(best_a, best_a))
    print(f"[global] start objective = {best_val:.9f}")

    # --------------------------------------------------------------
    # L‑BFGS optimisation phases (increasing λ)
    # --------------------------------------------------------------
    lambda_schedule = [5.0, 50.0, 200.0, 800.0, 2000.0, 8000.0]
    from scipy.optimize import fmin_l_bfgs_b  # may raise ImportError

    for lam in lambda_schedule:
        if time.time() >= deadline:
            break
        GLOBAL_LAMBDA = lam
        # a few restarts per λ to escape shallow basins
        for restart in range(2):
            if time.time() >= deadline:
                break
            # small random perturbation before each run
            if restart > 0:
                z += np.random.normal(scale=0.02, size=n)
            maxiter = 150          # generous but still quick
            try:
                z, fmin, info = fmin_l_bfgs_b(_loss_and_grad, z,
                                              maxiter=maxiter, disp=0)
            except Exception as exc:      # pragma: no cover
                print("[L‑BFGS] failed:", exc)
                continue

            # evaluate the true objective (exact max)
            a = _softmax(z)
            conv = _full_convolution(a)
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[L‑BFGS λ={lam:.0f} restart={restart}] new best = {best_val:.9f}")

    # --------------------------------------------------------------
    # Projected sub‑gradient refinement (same as reference code)
    # --------------------------------------------------------------
    a = best_a.copy()
    obj = best_val
    step = 0.02
    for it in range(1200):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the true max (g_i = 2 * a_{kmax-i})
        g = np.zeros_like(a)
        if kmax < n:
            g[:kmax + 1] = 2.0 * a[kmax::-1]
        else:
            offset = kmax - (n - 1)
            L = n - offset
            g[offset:] = 2.0 * a[:L][::-1]

        a_trial = _project_simplex(a - step * g)
        conv_trial = _full_convolution(a_trial)
        obj_trial = 2.0 * n * conv_trial.max()

        if obj_trial < obj - 1e-12:
            a = a_trial
            obj = obj_trial
            best_a = a.copy()
            best_val = obj
            step = min(step * 1.1, 0.5)
            print(f"[subgrad] improved to {obj:.9f}")
            if obj < 1.503:
                break
        else:
            step *= 0.7

    # --------------------------------------------------------------
    # Final scaling to the conventional sum √(2 n) and secure evaluation
    # --------------------------------------------------------------
    final_seq = (best_a * target_sum).tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
