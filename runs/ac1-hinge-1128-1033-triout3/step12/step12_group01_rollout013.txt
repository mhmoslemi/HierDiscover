def evaluate_sequence(sequence):
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)

def _mirror_descent(a, max_steps, eta_start, end_time):
    n = a.shape[0]
    best_O = np.inf
    best_a = a.copy()
    for step in range(max_steps):
        if time.time() >= end_time:
            break
        conv = _full_convolution(a)   # length 2n-1
        max_conv = conv.max()
        O = 2.0 * n * max_conv   # since sum = 1 (normalization ensures)
        if O < best_O:
            best_O = O
            best_a = a.copy()
        # compute subgradient
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi+1)
            g = np.zeros_like(a)
            g[idx] = 2.0 * a[k - idx]
        else:
            g = np.zeros_like(a)
        eta = eta_start * (1.0 - step / max_steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)
    return best_a, best_O

def _mirror_descent_max(a: np.ndarray, steps: int, time_deadline: float,
                        stagnation_thresh: int = 5000,

    """
    Entropic mirror descent that minimises the peak of the self‑convolution.
    Returns the best vector seen and its objective O = 2·n·max(conv).

    for it in range(steps):
