"""
Propose a step‑function that (approximately) minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))² .

The algorithm works in two phases:

* **Phase 1 – smoothed Adam** (≈35 % of the time budget):
  Replace the nondifferentiable “max” by a soft‑max with a temperature that
  grows from 1 to a large value.  The surrogate is differentiable; Adam
  updates are performed on the log‑probabilities (guaranteeing positivity)
  and the vector is renormalised after each step.

* **Phase 2 – exact entropy‑mirror descent** (the remaining time):
  Switch to the true objective.  For the current maximum‑convolution entry
  `k = argmax (a*a)`, the sub‑gradient is `g_i = 2·a_{k‑i}` for the
  contributing indices.  An exponentiated‑gradient (entropy‑mirror) update
  `a ← a·exp(−η·g)` is applied with a linearly decaying step size η.

* **Final polishing – short stochastic local search**:
  With the time left, a cheap random‑pair “swap” move is repeatedly tried.
  A move that lowers the objective is kept.  This often yields a tiny extra
  improvement (the remaining gap to the target).

All operations respect the 1000 s wall‑clock budget, keep the vector
non‑negative, and maintain a unit sum (the objective is scale‑invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two (possibly different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget (default 1000 s) and returns a list
    of floats (the heights).
    """
    np.random.seed(seed)
    start = time.time()
    # keep a short safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between flexibility and speed.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a probability vector (sum = 1, non‑negative).
    # ------------------------------------------------------------------
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        try:
            old = np.asarray(old_seq, dtype=float)
            if old.shape[0] != n:
                # interpolate to the required length
                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            print("[init] interpolated previous solution")
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # smooth Beta(2,2) shape – central bump, strictly positive
        i = np.arange(1, n + 1, dtype=np.float64)
        a0 = (i ** 1.0) * ((n - i + 1) ** 1.0)          # proportional to i·(n‑i+1)
        a0 = a0 / a0.sum()
        print("[init] generated Beta(2,2) start")

    # Work in the log‑domain for the Adam stage (positivity guaranteed)
    eps = 1e-30
    z = np.log(a0 + eps)    # logits

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    # time split
    stage1_frac = 0.35                     # 35 % of the budget → Adam
    # Adam
    beta1, beta2 = 0.90, 0.999
    eps_adam = 1e-8
    lr_start, lr_end = 0.40, 0.02         # relatively aggressive
    max_alpha = 400.0                     # final soft‑max temperature
    # Mirror descent
    eta_start, eta_end = 0.90, 0.01       # start fairly aggressive
    # Stagnation handling
    stagn_limit = 2000
    jitter_scale_adam = 0.04
    jitter_scale_mirror = 0.01

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_obj = np.inf
    best_iter = 0
    it = 0
    mirror_mode = False

    stagn_counter = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress as a number in [0, 1]
        prog = (now - start) / (deadline - start)

        # ------------------------------------------------------------------
        # Switch from Adam to mirror descent according to the allocated time
        # ------------------------------------------------------------------
        if not mirror_mode and (now - start) >= stage1_frac * (deadline - start):
            mirror_mode = True
            # materialise the probability vector from logits
            a = np.exp(z)
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else np.full_like(a, 1.0 / n)
            print(f"[mode] entering mirror descent at t={now-start:.1f}s")

        # ------------------------------------------------------------------
        # Current probability vector
        # ------------------------------------------------------------------
        if not mirror_mode:
            a = np.exp(z)
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else np.full_like(a, 1.0 / n)

        # ------------------------------------------------------------------
        # Full self‑convolution and objective value (sum(a)=1 ⇒ obj = 2·n·max)
        # ------------------------------------------------------------------
        conv = _full_convolution(a)
        max_c = conv.max()
        obj = 2.0 * n * max_c

        # ------------------------------------------------------------------
        # Record the best solution seen so far
        # ------------------------------------------------------------------
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            best_iter = it
            stagn_counter = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}  t={now-start:.1f}s")
        else:
            stagn_counter += 1

        # ------------------------------------------------------------------
        # Gradient / update step
        # ------------------------------------------------------------------
        if not mirror_mode:
            # ---- Phase 1: Adam on the soft‑max surrogate ----
            # Temperature grows with prog (more peaked soft‑max)
            temp = 1.0 + (max_alpha - 1.0) * (prog ** 1.5)

            # soft‑max weights (numerically stable)
            exp_scaled = np.exp(temp * (conv - max_c))
            weight_sum = exp_scaled.sum()
            weights = exp_scaled / weight_sum                     # Σ weights = 1

            # gradient of the surrogate w.r.t. a
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            # slice the middle n entries (gradient for each a_i)
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]

            # chain rule through the soft‑max (z = log a)
            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            # Adam update on logits
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = lr_start * (1.0 - prog) + lr_end * prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        else:
            # ---- Phase 2: exact entropy‑mirror descent on the max entry ----
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)

            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                # sub‑gradient of max entry
                g[idx] = 2.0 * a[k - idx]

            # linear decay of η inside the mirror phase
            phase_start = start + stage1_frac * (deadline - start)
            phase_prog = (now - phase_start) / ((deadline - start) * (1.0 - stage1_frac))
            phase_prog = max(0.0, min(1.0, phase_prog))
            eta = eta_start * (1.0 - phase_prog) + eta_end * phase_prog

            a = a * np.exp(-eta * g)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

            it += 1

        # ------------------------------------------------------------------
        # Escape from stagnation
        # ------------------------------------------------------------------
        if stagn_counter >= stagn_limit:
            if not mirror_mode:
                z += np.random.normal(scale=jitter_scale_adam, size=n)
                print(f"[reset] jitter added (Adam) at it={it}")
            else:
                perturb = np.random.normal(scale=jitter_scale_mirror, size=n)
                a = np.maximum(a + perturb, 0.0)
                a = a / a.sum()
                print(f"[reset] jitter added (Mirror) at it={it}")
            stagn_counter = 0

    # ------------------------------------------------------------------
    # Final stochastic local search (cheap hill‑climbing) on the best vector
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 2.0:                     # use at most a few seconds
        search_budget = min(5.0, time_left)
        a = best_a.copy()
        conv = _full_convolution(a)
        obj = 2.0 * n * conv.max()
        best_loc_a = a.copy()
        best_loc_obj = obj
        sa_start = time.time()
        max_frac = 0.10                     # max 10 % of a[i] may be moved

        while time.time() - sa_start < search_budget:
            i = np.random.randint(0, n)
            if a[i] <= 0.0:
                continue
            j = np.random.randint(0, n)
            if i == j:
                continue

            # move a small random fraction of a[i] to a[j]
            delta = max_frac * np.random.rand() * a[i]
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            conv_new = _full_convolution(a_new)
            obj_new = 2.0 * n * conv_new.max()

            if obj_new < best_loc_obj:
                best_loc_obj = obj_new
                best_loc_a = a_new.copy()
                a = a_new     # continue hill‑climbing from the improved point

        if best_loc_obj < best_obj - 1e-12:
            best_obj = best_loc_obj
            best_a = best_loc_a
            print(f"[sa] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (the wrapper will also evaluate)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx O={best_obj:.6f})")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
