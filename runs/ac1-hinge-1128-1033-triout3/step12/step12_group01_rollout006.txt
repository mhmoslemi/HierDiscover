    best_a = a.copy()

    stagnation = 0
    it = 0
    max_eta = 0.7
    min_eta = 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute full conv
        conv = _full_convolution(a)
        max_val = conv.max()
        if max_val < best_obj - 1e-12:
            best_obj = max_val
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # compute gradient at max index (choose first max)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g = np.zeros_like(a)
        # gradient = 2 * a[d] where d = k - i
        g[idx] = 2.0 * a[k - idx]

        # schedule eta based on iteration count
        # Use linear decay from max_eta to min_eta over, say, 200000 iterations
        decay_iters = 200000
        eta = max_eta * max(0.0, 1.0 - it / decay_iters) + min_eta * min(1.0, it / decay_iters)
        # update multiplicatively
        a = a * np.exp(-eta * g)
        # renormalize to target sum
        s = a.sum()
        if s > 0:
            a = a * (target / s)
        else:
            a.fill(target / n)

            else:
                a.fill(target / n)

"""
Mirror‑descent optimiser for the “flat autocorrelation” problem.

The evaluation metric is             O = 2·n·max(conv(a,a)) / (sum(a))² .
Because the objective is scale‑invariant we normalise every candidate
to  Σa = sqrt(2·n).  Then O = max(conv(a,a)), i.e. we only have to minimise
the maximal entry of the self‑convolution.

The optimiser works as follows:

* Initialise a non‑negative vector (use the previous best solution if
  available, otherwise a Dirichlet draw).
* Repeatedly:
    – compute the full convolution a*a,
    – locate the index k where it attains its maximum,
    – form the exact sub‑gradient of the max‑entry (g_i = 2·a_{k‑i} on the
      support of the convolution) and perform an exponentiated‑gradient /
      mirror‑descent update a_i ← a_i·exp(‑η·g_i),
    – renormalise a to the required sum.
  The step‑size η decays as η = η₀ / sqrt(t+1); occasional random jitter
  is injected when no improvement has been observed for a while, which
  helps escaping plateaux.
* The whole optimisation runs until the time budget (≈1000 s) is exhausted,
  always keeping the best vector seen so far.
* Finally the best vector is returned as a Python list.  ``evaluate_sequence``
  is called once at the end to obtain the official score.

This approach removes the heavy Adam phase of the reference code and
focuses all compute on the exact max‑convolution objective,
allowing many more iterations (hundreds of thousands) within the budget
and typically pushing the bound below the previous 1.508 value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolution (self‑convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Mirror‑descent optimiser for max(conv(a,a))
# ----------------------------------------------------------------------
def _mirror_max_descent(
    a0: np.ndarray,
    deadline: float,
    n: int,
    eta0: float = 0.8,
    jitter_scale: float = 0.01,
    stagnation_thr: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient (mirror) descent on the exact max‑entry
    of the self‑convolution.

    Parameters
    ----------
    a0 : np.ndarray
        Starting non‑negative vector (will be normalised internally).
    deadline : float
        Absolute time at which the routine must stop.
    n : int
        Length of the vector.
    eta0 : float
        Initial step‑size (decays as eta0 / sqrt(t+1)).
    jitter_scale : float
        Standard deviation of Gaussian jitter injected after long stagnation.
    stagnation_thr : int
        Number of iterations without improvement before jitter is added.

    Returns
    -------
    best_a : np.ndarray
        Vector with the smallest max‑convolution encountered.
    best_val : float
        Corresponding maximal convolution value (i.e. the O‑score for the
        normalised vector).
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise
    a = np.maximum(a0, 0.0).astype(float)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / n)

    best_a = a.copy()
    # compute initial objective
    conv = _full_convolution(a)
    best_val = float(conv.max())
    stagnation = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- convolution and max entry -----
        conv = _full_convolution(a)
        max_val = float(conv.max())
        k = int(conv.argmax())

        # record improvement
        if max_val < best_val - 1e-12:
            best_val = max_val
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # ----- sub‑gradient of max entry -----
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g = np.zeros_like(a)
        # derivative of c_k = Σ_i a_i a_{k-i} w.r.t a_j is 2·a_{k-j}
        g[idx] = 2.0 * a[k - idx]

        # ----- step‑size schedule -----
        eta = eta0 / math.sqrt(it + 1.0)

        # ----- mirror‑descent update (multiplicative) -----
        a = a * np.exp(-eta * g)

        # ----- renormalise to keep Σa = sqrt(2n) -----
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ----- occasional jitter to escape plateaus -----
        if stagnation >= stagnation_thr:
            jitter = np.random.normal(scale=jitter_scale, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a = a * (target_sum / s)
            else:
                a = np.full_like(a, target_sum / n)
            stagnation = 0
            print(f"[jitter] it={it} max={max_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights that (empirically) improves
    the metric O = 2·n·max(conv(seq,seq)) / (sum(seq))².
    The routine allocates the full time budget (≈1000 s) to a mirror‑descent
    optimisation starting from the previous best solution (if available) or
    from a random Dirichlet draw.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose final length – a larger n gives a lower theoretical bound.
    # 12000 is a good compromise between quality and speed.
    # ------------------------------------------------------------------
    final_n = 12000

    # ------------------------------------------------------------------
    # Initialise a candidate vector (use the previous best if present)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined globally by the surrounding harness
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        # interpolate old vector to the desired length
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, final_n)
        a0 = np.interp(x_new, x_old, old)
        print("[init] interpolated previous solution")
    except Exception:
        # fallback: a modestly spiky Dirichlet draw (helps break uniform symmetry)
        concentration = 0.05
        a0 = np.random.dirichlet(np.full(final_n, concentration))
        print("[init] random Dirichlet start (conc=0.05)")

    # ------------------------------------------------------------------
    # Run the mirror‑descent optimiser until time runs out
    # ------------------------------------------------------------------
    best_a, best_val = _mirror_max_descent(
        a0=a0,
        deadline=deadline,
        n=final_n,
        eta0=0.8,
        jitter_scale=0.01,
        stagnation_thr=5000,
    )
    print(f"[finish] best max‑conv = {best_val:.6f} (O‑score) after optimisation")

    # ------------------------------------------------------------------
    # Convert to a plain Python list and perform the final secure evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_val:.6f})")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
