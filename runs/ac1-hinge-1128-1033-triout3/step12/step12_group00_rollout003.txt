    a = a0.copy()
    cur_val = _objective(a)
    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break
        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        grad_norm_sq = (grad * grad).sum()
        if grad_norm_sq == 0:
            break
        # step size using Polyak
        step = (cur_val - best_val) / (grad_norm_sq + 1e-12)
        # Ensure step positive; if cur_val == best_val, use small step
        if step <= 0:
            step = 0.01 * (0.35 * (1.0 - (now - start_time)/(deadline - start_time)) + 1e-5)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        new_val = _objective(a_new)
        # accept if improvement else maybe random jitter
        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val

"""
Optimiser for the step‑function evaluation problem.

The algorithm proceeds in four stages:

1️⃣  **Initial seeds** – uniform, an interpolated historic solution (if
    available) and a quick sweep of random Beta‑shaped candidates.

2️⃣  **Global search with a mixture model** – the vector a is expressed as a
    mixture of a few (default 4) Beta‑like components.
    A low‑dimensional differential‑evolution optimiser searches the
    component parameters (log‑shape exponents and log‑weights).  The best
    mixture is kept as the current candidate.

3️⃣  **Exact sub‑gradient polishing** – on the full‑length vector a we run a
    projected sub‑gradient descent on the true objective
    f(a)=2·n·max(conv(a,a)).  The step size follows Polyak’s rule
    (distance to the best value divided by ‖grad‖²) and occasional jitter
    helps escape plateaus.

4️⃣  **Fine smoothing** – a short L‑BFGS‑B optimisation of a smooth
    log‑sum‑exp surrogate (ε≈1e‑12) refines the solution.

All phases respect a 1000 s wall‑clock budget.  The final sequence is
returned as a plain ``list[float]`` and is evaluated with the supplied
``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ----------------------------------------------------------------------
# FFT‑based convolutions (full linear)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.  The gradient already includes the factor 2·n.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → probability simplex)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random Beta family candidate (cheap initial sweep)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ~1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 1500) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smoothed objective.
    """
    # initialise in log‑space (soft‑max guarantees feasibility)
    x0 = np.log(a0 + 1e-12)

    # ----- objective ----------------------------------------------------
    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    # ----- gradient -----------------------------------------------------
    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    # respect the remaining time
    time_left = deadline - time.time()
    if time_left < 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:
        # fallback – return without changing best
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum of contributions from all maximal diagonals
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        grad_norm_sq = float(np.dot(grad, grad))
        if grad_norm_sq == 0.0:
            break

        # Polyak step (distance to current best divided by ‖grad‖²)
        step = (cur_val - best_val) / (grad_norm_sq + 1e-12)
        if step <= 0.0:
            # fallback to a diminishing schedule
            prog = (now - start_time) / (deadline - start_time)
            step = 0.35 * (1.0 - prog) + 1e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 400 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)
            cur_val = _objective(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Mixture‑of‑Beta helper (low‑dimensional parametric family)
# ----------------------------------------------------------------------
def _mixture_from_params(params: np.ndarray,
                         n: int,
                         log_i: np.ndarray,
                         log_ni: np.ndarray) -> np.ndarray:
    """
    Build a length‑n non‑negative vector from parameters.
    params are packed as [log(p1), log(q1), log(w1), log(p2), log(q2), log(w2), …].
    """
    M = len(params) // 3
    log_p = params[0::3]
    log_q = params[1::3]
    log_w = params[2::3]

    p = np.exp(log_p)            # shape (M,)
    q = np.exp(log_q)            # shape (M,)

    # broadcasting: (M,1)*(1,n) -> (M,n)
    p_minus1 = p[:, None] - 1.0
    q_minus1 = q[:, None] - 1.0
    log_comp = p_minus1 * log_i[None, :] + q_minus1 * log_ni[None, :] + log_w[:, None]

    # stability shift
    max_log = np.max(log_comp)
    comp = np.exp(log_comp - max_log)          # shape (M,n)

    a = np.sum(comp, axis=0)                   # shape (n,)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a list whose sum is 1 and that minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – a moderately large n gives a richer search space.
    # --------------------------------------------------------------
    n = 12000
    i = np.arange(1, n + 1, dtype=np.float64)
    log_i = np.log(i)
    log_ni = np.log(n - i + 1)

    # --------------------------------------------------------------
    # 0️⃣  Initial candidates (exact objective)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # uniform baseline
    best_val = _objective(best_a)

    # 0.1  Re‑use a previously discovered solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
        print(f"[Init] interpolated previous solution, value={val:.6f}")
    except Exception:
        pass

    # 0.2  Quick random Beta sweep
    for _ in range(200):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val
    print(f"[Init] after Beta sweep, best value={best_val:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Global search with a mixture of Beta components
    # --------------------------------------------------------------
    M = 4                               # number of mixture components
    dim = M * 3
    bounds = [(-5.0, 5.0)] * dim        # log‑scale bounds for (p,q,w)

    def obj_mixture(params):
        a = _mixture_from_params(params, n, log_i, log_ni)
        return _objective(a)

    # allocate a fraction of the budget to DE
    de_deadline = start_time + 0.30 * budget_s
    time_left = de_deadline - time.time()
    if time_left > 2.0:
        # modest settings – keep total runtime well under the allocated slice
        de_res = differential_evolution(
            obj_mixture,
            bounds,
            maxiter=30,
            popsize=12,
            polish=False,
            seed=seed,
            updating='deferred')
        a_mix = _mixture_from_params(de_res.x, n, log_i, log_ni)
        val_mix = _objective(a_mix)
        if val_mix < best_val:
            best_a, best_val = a_mix.copy(), val_mix
        print(f"[DE] best value after DE = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Exact sub‑gradient polishing (majority of remaining budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a,
        n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=250_000)

    # --------------------------------------------------------------
    # 3️⃣  Small blend with the uniform distribution (helps eliminate spikes)
    # --------------------------------------------------------------
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # 4️⃣  Final short L‑BFGS on the smooth surrogate (tiny epsilon)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(
            best_a,
            n,
            deadline,
            start_time,
            best_a,
            best_val,
            eps=1e-12,
            max_iter=800)

    # --------------------------------------------------------------
    # Final secure evaluation and return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
