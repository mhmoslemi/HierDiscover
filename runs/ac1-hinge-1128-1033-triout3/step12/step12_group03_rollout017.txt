"""
Hybrid search for the step‑function self‑convolution problem.

The optimiser works in three stages:

1. Initialise several diverse candidates (uniform, Dirichlet draws,
   exponential decay and a simple linear taper).  Each candidate is
   normalised to a probability vector (sum = 1).

2. Run a medium‑length Adam optimiser on a smooth‑max version of the
   objective.  The temperature λ grows from a modest value to a very
   large one; when λ is large we switch to the exact max (a one‑hot
   weight).  This quickly drives the solution into a region of low
   convolution peak.

3. polish the best solution with a cheap projected sub‑gradient loop,
   using the exact max‑gradient.  The step size is adapted
   multiplicatively and occasional noise is added to avoid stagnation.

All work respects the 1000 s budget; the best sequence found is finally
passed to `evaluate_sequence`.  The code is deliberately written with
only NumPy (no external solvers) and is self‑contained.
"""

import numpy as np

# Helper utilities – all pure NumPy, no external dependence

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

    computed via FFT for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of weight vector w with a.
    Returned vector has length len(a).  Implemented via FFT; for the
    problem sizes involved this is faster than a pure Python loop.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective = 2·n·max(self‑convolution) for a probability vector."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise `evaluate_sequence`.  The routine runs several short Adam
    optimisations from diverse starts, followed by a projected
    sub‑gradient refinement, and respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 works well and matches the reference code.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from a previous construction if it exists.
    # ------------------------------------------------------------------
    try:
        old_seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old_seq.size > 1:
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_seq.size)
            a0 = np.interp(new_x, old_x, old_seq)
            a0 = np.maximum(a0, 0.0)
            a0_sum = a0.sum()
            if a0_sum > 0:
                a0 = a0 / a0_sum
            else:
                a0 = np.full(n, 1.0 / n)
            print("[init] interpolated previous solution")
        else:
            raise Exception
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] uniform start")

    # ------------------------------------------------------------------
    # Global best tracker
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[global] start objective = {best_val:.9f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – many short restarts with Adam + refinement
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1️⃣  Select a random initial distribution
        # --------------------------------------------------------------
        init_type = np.random.choice(
            ["uniform", "dirichlet", "exponential", "linear"], p=[0.2, 0.5, 0.2, 0.1]
        )
        if init_type == "uniform":
            a = np.full(n, 1.0 / n)
        elif init_type == "dirichlet":
            # draw a Dirichlet (Gamma) vector with a random concentration
            alpha = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a = np.random.gamma(alpha, 1.0, size=n)
            a /= a.sum()
        elif init_type == "exponential":
            # truncated exponential (geometric‑like) decay, r close to 1
            r = np.random.uniform(0.99, 0.9999)
            idx = np.arange(n)
            a = (1 - r) * (r ** idx)
            a /= a.sum()
        else:   # linear taper
            delta = np.random.uniform(0.0, 2.0 / (n - 1))
            a = (1.0 + delta * ((n - 1) / 2.0 - np.arange(n))) / n
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # add a tiny Gaussian jitter to break symmetry
        a += np.random.normal(scale=1e-4, size=n)
        a = np.clip(a, 0.0, None)
        a /= a.sum()

        # --------------------------------------------------------------
        # 2️⃣  Adam optimisation on a smooth‑max approximation
        # --------------------------------------------------------------
        # work in log‑softmax space to keep the simplex constraint exact
        z = np.log(a + 1e-12)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        t_iter = 0
        max_inner = 18000          # enough steps for the temperature sweep
        lambda_start = 5.0
        lambda_end = 20000.0

        while time.time() < deadline and t_iter < max_inner:
            t_iter += 1
            prog = (t_iter - 1) / max_inner
            lam = lambda_start * (1.0 - prog) + lambda_end * prog

            # ----- soft‑max → probability vector -----
            mx = np.max(z)
            exp_z = np.exp(z - mx)
            a = exp_z / exp_z.sum()

            # ----- exact objective (for bookkeeping) -----
            conv = _full_convolution(a)
            conv_max = conv.max()
            cur_val = 2.0 * n * conv_max
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[adam] new best = {best_val:.9f} (iter {t_iter})")

            # ----- weighting for the max (smooth vs exact) -----
            if lam > 5000.0:
                w = np.zeros_like(conv)
                w[int(np.argmax(conv))] = 1.0
            else:
                # smooth‑max via soft‑max over the convolution values
                exp_vals = np.exp(lam * (conv - conv_max))
                w = exp_vals / exp_vals.sum()

            # ----- gradient of the (smooth) max w.r.t. a -----
            g_raw = _cross_corr(w, a)           # O(n log n) via FFT
            g = 2.0 * g_raw

            # ----- back‑prop through the soft‑max (logits) -----
            dot_ag = np.dot(a, g)
            grad_z = a * (g - dot_ag)

            # ----- Adam parameter update -----
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            z -= 0.04 * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 3️⃣  Cheap projected sub‑gradient refinement (exact max)
        # --------------------------------------------------------------
        a = best_a.copy()
        step = 0.02
        for _ in range(2500):
            if time.time() >= deadline:
                break
            conv = _full_convolution(a)
            kmax = int(np.argmax(conv))

            # sub‑gradient of the true max (vector g_i = 2 * a_{kmax-i})
            g = np.zeros_like(a)
            if kmax < n:
                g[:kmax + 1] = 2.0 * a[kmax::-1]
            else:
                offset = kmax - (n - 1)
                L = n - offset
                g[offset:] = 2.0 * a[:L][::-1]

            a_trial = _project_simplex(a - step * g)
            obj_trial = 2.0 * n * _full_convolution(a_trial).max()

            if obj_trial < best_val - 1e-12:
                a = a_trial
                best_a = a_trial.copy()
                best_val = obj_trial
                step = min(step * 1.1, 0.5)   # slowly increase after success
                print(f"[sgd] improved to {best_val:.9f}")
                if best_val < 1.503:
                    break
            else:
                step *= 0.7  # shrink step on no improvement

        # Early exit if target already reached
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation (the only place we call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
