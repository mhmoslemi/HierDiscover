"""
Propose a step‑function sequence that minimizes
    2·n·max(conv(a,a)) / (sum(a))²
under a_i ≥ 0.
The optimizer combines a richer systematic initial search, long‑running
Adam/gradient polish phases, an aggressive simulated‑annealing hill‑climber,
and a lightweight local‑parameter refinement (Beta‑family).  All work on the
simplex (Σa = 1) so the denominator of the true evaluation is constant.
The code respects the 1000 s wall‑clock budget and prints progress.
"""

import time
import numpy as np
from scipy.optimize import linprog   # optional – not used in the final version
from scipy import sparse


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) when sum(a)==1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Beta‑shaped vector on a log‑scale."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    """Deterministic Beta candidate for given log‑parameters."""
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Truncated exponential shape a_i ∝ r^i with r∈(0,1)."""
    log_r = rng.uniform(-8.0, -0.1)   # r between ~0.0003 and 0.9
    r = np.exp(log_r)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i ∝ (i+1)^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a /= a.sum()
    return a


def _mixture_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent Beta families."""
    w = rng.random()
    a1 = _random_beta_candidate(n, rng)
    a2 = _random_beta_candidate(n, rng)
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex for L(a)=2·n·smooth_max(a)."""
    w = np.log(a0 + 1e-12)          # work in log‑softmax space

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.40, 0.001      # a somewhat larger start LR
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 8000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule learning rate and smoothing
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient (log‑sum‑exp trick)
        nloc = n
        b = _full_convolution(a)                     # length 2n‑1
        max_b = b.max()
        shifted = (b - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()                  # shape (2n‑1,)

        # convolution of a with reversed p (gradient of smooth max)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
        segment = conv_ab[nloc - 1:2 * nloc - 1]      # n entries
        grad_a = 4.0 * nloc * segment[::-1]          # includes factor 2·n
        # (note that smooth_val is not needed for the update)

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.017:
            w += np.random.normal(scale=0.02, size=n)

        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (source→target moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """Local search with simulated‑annealing acceptance."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    ## schedule parameters (more aggressive early moves)
    max_frac_start, max_frac_end = 0.95, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 60000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # location of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index: sample proportionally to contribution
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index: favour low contribution slots
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # re‑inject a small amount of noise and continue
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule (more aggressive early on)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to escape flats
        if np.random.rand() < 0.004:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) polishing
# ----------------------------------------------------------------------
def _expgrad_polish(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient updates for the exact max‑convolution objective."""
    a = a0.copy()
    cur_val = _objective(a)

    eta_start, eta_end = 1.2, 1e-5

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # schedule eta
        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # EG update, stay strictly positive
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        new_val = _objective(a)

        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[ExpG] new best = {best_val:.6f}")

        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.018, size=n), 0.0)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 25):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Light smoothing (running‑average) – useful after noisy steps
# ----------------------------------------------------------------------
def _smooth_candidate(a: np.ndarray, width: int) -> np.ndarray:
    """Apply a moving‑average filter of given odd width and renormalise."""
    if width <= 1:
        return a
    kernel = np.ones(width) / width
    # convolution with 'same' mode using FFT for speed
    conv = np.convolve(a, kernel, mode='same')
    conv = np.maximum(conv, 0.0)
    conv /= conv.sum()
    return conv


# ----------------------------------------------------------------------
# Local Beta‑parameter refinement (stochastic hill‑climbing)
# ----------------------------------------------------------------------
def _beta_refine(a_best: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 best_val: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Fine‑tune log‑Beta parameters via a tiny simulated‑annealing loop."""
    # initialise from the current best candidate if it resembles a Beta shape
    # (fallback to centre of the grid)
    logp = -2.0
    logq = -2.0
    # quick guess: fit a power law to the current best
    # (use log‑linear regression on the tail)
    try:
        # use the first third as "left tail" and last third as "right tail"
        idx = np.arange(n)
        left = idx < n // 3
        right = idx > 2 * n // 3
        # avoid zeros
        eps = 1e-12
        lp = np.log(np.maximum(a_best[left], eps))
        lpos = np.log(idx[left] + 1)
        logp = np.dot(lpos, lp) / np.dot(lpos, lpos)
        lq = np.log(np.maximum(a_best[right], eps))
        rpos = np.log(n - idx[right] + 1)
        logq = np.dot(rpos, lq) / np.dot(rpos, rpos)
    except Exception:
        pass

    cur_logp, cur_logq = logp, logq
    cur_a = _beta_grid_candidate(n, cur_logp, cur_logq)
    cur_val = _objective(cur_a)

    # simple annealing schedule
    T0 = 0.025
    while True:
        now = time.time()
        if now >= deadline:
            break
        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog) + 1e-12

        # propose new log‑params
        cand_logp = cur_logp + rng.normal(scale=0.8 * (1.0 - prog) + 0.04 * prog)
        cand_logq = cur_logq + rng.normal(scale=0.8 * (1.0 - prog) + 0.04 * prog)

        cand = _beta_grid_candidate(n, cand_logp, cand_logq)
        cand_val = _objective(cand)

        if cand_val < cur_val:
            cur_logp, cur_logq, cur_a, cur_val = cand_logp, cand_logq, cand, cand_val
            if cand_val < best_val:
                best_val, best_a = cand_val, cand.copy()
                print(f"[BetaRef] new best = {best_val:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(cand_val - cur_val) / T)
            if rng.random() < prob:
                cur_logp, cur_logq, cur_a, cur_val = cand_logp, cand_logq, cand, cand_val

    # return the best beta‑candidate found (or the incoming best if none improved)
    return best_a, best_val


# ----------------------------------------------------------------------
# Large‑move simulated‑annealing (final phase)
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple annealing with relatively large source→target transfers."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.022

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)          # linear cooling
        max_frac = 0.18 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # occasional global jitter
        if rng.random() < 0.012:
            a = np.maximum(a + rng.normal(scale=0.009, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is ≈1 and (tries to) minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size (larger n gives a finer discretisation)
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Rich initial population (random + systematic)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # try to reuse a previously discovered solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # ----- systematic Beta grid (finer than before) -----
    logp_vals = np.linspace(-7.5, 2.0, 19)   # 19×19 ≈ 361 candidates
    logq_vals = np.linspace(-7.5, 2.0, 19)
    for lp in logp_vals:
        for lq in logq_vals:
            cand = _beta_grid_candidate(n, lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ----- random Beta draws (more than previous) -----
    for _ in range(7000):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- Dirichlet draws (various concentrations) -----
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(50):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ----- exponential family -----
    for _ in range(1200):
        cand = _exponential_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- power‑law family -----
    for p in np.linspace(-4.0, 4.0, 31):
        cand = _powerlaw_candidate(n, p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- mixture of two Betas -----
    for _ in range(2500):
        cand = _mixture_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- basic sinusoidal shapes (already used before) -----
    i = np.arange(n, dtype=np.float64)
    cand = np.sin(np.pi * (i + 0.5) / n)
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    cand = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    print(f"[Init] best after exhaustive init = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam optimisation (≈22 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.22 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase B – Aggressive hill‑climbing (≈55 % of the budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * budget_s
    best_a, best_val = _hillclimb(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase C – Local Beta‑parameter refinement (≈7 % of the budget)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.62 * budget_s
    best_a, best_val = _beta_refine(
        best_a, n,
        deadline=phase_c_deadline,
        start_time=start_time,
        best_val=best_val,
        rng=rng,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase D – Sub‑gradient polishing (≈14 % of the budget)
    # --------------------------------------------------------------
    phase_d_deadline = start_time + 0.76 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase_d_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=400_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase E – Mirror‑descent (exponentiated‑gradient) polishing
    # --------------------------------------------------------------
    phase_e_deadline = start_time + 0.90 * budget_s
    best_a, best_val = _expgrad_polish(
        best_a, n,
        deadline=phase_e_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=180_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase F – Light smoothing passes (helps tiny spikes)
    # --------------------------------------------------------------
    for w in (3, 5, 9):
        cand = _smooth_candidate(best_a, w)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Smooth] width={w} new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase G – Final simulated‑annealing with larger moves
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_anneal(
            best_a, n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            rng=rng,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation (for logging only)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return a plain Python list
    return best_a.tolist()
