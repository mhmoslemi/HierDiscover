def proj_simplex(v):
    # v is an array, returns projection onto simplex {x>=0, sum x = 1}
    # algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time, numpy as np, math

def full_convolution(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    # Actually we only need size minimal power of two >= 2*n-1
    size = 1 << ((2*n - 1).bit_length())
    fx = np.fft.rfft(a, size)
    conv = np.fft.irfft(fx*fx, size)[:2*n-1]
    return conv

def objective(a):
    n = a.shape[0]
    conv = full_convolution(a)
    M = conv.max()
    return 2.0 * n * M

def subgradient_max(a, conv, max_active=15, rel_tol=1e-6):
    n = a.shape[0]
    max_val = conv.max()
    # find indices within tolerance
    # find top K maybe
    # using argpartition
    if max_active >= conv.size:
        active = np.arange(conv.size)
    else:
        # get indices of top max_active values
        idx_top = np.argpartition(conv, -max_active)[-max_active:]
        # keep those within tolerance
        mask = conv[idx_top] >= max_val * (1 - rel_tol)
        active = idx_top[mask]
        # if none, include max index
        if active.size == 0:
            active = np.array([int(np.argmax(conv))])
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad

def proj_simplex(v):
    # projection onto simplex sum=1, v>=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero?
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        s = a0.sum()
        if s == 0:

        alpha = np.random.uniform(0.02, 5.0)
        a = np.random.gamma(alpha, 1.0, size=n)
        a_sum = a.sum()

        val = objective(a)

            best_val = val
            best_seq = a.copy()
            print(f"[randinit] new best val {best_val:.6f} alpha {alpha:.3f}")

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = objective(best_seq)
        print("[fallback] uniform start", best_val)

    # ADAM optimizer parameters
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999

        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[optim] iter {t} new best {best_val:.6f}")
        else:
            no_improve += 1

        # compute subgradient

        # Adam updates

        m = beta1 * m + (1 - beta1) * grad

        # linear learning rate decay

        lr = lr0 * (1 - progress) + 5e-5

        step = lr / (np.sqrt(v_hat) + eps)

            no_improve = 0

        print(f"[final] evaluate_sequence returned {final_score:.6f}")

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The algorithm works in three stages:

1️⃣  **Warm‑start** – build a large pool of random Dirichlet‑distributed
    candidates (different concentration parameters) and keep the one with
    the smallest cheap objective 2·n·max(conv(a,a)) (with Σa = 1).  If a
    previous solution is available in the global variable ``height_sequence_1``,
    it is interpolated to the current resolution and used as an additional
    start.

2️⃣  **Adam‑mirror descent** – from the best start we run an Adam optimiser.
    The sub‑gradient of the max‑convolution is obtained from the active set
    (the few largest convolution entries).  After each Adam step the iterate
    is projected onto the probability simplex (non‑negative, sum = 1).  A
    linear learning‑rate decay and occasional Gaussian jitter help to escape
    plateaux.

3️⃣  **Final evaluation** – the best sequence discovered is fed to the
    supplied ``evaluate_sequence`` for the official score and returned.

The whole routine respects the supplied time budget (default 1000 s) and
stores the best sequence found so far.  The implementation is fully
vectorised, uses FFT‑based convolutions and a standard simplex projection.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, full linear convolution)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself (length 2·len(a)‑1)
    using an FFT of the next power of two.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next pow‑2 ≥ 2·n‑1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def cheap_objective(a: np.ndarray) -> float:
    """
    Exact value of 2·n·max(conv(a,a)) assuming Σa = 1.
    Used as a cheap proxy while searching.
    """
    n = a.shape[0]
    b = full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def subgradient_max(a: np.ndarray,
                    conv: np.ndarray,
                    max_active: int = 15,
                    rel_tol: float = 1e-7) -> np.ndarray:
    """
    Return a sub‑gradient of max(conv) w.r.t. a.
    ``max_active`` limits the size of the active set; entries whose value
    lies within ``rel_tol`` of the maximum are kept.
    """
    n = a.shape[0]
    max_val = conv.max()
    # decide which lags belong to the active set
    if max_active >= conv.size:
        active = np.arange(conv.size)
    else:
        idx_top = np.argpartition(conv, -max_active)[-max_active:]
        mask = conv[idx_top] >= max_val * (1.0 - rel_tol)
        active = idx_top[mask]
        if active.size == 0:                     # fallback – the absolute max
            active = np.array([int(np.argmax(conv))])

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution of a_{k-i}
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return a uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (sequence) that minimizes
    ``evaluate_sequence``.

    The routine respects the supplied time budget, returns the best
    sequence it discovered (as a plain Python list) and prints progress.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # 0️⃣  Problem size – a compromise between expressivity and speed.
    # ------------------------------------------------------------------
    n = 16000                                 # 2ⁿⁿ‑1 fits into a 32768‑point FFT

    best_seq: np.ndarray | None = None
    best_val = np.inf

    def maybe_update(candidate: np.ndarray):
        """Record a candidate if it improves the cheap objective."""
        nonlocal best_seq, best_val
        val = cheap_objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[init] new best cheap objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previously saved solution (if available)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.ndim != 1:
            old = old.ravel()
        if old.shape[0] != n:
            # Linear interpolation to the current resolution
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s == 0 or not np.isfinite(s):
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / s
        maybe_update(a0)
        print("[init] interpolated previous solution")
    except Exception:
        # No previous solution – we will fall back to random starts later.
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet initialisations (time‑boxed)
    # ------------------------------------------------------------------
    init_deadline = deadline - 20.0            # keep ~20 s for the Adam phase
    while time.time() < init_deadline:
        # Draw a concentration parameter; small values give sparse vectors,
        # large values give near‑uniform vectors.
        alpha = np.random.uniform(0.02, 5.0)
        a = np.random.gamma(alpha, 1.0, size=n)
        total = a.sum()
        if total == 0.0 or not np.isfinite(total):
            continue
        a /= total
        maybe_update(a)

    # If nothing succeeded (unlikely) start from uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = cheap_objective(best_seq)
        print("[fallback] uniform start, cheap objective =", best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑mirror‑descent refinement
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)               # first moment
    v = np.zeros_like(a)               # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.20                         # initial learning rate
    iteration = 0
    no_improve = 0
    max_no_improve_restart = 25000
    jitter_prob = 0.001

    while time.time() < deadline:
        conv = full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[optim] iter {iteration:6d}  new best {best_val:.6f}")
        else:
            no_improve += 1

        # sub‑gradient of the max‑convolution
        grad = subgradient_max(a, conv, max_active=15, rel_tol=1e-7)

        # Adam update (with bias correction)
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # linear decay of the base learning rate
        elapsed = time.time() - start
        progress = elapsed / (deadline - start)
        lr = lr0 * (1.0 - progress) + 5e-5

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = proj_simplex(a)                     # stay on the simplex

        # occasional stochastic jitter to escape flat regions
        if np.random.rand() < jitter_prob or no_improve >= max_no_improve_restart:
            jitter = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = proj_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[jitter] iter {iteration}")

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
