import cvxpy as cp

def lp_step(f, M, active_lags):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_lags:
        # compute dot product efficiently using convolution?
        # we can compute index range
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # indices i from lo..hi, f_i coefficient with g_{k-i}
        # We'll build vector of size n
        coeff = np.zeros(n)
        # For i in lo..hi:
        # j = k - i
        # coeff[j] = f[i]
        # Because we want sum_i f_i * g_{k-i}
        # So set coeff[k - i] = f_i
        coeff[ k - np.arange(lo, hi+1) ] = f[lo:hi+1]
        constraints.append( coeff @ g <= M )
    # maybe also enforce sum(g) <= something? not needed for objective
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)  # adjust
    if g.value is None:
        return None
    g_vec = np.maximum(g.value, 0.0)  # ensure non-neg and maybe set small epsilon
    # Normalize to sum 1
    if g_vec.sum() > 0:
        g_vec = g_vec / g_vec.sum()
    else:
        g_vec = np.full(n, 1.0/n)
    return g_vec
```

def solve_lp(f, M, active_lags):
    A = build_constraint_matrix(f, active_lags)
    n = len(f)
    if A.shape[0] == 0:
        # no constraints, fallback to uniform
        return np.full(n, 1.0/n)
    b = np.full(A.shape[0], M)
    c = -np.ones(n)  # maximize sum
    bounds = [(0.0, 1000.0) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success or res.x is None:
        # fallback: return current f unchanged
        return f.copy()
    g = np.maximum(res.x, 0.0)
    # Normalize to sum 1
    sum_g = g.sum()
    if sum_g <= 0:
        g = np.full(n, 1.0/n)
    else:
        g = g / sum_g
    return g
```

    n = x.shape[0]

"""
Propose a step‐function (non‑negative sequence) that minimises the
evaluation function

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three stages:

1️⃣  Quick random‑support sampling – many candidate vectors are created by
    picking a random support set (size from a few hundred up to the full
    length) and assigning random positive weights on that support.
    The best candidate according to the cheap objective
        2·n·max(conv(a,a))          (sum(a) is forced to 1)
    is kept.

2️⃣  LP‑guided refinement – starting from the best candidate, an active‑set
    linear program is solved repeatedly.  For the current vector `f`,
    let `M = max(conv(f,f))`.  The LP maximises `∑ g_j` subject to
        g ≥ 0
        f * g ≤ M               (only the most violating lags are kept)
    i.e. the convolution of `f` with the new vector `g` never exceeds the
    current worst value `M`.  After solving, `g` is normalised to a
    probability vector and the update

        f ← (1‑t)·f + t·g

    is performed with a step size `t` that decays linearly with time.
    Occasionally a tiny Gaussian jitter is added to escape plateaus.

3️⃣  Final secure evaluation – the best vector found is passed to the
    supplied `evaluate_sequence` and returned.

The whole routine respects the supplied time budget (default 1000 s) and
uses only standard scientific libraries (`numpy`, `scipy`).  It improves
upon the reference implementation by employing an active‑set linear
program that directly enforces the current max‑convolution bound,
allowing a faster reduction of the upper bound.
"""

import time
import numpy as np
from typing import List

# optional import – SciPy gives a fast LP solver
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself, length 2·len(x)‑1."""
    n = x.shape[0]
    size = 1 << (2 * n - 2).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper to build the (few) active linear constraints  f * g ≤ M
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray, active_lags: np.ndarray) -> np.ndarray:
    """
    For each lag k in `active_lags` build a row vector `row`
    such that  row @ g == (f * g)[k]  (linear in g).
    Returns a dense matrix of shape (len(active_lags), n).
    """
    n = f.shape[0]
    rows = []
    for k in active_lags:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        length = hi - lo + 1
        if length <= 0:
            rows.append(np.zeros(n, dtype=np.float64))
            continue
        # slice of g that participates: indices j = k - i, i in [lo, hi]
        j_start = k - hi          # smallest j
        row = np.zeros(n, dtype=np.float64)
        # contribution from f[lo:hi+1] in reversed order
        row[j_start: j_start + length] = f[lo:hi + 1][::-1]
        rows.append(row)
    if rows:
        return np.vstack(rows)
    else:                         # pragma: no cover
        return np.empty((0, n), dtype=np.float64)


# ----------------------------------------------------------------------
# Linear‑programming step (active‑set)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, active_lags: np.ndarray) -> np.ndarray:
    """
    Solve  max 1·g   subject to   f * g ≤ M   for the supplied active lags.
    Returns a normalised vector (sum == 1).  If SciPy is unavailable or the
    LP fails, `f` itself is returned.
    """
    if not _SCIPY_AVAILABLE:
        return f.copy()

    A = _build_constraint_matrix(f, active_lags)   # shape (c, n)
    if A.shape[0] == 0:
        return f.copy()

    b = np.full(A.shape[0], M, dtype=np.float64)
    # maximise sum(g) → minimise -sum(g)
    c = -np.ones(f.shape[0], dtype=np.float64)

    # bounds 0 ≤ g_i ≤ 1000  (the same clipping used in evaluate_sequence)
    bounds = [(0.0, 1000.0) for _ in range(f.shape[0])]

    # use the high‑performance HiGHS interior‑point method
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})

    if not res.success or res.x is None:
        return f.copy()

    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s == 0.0 or not np.isfinite(s):
        g = np.full_like(g, 1.0 / g.shape[0])
    else:
        g = g / s
    return g
