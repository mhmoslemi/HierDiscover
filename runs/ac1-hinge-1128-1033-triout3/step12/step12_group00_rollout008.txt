The optimizer searches for a non‑negative step‑function `a` (length `n`) that minimizes  

```
E(a) = 2·n·max(conv(a,a)) / (sum(a)^2)
```

(the same objective used by `evaluate_sequence`).  
The main ideas are:

1. **Rich initial pool** – uniform, symmetric Beta families, Dirichlet samples,
   and mixtures of a Beta shape with the uniform distribution.  
   All candidates are normalised to sum = 1 and evaluated exactly.

2. **Symmetry enforcement** – the problem is symmetric (`a` ↔ `rev(a)`),
   therefore we keep the search restricted to (or projected onto) the
   symmetric sub‑space: after each update we replace `a ← (a+rev(a))/2`.

3. **Smooth surrogate optimisation** – we replace the non‑smooth `max`
   by a log‑sum‑exp with a temperature `ε`.  Optimisation is performed in
   the **log‑softmax** (logits) space, guaranteeing the simplex
   constraints automatically.  Two complementary optimisers are used:
   *SciPy* L‑BFGS‑B (as before) and a custom **Adam** optimiser with an
   aggressive learning‑rate schedule.  Both are run repeatedly with a
   decreasing temperature schedule.

4. **Exact sub‑gradient polishing** – after the smooth phases we switch to
   true sub‑gradient descent on the original objective.  The update
   respects the simplex and the symmetry constraint.

5. **Blending & final pass** – convex blends with the uniform distribution
   are tried, a final short L‑BFGS run with a tiny temperature finalises the
   solution.

The whole pipeline respects the 1000 s budget and returns the best
sequence found.

#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
Propose a non‑negative step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm builds a diverse initial pool, refines the best candidates
with two smooth optimisers (L‑BFGS‑B and a handcrafted Adam), then
polishes the result with exact sub‑gradient descent.  All iterates are
projected onto the probability simplex and symmetrised
(a ↔ rev(a)) – the problem is invariant under reversal.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT based convolutions (full linear)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.  The gradient already includes the factor 2·n.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()

    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # Convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # keep the n entries that correspond to the “forward” part
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (logits → probability simplex) – automatic non‑negativity & sum=1
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Symmetric Beta shape (one‑parameter family, automatically symmetric)
# ----------------------------------------------------------------------
def _symmetric_beta_shape(n: int, p: float) -> np.ndarray:
    """
    Returns a symmetric Beta‑like shape:
        a_i ∝ i^{p-1}·(n-i+1)^{p-1}
    for i = 1 … n.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (p - 1.0))
    return a


# ----------------------------------------------------------------------
# Random Dirichlet candidate (cheap)
# ----------------------------------------------------------------------
def _random_dirichlet_candidate(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `alpha`."""
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Project onto the probability simplex (used after sub‑gradient steps)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 1500) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smoothed objective.
    """
    # initialise in log‑space (soft‑max guarantees feasibility)
    x0 = np.log(a0 + 1e-12)

    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        # chain rule for soft‑max
        return a * (grad_a - dot)

    time_left = deadline - time.time()
    if time_left < 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    # final symmetrisation
    a_opt = (a_opt + a_opt[::-1]) / 2.0
    a_opt /= a_opt.sum()
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate (with symmetry)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   eps_start: float = 5e-1,
                   eps_end: float = 1e-12,
                   max_steps: int = 2500) -> tuple[np.ndarray, float]:
    """
    Custom Adam optimiser on the smooth objective.
    Symmetry is enforced after each soft‑max step.
    """
    # initialise logits
    x = np.log(a0 + 1e-12)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_num = 1e-8
    t = 0

    while time.time() < deadline and t < max_steps:
        t += 1
        # temperature schedule (exponential decay)
        prog = t / max_steps
        eps = eps_start * (eps_end / eps_start) ** prog

        a = _softmax(x)
        # enforce symmetry *before* computing the gradient
        a = (a + a[::-1]) / 2.0
        a /= a.sum()

        smooth_val, grad_a = _smooth_max_and_grad(a, eps)

        # chain rule for soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule: start large, decay like 1/√t
        lr = 0.5 / (np.sqrt(t) + 1.0)

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_num)

        # occasional exact evaluation (every 200 steps)
        if t % 200 == 0 or t == max_steps:
            a_candidate = _softmax(x)
            a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
            a_candidate /= a_candidate.sum()
            val = _objective(a_candidate)
            if val < best_val:
                best_val, best_a = val, a_candidate.copy()
                print(f"[Adam] new best = {best_val:.6f} (step {t})")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution) with symmetry
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a))
    using the analytic sub‑gradient.  After each projection the iterate is
    symmetrised.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        active = np.where(b == max_b)[0]

        # sub‑gradient: sum contributions from all active shifts
        grad = np.zeros_like(a)
        for k in active:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple diminishing step size
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        # enforce symmetry
        a_new = (a_new + a_new[::-1]) / 2.0
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 500 == 0 and np.random.rand() < 0.02:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)
            a = (a + a[::-1]) / 2.0
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        candidates.append((_objective(a0), a0))
        print(f"[Init] interpolated previous solution, value={candidates[-1][0]:.6f}")
    except Exception:
        pass

    # Uniform baseline
    uniform = np.full(n, 1.0 / n)
    candidates.append((_objective(uniform), uniform))

    # Symmetric Beta family (single parameter)
    for logp in np.linspace(-9, 2, 28):
        p = 10 ** logp
        a = _symmetric_beta_shape(n, p)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        candidates.append((_objective(a), a))

    # Random Dirichlet draws (various concentrations)
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            a = _random_dirichlet_candidate(n, alpha, rng)
            candidates.append((_objective(a), a))

    # Mixtures Beta–Uniform
    for _ in range(12):
        w = rng.uniform(0.0, 1.0)
        p = np.exp(rng.uniform(-9, 2))
        beta = _symmetric_beta_shape(n, p)
        beta /= beta.sum()
        a = w * beta + (1.0 - w) * uniform
        a /= a.sum()
        candidates.append((_objective(a), a))

    # Keep the top few for the exact phases
    candidates.sort(key=lambda t: t[0])
    top_k = 8
    best_val, best_a = candidates[0][0], candidates[0][1].copy()
    print(f"[Init] best after sweep: value={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Smooth optimisation (L‑BFGS & Adam)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    eps_start = 5e-1
    eps_final = 1e-12

    # L‑BFGS pass (multiple restarts)
    while time.time() < phase_a_deadline and time.time() < deadline:
        prog = (time.time() - start_time) / (phase_a_deadline - start_time)
        eps = eps_start * (eps_final / eps_start) ** prog
        for i in range(min(top_k, len(candidates))):
            if time.time() >= phase_a_deadline:
                break
            a_start = candidates[i][1]
            best_a, best_val = _lbfgs_optimize(
                a_start, n, deadline, start_time,
                best_a, best_val, eps=eps, max_iter=1200)
            best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # Adam fine‑tuning (starting from current best)
    if time.time() < deadline:
        best_a, best_val = _adam_optimize(
            best_a, n, deadline, start_time,
            best_a, best_val,
            eps_start=eps_start, eps_end=eps_final, max_steps=1500)

    # --------------------------------------------------------------
    # Phase B – Exact sub‑gradient polishing (≈40 % of remaining budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.90 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=400_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase C – Final short L‑BFGS with ultra‑tiny ε
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(
            best_a, n, deadline, start_time,
            best_a, best_val, eps=1e-12, max_iter=800)

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:
        print("[Final] evaluate_sequence raised:", exc)

    # Return plain list of floats
    return best_a.tolist()
