def simplex_projection(v):
    """Projects a vector v onto the probability simplex (non-negative entries summing to 1)."""
    # v can be any real vector
    # Using algorithm from Wang & Carreira-Perpiñán 2013
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

The algorithm consists of three stages:

1️⃣  **Randomised seeding** – many candidates with random supports and
    Dirichlet‑distributed weights are generated; the best cheap objective
    (2·n·max(conv)) is kept.

2️⃣  **Adam‑based mirror descent** – starting from the best candidate,
    a sub‑gradient of the max‑convolution (taken over the full active
    set) is used in an Adam optimiser.  The update is multiplicative
    (exponentiated‑gradient) which automatically stays on the probability
    simplex.  Adaptive step‑sizes, momentum and occasional Gaussian jitter
    help to escape plateaus.

3️⃣  **Local swap search** – after the Adam phase a simple coordinate‑wise
    hill‑climbing (moving a random fraction of mass from one index to
    another) is performed while time permits.  Only moves that improve
    the cheap objective are accepted, giving a final fine‑tuning.

The whole routine respects a 1000 s budget, prints progress and finally
calls ``evaluate_sequence`` on the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 2).bit_length())       # next power‑of‑two ≥ 2·n‑1
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Cheap proxy used during search: 2·n·max(conv(a,a)) assuming Σa = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simplex projection (L1‑norm = 1, non‑negative)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a real vector onto the probability simplex.
    Implementation follows the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (full active set)
# ----------------------------------------------------------------------


def _subgradient_full(a: np.ndarray, conv: np.ndarray, rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    All entries of ``conv`` that are within ``rel_tol`` of the maximum are
    treated as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    active = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # a[k-idx] is the partner element producing the product a[idx] * a[k-idx]
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad

# Main entry point
# ----------------------------------------------------------------------

        weights = rng.random(k) + 1e-6
        a = np.zeros(n, dtype=np.float64)
        a[idx] = weights
        a /= a.sum()
        _maybe_update(a)

        # (b) Small tent‑shaped perturbation on the same support
        if rng.random() < 0.07:
            t = np.arange(k, dtype=np.float64)
            tent = 1.0 - np.abs(t - (k - 1) / 2) / (k / 2)
            a2 = np.zeros_like(a)
            a2[idx] = tent + 1e-6
            a2 /= a2.sum()
            _maybe_update(a2)

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑based mirror‑descent refinement
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)
        print("[fallback] uniform start", best_val)

    a = best_seq.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[adam] iter {it:04d} new best cheap objective = {best_val:.6f}")

        # sub‑gradient over *all* active entries
        grad = _subgradient_full(a, conv, rel_tol=1e-12)

        # Adam moments
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size – √ decay (more aggressive early, gentle later)
        step = 0.5 / np.sqrt(it)

        # exponentiated‑gradient update (mirror descent on the simplex)
        log_factor = -step * (m_hat / (np.sqrt(v_hat) + eps))
        log_factor = np.clip(log_factor, -40.0, 40.0)   # protect against overflow
        a = a * np.exp(log_factor)

        # project back onto the simplex
        a = _project_onto_simplex(a)

        # occasional Gaussian jitter to break flat plateaux
        if rng.random() < 0.001:
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # 3️⃣  Local swap‑based hill climbing (fine‑tuning)
    # ------------------------------------------------------------------
    while time.time() < deadline - 0.5:
        # pick a donor with non‑trivial mass
        donors = np.where(a > 1e-6)[0]
        if donors.size == 0:
            break
        i = rng.choice(donors)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move a random fraction (10‑% .. 50‑%) of the donor's mass
        frac = rng.uniform(0.1, 0.5)
        delta = a[i] * frac
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate /= a_candidate.sum()
        conv_c = _full_convolution(a_candidate)
        val_c = 2.0 * n * conv_c.max()
        if val_c < best_val:
            best_val = val_c
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[swap] new best cheap objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final evaluation using the official ``evaluate_sequence`` function
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return a plain list as required by the framework
    return best_seq.tolist()
