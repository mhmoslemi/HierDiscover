def _softmax_variance_obj_grad(z, alpha, lambda_var):

   n = len(z)

   a = np.exp(z); a = a / a.sum()

   conv = _full_convolution(a)

   maxc = conv.max()

   # softmax weights
   exp_scaled = np.exp(alpha * (conv - maxc))
   sumexp = exp_scaled.sum()
   w_soft = exp_scaled / sumexp

   # Variance weights
   avg = conv.mean()  # ideally 1/(2*n-1) but compute
   w_var = 2.0 * lambda_var * (conv - avg)

   w = w_soft + w_var

   # convolution for g1
   w_rev = w[::-1]
   conv_g1 = _full_convolution_two(a, w_rev)  # length 3n-2?
   len_w = 2 * n - 1
   g1 = conv_g1[len_w-1::-1][:n]  # reversed slice

   # window sum:
   ones = np.ones(n)
   conv_win = _full_convolution_two(w, ones)
   s_window = conv_win[n-1 : n-1 + n]  # length n

   g2 = a * s_window

   g_a = g1 + g2

   dot = np.dot(g_a, a)
   grad_z = a * (g_a - dot)

   # objective
   soft_max = maxc + np.log(sumexp) / alpha
   var = ((conv - avg) ** 2).mean()
   obj = soft_max + lambda_var * var

   return obj, grad_z

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, scipy.optimize as opt

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine n and initialize a0
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        a0 = np.asarray(old_seq, dtype=float)
        n = a0.shape[0]
        # Interpolate if needed? assume length matches? Keep as is.
        if n != 8000: # we can keep as is.
            # For fairness use n from old_seq.
            pass
    else:
        n = 8000
        i = np.arange(1, n+1, dtype=np.float64)
        a0 = (i * (n - i + 1))  # beta(2,2) shape
        a0 = a0 / a0.sum()

    # Ensure nonnegative and sum to 1
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    a0 = a0 / s if s > 0 else np.full(n, 1.0/n)

    # Convert to logits
    eps = 1e-30
    z = np.log(a0 + eps)

    # Hyperparameters
    alpha = 150.0  # temperature for softmax
    lambda_var_candidates = [0.0, 0.05, 0.1, 0.2, 0.5]
    best_a = a0.copy()
    best_obj = 2.0 * n * _full_convolution(best_a).max()

    # Function to evaluate exact objective given vector a
    def exact_obj(a):
        conv = _full_convolution(a)
        return 2.0 * n * conv.max()

    # Outer loop over variance weights
    for lam in lambda_var_candidates:
        if time.time() >= deadline:
            break

        # Run L-BFGS starting from current logits
        try:
            res = opt.minimize(fun=lambda zz: _softmax_variance_obj_grad(zz, alpha, lam),
                               x0=z,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': 120, 'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        if not res.success:
            print("[lbfgs] not successful, status:", res.message)

        z_opt = res.x
        # convert to probabilities
        a_opt = np.exp(z_opt)
        a_opt = a_opt / a_opt.sum()

        # Polish with entropy‑mirror descent
        a_polished = _max_refine(a_opt, steps=2000, eta_start=0.45)

        obj_val = exact_obj(a_polished)

        if obj_val < best_obj:
            best_obj = obj_val
            best_a = a_polished.copy()
            print(f"[update] lam={lam:.3f} improved exact O={best_obj:.6f}")

    # Additional polishing on best_a
    best_a = _max_refine(best_a, steps=5000, eta_start=0.45)

    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (exact O={best_obj:.6f})")
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    return final_seq

def _full_convolution(x):
    n = x.shape[0]
    out_len = 2*n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv
```

def _full_convolution_two(x, y):

    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]

"""
Propose a candidate step‑function that (hopefully) beats the previous
record (≈1.508).  The method differs from the earlier Adam → mirror‑descent
pipeline:

* Start from the previous best solution (if available) or from a smooth
  Beta(2, 2) shape and work in the log‑domain (logits) to guarantee
  positivity.
* Optimise a *smooth* surrogate of the true objective:
      L(z) = softmax_α(conv(a)) + λ·Var(conv(a)),
  where a = softmax(z) (so Σa=1) and α is a large temperature.
  The variance term penalises large deviations of the self‑convolution
  and encourages a flatter profile, i.e. a more uniform sum‑distribution.
* The surrogate gradient is obtained analytically using only three
  FFT‑based convolutions (a*a, w*a_rev and w*1), which makes the
  evaluation cheap even for 8000 coefficients.
* The surrogate is minimised with L‑BFGS‑B (max ≈ 120 iterations) for a
  handful of λ values (0 … 0.5).  Each run yields an improved probability
  vector a.
* Each candidate is *polished* with the entropy‑mirror descent that was
  used in the previous solution (exact max‑convex sub‑gradient steps).
* The best vector found (lowest exact objective 2·n·max(conv)) is finally
  returned as a plain list of floats.  All operations respect the
  1000 s time budget.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# FFT‑based linear convolutions (real input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two (real) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the *exact* max‑entry of the self‑convolution
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 5000, eta_start: float = 0.5) -> np.ndarray:
    """Exact mirror‑descent polishing on max(conv(a,a))."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]                    # sub‑gradient

        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Surrogate objective (soft‑max of the convolution) plus variance term
# ----------------------------------------------------------------------
def _softmax_variance_obj_grad(z: np.ndarray,
                               alpha: float,
                               lambda_var: float) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) for the surrogate
        L(z) = softmax_α(conv(a)) + λ·Var(conv(a)),
    where a = softmax(z).  Gradient is w.r.t. the logits z.
    """
    n = z.shape[0]
    # probabilities
    a = np.exp(z)
    a = a / a.sum()

    # self‑convolution
    conv = _full_convolution(a)                     # length 2·n‑1
    maxc = conv.max()

    # ----- soft‑max part -------------------------------------------------
    exp_scaled = np.exp(alpha * (conv - maxc))       # ≤ 1
    sumexp = exp_scaled.sum()
    w_soft = exp_scaled / sumexp                    # Σ w_soft = 1

    # ----- variance part ------------------------------------------------
    m = 2 * n - 1                                    # number of convolution entries
    avg = conv.mean()                               # = 1/m  (exact, but computed)
    w_var = (2.0 * lambda_var / m) * (conv - avg)   # derivative of λ·Var

    # total weight for the linear combination of entries
    w = w_soft + w_var                               # shape (2·n‑1,)

    # ----- gradient w.r.t. a (size n) ------------------------------------
    # term g1_i = Σ_j w_{i+j} a_j  (cross‑correlation a ⋆ w)
    w_rev = w[::-1]
    conv_g1 = _full_convolution_two(a, w_rev)       # length 3·n‑2
    len_w = 2 * n - 1
    g1 = conv_g1[len_w - 1 :: -1][:n]                # g1_i = conv_g1[len_w-1-i]

    # term g2_i = a_i * Σ_{k=i}^{i+n-1} w_k  (window sums of w)
    ones = np.ones(n)
    conv_win = _full_convolution_two(w, ones)       # length 3·n‑2
    s_window = conv_win[n - 1 : n - 1 + n]          # window sums, length n
    g2 = a * s_window

    g_a = g1 + g2                                    # ∂L/∂a

    # ----- chain rule through the soft‑max (logits) ----------------------
    dot = np.dot(g_a, a)
    grad_z = a * (g_a - dot)                         # ∂L/∂z

    # ----- surrogate objective value ------------------------------------
    soft_max = maxc + np.log(sumexp) / alpha
    var = ((conv - avg) ** 2).mean()
    obj = soft_max + lambda_var * var

    return float(obj), grad_z.astype(np.float64)


# ----------------------------------------------------------------------
# Callable wrapper for L‑BFGS‑B (no closures, only default arguments)
# ----------------------------------------------------------------------
class _LbfgsCallable:
    """Encapsulates a surrogate objective for a given (α, λ)."""
    def __init__(self, alpha: float, lambda_var: float):
        self.alpha = alpha
        self.lambda_var = lambda_var

    def __call__(self, z: np.ndarray) -> tuple[float, np.ndarray]:
        return _softmax_variance_obj_grad(z, self.alpha, self.lambda_var)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s
    time budget and tries a few variance‑regularised L‑BFGS runs followed
    by an exact mirror‑descent polishing.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5            # small safety margin

    # ------------------------------------------------------------------
    # Determine problem size and initialise a probability vector
    # ------------------------------------------------------------------
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        a0 = np.asarray(old_seq, dtype=float)
        n = a0.shape[0]
        a0 = np.maximum(a0, 0.0)
    else:
        n = 8000
        i = np.arange(1, n + 1, dtype=np.float64)
        a0 = i * (n - i + 1)                     # discrete Beta(2,2)
    a0_sum = a0.sum()
    if a0_sum == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0_sum                         # normalise to sum = 1

    # work in the log‑domain (logits)
    eps = 1e-30
    z = np.log(a0 + eps)

    # ------------------------------------------------------------------
    # Hyper‑parameters for the surrogate optimisation
    # ------------------------------------------------------------------
    alpha = 150.0                                 # temperature for soft‑max
    lambda_candidates = [0.0, 0.05, 0.1, 0.2, 0.5]

    # ------------------------------------------------------------------
    # Helper for the exact objective (2·n·max(conv))
    # ------------------------------------------------------------------
    def _exact_obj(a_vec: np.ndarray) -> float:
        conv = _full_convolution(a_vec)
        return 2.0 * n * conv.max()

    best_a = a0.copy()
    best_obj = _exact_obj(best_a)

    # ------------------------------------------------------------------
    # Scan a few variance‑weights, each time solving the smooth problem
    # with L‑BFGS‑B and polishing the result.
    # ------------------------------------------------------------------
    for lam in lambda_candidates:
        if time.time() >= deadline:
            break

        callable_obj = _LbfgsCallable(alpha, lam)

        try:
            res = _opt.minimize(fun=callable_obj,
                                x0=z,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': 120, 'disp': False})
        except Exception as exc:
            print(f"[lbfgs] exception (λ={lam}):", exc)
            continue

        if not res.success:
            # L‑BFGS can fail to converge but still return a useful point.
            print(f"[lbfgs] warning – λ={lam}, status:", res.message)

        z_opt = res.x
        a_opt = np.exp(z_opt)
        a_opt = a_opt / a_opt.sum()

        # Polishing with exact mirror descent (much cheaper than the
        # full Adam stage of the previous code).
        a_polished = _max_refine(a_opt, steps=2000, eta_start=0.45)

        obj_val = _exact_obj(a_polished)

        if obj_val < best_obj:
            best_obj = obj_val
            best_a = a_polished.copy()
            print(f"[update] λ={lam:.3f} improved exact O={best_obj:.6f}")

        # keep the latest logits as the starting point for the next λ
        z = z_opt.copy()

    # ------------------------------------------------------------------
    # Final polishing on the best vector discovered
    # ------------------------------------------------------------------
    best_a = _max_refine(best_a, steps=5000, eta_start=0.45)

    # ------------------------------------------------------------------
    # Convert to plain list and evaluate (the wrapper does the safety checks)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(exact O≈{best_obj:.6f})")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
