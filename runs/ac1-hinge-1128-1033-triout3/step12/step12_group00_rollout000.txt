The optimiser searches for a non‑negative step‑function `a` (length *n*, `sum(a)=1`) that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}=2\,n\;\max (a*a)
\]

(the exact same quantity returned by `evaluate_sequence`).  
The code follows three phases:

1. **Diverse random sweep** – a large set of Beta‑ and Dirichlet‑distributed
   candidates (and, if available, the previously discovered sequence) is
   generated and evaluated with the exact objective.  
   The best few are kept.

2. **Adam optimisation on a smooth surrogate** – the non‑smooth `max`
   is replaced by a log‑sum‑exp (smooth‑max) with a geometric schedule for the
   smoothing parameter `ε`.  Adam (adaptive‑momentum) updates are performed on
   the surrogate gradient, followed by an Euclidean projection onto the simplex
   (non‑negative, unit‑sum).  Each restart is limited by the remaining time.

3. **Exact sub‑gradient polishing & random local moves** – after the smooth
   optimisation a classical projected sub‑gradient descent on the *true*
   objective is executed (as in the reference solution) and a cheap stochastic
   “mass‑shift” search is applied to escape tiny plateaus.

The whole pipeline respects the 1000 s budget, prints intermediate bounds,
and finally returns the best vector found (as a plain list of `float`s).

```python
"""
Propose a step‑function of non‑negative heights that minimises
evaluate_sequence(sequence).  The algorithm consists of:

1.   A wide random sweep (Beta / Dirichlet samples + possible reuse of a
     previously stored solution) – exact evaluation, keep the best few.
2.   Adam optimisation on a smooth max‑surrogate (log‑sum‑exp) with a
     geometric ε‑schedule.  After each update the vector is projected onto
     the probability simplex.
3.   Exact projected sub‑gradient polishing (as in the reference code) and
     a cheap stochastic “mass‑shift” local search.

All stages stop automatically when the overall time budget (≈1000 s) is
exhausted.  The returned list sums to 1, so evaluate_sequence simply
returns 2·n·max(conv(a,a)).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a)=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient.
# The returned gradient already includes the factor 2·n.
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_max, grad) where smooth_max approximates max(conv(a,a))
    and grad = ∇_a ( 2·n·smooth_max ).
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp stabilisation
    shifted = (b - max_b) / eps
    w = np.exp(shifted)
    p = w / w.sum()                               # probabilities, length 2n‑1

    # convolution a * reversed(p)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 3n‑2, but we need the middle part
    segment = conv_ab[n - 1:2 * n - 1]            # exactly n entries
    grad = 4.0 * n * segment[::-1]                # ∇ (2·n·smooth_max)

    smooth_val = max_b + eps * np.log(w.sum())
    return smooth_val, grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random Beta‑family candidate (cheap initial sweep)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Beta‑shaped density on {1,…,n} and normalise it."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ≈ 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   eps_start: float,
                   eps_end: float,
                   max_iters: int = 2500) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth‑max surrogate.
    eps is annealed geometrically from eps_start to eps_end.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0
    base_lr = 0.4          # initial learning rate
    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # geometric schedule for ε
        prog = it / max_iters
        eps = eps_start * (eps_end / eps_start) ** prog

        smooth_val, grad = _smooth_max_and_grad(a, eps)   # grad already = ∇(2·n·smooth)
        # Adam moments
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (more aggressive early)
        lr = base_lr * (1.0 - prog) + 1e-4 * prog

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = _project_simplex(a)

        # occasional exact evaluation (cheap, every 100 iter)
        if (it & 0x3F) == 0:
            val_exact = _objective(a)
            if val_exact + 1e-12 < best_val:
                best_val = val_exact
                best_a = a.copy()
                print(f"[Adam] iter {it:4d} new best = {best_val:.6f}")

    # final check
    val_final = _objective(a)
    if val_final < best_val - 1e-12:
        best_val, best_a = val_final, a.copy()
        print(f"[Adam] final improvement = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum of contributions from all maximal diagonals
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple diminishing step size
        progress = (time.time() - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if (it % 400) == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a set of t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap stochastic local moves (mass‑shift)
# ----------------------------------------------------------------------
def _random_local_search(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a small amount of mass from one index to another.
    Accept only improving moves.
    """
    a = a0.copy()
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # transfer at most 2 % of the source mass
        delta = min(a[i], 0.02 * rng.random())
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta
        # No projection needed – sum stays 1 and all entries stay ≥0.
        val = _objective(a)
        if val + 1e-12 < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Local] iter {it:6d} new best = {best_val:.6f}")
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta
        it += 1
    return best_a, best_val
```
