        # ensure length exactly n
        if a.shape[0] != n:
            a = a[:n]
        # Clip tiny negatives due to rounding
        a = np.maximum(a, 0.0)
        # Normalize sum to 1 (just in case rounding)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        return a

    # Convolution function using FFT
    def conv_max(a):
        conv = _full_convolution(a)
        return conv.max()

    # Objective (sum normalized to 1)
    def objective(a):
        return 2.0 * n * conv_max(a)

    # Initial candidate uniform
    p_uniform = np.ones(K, dtype=float)
    a_uniform = build_a(p_uniform)
    best_a = a_uniform.copy()
    best_score = objective(best_a)
    best_p = p_uniform.copy()
    print("[init] uniform score =", best_score)

    # keep current candidate for local search
    cur_p = best_p.copy()
    cur_a = best_a.copy()
    cur_score = best_score

    iter_cnt = 0
    stagn = 0
    max_stagn = 5000

    while time.time() < deadline:
        iter_cnt += 1
        # decide explorer vs local
        if np.random.rand() < 0.2:
            # pure random point (exponential distribution)
            p_candidate = np.random.exponential(scale=1.0, size=K)
        else:
            # perturb best known parameters
            # scale relative perturbation
            sigma = 0.05 * (1 + np.exp(-stagn / 2000.0))
            perturb = np.random.randn(K) * sigma
            p_candidate = best_p * (1.0 + perturb)
            # enforce positivity
            p_candidate = np.where(p_candidate > 1e-12, p_candidate, 1e-12)

        a_candidate = build_a(p_candidate)
        cand_score = objective(a_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            best_p = p_candidate.copy()
            stagn = 0
            print(f"[iter {iter_cnt:06d}] new best = {best_score:.9f} "
                  f"(time {time.time() - start:.1f}s)")
        else:
            stagn += 1
            # occasionally accept worse to maintain diversity
            if np.random.rand() < 0.001:
                cur_p = p_candidate
                cur_a = a_candidate
                cur_score = cand_score

        # periodic random restart if stagnated too long

            cur_score = objective(cur_a)

        print("[final] evaluate_sequence =", final_score)
    except Exception as e:
        print("[final] evaluate_sequence error:", e)

"""
Propose a candidate step‑function using a low‑dimensional (piece‑wise‑constant)
parameterisation combined with a simple evolutionary search.

The idea is to split the length‑n vector into K contiguous blocks (K≈200 for
n=8000).  All entries inside a block share the same height; the block heights
are the optimisation variables.  After normalising to sum‑to‑one the
objective 2·n·max(a*a) can be evaluated quickly with a real FFT.  The search
alternates between random exploration (exponential draws) and local
perturbations of the current best parameters, with occasional restarts when
the search stagnates.  The final best vector is returned and evaluated once
with the provided ``evaluate_sequence`` function.

This approach explores a much broader space than a plain sub‑gradient walk on
the full 8000‑dimensional vector while still allowing enough resolution to
beat the previous best bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helper – linear (non‑circular) convolution of a real vector with itself
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding the FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*x (real FFT, length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence discovered (as a list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – we keep the same length used by the reference code.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Piece‑wise constant representation:
    #   K blocks of (approximately) equal length.
    # --------------------------------------------------------------
    K = 200                         # number of blocks (tunable)
    seg_len = n // K
    remainder = n % K
    seg_lengths = np.full(K, seg_len, dtype=int)
    seg_lengths[:remainder] += 1    # first `remainder` blocks are 1 longer

    # --------------------------------------------------------------
    # Helper: build the full vector from a block‑height vector `p`.
    # --------------------------------------------------------------
    def _build_a(p: np.ndarray) -> np.ndarray:
        """
        Given a positive K‑dim vector p, create a length‑n vector whose entries
        are constant on each block.  The result is normalised to sum to 1.
        """
        total_mass = np.dot(p, seg_lengths)          # ∑ p_i·len_i
        if total_mass <= 0.0:
            # Fallback to the uniform distribution (should never happen)
            return np.full(n, 1.0 / n, dtype=float)

        # value taken by every coordinate inside block i
        per_block_val = p / total_mass                # because of later normalisation
        # repeat each block value according to its length
        a = np.repeat(per_block_val, seg_lengths)
        # safety trim (numerical rounding could give a length off by 1)
        if a.shape[0] != n:
            a = a[:n]
        # Clip tiny negatives (should not appear) and re‑normalise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        return a

    # --------------------------------------------------------------
    # Objective (sum is already forced to 1)
    # --------------------------------------------------------------
    def _objective(a: np.ndarray) -> float:
        conv = _full_convolution(a)
        return 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # Initial candidate – uniform block heights
    # --------------------------------------------------------------
    p_uniform = np.ones(K, dtype=float)
    a_uniform = _build_a(p_uniform)
    best_a = a_uniform.copy()
    best_p = p_uniform.copy()
    best_score = _objective(best_a)
    print("[init] uniform score =", best_score)

    # --------------------------------------------------------------
    # Search loop
    # --------------------------------------------------------------
    stagnation = 0           # number of iterations without improvement
    max_stagn = 5000         # after this we trigger a random restart
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ----------------------------------------------------------
        # 1) generate a new candidate: 20 % pure random, else local perturb.
        # ----------------------------------------------------------
        if np.random.rand() < 0.20:
            # pure exploration – exponential draws give a positive vector
            p_candidate = np.random.exponential(scale=1.0, size=K)
        else:
            # local perturbation of the current best parameters
            # relative perturbation size slowly shrinks with stagnation
            sigma = 0.05 * (1.0 + np.exp(-stagnation / 2000.0))
            perturb = np.random.randn(K) * sigma
            p_candidate = best_p * (1.0 + perturb)
            # enforce strict positivity
            p_candidate = np.where(p_candidate > 1e-12, p_candidate, 1e-12)

        a_candidate = _build_a(p_candidate)
        cand_score = _objective(a_candidate)

        # ----------------------------------------------------------
        # 2) accept if we improve the global best
        # ----------------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            best_p = p_candidate.copy()
            stagnation = 0
            print(f"[iter {iteration:06d}] new best = {best_score:.9f} "
                  f"(time {time.time() - start_time:.1f}s)")
        else:
            stagnation += 1
            # very occasional uphill move to keep diversity
            if np.random.rand() < 0.001:
                pass  # ignore – we keep the current best only

        # ----------------------------------------------------------
        # 3) restart if we have been stuck for a while
        # ----------------------------------------------------------
        if stagnation > max_stagn:
            # completely new random point (keeps global best untouched)
            p_candidate = np.random.exponential(scale=1.0, size=K)
            a_candidate = _build_a(p_candidate)            # noqa: F841
            stagnation = 0
            print(f"[reset] random restart at iter {iteration} "
                  f"(time {time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # Final secure evaluation (single call)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] evaluate_sequence =", final_score)
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return the best raw vector (list of floats)
    return best_a.tolist()
