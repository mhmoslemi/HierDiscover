        val_new = _objective(a_new)

    # evaluate using evaluate_sequence

"""
Optimiser for the step‑function challenge (≈1000 s budget).

Strategy
--------
1. **Coarse sweep on a small problem** –  n≈2500 – many Beta/Dirichlet
   draws, exact objective evaluation, keep the best few seeds.

2. **Smooth‑max L‑BFGS** – minimise a log‑sum‑exp surrogate of max(conv(a,a))
   (variables are in soft‑max form, guaranteeing non‑negativity and unit sum).

3. **Exact sub‑gradient polishing** – projected sub‑gradient descent on the
   true objective (max‑convolution).  Performed on the small problem where
   it is cheap.

4. **Upsample** the best small‑n solution to the target size (n=12000).

5. **Linear‑programming refinement** – with the current bound M=
   max(conv(a,a)) we maximise Σg under the linear constraints (a∗g)≤M.
   Only the currently active (binding) constraints are kept, making the
   LP tiny.  We then convex‑blend the obtained g into a.

6. **Final L‑BFGS + sub‑gradient** on the full size and a short simulated‑annealing
   perturbation loop.

All steps respect a global time‑budget; the best sequence found is returned
as a plain list of floats.  The code works without any file or network IO.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Convolution helpers (FFT – fully vectorised, zero‑padded)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient.
# Both already include the prefactor 2·n.
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns smooth max(b) (where b = conv(a,a)) and the gradient of 2·n·smooth_max
    w.r.t. a.  ``eps`` is the smoothing parameter.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p (needed for gradient)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # the n entries that matter
    grad_a = 4.0 * n * segment[::-1]             # already contains factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max re‑parameterisation (guarantees a≥0, Σa=1)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 1200) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on the smoothed objective."""
    # initialise in log‑space
    x0 = np.log(a0 + 1e-12)

    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    def jac(x):
        a = _softmax(x)
        _, grad_a = _smooth_max_and_grad(a, eps)   # grad_a already includes 2·n
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    # respect remaining budget
    if deadline - time.time() < 0.5:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:          # pragma: no cover
        print("[L‑BFGS] failed:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (uses the true max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    cur_val = _objective(a)
    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple step‑size schedule (aggressive → tiny)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 400 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps to smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta‑family candidate (cheap initial sweep)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ~1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set LP)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               deadline: float,
               start_time: float,
               max_iters: int = 50,
               blend_t: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Given a current vector a (Σa=1), repeatedly:
        * compute M = max(conv(a,a))
        * solve max Σg  subject to g≥0 and (a∗g)≤M  (only active constraints)
        * blend g into a.
    This is a cheap surrogate of the alternating LP scheme described in the
    problem statement.
    """
    n = a.shape[0]
    best_a, best_val = a.copy(), _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        M = b.max()
        # select indices with value very close to the maximum (active set)
        active = np.where(b >= M * (1 - 1e-8))[0]
        # cap the active set size to keep the LP tiny
        if active.size > 30:
            active = active[:30]

        # build the (few) linear constraints a * g <= M
        rows = []
        for k in active:
            row = np.zeros(n)
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            # indices j where a_{k-j} exists
            idx = k - np.arange(j_start, j_end + 1)
            row[j_start:j_end + 1] = a[idx]
            rows.append(row)

        A_ub = np.vstack(rows) if rows else np.empty((0, n))
        b_ub = np.full(A_ub.shape[0], M)

        # maximise Σg  <=> minimise -Σg
        c = -np.ones(n)
        bounds = [(0, None)] * n
        # solve tiny LP (highs is fast for this size)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception as exc:               # pragma: no cover
            print("[LP] linprog failed:", exc)
            break

        if not res.success:
            # if the LP failed, give up on LP‑refinement
            break

        g = res.x
        s = g.sum()
        if s < 1e-12:
            break
        g = g / s                                 # enforce Σg=1

        # convex blend
        a_new = (1.0 - blend_t) * a + blend_t * g
        a_new = _project_simplex(a_new)
        val_new = _objective(a_new)

        if val_new + 1e-12 < best_val:
            best_val, best_a = val_new, a_new.copy()
            a = a_new
            print(f"[LP] iter {it+1} new best = {best_val:.6f}")
        else:
            # no improvement – stop early
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (Σ=1) that minimises the
    ``evaluate_sequence`` score, using a 1000 s time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a small safety margin

    # --------------------------------------------------------------
    # 1️⃣  Small‑scale exploration (n≈2500) – cheap but expressive.
    # --------------------------------------------------------------
    n_small = 2500
    candidates = []

    # uniform seed
    uniform_small = np.full(n_small, 1.0 / n_small)
    candidates.append((_objective(uniform_small), uniform_small))

    # possibly reuse a previously found solution (if available)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.size > 0:
            xs_old = np.linspace(0.0, 1.0, old.size)
            xs_new = np.linspace(0.0, 1.0, n_small)
            a_interp = np.interp(xs_new, xs_old, old)
            a_interp = np.maximum(a_interp, 0.0)
            a_interp /= a_interp.sum()
            candidates.append((_objective(a_interp), a_interp))
            print("[Init] interpolated previous solution (small).")
    except Exception:
        pass

    # random Beta family
    for _ in range(300):
        cand = _random_beta_candidate(n_small, rng)
        candidates.append((_objective(cand), cand))

    # Dirichlet draws with several concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            cand = rng.dirichlet(np.full(n_small, alpha))
            candidates.append((_objective(cand), cand))

    # keep the best few seeds
    candidates.sort(key=lambda t: t[0])
    top_k = 5
    best_a, best_val = candidates[0][1], candidates[0][0]
    print(f"[Init] small‑scale best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smooth‑max L‑BFGS on the small problem
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.35 * budget_s
    eps_start, eps_end = 5e-1, 1e-12

    while time.time() < phase_a_deadline and time.time() < deadline:
        prog = (time.time() - start_time) / (phase_a_deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** prog
        for i in range(min(top_k, len(candidates))):
            if time.time() >= phase_a_deadline:
                break
            a_start = candidates[i][1]
            best_a, best_val = _lbfgs_optimize(
                a_start, n_small, deadline, start_time,
                best_a, best_val, eps=eps, max_iter=800)
        best_a, best_val = _blend_with_uniform(best_a, n_small, best_val)

    # --------------------------------------------------------------
    # 3️⃣  Exact sub‑gradient polishing on the small problem
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n_small,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=200_000)

    # --------------------------------------------------------------
    # 4️⃣  Upsample to the target size (n=12000)
    # --------------------------------------------------------------
    n_large = 12000
    xs_small = np.linspace(0.0, 1.0, n_small)
    xs_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(xs_large, xs_small, best_a)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    best_a, best_val = a_large, _objective(a_large)
    print(f"[Upsample] value = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  LP‑based active‑set refinement (large scale)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _lp_refine(
        best_a, deadline=phase_c_deadline, start_time=start_time,
        max_iters=30, blend_t=0.2)

    # --------------------------------------------------------------
    # 6️⃣  Final smooth‑max L‑BFGS (tiny ε) and sub‑gradient polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(
            best_a, n_large, deadline, start_time,
            best_a, best_val, eps=1e-12, max_iter=800)

    if time.time() < deadline:
        best_a, best_val = _subgradient_polish(
            best_a, n_large,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            max_iters=100_000)

    # --------------------------------------------------------------
    # 7️⃣  Light simulated‑annealing perturbation to escape tiny plateaus
    # --------------------------------------------------------------
    temp_start, temp_end = 0.08, 0.001
    while time.time() < deadline:
        now = time.time()
        progress = (now - start_time) / (deadline - start_time)
        temp = temp_start * (temp_end / temp_start) ** progress
        i = rng.integers(n_large)
        j = rng.integers(n_large)
        if i == j:
            continue
        delta = temp * best_a[i]                     # move a small fraction
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑project onto the simplex (cheap)
        cand = _project_simplex(cand)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Anneal] new best = {best_val:.6f}")

    # final blend with uniform (helps with occasional spikes)
    best_a, best_val = _blend_with_uniform(best_a, n_large, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
