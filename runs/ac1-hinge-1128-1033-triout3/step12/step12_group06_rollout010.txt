"""
Propose a non‑negative step function that (hopefully) beats the 1.503 bound.

Idea
----
The evaluation function is

    2·n·max(conv(a,a)) / (sum(a)^2)

and it is scale‑invariant, so we may assume sum(a)=1.  For a sparse
vector consisting of *m* equal spikes (a[pos]=1/m) the convolution at any
lag contains at most one product 1/m², because each absolute difference
between two spike positions appears only once.  Consequently

    max(conv) = 1/m²  →  objective = 2·n / m².

If we can find *m* positions with pairwise distinct absolute differences
(a Sidon / Golomb‑ruler set) we obtain the bound above.
The theoretical limit is m ≤ √(2·n) (≈126 for n=8000).  Achieving
≈104 spikes gives an objective below 1.5.

The implementation therefore

1. Searches for a large Sidon set by random greedy construction,
   keeping the best one found within the time budget.
2. Builds the step function with equal weight 1/m on those positions.
3. If the obtained size is insufficient, falls back to the monotone
   power‑law shape used previously.

The whole search fits easily into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast convolution (FFT) and the cheap surrogate objective
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a sequence with sum(a)=1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Parametric families – kept as fall‑back
# ----------------------------------------------------------------------


def _monotone_power_shape(p: float, n: int) -> np.ndarray | None:
    """a_i ∝ (i+1)^p, normalised to sum=1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Sidon set construction (greedy random search)
# ----------------------------------------------------------------------


def _random_sidons(n: int, rng: np.random.Generator) -> list[int]:
    """
    Return a set of positions 0…n‑1 whose pairwise absolute differences
    are all distinct (a Sidon/Golomb‑ruler set).  The greedy algorithm
    processes a random permutation of the indices and keeps a point if it
    does not create a duplicate difference.
    """
    order = rng.permutation(n)
    positions: list[int] = []
    diffs: set[int] = set()          # stores positive differences only

    for cand in order:
        # compute the new differences
        conflict = False
        for p in positions:
            d = cand - p if cand > p else p - cand
            if d in diffs:
                conflict = True
                break
        if conflict:
            continue
        # accept the candidate
        for p in positions:
            diffs.add(cand - p if cand > p else p - cand)
        positions.append(int(cand))
    return positions


# ----------------------------------------------------------------------
# Main entry point required by the outer framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The search consists of:
      * greedy random construction of a large Sidon set,
      * building an equal‑weight spike vector on that set,
      * (optionally) falling back to a monotone power‑law shape.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Problem size – 8000 works well for the current budget.
    # --------------------------------------------------------------
    n = 8000

    # Target size of a Sidon set that guarantees an objective < 1.503
    #   2·n / m² < 1.503  →  m > sqrt(2·n / 1.503)
    target_m = int(np.ceil(np.sqrt(2.0 * n / 1.503)))   # ≈104 for n=8000

    best_positions: list[int] = []
    best_m = 0
    best_seq: np.ndarray | None = None
    best_val = np.inf

    attempts = 0
    # --------------------------------------------------------------
    # 1️⃣  Random greedy Sidon construction – keep the longest set.
    # --------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1
        pos = _random_sidons(n, rng)
        m = len(pos)

        if m > best_m:
            best_m = m
            best_positions = pos
            # build the corresponding spike vector
            seq = np.zeros(n, dtype=np.float64)
            if m > 0:
                seq[pos] = 1.0 / m
            val = _objective(seq)
            best_seq = seq
            best_val = val
            print(f"[search] attempt {attempts:04d} new best size {best_m} "
                  f"objective={best_val:.6f}")

        # stop early if we already surpassed the required size
        if best_m >= target_m:
            break

        # occasionally give the loop a short breather
        if attempts % 100 == 0 and time.time() > deadline - 30:
            break

    # --------------------------------------------------------------
    # 2️⃣  If the Sidon set is still too small, fall back to the
    #      monotone power‑law family (the previous state‑of‑the‑art).
    # --------------------------------------------------------------
    if best_m < target_m:
        print("[fallback] Sidon set did not reach target size, "
              f"using monotone shape (best size {best_m})")
        # try a fine grid of exponents (same idea as earlier code)
        exponents = np.arange(0.5, 10.0, 0.5)
        for exp in exponents:
            cand = _monotone_power_shape(exp, n)
            if cand is None:
                continue
            val = _objective(cand)
            if val < best_val:
                best_val = val
                best_seq = cand
                print(f"[fallback] new best via power p={exp:.2f} "
                      f"objective={best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final safety: make sure we have a valid sequence.
    # --------------------------------------------------------------
    if best_seq is None:
        # uniform sequence as last resort (objective exactly 2)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)

    # --------------------------------------------------------------
    # 4️⃣  Log the final surrogate value (for debugging) and return.
    # --------------------------------------------------------------
    print(f"[final] best sequence length {n}, "
          f"size of Sidon set {best_m}, surrogate objective {best_val:.6f}")

    # The outer harness will invoke evaluate_sequence on the returned list.
    return best_seq.tolist()
