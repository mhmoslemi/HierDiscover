"""
Propose a step‑function with a smaller value of
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))²

Algorithm
---------

1. **Multiple independent restarts** – the total budget (≈1000 s) is split
   into short optimisation runs (≈30 s each).  Each run uses a different
   random seed, a different initial shape (Beta, Dirichlet, sparse,
   possibly the previous best) and different optimiser hyper‑parameters.
   The best result of all runs is kept.

2. **Stage 1 – smoothed Adam in the log‑domain** – we optimise the
   soft‑max approximation of the max‑convolution entry.
   The temperature α grows from 1 to a large value (200‑600) so that the
   surrogate becomes close to the true max.  Adam is applied to the
   logits `z = log(a)`; after each step the probabilities are re‑normalised.

3. **Stage 2 – entropy mirror descent on the exact max‑entry** – once the
   temperature exceeds a fraction of its final value we switch to a
   sub‑gradient step that directly minimises the maximal convolution entry.
   The update is
        a ← a · exp(‑η · g) ,  a ← a / sum(a)
   with a linearly decaying step‑size η.

4. **Escaping stagnation** – if no improvement is seen for a while a
   small Gaussian jitter is added (different magnitude in the two stages).

5. **Polishing** – after every restart a longer mirror‑descent run
   (10 000 steps) is applied to the personal best of that run.
   After all restarts a final polishing (another 10 000 steps) is run
   on the globally best vector.

All operations are performed with FFT‑based convolutions (linear, not circular)
which makes each iteration O(n log n).  The code respects the overall time‑budget
and returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two different vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max entry (polishing)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 5000, eta_start: float = 0.5) -> np.ndarray:
    """Entropy‑mirror descent on the max entry of a*a."""
    n = a.shape[0]
    a = a.copy()
    eps = 1e-30
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]                 # sub‑gradient
        stage = t / steps
        eta = eta_start * (1.0 - stage) + 0.01 * stage
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# One optimisation run (a “restart”) – returns refined vector and its
#          (approximate) objective 2·n·max(conv) (sum = 1)
# ----------------------------------------------------------------------
def _run_one_restart(
    n: int,
    seed: int,
    init_type: str,
    init_params: dict,
    hyper: dict,
    budget_s: float,
) -> tuple[np.ndarray, float]:
    """Run a single Adam → mirror‑descent optimisation for ~budget_s seconds."""
    np.random.seed(seed)

    # ----- initialise a probability vector ---------------------------------
    if init_type == "beta":
        p = init_params.get("p", 2.0)
        q = init_params.get("q", 2.0)
        i = np.arange(1, n + 1, dtype=np.float64)
        a0 = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    elif init_type == "dirichlet":
        conc = init_params.get("conc", 1.0)
        a0 = np.random.gamma(shape=conc, scale=1.0, size=n)
    elif init_type == "sparse":
        frac = init_params.get("frac", 0.05)
        m = max(1, int(frac * n))

        idx = np.random.choice(n, size=m, replace=False)

    elif init_type == "previous":
        old = np.asarray(globals().get("height_sequence_1", []), dtype=np.float64)
        if old.shape[0] != n:
            # linear interpolation onto the new grid
            x_new = np.linspace(0.0, 1.0, n)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            a0 = np.interp(x_new, x_old, old)
        else:
            a0 = old.copy()
    else:

    a0 = _project_simplex(a0)
    eps = 1e-30
    z = np.log(a0 + eps)                           # logits for Adam

    # ----- Adam bookkeeping -------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m_buf = np.zeros_like(z)
    v_buf = np.zeros_like(z)

    # ----- hyper‑parameters -------------------------------------------------
    lr_start = hyper["lr_start"]
    lr_end = hyper["lr_end"]
    max_alpha = hyper["max_alpha"]
    switch_alpha = hyper["switch_alpha"]
    stagn_limit = hyper["stagn_limit"]
    jitter_adam = hyper["jitter_adam"]
    jitter_mirror = hyper["jitter_mirror"]

    # ----- time management --------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.1          # small safety margin

    best_a = None
    best_obj = np.inf
    stagn_counter = 0
    it = 0
    mirror_mode = False
    switch_prog = None

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start) / (deadline - start)       # 0 → 1 inside this run
        alpha = 1.0 + (max_alpha - 1.0) * (prog ** 1.5)  # temperature growth

        # -------- current probability vector --------------------------------
        if not mirror_mode:
            a = np.exp(z)
            a = _project_simplex(a)

        # -------- convolution & objective -----------------------------------
        conv = _full_convolution(a)
        maxc = conv.max()
        obj = 2.0 * n * maxc                         # sum(a)=1 → denominator =1

        # -------- bookkeeping ------------------------------------------------
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagn_counter = 0
        else:
            stagn_counter += 1

        # -------- transition to mirror descent -------------------------------
        if not mirror_mode and alpha >= switch_alpha:
            mirror_mode = True
            switch_prog = prog
            # keep a consistent with logits
            a = np.exp(z)
            a = _project_simplex(a)

        # -------- gradient / update -----------------------------------------
        if not mirror_mode:
            # --- Stage 1: Adam on a soft‑max surrogate of max(conv) ---------
            maxc = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc))
            weights = exp_scaled / exp_scaled.sum()          # Σweights = 1

            # gradient w.r.t. a (derivative of soft‑maxed max)
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            g_a = 2.0 * conv_rev[n - 1 : n - 1 + n]          # length n

            # chain rule through soft‑max(z)
            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            # Adam step on logits
            it += 1
            m_buf = beta1 * m_buf + (1.0 - beta1) * grad_z
            v_buf = beta2 * v_buf + (1.0 - beta2) * (grad_z ** 2)

            v_hat = v_buf / (1.0 - beta2 ** it)

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        else:
            # --- Stage 2: exact entropy‑mirror descent on the max entry ------
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                g[idx] = 2.0 * a[k - idx]                     # sub‑gradient

            # linear decay of η inside the mirror phase
            stage_prog = (prog - switch_prog) / (1.0 - switch_prog) if switch_prog else prog
            eta = 0.5 * (1.0 - stage_prog) + 0.01 * stage_prog

                print(f"[restart] Adam jitter (seed={seed}) at iter={it}")
            else:

                a = _project_simplex(np.maximum(a + perturb, 0.0))
                z = np.log(a + eps)
                print(f"[restart] Mirror jitter (seed={seed}) at iter={it}")

    if best_a is not None:

        conv_ref = _full_convolution(refined)

            refined = refined
        return refined, best_obj
    # fallback (should never happen)
    return np.full(n, 1.0 / n), 2.0 * n * n * (1.0 / n ** 2)

        # very unlikely fallback – uniform distribution (objective = 2)
        global_best = np.full(n, 1.0 / n)

    print(f"[result] returning O≈{global_best_obj:.6f} with length {len(global_best)}")
    return global_best.tolist()
