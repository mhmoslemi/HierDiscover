"""
Propose an improved step‑function for ``evaluate_sequence``.

The algorithm consists of three phases:
1.  A systematic search over the monotone power‑law family
    a_i ∝ (i+1)^p.  First a coarse grid (step 0.25) is scanned,
    then a 1‑dimensional golden‑section refinement is performed
    around the best exponent.  The cost of each evaluation is
    cheap (FFT‑based convolution) so thousands of candidates
    can be tried well within the 1000 s budget.
2.  A short stochastic “mass‑transfer’’ local search that moves a
    tiny amount of probability from one index to another while
    keeping the vector on the simplex.  This can push the maximum
    convolution down by a few per‑mille.
3.  A final mirror‑descent (exponentiated‑gradient) optimisation,
    using a sub‑gradient of the max‑convolution, to polish the
    solution.

All intermediate candidates are kept on the probability simplex,
so the objective 2·n·max(conv)/sum² is invariant to scaling.
The routine respects the time budget and prints progress,
finally returning the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (size = next power of two)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming Σa = 1.
    The sequence must be non‑negative.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Sub‑gradient for the max‑convolution (used in mirror‑descent)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 5,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Sub‑gradient of max(b) w.r.t. a.
    Returns a vector of the same length as ``a``.
    """
    n = a.shape[0]
    max_val = b.max()
    # choose a few active indices (the largest entries of b)
    if max_active >= b.size:
        active = np.arange(b.size)
    else:
        active = np.argpartition(b, -max_active)[-max_active:]
        mask = b[active] >= max_val * (1.0 - rel_tol)
        active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def _power_shape(p: float, n: int) -> np.ndarray:
    """
    a_i ∝ (i+1)^p   (i = 0 … n‑1),   Σa = 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)  # 1 … n
    a = np.power(i, p)
    total = a.sum()
    if total == 0.0 or not np.isfinite(total):
        # safety fallback: uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    a /= total
    return a


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Original Beta family a_i ∝ (i+1)^{p‑1}·(n‑i)^{q‑1}, Σa = 1.
    Returns None if the parameters produce an invalid vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimizing ``evaluate_sequence``.
    Returns a plain Python list.  The search respects the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good balance.
    # --------------------------------------------------------------
    n = 8000

    best_seq: np.ndarray | None = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Helper to possibly update the global best
    # ------------------------------------------------------------------
    def _maybe_update(candidate: np.ndarray, source: str = ""):
        nonlocal best_seq, best_val
        val = _objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[update] {source:6s} val={best_val:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previously found solution if available
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old.size > 1:
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            _maybe_update(a0, "old")
            print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse grid over the power‑law exponent
    # ------------------------------------------------------------------
    print("[phase] coarse grid over exponent p")
    for p in np.arange(0.5, 12.5, 0.25):
        cand = _power_shape(p, n)
        _maybe_update(cand, f"grid{p:.2f}")
        if time.time() > deadline - 120:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Golden‑section refinement around the best exponent
    # ------------------------------------------------------------------
    if best_seq is not None:
        # recover the exponent that produced best_seq (brute‑force lookup)
        # – during the grid search we stored the exponent in the source string,
        #   but for safety we now perform a quick re‑search.
        # The rough optimum lies somewhere between 3 and 9.
        lo, hi = 2.0, 9.0
        # narrow the interval around previously best grid point if we can
        # (the best grid exponent is embedded in `best_val` updates)
        # We'll just keep a reasonable window.
    else:
        lo, hi = 0.5, 12.0

    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate

    def _eval_power(p_val: float) -> float:
        """Return objective for power‑law shape with exponent p_val."""
        cand = _power_shape(p_val, n)
        return _objective(cand)

    # golden‑section loop (stop once interval width is ~1e‑4 or time runs out)
    while hi - lo > 1e-4 and time.time() < deadline - 80:
        x1 = hi - phi * (hi - lo)
        x2 = lo + phi * (hi - lo)
        f1 = _eval_power(x1)
        f2 = _eval_power(x2)

        if f1 < f2:
            hi = x2
            if f1 < best_val:
                _maybe_update(_power_shape(x1, n), f"gold{x1:.4f}")
        else:
            lo = x1
            if f2 < best_val:
                _maybe_update(_power_shape(x2, n), f"gold{x2:.4f}")

    # ------------------------------------------------------------------
    # 3️⃣  Random Beta‑family exploration (adds a second shape degree of freedom)
    # ------------------------------------------------------------------
    print("[phase] random Beta family exploration")
    rng = np.random.default_rng()
    for _ in range(1500):
        p = rng.uniform(0.5, 10.0)
        q = rng.uniform(0.5, 10.0)
        cand = _beta_shape(p, q, n)
        if cand is None:
            continue
        _maybe_update(cand, f"beta{p:.2f}{q:.2f}")
        if time.time() > deadline - 70:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Simple mass‑transfer hill‑climbing (stochastic local search)
    # ------------------------------------------------------------------
    print("[phase] stochastic mass‑transfer hill‑climbing")
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    a = best_seq.copy()
    while time.time() < deadline - 60:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = 1e-5  # transfer magnitude as a fraction of total mass
        delta = eps * a[i]  # move a tiny amount from i to j
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta
        # keep non‑negativity (a[i] might become negative due to rounding)
        if a[i] < 0:
            a[i] = 0.0
            a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[hill] new best {best_val:.6f}")
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta

    # ------------------------------------------------------------------
    # 5️⃣  Mirror‑descent polishing (exponentiated gradient)
    # ------------------------------------------------------------------
    print("[phase] mirror‑descent polishing")
    a = best_seq.copy()
    m = np.zeros_like(a)
    beta_mom = 0.9
    it = 0

    while time.time() < deadline - 1.0:
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[optim] iter {it:4d} new best {best_val:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_max(a, conv, max_active=5, rel_tol=1e-6)

        # momentum on the gradient
        m = beta_mom * m + (1.0 - beta_mom) * grad

        # step‑size decays linearly with the remaining time
        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)
        step = 0.2 * (1.0 - progress) + 1e-6

        # multiplicative (exponentiated‑gradient) update – stays on simplex
        log_factor = -step * m
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # occasional jitter to escape shallow plateaus
        if np.random.rand() < 0.001:
            jitter = np.random.normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            m.fill(0.0)

        it += 1

    # ------------------------------------------------------------------
    # 6️⃣  Final secure evaluation (just for information)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return a plain Python list as required by the outer framework
    return best_seq.tolist()
