"""
Hybrid optimizer with a linear‐programming direction step.

The algorithm works on a probability vector `a` (non‑negative, sum = 1) of
length `n`.  At each outer iteration it:

1. Computes the self‑convolution `c = a*a` and the current objective
   `obj = 2*n*max(c)`.
2. If the objective is already below the target it stops.
3. Takes the `K` largest convolution entries (the “active set”) and
   solves a tiny linear program:
       maximise   Σ_j b_j
       subject to b_j ≥ 0
                  (a * b)_k ≤ max(c)   for every k in the active set
   where `(a*b)_k` is linear in `b`.  The LP gives a direction `b`
   that tries to spread the mass without increasing the current peak
   value.
4. Mixes the current `a` with the LP solution (`a ← (1‑t)*a + t*b`) and
   runs a short Adam optimisation on a smooth‑max surrogate of the
   `max` (temperature λ grows from a modest start to a very large
   value).  This refines the solution where the LP is only a first‑order
   guide.
5. Finishes the outer iteration with a few projected sub‑gradient steps
   on the exact max (the true sub‑gradient is cheap to compute).

The loop repeats until the 1000 s budget expires (or the target is met).
All intermediate vectors are projected onto the probability simplex to
respect the `evaluate_sequence` constraints.  The best vector found is
returned at the end.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures or lambdas
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a real vector with itself.
    Length = 2*len(x)-1, computed via FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of weight vector w with a.
    Returned vector has length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """Objective 2·n·max(self‑convolution) for a probability vector."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


def _smooth_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max (temperature λ) weights for the smooth‑max of conv."""
    cmax = conv.max()
    if lam > 2000.0:                         # switch to exact max for speed
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    # stable soft‑max around the maximum
    exp_vals = np.exp(lam * (conv - cmax))
    w = exp_vals / exp_vals.sum()
    return w


def _build_A_ub(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A for the LP constraints (a * b)_k ≤ M,
    one row per index k in `ks`.  A[k,i] = a_{k‑i} when the index is
    valid, otherwise 0.
    """
    n = a.shape[0]
    K = ks.size
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(ks):
        # valid i satisfy 0 ≤ k‑i < n  →  max(0, k‑(n‑1)) ≤ i ≤ min(k, n‑1)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        A[row, i_range] = a[k - i_range]
    return A

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise `evaluate_sequence`.  The routine mixes a linear‑programming
    direction step with Adam + sub‑gradient refinement under a strict
    1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny margin for the final secure evaluation
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 8000 values work well for the given budget.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise from a previous construction if available.
    # --------------------------------------------------------------
    try:
        old_seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old_seq.size > 1:
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_seq.size)
            a = np.interp(new_x, old_x, old_seq)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum < 1e-12:
                a = np.full(n, 1.0 / n)
            else:
                a = a / a_sum
            print("[init] interpolated previous solution")
        else:
            raise Exception
    except Exception:
        a = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    # --------------------------------------------------------------
    # Global best tracker
    # --------------------------------------------------------------
    best_a = a.copy()
    best_val = _objective(best_a, n)
    print(f"[global] start objective = {best_val:.9f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    target_obj = 1.503                     # stop when we beat this
    K_active = 300                         # size of the active set for LP
    t_mix = 0.25                           # mixing factor (a ← (1‑t)*a + t*b)
    adam_lr = 0.04
    b1, b2 = 0.9, 0.999
    adam_eps = 1e-8
    max_adam_steps = 5000
    max_subgrad_steps = 2000

    # ------------------------------------------------------------------
    # Helper for a quick check whether we are out of time.
    # ------------------------------------------------------------------
    def _out_of_time() -> bool:
        return time.time() >= deadline

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while not _out_of_time():
        # ----------------------------------------------------------
        # 0️⃣  Compute current convolution and objective
        # ----------------------------------------------------------
        conv = _full_convolution(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_val - 1e-12:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[global] new best = {best_val:.9f} (t={time.time() - start_time:.1f}s)")

        # early exit if target met
        if best_val < target_obj:
            print("[early] target achieved")
            break

        # ----------------------------------------------------------
        # 1️⃣  LP direction step based on the active set of constraints
        # ----------------------------------------------------------
        # pick the K largest convolution entries (indices in the full conv array)
        K = min(K_active, conv.size)
        top_inds = np.argpartition(conv, -K)[-K:]          # unsorted
        # Build the LP matrices
        A_ub = _build_A_ub(a, top_inds)
        b_ub = np.full(A_ub.shape[0], conv.max())          # same bound M for all
        c_obj = -np.ones(n)                                 # maximise Σ b_j

        try:
            lp_res = linprog(c=c_obj,
                             A_ub=A_ub,
                             b_ub=b_ub,
                             bounds=[(0, None)] * n,
                             method='highs',
                             options={'presolve': True, 'time_limit': max(deadline - time.time(), 0.0)})
            if lp_res.success and lp_res.x is not None:
                b = lp_res.x
                b_sum = b.sum()
                if b_sum > 0:
                    b = b / b_sum                     # renormalise to a probability vector
                    # Convex mix of the current a and the LP direction
                    a = (1.0 - t_mix) * a + t_mix * b
                    # (mixing retains sum=1 automatically)
            else:
                # LP failed – fall back to a tiny random perturbation
                a += np.random.normal(scale=1e-4, size=n)
                a = _project_simplex(a)
        except Exception as e:  # pragma: no cover – safety net
            print("[lp] exception:", e)
            a += np.random.normal(scale=1e-4, size=n)
            a = _project_simplex(a)

        # ----------------------------------------------------------
        # 2️⃣  Short Adam optimisation on a smooth‑max surrogate
        # ----------------------------------------------------------
        # Work in log‑space (mirrors a soft‑max parameterisation)
        eps = 1e-12
        z = np.log(a + eps) + np.random.normal(scale=0.01, size=n)   # small jitter
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        for step in range(1, max_adam_steps + 1):
            if _out_of_time():
                break
            # probability vector from logits
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            a = exp_z / exp_z.sum()

            # convolution and smooth‑max weights
            conv = _full_convolution(a)
            prog = step / max_adam_steps
            lam = 5.0 * (1.0 - prog) + 8000.0 * prog
            w = _smooth_weights(conv, lam)

            # gradient of smooth max w.r.t. a (2 * cross‑correlation)
            g_raw = _cross_corr(w, a)
            g = 2.0 * g_raw

            # back‑prop through the soft‑max (mirror descent)
            dot_ag = np.dot(a, g)
            grad_z = a * (g - dot_ag)

            # Adam update
            m = b1 * m + (1.0 - b1) * grad_z
            v = b2 * v + (1.0 - b2) * (grad_z ** 2)
            m_hat = m / (1.0 - b1 ** step)
            v_hat = v / (1.0 - b2 ** step)
            z -= adam_lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # occasional noise to escape plateaus
            if step % 1500 == 0:
                z += np.random.normal(scale=0.02, size=n)

        # ----------------------------------------------------------
        # 3️⃣  Projected sub‑gradient sweep on the exact max
        # ----------------------------------------------------------
        step_size = 0.03
        a_proj = a.copy()
        for _ in range(max_subgrad_steps):
            if _out_of_time():
                break
            conv = _full_convolution(a_proj)
            kmax = int(np.argmax(conv))

            # exact sub‑gradient of max (vector of length n)
            g = np.zeros_like(a_proj)
            if kmax < n:
                # overlapping range 0 … kmax
                g[:kmax + 1] = 2.0 * a_proj[kmax::-1]
            else:
                offset = kmax - (n - 1)
                L = n - offset
                g[offset:] = 2.0 * a_proj[:L][::-1]

            a_trial = _project_simplex(a_proj - step_size * g)
            obj_trial = 2.0 * n * _full_convolution(a_trial).max()
            if obj_trial < cur_obj - 1e-12:
                a_proj = a_trial
                cur_obj = obj_trial
                step_size = min(step_size * 1.05, 0.5)
                if cur_obj < best_val - 1e-12:
                    best_val = cur_obj
                    best_a = a_proj.copy()
                    print(f"[subgrad] improved to {best_val:.9f}")
            else:
                step_size *= 0.7

        a = a_proj  # carry the refined vector forward

        # ----------------------------------------------------------
        # 4️⃣  Safety check – break if we are very close to the target
        # ----------------------------------------------------------
        if best_val < target_obj:
            print("[early] reached target after refinement")
            break

    # ------------------------------------------------------------------
    # Final secure evaluation (the only place we call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
