def compute_obj_and_grad(y, n, alpha, size_fft):
    a = softmax(y)
    # Compute convolution a*a via FFT
    fft_a = np.fft.rfft(a, n=size_fft)
    conv_full = np.fft.irfft(fft_a * fft_a, n=size_fft)
    conv = conv_full[: 2*n - 1]
    # Smooth max via log-sum-exp
    max_val = conv.max()
    diff = conv - max_val
    exp_terms = np.exp(alpha * diff)
    sum_exp = exp_terms.sum()
    smooth_max = max_val + np.log(sum_exp) / alpha
    f = 2.0 * n * smooth_max
    # weights for gradient
    w = exp_terms / sum_exp  # shape 2n-1
    # compute gradient w.r.t a
    w_rev = w[::-1]
    fft_w_rev = np.fft.rfft(w_rev, n=size_fft)
    corr_full = np.fft.irfft(fft_a * fft_w_rev, n=size_fft)
    start = 2*n - 2
    grad_a = 4.0 * n * corr_full[start : start + n]
    # chain rule through softmax
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)
    return f, grad_y
```

"""
Propose a step‐function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------
1.  Choose a moderate length `n` (8000).  The objective depends only on the
    shape of the distribution, not on its absolute scale, so we restrict
    all candidates to the probability simplex (sum = 1).

2.  Optimisation is performed in the *log‑probability* space.
    For a vector `y` we set `a = softmax(y)` which guarantees
    `a_i ≥ 0` and `∑ a_i = 1`.

3.  A smooth approximation of the original objective is used:
        f(y) = 2·n·smooth_max(a*a) ,   smooth_max(z) = (1/α)·log∑exp(α·z)
    with a smoothing parameter `α`.  The gradient of `f` can be written
    analytically using a single FFT for the convolution and a second FFT
    for the cross‑correlation needed in the chain rule.

4.  The smooth problem is solved with a few L‑BFGS‑B restarts (SciPy).
    Early restarts use a modest `α` (smooth landscape) and later ones
    increase `α` to focus on the true maximum.

5.  The best candidate from the L‑BFGS phase is refined with a short
    mirror‑descent / exponentiated‑gradient loop (the same update as the
    previous baseline) to exploit any remaining improvement.

6.  The final sequence is evaluated with the provided
    ``evaluate_sequence`` and returned.

The code respects the given time budget, prints progress, and works
without any file or network access.
"""

import time
import math
import numpy as np
from scipy import optimize as opt

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    if x <= 0:
        return 1
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Full linear convolution a * a using a pre‑allocated FFT size.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    # FFT of a (zero‑padded to size_fft)
    fft_a = np.fft.rfft(a, n=size_fft)
    conv_full = np.fft.irfft(fft_a * fft_a, n=size_fft)
    # keep only the linear part
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------
# Softmax (log‑probability → probability simplex)
# ----------------------------------------------------------------------


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth objective and its gradient (returns (f, grad_y))
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(
    y: np.ndarray, n: int, alpha: float, size_fft: int
) -> tuple[float, np.ndarray]:
    """
    Compute a smooth surrogate of the official objective and its gradient
    w.r.t. the log‑probabilities `y`.

    Parameters
    ----------
    y : (n,) array
        Log‑probabilities.
    n : int
        Length of the sequence.
    alpha : float
        Smoothing parameter (larger → closer to true max).
    size_fft : int
        Length of the zero‑padded FFT (≥ 3·n‑2).

    Returns
    -------
    f : float
        Smooth objective value.
    grad_y : (n,) array
        Gradient w.r.t. `y`.
    """
    # probabilities on the simplex
    a = _softmax(y)

    # ---- 1) convolution a*a ------------------------------------------------
    fft_a = np.fft.rfft(a, n=size_fft)
    conv_full = np.fft.irfft(fft_a * fft_a, n=size_fft)
    conv = conv_full[: 2 * n - 1]                # length 2n‑1

    # ---- 2) smooth max via log‑sum‑exp --------------------------------------
    max_val = conv.max()
    diff = conv - max_val                         # ≤ 0
    exp_terms = np.exp(alpha * diff)              # stable (max term → 1)
    sum_exp = exp_terms.sum()
    smooth_max = max_val + np.log(sum_exp) / alpha
    f = 2.0 * n * smooth_max                      # smooth surrogate of evaluate_sequence

    # ---- 3) gradient of the smooth max ---------------------------------------
    # weights that sum to 1 (softmax‑like over the convolution entries)
    w = exp_terms / sum_exp                       # shape (2n‑1,)

    # cross‑correlation a ⋆ w  (gradient of Σ w_k·(a*a)_k)
    w_rev = w[::-1]
    fft_w_rev = np.fft.rfft(w_rev, n=size_fft)
    corr_full = np.fft.irfft(fft_a * fft_w_rev, n=size_fft)

    start = 2 * n - 2                              # index where the length‑n slice begins
    grad_a = 4.0 * n * corr_full[start : start + n]   # d f / d a (chain rule applied to 2·n·Σ w_k·(a*a)_k)

    # ---- 4) chain rule through softmax ---------------------------------------
    # ∂a_i/∂y_j = a_i (δ_ij – a_j)
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    return float(f), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient used by the optional mirror‑descent refinement
# ----------------------------------------------------------------------


def _subgradient_max(
    a: np.ndarray,
    conv: np.ndarray,
    max_active: int = 7,
    rel_tol: float = 1e-6,
) -> np.ndarray:
    """Return a sub‑gradient of max(conv) w.r.t. `a` (active‑set version)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_active >= conv.size:
        active = np.arange(conv.size)
    else:
        active = np.argpartition(conv, -max_active)[-max_active:]
        mask = conv[active] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            active = np.array([int(np.argmax(conv))])
        else:
            active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Cheap (exact) objective used during the search (Σa = 1)
# ----------------------------------------------------------------------


def _exact_objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(fft_a * fft_a, n=size)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine uses a smooth L‑BFGS optimisation in log‑probability space,
    several random restarts, and a short mirror‑descent polishing phase.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    start = time.time()
    deadline = start + budget_s - 0.5          # keep a safety margin for the final call

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 8000
    size_fft = _next_pow2(3 * n - 2)          # enough for both convolutions

    # ------------------------------------------------------------------
    # Keep the best candidate found so far
    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Helper to possibly replace the current best
    # ------------------------------------------------------------------
    def _maybe_update(candidate: np.ndarray):
        nonlocal best_seq, best_val
        val = _exact_objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[update] new best cheap objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previously saved solution (if available)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # Interpolate to the required resolution
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            old = np.interp(new_x, old_x, old)
        old = np.maximum(old, 0.0)
        s = old.sum()
        if s > 0.0 and np.isfinite(s):
            old = old / s
        else:
            old = np.full_like(old, 1.0 / n)
        _maybe_update(old)
        print("[init] loaded previous solution")
    except Exception:
        # No previous solution available
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random L‑BFGS restarts with smooth surrogate
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0:
        # Schedule smoothing parameter: early restarts are smoother
        if restart < 8:
            alpha = 30.0
        else:
            alpha = 150.0

        # Initialise log‑probabilities
        if best_seq is not None and rng.random() < 0.35:
            # Start near the current best (add small Gaussian noise)
            y0 = np.log(best_seq + 1e-12) + rng.normal(scale=0.2, size=n)
        else:
            # Pure random start
            y0 = rng.normal(loc=0.0, scale=1.0, size=n)

        # L‑BFGS‑B optimisation (smooth objective)
        try:
            res = opt.minimize(
                lambda y: _smooth_obj_and_grad(y, n, alpha, size_fft),
                x0=y0,
                jac=True,
                method="L-BFGS-B",
                options={"maxiter": 200, "ftol": 1e-12, "disp": False},
            )
        except Exception as exc:
            print(f"[lbfgs] failed on restart {restart}: {exc}")
            restart += 1
            continue

        a_opt = _softmax(res.x)                # back to the probability simplex
        _maybe_update(a_opt)

        restart += 1

    # ------------------------------------------------------------------
    # 2️⃣  Optional mirror‑descent polishing (same idea as the baseline)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _exact_objective(best_seq)

    a = best_seq.copy()
    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    no_improve = 0
    max_no_improve = 18000

    while time.time() < deadline - 0.5:
        conv = _full_convolution(a, size_fft)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[mirror] iter {it} new best cheap = {best_val:.6f}")
        else:
            no_improve += 1

        # Sub‑gradient from a few active lags
        grad = _subgradient_max(a, conv, max_active=7, rel_tol=1e-6)

        # Momentum + exponential (mirror) update
        momentum = beta * momentum + (1.0 - beta) * grad
        elapsed = time.time() - start
        prog = elapsed / (deadline - start)
        step = 0.4 * (1.0 - prog) + 5e-5
        log_factor = -step * momentum
        log_factor = np.clip(log_factor, -40.0, 40.0)
        a = a * np.exp(log_factor)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # Occasional jitter to escape plateaus
        if rng.random() < 0.001 or no_improve >= max_no_improve:
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            momentum.fill(0.0)
            no_improve = 0

        it += 1

    # ------------------------------------------------------------------
    # 3️⃣  Final official evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
