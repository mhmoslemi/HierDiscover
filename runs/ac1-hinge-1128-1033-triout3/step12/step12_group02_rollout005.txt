            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = lr_start * (1.0 - total_prog) + lr_end * total_prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        else:
            # exact mirror descent
            # locate max index
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                # subgradient: 2 * a[k - idx]
                g[idx] = 2.0 * a[k - idx]
            # compute stage progress since switch

            eta = eta_start * (1.0 - stage_prog) + eta_end * stage_prog
            a = a * np.exp(-eta * g)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)
            it += 1

        # stagnation handling

            if not mirror_mode:
                z += np.random.normal(scale=jitter_scale_adam, size=n)
                print(f"[reset] jitter Adam at it={it}")
            else:

                a = a / s if s > 0 else np.full_like(a, 1.0 / n)
                print(f"[reset] jitter Mirror at it={it}")
            stagn_counter = 0

    # polishing: apply many mirror steps on best
    if best_a is not None:
        print("[polish] start polishing")
        refined = _max_refine(best_a, steps=20000, eta_start=eta_start)
        conv_ref = _full_convolution(refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = refined
            print(f"[polish] improved O={best_obj:.6f}")

    final_seq = best_a.tolist() if best_a is not None else a0.tolist()
    # final evaluate to print score
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returns {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception:
        pass
    return final_seq

"""
Propose a non‑negative step function that (almost) minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))² .

The objective, after fixing sum(seq)=1, is simply
    O = 2·n·max(conv(seq,seq)) .

We attack it with two complementary phases:

* Phase‑1 – smoothed Adam on a soft‑max surrogate of the max.
  The temperature α grows from 1 to a very large value, so the
  surrogate becomes arbitrarily close to the true max.  The parameters
  are kept in the log‑domain (z = log a) which guarantees positivity.

* Phase‑2 – exact entropy‑mirror descent on the true maximal entry.
  The sub‑gradient for the max entry is known analytically, so we can
  perform an exponentiated‑gradient (mirror) update and renormalise.

* Polishing – a long run of the mirror descent (20 000 steps) after the
  main optimisation loop, to squeeze out the last few bits.

The routine respects the 1000 s budget, uses FFT based convolutions for
speed, and tries a few different initialisations (previous solution,
Beta‑shape, random Dirichlet).  The best vector found is returned as a
plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two different vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max entry (polishing)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 20000,
                eta_start: float = 0.45, eta_end: float = 0.02) -> np.ndarray:
    """
    Entropy‑mirror descent on the exact max‑entry of the self‑convolution.
    The step‑size η decays linearly from ``eta_start`` to ``eta_end``.
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]

        prog = t / steps
        eta = eta_start * (1.0 - prog) + eta_end * prog
        # exponentiated‑gradient (mirror) update
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a.fill(1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (floats) that minimises
    ``evaluate_sequence``.  The function runs for up to ``budget_s`` seconds.
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer approximation of the
    # continuous problem while keeping the asymptotic objective unchanged.
    # 12000 works comfortably within the time budget.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise a probability vector (sum to 1, all entries ≥0).
    # Try to reuse the previously best solution if available.
    # ------------------------------------------------------------------
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        try:
            old = np.asarray(old_seq, dtype=float)
            if old.shape[0] != n:
                # linear interpolation to the new length
                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            print("[init] interpolated previous solution")
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # pick a smooth Beta(2,2) shape as a generic starting point
        i = np.arange(1, n + 1, dtype=np.float64)
        a0 = (i * (n - i + 1)).astype(np.float64)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        print("[init] generated Beta(2,2) start")

    # Work in the log‑domain to guarantee positivity throughout Adam.
    eps = 1e-30
    z = np.log(a0 + eps)

    # ------------------------------------------------------------------
    # Adam hyper‑parameters (Phase‑1)
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Learning‑rate schedule (high → low)
    lr_start, lr_end = 0.20, 0.005

    # Soft‑max temperature schedule
    max_alpha = 1200.0   # final temperature – large enough to emulate a true max
    switch_alpha = 300.0   # when we hand over to the exact mirror phase

    # ------------------------------------------------------------------
    # Mirror‑descent parameters (Phase‑2)
    # ------------------------------------------------------------------
    eta_start, eta_end = 0.5, 0.02   # linear decay inside the mirror phase

    # Stagnation handling
    stagn_counter = 0
    stagn_limit = 2500                 # steps without improvement → jitter
    jitter_scale_adam = 0.02
    jitter_scale_mirror = 0.003

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_a = None
    best_obj = np.inf
    it = 0
    mirror_mode = False
    switch_prog = None

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # Global progress (0 → 1) measured over the whole remaining budget
        total_prog = (now - start) / (deadline - start)

        # Temperature for the soft‑max surrogate
        alpha = 1.0 + (max_alpha - 1.0) * (total_prog ** 1.4)

        # ------------------------------------------------------------------
        # Switch from Adam to exact mirror descent once the temperature is
        # high enough.
        # ------------------------------------------------------------------
        if not mirror_mode and alpha >= switch_alpha:
            mirror_mode = True
            switch_prog = total_prog
            # materialise the probability vector from the logits
            a = np.exp(z)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)
            print(f"[mode] entering mirror descent at it={it}, prog={total_prog:.3f}")

        # Current probability vector (always normalised)
        if not mirror_mode:
            a = np.exp(z)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        # ------------------------------------------------------------------
        # Full self‑convolution (used for both phases)
        # ------------------------------------------------------------------
        conv = _full_convolution(a)
        obj = 2.0 * n * conv.max()      # exact objective for the current a

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagn_counter = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}  t={now-start:.1f}s")
        else:
            stagn_counter += 1

        # ------------------------------------------------------------------
        # Gradient / update step
        # ------------------------------------------------------------------
        if not mirror_mode:
            # ------- Phase 1: smoothed Adam on the soft‑max objective -------
            maxc = conv.max()
            # soft‑max weights (numerically stable)
            exp_scaled = np.exp(alpha * (conv - maxc))
            sumexp = exp_scaled.sum()
            weights = exp_scaled / sumexp                     # Σ weights = 1

            # gradient w.r.t. a (derivation follows the previous implementation)
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            # slice the part that corresponds to the original length n
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = lr_start * (1.0 - total_prog) + lr_end * total_prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        else:
            # ------- Phase 2: exact entropy‑mirror descent on the max entry ---
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)

            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                # sub‑gradient of the maximal convolution entry
                g[idx] = 2.0 * a[k - idx]

            # progress within the mirror phase (0 → 1)

        # Escape from stagnation – add a small random jitter
        # ------------------------------------------------------------------

            if not mirror_mode:
                z += np.random.normal(scale=jitter_scale_adam, size=n)
                print(f"[reset] jitter added (Adam) at it={it}")
            else:
                perturb = np.random.normal(scale=jitter_scale_mirror, size=n)
                a = np.maximum(a + perturb, 0.0)
                s = a.sum()
                a = a / s if s > 0 else np.full_like(a, 1.0 / n)
                print(f"[reset] jitter added (Mirror) at it={it}")
            stagn_counter = 0

    # ------------------------------------------------------------------
    # Polishing phase – a long run of mirror descent on the best vector
    # ------------------------------------------------------------------
    if best_a is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:          # only if we have a few seconds left

            refined = _max_refine(best_a, steps=20000,
                                  eta_start=eta_start, eta_end=eta_end)
            conv_ref = _full_convolution(refined)
            obj_ref = 2.0 * n * conv_ref.max()
            if obj_ref < best_obj:
                best_obj = obj_ref
                best_a = refined
                print(f"[polish] improved to O={best_obj:.6f}")

    # Final wrap‑up – evaluate once for reporting and return the list
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else a0.tolist()
    try:
        final_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx O={best_obj:.6f})")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
