def project_simplex(v):
    """Projects v onto probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _refine_sequence(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    size_fft = 1 << (3 * n - 2).bit_length()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lam = 1.0
    lam_max = 2000.0
    lam_growth = 1.001
    best_a = a.copy()
    best_val = _objective(a)
    iter_num = 0
    while time.time() < deadline:
        # compute FFT of a
        fft_a = np.fft.rfft(a, size_fft)
        # conv via fft_a**2
        conv = np.fft.irfft(fft_a * fft_a, size_fft)[: 2 * n - 1]
        cur_val = 2.0 * n * conv.max() / (a.sum()**2)
        # update best
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[refine] iter {iter_num} new best {best_val:.6f}")
        # compute softmax weights for gradient
        max_conv = conv.max()
        # shift for stability
        exp_term = np.exp(lam * (conv - max_conv))
        w = exp_term / np.sum(exp_term)
        # FFT of w
        fft_w = np.fft.rfft(w, size_fft)
        # correlation via FFT
        corr_full = np.fft.irfft(fft_a * np.conj(fft_w), size_fft)[: (n + (2 * n - 1) - 1)]
        start = (2 * n - 1) - 1
        grad = 2.0 * corr_full[start : start + n]
        # Adam update
        iter_num += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iter_num)
        v_hat = v / (1 - beta2 ** iter_num)
        # learning rate schedule: decaying
        # compute progress fraction
        # we have start time info accessible via closure not allowed - we can pass start_time param.
        # We'll compute step = 0.2 / (1 + iter_num * 1e-3) maybe.
        step = 0.5 / (1 + iter_num * 1e-3)  # decreasing
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex (simple clip+renorm)

Algorithm
---------

1. **Random‑support exploration** (≈30 % of the time budget) – many
   candidates are sampled, each with a random subset of indices and

   *smooth‑max* Adam optimiser.

     for a weight vector ``w`` (the softmax probabilities) the gradient

     cross‑correlation.
   * Adam steps are taken, the step size decays, and the distribution
     is projected back onto the probability simplex after each step.
   * The smoothing parameter λ is slowly increased (annealing) from
     1.0 up to 2000, gradually turning the smooth objective into the
     true max.
   * Small Gaussian jitter is added occasionally to escape plateaus.

3. The best sequence found is finally evaluated with the official
   ``evaluate_sequence`` function and returned.

The whole procedure respects the supplied time budget (default 1000 s)
and never exceeds it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Cheap objective used during the search.
    Returns 2·n·max(conv(a,a)) / (sum(a)**2).
    """
    n = a.shape[0]
    s = a.sum()
    if s <= 0.0:
        return np.inf
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max() / (s * s))


def _random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector of length n."""
    # support size between 20 % and 80 % of n
    frac = rng.uniform(0.2, 0.8)
    k = max(2, int(frac * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k) + 1e-6          # avoid exact zeros on the support
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Return the soft‑max probabilities of a convolution vector.
    ``lam`` controls the smoothness (larger → tighter to the true max).
    """
    # shift by the maximum for numerical stability
    shift = conv.max()
    exps = np.exp(lam * (conv - shift))
    return exps / np.sum(exps)


def _smooth_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     lam: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max) objective w.r.t. ``a``.
    Implements:
        w  = softmax_λ(conv)
        ∇ = 2·corr(a, w)
    where ``corr`` is the non‑reversed cross‑correlation.
    """
    n = a.shape[0]
    # soft‑max weights
    w = _softmax_weights(conv, lam)

    # FFT of a and w (zero‑padded to the same size)
    fa = np.fft.rfft(a, size_fft)
    fw = np.fft.rfft(w, size_fft)

    # correlation = irfft( fa * conj(fw) )
    corr_full = np.fft.irfft(fa * np.conj(fw), size_fft)
    # length of w is 2·n‑1
    start = (2 * n - 1) - 1
    grad = 2.0 * corr_full[start : start + n]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Simple projection onto the probability simplex:
    clip negatives to zero and renormalise.
    (This is not the Euclidean projection but inexpensive and works well
    together with the exponentiated‑gradient style updates.)
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    else:
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])


def _refine_sequence(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style refinement of a candidate distribution using a smooth‑max
    surrogate.  Returns the best sequence encountered and its cheap objective.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT size sufficient for a (n) + w (2n‑1) → length 3n‑2
    size_fft = _next_pow2(3 * n - 2)

    # Adam statistics
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # smoothing parameter schedule
    lam = 1.0
    lam_max = 2000.0
    lam_growth = 1.001

    # best‑so‑far
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- convolution of a with itself -----
        fa = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]

        # current cheap objective (exact max, not the smoothed one)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[refine] iter {it} new best {best_val:.6f}")

        # ----- smooth‑max gradient -----
        grad = _smooth_gradient(a, conv, lam, size_fft)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # linear decay of the (raw) step size
        progress = (time.time() - start_time) / (deadline - start_time + 1e-12)
        step = 0.5 * (1.0 - progress) + 5e-5

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # keep a on the simplex
        a = _project_onto_simplex(a)

        # ----- anneal λ -----
        lam = min(lam * lam_growth, lam_max)

        # occasional jitter to escape flat plateaux
        if np.random.rand() < 0.0005:
            jitter = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_onto_simplex(a)
            m.fill(0.0)
            v.fill(0.0)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the official
    ``evaluate_sequence`` score.

    The routine consists of a fast random‑support phase (≈30 % of the
    time budget) followed by a smooth‑max Adam refinement of the best
    candidate found.  The final sequence is returned as a Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # keep a safety margin

    # ------------------------------------------------------------------
    # problem size – 8000 works well for the trade‑off between expressivity
    # and speed.  Feel free to change it if you wish to explore other lengths.
    # ------------------------------------------------------------------
    n = 8000

    best_seq: np.ndarray | None = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previously saved solution, if present.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        if a0.sum() <= 0.0:
            a0 = np.full_like(a0, 1.0 / n)
        else:
            a0 = a0 / a0.sum()
        val0 = _objective(a0)
        best_seq = a0.copy()
        best_obj = val0
        print(f"[init] loaded previous solution, objective={best_obj:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random‑support exploration (≈30 % of the total time)
    # ------------------------------------------------------------------
    rs_deadline = start + 0.30 * budget_s
    while time.time() < rs_deadline:
        cand = _random_candidate(n, rng)
        val = _objective(cand)
        if val < best_obj:
            best_obj = val
            best_seq = cand.copy()
            print(f"[rand] new best objective={best_obj:.6f}")

    # Fallback – should never be needed, but keeps the function safe.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_obj = _objective(best_seq)
        print("[fallback] uniform start, objective=", best_obj)

    # ------------------------------------------------------------------
    # 2️⃣  Refine the current best candidate with smooth‑max Adam.
    # ------------------------------------------------------------------
    refine_deadline = deadline - 0.2   # tiny buffer for the final safe‑call
    refined_seq, refined_obj = _refine_sequence(best_seq, refine_deadline, start)
    if refined_obj < best_obj:
        best_seq = refined_seq
        best_obj = refined_obj
        print(f"[final‑refine] improved to {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final official evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
