"""
Hybrid optimiser for the self‑convolution max problem.

The routine keeps the best probability vector seen so far and performs
repeated Adam optimisations on a smooth‑max surrogate of the max‑convolution.
The temperature λ grows from a modest value to a very large one; for the
last part of the run we switch to the exact max (a one‑hot weight vector)
so the optimiser follows the true sub‑gradient.  After each Adam run a
long projected sub‑gradient refinement is executed which often brings the
objective a little further down.  The whole loop respects the 1000 s
budget and finally returns the best sequence found.

No external files are accessed; only NumPy is required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2*len(x)-1, computed via FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k-i} (cross‑correlation of w with a).
    Returned vector has length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1                         # extract indices n‑1 … n‑1+n‑1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """
    Compute the evaluation function for a probability vector a
    (i.e. Σ a = 1).  The returned value is exactly
        2 * n * max(conv(a, a))
    """
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()
