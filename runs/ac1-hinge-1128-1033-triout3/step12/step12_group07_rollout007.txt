def _beta_mixture_candidate(log_params):
    # log_params: array of length 9
    # NCOMP = 3
    p = np.exp(log_params[:NCOMP])
    q = np.exp(log_params[NCOMP:2*NCOMP])
    w_logits = log_params[2*NCOMP:3*NCOMP]
    # compute components
    # shape = sum_i w_i * comp_i
    weights = np.exp(w_logits - np.max(w_logits))
    weights = weights / np.sum(weights)
    a = np.zeros(N, dtype=np.float64)
    for i in range(NCOMP):
        comp_log = (p[i] - 1.0) * LOG_I + (q[i] - 1.0) * LOG_REV
        # subtract max for stability
        comp = np.exp(comp_log - np.max(comp_log))
        a += weights[i] * comp
    # normalize
    s = a.sum()
    if s == 0:
        a = np.full(N, 1.0/N)
    else:
        a /= s
    return a

"""
propose_candidate – hybrid global‑+‑local optimiser for the
step‑function attack.

The algorithm proceeds in three stages:

1. **Rich initial pool** – a quick sweep over several analytic families
   (Beta, Dirichlet, exponential, power‑law) to obtain a decent starter.

2. **Global optimisation of a 3‑component Beta mixture** – the search
   space is only 9 scalar parameters (log‑shape parameters and mixture
   weights).  We use SciPy’s differential‑evolution with an early‑stop
   callback that respects the global 1000 s wall‑clock budget.

3. **Heavy local polishing** – the best sequence found so far is refined
   by a cascade of optimiser phases (Adam on a smooth max, aggressive
   hill‑climbing, exact sub‑gradient descent, exponentiated‑gradient
   descent and a final simulated‑annealing run).  Each phase runs only
   until the remaining time budget expires.

All sequences are kept on the probability simplex (non‑negative,
sum‑to‑one) so the objective simplifies to

    F(a) = 2·n·max(conv(a,a))

which is exactly what ``evaluate_sequence`` returns (apart from the
sanity‑checks).  The final list of floats is returned; the host will
call ``evaluate_sequence`` to obtain the official score.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Global constants (problem size)
# ----------------------------------------------------------------------
N = 8000                     # length of the step function
LOG_I = np.log(np.arange(1, N + 1, dtype=np.float64))
LOG_REV = np.log(np.arange(N, 0, -1, dtype=np.float64))


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    b = _full_convolution(a)
    return float(2.0 * N * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """Smooth approximation of max(conv(a,a)) and its gradient w.r.t. a."""
    b = _full_convolution(a)                     # length 2N‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2N‑1,)

    # convolution of a with reversed p
    conv_ab = _conv_fft(a, p[::-1])              # length 2N‑1
    # central segment corresponds to the gradient contributions
    segment = conv_ab[N - 1:2 * N - 1]            # N entries
    grad_a = 4.0 * N * segment[::-1]             # factor 2·N already included
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Candidate families (used for the quick initial sweep)
# ----------------------------------------------------------------------
def _random_beta_candidate(rng: np.random.Generator) -> np.ndarray:
    i = np.arange(1, N + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p, q = np.exp(logp), np.exp(logq)
    a = (i ** (p - 1.0)) * ((N - i + 1) ** (q - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(N, 1.0 / N)


def _beta_grid_candidate(logp: float, logq: float) -> np.ndarray:
    i = np.arange(1, N + 1, dtype=np.float64)
    p, q = np.exp(logp), np.exp(logq)
    a = (i ** (p - 1.0)) * ((N - i + 1) ** (q - 1.0))
    s = a.sum()
    return a / s if s > 0 else np.full(N, 1.0 / N)


def _exponential_candidate(rng: np.random.Generator) -> np.ndarray:
    log_r = rng.uniform(-8.0, -0.1)
    r = np.exp(log_r)
    a = r ** np.arange(N, dtype=np.float64)
    a /= a.sum()
    return a


def _powerlaw_candidate(p: float) -> np.ndarray:
    i = np.arange(1, N + 1, dtype=np.float64)
    a = i ** p
    a /= a.sum()
    return a


def _mixture_beta_candidate(rng: np.random.Generator) -> np.ndarray:
    w = rng.random()
    a1 = _random_beta_candidate(rng)
    a2 = _random_beta_candidate(rng)
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 3‑component Beta mixture (global DE optimisation)
# ----------------------------------------------------------------------
NCOMP = 3  # number of Beta components in the mixture


def _beta_mixture_candidate(log_params: np.ndarray) -> np.ndarray:
    """
    Build a probability vector from a mixture of `NCOMP` Beta‑shaped
    components.  ``log_params`` holds:
        – first NCOMP entries : log(p_i)
        – next NCOMP entries  : log(q_i)
        – last NCOMP entries  : raw mixture logits (softmaxed)
    """
    p = np.exp(log_params[:NCOMP])
    q = np.exp(log_params[NCOMP:2 * NCOMP])
    w_raw = log_params[2 * NCOMP:3 * NCOMP]
    # softmax the raw mixture weights
    w = np.exp(w_raw - np.max(w_raw))
    w = w / w.sum()

    a = np.zeros(N, dtype=np.float64)
    for idx in range(NCOMP):
        comp_log = (p[idx] - 1.0) * LOG_I + (q[idx] - 1.0) * LOG_REV
        comp = np.exp(comp_log - np.max(comp_log))  # stabilise exponentiation
        a += w[idx] * comp

    s = a.sum()
    if s == 0.0:
        a = np.full(N, 1.0 / N)
    else:
        a /= s
    return a


def _de_objective(log_params: np.ndarray) -> float:
    """Objective for differential evolution (to be minimised)."""
    a = _beta_mixture_candidate(log_params)
    return _objective(a)


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    w = np.log(a0 + 1e-12)          # work in log‑softmax space
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.001
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 6000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        prog = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** prog
        eps = eps_start * (eps_end / eps_start) ** prog

        _, grad = _smooth_max_and_grad(a, eps)
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=N)

        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(N) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (source→target moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * N * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.9, 0.001
    temp_start, temp_end = 0.03, 1e-9
    no_improve = 0
    IMPROVE_LIMIT = 50000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(N)
        mask = (idx >= 0) & (idx < N)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(N, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(N, p=tgt_weights / wsum)
        else:
            j = np.random.randint(N - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(N))
        b_new = b.copy()
        b_new[j:j + N] += 2.0 * delta * a
        b_new[i:i + N] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * N * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation (it={iteration}), re‑perturb")
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=N)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * N * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(N)
            mask = (idx >= 0) & (idx < N)
            grad[mask] += 2.0 * a[idx[mask]]

        prog = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - prog) + 1e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=N), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------
def _expgrad_polish(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_val = _objective(a)

    eta_start, eta_end = 1.0, 1e-5

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(N)
            mask = (idx >= 0) & (idx < N)
            grad[mask] += 2.0 * a[idx[mask]]

        prog = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - prog) + eta_end * prog

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        new_val = _objective(a)

        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[ExpG] new best = {best_val:.6f}")

        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.018, size=N), 0.0)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray, cur_best_val: float) -> tuple[np.ndarray, float]:
    u = np.full(N, 1.0 / N)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Larger‑move simulated‑annealing (final phase)
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * N * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(N)
        if a[i] < 1e-12:
            continue
        j = rng.integers(N - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[j:j + N] += 2.0 * delta * a
        b_new[i:i + N] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * N * b_new.max()
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=N), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * N * b.max()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and (tries to) minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 0️⃣  Initialise (quick sweep over analytic families)
    # --------------------------------------------------------------
    best_a = np.full(N, 1.0 / N)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value = {best_val:.6f}")

    # try to reuse a previously discovered solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, N)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0, val
            print(f"[Init] interpolated previous solution, value = {best_val:.6f}")
    except Exception:
        pass

    # ---- random Beta draws (moderate amount) ----
    for _ in range(2000):
        cand = _random_beta_candidate(rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ---- systematic Beta grid (coarse) ----
    logp_vals = np.linspace(-7.5, 2.0, 12)
    logq_vals = np.linspace(-7.5, 2.0, 12)
    for lp in logp_vals:
        for lq in logq_vals:
            cand = _beta_grid_candidate(lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ---- Dirichlet draws (different concentrations) ----
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(15):
            cand = rng.dirichlet(np.full(N, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ---- Exponential family ----
    for _ in range(400):
        cand = _exponential_candidate(rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ---- Power‑law family ----
    for p in np.linspace(-4.0, 4.0, 20):
        cand = _powerlaw_candidate(p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ---- Mixture of two Betas ----
    for _ in range(800):
        cand = _mixture_beta_candidate(rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ---- Hand‑crafted trig shapes (already known to be useful) ----
    i = np.arange(N, dtype=np.float64)
    cand = np.sin(np.pi * (i + 0.5) / N)
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    cand = np.cos(np.pi * (i - (N - 1) / 2) / N) ** 2
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    print(f"[Init] best after quick sweep = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Global search: 3‑component Beta mixture via DE
    # --------------------------------------------------------------
    de_frac = 0.35
    de_deadline = start_time + de_frac * budget_s
    bounds = [(-8.0, 3.0)] * NCOMP + [(-8.0, 3.0)] * NCOMP + [(-5.0, 5.0)] * NCOMP

    def de_callback(xk, convergence):
        # Stop DE as soon as we run out of its dedicated budget
        return time.time() > de_deadline

    print("[DE] starting differential evolution...")
    result = differential_evolution(
        func=_de_objective,
        bounds=bounds,
        maxiter=200,               # large enough; callback will stop early
        popsize=30,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=False,
        updating='deferred',
        callback=de_callback,
    )
    a_de = _beta_mixture_candidate(result.x)
    val_de = _objective(a_de)
    if val_de < best_val:
        best_a, best_val = a_de, val_de
        print(f"[DE] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Remaining time is split among the local polishing phases
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # Allocate fractions reminiscent of the original pipeline
    frac_adam  = 0.20 * remaining
    frac_hill  = 0.45 * remaining
    frac_subg  = 0.15 * remaining
    frac_exp   = 0.10 * remaining
    # anything left goes to final annealing

    # ---- Adam ----
    phase_dead = now + frac_adam
    best_a, best_val = _adam_optimize(
        best_a, phase_dead, start_time, best_a, best_val
    )
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ---- Hill‑climbing ----
    now = time.time()
    phase_dead = now + frac_hill
    best_a, best_val = _hillclimb(
        best_a, phase_dead, start_time, best_a, best_val
    )
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ---- Sub‑gradient polishing ----
    now = time.time()
    phase_dead = now + frac_subg
    best_a, best_val = _subgradient_polish(
        best_a, phase_dead, start_time, best_a, best_val,
        max_iters=400_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ---- Exponentiated‑gradient polishing ----
    now = time.time()
    phase_dead = now + frac_exp
    best_a, best_val = _expgrad_polish(
        best_a, phase_dead, start_time, best_a, best_val,
        max_iters=250_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ---- Final simulated‑annealing (using whatever time remains) ----
    if time.time() < deadline:
        best_a, best_val = _random_anneal(
            best_a, deadline, start_time, best_a, best_val, rng
        )
        best_a, best_val = _blend_with_uniform(best_a, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation (just for logging)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Ensure a plain list of floats is returned
    return best_a.tolist()
