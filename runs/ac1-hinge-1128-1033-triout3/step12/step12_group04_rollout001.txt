"""
Propose a step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
* Work on the simplex Σ a_i = 1, a_i ≥ 0 (the scale of a does not matter for the
  true objective).  The vector length is fixed at ``n = 8000`` – this gives a
  very good trade‑off between solution quality and runtime.

* Optimisation is performed on a smooth surrogate of the
  ‑∞‑norm:
        F(a) = (1/α)·log Σ_k exp(α·(a*a)_k) .
  For large α the surrogate approaches the true max‑entry.  Its gradient is

        ∂F/∂a_j = 2·α· Σ_k p_k·a_{k‑j} ,   p_k = exp(α·c_k)/Σ exp(α·c_k),

  where c = a*a (full linear convolution).  The gradient can be obtained
  with a single FFT‑based convolution.

* The search uses the Adam optimiser (first‑ and second‑moment estimates) to
  accelerate convergence, followed by a Euclidean projection onto the simplex
  after each step.

* If no improvement is seen for a while, a small Gaussian jitter is added,
  the Adam moments are reset and the optimisation continues.  This simple
  “restart” mechanism helps to escape flat plateaux.

* The whole routine respects the 1000 s budget (a 0.5 s safety margin is kept
  for the final secure call to ``evaluate_sequence``).  At any moment the best
  sequence found is stored; after the time expires it is evaluated once and
  returned.

The method is deliberately different from the previously supplied sub‑gradient
code and typically obtains a slightly lower bound (target ≈ 1.503).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT length)."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    # Implementation of algorithm from
    #   Wang & Carreira‑Perpiñán, “Projection onto the probability simplex:
    #   An efficient algorithm with a simple proof, and an application”.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _full_conv_fft(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of a and b using real FFT, output length len(a)+len(b)-1."""
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    out_len = a.shape[0] + b.shape[0] - 1
    return conv[:out_len]

# ----------------------------------------------------------------------

                     budget_s: int = 1000,

    Returns the best list of heights found before the time budget expires.
    """

    # --------------------------------------------------------------

    n = 8000                     # length of the sequence

    # --------------------------------------------------------------

    try:

    # --------------------------------------------------------------

    beta1 = 0.9
    beta2 = 0.999

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    eta = 0.12                 # base learning rate

    alpha_ramp_iters = 20000   # iterations over which α grows linearly
    stagnation_limit = 5000    # steps without improvement before jitter

    best_a = a.copy()

        a_candidate = _project_simplex(a_candidate)

        conv_candidate = _full_conv_fft(a_candidate, a_candidate, L)[:2 * n - 1]

            cur_obj = cand_obj
            stagnation = 0
            if cand_obj < best_val:

                best_a = a.copy()
                elapsed = time.time() - start_time

            # the dynamics alive

        print("[final] reported score =", final_score)
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
