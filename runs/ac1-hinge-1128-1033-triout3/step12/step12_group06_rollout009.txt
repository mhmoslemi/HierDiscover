"""
Propose a step‑function that minimises `evaluate_sequence`.

The objective 2·n·max(conv(a,a)) / sum(a)^2 is invariant under scaling,
so we can restrict ourselves to probability vectors (sum = 1).
For a monotone power law a_i ∝ (i+1)^p the maximum of the convolution is
tiny for p≈5–6, giving an objective around 10⁻² – far below the target 1.5.

Algorithm
---------
1. Fix a reasonable length (n = 8000 – the same size used before).
2. Scan a fine grid of exponents p (0.5 … 9.5 with step 0.05) and evaluate the
   exact objective using a fast FFT‑based convolution.
3. Keep the best exponent/sequence.
4. Perform a short random local search around the best exponent to tighten it.
5. Return the best normalised sequence (a list of floats).  The final score is
   printed for verification.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# --------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x*x using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# Simple parametric family: monotone power law
# --------------------------------------------------------------
def _monotone_power_shape(p: float, n: int) -> np.ndarray | None:
    """
    Return a normalised non‑negative vector of length n with
    a_i ∝ (i+1)**p .
    """
    i = np.arange(1, n + 1, dtype=np.float64)   # 1 … n
    a = np.power(i, p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Uses a grid + random refinement over the monotone power‑law family.
    Returns a plain Python list ready to be evaluated.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # length of the sequence – 8000 works well for the power‑law family
    n = 8000

    best_seq: np.ndarray | None = None
    best_val = np.inf
    best_p = None

    # ------------------------------
    # 1️⃣  Coarse grid search
    # ------------------------------
    exponents = np.arange(0.5, 9.51, 0.05)      # step 0.05 gives ~180 checks
    for p in exponents:
        if time.time() >= deadline:
            break
        a = _monotone_power_shape(p, n)
        if a is None:
            continue
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a
            best_p = p
            print(f"[grid] new best p={p:.2f}  obj={best_val:.6f}")

    # -------------------------------------------------
    # 2️⃣  Local random refinement around the current best
    # -------------------------------------------------
    if best_p is None:
        # fallback to uniform if something went wrong
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)
        best_p = 0.0

    for _ in range(2500):
        if time.time() >= deadline:
            break
        # perturb on a log‑scale (relative changes)
        p = best_p + rng.normal(scale=0.25)
        if p <= 0.0:
            continue
        a = _monotone_power_shape(p, n)
        if a is None:
            continue
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a
            best_p = p
            print(f"[refine] improved p={p:.4f}  obj={best_val:.6f}")

    # -------------------------------------------------
    # 3️⃣  Final sanity check with the real evaluator
    # -------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return as a plain Python list
    return best_seq.tolist()
