def _interpolate_to_length(seq, target_n):
    old_n = len(seq)
    new_x = np.linspace(0.0, 1.0, target_n)
    old_x = np.linspace(0.0, 1.0, old_n)
    a = np.interp(new_x, old_x, seq)

    a_sum = a.sum()

        a = np.full(target_n, 1.0 / target_n)
    else:
        a = a / a_sum
    return a

"""
Propose the best non‑negative step function (a list of heights) that minimizes
``evaluate_sequence``.  

The algorithm consists of three stages:

1. **Coarse parametric search** – on a modest size (n≈2000) we scan a
   family of power‑law shapes and a two‑parameter family
   a_i ∝ (i+1)^p·(1‑i/n)^q (including the pure power law as q=0).  
   The cheap exact objective 2·n·max(conv) is evaluated via FFT and the
   best shape is kept.

2. **Upscaling** – the selected shape is interpolated to the final problem
   size (default n≈8000) and normalised to sum to 1.  If a previous solution
   ``height_sequence_1`` is present it is used as an alternative starting
   point.

3. **Smooth‑max mirror descent** – we repeatedly minimise a smooth
   approximation of the max‑convolution:
       f(a)=2·n·(1/λ)·log Σ_k exp(λ·(a⋆a)_k)
   The gradient of the smooth max is computed efficiently by a
   weighted convolution (using FFT).  An additive step with Euclidean
   projection onto the probability simplex implements mirror descent.
   The smoothing parameter λ is increased slowly so that the optimisation
   gradually focuses on the true maximum.  The best exact
   objective seen during the search is stored and finally returned.

The whole routine respects a user‑supplied time budget (default 1000 s)
and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using FFT (real input, real output).
    Returns a length len(a)+len(b)-1 array.
    """
    n, m = a.shape[0], b.shape[0]
    size = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[: n + m - 1]
    return conv


def _objective_fast(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) with sum(a)=1."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _shape_power(p: float, n: int) -> np.ndarray | None:
    """a_i ∝ (i+1)^p, normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    tot = a.sum()
    if tot == 0.0:
        return None
    a /= tot
    return a


def _shape_two_param(p: float, q: float, n: int) -> np.ndarray | None:
    """
    a_i ∝ (i+1)^p * (1 - i/n)^q   (i = 0 … n‑1)
    Generalises a pure power law (q=0) and produces unimodal shapes.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    factor = 1.0 - i / n                               # in (0,1]
    # avoid negative numbers (factor is 0 exactly at i=n)
    a = (i ** p) * (factor ** q)
    tot = a.sum()
    if tot <= 0.0:
        return None
    a /= tot
    return a


def _interpolate(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_n = seq.shape[0]
    if old_n == target_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    out = np.interp(new_x, old_x, seq)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out = np.full(target_n, 1.0 / target_n, dtype=np.float64)
    else:
        out /= s
    return out


def _smooth_max_weights(b: np.ndarray, lam: float):
    """
    Returns (smooth_max, weight_vector) where
        smooth_max = (1/λ)·log Σ exp(λ·b_k)
    and weight_vector = softmax(λ·b).
    Computation is numerically stable.
    """
    b_max = b.max()
    exp_shift = np.exp(lam * (b - b_max))
    w_sum = exp_shift.sum()
    w = exp_shift / w_sum
    smooth = b_max + math.log(w_sum) / lam
    return smooth, w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    Returns a list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin for the outer call

    # --------------------------------------------------------------
    # 0️⃣  Problem size
    # --------------------------------------------------------------
    final_n = 8000          # final dimensionality (good trade‑off)

    # --------------------------------------------------------------
    # 1️⃣  Coarse parametric search on a smaller instance
    # --------------------------------------------------------------
    n0 = 2000
    best_a_small = None
    best_val_small = np.inf

    def _try_candidate(a: np.ndarray, descr: str):
        nonlocal best_a_small, best_val_small
        if a is None:
            return
        val = _objective_fast(a)
        if val < best_val_small:
            best_val_small = val
            best_a_small = a.copy()
            print(f"[grid] new best {val:.6f} via {descr}")

    # pure power‑law family
    for p in np.linspace(0.5, 12.0, 24):      # 24 values ≈ step 0.5
        _try_candidate(_shape_power(p, n0), f"power(p={p:.2f})")
        if time.time() >= deadline - 300:
            break

    # two‑parameter family a_i ∝ i^p·(1‑i/n)^q
    p_vals = np.linspace(0.5, 8.0, 16)        # 16 values
    q_vals = np.linspace(0.5, 8.0, 16)        # 16 values
    for p in p_vals:
        for q in q_vals:
            _try_candidate(_shape_two_param(p, q, n0),
                           f"two(p={p:.2f},q={q:.2f})")
        if time.time() >= deadline - 300:
            break

    # a few random draws to jitter the grid a bit
    rng = np.random.default_rng(seed + 12345)
    for _ in range(800):
        p = rng.uniform(0.1, 12.0)
        q = rng.uniform(0.1, 12.0)
        _try_candidate(_shape_two_param(p, q, n0),
                       f"rand(p={p:.2f},q={q:.2f})")
        if time.time() >= deadline - 300:
            break

    # --------------------------------------------------------------
    # 2️⃣  Upscale to the final dimension and choose a starting point
    # --------------------------------------------------------------
    # If a previous solution exists, keep it as an alternative start.
    init_candidates = []

    if best_a_small is not None:
        init_candidates.append(_interpolate(best_a_small, final_n))

    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        init_candidates.append(_interpolate(prev, final_n))
        print("[init] incorporated previous global solution")
    except Exception:
        pass

    if not init_candidates:
        # fallback – uniform distribution
        init_candidates.append(np.full(final_n, 1.0 / final_n, dtype=np.float64))

    # pick the best of the initial candidates (cheap exact objective)
    a = init_candidates[0]
    best_seq = a.copy()
    best_val = _objective_fast(a)

    for cand in init_candidates[1:]:
        val = _objective_fast(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            a = cand.copy()
    print(f"[init] start objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max mirror‑descent optimisation
    # --------------------------------------------------------------
    n = final_n
    grad_mom = np.zeros_like(a)               # momentum accumulator
    beta = 0.9                                 # exponential decay for momentum
    lam = 1.0                                  # smoothing parameter λ
    max_lambda = 200.0
    it = 0
    no_improve = 0
    max_no_improve = 4000

    while time.time() < deadline:
        # convolution a⋆a
        conv = _full_convolution_fft(a, a)           # length 2n‑1
        cur_exact = 2.0 * n * conv.max()
        if cur_exact < best_val:
            best_val = cur_exact
            best_seq = a.copy()
            no_improve = 0
            print(f"[optim] iter {it:5d} new best {best_val:.6f}")
        else:
            no_improve += 1

        # smooth max and weight vector
        smooth, w = _smooth_max_weights(conv, lam)

        # gradient of the smoothed objective (2·n·smooth) – momentum version
        conv_w_a = _full_convolution_fft(w, a)
        conv_w_rev_a = _full_convolution_fft(w[::-1], a)

        # central part (indices n‑1 … 2n‑2) contains the required n components
        grad = 2.0 * (conv_w_a[n - 1: n - 1 + n] + conv_w_rev_a[n - 1: n - 1 + n])
        grad *= 2.0 * n                               # scale to match the objective

        # momentum
        grad_mom = beta * grad_mom + (1.0 - beta) * grad

        # adaptive step size – we keep updates modest
        step = 0.2 / (np.max(np.abs(grad_mom)) + 1e-12)

        # additive mirror‑descent step with Euclidean projection onto simplex
        a_new = a - step * grad_mom
        a = _project_simplex(a_new)

        # increase λ gradually so that the optimisation increasingly
        # concentrates on the true maximum
        lam = min(max_lambda, lam * 1.001 + 0.01)

        # occasional jitter to escape plateaus
        if no_improve > max_no_improve or (np.random.rand() < 0.001):
            jitter = np.random.normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            grad_mom.fill(0.0)
            no_improve = 0
            print("[optim] jitter applied")

        it += 1

    # --------------------------------------------------------------
    # 4️⃣  Final secure evaluation (just for logging)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised exception:", exc)

    return best_seq.tolist()
