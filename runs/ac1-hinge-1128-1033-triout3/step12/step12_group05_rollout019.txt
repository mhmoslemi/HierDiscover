def _simplex_projection(v):
    """Project onto the probability simplex."""
    # Input: vector v (numpy array)
    # Output: projected vector w s.t. w >= 0, sum w = 1
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    size = 1 << (2 * n - 2).bit_length()          # next power of two ≥ 2·n‑1
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Cheap objective – exact value of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     max_active: int = 15,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Return a sub‑gradient of max(conv) w.r.t. `a`.

    The active set contains at most ``max_active`` largest entries of `conv`.
    If several values are within ``rel_tol`` of the maximum they are all kept.
    """
    n = a.shape[0]
    max_val = conv.max()
    # find the active indices (largest values, possibly a few more because of tolerance)
    if max_active >= conv.size:
        active = np.arange(conv.size)
    else:
        # use partition for speed, then apply the relative tolerance filter
        cand = np.argpartition(conv, -max_active)[-max_active:]
        mask = conv[cand] >= max_val * (1.0 - rel_tol)
        active = cand[mask] if np.any(mask) else np.array([int(np.argmax(conv))])

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # for each i in idx we add a_{k-i}
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector `v` onto the simplex { x ≥ 0 , sum x = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]                # sort in descending order
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (sequence) that minimises
    ``evaluate_sequence``.  The algorithm performs several Adam‑optimisation
    restarts starting from the previous best solution (if any) and from random
    Dirichlet draws.  The best sequence discovered is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 0.5       # keep a small safety margin

    # --------------------------------------------------------------
    # Problem size – a compromise between expressivity and speed.
    # --------------------------------------------------------------
    n = 8000

    best_seq: np.ndarray | None = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Helper to record a new best candidate (cheap objective)
    # ------------------------------------------------------------------
    def maybe_update(candidate: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[update] new best cheap objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previously saved solution (if available)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0 or not np.isfinite(s):
            a0 = np.full_like(a0, 1.0 / n)
        else:
            a0 = a0 / s
        maybe_update(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random initialisations (Dirichlet draws) to diversify the search
    # ------------------------------------------------------------------
    n_initial = 6                # total restarts (including possible previous best)
    init_vectors = []

    # start from the current best (if we already have one)
    if best_seq is not None:
        init_vectors.append(best_seq.copy())

    # fill the rest with random Dirichlet samples (concentration < 1 → sparse)
    while len(init_vectors) < n_initial:
        alpha = 0.4                      # smaller than 1 encourages sparsity
        vec = rng.gamma(alpha, 1.0, size=n)
        vec /= vec.sum()
        init_vectors.append(vec)

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimiser with simplex projection
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_start = 0.7
    lr_end = 5e-4

    # Parameters for the optimiser
    max_active = 20          # size of the active set for the sub‑gradient
    rel_tol = 5e-7           # tolerance for active‑set expansion

    for restart_idx, a in enumerate(init_vectors):
        if time.time() >= deadline:
            break
        print(f"\n[restart] #{restart_idx + 1}/{n_initial}")

        m = np.zeros_like(a)          # first moment
        v = np.zeros_like(a)          # second moment
        t = 0

        # No‑improvement counter to cut a restart early if it stalls
        no_improve = 0
        max_no_improve = 8000

        while time.time() < deadline - 0.5:
            t += 1
            # ----------------------------------------------------------
            # 2a.  Evaluate current objective
            # ----------------------------------------------------------
            conv = _full_convolution(a)
            cur_max = conv.max()
            cur_obj = 2.0 * n * cur_max

            # Record best global solution
            if cur_obj < best_val:
                best_val = cur_obj
                best_seq = a.copy()
                no_improve = 0
                print(f"[optim] restart {restart_idx + 1} iter {t} new best cheap = {best_val:.6f}")
            else:
                no_improve += 1

            # ----------------------------------------------------------
            # 2b.  Compute sub‑gradient (active‑set version)
            # ----------------------------------------------------------
            grad = _subgradient_max(a, conv, max_active=max_active, rel_tol=rel_tol)

            # ----------------------------------------------------------
            # 2c.  Adam update (projected onto the simplex)
            # ----------------------------------------------------------
            # bias‑corrected learning‑rate (linear decay)
            progress = (time.time() - start) / (deadline - start)
            lr = lr_start * (1.0 - progress) + lr_end

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step                     # minimise → move opposite to gradient

            # Project back onto the probability simplex (ensures non‑negativity & sum = 1)
            a = _simplex_projection(a)

            # ----------------------------------------------------------
            # 2d.  Occasional jitter to shake off plateaus
            # ----------------------------------------------------------
            if rng.random() < 0.001 or no_improve >= max_no_improve:
                jitter = rng.normal(scale=1e-3, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = a / a.sum()
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print("[jitter] injected Gaussian noise")

            # ----------------------------------------------------------
            # 2e.  Early‑stop a stagnating restart
            # ----------------------------------------------------------
            if no_improve >= max_no_improve:
                print("[restart] early stop after stagnation")
                break

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation (real objective)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
