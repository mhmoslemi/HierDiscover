"""
Optimization of a non‑negative step function to minimise the evaluation
objective.  The algorithm proceeds in two phases:

1️⃣ Random‑support sampling – many candidates are built by putting equal
   mass on a random subset of indices (different densities).  The best
   candidate according to the cheap proxy 2·n·max(conv) is kept.

2️⃣ Smoothed‑max mirror descent – starting from the best candidate we
   minimise a smooth approximation of the max‑convolution using an
   exponentiated‑gradient (Kullback‑Leibler mirror) update.  The smooth
   approximation is a soft‑max (log‑sum‑exp) with a temperature that is
   gradually increased, turning the problem from a smooth one into the
   original nonsmooth objective.  Momentum, a linearly decaying step‑size
   and occasional Gaussian jitter are employed to escape plateaux.

The routine respects the supplied time budget (default 1000 s) and finally
calls ``evaluate_sequence`` on the best sequence discovered.
"""

import time
import numpy as np

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself using an FFT."""
    n = x.shape[0]
    # size = next power of two ≥ 2·n‑1
    size = 1 << (2 * n - 2).bit_length()
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
    return conv

def _cheap_objective(a: np.ndarray, n: int) -> float:
    """Exact inexpensive objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

def _gradient_and_max(a: np.ndarray, n: int, tau: float):
    """
    Return the gradient of the smooth log‑sum‑exp proxy of max(conv)
    together with the current max(conv) value.
    """
    b = _full_convolution(a)
    max_b = b.max()
    # Soft‑max weights, numerically stable
    w = np.exp(tau * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w = w / w_sum
    # Gradient: 2 * (a_rev * w) restricted to the original length
    a_rev = a[::-1]
    c = np.convolve(a_rev, w)          # full convolution
    grad = 2.0 * c[n - 1 : n - 1 + n]   # slice of length n
    return grad, max_b

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Simple Euclidean projection onto the probability simplex.
    The implementation follows the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    # shift to make sure the vector is non‑negative after projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        if rng.random() < 0.02:

    if best_seq is None:

    # --------------------------------------------------------------
    a = best_seq.copy()

        elapsed = time.time() - start_time

        if rng.random() < 0.001:

            momentum.fill(0.0)

    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()

"""
Optimization of a non‑negative step function to minimise the official
objective.  The search works in two stages.

1️⃣ *Random‑support exploration* – many candidates are built by putting
   equal mass on a random subset of indices (different densities).  The
   best candidate with respect to the cheap proxy
   ``2·n·max(conv(a,a))`` (assuming Σa = 1) is kept.

2️⃣ *Smoothed‑max mirror descent* – starting from the best candidate we
   minimise a smooth approximation of the max‑convolution using an
   exponentiated‑gradient (KL‑mirror) update.  The smooth proxy is a
   soft‑max (log‑sum‑exp) of the convolution values whose temperature is
   gradually increased, turning the problem from a smooth one into the
   original non‑smooth objective.  Momentum, a linearly decaying step‑size
   and occasional Gaussian jitter are employed to escape plateaux.

The routine respects the supplied time budget (default 1000 s) and finally
calls ``evaluate_sequence`` on the best sequence discovered.
"""

import time
import numpy as np


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself using an FFT."""
    n = x.shape[0]
    # size = next power of two ≥ 2·n‑1
    size = 1 << (2 * n - 2).bit_length()
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
    return conv


def _cheap_objective(a: np.ndarray, n: int) -> float:
    """Exact inexpensive objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _gradient_and_max(a: np.ndarray, n: int, tau: float):
    """
    Gradient of the soft‑max proxy of ``max(conv(a,a))`` together with the
    current maximum convolution value.

    The proxy is   L_tau(a) = (1/τ)·log Σ_k exp(τ·b_k),   b = a*a.
    Its gradient w.r.t. a is
        grad_i = 2·Σ_k w_k·a_{k‑i},
    where w_k = exp(τ·b_k) / Σ_j exp(τ·b_j).
    """
    b = _full_convolution(a)
    max_b = b.max()
    # Soft‑max weights (numerically stable)
    w = np.exp(tau * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w = w / w_sum

    # Gradient via a reversed‑vector convolution:
    a_rev = a[::-1]
    c = np.convolve(a_rev, w)              # full convolution
    grad = 2.0 * c[n - 1 : n - 1 + n]       # slice of length n
    return grad, max_b


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the algorithm of
    Wang & Carreira‑Perpiñán (ICML 2013).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
