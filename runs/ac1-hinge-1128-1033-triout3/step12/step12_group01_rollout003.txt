"""
Improved search for a step‑function with a low value of

    O = 2·n·max(conv(seq,seq)) / (sum(seq))²

The method expands on the previous Adam + mirror‑descent scheme:

*   A larger and more diverse set of seeds (beta‑shapes, Dirichlet draws,
    power‑law, sinusoidal and exponential patterns) – especially very
    front‑loaded distributions – are used.
*   The soft‑max temperature (α) is driven much higher (up to a few thousand)
    and the learning‑rate schedule is more aggressive, which lets the
    optimisation focus on the true max‑entry sooner.
*   Stagnation is handled with occasional Gaussian jitter.
*   After the Adam phase a longer entropy‑mirror‑descent polishing is
    performed, again on the exact max‑convolution.
*   The whole routine respects the 1000 s budget, keeps a small safety
    margin for the final evaluation and prints progress information.

The returned sequence is normalised (sum = 1); scaling does not affect the
objective.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (full, i.e. length 2·len‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two (possibly different) vectors (full length)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent polishing (exact max entry)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 3000, eta_start: float = 0.45) -> np.ndarray:
    """
    Mirrors‑descent on the exact L∞‑norm of the self‑convolution.
    The iterate stays in the probability simplex (sum = 1, a_i≥0).
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of the max entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]           # derivative of a_i·a_{k-i}
        eta = eta_start * (1.0 - t / steps)      # linear decay
        a = a * np.exp(-eta * g)                # multiplicative update
        s = a.sum()
        if s > 0.0:
            a /= s
        else:                                    # safety fallback
            a.fill(1.0 / n)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights (a probability vector)
    that (hopefully) yields a lower value of ``evaluate_sequence`` than
    the previous best (~1.508). The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # a small safety margin for the final evaluation call
    deadline = start + budget_s - 0.8

    # --------------------------------------------------------------
    # Problem size – a slightly larger n gives more degrees of freedom
    # --------------------------------------------------------------
    n = 12000               # can be tuned, but stays well within the budget

    # --------------------------------------------------------------
    # Build a rich pool of starting vectors
    # --------------------------------------------------------------
    init_vectors = []

    # 0) previous solution (if the global variable exists)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        # interpolate to the current length n
        x_new = np.linspace(0.0, 1.0, n)
        x_old = np.linspace(0.0, 1.0, old_len)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)
        init_vectors.append(a)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # 1) uniform distribution – serves as a sanity check
    init_vectors.append(np.full(n, 1.0 / n))

    # 2) smooth Beta shapes (including very front‑loaded ones)
    beta_params = [
        (0.2, 0.2), (0.3, 0.3), (0.5, 0.5),
        (0.2, 1.0), (1.0, 0.2), (0.1, 0.1),
        (5.0, 0.5), (0.5, 5.0)
    ]
    idx = np.arange(1, n + 1, dtype=np.float64) / n
    for p, q in beta_params:
        a = (idx ** (p - 1.0)) * ((1.0 - idx) ** (q - 1.0))
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)
        init_vectors.append(a)

    # 3) random Dirichlet draws with a wide range of concentration parameters
    conc_params = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 5.0]
    for conc in conc_params:
        a = np.random.dirichlet(np.full(n, conc))
        init_vectors.append(a)

    # 4) power‑law (front‑loaded) shapes a_i ∝ (i+1)^(-p)
    for p in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
        i = np.arange(n, dtype=np.float64) + 1.0
        a = i ** (-p)
        a /= a.sum()
        init_vectors.append(a)

    # 5) exponential decay a_i ∝ exp(-λ·i)
    for lam in [0.0002, 0.0005, 0.001, 0.002, 0.005]:
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * i)
        a /= a.sum()
        init_vectors.append(a)

    # 6) Sinusoidal shapes (half‑sine, raised‑cosine, etc.)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    sinusoid = np.sin(np.pi * x)          # zero at the ends, peak in the centre
    init_vectors.append(sinusoid / sinusoid.sum())
    raised_cos = (1 - np.cos(2 * np.pi * x)) / 2.0
    init_vectors.append(raised_cos / raised_cos.sum())

    # --------------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    eps_log = 1e-30

    # Reserve a fraction of the total budget for the final polishing pass
    polishing_reserve = 0.12 * budget_s   # ≈12 % of total time

    best_obj = np.inf
    best_a = None

    # --------------------------------------------------------------
    # Main optimisation loop (multiple restarts)
    # --------------------------------------------------------------
    for restart_id, a0 in enumerate(init_vectors):
        now = time.time()
        if now >= deadline - polishing_reserve:
            break

        print(f"[restart] #{restart_id + 1}/{len(init_vectors)} – start")
        # work in the log‑domain (ensures positivity)
        z = np.log(a0 + eps_log)
        m = np.zeros_like(z)
        v = np.zeros_like(z)

        it = 0
        stagnation = 0

        while True:
            now = time.time()
            if now >= deadline - polishing_reserve:
                break

            # ----------------------------------------------------------
            # Scheduling of learning‑rate and soft‑max sharpness
            # ----------------------------------------------------------
            prog = (now - start) / (deadline - start)          # 0 → 1
            # learning‑rate decays from 0.5 to 0.01
            lr = 0.5 * (1.0 - prog) + 0.01 * prog
            # α grows from ≈1 to about 2000 (soft‑max → hard max)
            alpha = 1.0 + 1999.0 * (prog ** 1.2)

            # ----------------------------------------------------------
            # Recover the probability vector a from its logits
            # ----------------------------------------------------------
            a = np.exp(z)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

            # ----------------------------------------------------------
            # Exact objective (max entry of the self‑convolution)
            # ----------------------------------------------------------
            conv = _full_convolution(a)               # length 2·n‑1
            max_conv = conv.max()
            exact_obj = 2.0 * n * max_conv

            if exact_obj < best_obj - 1e-12:
                best_obj = exact_obj
                best_a = a.copy()
                stagnation = 0
                print(f"[iter] new best O={best_obj:.6f}  it={it}  t={now-start:.1f}s")
            else:
                stagnation += 1

            # ----------------------------------------------------------
            # Soft‑max of the convolution (smooth approximation of max)
            # ----------------------------------------------------------
            maxc = max_conv
            exp_scaled = np.exp(alpha * (conv - maxc))
            sumexp = exp_scaled.sum()
            weights = exp_scaled / sumexp                     # Σ weights = 1

            # ----------------------------------------------------------
            # Gradient w.r.t. a (derived in the reference implementation)
            # ----------------------------------------------------------
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)  # length 3·n‑2
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]          # slice length n

            # ----------------------------------------------------------
            # Chain‑rule through the soft‑max (a = softmax(z))
            # ----------------------------------------------------------
            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            # ----------------------------------------------------------
            # Adam update on logits
            # ----------------------------------------------------------
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ----------------------------------------------------------
            # Jitter when we appear to be stuck
            # ----------------------------------------------------------
            if stagnation >= 4000:
                jitter = np.random.normal(scale=0.02, size=n)
                z += jitter
                stagnation = 0
                print(f"[reset] jitter injected at it={it}")

    # --------------------------------------------------------------
    # Polishing phase – entropy‑mirror descent on the exact max entry
    # --------------------------------------------------------------
    if best_a is not None:
        time_left = deadline - time.time()
        # Roughly 50 µs per mirror‑descent step → convert time into steps
        steps = max(2000, int(min(15000, time_left * 18000)))
        print(f"[polish] mirror‑descent start – {steps} steps")
        a_refined = _max_refine(best_a, steps=steps, eta_start=0.45)
        conv_ref = _full_convolution(a_refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = a_refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # Final secure evaluation (the platform will call evaluate_sequence)
    # --------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else init_vectors[0].tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
