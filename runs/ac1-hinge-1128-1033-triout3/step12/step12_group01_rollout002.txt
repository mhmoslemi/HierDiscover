   convolution entry (the one attaining the maximum).  The step‑size
   ``η`` decays linearly from a large initial value to a small final
   value.  The best vector over all runs is kept.

3. **Stochastic local refinement** – starting from the best vector,
   repeated random transfers of a tiny amount of mass between two
   positions are accepted only when they lower the exact objective.
   This “hill‑climbing’’ can escape shallow plateaus left by the
   mirror‑descent phase.

All work respects the supplied time budget (default 1000 s).  The
final vector is returned as a plain Python list; a final call to
``evaluate_sequence`` prints the exact score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full length = 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the maximum convolution entry
# ----------------------------------------------------------------------


def _max_subgradient(a: np.ndarray):
    """
    Compute the sub‑gradient of ``max(conv(a,a))`` at the currently active
    shift.  Returns (g, max_val) where ``g`` has the same shape as ``a``.
    """
    conv = _full_convolution(a)
    k = int(conv.argmax())
    max_val = conv[k]

    n = a.shape[0]
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    # for each i in [lo, hi] the contribution is 2 * a_{k‑i}
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k - idx]
    return g, max_val


# ----------------------------------------------------------------------
# Entropy mirror‑descent on the exact max
# ----------------------------------------------------------------------


def _mirror_descent(a0: np.ndarray,
                    max_steps: int = 20000,
                    eta_start: float = 0.8,
                    eta_end: float = 0.02,
                    time_deadline: float = np.inf):
    """
    Perform mirror‑descent (multiplicative updates on the simplex) using
    the exact sub‑gradient of the max‑convolution.
    The step‑size η decays linearly from ``eta_start`` to ``eta_end``.
    The loop stops when ``max_steps`` is reached or the wall‑clock time
    exceeds ``time_deadline``.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    # objective = 2·n·max(conv)
    n = a.shape[0]
    conv = _full_convolution(a)
    best_obj = 2.0 * n * conv.max()

    for step in range(max_steps):
        if time.time() >= time_deadline:
            break

        # sub‑gradient of the current max entry
        g, max_val = _max_subgradient(a)

        # linear decay of η
        eta = eta_start + (eta_end - eta_start) * (step / max_steps)

        # mirror update (entropy mirror = multiplicative)
        a = a * np.exp(-eta * g)

        # re‑project onto the probability simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)

        # keep the best
        obj = 2.0 * n * max_val
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


# ----------------------------------------------------------------------
# Local stochastic refinement (hill‑climbing)
# ----------------------------------------------------------------------


def _local_refine(a: np.ndarray,
                  max_iters: int = 20000,
                  move_frac: float = 0.03,
                  time_deadline: float = np.inf):
    """
    Randomly move a small fraction of mass between two positions.
    Accept only improving moves (strict descent).  The routine stops
    after ``max_iters`` iterations or when the wall‑clock time exceeds
    ``time_deadline``.
    """
    n = a.shape[0]
    best_a = a.copy()
    conv = _full_convolution(best_a)
    best_obj = 2.0 * n * conv.max()

    for it in range(max_iters):
        if time.time() >= time_deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # choose a signed transfer amount
        delta = (np.random.rand() - 0.5) * 2.0 * move_frac * best_a[i]
        # ensure the move does not create negativity
        if delta > 0.0:
            delta = min(delta, best_a[i])
        else:
            delta = -min(-delta, best_a[j])

        if np.isclose(delta, 0.0):
            continue

        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # compute the new objective (full convolution)
        conv_new = _full_convolution(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj:
            best_a = a_new
            best_obj = obj_new
            # Occasionally print progress
            if (it + 1) % 5000 == 0:
                print(f"[refine] iter={it+1}  new best O={best_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimizes
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds and prints intermediate progress.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start + budget_s - 0.8

    # ------------------------------------------------------------------
    # Problem size – the same as the reference implementation
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Build a diversified pool of starting vectors (all are normalized to sum=1)
    # ------------------------------------------------------------------
    init_vectors = []

    # 1) previous best, if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        x_new = np.linspace(0.0, 1.0, n)
        x_old = np.linspace(0.0, 1.0, old_len)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)
        init_vectors.append(a)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # 2) smooth Beta shapes
    betas = [(1.0, 1.0), (2.0, 2.0), (0.5, 0.5), (5.0, 5.0),
             (0.3, 0.7), (2.5, 1.0), (3.0, 0.5), (0.8, 3.0)]
    idx = np.arange(1, n + 1, dtype=np.float64) / n
    for p, q in betas:
        a = (idx ** (p - 1.0)) * ((1.0 - idx) ** (q - 1.0))
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)
        init_vectors.append(a)

    # 3) monotone power‑law (decreasing)
    for power in [0.2, 0.5, 1.0, 1.5, 2.0]:
        a = (1.0 - idx) ** power
        a = a / a.sum()
        init_vectors.append(a)

    # 4) random Dirichlet draws (various concentration parameters)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 5.0, 10.0]
    for conc in concs:
        a = np.random.dirichlet(np.full(n, conc))
        init_vectors.append(a)

    # 5) a few “saw‑tooth’’ patterns
    for period in [2, 4, 8, 16]:
        a = np.tile([0.9, 0.1], n // 2 + 1)[:n]
        a = a * (1.0 / period) + (1.0 - 1.0 / period) / n
        a = a / a.sum()
        init_vectors.append(a)

    # ------------------------------------------------------------------
    # Main optimisation loop (mirror‑descent on each start)
    # ------------------------------------------------------------------
    best_obj = np.inf
    best_a = None

    # Reserve ~15 % of the budget for the final stochastic refinement
    refine_reserve = 0.15 * budget_s

    for idx_start, a0 in enumerate(init_vectors):
        if time.time() >= deadline - refine_reserve:
            break

        print(f"[restart] start #{idx_start + 1}/{len(init_vectors)}")

        # Define a per‑restart deadline (leaving room for later phases)
        per_restart_deadline = min(deadline - refine_reserve,
                                   time.time() + (deadline - time.time()) * 0.4)

        a_opt, obj_opt = _mirror_descent(
            a0,
            max_steps=30000,
            eta_start=0.8,
            eta_end=0.02,
            time_deadline=per_restart_deadline,
        )

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            print(f"[best] new overall O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Stochastic local refinement (hill‑climbing) – uses the remaining time
    # ------------------------------------------------------------------
    if best_a is not None:
        time_left = deadline - time.time()
        if time_left > 2.0:   # need at least a sliver of time
            print(f"[refine] starting local refinement with {time_left:.1f}s left")
            a_refined, obj_refined = _local_refine(
                best_a,
                max_iters=50000,
                move_frac=0.025,
                time_deadline=deadline,
            )
            if obj_refined < best_obj:
                best_obj = obj_refined
                best_a = a_refined
                print(f"[refine] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (the secure wrapper)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else init_vectors[0].tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(internal O≈{best_obj:.6f})")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
