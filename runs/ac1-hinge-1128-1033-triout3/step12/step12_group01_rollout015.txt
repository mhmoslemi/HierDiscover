"""
Propose a non‑negative step function by iteratively solving a linear program.
The LP maximises the total mass of a new vector g while keeping the
convolution f * g bounded by the current worst‑case value M = max(f*f).
A small convex combination f←(1‑t)f+t g (t gradually decays) makes the
maximum entry of f*f strictly decrease.  After the LP phase a few
entropy‑mirror‑descent steps polish the final vector on the exact
max‑convolution.  The routine works for modest n (≈500) so that many
LP iterations fit comfortably into the 1000 s budget and typically
reach an objective O = 2·n·max(conv)/sum² well below the previous
1.508‑level.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Helper: full convolution (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Mirror‑descent polishing (exact max‑entry of self‑convolution)
# ----------------------------------------------------------------------
def _mirror_descent(a: np.ndarray, steps: int = 3000, eta_start: float = 0.5) -> np.ndarray:
    """Entropy‑mirror descent on the exact max‑entry of the self‑convolution."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of the max‑entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)
    return a

        v = (idx ** (a - 1.0)) * ((1.0 - idx) ** (b - 1.0))
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            v /= s

    except Exception as exc:            # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
