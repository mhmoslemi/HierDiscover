def softmax(beta, arr):
    z = arr * beta
    max_z = np.max(z)
    e_z = np.exp(z - max_z)
    sum_e = np.sum(e_z)
    w = e_z / sum_e
    return w

def project_simplex(v):
    # Ensure non-negative after projection, sum=1.
    if np.sum(v) == 0:
        # return uniform
        return np.full_like(v, 1.0/v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent optimisation of the max‑convolution objective.
The algorithm smooths the max via a soft‑max (temperature β) and
performs additive updates on the probability simplex (projection
onto the simplex).  β and the step size are annealed during the
run.  The final best sequence is evaluated with ``evaluate_sequence``.

import time
import numpy as np

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using an FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (≥0, sum=1)."""
    # See https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All components are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise
    s = w.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full_like(v, 1.0 / n)
    return w / s

        print("[final] evaluate_sequence raised:", exc)

"""
Gradient‑descent optimisation of the max‑convolution objective.

The algorithm smooths the max via a temperature β (soft‑max) and
performs additive updates on the probability simplex (projection by

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real 1‑D arrays using an FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]

# ----------------------------------------------------------------------


def _cheap_objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))   (the denominator Σa is 1)."""
    n = a.shape[0]
    conv = _fft_convolve(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (sequence) that minimises
    ``evaluate_sequence``.  The method uses a smoothed max‑convolution
    (soft‑max) together with a projected gradient descent on the simplex.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # reserve a little time for the final call

    # ------------------------------------------------------------------
    # Problem size – a trade‑off between expressivity and speed.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise: reuse a previously found solution if available,
    # otherwise start from the uniform distribution.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old.shape[0] != n:
            # Linear interpolation to the new resolution
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, old)
        else:
            a = old.copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0.0 and np.isfinite(s) else np.full(n, 1.0 / n)
    except Exception:
        a = np.full(n, 1.0 / n)

    # Keep the best sequence seen so far (cheap objective is enough for
    # intermediate comparisons because the true objective is scale‑invariant).
    best_seq = a.copy()
    best_val = _cheap_objective(a)
    print(f"[init] cheap objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 999)
    momentum = np.zeros_like(a)

    beta_start = 5.0            # soft‑max temperature at the beginning
    beta_end = 2000.0           # final temperature (sharp approximation of max)
    step_start = 0.2            # initial additive step size
    step_end = 1e-5             # final step size
    mom_coeff = 0.9             # momentum coefficient

    it = 0
    while time.time() < deadline - 0.5:
        # --------------------------------------------------------------
        # 1.  Self‑convolution a*a (needed for the soft‑max)
        # --------------------------------------------------------------
        conv_aa = _fft_convolve(a, a)                 # length 2·n‑1
        cur_obj = 2.0 * n * conv_aa.max()
        if cur_obj < best_val:
            best_val = cur_obj
            best_seq = a.copy()
            print(f"[iter {it:06d}] new best cheap={best_val:.6f}")

        # --------------------------------------------------------------
        # 2.  Soft‑max (smooth max) and its gradient
        # --------------------------------------------------------------
        elapsed = time.time() - start
        progress = elapsed / (deadline - start)      # 0 → 1
        beta = beta_start + (beta_end - beta_start) * progress

        # stable soft‑max
        scaled = beta * conv_aa
        mx = scaled.max()
        exp_scaled = np.exp(scaled - mx)
        w = exp_scaled / exp_scaled.sum()            # length 2·n‑1, sums to 1

        # gradient of the smoothed max:
        #   ∂M/∂a_i = 2 * Σ_k w_k a_{k-i}
        a_rev = a[::-1]
        conv_wa = _fft_convolve(w, a_rev)            # length 3·n‑2
        grad = 2.0 * conv_wa[n - 1: n - 1 + n]        # size n

        # --------------------------------------------------------------
        # 3.  Momentum + additive update on the simplex
        # --------------------------------------------------------------
        momentum = mom_coeff * momentum + (1.0 - mom_coeff) * grad
        step = step_start * (1.0 - progress) + step_end * progress
        a_tmp = a - step * momentum
        a_tmp = np.maximum(a_tmp, 0.0)                # keep non‑negative
        s = a_tmp.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a_tmp / s                             # renormalise to sum = 1

        # --------------------------------------------------------------
        # 4.  Small jitter every few thousand iterations to break plateaux
        # --------------------------------------------------------------
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            momentum.fill(0.0)

        it += 1

    # ------------------------------------------------------------------
    # Optional local refinement (tiny random transfers) while we still have
    # a fraction of the budget left.
    # ------------------------------------------------------------------
    while time.time() < deadline - 0.2:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a tiny amount δ from i to j
        delta = (rng.random() - 0.5) * 0.01
        if best_seq[i] - delta < 0 or best_seq[j] + delta < 0:
            continue
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        obj_cand = _cheap_objective(cand)
        if obj_cand < best_val:
            best_val = obj_cand
            best_seq = cand
            print(f"[local] improved cheap={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
