def beta_initial(alpha, n):
    x = np.linspace(0, 1, n)
    eps = 1e-12
    x = np.clip(x, eps, 1-eps)  # avoid zero
    pdf = (x**(alpha-1) * (1-x)**(alpha-1))

"""
Robust optimizer for the self‑convolution max problem.

The algorithm repeatedly samples diverse initial probability vectors,
optimises them with Adam on a smooth‑max surrogate of the max‑convolution,
and finishes each run with a cheap projected sub‑gradient sweep.
A small collection of analytically‑motivated initial shapes (Beta‑densities)
is also tested.  The best solution found before the time budget expires
is finally evaluated with the provided `evaluate_sequence` function.
"""

import time
import math
import numpy as np
from scipy.special import gammaln

# ----------------------------------------------------------------------
# Low‑level utilities (must stay at module level – no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself using FFT.
    Length = 2*len(x)-1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of weight vector w (length 2n‑1) with a (length n).
    Returns a vector of length n:
        (w ⋆ a)_i = Σ_k w_k * a_{k-i}
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1                         # indices n‑1 … n‑1+n‑1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_initial(alpha: float, n: int) -> np.ndarray:
    """
    Return a discretised symmetric Beta(α,α) density on n points,
    normalised to sum to 1.  Very small values near the boundaries
    are safely clipped.
    """
    eps = 1e-12
    x = np.linspace(0.0, 1.0, n)
    x = np.clip(x, eps, 1.0 - eps)
    # log pdf = (α-1)(log x + log(1-x)) - log B(α,α)
    log_pdf = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    log_beta = gammaln(alpha) + gammaln(alpha) - gammaln(2.0 * alpha)
    log_pdf -= log_beta
    pdf = np.exp(log_pdf)
    pdf = np.maximum(pdf, 0.0)
    s = pdf.sum()
    if s == 0.0:
        pdf = np.full(n, 1.0 / n)
    else:
        pdf = pdf / s
    return pdf


def _objective_val(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(self‑convolution) for a probability vector."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Core optimiser for a single start vector
# ----------------------------------------------------------------------
def _optimize_candidate(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Run a medium‑length Adam optimisation from `a0` (already normalised).
    Returns the best vector found and its objective value.
    """
    # hyper‑parameters
    lambda_start = 5.0
    lambda_end = 20000.0
    adam_lr = 0.05
    b1, b2 = 0.9, 0.999
    adam_eps = 1e-8
    max_iters = 100000
    stagnation_limit = 3000
    smooth_thresh = 5000.0          # switch to exact max above this λ

    # initialise log‑softmax parameters
    eps = 1e-12
    z = np.log(a0 + eps)
    # a small random perturbation helps to escape plateaus
    z += np.random.normal(scale=0.02, size=n)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_local_a = a0.copy()
    best_local_val = _objective_val(a0, n)
    stagnation = 0

    t_iter = 0
    while time.time() < deadline and t_iter < max_iters:
        t_iter += 1

        # ---- 1️⃣  softmax → probability vector
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # ---- 2️⃣ evaluate exact objective
        conv = _full_convolution(a)
        conv_max = conv.max()
        cur_val = 2.0 * n * conv_max

        # ---- 3️⃣ track best solution
        if cur_val < best_local_val - 1e-12:
            best_local_val = cur_val
            best_local_a = a.copy()
            stagnation = 0
            # early exit if we beat the known target
            if cur_val < 1.503:
                break
        else:
            stagnation += 1

        # ---- 4️⃣ temperature for smooth‑max
        prog = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()))
        lam = lambda_start * (1.0 - prog) + lambda_end * prog

        # ---- 5️⃣ compute weights over convolution entries
        if lam > smooth_thresh:
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            # smooth‑max via soft‑max
            exp_vals = np.exp(lam * (conv - conv_max))
            w = exp_vals / exp_vals.sum()

        # ---- 6️⃣ gradient of (smooth) max w.r.t. a  (2·cross‑correlation)
        g_raw = _cross_corr(w, a)
        g = 2.0 * g_raw

        # ---- 7️⃣ back‑prop through the soft‑max
        dot_ag = np.dot(a, g)
        grad_z = a * (g - dot_ag)

        # ---- 8️⃣ Adam update
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z ** 2)
        m_hat = m / (1.0 - b1 ** t_iter)
        v_hat = v / (1.0 - b2 ** t_iter)
        z -= adam_lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ---- 9️⃣ occasional noise to avoid stagnation
        if stagnation > stagnation_limit:
            z += np.random.normal(scale=0.05, size=n)
            stagnation = 0

    # ------------------------------------------------------------------
    # Sub‑gradient refinement (cheap, direct on the true max)
    # ------------------------------------------------------------------
    a = best_local_a.copy()
    obj = best_local_val
    step = 0.02
    for _ in range(3000):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a)
        kmax = int(np.argmax(conv))

        # gradient of the true max (vector g_i = 2·a_{kmax-i})
        g = np.zeros_like(a)
        if kmax < n:
            # region 0 … kmax
            g[:kmax + 1] = 2.0 * a[kmax::-1]
        else:
            # region kmax-(n‑1) … n‑1
            offset = kmax - (n - 1)
            L = n - offset
            g[offset:] = 2.0 * a[:L][::-1]

        a_trial = _project_simplex(a - step * g)
        obj_trial = 2.0 * n * _full_convolution(a_trial).max()

        if obj_trial < obj - 1e-12:
            a = a_trial
            obj = obj_trial
            best_local_a = a.copy()
            best_local_val = obj
            step = min(step * 1.1, 0.5)
            if obj < 1.503:
                break
        else:
            step *= 0.7

    return best_local_a, best_local_val
