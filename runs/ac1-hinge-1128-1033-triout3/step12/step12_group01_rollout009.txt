"""
Propose a step‑function sequence that (empirically) beats the previous
best O≈1.50813.  

The algorithm works in three stages  

1. **Initialisation** – reuse the previous solution (if available) and
   normalise it to a probability distribution (sum = 1).  

2. **Targeted greedy mass transfer** – repeatedly locate the index `k`
   where the self‑convolution attains its maximum.  The contribution of
   every entry `a[i]` to that maximum is `c_i = a[i] * a[k‑i]`
   (zero outside the valid range).  The algorithm moves a small amount
   of mass from the entry with the largest `c_i` to an entry that does
   not contribute (or contributes the least) to the same convolution
   index.  The convolution is updated *incrementally* using slice
   operations, so each iteration costs O(n) work only.  Moves are kept
   only if they strictly decrease the current maximum; a tiny
   simulated‑annealing acceptance probability is kept to escape plateaus.

3. **Mirror‑descent polishing** – the best vector found is refined by
   the entropy‑mirror‑descent routine from the reference implementation,
   using the remaining time budget.

All numbers stay non‑negative, the sum stays equal to one (scale‑invariant
objective) and the final list is returned for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution (FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of the vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

# Mirror‑descent polishing (exact max of the self‑convolution)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 3000, eta_start: float = 0.5) -> np.ndarray:
    """Entropy‑mirror descent on the exact max‑entry of the self‑convolution."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of max entry
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[idx] = 2.0 * a[k - idx]
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)
    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget
    (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start + budget_s - 0.8

    # ------------------------------------------------------------------
    # Problem size – keep the same n as the reference implementation
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from the previous solution if it exists
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        x_old = np.linspace(0.0, 1.0, old_len, endpoint=False)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
    except Exception:
        # fall back to a uniform start
        a = np.full(n, 1.0 / n)

    # normalise to a probability vector (sum = 1)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Compute initial self‑convolution and objective
    # ------------------------------------------------------------------
    conv = _full_convolution(a)           # length 2·n‑1
    cur_max = float(conv.max())
    best_obj = 2.0 * n * cur_max
    best_a = a.copy()
    print(f"[init] O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Greedy mass‑transfer loop (targeted reduction of the max entry)
    # ------------------------------------------------------------------
    # hyper‑parameters
    max_move_abs = 1e-3          # absolute upper bound for a single transfer
    step_fraction = 0.05         # fraction of source mass to move
    eps = np.finfo(float).eps

    it = 0
    while True:
        now = time.time()
        if now >= deadline - 0.5:         # reserve a little time for polishing
            break

        # locate the current maximum of the convolution
        k = int(conv.argmax())
        cur_max = float(conv[k])

        # range of indices that actually influence this entry
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)

        # contribution of each a[i] to this max (c_i = a[i] * a[k-i])
        contrib = a[idx] * a[k - idx]

        # source index – the one with the largest contribution
        i_src = lo + int(np.argmax(contrib))

        # candidate target indices that do **not** affect the current max
        if lo > 0:
            cand_target = np.arange(0, lo)               # left side, partner = 0
        elif hi < n - 1:
            cand_target = np.arange(hi + 1, n)           # right side, partner = 0
        else:
            # the max sits in the centre; choose an index with the smallest partner value
            partner = np.full(n, np.inf)
            valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
            partner[valid] = a[k - np.arange(n)[valid]]
            min_partner = partner.min()
            cand_target = np.where(partner == min_partner)[0]

        # avoid trivial move (source == target)
        if cand_target.size == 0:
            # fallback: pick any index different from i_src
            cand_target = np.delete(np.arange(n), i_src)
        # choose the target with the smallest actual a‑value (helps keep the vector smooth)
        j_tgt = cand_target[int(np.argmin(a[cand_target]))]

        # amount of mass to move
        delta = min(a[i_src] * step_fraction, max_move_abs)
        if delta <= eps:
            # nothing meaningful to move – abort the loop
            break

        # --------------------------------------------------------------
        # Incremental update of the full convolution
        # conv' = conv - 2·δ·shift_i + 2·δ·shift_j    (shift vectors are zero‑padded a)
        # --------------------------------------------------------------
        conv_new = conv.copy()
        conv_new[i_src:i_src + n] -= 2.0 * delta * a
        conv_new[j_tgt:j_tgt + n] += 2.0 * delta * a

        new_max = float(conv_new.max())

        # --------------------------------------------------------------
        # Acceptance test
        # --------------------------------------------------------------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            # very small probability to accept a worsening move (simulated annealing)
            progress = (now - start) / (deadline - start)
            temperature = 1e-8 * (1.0 - progress) + 1e-12
            if np.random.rand() < np.exp(-(new_max - cur_max) / temperature):
                accept = True

        if accept:
            # apply the move permanently
            a[i_src] -= delta
            a[j_tgt] += delta
            conv = conv_new
            cur_max = new_max
            it += 1

            # record best solution so far
            obj = 2.0 * n * cur_max
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                print(f"[iter] it={it:6d}  O={best_obj:.6f}  t={now-start:.1f}s")
        else:
            # occasional random jitter to escape plateaus
            if np.random.rand() < 5e-4:
                # pick two random positions and exchange a tiny amount
                i, j = np.random.choice(n, size=2, replace=False)
                eps_jit = min(a[i], a[j]) * 0.01
                if eps_jit > eps:
                    delta_j = np.random.uniform(-eps_jit, eps_jit)
                    a[i] = max(0.0, a[i] + delta_j)
                    a[j] = max(0.0, a[j] - delta_j)
                    # re‑normalise (sum stays 1 up to round‑off)
                    a_sum = a.sum()
                    if a_sum > 0:
                        a /= a_sum
                    conv = _full_convolution(a)

    # ------------------------------------------------------------------
    # Final polishing with the entropy‑mirror‑descent routine
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_a is not None and time_left > 1.0:
        steps = max(2000, int(min(20000, time_left * 18000)))
        print(f"[polish] mirror descent start – {steps} steps")
        a_refined = _max_refine(best_a, steps=steps, eta_start=0.45)
        conv_ref = _full_convolution(a_refined)
        obj_ref = 2.0 * n * float(conv_ref.max())
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = a_refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
