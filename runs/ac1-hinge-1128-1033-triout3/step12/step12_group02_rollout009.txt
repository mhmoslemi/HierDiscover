"""
Propose a step‑function (list of non‑negative heights) that aims to
minimise
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))² .

Algorithm
---------

* Initialise from the previously found best solution (global
  ``height_sequence_1``); if it is missing we start from a random
  Dirichlet distribution.

* **Stage 1 – soft‑max Adam** (first ≈20 % of the time budget):
  We minimise a smooth surrogate of the max‑convolution by replacing the
  hard max with a temperature‑controlled soft‑max.
  The parameters live in the log‑domain so positivity is guaranteed.
  Adam updates are applied to the logits.

* **Stage 2 – exact entropy‑mirror descent** (remainder of the budget):
  Once the temperature is large we switch to sub‑gradient descent on the
  true maximal convolution entry.  The update
      a ← a·exp(‑η·g)   (g_i = 2·a_{k‑i} for the active index k)
  is an entropy‑mirror step that automatically respects the simplex
  constraint (a stays non‑negative and is re‑normalised each iteration).

* **Stagnation handling** – after a configurable number of iterations
  without improvement a small random jitter is added to the current
  iterate (different scale for Adam / Mirror phases).

* **Polishing** – after the main loop the best vector found is refined
  by a longer mirror‑descent run whose length is limited only by the
  remaining time.

All operations are performed with FFT‑based linear convolutions
(`_full_convolution` / `_full_convolution_two`) which run in
O(n log n).  The routine respects the 1000 s time budget; the final
list of floats is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‐based full convolutions (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two (possibly different length) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max‑entry (polishing phase)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 5000, eta_start: float = 0.2) -> np.ndarray:
    """Run a longer mirror‑descent on the exact max‑convolution entry."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]          # sub‑gradient of the active entry
        # linear decay of the step size
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:                                   # extremely unlikely, keep simplex
            a = np.full_like(a, 1.0 / n)
    return a

            # slice the central n entries
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]

            # chain rule through the softmax(z)
            dot = (g_a * a).sum()
            grad_z = a * (g_a - dot)

            # Adam update on logits
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = lr_start * (1.0 - prog) + lr_end * prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        else:
            # ---- Stage 2: exact entropy‑mirror descent on the max entry ----
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)

            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                g[idx] = 2.0 * a[k - idx]                     # sub‑gradient

            # linear schedule for η inside the mirror phase
            stage_prog = (prog - switch_prog) / (1.0 - switch_prog) if switch_prog else prog
            eta = eta_start * (1.0 - stage_prog) + eta_end * stage_prog

            a = a * np.exp(-eta * g)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full_like(a, 1.0 / n)

            # keep a compatible log‑vector for stagnation handling
            z = np.log(a + eps)
            it += 1

        # ------------------------------------------------------------------
        # Escape from stagnation
        # ------------------------------------------------------------------
        if stagn_counter >= stagn_limit:
            if not mirror_mode:
                z += np.random.normal(scale=jitter_scale_adam, size=n)
                print(f"[{now-start:.1f}s] jitter added (Adam) at iter={it}")
            else:
                perturb = np.random.normal(scale=jitter_scale_mirror, size=n)
                a = np.maximum(a + perturb, 0.0)
                a = a / a.sum()
                z = np.log(a + eps)
                print(f"[{now-start:.1f}s] jitter added (Mirror) at iter={it}")
            stagn_counter = 0

    # ------------------------------------------------------------------
    # Polishing phase – run a longer mirror descent on the best vector
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and best_a is not None:
        # Estimate how many further mirror steps we can afford.
        # Empirically a step costs ~1–2 ms; we are conservative.
        est_step_ms = 1.5
        max_steps = int(remaining * 1000.0 / est_step_ms)
        polish_steps = min(30000, max_steps)
        print(f"[polish] refining with {polish_steps} mirror steps (time left {remaining:.1f}s)")
        refined = _max_refine(best_a, steps=polish_steps, eta_start=eta_start)
        conv_ref = _full_convolution(refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    return final_seq
