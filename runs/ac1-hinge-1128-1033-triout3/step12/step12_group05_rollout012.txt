"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The algorithm works in three stages:

1️⃣  **Random support exploration** – cheap generation of many candidates
    with uniform or perturbed weights on a random subset of indices.
    The best candidate (lowest cheap objective 2·n·max(conv(a,a))) is kept.

2️⃣  **Linear‑programming refinement** – from the current best, an LP is
    solved that maximises the total mass of a new distribution `g`
    under the linear constraints ``(f * g)[k] ≤ M`` for the *most
    critical lags* (the largest entries of the current autocorrelation
    `M = max(f*f)`).  The solution `g` is normalised and mixed with `f`
    (exponential update) which strictly reduces the bound `M`.  This
    step is repeated while time permits.

3️⃣  **Adam‑mirrored gradient descent** – a smooth sub‑gradient of the
    max‑convolution (active‑set version) is used together with Adam‑style
    moments.  The update is performed in the log‑domain (`a ← a·exp(–step)`)
    which automatically stays on the probability simplex.  Momentum,
    a linear learning‑rate decay and occasional Gaussian jitter avoid
    plateaus.

The whole routine respects the supplied time budget (default 1000 s),
returns the best sequence found, and finally evaluates it with the
official `evaluate_sequence` for the true score.
"""

import time
import numpy as np
import cvxpy as cp


# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 2).bit_length())      # next power of two ≥ 2·n‑1
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Cheap proxy for the official score.
    Assumes Σ a = 1 (the vector is kept normalised during the search).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 7,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Return a sub‑gradient of ``max(b)`` w.r.t. ``a``.
    The active set contains the ``max_active`` largest entries of ``b``
    (or all entries within ``rel_tol`` of the maximum).
    """
    n = a.shape[0]
    max_val = b.max()
    if max_active >= b.size:
        active = np.arange(b.size)
    else:
        active = np.argpartition(b, -max_active)[-max_active:]
        mask = b[active] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            active = np.array([int(np.argmax(b))])
        else:
            active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Linear programming step (maximise total mass under active constraints)
# ----------------------------------------------------------------------


def _solve_lp(f: np.ndarray,
              M: float,
              active_idx: np.ndarray,
              max_g_val: float = 1.0) -> np.ndarray | None:
    """
    Solve the LP

        maximise    Σ g_i
        subject to  (f * g)[k] ≤ M   for every k in ``active_idx``
                    0 ≤ g_i ≤ ``max_g_val``

    Returns the optimal vector ``g`` (as a dense NumPy array) or ``None``
    if the problem is infeasible / solver fails.
    """
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = [g <= max_g_val]              # simple upper bound

    # each active convolution constraint is linear in g
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        expr = 0.0
        for i in range(i_start, i_end + 1):
            expr += f[i] * g[k - i]               # (f * g)[k] component
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver='ECOS', warm_start=True, max_iters=3000)
    except Exception:
        return None

    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)                # numerical safety

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------

        best_val = _objective(best_seq)
        print("[fallback] uniform start, objective=", best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Linear‑programming refinement (active‑set LP)
    # ------------------------------------------------------------------
    f = best_seq.copy()
    lp_no_improve = 0
    while time.time() < deadline - 15.0:
        conv_f = _full_convolution(f)
        M = conv_f.max()
        # choose a modest active set (largest lags)
        LP_MAX = 20
        active_idx = np.argpartition(conv_f, -LP_MAX)[-LP_MAX:]

        g = _solve_lp(f, M, active_idx, max_g_val=1.0)
        if g is None:
            print("[LP] solver failed – aborting LP phase")
            break

        # normalise g to sum = 1 (the scaling does not change the bound)
        sum_g = g.sum()
        if sum_g <= 0.0 or not np.isfinite(sum_g):
            print("[LP] degenerate solution – aborting")
            break
        g /= sum_g

        # try a few mixing rates, keep the best improvement
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            f_new = (1.0 - t) * f + t * g
            val_new = _objective(f_new)
            if val_new < best_val:
                best_val = val_new
                best_seq = f_new.copy()
                f = f_new
                improved = True
                lp_no_improve = 0
                print(f"[LP] mix t={t:.2f} new best cheap objective={best_val:.6f}")
                break
        if not improved:
            lp_no_improve += 1
            if lp_no_improve >= 8:
                # no progress for several attempts → exit LP loop
                break

    # ------------------------------------------------------------------
    # 3️⃣  Adam‑mirrored gradient descent refinement
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)           # first moment
    v = np.zeros_like(a)           # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    no_improve = 0
    MAX_NO_IMPROVE = 12000

    while time.time() < deadline - 0.5:
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[adam] iter {it} new best cheap objective={best_val:.6f}")
        else:
            no_improve += 1

        # sub‑gradient from the most active lags
        grad = _subgradient_max(a, conv, max_active=10, rel_tol=1e-6)

        # Adam moments
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # linear learning‑rate decay (from ≈0.4 down to ≈5·10⁻⁵)
        elapsed = time.time() - start
        progress = elapsed / (deadline - start)
        lr = 0.4 * (1.0 - progress) + 5e-5

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # exponentiated (multiplicative) update – stays on the simplex
        log_factor = -step
        log_factor = np.clip(log_factor, -40.0, 40.0)   # avoid overflow
        a = a * np.exp(log_factor)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional Gaussian jitter to escape flat plateaux
        if rng.random() < 0.001 or no_improve >= MAX_NO_IMPROVE:
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
