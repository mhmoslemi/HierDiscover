            m_hat = m / (1 - beta1**step)

            lr = 0.3 * (1 - prog) + 0.005 * prog
            log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        else:
            # Exact mirror descent on max entry
            # Determine max index k
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # subgradient in a (g_i)
            g = np.zeros(n)
            if lo <= hi:
                idx = np.arange(lo, hi+1)
                g[idx] = 2.0 * a[k - idx]   # (note a used as current)
            # step size schedule for mirror

            # simple linear decay

            # update log_a

            # re-normalize to keep sum=1
            # to avoid overflow:

            a_tmp = np.exp(log_a)
            a_tmp /= a_tmp.sum()
            # re-set log_a for next iteration
            log_a = np.log(a_tmp + eps)

            step += 1

        # stagnation: jitter

            # add small random noise to log_a
            jitter = np.random.normal(scale=0.01, size=n)
            log_a += jitter
            stagn_counter = 0

    # after loop, refine best_a with long mirror descent
    refined = _max_refine(best_a, steps=20000, eta_start=0.3)
    # evaluate refined:
    obj_ref = 2.0 * n * _full_convolution(refined).max()

        print("[final] eval", final_score, "obj approx", best_obj)
    except Exception as e:
        print("[final] evaluation error", e)

    return final_seq

"""
Propose a step‑function sequence that (hopefully) beats the previous
best bound 1.508158.  The algorithm consists of three phases:

1.  Smoothed Adam – an Adam optimiser works on a soft‑max surrogate of
    the max‑convolution.  The temperature (α) grows from 1 to a large
    value (≈600) so the surrogate becomes increasingly sharp.

2.  Exact entropy‑mirror descent – once α is large enough we switch to
    a true sub‑gradient descent on the maximal convolution entry.  The
    update is multiplicative (a ← a·exp(‑η g)) which automatically
    preserves non‑negativity.  η decays slowly (0.6 → 0.001).

3.  Long polishing – a final call to the specialised _max_refine()
    routine performs many extra mirror steps with a very small step
    size.

The routine respects the 1000 s budget, keeps the best vector seen,
and finally returns it as a plain list of floats.  If a previous
solution (global variable ``height_sequence_1``) exists it is
interpolated and used as a warm start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full convolutions (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)−1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two (possibly different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max entry (polishing)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 50000,
                eta_start: float = 0.6, eta_end: float = 0.001) -> np.ndarray:
    """Long mirror‑descent polishing of a (a already sums to 1)."""
    n = a.shape[0]
    a = a.copy()
    # work in log‑domain to avoid under‑flow
    log_a = np.log(a + 1e-30)
    for t in range(steps):
        # compute convolution and the index attaining the max
        conv = _full_convolution(np.exp(log_a - np.max(log_a)))
        conv = conv / conv.sum() * 1.0           # rescales – not used, only max index matters
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

        # sub‑gradient of the max entry
        g = np.zeros_like(log_a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * np.exp(log_a[k - idx] - np.max(log_a))

        # linearly decaying step size
        eta = eta_start * (1.0 - t / steps) + eta_end * (t / steps)

        # exponentiated‑gradient update in the log‑domain
        log_a -= eta * g
        # numerical stabilisation (subtract max then renorm)
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum
        log_a = np.log(a + 1e-30)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the given
    time budget (≈1000 s) and prints occasional progress reports.
    """
    np.random.seed(seed)
    start = time.time()
    # small safety margin – the final evaluate_sequence call costs a few ms
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem dimension – 8000 works well and is the size used previously.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a probability vector.
    # ------------------------------------------------------------------
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        try:
            old = np.asarray(old_seq, dtype=float)
            if old.shape[0] != n:
                # linear interpolation onto the new grid
                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                a = np.interp(x_new, x_old, old)
            else:
                a = old.copy()
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 0 else np.full(n, 1.0 / n)
            print("[init] interpolated previous solution")
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # smooth, centrally‑peaked Beta(2,2) density – a good generic start
        i = np.arange(1, n + 1, dtype=np.float64)
        p = q = 2.0
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        a = a / a.sum()
        print("[init] generated Beta(2,2) start")

    # work in log‑space for the Adam phase (positivity guaranteed)
    eps = 1e-30
    log_a = np.log(a + eps)

    # ------------------------------------------------------------------
    # Adam hyper‑parameters (Stage 1)
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    lr_start, lr_end = 0.50, 0.005      # more aggressive learning‑rate schedule
    max_alpha = 600.0                  # final temperature for the soft‑max surrogate
    switch_alpha = 250.0               # when to hand over to exact mirror descent

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_a = None
    best_obj = np.inf
    stagn_counter = 0
    stagn_limit = 4000                 # jitter after ~4 k idle iterations
    jitter_scale_adam = 0.03
    jitter_scale_mirror = 0.008
    it = 0
    mirror_mode = False
    switch_prog = None

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress fraction (0 → 1) over the whole budget
        prog = (now - start) / (deadline - start)

        # temperature schedule (grows super‑linearly)
        alpha = 1.0 + (max_alpha - 1.0) * (prog ** 1.3)

        # ------------------------------------------------------------------
        # Transition from Adam to exact mirror descent
        # ------------------------------------------------------------------
        if not mirror_mode and alpha >= switch_alpha:
            mirror_mode = True
            switch_prog = prog
            # materialise the probability vector from the logits
            a = np.exp(log_a - np.max(log_a))
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else np.full_like(a, 1.0 / n)
            print(f"[mode] entering mirror descent at it={it}, prog={prog:.3f}")

        # ------------------------------------------------------------------
        # Current probability vector (always normalised)
        # ------------------------------------------------------------------
        if not mirror_mode:
            a = np.exp(log_a - np.max(log_a))
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else np.full_like(a, 1.0 / n)

        # ------------------------------------------------------------------
        # Full self‑convolution (used for both stages)
        # ------------------------------------------------------------------
        conv = _full_convolution(a)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv                # exact objective for this a

        # ------------------------------------------------------------------
        # Keep the best vector seen so far
        # ------------------------------------------------------------------
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagn_counter = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}  t={now-start:.1f}s")
        else:
            stagn_counter += 1

        # ------------------------------------------------------------------
        # Gradient / update step
        # ------------------------------------------------------------------
        if not mirror_mode:
            # ----- Stage 1: smoothed Adam on the soft‑max surrogate -----
            # soft‑max weights over the convolution entries
            maxc = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc))
            weights = exp_scaled / exp_scaled.sum()          # Σ weights = 1

            # gradient of the surrogate w.r.t. a
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]        # length‑n slice

            # chain rule through the softmax(logits) → a mapping
            dot = np.dot(g_a, a)
            grad_log = a * (g_a - dot)

            # Adam update on the logits
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = lr_start * (1.0 - prog) + lr_end * prog
            log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        else:
            # ----- Stage 2: exact entropy‑mirror descent on the max entry -----
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)

            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                g[idx] = 2.0 * a[k - idx]                     # sub‑gradient

            # stage‑relative progress (0 → 1) inside the mirror phase
            stage_prog = (prog - switch_prog) / (1.0 - switch_prog) if switch_prog else prog
            eta_start, eta_end = 0.6, 0.001
            eta = eta_start * (1.0 - stage_prog) + eta_end * stage_prog

            # multiplicative update (exponentiated gradient)
            a = a * np.exp(-eta * g)
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else np.full_like(a, 1.0 / n)

            it += 1

        # ------------------------------------------------------------------
        # Escape from stagnation (add a tiny random jitter)
        # ------------------------------------------------------------------
        if stagn_counter >= stagn_limit:
            if not mirror_mode:
                log_a += np.random.normal(scale=jitter_scale_adam, size=n)
                print(f"[reset] jitter added (Adam) at it={it}")
            else:
                perturb = np.random.normal(scale=jitter_scale_mirror, size=n)
                a = np.maximum(a + perturb, 0.0)
                a = a / a.sum()
                print(f"[reset] jitter added (Mirror) at it={it}")
            stagn_counter = 0

    # ------------------------------------------------------------------
    # Polishing phase – a longer mirror‑descent on the best vector found
    # ------------------------------------------------------------------
    if best_a is not None:
        refined = _max_refine(best_a, steps=50000,
                              eta_start=0.45, eta_end=0.001)
        conv_ref = _full_convolution(refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (the wrapper secures the result)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx O={best_obj:.6f})")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
