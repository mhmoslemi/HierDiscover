"""
Proposes a step‑function (non‑negative sequence) that tries to minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the full linear convolution (np.convolve).  The algorithm
consists of three stages:

1️⃣  Generate a diverse pool of initial candidates (uniform, several
    deterministic “U‑shaped’’ profiles, Dirichlet random draws and the
    previously saved solution if it exists).  Their cheap objective
    (2·n·max(conv)) is evaluated; the best candidate is kept.

2️⃣  Refine the best candidate with an Adam–style optimiser on the
    log‑probability (soft‑max) parametrisation.  The gradient is a
    sub‑gradient of the max‑convolution obtained from the currently
    active lag(s).  An exponentially decreasing step‑size, Adam‑moments,
    and occasional Gaussian jitter are used to escape plateaux.

3️⃣  The final best sequence is evaluated with the official
    ``evaluate_sequence`` function and returned.

The whole routine respects a user‑supplied time budget (default 1000 s)
and never returns ``None`` – a uniform distribution is used as a safe
fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2·n‑1
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Cheap objective: 2·n·max(conv(a,a))  under the assumption Σa = 1.
    This is exactly the value returned by ``evaluate_sequence`` when the
    sequence is normalised to sum = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(
    a: np.ndarray,
    conv: np.ndarray,
    max_active: int = 20,
    rel_tol: float = 1e-5,
) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. ``a``.
    The active set contains the ``max_active`` largest entries of ``conv``,
    optionally filtered by a relative tolerance.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_active >= conv.size:
        active = np.arange(conv.size)
    else:
        # indices of the largest ``max_active`` entries (unsorted)
        active = np.argpartition(conv, -max_active)[-max_active:]
        mask = conv[active] >= max_val * (1.0 - rel_tol)
        if np.any(mask):
            active = active[mask]
        else:
            active = np.array([int(np.argmax(conv))])

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Helper: generate a few deterministic “U‑shaped’’ profiles
# ----------------------------------------------------------------------


def _make_u_shaped(n: int, power: float) -> np.ndarray:
    """
    Returns a symmetric distribution that is large at the two ends
    and small in the centre:

        a[i] ∝ |cos(π·i/(n‑1))|**power
    """
    i = np.arange(n, dtype=float)
    vals = np.abs(np.cos(np.pi * i / (n - 1))) ** power
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (sequence) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget,
    performs a fast random‑initialisation phase and then refines the
    best candidate with an Adam optimiser on the simplex.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Build a pool of initial candidates
    # ------------------------------------------------------------------
    candidates: list[np.ndarray] = []

    # (a) Uniform distribution – the theoretical baseline
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # (b) Deterministic “U‑shaped’’ families (different powers)
    for p in (0.5, 1.0, 2.0, 4.0, 8.0):
        candidates.append(_make_u_shaped(n, power=p))

    # (c) Inverted sine (high at the ends, zero in the middle)
    i = np.arange(n, dtype=float)
    inv_sin = 1.0 - np.sin(np.pi * i / (n - 1))
    inv_sin = np.maximum(inv_sin, 0.0)
    if inv_sin.sum() > 0:
        candidates.append(inv_sin / inv_sin.sum())

    # (d) Random Dirichlet draws (different concentration parameters)
    rng = np.random.default_rng()
    for _ in range(6):
        # draw Gamma variables with random shape parameters
        shape_param = rng.uniform(0.2, 2.0)  # controls sparsity
        gamma_samples = rng.gamma(shape=shape_param, scale=1.0, size=n)
        cand = gamma_samples / gamma_samples.sum()
        candidates.append(cand)

    # (e) Previously found solution, if the global variable exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the current length
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            old = np.interp(new_x, old_x, old)
        old = np.maximum(old, 0.0)
        if old.sum() > 0:
            old = old / old.sum()
        else:
            old = np.full(n, 1.0 / n, dtype=np.float64)
        candidates.append(old)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Select the best among the initial pool (cheap objective)
    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = np.inf

    for idx, a in enumerate(candidates):
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # progress output (optional, not too chatty)
        if idx % 5 == 0:
            print(f"[init] candidate {idx:02d} value={val:.6f}")

    if best_seq is None:  # should never happen
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)

    print(f"[init] best initial cheap objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Adam‑style refinement on the probability simplex
    # ------------------------------------------------------------------
    # Initialise logits w such that softmax(w) = best_seq
    eps_small = 1e-30
    w = np.log(np.maximum(best_seq, eps_small))

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    step0 = 0.7               # initial step size (on the logits)
    it = 0
    no_improve = 0
    max_no_improve = 25000   # after this we inject jitter

    while time.time() < deadline - 0.5:
        # ----- soft‑max to obtain a probability vector (sum = 1) -----
        max_w = np.max(w)
        exp_w = np.exp(w - max_w)          # improve numerical stability
        a = exp_w / exp_w.sum()            # a lives on the simplex

        # ----- evaluate cheap objective -----
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[optim] iter {it:06d} new best {best_val:.6f}")
        else:
            no_improve += 1

        # ----- sub‑gradient of the max‑convolution -----
        grad_a = _subgradient_max(
            a,
            conv,
            max_active=30,          # consider a reasonable active set
            rel_tol=1e-5,
        )  # shape (n,)

        # chain rule: ∂f/∂w = (∂f/∂a) * (∂a/∂w) ;  ∂a/∂w = diag(a) - a aᵀ
        # Using the soft‑max trick we can simply multiply by a (the
        # “straight‑through’’ estimator works well in practice):
        grad_w = grad_a * a

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step decays like 1/sqrt(t) – keeps early steps aggressive
        step = step0 / np.sqrt(it)

        w -= step * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional jitter to escape flat plateaux -----
        if np.random.rand() < 0.001 or no_improve >= max_no_improve:
            jitter = np.random.normal(scale=0.001, size=n)
            w += jitter
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[optim] jitter injected at iter {it}")

    # ------------------------------------------------------------------
    # 2️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Ensure we return a plain Python list of floats
    return best_seq.tolist()
