"""
Hybrid optimizer for the self‑convolution max problem.

The routine builds a non‑negative step function whose L1‑norm is 1
(soft‑max parametrisation).  It starts from an anti‑correlated
ordering of random weights (large values paired with small values)
or, if a previous construction ``height_sequence_1`` exists, it
interpolates it.  An Adam optimiser with a smooth‑max temperature that
grows from a modest value to a very large one is run for many
iterations; when the temperature is large the exact max‑gradient is
used (a one‑hot weight).  After the Adam phase a long projected
sub‑gradient / exponentiated‑gradient refinement with the true
maximum weight is performed.  The whole loop respects the 1000 s
budget and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2*len(x)-1, computed via FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k-i}  (cross‑correlation of weight vector w
    with a).  Returned vector has length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1                         # extract indices n‑1 … n‑1+n‑1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(self‑convolution) for a probability vector a."""
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())

    Produce a non‑negative step function (list of floats) that aims to
    minimise `evaluate_sequence`.  The routine runs a long Adam optimisation
    with an increasing smooth‑max temperature, then a thorough projected
    sub‑gradient (exponentiated‑gradient) refinement.
    """

    start_time = time.time()
    # small safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical bound.
    # --------------------------------------------------------------
    n = 12000                     # you may increase this if time permits

    # --------------------------------------------------------------
    # Initialise from a previous construction if available, otherwise
    # build an anti‑correlated start (large with small, small with large).
    # --------------------------------------------------------------

        if old_seq.size > 1:

            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_seq.size)
            a0 = np.interp(new_x, old_x, old_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s < 1e-12:

            else:
                a0 = a0 / s
            print("[init] interpolated previous solution")
        else:
            raise Exception
    except Exception:
        # anti‑correlated random start
        vals = np.random.exponential(scale=1.0, size=n)
        vals.sort()                                 # ascending
        half = n // 2
        a0 = np.empty(n, dtype=float)
        # first half gets the largest values (reversed)
        a0[:half] = vals[half:][::-1]
        # second half gets the smallest values (also reversed to pair)
        a0[half:] = vals[:half][::-1]
        a0 = np.maximum(a0, 0.0)

    best_val = _objective(best_a, n)

            w = exp_vals / exp_vals.sum()

        g = 2.0 * g_raw

        # ---- back‑propagation through soft‑max (logits) ----
        dot_ag = np.dot(a, g)               # scalar
        grad_z = a * (g - dot_ag)           # ∂f/∂z

        # ---- Adam update of logits ----
        iteration += 1

        v = b2 * v + (1.0 - b2) * (grad_z ** 2)

    obj = best_val
