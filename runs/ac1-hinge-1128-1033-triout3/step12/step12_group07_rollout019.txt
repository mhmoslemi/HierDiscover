"""
Optimiser for the “step‑function” upper‑bound problem.

Main ideas
----------
* Build a diverse initial population (beta, exponential, power‑law,
  Dirichlet etc.) and keep the best candidate.
* Run a *linear‑programming blending* phase:
      – given the current f, compute M = max(conv(f,f)).
      – pick the K largest convolution entries (the “active” constraints).
      – solve a tiny LP that maximises Σ g_i under the linear constraints
        f * g ≤ M for the active shifts (g ≥ 0, g ≤ 1000).
      – rescale the LP solution to the required ℓ₁‑norm
        (√(2·n)) and blend it into f.
  This replicates the LP‑based update described in the paper and usually
  pushes the maximum of the convolution down faster than pure gradient
  methods.
* Finish with a short hill‑climbing / simulated‑annealing phase to polish
  the result.

The whole routine respects the 1000 s wall‑clock limit,
keeps all entries non‑negative and automatically normalises the sum
to √(2·n) (the value required by ``evaluate_sequence``).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution helpers (FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact evaluation (the same formula used by ``evaluate_sequence``)
# ----------------------------------------------------------------------
def _evaluate(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max() / (np.sum(a) ** 2))


# ----------------------------------------------------------------------
# Projection onto the probability simplex (used after renormalisation)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators (systematic families + random draws)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-7.5, 2.0)
    logq = rng.uniform(-7.5, 2.0)
    p, q = np.exp(logp), np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    p, q = np.exp(logp), np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    log_r = rng.uniform(-8.0, -0.1)
    r = np.exp(log_r)
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a /= a.sum()
    return a


def _mixture_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    w = rng.random()
    a1 = _random_beta_candidate(n, rng)
    a2 = _random_beta_candidate(n, rng)
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming blending step
# ----------------------------------------------------------------------
def _build_A_ub(f: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    Build the matrix A for the constraints  (f * g)[k] ≤ M  for the
    selected shift indices ``ks``.
    Row r corresponds to shift k = ks[r] and contains the coefficients
    of the linear form in the variables g_0 … g_{n‑1}.
    """
    n = f.shape[0]
    rows = np.zeros((len(ks), n), dtype=np.float64)
    for r, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start > i_end:
            continue
        # f[i_start … i_end] contributes to g[k‑i] (reversed order)
        segment = f[i_start:i_end + 1][::-1]
        j_start = k - i_end
        j_end = k - i_start
        rows[r, j_start:j_end + 1] = segment
    return rows


def _lp_step(f: np.ndarray, M: float, target_sum: float, rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve the tiny LP ``max Σ g_i``  subject to  f * g ≤ M  for the
    currently active shifts.  Return the scaled vector g (or None
    if the LP fails).
    """
    n = f.shape[0]

    # pick the K largest entries of the self‑convolution
    b = _full_convolution(f)
    K = 12                     # a very small active set keeps the LP cheap
    # indices of the K biggest values (ties are broken arbitrarily)
    top_idx = np.argpartition(-b, K)[:K]

    # build the linear constraints (A_ub @ g ≤ M)
    A = _build_A_ub(f, top_idx)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)

    # maximise Σ g_i  ⇒  minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # solve – a short, tiny LP, usually finishes in a few milliseconds
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        return None

    g0 = res.x
    sum_g0 = np.sum(g0)
    if sum_g0 == 0.0:
        return None

    # rescale to the required ℓ₁‑norm (√(2·n))
    g = g0 * (target_sum / sum_g0)
    # ensure feasibility after rescaling (very mild jitter is OK)
    g = np.clip(g, 0.0, 1000.0)
    # final projection onto the simplex with the exact target sum
    g = _project_simplex(g) * target_sum
    return g


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (the final polishing phase)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing hill‑climber (large moves)."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = _evaluate(a)
    no_improve = 0
    IMP_LIMIT = 80000

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = 0.04 * (1.0 - prog) + 1e-9
        max_frac = 0.6 * (1.0 - prog) + 0.001 * prog

        # pick source (proportional to its contribution to the current max)
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            src = rng.choice(n, p=contrib / total)
        else:
            src = rng.integers(n)

        # pick a target that is currently *weak* in the max shift
        tgt_weights = np.ones_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + 1e-12)
        tgt_weights[src] = 0.0
        tgt = rng.choice(n, p=tgt_weights / tgt_weights.sum())

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast convolution update (O(n))
        b_new = b.copy()
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        b_new[2 * tgt]   += delta * delta
        b_new[2 * src]   += delta * delta
        b_new[tgt + src] -= delta * delta

        new_val = _evaluate(a)  # recompute exact value (fast enough)
        # accept if improvement or according to SA temperature
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if rng.random() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMP_LIMIT:
            # re‑initialise around the current best to escape plateaus
            a = best_a.copy()
            a += rng.normal(scale=0.02, size=n)
            a = np.clip(a, 0.0, 1000.0)
            a = _project_simplex(a) * math.sqrt(2 * n)
            b = _full_convolution(a)
            cur_val = _evaluate(a)
            no_improve = 0

    return best_a, best_val

# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,

    """

    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 8000
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 0️⃣  Initial population (systematic + random)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n) * target_sum   # uniform, already respects sum

    # systematic Beta grid

    # random draws from several families

    for _ in range(1200):
        cand = _exponential_candidate(n, rng) * target_sum

    for p in np.linspace(-4.0, 4.0, 21):
        cand = _powerlaw_candidate(n, p) * target_sum
        val = _evaluate(cand)
        if val < best_val:

    for _ in range(800):
        cand = _mixture_beta_candidate(n, rng) * target_sum
        val = _evaluate(cand)

    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    cand *= target_sum
    val = _evaluate(cand)
    if val < best_val:
        best_a, best_val = cand, val

    print(f"[Init] best after exhaustive init = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – LP‑blending iterations (≈30 % of the total budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s

        if g is None:
            # fallback – tiny random perturbation

            g = np.clip(g, 0.0, 1000.0)

        # convex combination (blend)

        cur_val = _evaluate(best_a)
        if cur_val < best_val:

    # ------------------------------------------------------------------
    phase_b_deadline = deadline

        deadline=phase_b_deadline,

        best_a=best_a,
        best_val=best_val,
        rng=rng,
    )

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
