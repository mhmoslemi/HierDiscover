def _greedy_sidon(order, n):
    S = []
    used = set()
    for x in order:
        collision = False
        for s in S:
            if (x + s) in used:
                collision = True
                break
        if (x + x) in used:
            collision = True
        if not collision:
            for s in S:
                used.add(x + s)
            used.add(x + x)
            S.append(x)

"""
Propose a sequence of non‑negative heights that minimizes
evaluate_sequence.  The algorithm consists of three stages:

1.  Build a good initial guess.
    •  Try a uniform vector.
    •  Try a “Sidon‑set” vector – a sparse 0/1 pattern for which
       all pairwise sums i+j are distinct.  Such a pattern reduces the
       maximal self‑convolution considerably.
    •  If a previous solution (height_sequence_1) exists, normalise it.
    The best of these three starts (lowest objective) is kept.

2.  Refine the start with a smooth‑max Adam optimiser.
    •  The true objective is 2·n·max_k (a*a)_k (the sum is forced to 1,
       therefore the scaling‑invariance of the problem can be used).
    •  To obtain a useful gradient we replace max by a log‑sum‑exp
       smoothing:
           L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k) .
       ∇L_α can be written in closed form using a single cross‑correlation
       between the weight vector w_k ∝ exp(α·(a*a)_k) and a.
    •  Adam (adaptive moment estimation) updates a, after which the vector
       is projected back onto the probability simplex (non‑negative,
       Σ a_i = 1).

3.  Keep the best vector seen during the optimisation and finally evaluate
    it with the supplied evaluate_sequence().

The whole routine respects the 1000 s budget and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a * a (real FFT, length 2·len(a)-1)."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(p: np.ndarray, a: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the “valid” cross‑correlation of p (length 2·n‑1) with a (length n):
        g[i] = Σ_j p[i+j] * a[j]   for i = 0 … n‑1 .
    The result is an array of length n.
    """
    fp = np.fft.rfft(p, size)
    fa_rev = np.fft.rfft(a[::-1], size)          # FFT of reversed a
    corr = np.fft.irfft(fp * np.conj(fa_rev), size)
    start = a.shape[0] - 1                       # slice where the “valid” part begins
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _greedy_sidon(order: np.ndarray, n: int):
    """Greedy construction of a Sidon set (all i+j distinct) using the given order."""
    S = []
    used = set()
    for x in order:
        # test whether any sum collides
        bad = (x + x) in used
        if not bad:
            for s in S:
                if (x + s) in used:
                    bad = True
                    break
        if not bad:
            for s in S:
                used.add(x + s)
            used.add(x + x)
            S.append(x)
    return np.array(S, dtype=int)


def _make_sidon_vector(n: int, trials: int = 10) -> np.ndarray:
    """Return the best Sidon‑set vector found after a few random trials."""
    best_vec = None
    best_k = -1
    for _ in range(trials):
        order = np.arange(n)
        np.random.shuffle(order)
        S = _greedy_sidon(order, n)
        k = S.size
        if k > best_k:
            vec = np.zeros(n, dtype=float)
            if k > 0:
                vec[S] = 1.0 / k
            best_vec = vec
            best_k = k
    return best_vec
