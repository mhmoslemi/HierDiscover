"""
Gradient‑based search for a non‑negative step function that minimizes
``evaluate_sequence``.  The objective is 2·n·max(conv(a,a)) when the sequence
is normalized to sum = 1.  The algorithm:

  * starts from a small set of simple parametric candidates
    (power‑law shapes, uniform, random Dirichlet);
  * picks the best of them as a warm‑start;
  * runs an Adam‑style mirror‑descent on a smooth approximation of the
    max‑convolution (soft‑max with an increasing temperature);
  * after each step projects the vector onto the probability simplex
    (keeps all entries non‑negative and sum = 1);
  * stops when the time budget expires and returns the best sequence found.

The code works with any length; a moderate size (2000) gives a good trade‑off
between speed and expressive power.  The final sequence is returned as a plain
Python list, ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·len(a)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a sequence with sum = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _cross_correlation_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    It equals 2·(a ⋆ w) where ⋆ denotes the (non‑reversed) cross‑correlation:
        (a ⋆ w)[i] = Σ_j a_j·w_{i+j}.
    """
    n = a.shape[0]
    # length needed for full cross‑correlation: n + len(w) - 1 = 3n‑2
    size = 1 << ((3 * n - 2).bit_length())
    a_fft = np.fft.rfft(a, size)
    w_fft = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, size)[:n]
    return 2.0 * corr


# ----------------------------------------------------------------------
# Parametric families (used for cheap initialisation)
# ----------------------------------------------------------------------


def _power_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^p, i=0,…,n‑1  (strictly increasing for p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)   # 1 … n
    a = np.power(i, p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def _uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex using a Dirichlet draw."""
    return rng.dirichlet(np.ones(n)).astype(np.float64)

# ----------------------------------------------------------------------

    best_seq = None
    best_val = np.inf

    for idx, cand in enumerate(candidates):
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[init] candidate {idx} gives val={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑style optimisation on the smooth max‑approximation
    # ------------------------------------------------------------------
    if best_seq is None:                     # safety‑net (should never happen)
        best_seq = _uniform_shape(n)
        best_val = _objective(best_seq)

    a = best_seq.copy()
    m = np.zeros_like(a)                     # first moment
    v = np.zeros_like(a)                     # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    base_lr = 0.2
    iteration = 0
    no_improve = 0
    max_no_improve = 3000

    while time.time() < deadline:
        # full convolution and objective
        conv = _full_convolution(a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[optim] iter {iteration:5d} new best {best_val:.6f}")
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # restart from a fresh random point to escape possible stall
                a = _random_simplex(n, rng)
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print(f"[optim] iter {iteration:5d} restart due to stall")
                continue

        # ----- smooth approximation of the max (soft‑max) -----
        # temperature grows slowly so that the gradient focuses more on the true maximum
        beta = 5.0 + 0.01 * iteration
        beta = min(beta, 80.0)
        exp_term = np.exp(beta * (conv - cur_max))           # max → 1, others → <1
        weights = exp_term / exp_term.sum()                  # sum‑to‑1 mask

        # ----- gradient of the weighted sum of convolutions -----
        grad = _cross_correlation_grad(a, weights)

        # ----- Adam update (mirrored onto the simplex) -----
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        # learning‑rate schedule (decays linearly with time)
        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)
        lr = base_lr * (1.0 - progress) + 1e-5
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # multiplicative (exponentiated) update keeps positivity automatically
        log_factor = -step
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)

        # simplex projection
        a_sum = a.sum()
        if a_sum <= 0.0 or not np.isfinite(a_sum):
            a = _uniform_shape(n)
        else:
            a = _project_simplex(a)

        # occasional tiny jitter to shake out of plateaus
        if np.random.rand() < 0.001:
            a += np.random.normal(scale=1e-4, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    # ------------------------------------------------------------------
    # 3️⃣  Final reporting (calls the external evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
