"""
Propose a candidate step‑function by combining several heuristic initialisations
(uniform, Hann window, random simplex, greedy Sidon set) and then refining each
candidate with a fast Adam optimiser on a smooth‑max surrogate of the
self‑convolution, followed by a long projected sub‑gradient descent that
optimises the exact max.  The routine runs inside the 1000 s budget,
keeps the best sequence seen and finally evaluates it with
`evaluate_sequence`.  The optimiser works on the probability simplex
(`∑ a_i = 1, a_i ≥ 0`) so the evaluation is scale‑invariant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of weight vector ``w`` (length 2n‑1) with ``a``.
    Returns a vector of length n:
        g_i = Σ_k w_k * a_{k-i}
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """Return the quantity 2·n·max(self‑convolution) (sum is assumed = 1)."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   time_deadline: float,
                   max_steps: int = 25000,
                   lambda_start: float = 5.0,
                   lambda_end: float = 8000.0,
                   lr: float = 0.04,
                   b1: float = 0.9,
                   b2: float = 0.999,
                   eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimisation with a smooth‑max surrogate of the max‑convolution.
    Returns the best vector found (still on the simplex).
    """
    best_a = a0.copy()
    best_val = _objective(best_a, n)

    # initialise log‑softmax parameters
    z = np.log(best_a + 1e-12) + np.random.normal(scale=0.02, size=n)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    for t_iter in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        # soft‑max to obtain a probability vector
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        conv = _full_convolution(a)
        conv_max = conv.max()
        cur_val = 2.0 * n * conv_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            # printing inside the optimiser is cheap compared to the whole run
            print(f"[Adam] step {t_iter:06d} new best {best_val:.9f}")

        # temperature schedule (linear in iteration count)
        prog = t_iter / max_steps
        lam = lambda_start * (1.0 - prog) + lambda_end * prog

        if lam > 2000.0:
            # exact max – one‑hot weight vector
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            # smooth‑max
            exp_vals = np.exp(lam * (conv - conv_max))
            w = exp_vals / exp_vals.sum()

        # gradient of the (smooth) max w.r.t. a  (2·cross‑corr)
        g_raw = _cross_corr(w, a)
        g = 2.0 * g_raw

        # back‑prop through the soft‑max
        dot_ag = np.dot(a, g)
        grad_z = a * (g - dot_ag)

        # Adam updates
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z ** 2)
        m_hat = m / (1.0 - b1 ** t_iter)
        v_hat = v / (1.0 - b2 ** t_iter)

def _greedy_sidon_positions(n: int) -> np.ndarray:

        conflict = False
        for p in pos:

        # (the theoretical maximum is ≈ sqrt(2n))
        if len(pos) * (len(pos) + 1) // 2 > 2 * n:
            break
    return np.array(pos, dtype=int)

"""
Propose a candidate step‑function by mixing several simple initialisations
(uniform, Hann window, random Dirichlet, greedy Sidon set, optional previous
solution) and then refining each candidate with:

1. An Adam optimiser on a smooth‑max surrogate of the self‑convolution
   (temperature λ grows from a few to several thousand).  This quickly moves
   the vector into a region where the maximal convolution is already small.

2. A long projected sub‑gradient descent that uses the *exact* max‑convolution
   (a one‑hot weight vector).  The update is
       a ← Π_{simplex}(a – η·g) ,   g_i = 2·a_{k_max‑i},
   where Π_{simplex} projects onto the probability simplex.
   The step‑size η is increased after successful moves and shrunk otherwise.

Both phases respect the global 1000 s budget.  The best vector found
(over all initialisations) is finally evaluated with ``evaluate_sequence``
and returned.  All vectors are kept on the simplex (∑a_i = 1, a_i ≥ 0) so the
objective is scale‑invariant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a real vector using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of weight vector ``w`` (length 2n‑1) with ``a``
    (length n).  Returns a vector of length n:
        g_i = Σ_k w_k·a_{k‑i} .
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(self‑convolution) for a probability vector a."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   max_steps: int = 25000,
                   lambda_start: float = 5.0,
                   lambda_end: float = 8000.0,
                   lr: float = 0.04,
                   b1: float = 0.9,
                   b2: float = 0.999,
                   eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimisation on a smooth‑max surrogate of the maximum convolution.
    Operates on log‑softmax parameters ``z`` (so the iterate is always on the
    probability simplex).  Returns the best vector seen.
    """
    best_a = a0.copy()
    best_val = _objective(best_a, n)

    # start from a slightly perturbed log‑softmax representation
    z = np.log(best_a + 1e-12) + np.random.normal(scale=0.02, size=n)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    for t in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # probabilities from logits
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # evaluate smooth objective
        conv = _full_convolution(a)
        conv_max = conv.max()
        cur_val = 2.0 * n * conv_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam] step {t:06d} new best {best_val:.9f}")

        # temperature schedule (linear)
        prog = t / max_steps
        lam = lambda_start * (1.0 - prog) + lambda_end * prog

        if lam > 2000.0:
            # exact max: one‑hot weight
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            # smooth max via soft‑max over convolution entries
            exp_vals = np.exp(lam * (conv - conv_max))
            w = exp_vals / exp_vals.sum()

        # gradient of (smooth) max w.r.t. a  (2·cross‑corr)
        g_raw = _cross_corr(w, a)
        g = 2.0 * g_raw

        # back‑prop through the soft‑max
        dot_ag = np.dot(a, g)
        grad_z = a * (g - dot_ag)

        # Adam update
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a

    while time.time() < deadline and global_best_val > 1.503:
        a = global_best.copy()
        i, j = np.random.randint(n, size=2)
        if i == j or a[i] < 1e-12:
            continue
        delta = min(a[i], 0.01) * np.random.rand()
        a[i] -= delta
        a[j] += delta
        a = _project_simplex(a)
        val = _objective(a, n)
        if val < global_best_val - 1e-12:
            global_best_val = val
            global_best = a.copy()
            print(f"[fine] improved to {global_best_val:.9f}")

    # ------------------------------------------------------------------
    # Final secure evaluation (the only place we call evaluate_sequence)
    # ------------------------------------------------------------------
    if global_best is None:
        # fallback – should never happen
        global_best = np.full(n, 1.0 / n)

    try:
        final_score = evaluate_sequence(global_best.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best.tolist()
