The code builds a non‑negative step‑function a (∑a = 1) and minimises
    2·n·max_k (a*a)_k
by repeatedly
1. computing the full linear convolution a*a,
2. forming a smooth approximation of the max using a soft‑max
   (log‑sum‑exp) with a temperature α that grows during the run,
3. evaluating the gradient of this smooth objective via a second
   convolution (the gradient of max is a weighted sum of the basic
   sub‑gradient 2·a_{k‑i}),
4. applying the Adam optimiser and projecting back onto the probability
   simplex,
5. injecting occasional Gaussian jitter when progress stalls.

The routine respects a 1000 s time budget and finally returns the best
sequence found (as a list of floats) together with a printed score from
evaluate_sequence().
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using real FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w≥0, Σw=1}.
    Reference: https://arxiv.org/abs/1101.6081 (Algorithm 1)
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def smoothmax_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max (log‑sum‑exp) of the self‑convolution.
    Returns (grad, conv) where conv = a*a (full linear convolution).
    """
    conv = full_convolution(a, a)                     # length 2·n-1
    max_val = conv.max()
    # soft‑max weights, stabilised by subtracting the maximum
    shifted = alpha * (conv - max_val)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()               # sum_i w_i = 1

    # gradient = 2 * Σ_i w_i * a_{i‑j}  (j is the variable index)
    # This is a correlation: g = 2 * (w_rev * a) sliced appropriately.
    w_rev = w[::-1]
    d = full_convolution(w_rev, a)                    # length (2n-1)+n-1
    L = conv.shape[0]                                 # = 2·n‑1
    idx = L - 1 - np.arange(a.shape[0])               # mapping to g_i
    grad = 2.0 * d[idx]
    return grad, conv


def objective_value(a: np.ndarray, n: int) -> float:
    """
    Returns the exact objective used by evaluate_sequence for a vector with
    sum(a)=1, i.e. 2·n·max(a*a).
    """
    conv = full_convolution(a, a)
    return 2.0 * n * conv.max()

# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:

    Search for a non‑negative step sequence that minimises ``evaluate_sequence``.
    The function respects the given time budget (default 1000 s) and returns
    the best sequence discovered before the time expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise – try to reuse a previously stored solution if available.
    # ------------------------------------------------------------------

        old_len = old.shape[0]
        x_old = np.linspace(0.0, 1.0, old_len)
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum
        print("[init] interpolated previous solution (len:", old_len,
              "→", n, ")")
    except Exception:
        # uniform start with a tiny random perturbation
        a = np.full(n, 1.0 / n) + np.random.normal(scale=0.001, size=n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        print("[init] uniform start, n =", n)

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                 # initial learning rate
    lr_decay = 0.9999

    # smooth‑max temperature schedule
    alpha = 10.0
    alpha_max = 50000.0
    alpha_inc = 1.001

    # bookkeeping
    best_a = a.copy()
    best_val = objective_value(a, n)
    print("[init] objective =", best_val)

    stagnation = 0
    max_stagn = 30000   # after this many non‑improving steps we jitter

    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) gradient of smooth max and current convolution
        # --------------------------------------------------------------
        grad, conv = smoothmax_gradient(a, alpha)

        # --------------------------------------------------------------
        # 2) Adam update + simplex projection
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = project_simplex(a_candidate)

        # --------------------------------------------------------------
        # 3) evaluate (fast) objective
        # --------------------------------------------------------------
        cur_val = objective_value(a_candidate, n)

        # --------------------------------------------------------------
        # 4) acceptance / bookkeeping
        # --------------------------------------------------------------
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_candidate.copy()
            print(f"[iter {iteration:06d}] new best = {best_val:.9f} "
                  f"(time {time.time() - start_time:.1f}s)")
            stagnation = 0
        else:
            stagnation += 1

        # Accept the candidate if it improves the *current* objective;
        # otherwise keep the old a (with a tiny probability to escape).
        cur_obj = objective_value(a, n)
        if cur_val < cur_obj:
            a = a_candidate
        else:
            if np.random.rand() < 0.001:       # rare uphill move
                a = a_candidate

        # --------------------------------------------------------------
        # 5) schedule updates
        # --------------------------------------------------------------
        if alpha < alpha_max:
            alpha *= alpha_inc
        lr = max(lr * lr_decay, 1e-5)

        # --------------------------------------------------------------
        # 6) stagnation handling – inject Gaussian jitter
        # --------------------------------------------------------------
        if stagnation > max_stagn:
            noise = np.random.normal(scale=0.02, size=n)
            a = best_a + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0
            print(f"[reset] jitter at iter {iteration} (time {time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # Final secure evaluation (only one call to the provided function)
    # ------------------------------------------------------------------

        print("[final] reported score =", final_score)
    except Exception as exc:            # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
