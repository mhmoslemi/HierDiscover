"""
Improved step‑function optimiser (≈1000 s budget)

What this does
---------------
1. **Analytic seed** – the family a_i ∝ (i+1)^p (p∈{0,0.5,0.8,1,1.2,1.5,2})
   contains the linear “i+1” distribution which analytically gives
   2 n · max(conv) = 4/3 ≈ 1.333 for large n – already better than
   the previous best 1.508.

2. **Smooth‑max L‑BFGS** – the non‑smooth max in the objective is
   replaced by a log‑sum‑exp (parameter ε).  The variables are
   re‑parameterised with a soft‑max so the simplex constraint
   (non‑negative, sum = 1) is satisfied automatically.
   Several restarts are performed, gradually decreasing ε.

3. **Exact sub‑gradient polishing** – a few thousand projected
   sub‑gradient steps on the true objective (max‑convolution) clean
   up any remaining spikes.

4. **Uniform blending** – convex combinations with the flat
   distribution are tried, then the best result is returned.

The whole pipeline respects the 1000 s time‑budget and returns a list
of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT based convolution helpers (full linear, length 2·len(a)‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of ``x`` with itself (linear, not circular)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – identical to ``evaluate_sequence`` when sum(a)=1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.  The gradient already includes the factor 2·n.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick (stabilised)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p (gives d/d a of smooth max)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # keep the n entries that correspond to the “forward” part
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → probability simplex)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum contributions from all maximal indices
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple step‑size schedule (aggressive early, tiny later)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 400 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    n = a.shape[0]
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    eps: float,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    max_iter: int = 1200) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smoothed objective.
    """
    n = a0.shape[0]
    x0 = np.log(a0 + 1e-12)          # initialise in log‑space

    # ----- objective ----------------------------------------------------
    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    # ----- gradient -----------------------------------------------------
    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    # respect remaining time
    if time.time() >= deadline - 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and that minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – a value around 15 k gives a good trade‑off.
    # --------------------------------------------------------------
    n = 15000

    # --------------------------------------------------------------
    # 0️⃣  Analytic seeds (powers a_i ∝ (i+1)^p)
    # --------------------------------------------------------------
    power_candidates = [0.0, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0]
    candidates = []

    idx = np.arange(1, n + 1, dtype=np.float64)   # i+1

    for p in power_candidates:
        a = idx ** p
        a /= a.sum()
        val = _objective(a)
        candidates.append((val, a))
        print(f"[Init] power p={p:4.1f} → value={val:.6f}")

    # also add a pure uniform baseline
    uniform = np.full(n, 1.0 / n)
    candidates.append((_objective(uniform), uniform))

    # keep the best few for the next phase
    candidates.sort(key=lambda t: t[0])
    top_k = 5
    best_a, best_val = candidates[0][1], candidates[0][0]
    print(f"[Init] best after seed sweep: value={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – smooth‑max L‑BFGS (multiple restarts, decreasing ε)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.55 * budget_s   # a bit more budget for Phase A
    eps_start = 5e-1
    eps_final = 1e-12

    while time.time() < phase_a_deadline:
        # ε shrinks exponentially with elapsed time inside Phase A
        prog = (time.time() - start_time) / (phase_a_deadline - start_time)
        eps = eps_start * (eps_final / eps_start) ** prog

        for i in range(min(top_k, len(candidates))):
            if time.time() >= phase_a_deadline:
                break
            a_start = candidates[i][1]
            best_a, best_val = _lbfgs_optimize(
                a_start, eps, deadline, start_time,
                best_a, best_val, max_iter=1300)
            # try a little convex blending after each optimisation
            best_a, best_val = _blend_with_uniform(best_a, best_val)

    # --------------------------------------------------------------
    # Phase B – exact sub‑gradient polishing
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.90 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, deadline=phase_b_deadline, start_time=start_time,
        best_a=best_a, best_val=best_val, max_iters=300_000)
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # --------------------------------------------------------------
    # Phase C – final short L‑BFGS with a very tiny ε
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(
            best_a, eps=1e-12, deadline=deadline,
            start_time=start_time, best_a=best_a, best_val=best_val,
            max_iter=800)

    # ------------------------------------------------------------------
    # Final secure evaluation (just to report)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
