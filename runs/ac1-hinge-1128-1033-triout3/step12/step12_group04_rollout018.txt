    n = 8000

        size = 1 << ((2*n - 1 - 1).bit_length())
        f = np.fft.rfft(arr.astype(np.float64), size)
        conv = np.fft.irfft(f * f, size)[:2*n-1]
        return np.rint(conv).astype(np.int32)

    best_score = float('inf')
    best_vec = None

    # function to evaluate set S
    def evaluate_set(S):

        m = int(conv.max())
        k = len(S)
        score = 2.0 * n * m / (k * k)
        return score, m, sel, conv

    # main loop trying different k values
    while time.time() < deadline:
        # pick a size k aiming at high density
        k = np.random.randint(180, 215)  # inclusive 214

"""
Propose a step‑function that beats the previous 1.5083 bound.

The algorithm searches for a *binary* mask S ⊂ {0,…,n‑1} of size k.
All selected indices receive the same weight 1/k, so the sum of the
sequence is 1 and the evaluation function reduces to

    score = 2·n·max_k (s*s)_k / k² ,

where s is the 0‑1 indicator of S and (s*s) is its linear
convolution (ordered pair count).  The goal is to keep the
maximum entry of this convolution small.
For a given k the target maximal count that guarantees a score
≤ 1.503 is

    max_allowed = floor(1.503·k²/(2·n))

(e.g. max_allowed=3 for k≈180, =4 for k≈210).

We perform a stochastic hill‑climbing search:
* pick a random size k in a promising range,
* initialise a random mask,
* repeatedly try to swap one selected index for an unselected one,
  accepting the swap if it lowers the current maximum count (or
  occasionally accepts an equal move to escape plateaus),
* keep the best mask ever seen.

Convolution of the binary mask is computed with a real FFT; the
result is rounded to the nearest integer because the true values
are integral.  The whole routine respects the 1000 s budget and
returns the best sequence it found (as a list of floats).  A final
call to the provided ``evaluate_sequence`` reports the official
score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – deliberately defined at top level (no closures).
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_counts(binary: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a 0‑1 array with itself.
    Returns an integer array of length 2·len(binary)‑1 containing the
    ordered pair counts (self‑pairs count 1, i≠j pairs count 2).
    """
    n = binary.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    f = np.fft.rfft(binary.astype(np.float64), size)
    conv = np.fft.irfft(f * f, size)[:out_len]
    # rounding is safe because the true result is integral
    return np.rint(conv).astype(np.int32)


def _score_from_counts(max_count: int, k: int, n: int = 8000) -> float:
    """Score = 2·n·max_count / k²  (sum of coefficients equals 1)."""
    return 2.0 * n * max_count / (k * k)


def _initial_mask(k: int, n: int = 8000) -> np.ndarray:
    """Return a random 0‑1 mask of length n with exactly k ones."""
    mask = np.zeros(n, dtype=np.int8)
    mask[np.random.choice(n, size=k, replace=False)] = 1
    return mask

# Main entry point requested by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence`` under the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    n = 8000                     # length of the step function
    best_score = float("inf")    # lowest score seen so far
    best_vec = None              # corresponding float vector (weights)

    # ------------------------------------------------------------------
    # Main outer loop – pick a size k and try to improve a random mask.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Choose a size that gives a realistic chance of beating the target.
        # 180 – 214 are good compromises (k≈210 yields max_allowed=4).
        k = np.random.randint(180, 215)          # 180 ≤ k < 215
        # The maximal ordered count that still satisfies the 1.503 target.
        max_allowed = int(np.floor(1.503 * (k * k) / (2.0 * n)))
        max_allowed = max(1, max_allowed)

        # Random initial mask and its convolution.
        mask = _initial_mask(k, n)
        conv = _full_counts(mask)
        cur_max = int(conv.max())
        cur_score = _score_from_counts(cur_max, k, n)

        # Record the initial solution if it is the best so far.
        if cur_score < best_score:
            best_score = cur_score
            best_vec = mask.astype(np.float64) / k
            print(f"[init] k={k} max={cur_max} score={cur_score:.6f}")

        # If the random start already meets the target, stop early.
        if cur_max <= max_allowed and cur_score <= 1.503:
            best_score = cur_score
            best_vec = mask.astype(np.float64) / k
            print("[early] found satisfying solution, exiting")
            break

        # --------------------------------------------------------------
        # Hill‑climbing: random swaps of one selected index for an unselected.
        # --------------------------------------------------------------
        # Constrain the inner loop to a reasonable number of attempts.
        inner_limit = 25000
        it = 0
        while time.time() < deadline and it < inner_limit:
            it += 1

            # pick a random index that is currently selected
            selected = np.flatnonzero(mask)
            out_idx = np.random.choice(selected)

            # propose a random new index (must be currently zero)
            new_idx = np.random.randint(0, n)
            while mask[new_idx] == 1:
                new_idx = np.random.randint(0, n)

            # perform the swap on a copy (fast, uses FFT for the new conv)
            mask_new = mask.copy()
            mask_new[out_idx] = 0
            mask_new[new_idx] = 1
            conv_new = _full_counts(mask_new)
            new_max = int(conv_new.max())

            # simple acceptance rule – accept any improvement,
            # occasionally accept equal‑max moves to explore.
            accept = False
            if new_max < cur_max:
                accept = True
            elif new_max == cur_max and np.random.rand() < 0.01:
                accept = True

            if accept:
                mask = mask_new
                conv = conv_new
                cur_max = new_max
                cur_score = _score_from_counts(cur_max, k, n)

                # keep the globally best mask
                if cur_score < best_score:
                    best_score = cur_score
                    best_vec = mask.astype(np.float64) / k
                    print(
                        f"[improve] iter {it:05d} k={k} max={cur_max} "
                        f"score={cur_score:.6f} (elapsed {time.time()-start_time:.1f}s)"
                    )

                # stop early if we already satisfy the target bound
                if cur_max <= max_allowed and cur_score <= 1.503:
                    print("[found] target satisfied within inner loop")
                    break

        # If a satisfactory solution has been found, exit the outer loop.
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: uniform vector (never worse than 2.0) in the extremely
    # unlikely case that the search never produced a mask.
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final secure evaluation (single call as required).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_vec.tolist())
        print("[final] reported score =", final_score)
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_vec.tolist()
