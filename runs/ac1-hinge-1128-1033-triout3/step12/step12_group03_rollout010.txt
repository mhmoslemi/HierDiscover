def _cross_corr(w, a):
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1  # = 3n-2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]

"""
Hybrid optimiser for the self‑convolution supremum problem.

Main ideas
---------
* Initialise from the previously found solution (if available) and
  down‑sample it to a modest size (default n=2000).  A smaller problem
  is cheap to solve with a linear programme while still carrying the
  same “shape’’ information.
* **LP boost** – given a current probability vector `a`, compute
  `M = max(a * a)` and solve the linear programme
      maximise   Σ b_j
      subject to A(a)·b ≤ M ,   b ≥ 0,
  where `A(a)` is the full convolution matrix (each column `j` contains the
  coefficient `a_j` on a band of length `n`).  The optimum sum‐weight
  `Σ b_j` is ≥ 1, therefore after normalising `g = b / Σb` we can mix
  `a ← (1‑t)·a + t·g` (t≈0.2) and strictly lower the maximum
  convolution.
* Repeat the LP‑boost a few dozen times (or until the time budget runs
  out).  This phase drives the objective close to the theoretical limit
  ≈ 1.5.
* **Fine‑tuning** – run a longer Adam optimisation on a smooth‑max
  approximation of the peak, finishing with a projected sub‑gradient
  sweep.  Both steps respect the probability simplex and improve the
  final bound further.
* The whole routine respects the 1000 s budget and finally returns the
  best sequence found (as a plain list of floats).

The evaluation function `evaluate_sequence` is called only once at the
end, guaranteeing a valid return even if the optimisation is cut short.
"""
import time
import numpy as np
import scipy.optimize
import scipy.sparse


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself using FFT."""
    n = x.shape[0]
    length = 2 * n - 1
    size = _next_pow2(length)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:length]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of weight vector `w` (length 2n‑1) with `a`
    (length n).  Returns a vector of length n containing
    Σ_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1  # = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(self‑convolution) for a probability vector."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------

    max_lp_iters = 25

        A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        c = -np.ones(n)

        exp_z = np.exp(z - max_z)

        conv_max = conv.max()
        cur_val = 2.0 * n * conv_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()

            stagnation += 1

        if lam > 2000.0:
