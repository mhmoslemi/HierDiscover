def simplex_projection(v):

    n = len(v)

"""
Propose a step‐function (non‑negative sequence) that minimizes
``evaluate_sequence``.  

The search consists of three phases:

1️⃣  **Sidon‑like support construction** – greedily build a set of indices
    whose pairwise sums appear only a few times (≤ ``max_allowed``).  The
    resulting binary mask, after normalising the coefficients, already
    yields a cheap objective far below the uniform benchmark (2.0).

2️⃣  **Adam‑based refinement on the cheap objective** – with the
    sequence fixed to sum to 1 we minimise  
    ``2·n·max(conv(a,a))`` (the value returned by ``evaluate_sequence`` when
    the sum is 1).  The sub‑gradient of the max‑convolution is obtained
    from the most active lags (an “active‑set” of the largest convolution
    entries).  Adam with bias correction and a linear step‑size decay
    stays on the probability simplex via a projection after each update.

3️⃣  **Final evaluation** – the best sequence found is passed to the
    official ``evaluate_sequence`` function and returned.

The routine respects the supplied time budget (default 1000 s) and works
with a single CPU only.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    # size = next power of two ≥ 2·n‑1
    size = 1 << ((2 * n - 2).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact cheap objective 2·n·max(conv(a,a)) assuming Σa = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 7,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Return a sub‑gradient of max(b) w.r.t. a.
    The active set contains the ``max_active`` largest entries of b,
    filtered by a relative tolerance.
    """
    n = a.shape[0]
    max_val = b.max()
    if max_active >= b.size:
        active = np.arange(b.size)
    else:
        active = np.argpartition(b, -max_active)[-max_active:]
        mask = b[active] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            active = np.array([int(np.argmax(b))])
        else:
            active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x_i ≥ 0)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Greedy Sidon‑like support construction
# ----------------------------------------------------------------------


def _greedy_sidonic_support(n: int,
                            max_allowed: int = 4,
                            rng: np.random.Generator = np.random.default_rng()
                            ) -> np.ndarray:
    """
    Build a set of indices S ⊂ {0,…,n‑1} such that each (ordered) sum
    i+j (0 ≤ i,j < n) appears at most ``max_allowed`` times.
    Returns the binary mask (float array) with equal weight on the chosen
    positions, normalised to sum = 1.  The routine is stochastic – a new
    random order is tried each call.
    """
    cnt = np.zeros(2 * n - 1, dtype=np.int32)   # ordered‑pair counters
    S = []
    order = rng.permutation(n)
    for idx in order:
        # check whether adding idx would violate the bound
        ok = True
        self_sum = idx + idx
        if cnt[self_sum] + 1 > max_allowed:
            ok = False
        else:
            for i in S:
                s = idx + i
                if cnt[s] + 2 > max_allowed:
                    ok = False
                    break
        if not ok:
            continue
        # accept idx
        S.append(idx)
        cnt[self_sum] += 1
        for i in S[:-1]:           # all previous elements
            s = idx + i
            cnt[s] += 2
    if len(S) == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[S] = 1.0 / len(S)
    return a

# Main entry point required by the evaluation framework

    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin for final call

    # Problem size – a compromise between expressivity and speed.
    n = 8000

    best_seq: np.ndarray | None = None

    # ------------------------------------------------------------------

    try:
