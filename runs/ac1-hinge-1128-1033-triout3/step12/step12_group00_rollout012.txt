def adam_optimizer(a_init, deadline):
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_val = _objective(a)
    best_a = a.copy()
    while time.time() < deadline:
        t += 1
        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.flatnonzero(b == max_b)   # vector of indices
        # For speed just pick one or random subset
        k = k_maxes[0]   # pick first
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]
        # adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.1 * m_hat / (np.sqrt(v_hat) + eps)   # 0.1 initial LR
        a = a - step
        a = _project_simplex(a)

        # Occasionally evaluate full objective (every 20 it)
        if t % 20 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[Adam] new best {best_val:.6f}")

    return best_a, best_val

def _spectral_equalize(a):
    n = a.size
    L = 1 << ((2*n - 1)-1).bit_length()
    A = np.fft.rfft(a, L)
    # preserve DC component
    mag_const = np.median(np.abs(A[1:]))  # Could also use mean

    phases = np.angle(A[1:])

    a_eq = np.fft.irfft(A_new, L)[:n]

    if a_eq.sum() == 0:
        a_eq = np.full(n, 1.0 / n)
    else:
        a_eq = a_eq / a_eq.sum()
    return a_eq

def _random_spectral_candidate(a, rng):

    L = 1 << ((2*n-1)-1).bit_length()

We combine several ideas that complement the previous pipeline:

*   A broad random initialise (uniform, Dirichlet, Beta families and the

    spikes.

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _smooth_max_and_grad(a: np.ndarray, eps: float, n: int):
    """
    Smooth approximation of max(conv(a,a)) using log‑sum‑exp.
    Returns the smooth value and the gradient of 2·n·smooth_max w.r.t. a.
    """
    b = _full_convolution(a)            # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()         # shape (2n‑1,)

    # convolution of a with reversed p (same length 2n‑1)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)
    segment = conv_ab[n - 1:2 * n - 1]   # n entries that correspond to “forward” part
    grad_a = 4.0 * n * segment[::-1]    # includes the factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random Beta‑family candidate (cheap initial sweep)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ≈ 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Spectral equalisation – make all FFT magnitudes (except DC) equal
# ----------------------------------------------------------------------
def _spectral_equalise(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force all non‑DC FFT magnitudes to a common value while keeping phases."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, L)

    # target magnitude = sqrt(mean of |A|^2) over non‑DC components
    target_mag = np.sqrt(np.mean(np.abs(A[1:]) ** 2))

    # randomise phases for the non‑DC frequencies
    phases = rng.uniform(0.0, 2.0 * np.pi, size=A[1:].shape)
    A_new = A.copy()
    A_new[0] = A[0]                     # keep the DC term (sum = 1)
    A_new[1:] = target_mag * np.exp(1j * phases)

    a_new = np.fft.irfft(A_new, L)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full(n, 1.0 / n)
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 1500) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smoothed objective.
    """
    x0 = np.log(a0 + 1e-12)                 # initialise in log‑space

    # ----- objective ----------------------------------------------------
    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps, n)
        return 2.0 * n * smooth_val

    # ----- gradient -----------------------------------------------------
    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps, n)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    # respect remaining time
    if time.time() >= deadline - 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt, n)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    cur_val = _objective(a, n)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.flatnonzero(b == max_b)

        # sub‑gradient: sum of contributions from all maximal diagonals
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple linear schedule, aggressive early, tiny later
        progress = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new, n)

        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 400 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (projected onto simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   max_iters: int = 60_000) -> tuple[np.ndarray, float]:
    """
    Adam (projected) optimiser using the true sub‑gradient of max(conv(a,a)).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    cur_best_val = best_val
    cur_best_a = best_a.copy()

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        t += 1
        b = _full_convolution(a)
        max_b = b.max()
        # pick one of the maximal indices (any works for a sub‑gradient)
        k = np.flatnonzero(b == max_b)[0]

        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = 0.15 * m_hat / (np.sqrt(v_hat) + eps)      # learning‑rate ≈0.15
        a = a - step
        a = _project_simplex(a)

        # occasional evaluation (every 30 steps)
        if it % 30 == 0:
            val = _objective(a, n)
            if val < cur_best_val:
                cur_best_val, cur_best_a = val, a.copy()
                print(f"[Adam] new best = {cur_best_val:.6f}")

    return cur_best_a, cur_best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list (sum ≈1) that minimises ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 12 k entries give a rich search space while staying fast.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # 0️⃣  Initial sweep (cheap candidates)
    # ------------------------------------------------------------------
    candidates = []

    # 0.1  Re‑use previous solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        candidates.append((_objective(a0, n), a0))
        print(f"[Init] interpolated previous solution, value={candidates[-1][0]:.6f}")
    except Exception:
        pass

    # 0.2  Uniform baseline
    uniform = np.full(n, 1.0 / n)
    candidates.append((_objective(uniform, n), uniform))

    # 0.3  Random Beta families
    for _ in range(200):
        cand = _random_beta_candidate(n, rng)
        candidates.append((_objective(cand, n), cand))

    # 0.4  Dirichlet draws with several concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(5):
            cand = rng.dirichlet(np.full(n, alpha))
            candidates.append((_objective(cand, n), cand))

    # keep the top few seeds for the next phases
    candidates.sort(key=lambda t: t[0])
    top_k = min(6, len(candidates))
    best_a, best_val = candidates[0][1].copy(), candidates[0][0]
    print(f"[Init] best after sweep: value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Spectral equalisation + Adam optimiser (≈30 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.33 * budget_s
    for i in range(top_k):
        if time.time() >= phase_a_deadline:
            break
        seed_vec = candidates[i][1]

        # spectral equalisation with random phases
        cand_eq = _spectral_equalise(seed_vec, rng)

        # run Adam optimiser for a slice of the remaining time
        slice_deadline = min(deadline,
                             phase_a_deadline + (i + 1) * (budget_s * 0.03))
        cand_opt, val_opt = _adam_optimize(
            cand_eq, n, slice_deadline, start_time, best_a, best_val,
            max_iters=30_000)
        if val_opt < best_val:
            best_val, best_a = val_opt, cand_opt.copy()
            print(f"[PhaseA] improved best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – L‑BFGS on the smooth‑max surrogate (≈20 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.60 * budget_s
    eps_start, eps_final = 5e-1, 1e-12
    while time.time() < phase_b_deadline:
        # schedule ε geometrically
        prog = (time.time() - start_time) / (phase_b_deadline - start_time)
        eps = eps_start * (eps_final / eps_start) ** prog
        best_a, best_val = _lbfgs_optimize(best_a, n,
                                            deadline,
                                            start_time,
                                            best_a, best_val,
                                            eps=eps,
                                            max_iter=1200)
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Exact sub‑gradient polishing (≈30 % of remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_polish(best_a, n,
                                            deadline=deadline,
                                            start_time=start_time,
                                            best_a=best_a,
                                            best_val=best_val,
                                            max_iters=300_000)

    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Final short L‑BFGS with a very tiny ε (cleanup)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(best_a, n,
                                            deadline, start_time,
                                            best_a, best_val,
                                            eps=1e-12, max_iter=800)

    # ------------------------------------------------------------------
    # Final evaluation through the protected function
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
