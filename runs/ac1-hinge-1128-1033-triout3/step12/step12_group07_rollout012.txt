"""
Improved optimizer for the step‑function attack.

Changes compared with the reference implementation:
* A finer systematic search over Beta‑distribution shapes – a 60×60
  grid on log‑parameters (≈3 600 candidates) – to obtain a better
  starting point.
* After the exponential‑gradient polishing we run a deterministic
  projected‑gradient descent with backtracking line‑search on the exact
  max‑convolution objective.  This often squeezes a few more points out
  of the remaining gap.
* A short “smoothing” phase that averages each entry with its neighbours,
  re‑normalises and keeps the best result; this removes tiny spikes
  that survive the stochastic phases.
* The remaining budget is spent on the final simulated‑annealing run as
  before.

All phases respect the 1000 s wall‑clock budget, maintain a total sum
of 1 (so that scaling does not affect the score) and return a plain list
of floats that can be handed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    log_r = rng.uniform(-8.0, -0.1)
    r = np.exp(log_r)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a /= a.sum()
    return a


def _mixture_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    w = rng.random()
    a1 = _random_beta_candidate(n, rng)
    a2 = _random_beta_candidate(n, rng)
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    w = np.log(a0 + 1e-12)          # work in log‑softmax space

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.001
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 5000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedules
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max gradient (same pattern as exact but weighted)
        b = _full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1:2 * n - 1]
        grad = 4.0 * n * segment[::-1]       # gradient of 2·n·smooth_max

        # chain‑rule through softmax
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)

        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (source→target moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.9, 0.001
    temp_start, temp_end = 0.03, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 40000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # locate the peak
        k_max = int(np.argmax(b))

        # contributions to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # choose source proportional to its contribution
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # choose target (favor low contributions)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation (it={iteration}), re‑perturb")
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # exact sub‑gradient of the objective
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        grad_f = 2.0 * n * grad          # factor 2·n from the objective

        # linear step‑size schedule (more aggressive early)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad_f
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------
def _expgrad_polish(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_val = _objective(a)

    eta_start, eta_end = 1.0, 1e-5

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        grad_f = 2.0 * n * grad

        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        a = a * np.exp(-eta * grad_f)
        a = np.maximum(a, 1e-30)          # keep strictly positive
        a /= a.sum()

        new_val = _objective(a)
        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[ExpG] new best = {best_val:.6f}")

        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.018, size=n), 0.0)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic projected gradient descent on the exact objective
# ----------------------------------------------------------------------
def _projgrad_exact(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent with back‑tracking line‑search on the exact max‑conv objective."""
    a = a0.copy()
    cur_val = best_val

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        max_idx = np.where(np.abs(b - max_b) < 1e-14)[0]

        # build indicator vector for the max entries
        p = np.zeros_like(b)
        p[max_idx] = 1.0
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1:2 * n - 1]          # length n
        grad_f = 4.0 * n * segment[::-1]           # gradient of 2·n·max_b

        # line‑search (simple back‑tracking)
        step = 1.0
        improved = False
        for _ in range(20):
            a_new = a - step * grad_f
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            new_val = _objective(a_new)
            if new_val < cur_val - 1e-15:
                a = a_new
                cur_val = new_val
                improved = True
                if new_val < best_val:
                    best_val, best_a = new_val, a.copy()
                    print(f"[ProjG] new best = {best_val:.6f}")
                break
            step *= 0.5

        if not improved:
            # unable to make progress – stop this phase
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Small smoothing phase (neighbour averaging)
# ----------------------------------------------------------------------
def _smooth_pass(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 best_a: np.ndarray,
                 best_val: float) -> tuple[np.ndarray, float]:
    """Apply a few neighbour‑averaging kernels and keep any improvement."""
    kernels = [
        np.array([0.25, 0.5, 0.25]),
        np.array([0.1, 0.8, 0.1]),
        np.array([0.05, 0.9, 0.05]),
        np.array([0.2, 0.6, 0.2]),
    ]
    a = a0.copy()
    cur_val = best_val

    for ker in kernels:
        now = time.time()
        if now >= deadline:
            break
        a_s = np.convolve(a, ker, mode='same')
        a_s = np.maximum(a_s, 0.0)
        a_s /= a_s.sum()
        val = _objective(a_s)
        if val < cur_val - 1e-14:
            a = a_s
            cur_val = val
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[Smooth] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Larger‑move simulated‑annealing (final phase)
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and (tries to) minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size (larger n gives finer granularity)
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Rich initial population (systematic Beta grid + random draws)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # try to reuse a previously discovered solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # ----- systematic Beta grid (60×60) -----
    logp_vals = np.linspace(-8.0, 2.5, 60)
    logq_vals = np.linspace(-8.0, 2.5, 60)
    for lp in logp_vals:
        for lq in logq_vals:
            cand = _beta_grid_candidate(n, lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ----- random Beta draws -----
    for _ in range(3000):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- Dirichlet draws (various concentrations) -----
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(30):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ----- exponential family -----
    for _ in range(800):
        cand = _exponential_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- power‑law family -----
    for p in np.linspace(-4.0, 4.0, 25):
        cand = _powerlaw_candidate(n, p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- mixture of two Betas -----
    for _ in range(1200):
        cand = _mixture_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- simple sinusoid / cosine‑square candidates -----
    i = np.arange(n, dtype=np.float64)
    cand = np.sin(np.pi * (i + 0.5) / n)
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    cand = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    print(f"[Init] best after exhaustive init = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam optimisation (≈20 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.20 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase B – Hill‑climbing (≈35 % of the budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * budget_s
    best_a, best_val = _hillclimb(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase C – Sub‑gradient polishing (≈25 % of the budget)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase_c_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=500_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase D – Mirror‑descent (exponentiated‑gradient) polishing (≈15 % of budget)
    # --------------------------------------------------------------
    phase_d_deadline = start_time + 0.95 * budget_s
    best_a, best_val = _expgrad_polish(
        best_a, n,
        deadline=phase_d_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=200_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase E – Deterministic projected gradient descent (exact max)
    # --------------------------------------------------------------
    phase_e_deadline = start_time + 0.985 * budget_s
    best_a, best_val = _projgrad_exact(
        best_a, n,
        deadline=phase_e_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase F – Small smoothing passes
    # --------------------------------------------------------------
    phase_f_deadline = start_time + 0.997 * budget_s
    best_a, best_val = _smooth_pass(
        best_a, n,
        deadline=phase_f_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase G – Final simulated‑annealing (use any leftover time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_anneal(
            best_a, n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            rng=rng,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure a plain list of floats is returned
    return best_a.tolist()
