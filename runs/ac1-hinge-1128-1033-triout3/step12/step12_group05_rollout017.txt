```python
import time, numpy as np

def _full_convolution(x):
    n = x.shape[0]
    size = 1 << (2*n - 2).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx*fx, size)[:2*n-1]
    return conv

def _objective(a):
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0*n*conv.max())

def _active_gradient(a, conv, rel_tol=0.0):
    n = a.shape[0]
    max_val = conv.max()
    # find indices equal within tolerance
    if rel_tol == 0.0:
        active = np.flatnonzero(conv == max_val)
    else:
        active = np.flatnonzero(conv >= max_val * (1 - rel_tol))
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad
```

"""
Propose a step‑function with a very low value of ``evaluate_sequence``.

The search consists of three stages:

1️⃣  **Broad random exploration** – many candidates are created by
    drawing a random support (5 %‑50 % of the indices) and random
    Dirichlet weights on that support.  A few deterministic families
    (geometric, symmetric‑geometric and power‑law sequences) are also
    evaluated.  The ten best candidates (according to the cheap proxy
    ``2·n·max(conv)``) are kept.

2️⃣  **Refinement** – each of the top candidates is optimised with an
    exponentiated‑gradient (mirror‑descent) optimiser on the simplex.
    The gradient is the exact sub‑gradient of the *max* of the
    convolution (all entries whose value is within a tiny relative
    tolerance of the maximum are used).  Momentum (β≈0.99) and an
    exponentially decaying step‑size are employed.  Occasionally a
    tiny Gaussian jitter is added to escape plateaus.

3️⃣  **Local search** – after the mirror‑descent phase the best result
    undergoes a short stochastic hill‑climbing where a few coordinates
    are perturbed randomly; any improvement is kept.

All stages respect the given time budget (default 1000 s).  The final
sequence is returned as a plain ``list[float]``; the official score is
printed for reference.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of ``x`` with itself (length ``2·len(x)‑1``)."""
    n = x.shape[0]
    size = 1 << (2 * n - 2).bit_length()      # next power‑of‑two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


def _cheap_objective(a: np.ndarray) -> float:
    """
    Cheap surrogate of ``evaluate_sequence``.
    The sequence is assumed to be normalised to sum = 1, therefore
    the denominator is constant and we only need ``2·n·max(conv)``.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (full active set)
# ----------------------------------------------------------------------


def _subgradient_max(
    a: np.ndarray,
    conv: np.ndarray,
    rel_tol: float = 1e-8,
) -> np.ndarray:
    """
    Exact sub‑gradient of ``max(conv)`` w.r.t. ``a``.
    All indices whose value is within ``rel_tol`` of the maximum are
    taken into the active set.
    """
    n = a.shape[0]
    max_val = conv.max()
    # indices that are “active”
    active = np.flatnonzero(conv >= max_val * (1.0 - rel_tol))
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution of a_i a_{k-i}
        grad[idx] += a[k - idx]

    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Candidate generation helpers
# ----------------------------------------------------------------------


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """Geometric decay a_i ∝ exp(-lam·i) (normalised)."""
    x = np.exp(-lam * np.arange(n, dtype=np.float64))
    return x / x.sum()


def _make_symmetric_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric decay a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    d = np.arange(n, dtype=np.float64)
    dist = np.minimum(d, n - 1 - d)
    x = np.exp(-lam * dist)
    return x / x.sum()


def _make_powerlaw(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """Power‑law a_i ∝ (i+shift)^{-p} (normalised)."""
    idx = np.arange(n, dtype=np.float64) + shift
    # avoid division by zero
    idx[idx == 0.0] = 1.0
    x = idx ** (-p)
    return x / x.sum()

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (sequence) that minimises
    ``evaluate_sequence``.  The function respects the available time
    budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # leave a safety margin

    # --------------------------------------------------------------
    # Problem size – 8000 is a good trade‑off between speed and quality.
    # --------------------------------------------------------------
    n = 8000

    best_seq: np.ndarray | None = None
    best_val = np.inf

    def maybe_update(candidate: np.ndarray):
        """Update the global best if the cheap objective improves."""
        nonlocal best_seq, best_val
        val = _cheap_objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[update] new best cheap={best_val:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previously saved solution (if available)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # Interpolate to current resolution
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0 or not np.isfinite(s):
            a0 = np.full_like(a0, 1.0 / n)
        else:
            a0 = a0 / s
        maybe_update(a0)
        print("[init] re‑used previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad random exploration + deterministic families
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    top_candidates: list[tuple[float, np.ndarray]] = []   # (objective, seq)

    def add_candidate(arr: np.ndarray):
        val = _cheap_objective(arr)
        # keep a small sorted list of the best few candidates
        if len(top_candidates) < 12:
            top_candidates.append((val, arr.copy()))
            top_candidates.sort(key=lambda t: t[0])
        elif val < top_candidates[-1][0]:
            top_candidates[-1] = (val, arr.copy())
            top_candidates.sort(key=lambda t: t[0])

    # random Dirichlet‑type candidates
    while time.time() < deadline - 30.0:      # reserve ~30 s for refinement
        # support size between 5 % and 50 %
        k = int(rng.uniform(0.05, 0.5) * n)
        k = max(2, min(k, n))
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.gamma(shape=1.0, scale=1.0, size=k)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = weights
        a /= a.sum()
        maybe_update(a)
        add_candidate(a)

        # occasionally try a dense Dirichlet draw (full support)
        if rng.random() < 0.005:
            a_full = rng.gamma(shape=1.0, scale=1.0, size=n)
            a_full /= a_full.sum()
            maybe_update(a_full)
            add_candidate(a_full)

        a_exp = _make_exponential(n, lam)
        maybe_update(a_exp)
        add_candidate(a_exp)

        a_sym = _make_symmetric_exponential(n, lam)
        maybe_update(a_sym)
        add_candidate(a_sym)

    for p in np.linspace(0.35, 0.65, 9):
        a_pow = _make_powerlaw(n, p)
        maybe_update(a_pow)
        add_candidate(a_pow)

    # ensure we have at least one candidate
    if not top_candidates:
        uniform = np.full(n, 1.0 / n)
        maybe_update(uniform)
        top_candidates.append((best_val, uniform))

    print(f"[explore] kept {len(top_candidates)} top candidates; best cheap={best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Refinement – mirror‑descent (exponentiated gradient)
    # --------------------------------------------------------------
    # hyper‑parameters
    beta = 0.99                # momentum
    step0 = 0.4                # initial step (log‑space)

        a = init_seq.copy()
        momentum = np.zeros_like(a)
        it = 0
        no_improve = 0

        while time.time() < deadline - 5.0:
            conv = _full_convolution(a)
            cur_val = 2.0 * n * conv.max()

            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                no_improve = 0
                print(f"[optim #{rank+1}] iter {it} new global best {best_val:.6f}")
            else:
                no_improve += 1

            grad = _subgradient_max(a, conv, rel_tol=1e-8)

            # momentum update
            momentum = beta * momentum + (1.0 - beta) * grad

            # linear to exponential step‑size decay
            elapsed = time.time() - start
            progress = elapsed / (deadline - start)
            step = step0 * (1.0 - progress) + step_min

            # exponentiated (multiplicative) update – stays on the simplex
            log_factor = -step * momentum
            log_factor = np.clip(log_factor, -40.0, 40.0)   # protect from overflow
            a = a * np.exp(log_factor)
            s = a.sum()
            if s == 0.0 or not np.isfinite(s):

            else:
                a /= s

            # jitter occasionally or after long stagnation
            if rng.random() < 0.001 or no_improve >= max_no_improve:
                jitter = rng.normal(scale=0.001, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = a / a.sum()
                momentum.fill(0.0)
                no_improve = 0

            it += 1

        # ----------------------------------------------------------
        # 3️⃣  Short local hill‑climbing around the obtained optimum
        # ----------------------------------------------------------
        if best_seq is not None:
            a_local = best_seq.copy()
            for _ in range(3000):

                delta = rng.normal(scale=0.0007, size=5)

                    # keep using the improved point

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:

        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
