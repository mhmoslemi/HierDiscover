def mirror_descent(a, max_steps, eta_start=0.5, eta_end=0.001):
    best_obj = np.inf
    best_a = a.copy()
    stagnation = 0
    n = a.shape[0]
    out_len = 2*n - 1
    # for reference inside loop
    step = 0
    while step < max_steps and time.time() < deadline:
        conv = conv_self(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagnation = 0
            print(f"[mirror] step={step} new best O={obj:.6f}")
        else:
            stagnation += 1
        # compute gradient for max index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]
        # step size
        eta = eta_start * (1.0 - step / max_steps) + eta_end * (step / max_steps)
        # update
        a = a * np.exp(-eta * g)
        # ensure positivity
        eps = 1e-15
        a = np.maximum(a, eps)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)
        # jitter occasionally
        if stagnation >= 20000:
            jitter_noise = np.random.normal(scale=0.01, size=n)
            a = a + jitter_noise
            a = np.maximum(a, eps)
            a /= a.sum()
            stagnation = 0
            print(f"[mirror] jitter at step {step}")
        step += 1
    return best_a, best_obj
```

"""
Propose a step function that (empirically) improves the objective

    O = 2·n·max(conv(seq, seq)) / (sum(seq))² .

The algorithm consists of three phases:

1. **Initial pool** – a few deterministic shapes (beta‑distributions),
   a possible previous solution (interpolated) and a handful of Dirichlet
   draws.  All candidates are normalised to sum to 1.

2. **Short mirror‑descent runs** – each candidate is refined for a few
   thousand iterations using an exponentiated‑gradient (entropy‑mirror)
   update that directly minimises the maximal entry of the *self*
   convolution.  The best candidate is kept.

3. **Long polishing** – on the best candidate a long mirror‑descent
   phase runs for the remaining time budget, with a linearly decreasing
   step‑size and occasional random jitter to escape plateaus.

All updates stay inside the probability simplex (non‑negative, sum = 1)
and the convolution is computed by FFT for speed.  The final sequence
is returned as a plain Python list; a secure call to ``evaluate_sequence``
checks the achieved value.
"""

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_self(a: np.ndarray, fft_len: int, out_len: int) -> np.ndarray:
    """
    Linear convolution of a with itself (full length = 2·len(a)‑1)
    using an FFT of length ``fft_len`` (a power of two).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:out_len]
    return conv


def _mirror_descent(
    a: np.ndarray,
    n: int,
    fft_len: int,
    out_len: int,
    deadline: float,
    max_steps: int,
    eta_start: float = 0.5,
    eta_end: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror descent on the exact max entry of the self‑convolution.

    Parameters
    ----------
    a : np.ndarray
        Starting distribution (non‑negative, sum = 1).
    n : int
        Length of ``a``.
    fft_len, out_len : int
        Parameters for the FFT‑based convolution routine.
    deadline : float
        Absolute time (as returned by ``time.time()``) after which the
        routine must stop.
    max_steps : int
        Upper bound on the number of iterations.
    eta_start, eta_end : float
        Linear schedule for the step‑size ``eta`` (start → end).

    Returns
    -------
    best_a : np.ndarray
        The best distribution seen during the run.
    best_obj : float
        Corresponding objective value 2·n·max(conv).
    """
    best_obj = np.inf
    best_a = a.copy()
    stagnation = 0
    eps = 1e-15          # avoid exact zeros

    step = 0
    while step < max_steps and time.time() < deadline:
        # full self‑convolution
        conv = _full_convolution_self(a, fft_len, out_len)
        max_val = conv.max()
        obj = 2.0 * n * max_val

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagnation = 0
            print(f"[mirror] step={step:>7d}  new best O={obj:.6f}")
        else:
            stagnation += 1

        # index of the (first) maximal entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max‑entry w.r.t. a (2·a_{k‑i})
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        # linearly decaying step size
        eta = eta_start * (1.0 - step / max_steps) + eta_end * (step / max_steps)

        # exponentiated‑gradient (entropy‑mirror) update
        a = a * np.exp(-eta * g)

        # enforce positivity and renormalise
        a = np.maximum(a, eps)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)

        # occasional jitter to escape plateaus
        if stagnation >= 20000:
            jitter = np.random.normal(scale=0.01, size=n)
            a = a + jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            stagnation = 0
            print(f"[mirror] jitter injected at step {step}")

        step += 1

    return best_a, best_obj


def _beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Return a normalised beta‑density sampled on a uniform grid."""
    x = np.arange(1, n + 1, dtype=np.float64) / n
    a = (x ** (p - 1.0)) * ((1.0 - x) ** (q - 1.0))
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a.fill(1.0 / n)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights that (hopefully) yields a lower
    value of ``evaluate_sequence`` than the previous best (~1.5081).
    The routine respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer discretisation and a
    # potentially lower maximal autocorrelation.
    # ------------------------------------------------------------------
    n = 12000
    out_len = 2 * n - 1
    fft_len = _next_pow2(out_len)

    # ------------------------------------------------------------------
    # Build a small, diverse pool of starting vectors
    # ------------------------------------------------------------------
    init_vectors = []

    # 1) reuse previous solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old.ndim == 1 and old.size > 0:
            old_len = old.shape[0]
            x_new = np.linspace(0.0, 1.0, n)
            x_old = np.linspace(0.0, 1.0, old_len)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full(n, 1.0 / n)
            init_vectors.append(a)
            print("[init] interpolated previous solution")
    except Exception:
        pass

    # 2) a few smooth Beta shapes (including flat, bell‑ and U‑shaped)
    for p, q in [(1.0, 1.0), (0.5, 0.5), (2.0, 2.0), (5.0, 5.0), (0.3, 0.7), (2.5, 1.0)]:
        init_vectors.append(_beta_shape(n, p, q))

    # 3) random Dirichlet draws with a range of concentration parameters
    for conc in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 5.0]:
        a = np.random.dirichlet(np.full(n, conc))
        init_vectors.append(a)

    # ------------------------------------------------------------------
    # Phase 1 – short mirror‑descent refinements on every start vector
    # ------------------------------------------------------------------
    best_a = None
    best_obj = np.inf

    # allocate roughly 15 % of the total budget for this phase
    init_deadline = start_time + 0.15 * budget_s
    short_steps = 8000

    print("[phase 1] quick refinements on each initial vector")
    for idx, a0 in enumerate(init_vectors):
        if time.time() >= init_deadline:
            break
        print(f"[phase 1] start candidate #{idx + 1}/{len(init_vectors)}")
        a_refined, obj_refined = _mirror_descent(
            a0.copy(),
            n,
            fft_len,
            out_len,
            deadline=init_deadline,
            max_steps=short_steps,
            eta_start=0.5,
            eta_end=0.01,
        )
        if obj_refined < best_obj:
            best_obj = obj_refined
            best_a = a_refined.copy()
            print(f"[phase 1] new overall best O={best_obj:.6f}")

    # If nothing succeeded (should not happen), fall back to a uniform vector
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_obj = 2.0 * n * _full_convolution_self(best_a, fft_len, out_len).max()
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Phase 2 – long polishing on the best candidate
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 5.0:  # need a few seconds at least
        # we do not know the exact time per iteration, so we give a large
        # upper bound and let the routine stop by the deadline.
        max_polish_steps = int(5e6)  # effectively “as many as possible”
        print("[phase 2] long polishing run")
        best_a, best_obj = _mirror_descent(
            best_a,
            n,
            fft_len,
            out_len,
            deadline=deadline,
            max_steps=max_polish_steps,
            eta_start=0.45,
            eta_end=0.0005,
        )
        print(f"[phase 2] polishing finished – best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return final_seq
