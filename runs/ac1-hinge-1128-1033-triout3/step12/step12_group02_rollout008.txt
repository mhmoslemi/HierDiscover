"""
Propose a candidate step‑function (non‑negative sequence) that
minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))²

The implementation works in three phases:

1. **Exploratory sampling** – a quick grid over Beta‑shaped
   sequences (including very asymmetric shapes) and a batch of
   random Dirichlet draws are evaluated.  The best sequence found
   becomes the starting point for refinement.

2. **Mirror‑descent refinement** – starting from the best candidate,
   we repeatedly minimise the exact maximal entry of the
   self‑convolution using an entropy‑mirror (exponentiated‑gradient)
   update.  The step‑size η is linearly decayed and a simple
   stagnation detector adds a small random jitter when no improvement
   has been seen for a while.

3. **Final polishing** – a short additional run of the same mirror
   descent is performed on the best vector discovered.

All work respects the supplied time‑budget (default 1000 s) and
returns a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution (self‑convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of ``x`` with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective for a probability vector (sum == 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) with n = len(a) (a is already normalised)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Mirror‑descent step for the exact max‑entry of the self‑convolution
# ----------------------------------------------------------------------
def _mirror_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform one entropy‑mirror descent step on ``a``.
    ``eta`` is the step size (positive).
    Returns the updated (and normalised) vector.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)

    # sub‑gradient of the maximal entry
    g = np.zeros_like(a)
    if lo <= hi:
        idx = np.arange(lo, hi + 1, dtype=int)
        # derivative 2·a_{k‑i}
        g[idx] = 2.0 * a[k - idx]

    # exponentiated‑gradient update (entropy mirror)
    a = a * np.exp(-eta * g)

    # renormalise to keep a a probability vector
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a /= s
    return a

    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
