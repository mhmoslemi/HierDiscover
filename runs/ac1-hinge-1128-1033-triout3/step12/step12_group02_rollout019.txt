"""
Propose a candidate step‑function that improves on the previous
1.508158 bound.

Algorithm
---------

1. **Initialisation** – If a previous solution is available it is
   interpolated to the current length (8000).  Otherwise a smooth
   Beta(2, 2) density is used and normalised to a probability vector.

2. **Stage 1 – smoothed Adam** (≈40 % of the time budget) –  The
   non‑smooth max is replaced by a soft‑max with an increasing
   temperature α.  Adam updates are performed in the log‑domain,
   guaranteeing positivity.

3. **Stage 2 – entropy‑mirror descent** (≈30 % of the budget) –  Once
   α is large (≈150) the surrogate becomes very close to the true max.
   We switch to a sub‑gradient step on the exact maximal convolution
   entry, using the entropy mirror map (multiplicative update).

4. **Stage 3 – local neighbourhood search** (remaining time) –  The
   best vector found so far is refined by tiny mass transfers between
   two indices.  For a transfer δ from i→j the convolution updates can
   be computed by two O(n) slice‑operations, allowing millions of cheap
   proposals.  Moves that improve the objective are always accepted;
   occasional worsening moves are accepted with a simple annealing
   probability.  After a long stretch without improvement a small
   random jitter is applied to escape possible plateaux.

5. **Polishing** –  the best vector is finally polished by a longer
   entropy‑mirror descent (5 000 steps) to squeeze out the last bits.

The routine respects the 1000 s budget, prints occasional progress
messages and finally returns the best sequence (list of floats) it
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution (FFT, length 2·n−1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (length 2·len(x)−1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max entry (polishing)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 5000, eta_start: float = 0.45) -> np.ndarray:
    """Entropy‑mirror descent on the exact max‑entry of a*a."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]               # sub‑gradient
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a
