"""
Projected gradient descent with soft‑max weighted gradient.

The algorithm tries to flatten the self‑convolution a*a.
For each iteration we

  1. compute the full convolution c = a*a,
  2. build a probability vector w by a soft‑max on c
     (high values receive larger weight; α controls sharpness),
  3. compute the gradient of the smoothed objective:
        g_i = 2 * Σ_j a_j * w_{i+j}
     which can be written as a convolution of a_rev and w,
  4. apply a momentum‑augmented step a←a‑η·g,
  5. project back onto the probability simplex (a ≥0, Σ a = 1).

The step size η decays, α is increased slowly to focus on the true
maximum, and occasional Gaussian jitter escapes plateaus.  The best
vector seen is finally evaluated with ``evaluate_sequence``.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a vector with itself using real FFT.
    Returns length 2·len(x)-1 array.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two vectors using real FFT.
    Returns length len(x)+len(y)-1 array.
    """
    size = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: len(x) + len(y) - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Main search routine
# -------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best list found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # ---------------------------------------------------------
    # Problem size
    # ---------------------------------------------------------
    n = 8000

    # ---------------------------------------------------------
    # Initialise – try to reuse a previously stored construction.
    # ---------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_len = old.shape[0]
        x_old = np.linspace(0.0, 1.0, old_len)
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total
        print("[init] interpolated previous solution (len:", old_len, "→", n, ")")
    except Exception:
        a = np.full(n, 1.0 / n)
        print("[init] uniform start, n =", n)

    # Ensure exact simplex feasibility
    a = _project_simplex(a)

    # ---------------------------------------------------------
    # Helper to compute the (scaled) objective for sum=1:
    #   obj = 2 * n * max(conv)
    # ---------------------------------------------------------
    def _objective(x: np.ndarray) -> float:
        conv = _full_convolution(x)
        return 2.0 * n * conv.max()

    cur_obj = _objective(a)
    best_a = a.copy()
    best_val = cur_obj
    print("[init] objective =", best_val)

    # ---------------------------------------------------------
    # Optimisation hyper‑parameters
    # ---------------------------------------------------------
    eta = 0.1          # initial step size
    eta_min = 0.001
    decay = 0.9995     # geometric decay per iteration
    alpha = 1.0        # initial soft‑max sharpness
    alpha_max = 30.0
    alpha_inc = 1.001  # multiplicative increase each iteration
    beta = 0.9         # momentum coefficient
    m = np.zeros_like(a)  # momentum buffer

    stagnation = 0
    max_stagn = 3000   # after this many non‑improving steps we jitter
    iteration = 0

    # ---------------------------------------------------------
    # Main optimisation loop
    # ---------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ---- 1) current convolution and objective ----
        conv = _full_convolution(a)
        max_c = conv.max()
        cur_obj = 2.0 * n * max_c

        # ---- keep the best seen ----
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[iter {iteration:07d}] new best = {best_val:.9f} "
                  f"(time {time.time() - start_time:.1f}s)")

        # ---- 2) soft‑max weights on the convolution ----
        # use α·(conv - max_c) for stability; subtract max afterwards
        scaled = alpha * (conv - max_c)
        # scaling to avoid overflow
        max_scaled = scaled.max()   # this is 0.0
        w = np.exp(scaled - max_scaled)   # w ≥ 0, sum ≤ length
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum  # now w is a probability distribution

        # ---- 3) gradient of the smoothed max (soft‑max) ----
        # g_i = 2 * Σ_j a_j * w_{i+j}
        a_rev = a[::-1]
        conv_rev = _convolve_fft(a_rev, w)          # length = 3n-2
        g = 2.0 * conv_rev[n - 1 : n - 1 + n]       # slice of length n

        # ---- 4) momentum update + projection ----
        m = beta * m + (1.0 - beta) * g
        v = a - eta * m
        a_new = _project_simplex(v)

        # ---- 5) evaluate the candidate ----
        conv_new = _full_convolution(a_new)
        new_obj = 2.0 * n * conv_new.max()

        # ---- 6) acceptance logic ----
        if new_obj < cur_obj:
            a = a_new
            cur_obj = new_obj
            stagnation = 0
        else:
            # tiny probability to accept worsening move
            if np.random.rand() < 0.001:
                a = a_new
                cur_obj = new_obj
            stagnation += 1

        # ---- 7) adapt step size and soft‑max sharpness ----
        eta = max(eta * decay, eta_min)
        alpha = min(alpha * alpha_inc, alpha_max)

        # ---- 8) handle stagnation ----
        if stagnation > max_stagn:
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            m = np.zeros_like(a)
            stagnation = 0
            print(f"[reset] jitter at iter {iteration} (time {time.time() - start_time:.1f}s)")

        # occasional status report (cheap)
        if iteration % 2000 == 0:
            elapsed = time.time() - start_time
            print(f"[status] iter {iteration}, best {best_val:.9f}, eta {eta:.5f}, α {alpha:.2f}, time {elapsed:.1f}s")

        # early exit if we already beat the target (nice but not required)
        if best_val < 1.503:
            print(f"[early stop] reached target at iter {iteration}")
            break

    # -----------------------------------------------------------------
    # Final evaluation (single secure call)
    # -----------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score =", final_score)
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
