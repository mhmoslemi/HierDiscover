Algorithm
---------

  - uniform,

        F(a) = 2·n·max_k (a * a)[k]        ( Σ a = 1 )

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    size = 1 << (2 * n - 2).bit_length()       # next power of two ≥ 2·n‑1
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` (len n) and ``y`` (len m)."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << (n + m - 2).bit_length()       # power‑of‑two ≥ n+m‑1
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: n + m - 1]
    return conv

# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return a probability vector proportional to exp(beta·conv)."""
    # subtract the max for numerical stability
    m = conv.max()
    exp_term = np.exp(beta * (conv - m))
    w = exp_term / exp_term.sum()
    return w

    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

                print(f"[jitter] start {start_idx} iter {it}")

            it += 1

            # early exit if the time budget is almost exhausted
            if time.time() >= deadline - 0.5:
                break

        # move to the next start point unless we ran out of time
        if time.time() >= deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Fallback safety net (should never be triggered)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation (official score)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
