"""
Hybrid optimiser for the step‑function problem
------------------------------------------------
* Phase 0 – Light sweep (uniform, previous solution, Dirichlet / Beta draws)
  to obtain a decent starting point.
* Phase 1 – Long sub‑gradient descent with a simple decaying stepsize.
  The true sub‑gradient of  f(a)=2·n·max(conv(a,a))  is computed by
  exploiting the structure of the convolution (only the indices where the
  maximum is attained matter).  After each step the iterate is projected
  onto the probability simplex.
* Phase 2 – (Optional) a short “final polish’’ with L‑BFGS on a smooth
  log‑sum‑exp surrogate of the max‑convolution.  This refines the
  solution found by the sub‑gradient phase.
* Phase 3 – Final secure evaluation and return.

The routine adapts its behaviour to the remaining time budget and never
exceeds the 1000 s limit.  It works purely with NumPy / SciPy, does not
touch the filesystem and respects all security checks imposed by
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return conv(x, x) in full mode using a power‑of‑two FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – the quantity minimised by evaluate_sequence."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x ≥ 0, Σx = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """Smooth approximation of max(conv(a,a)) together with ∇ (including 2·n)."""
    n = a.shape[0]
    b = _full_convolution(a)                 # length 2·n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    w = np.exp(shifted)
    p = w / w.sum()                          # soft‑max weights (length 2·n‑1)

    # grad = 2·n·conv(a, p_rev)  (the factor 2 comes from the derivative of a*a)
    p_rev = p[::-1]
    conv_ab = np.convolve(a, p_rev)          # full convolution, length 3·n‑2
    # take the central n entries that correspond to the “forward” part
    start = n - 1
    segment = conv_ab[start:start + n]
    grad_a = 4.0 * n * segment[::-1]         # factor 4 = 2 (from a*a) * 2·n (objective)
    smooth_val = max_b + eps * np.log(w.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → probability simplex)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    x = x - np.max(x)
    e = np.exp(x)
    return e / e.sum()


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smooth surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-12,
                    max_iter: int = 800) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on 2·n·smooth_max with a tiny ε."""
    n = a0.shape[0]
    x0 = np.log(a0 + 1e-12)    # log‑its → soft‑max guarantees feasibility

    def fun(x):
        a = _softmax(x)
        sm, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * sm

    def jac(x):
        a = _softmax(x)
        sm, grad_a = _smooth_max_and_grad(a, eps)
        # chain‑rule for the soft‑max
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    if deadline - time.time() < 0.5:
        return best_a, best_val
    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        print(f"[L‑BFGS] new best = {val_opt:.6f}")
        best_val, best_a = val_opt, a_opt.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true (non‑smooth) objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_a: np.ndarray,
                         best_val: float,
                         step0: float = 0.35,
                         min_step: float = 1e-5,
                         jitter_every: int = 400,
                         jitter_prob: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on
        f(a)=2·n·max(conv(a,a))
    using the exact sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        # full convolution and max
        b = _full_convolution(a)
        max_b = b.max()
        ks = np.flatnonzero(b == max_b)                # indices where the max is attained

        # sub‑gradient of max(b) : g_i = 2·∑_{k∈K} a_{k-i}
        grad = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # reversed slice of a contributes to the gradient
            grad_slice = a[i_start:i_end + 1][::-1]
            grad[i_start:i_end + 1] += 2.0 * grad_slice

        # decaying stepsize – aggressive early, tiny later
        progress = (time.time() - start_time) / (deadline - start_time)
        step = step0 * (1.0 - progress) + min_step * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        # accept strict improvements
        if new_val + 1e-12 < cur_val:
            a, cur_val = a_new, new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Sub]   it {it:6d}  new best = {best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % jitter_every == 0 and np.random.rand() < jitter_prob:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)
            cur_val = _objective(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Random candidate generators (cheap, used for the initial sweep)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Beta‑shaped vector on a log‑scale and normalise."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ≈10⁻⁴ … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

    # ------------------------------------------------------------------
    # Problem size – a moderate n gives a good trade‑off between quality
    # and speed while still allowing many iterative updates.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # 0️⃣  Coarse initialisation (exact objective)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # uniform start
    best_val = _objective(best_a)

    # 0.1  Re‑use a previously discovered solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        cand = np.interp(new_x, old_x, old)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
        print(f"[Init] interpolated previous solution, value={val:.6f}")
    except Exception:
        pass

    # 0.2  Random Beta candidates (cheap)
    for _ in range(200):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # 0.3  Dirichlet draws with a handful of concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    print(f"[Init] best after sweep: value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long sub‑gradient descent (≈70 % of the remaining budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.70 * budget_s
    best_a, best_val = _subgradient_descent(
        best_a,
        deadline=phase1_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        step0=0.35,
        min_step=1e-5,
    )
    print(f"[Phase 1] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short L‑BFGS refinement on a smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(
            best_a,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            eps=1e-12,
            max_iter=800,
        )
        print(f"[Phase 2] post‑L‑BFGS best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final quick sub‑gradient polishing (≈5 % of the budget)
    # ------------------------------------------------------------------
    final_phase_deadline = deadline
    best_a, best_val = _subgradient_descent(
        best_a,
        deadline=final_phase_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        step0=0.15,
        min_step=1e-6,
        jitter_every=800,
        jitter_prob=0.02,
    )
    print(f"[Phase 3] final best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation (the score reported by the platform)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
