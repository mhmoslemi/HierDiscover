"""
Propose a high‑quality step‑function using a smoothed‐max gradient descent
with Adam‑style updates, simplex projection and occasional jitter.
The method improves on the plain sub‑gradient optimiser by using a
temperature‑scaled softmax of the autocorrelation (so the gradient is
well‑behaved) and by keeping momentum (Adam) which lets it escape flat
regions faster.  The whole search respects the 1000 s budget and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán, 2013 (arXiv:1309.1541)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    shift = np.max(x)
    e = np.exp(x - shift)
    s = np.sum(e)
    if s == 0.0:  # should never happen, but guard against divide‑by‑zero
        return np.full_like(x, 1.0 / x.size)
    return e / s


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of given size."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    # retain the linear (non‑circular) part
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _correlation(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Linear correlation r[i] = Σ_j a[j] * w[i + j] for i = 0…len(a)-1.
    Returned array has length len(a); zero‑padding is handled by the FFT size.
    """
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr_full = np.fft.irfft(fa * np.conj(fw), size)
    # The linear correlation occupies the first (len(a)+len(w)-1) entries.
    # The needed portion is the first len(a) values.
    return corr_full[: a.shape[0]]

# Main search routine
# ----------------------------------------------------------------------

    * after a long stagnation period inject a small Gaussian jitter;
    * keep the best vector encountered (according to the exact objective);
    * finally evaluate the best vector with the provided ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Choose problem size (use the same n as the reference run)
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initial sequence: try to reuse a previously stored solution.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_len = old.shape[0]
        # Linear interpolation to the target length.
        x_old = np.linspace(0.0, 1.0, old_len, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        a = np.interp(x_new, x_old, old)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= total
        print(f"[init] interpolated previous solution (len {old_len} → {n})")
    except Exception:
        a = np.full(n, 1.0 / n)
        print(f"[init] uniform start, n = {n}")

    # --------------------------------------------------------------
    # Adam‑style state
    # --------------------------------------------------------------
    m = np.zeros(n)                     # first moment
    v = np.zeros(n)                     # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    eta = 0.2                           # base learning rate (will be scaled by Adam)
    min_eta = 0.01

    # --------------------------------------------------------------
    # Temperature schedule for the softmax (controls how sharp the
    # approximation of the max is).  Start fairly smooth and slowly
    # sharpen.
    # --------------------------------------------------------------
    beta_temp = 5.0                     # initial “inverse temperature”
    beta_factor = 1.0005
    max_beta = 5000.0

    # --------------------------------------------------------------
    # Pre‑compute the FFT size that can hold the correlation (a ⋆ w)
    # --------------------------------------------------------------
    size = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Helper to compute the true objective (sum is forced to 1)
    # --------------------------------------------------------------
    def _objective(x: np.ndarray) -> float:
        conv = _full_convolution(x, size)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # Initialise bookkeeping
    # --------------------------------------------------------------
    cur_obj = _objective(a)
    best_a = a.copy()
    best_val = cur_obj
    print(f"[init] objective = {best_val:.9f}")

    stagnation = 0                # iterations without improvement of best_val
    max_stagn = 5000              # after this many we inject jitter
    it = 0
    t_start = time.time()

    while time.time() < deadline:
        it += 1

        # ---- 1) full self‑convolution and the current max ----
        conv = _full_convolution(a, size)          # length 2n‑1
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv                # exact objective

        # ---- 2) softmax weights (smooth approximation of the max) ----
        w = _softmax(beta_temp * conv)              # length 2n‑1, sums to 1

        # ---- 3) gradient of the smoothed objective ----
        # g_i = 2 * Σ_j a_j * w_{i+j}  (i = 0…n‑1)
        corr = _correlation(a, w, size)            # length n
        g = 2.0 * corr

        # ---- 4) Adam update on the probability vector ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        # ---- 5) evaluate candidate ----
        cand_obj = _objective(a_candidate)

        # ---- 6) accept if it improves the current point ----
        if cand_obj < cur_obj:
            a = a_candidate
            cur_obj = cand_obj
            stagnation = 0
        else:
            # occasional acceptance to add stochasticity
            if np.random.rand() < 0.001:
                a = a_candidate
                cur_obj = cand_obj
            stagnation += 1

        # ---- 7) keep the best overall solution ----
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[iter {it:07d}] new best = {best_val:.9f} "
                  f"(time {time.time() - start_time:.1f}s)")

        # ---- 8) decay learning rate and increase temperature ----
        eta = max(eta * 0.9999, min_eta)
        beta_temp = min(beta_temp * beta_factor, max_beta)

        # ---- 9) jitter on long stagnation ----
        if stagnation > max_stagn:
            noise = np.random.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            cur_obj = _objective(a)
            stagnation = 0
            print(f"[reset] jitter at iter {it} (time {time.time() - start_time:.1f}s)")

        # ---- 10) final‑phase fallback to pure sub‑gradient steps
        # (use the remaining few seconds for a cheap fine‑tune)
        if time.time() > deadline - 5.0:
            break

    # ---- Optional cheap sub‑gradient refinement (last few seconds) ----
    # We reuse the simple max‑entry sub‑gradient used in the reference code.
    eta_sub = 0.04
    decay_sub = 0.9997
    stagn_sub = 0
    max_stagn_sub = 3000

    while time.time() < deadline:
        # 1) convolution and its maximiser
        conv = _full_convolution(a, size)
        k_max = int(conv.argmax())
        # 2) sub‑gradient of the *max* entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        counterpart = k_max - idx
        g[idx] = 2.0 * a[counterpart]

        # 3) gradient step + projection
        a = _project_simplex(a - eta_sub * g)

        # 4) evaluate
        cur_obj = _objective(a)

        # 5) bookkeeping
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[subgrad {it:07d}] new best = {best_val:.9f} "
                  f"(time {time.time() - start_time:.1f}s)")

        # 6) decay & stagnation handling
        eta_sub = max(eta_sub * decay_sub, 0.001)
        stagn_sub += 1
        if stagn_sub > max_stagn_sub:
            a = np.maximum(a + np.random.normal(scale=0.01, size=n), 0.0)
            a /= a.sum()
            stagn_sub = 0
            print(f"[reset sub] jitter at iter {it} (time {time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # Final secure evaluation (single call)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
```
