def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of ``x`` with itself (length ``2·len(x)-1``)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two (possibly different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

def _interpolate_to_len(old: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``old`` to a new length ``new_len``."""
    old_len = old.shape[0]
    if old_len == new_len:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    return np.interp(x_new, x_old, old)


def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` (without clipping)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    maxc = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def _mirror_descent_one_step(a: np.ndarray, eta: float) -> tuple[np.ndarray, float]:
    """
    Perform one entropy‑mirror‑descent update on ``a`` with step size ``eta``.
    Returns the new vector and the maximal convolution value of the new vector.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)

    g = np.zeros_like(a)
    if lo <= hi:
        idx = np.arange(lo, hi + 1, dtype=int)
        # sub‑gradient of the max entry
        g[idx] = 2.0 * a[k - idx]

    a_new = a * np.exp(-eta * g)
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        a_new = np.full_like(a, 1.0 / n)

    max_conv = _full_convolution(a_new).max()
    return a_new, max_conv


def _max_refine(a: np.ndarray, steps: int = 5000, eta_start: float = 0.45) -> np.ndarray:
    """
    Longer mirror‑descent polishing with linearly decreasing step size.
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]

        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)
    return a

    stage1_deadline = start + 0.55 * total

    # ------------------------------------------------------------------
    # Problem sizes
    # ------------------------------------------------------------------
    n1 = 4000      # coarse grid
    n2 = 12000     # fine grid (still cheap for FFT)

    # ------------------------------------------------------------------
    # Initialise the coarse vector (try to reuse the previous solution)
    # ------------------------------------------------------------------
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        try:
            old = np.asarray(old_seq, dtype=float)
            a0_small = _interpolate_to_len(old, n1)
            print("[init] interpolated previous solution to n1")
        except Exception:
            a0_small = None
    else:
        a0_small = None

    if a0_small is None:
        # smooth centrally‑peaked Beta(2,2) density – a good generic start
        i = np.arange(1, n1 + 1, dtype=np.float64)
        a0_small = (i ** 1.0) * ((n1 - i + 1) ** 1.0)
        print("[init] generated Beta(2,2) start for n1")

    a0_small = np.maximum(a0_small, 0.0)
    # Normalise to L1 = 1 (the absolute scale does not affect the objective)
    s0 = a0_small.sum()
    a0_small = a0_small / s0 if s0 > 0 else np.full_like(a0_small, 1.0 / n1)

    # ------------------------------------------------------------------
    # Coarse optimisation (stage 1) – mirror descent with line‑search
    # ------------------------------------------------------------------
    a = a0_small.copy()
    best_a_small = a.copy()
    best_obj_small = _objective(best_a_small)

    stagn_counter = 0
    stagn_limit = 2000
    it = 0
    eta_candidates_stage1 = (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01)

    while time.time() < stage1_deadline:
        improved = False
        for eta in eta_candidates_stage1:
            a_new, max_conv = _mirror_descent_one_step(a, eta)
            obj = 2.0 * n1 * max_conv  # denominator = 1 because sum(a_new)=1
            if obj < best_obj_small - 1e-12:
                a = a_new
                best_a_small = a.copy()
                best_obj_small = obj
                stagn_counter = 0
                improved = True
                print(f"[stage1] it={it:06d} obj={obj:.6f} eta={eta:.3f}")
                break
        if not improved:
            stagn_counter += 1
            if stagn_counter >= stagn_limit:
                # escape from a flat region
                perturb = np.random.normal(scale=0.005, size=n1)
                a = np.maximum(a + perturb, 0.0)
                a /= a.sum()
                stagn_counter = 0
                print(f"[stage1] jitter added at it={it}")
        it += 1

    # ------------------------------------------------------------------
    # Upsample to the fine grid (stage 2)
    # ------------------------------------------------------------------
    a_up = _interpolate_to_len(best_a_small, n2)
    a_up = np.maximum(a_up, 0.0)
    a_up /= a_up.sum()
    a = a_up.copy()
    best_a = a.copy()
    best_obj = _objective(best_a)

    # ------------------------------------------------------------------
    # Fine optimisation (stage 2) – same scheme, finer eta list
    # ------------------------------------------------------------------
    stagn_counter = 0
    stagn_limit = 3000
    it = 0
    eta_candidates_stage2 = (0.4, 0.2, 0.1, 0.05, 0.025, 0.01, 0.005)

    while time.time() < deadline:
        improved = False
        for eta in eta_candidates_stage2:
            a_new, max_conv = _mirror_descent_one_step(a, eta)
            obj = 2.0 * n2 * max_conv
            if obj < best_obj - 1e-12:
                a = a_new
                best_a = a.copy()
                best_obj = obj
                stagn_counter = 0
                improved = True
                print(f"[stage2] it={it:07d} obj={obj:.6f} eta={eta:.3f}")
                break
        if not improved:
            stagn_counter += 1
            if stagn_counter >= stagn_limit:
                perturb = np.random.normal(scale=0.002, size=n2)
                a = np.maximum(a + perturb, 0.0)
                a /= a.sum()
                stagn_counter = 0
                print(f"[stage2] jitter added at it={it}")
        it += 1

    # ------------------------------------------------------------------
    # Polishing phase – longer mirror descent with a smoothly decaying step
    # ------------------------------------------------------------------
    refined = _max_refine(best_a, steps=8000, eta_start=0.3)
    refined_obj = _objective(refined)
    if refined_obj < best_obj:
        best_a = refined
        best_obj = refined_obj
        print(f"[polish] improved to obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(internal obj ≈ {best_obj:.6f})")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
