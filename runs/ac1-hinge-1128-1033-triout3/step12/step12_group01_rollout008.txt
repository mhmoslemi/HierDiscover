"""
Propose a non‑negative step function that (empirically) beats the previous
objective O ≈ 1.50813.

Algorithm
---------
1.  Random search over a small parametric family.
    *   Single beta densities   a_i ∝ x^{p‑1}(1‑x)^{q‑1}.
    *   Mixtures of two betas   w·β(p₁,q₁)+(1‑w)·β(p₂,q₂).
    *   Dirichlet draws with various concentration values.
    *   A few deterministic seeds (uniform, sinusoid, previously found solution).

    The density is evaluated on a fixed grid (n = 8000) and normalised to
    sum‑to‑one.  The objective  O = 2·n·max(conv(a,a)) is computed with an
    FFT‑based convolution (≈ O(n log n) per evaluation) so that several
    hundred thousand candidates can be screened within the time budget.

2.  Keep the best candidate found in the search phase.

3.  Polish the survivor with entropy‑mirror descent on the *exact* max‑entry
    of the self‑convolution.  The update is
        a ← a·exp(‑η·g)   (g is the sub‑gradient of the max‑entry)
    followed by projection onto the probability simplex.  The step size η
    decays linearly and the number of steps is chosen from the remaining
    time budget (typical run: 20 000‑30 000 steps).

4.  Return the final sequence; a single call to ``evaluate_sequence`` is
    performed at the very end to obtain the official score.

All helper functions are defined at the top level; the only public entry point
required by the grader is ``propose_candidate``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolution (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (full length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max‑entry of the self‑convolution
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 3000, eta_start: float = 0.45) -> np.ndarray:
    """Refine ``a`` (already on the simplex) by entropy mirror descent."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())                     # active index
        # sub‑gradient of max‑entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # linearly decaying step size
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)
    return a


# ----------------------------------------------------------------------
# Helpers for families of candidate vectors
# ----------------------------------------------------------------------
def _beta_shape(p: float, q: float, n: int) -> np.ndarray:
    """Discrete beta density on a regular grid (sum not normalised)."""
    #  x in (0,1) exclusive to avoid zero‑to‑negative powers
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1))
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    return a


def _objective(a: np.ndarray) -> float:
    """Return the exact objective 2·n·max(conv(a,a))  (sum(a) assumed = 1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a probability vector (step heights) that minimises
    O = 2·n·max(conv(seq,seq)) / (sum(seq))².
    """
    np.random.seed(seed)
    start = time.time()
    # reserve a small margin for the final call to evaluate_sequence
    deadline = start + budget_s - 0.8

    # ------------------------------------------------------------------
    # Fixed problem size – 8000 points give a good trade‑off between
    # fidelity and speed (the objective is essentially independent of n).
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate quickly (avoid the slow np.convolve).
    # ------------------------------------------------------------------
    def fast_obj(vec):
        s = vec.sum()
        if s == 0.0:
            return np.inf
        a = vec / s
        return _objective(a)

    # ------------------------------------------------------------------
    # Initialise the incumbent with a few deterministic seeds
    # ------------------------------------------------------------------
    best_a = None
    best_obj = np.inf

    # 0) uniform distribution – gives O = 2.0 (reference)
    a_uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_a = a_uniform
    best_obj = fast_obj(a_uniform)

    # 1) sinusoidal shape (smooth, zero at the ends)
    a_sin = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a_sin = a_sin.clip(min=0)
    best_obj = min(best_obj, fast_obj(a_sin))
    if best_obj == fast_obj(a_sin):
        best_a = a_sin / a_sin.sum()

    # 2) try to reuse a previous best solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.size != n:
            # interpolate to the required grid
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a_old = np.interp(x_new, x_old, old, left=0.0, right=0.0)
        else:
            a_old = old.copy()
        a_old = np.maximum(a_old, 0.0)
        if a_old.sum() > 0.0:
            a_old /= a_old.sum()
            obj_old = fast_obj(a_old)
            if obj_old < best_obj:
                best_obj = obj_old
                best_a = a_old
                print("[init] imported previous solution, O≈{:.6f}".format(obj_old))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Random search over parametrised families
    # ------------------------------------------------------------------
    # Allocate roughly 70 % of the budget to the random search; the rest
    # will be used for the polishing phase.
    search_deadline = start + 0.70 * budget_s

    # Pre‑compute an array of log‑spaced bounds for the beta exponents.
    log_min, log_max = np.log(0.2), np.log(5.0)          # p,q ∈ [0.2, 5]
    log_min_c, log_max_c = np.log(0.02), np.log(5.0)    # Dirichlet concentration

    it = 0
    while time.time() < search_deadline:
        it += 1
        r = np.random.rand()

        if r < 0.35:
            # ----- single beta ------------------------------------------------
            p = np.exp(np.random.uniform(log_min, log_max))
            q = np.exp(np.random.uniform(log_min, log_max))
            a = _beta_shape(p, q, n)

        elif r < 0.70:
            # ----- mixture of two betas ---------------------------------------
            p1 = np.exp(np.random.uniform(log_min, log_max))
            q1 = np.exp(np.random.uniform(log_min, log_max))
            p2 = np.exp(np.random.uniform(log_min, log_max))
            q2 = np.exp(np.random.uniform(log_min, log_max))
            w = np.random.rand()
            a = w * _beta_shape(p1, q1, n) + (1.0 - w) * _beta_shape(p2, q2, n)

        elif r < 0.90:
            # ----- Dirichlet draw ---------------------------------------------
            conc = np.exp(np.random.uniform(log_min_c, log_max_c))
            a = np.random.dirichlet(np.full(n, conc))

        else:
            # ----- a random convex combination of three deterministic bases ---
            # (uniform, sinusoid, a narrow spike at the centre)
            w1, w2 = np.random.rand(2)
            w3 = 1.0 - w1 - w2
            if w3 < 0:
                w1, w2, w3 = np.abs(w1), np.abs(w2), np.abs(1.0 - w1 - w2)
            a = (w1 * a_uniform +
                 w2 * a_sin +
                 w3 * np.exp(-0.5 * ((np.arange(n) - n/2) / (0.05 * n))**2))

        # normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            continue
        a /= s

        # skip extremely peaked vectors which would be clipped by evaluate_sequence
        if a.max() > 1000.0:
            continue

        obj = fast_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[search] it={it:,}  new best O≈{best_obj:.6f}")

    # ------------------------------------------------------------------
    # Polishing phase – entropy mirror descent on the *exact* max entry
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_a is None:
        # fall back to uniform (should never happen)
        best_a = a_uniform

    if time_left > 0.5:
        # Rough estimate: ~5 µs per mirror‑descent step on typical hardware
        max_steps = int(min(60000, time_left * 18000))
        steps = max(5000, max_steps)
        print(f"[polish] running mirror descent for {steps} steps "
              f"(≈{time_left:.1f}s remaining)")
        best_a = _max_refine(best_a, steps=steps, eta_start=0.5)

    # ------------------------------------------------------------------
    # Final secure evaluation (the only call to the provided function)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx O≈{best_obj:.6f})")
    except Exception as exc:                       # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
