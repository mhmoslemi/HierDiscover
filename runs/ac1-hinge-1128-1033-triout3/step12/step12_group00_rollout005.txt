def _subgradient_descent(a0, n, deadline, start_time, max_iters):

    a = a0.copy()
    best = a.copy()
    best_val = _objective(a)
    step0 = 0.2

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute gradient
        b = _full_convolution(a)
        max_b = np.max(b)
        idxs = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in idxs:
            i = k - np.arange(n)
            mask = (i >= 0) & (i < n)
            grad[mask] += 2 * a[i[mask]]

        step = step0 / np.sqrt(it+1)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        val = _objective(a_new)
        if val < best_val:
            best_val, best = val, a_new.copy()
            print(f"[SubG] {it}: new best {best_val:.6f}")

        # acceptance: if improves, adopt; else maybe keep old a
        if val < _objective(a):
            a = a_new
        else:
            # occasional random perturbation to escape
            if np.random.rand() < 0.005:
                a = np.maximum(a + np.random.normal(scale=0.01, size=n), 0.0)
                a = _project_simplex(a)
    return best, best_val

"""
Improved step‑function optimiser (≈1000 s budget)

Key ideas
---------
1. **Broad initial sweep** – besides the random Beta/Dirichlet draws used
   previously we also enumerate cheap one‑parameter families
   (exponential and power‑law shapes) on a fine grid.  This discoveres
   high‑quality starting points that were missed before.

2. **More restarts & tighter schedule** – the top‑k candidates (k≈15)
   are each refined many times with L‑BFGS on a log‑sum‑exp smooth
   surrogate.  We start with a large smoothing ε (≈5e‑1) and drive it
   down to 1e‑12, using many short runs (max 800 iterations) to keep
   each run cheap while exploring many restarts.

3. **Extended sub‑gradient polishing** – after the L‑BFGS phase we run a
   pure projected sub‑gradient descent on the exact objective for the
   remaining time (≈40 %).  The step‑size decays as 1/√t and we add
   occasional jitter to escape tiny plateaus.

4. **Uniform blending & final fine‑tune** – we try convex blends with the
   uniform distribution and finish with a short L‑BFGS run at a very
   small ε (1e‑12).

All steps respect the 1000 s wall‑clock limit; the algorithm returns
the best feasible sequence it found.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.  The gradient already includes the factor 2·n.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # keep the n entries that correspond to the “forward” part
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → probability simplex)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray, cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    n = a.shape[0]
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 800) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smoothed objective.
    """
    # initialise in log‑space (soft‑max guarantees feasibility)
    x0 = np.log(a0 + 1e-12)

    # ----- objective ----------------------------------------------------
    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    # ----- gradient -----------------------------------------------------
    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    # respect remaining time
    if time.time() >= deadline - 0.5:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:        # pragma: no cover
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a))
    """
    a = a0.copy()
    cur_val = _objective(a)
    best_a, best_val = a.copy(), cur_val

    # initial step size – we start fairly aggressive
    step0 = 0.4

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # compute sub‑gradient (sum of contributions from all maximal diagonals)
        b = _full_convolution(a)
        max_b = b.max()
        idxs = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in idxs:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step‑size that decays like 1/√t
        step = step0 / math.sqrt(it + 1.0)

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val + 1e-12 < best_val:
            best_val, best_a = new_val, a_new.copy()
            print(f"[SubG] iter {it:06d} new best = {best_val:.6f}")

        # accept the move if it improves (otherwise stay)
        if new_val < cur_val:
            a = a_new
            cur_val = new_val
        else:
            # occasional jitter to get out of tiny plateaus
            if (it % 500 == 0) and (np.random.rand() < 0.03):
                a = np.maximum(a + np.random.normal(scale=0.012, size=n), 0.0)
                a = _project_simplex(a)
                cur_val = _objective(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def _candidate_exponential(n: int, r: float) -> np.ndarray:
    """Geometric‑decay candidate a_i ∝ r**i, normalised to sum 1."""
    if r <= 0.0:
        return np.full(n, 1.0 / n)
    if r >= 1.0 - 1e-12:
        return np.full(n, 1.0 / n)
    i = np.arange(n, dtype=np.float64)
    # use stable formula for the geometric series sum
    r_pow_n = r ** n
    denom = (1.0 - r_pow_n) / (1.0 - r)
    a = r ** i / denom
    return a


def _candidate_powerlaw(n: int, alpha: float) -> np.ndarray:
    """Power‑law candidate a_i ∝ (i+1)**(-alpha), normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-alpha)
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    return a / a_sum


def _candidate_beta(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑shape a_i ∝ i**(p-1) * (n-i+1)**(q-1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s
