"""
Hybrid optimiser for the self‑convolution max problem.

The routine consists of three stages:

1️⃣  *Parameter sweep* – a cheap systematic search over a few
    analytically‑motivated families (linear decay, power‑law,
    exponential, and mixtures).  The best of these simple constructions
    is used as a warm‑start.

2️⃣  *Long Adam optimisation* – the coefficients are represented in
    log‑softmax form (logits `z`).  The smooth‑max temperature
    λ grows from a modest value to a very large one, which makes the
    gradient focus on the true maximum.  A relatively large number of
    Adam steps (up to ~80 000 per run) are performed, with occasional
    Gaussian jitter when progress stalls.

3️⃣  *Projected sub‑gradient refinement* – after each Adam run a cheap
    exact‑gradient (single active shift) step with an adaptive step‑size
    and back‑tracking line‑search is applied.  This often squeezes a
    little more out of the current best solution.

The whole optimisation respects the 1000 s budget and finally returns
the best sequence found.  The returned list is ready to be handed to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2*len(x)-1, computed via FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of weight vector w with a.
    Returns a vector of length len(a) defined by
        g_i = Σ_k w_k * a_{k-i}
    (indices outside [0,len(a)-1] are treated as zero).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1                         # indices n‑1 … n‑1+n‑1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """
    Compute 2 * n * max(self‑convolution) for a probability vector a.
    The vector `a` is assumed to sum to 1 (or close to it); the
    scaling invariant nature of the original problem means this is
    sufficient for comparison.
    """
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims
    to minimise `evaluate_sequence`.  The routine runs a light
    parametric sweep, follows it with a long Adam optimisation using a
    smooth‑max surrogate, and finishes with a few exact
    projected sub‑gradient refinements.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise from the previous construction if possible.
    # --------------------------------------------------------------
    try:
        old_seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old_seq.size > 1:
            # linear interpolation onto the new grid
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_seq.size)
            a0 = np.interp(new_x, old_x, old_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s < 1e-12:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / s
            print("[init] interpolated previous solution")
        else:
            raise Exception
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[global] start objective = {best_val:.9f}")

    # --------------------------------------------------------------
    # 1️⃣  Parameter sweep – cheap families of analytically‑motivated
    #      shapes (linear decay, power‑law, exponential, mixtures).
    # --------------------------------------------------------------
    def _maybe_update(candidate: np.ndarray):
        """Update the global best if `candidate` improves."""
        nonlocal best_a, best_val
        cand = candidate / np.maximum(candidate.sum(), 1e-12)   # normalise
        val = _objective(cand, n)
        if val < best_val - 1e-12:
            best_val = val
            best_a = cand.copy()
            print(f"[param] new best = {val:.9f}")

    # linear decreasing (γ = 1 gives the simple triangle)
    for gamma in np.linspace(0.0, 5.0, 31):
        a = ((np.arange(n, 0, -1) / n) ** gamma)
        _maybe_update(a)

    # power‑law decreasing a_i ∝ (i+1)^{-p}
    for p in np.linspace(0.0, 3.0, 31):
        a = (np.arange(1, n + 1) ** (-p))
        _maybe_update(a)

    # exponential decay a_i ∝ exp(-α·i/n)
    for alpha in np.linspace(0.0, 8.0, 41):
        a = np.exp(-alpha * np.arange(n) / n)
        _maybe_update(a)

    # mixtures of the above with the uniform distribution
    for gamma in np.linspace(0.5, 3.0, 5):
        shape = ((np.arange(n, 0, -1) / n) ** gamma)
        for lam in np.linspace(0.1, 0.5, 5):
            a = lam * shape + (1.0 - lam) * np.ones(n)
            _maybe_update(a)

    print(f"[global] after sweep best = {best_val:.9f}")

    # --------------------------------------------------------------
    # 2️⃣  Long Adam optimisation with a growing smooth‑max temperature.
    # --------------------------------------------------------------
    lambda_start = 5.0          # start temperature
    lambda_end   = 20000.0      # final (very sharp) temperature
    adam_lr      = 0.06         # base learning rate
    b1, b2       = 0.9, 0.999   # Adam moments
    adam_eps     = 1e-8

    # Main optimisation loop – we keep restarting from the current
    # best but add a little noise when progress stalls.
    while time.time() < deadline:
        # initialise a fresh run from the current global best (+ jitter)
        eps = 1e-12
        z = np.log(best_a + eps) + np.random.normal(scale=0.025, size=n)

        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t_iter = 0
        stagnation = 0
        IMPROVE_LIMIT = 4000  # after this many non‑improving steps we inject noise

        # inner Adam loop
        while time.time() < deadline:
            if t_iter >= 80000:      # generous upper bound per run
                break
            t_iter += 1

            # ---- logits → probability vector (soft‑max) ----
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            a = exp_z / exp_z.sum()

            # ---- evaluate exact objective ----
            conv = _full_convolution(a)
            conv_max = conv.max()
            cur_val = 2.0 * n * conv_max

            # ---- update global best if we improved ----
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                stagnation = 0
                now = time.time()
                print(f"[iter {t_iter:06d}] new best = {best_val:.9f} (t={now - start_time:.1f}s)")
            else:
                stagnation += 1

            # ---- smooth‑max temperature (linear schedule) ----
            prog = min(1.0, (time.time() - start_time) / (deadline - start_time))
            lam = lambda_start * (1.0 - prog) + lambda_end * prog

            # ---- compute weighting w for the (smooth) max ----
            if lam > 15000.0:
                # exact max – single active index
                kmax = int(np.argmax(conv))
                w = np.zeros_like(conv)
                w[kmax] = 1.0
            else:
                # smooth max via a soft‑max over the convolution entries
                exp_vals = np.exp(lam * (conv - conv_max))
                w = exp_vals / exp_vals.sum()

            # ---- gradient of the (smooth) max w.r.t. a ----
            # g_raw = cross‑correlation(w, a)   →   2·a_{kmax‑i} when w is one‑hot
            g_raw = _cross_corr(w, a)
            g = 2.0 * g_raw

            # ---- back‑prop through the soft‑max (logits) ----
            dot_ag = np.dot(a, g)
            grad_z = a * (g - dot_ag)

            # ---- Adam update ----
            m = b1 * m + (1.0 - b1) * grad_z
            v = b2 * v + (1.0 - b2) * (grad_z ** 2)
            m_hat = m / (1.0 - b1 ** t_iter)
            v_hat = v / (1.0 - b2 ** t_iter)
            z -= adam_lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # ---- occasional jitter when stuck ----
            if stagnation > IMPROVE_LIMIT:
                z += np.random.normal(scale=0.07, size=n)
                stagnation = 0
                print(f"[reset] added Gaussian noise at iter {t_iter}")

        # ----------------------------------------------------------------
        # 3️⃣  Exact projected sub‑gradient sweep (cheap refinement)
        # ----------------------------------------------------------------
        a = best_a.copy()
        obj = best_val
        step = 0.03  # initial step size

        for _ in range(1500):
            if time.time() >= deadline:
                break
            conv = _full_convolution(a)
            kmax = int(np.argmax(conv))

            # sub‑gradient of the true max (vector g_i = 2·a_{kmax‑i})
            g = np.zeros_like(a)
            if kmax < n:
                # overlap region: indices 0 … kmax
                g[:kmax + 1] = 2.0 * a[kmax::-1]
            else:
                # overlap region: indices kmax‑(n‑1) … n‑1
                offset = kmax - (n - 1)
                L = n - offset
                g[offset:] = 2.0 * a[:L][::-1]

            # try a back‑tracking line search (step, step/2, step*2)
            improved = False
            for trial_step in (step, step * 0.5, step * 2.0):
                a_trial = _project_simplex(a - trial_step * g)
                obj_trial = 2.0 * n * _full_convolution(a_trial).max()
                if obj_trial < obj - 1e-12:
                    a = a_trial
                    obj = obj_trial
                    best_a = a.copy()
                    best_val = obj
                    step = min(step * 1.2, 0.5)  # gently increase after success
                    improved = True
                    print(f"[subgrad] improved to {obj:.9f}")
                    break
            if not improved:
                step *= 0.7               # shrink step if no progress

            if obj < 1.503:                # early exit when target reached
                break

        # stop early if we already beat the target
        if best_val < 1.503:
            break

    # --------------------------------------------------------------
    # 4️⃣  Final secure evaluation (the only place we call
    #     `evaluate_sequence`).  The result is printed for logging.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return the best sequence found (as a plain Python list)
    return best_a.tolist()
