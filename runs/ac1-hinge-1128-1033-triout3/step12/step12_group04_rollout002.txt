"""
Propose a sequence that (approximately) minimizes the L∞ self‑convolution
objective used by ``evaluate_sequence``.
Algorithm
---------
* Initialise a feasible vector on the probability simplex (uniform or,
  if available, the previous best ``height_sequence_1`` interpolated to the
  target length).
* Optimise the smooth approximation of the max‑convolution:
      L(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
  The gradient of L is
      ∇L_i = 2· Σ_k w_k·a_{k‑i},
  where w_k = exp(β·c_k)/Σ_j exp(β·c_j) and c = a*a.
* Use the Adam optimiser (adaptive first‑/second‑moment estimates) to
  update the coefficients, projecting the result back onto the simplex
  after each step.
* Increase the smoothing parameter β gradually (β≈1 → β≈30) so that the
  optimisation first sees a smooth landscape and later focuses on the
  true max‑norm.
* If the search stagnates for many iterations a small Gaussian jitter is
  added and the Adam moments are reset.
* The best vector found during the run is finally evaluated with the
  provided ``evaluate_sequence`` and returned.

The routine respects the 1000 s time budget (a small safety margin is
kept for the final secure call).  It works for any length; we use
``n = 8000`` which matches the reference implementation.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of ``x`` with itself (real FFT).
    Length of the result is 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_rev_weight(rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of the reversed vector ``rev`` with weight vector ``w``
    (full linear convolution, real FFT).
    """
    size = _next_pow2(len(rev) + len(w) - 1)
    fr = np.fft.rfft(rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fr * fw, size)[: len(rev) + len(w) - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """
    Exact value of 2·n·max(conv) (the denominator Σa² is 1 because a lies
    on the simplex).
    """
    return 2.0 * n * conv.max()

        #    g_i = 2· Σ_k w_k·a_{k‑i}
        # -------------------------------------------------------------
        rev_a = a[::-1]                # reversed a, shape (n,)
        conv_rev = _conv_rev_weight(rev_a, w)   # length = 3n‑2
        g = 2.0 * conv_rev[n - 1 : n - 1 + n]    # extract the relevant n entries

        # -------------------------------------------------------------
        # 3) Adam update + projection onto the simplex
        # -------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)

        # -------------------------------------------------------------
        # 4) Exact evaluation for the new point
        # -------------------------------------------------------------
        conv_new = _full_convolution(a_new)
        new_obj = _objective_from_conv(conv_new, n)

        # -------------------------------------------------------------
        # 5) Keep the best point seen so far
        # -------------------------------------------------------------
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            stagnation = 0
            print(f"[iter {iteration:06d}] new best = {best_obj:.9f} "
                  f"(time {time.time() - start_time:.1f}s)")
        else:
            stagnation += 1

        # -------------------------------------------------------------
        # 6) Prepare for next iteration
        # -------------------------------------------------------------
        a = a_new
        conv = conv_new
        cur_obj = new_obj

        # -------------------------------------------------------------
        # 7) Stagnation handling – random jitter & moment reset
        # -------------------------------------------------------------
        if stagnation > max_stagn:
            jitter = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            # reset Adam moments because the gradient history is no longer valid
            m.fill(0.0)
            v.fill(0.0)
            conv = _full_convolution(a)
            cur_obj = _objective_from_conv(conv, n)
            stagnation = 0
            print(f"[reset] jitter at iter {iteration} (time {time.time() - start_time:.1f}s)")

        # -------------------------------------------------------------
        # 8) Occasionally accept a worsening move (helps escape flat plateaus)
        # -------------------------------------------------------------
        if stagnation > max_stagn // 2 and np.random.rand() < 0.001:
            a = a_new
            conv = conv_new
            cur_obj = new_obj

    # -----------------------------------------------------------------
    # Final secure evaluation (single call to the provided function)
    # -----------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score =", final_score)
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = float("inf")

    # Return the best raw sequence (the harness will invoke evaluate_sequence on it)
    return best_a.tolist()
