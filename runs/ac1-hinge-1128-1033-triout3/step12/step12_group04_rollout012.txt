#!/usr/bin/env python
"""
Adam‑based smooth‑max optimisation for the L∞ self‑convolution problem.

The algorithm proceeds in two phases:

1.  **Smooth‑max descent** –  minimise a differentiable
    approximation of  max_k (a*a)_k  using the soft‑max
    (log‑sum‑exp) with an increasing temperature parameter.
    Adam (adaptive moment estimation) is employed and each
    iterate is projected onto the probability simplex
    (non‑negative entries summing to 1).

2.  **Hard‑max refinement** –  after the smooth phase the
    current best vector is handed to a plain sub‑gradient
    descent (the same scheme as the reference implementation)
    for a few hundred thousand additional steps.  This refines
    the solution w.r.t. the true L∞ objective.

Both phases respect the 1000 s time budget.  The best vector
encountered is finally evaluated with the supplied
``evaluate_sequence`` function and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution x*x (real FFT, length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w ≥ 0, Σ w = 1}."""
    # implementation of Algorithm 1 from
    #   https://arxiv.org/abs/1101.6081
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return the soft‑max weights for a given convolution vector.
    We compute w_i ∝ exp(alpha * (conv_i - max(conv))) to keep the
    exponent well‑scaled.
    """
    max_c = conv.max()
    # shift for numerical stability
    shifted = conv - max_c
    exp_scaled = np.exp(alpha * shifted)
    w = exp_scaled / exp_scaled.sum()
    return w


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k (a*a)_k = 2 * (w ⋆ a) .
    The cross‑correlation is obtained via a single FFT‑based convolution.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    size = _next_pow2(len(w) + len(a_rev) - 1)
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a_rev, size)
    c_full = np.fft.ifft(fw * fa_rev, size).real
    # g_i = 2 * c_full[n-1 + i] for i = 0 … n-1
    start = n - 1
    g = 2.0 * c_full[start:start + n]
    return g


def _subgradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One sub‑gradient step for the true max‑objective.
    Returns the projected new iterate.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k_max = int(conv.argmax())
    # build sub‑gradient of the maximal entry
    g = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    counterpart = k_max - idx
    g[idx] = 2.0 * a[counterpart]
    # gradient descent + simplex projection
    v = a - eta * g
    a_new = _project_simplex(v)
    return a_new


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    Returns a list of floats (the best sequence discovered before the time
    budget expires).  The routine uses a two‑phase Adam/soft‑max optimisation
    followed by a sub‑gradient refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size – 8192 is a power of two and works well with FFT.
    # ------------------------------------------------------------------
    n = 8192

    # ------------------------------------------------------------------
    # initialise – try to reuse a previously found solution if present.
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined in the global scope of the runner.
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        x_old = np.linspace(0.0, 1.0, old_len)
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total
        print("[init] interpolated previous solution (len:", old_len, "→", n, ")")
    except Exception:
        a = np.full(n, 1.0 / n)
        print("[init] uniform start, n =", n)

    # keep a copy of the best vector seen so far
    best_a = a.copy()
    cur_conv = _full_convolution(a)
    best_val = 2.0 * n * cur_conv.max()
    print("[init] objective =", best_val)

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max optimisation (Adam)
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)

    # step‑size schedule (initial, then exponential decay)
    step_size = 0.05
    step_decay = 0.9995
    step_min = 0.0005

    # soft‑max temperature schedule (α grows slowly)
    alpha = 5.0          # start with a fairly smooth approximation
    alpha_max = 50.0
    alpha_rate = 1.001   # multiplicative increase per iteration

    # stagnation detection for occasional jitter
    stagnation = 0
    max_stagn = 3000

    # decide how long to spend in phase 1 (≈70 % of the budget)
    phase1_deadline = start_time + 0.70 * (budget_s - 0.5)

    it = 0
    while time.time() < deadline:
        it += 1
        # --------------------------------------------------------------
        # common data: convolution of the current iterate
        # --------------------------------------------------------------
        conv = _full_convolution(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()
            stagnation = 0
            print(f"[iter {it:06d}] new best = {best_val:.9f} "
                  f"(time {time.time() - start_time:.1f}s)")
        else:
            stagnation += 1

        # --------------------------------------------------------------
        # choose the optimisation method according to the elapsed time
        # --------------------------------------------------------------
        now = time.time()
        if now < phase1_deadline:
            # ----- Smooth‑max (Adam) ------------------------------------
            w = _softmax_weights(conv, alpha)

            # gradient of the smooth approximation
            g = _gradient_softmax(a, w)

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a = a - step_size * m_hat / (np.sqrt(v_hat) + eps)

            # project back onto the probability simplex
            a = _project_simplex(a)

            # schedules
            step_size = max(step_size * step_decay, step_min)
            alpha = min(alpha * alpha_rate, alpha_max)

        else:
            # ----- Hard‑max sub‑gradient (same as reference) ----------
            # sub‑gradient step with a modest fixed step size
            eta = max(0.01, step_size)   # reuse the current step size
            a = _subgradient_step(a, eta)

        # --------------------------------------------------------------
        # occasional jitter to escape flat plateaus
        # --------------------------------------------------------------
        if stagnation > max_stagn:
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            # recompute objective after the perturbation
            conv = _full_convolution(a)
            cur_obj = 2.0 * n * conv.max()
            stagnation = 0
            print(f"[jitter] at iter {it} (time {time.time() - start_time:.1f}s)")

        # --------------------------------------------------------------
        # early exit if we are out of time
        # --------------------------------------------------------------
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – final refinement with a pure sub‑gradient loop.
    #          (Only a few thousand iterations – cheap but can shave a
    #          tiny amount off the objective.)
    # ------------------------------------------------------------------
    # Reset schedule for a short aggressive sub‑gradient run.
    eta = 0.04
    decay = 0.9999
    eta_min = 0.001
    stagnation = 0
    max_stagn = 2000

    while time.time() < deadline:
        conv = _full_convolution(a)
        k_max = int(conv.argmax())
        # sub‑gradient of the maximal entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        counterpart = k_max - idx
        g[idx] = 2.0 * a[counterpart]

        v = a - eta * g
        a = _project_simplex(v)

        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()
            stagnation = 0
            print(f"[refine {it:06d}] new best = {best_val:.9f} "
                  f"(time {time.time() - start_time:.1f}s)")
        else:
            stagnation += 1

        eta = max(eta * decay, eta_min)

        if stagnation > max_stagn:
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            stagnation = 0
            print(f"[refine jitter] at time {time.time() - start_time:.1f}s")

        # check deadline
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final evaluation with the provided function (single call)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score =", final_score)
    except Exception as exc:                         # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
