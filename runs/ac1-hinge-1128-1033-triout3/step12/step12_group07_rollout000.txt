"""
Propose a non‑negative step‑function (a simplex‑constrained vector) that
minimises 2·n·max(conv(a,a)) / (sum(a)²).  

The algorithm combines a fast systematic initialisation (Beta, Dirichlet,
edge‑weighted power‑laws, exponential decays) with a long
hill‑climbing / simulated‑annealing phase, followed by several gradient‑type
polishing steps (Adam, sub‑gradient, exponentiated‑gradient).  The
budget (default 1000 s) is split dynamically: a small fraction for
candidate generation, the bulk for hill‑climbing, and the remainder for
polishing.  The code respects the security checks of ``evaluate_sequence``,
returns a plain list of floats and prints progress.
"""

import time
import numpy as np
from typing import Tuple

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> Tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a (includes the 2·n factor).
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp smoothing
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # gradient for 2·n·smooth_max
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    log_r = rng.uniform(-8.0, -0.1)
    r = np.exp(log_r)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _mixture_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    w = rng.random()
    a1 = _random_beta_candidate(n, rng)
    a2 = _random_beta_candidate(n, rng)
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


def _edge_powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """Symmetric edge‑heavy shape a_i ∝ (i·(n+1‑i))**p (p may be negative)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i * (n + 1 - i)) ** p
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exp_decay_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric exponential that grows towards the edges."""
    alpha = rng.uniform(0.001, 0.01)   # moderate edge emphasis
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(alpha * np.abs(i - center))
    a /= a.sum()
    return a


def _sinusoid_candidate(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _cosine_sq_candidate(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> Tuple[np.ndarray, float]:
    """Adam optimisation on a smooth approximation of max(conv)."""
    w = np.log(a0 + 1e-12)                     # work in log‑softmax space
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.6, 0.002              # a little more aggressive
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 12000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule
        prog = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** prog
        eps = eps_start * (eps_end / eps_start) ** prog

        # smooth max & gradient
        _, grad = _smooth_max_and_grad(a, eps)

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.025, size=n)

        # restart if stalled too long
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> Tuple[np.ndarray, float]:
    """
    Aggressive source‑→‑target hill‑climbing with simulated‑annealing acceptance.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.85, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 30000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # current maximal convolution index
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source: biased by its contribution to the peak
        if total > 0.0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target: favour positions with small contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # fast convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # re‑perturb from the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        max_pos = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in max_pos:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step size schedule (linear decay)
        prog = (now - start_time) / (deadline - start_time)
        step = 0.25 * (1.0 - prog) + 1e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to escape flat spots
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


def _expgrad_polish(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    max_iters: int = 150_000) -> Tuple[np.ndarray, float]:
    """Exponentiated‑gradient polishing for the exact max‑convolution objective."""
    a = a0.copy()
    cur_val = _objective(a)

    eta_start, eta_end = 1.0, 1e-5

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        max_pos = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in max_pos:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        prog = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - prog) + eta_end * prog

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        new_val = _objective(a)

        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[ExpG] new best = {best_val:.6f}")

        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.018, size=n), 0.0)
            a /= a.sum()

    return best_a, best_val


def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> Tuple[np.ndarray, float]:
    """Try convex combinations with the uniform distribution."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


def _random_anneal(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> Tuple[np.ndarray, float]:
    """Simple simulated annealing with relatively large moves."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)            # linear cooling
        max_frac = 0.12 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # occasional jitter for diversity
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a plain list (sum≈1) that (hopefully) yields a low
    ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Initial population (quick systematic + random)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # Try to reuse a previously discovered solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # ---------- Random Beta draws ----------
    for _ in range(800):
        if time.time() >= deadline:
            break
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ---------- Systematic Beta grid ----------
    logp_vals = np.linspace(-7.5, 2.0, 11)
    logq_vals = np.linspace(-7.5, 2.0, 11)
    for lp in logp_vals:
        for lq in logq_vals:
            if time.time() >= deadline:
                break
            cand = _beta_grid_candidate(n, lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ---------- Dirichlet draws ----------
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(15):
            if time.time() >= deadline:
                break
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ---------- Exponential family ----------
    for _ in range(300):
        if time.time() >= deadline:
            break
        cand = _exponential_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ---------- Power‑law family ----------
    for p in np.linspace(-4.0, 4.0, 15):
        if time.time() >= deadline:
            break
        cand = _powerlaw_candidate(n, p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ---------- Edge‑weighted power‑law ----------
    for p in (-4.0, -3.0, -2.5, -2.0, -1.5, -1.0, -0.8, -0.6):
        if time.time() >= deadline:
            break
        cand = _edge_powerlaw_candidate(n, p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ---------- Exponential decay (edge‑heavy) ----------
    for _ in range(200):
        if time.time() >= deadline:
            break
        cand = _exp_decay_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ---------- Simple sinusoid / cosine‑square ----------
    cand = _sinusoid_candidate(n)
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    cand = _cosine_sq_candidate(n)
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    print(f"[Init] best after exhaustive init = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam optimisation (≈10 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.10 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase B – Aggressive hill‑climbing (≈65 % of the budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.75 * budget_s
    best_a, best_val = _hillclimb(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase C – Sub‑gradient polishing (≈10 % of the budget)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.85 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase_c_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=400_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase D – Mirror‑descent (exponentiated‑gradient) polishing (≈5 % of budget)
    # --------------------------------------------------------------
    phase_d_deadline = start_time + 0.90 * budget_s
    best_a, best_val = _expgrad_polish(
        best_a, n,
        deadline=phase_d_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=120_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase E – Final simulated‑annealing with larger moves (≈5 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_anneal(
            best_a, n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            rng=rng,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return a plain list of floats
    return best_a.tolist()
