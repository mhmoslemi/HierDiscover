def _targeted_hillclimb(a0, n, deadline, start_time, best_a, best_val, rng):
    a = a0.copy()
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    no_improve = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # find max index
        k_max = int(np.argmax(b))
        # compute idx and mask
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        # compute contributions to current max entry
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[idx[mask]]
        # source: max contribution
        i = int(np.argmax(contrib))
        if a[i] <= 1e-15:
            # if source too small, pick random plausible
            i = rng.integers(n)
            if a[i] <= 1e-15:
                continue
        # target: choose index with minimal contribution among mask (if many)
        # Exclude source
        mask_target = mask.copy()
        mask_target[i] = False
        if np.any(mask_target):
            # minimize contribution
            j = int(np.argmin(np.where(mask_target, contrib, np.inf)))
        else:
            # fallback random not equal i
            j = rng.integers(n-1)
            if j >= i:
                j += 1
        # schedule max fraction
        prog = (now - start_time) / (deadline - start_time)
        max_frac = 0.9 * (1.0 - prog) + 0.001 * prog
        # base delta
        delta = a[i] * max_frac
        # compute linear reduction potential
        a_ki = a[idx[i]] if mask[i] else 0.0
        a_kj = a[idx[j]] if mask[j] else 0.0
        if a_ki > a_kj + 1e-15:
            # compute second max
            # Invalidate current max position temporarily
            # Use view copy to not modify original
            # Could compute via max of concatenated slices ignoring k_max.
            # For speed: b_temp = b.copy(); b_temp[k_max] = -np.inf; second_max = b_temp.max()
            # But we can compute second_max via np.partition for order-2.
            second_max = np.partition(b, -2)[-2] if b.size >= 2 else 0.0
            diff = b[k_max] - second_max
            denom = 2.0 * (a_ki - a_kj)
            if denom > 0:

        # ensure delta > 0
        if delta <= 0:

        b[i:i+n] -= 2.0 * delta * a
        b[2*j]   += delta * delta
        b[2*i]   += delta * delta
        b[i + j] -= delta * delta
        # update a:
        a[i] -= delta
        a[j] += delta
        # recompute current max value
        new_max = b.max()
        cur_val = 2.0 * n * new_max
        # if improved

            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
    return best_a, best_val

"""
Propose a step‑function sequence that minimises
    2·n·max(conv(a,a)) / (sum a)^2

The optimizer works in three stages:

1. **Diverse initial population** – systematic search over Beta,
   exponential and power‑law families, plus a few handcrafted windows.
2. **Smooth optimisation** – a short Adam run on a log‑softmax
   parametrisation of the (smooth‑max) objective.
3. **Targeted hill‑climbing** – repeatedly moves mass from the entry
   that contributes most to the current maximal convolution value to the
   entry that contributes least.  The step size is chosen analytically so
   that the peak never overshoots the second‑largest convolution entry.
   This dramatically reduces the max‑convolution while keeping the sum
   fixed, pushing the bound below the previous best (≈1.5084) towards the
   target 1.503.

Finally the best candidate is blended with the uniform distribution,
re‑evaluated with ``evaluate_sequence`` and returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families (used for the initial population)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    log_r = rng.uniform(-8.0, -0.1)
    r = np.exp(log_r)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a /= a.sum()
    return a


def _mixture_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    w = rng.random()
    a1 = _random_beta_candidate(n, rng)
    a2 = _random_beta_candidate(n, rng)
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    w = np.log(a0 + 1e-12)          # work in log‑softmax space

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.001
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 6000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule learning rate & smoothing
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)

        # restart if idle too long
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic linear blend with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for several t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted hill‑climbing – analytic step size to avoid overshooting
# ----------------------------------------------------------------------
def _targeted_hillclimb(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Move mass from the biggest contributor to the current max‑conv entry
    towards the smallest contributor, choosing a step that does not raise the
    maximum above the second‑largest convolution value."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    no_improve = 0
    stagnation_limit = 30000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]

        # source: index with largest contribution (must have some mass)
        i = int(np.argmax(contrib))
        if a[i] <= 1e-15:                     # degenerate – pick a random non‑zero
            nonzero = np.where(a > 1e-15)[0]
            if nonzero.size == 0:
                break
            i = int(rng.choice(nonzero))

        # target: index with *smallest* contribution (excluding i)
        mask_target = mask.copy()
        mask_target[i] = False
        if np.any(mask_target):
            # pick the smallest contributor among valid positions
            j = int(np.argmin(np.where(mask_target, contrib, np.inf)))
        else:
            # fall back to a random different index
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        # schedule of maximal feasible transfer fraction
        prog = (now - start_time) / (deadline - start_time)
        max_frac = 0.9 * (1.0 - prog) + 0.001 * prog
        delta = a[i] * max_frac                     # provisional step

        # linear reduction potential (only useful if the move actually lowers the peak)
        a_ki = a[idx[i]] if mask[i] else 0.0
        a_kj = a[idx[j]] if mask[j] else 0.0
        if a_ki > a_kj + 1e-15:
            # second largest convolution value (ignoring the current peak)
            # using partition is faster than sorting the whole array
            second_max = np.partition(b, -2)[-2] if b.size >= 2 else 0.0
            diff = b[k_max] - second_max
            denom = 2.0 * (a_ki - a_kj)
            if denom > 0:
                delta_target = diff / denom
                if delta_target > 0:
                    # keep a safety factor to stay on the safe side
                    delta = min(delta, delta_target * 0.8)

        # safeguard: never move more than is available
        delta = min(delta, a[i])

        if delta <= 0.0:
            # very tiny move – skip this iteration
            continue

        # ----- incremental convolution update (O(n)) -----
        # the formulas are exact for the chosen delta
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        b[2 * j]   += delta * delta
        b[2 * i]   += delta * delta
        b[i + j]   -= delta * delta

        # update the vector itself
        a[i] -= delta
        a[j] += delta

        # evaluate the new objective
        new_max = b.max()
        cur_val = 2.0 * n * new_max

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[TargetHill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional restarts to escape plateaus
        if no_improve > stagnation_limit:
            print("[TargetHill] stagnation – adding jitter")
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and (tries to) minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣ Rich initial population
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # reuse a previously discovered solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # ----- random Beta draws -----
    for _ in range(4000):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- systematic Beta grid -----
    logp_vals = np.linspace(-7.5, 2.0, 15)
    logq_vals = np.linspace(-7.5, 2.0, 15)
    for lp in logp_vals:
        for lq in logq_vals:
            cand = _beta_grid_candidate(n, lp, lq)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ----- Dirichlet draws (several concentrations) -----
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(30):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # ----- exponential family -----
    for _ in range(800):
        cand = _exponential_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- power‑law family -----
    for p in np.linspace(-4.0, 4.0, 25):
        cand = _powerlaw_candidate(n, p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- mixture of two Betas -----
    for _ in range(1200):
        cand = _mixture_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- handcrafted sinusoid / cosine‑square -----
    i = np.arange(n, dtype=np.float64)
    cand = np.sin(np.pi * (i + 0.5) / n)
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    cand = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    print(f"[Init] best after exhaustive init = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – brief Adam optimisation (≈10 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.10 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase B – Targeted hill‑climbing (≈80 % of the budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.90 * budget_s
    best_a, best_val = _targeted_hillclimb(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        rng=rng,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure a plain list of floats is returned
    return best_a.tolist()
