"""
Optimiser for the “step‑function attack”.

We search for a non‑negative vector a (∑a = 1) that minimises

    F(a) = 2·n·max(conv(a,a))

which is exactly the value returned by ``evaluate_sequence`` after the
normalisation performed by the verifier.

Main ideas
---------
* Try several problem sizes (n = 8000 and n = 20000).  The theoretical
  lower bound for the objective is 1, and larger n usually allow a
  tighter bound.
* Initialise the population with a rich family of analytical shapes
  (Beta, exponential, power‑law, edge‑concentrated, triangular,
  sparse Dirichlet etc.) and a systematic grid over a few parameters.
* Refine the best candidate of each size with a sequence of simple,
  cheap optimisation phases that have proven effective on this task:
  – Adam on a smooth approximation of the max‑convolution (log‑sum‑exp)
  – aggressive source→target hill‑climbing
  – projected sub‑gradient descent on the exact max
  – exponentiated‑gradient (mirror descent)
  – final large‑move simulated annealing.
* All phases respect the simplex constraint (∑a = 1, a ≥ 0) by
  projecting onto the probability simplex.
* The whole search is split into equal time slices for the two problem
  sizes; the best vector over all sizes is returned.

The code is written without closures or lambdas and uses only NumPy (and
time for budgeting).  ``evaluate_sequence`` is assumed to be already
imported in the execution environment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) (∑a = 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed weight vector
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # already includes 2·n factor
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families (all return a vector with sum = 1)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-7.5, 2.0)
    logq = rng.uniform(-7.5, 2.0)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _beta_grid_candidate(n: int, logp: float, logq: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    log_r = rng.uniform(-8.0, -0.1)
    r = np.exp(log_r)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _edge_candidate(n: int, p: float) -> np.ndarray:
    """Mass concentrated near the two ends: a_i ∝ (i+1)^(-p)+(n-i)^(-p)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _triangular_candidate(n: int) -> np.ndarray:
    """Symmetric triangle, peak at centre."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - (n - 1) / 2.0) * 2.0 / (n - 1)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _sparse_random_candidate(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Place mass on k random positions (k << n)."""
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.random(k)
    weights /= weights.sum()
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on 2·n·smooth_max."""
    w = np.log(a0 + 1e-12)          # work in log‑softmax space

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.001
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 4000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
        else:
            idle += 1

        # schedule
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max gradient
        _, grad = _smooth_max_and_grad(a, eps)

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)

        if idle > MAX_IDLE:
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (source→target moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray, n: int,
               deadline: float, start_time: float,
               best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Hill‑climbing with simulated‑annealing acceptance."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.9, 0.001
    temp_start, temp_end = 0.03, 1e-9
    no_improve = 0
    IMPROVE_LIMIT = 40000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # locate the currently maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        if total > 0.0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index favours low‑contribution coordinates
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # fast update of the full convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # restart from the current global best with a little jitter
            a = best_a.copy()
            a += np.random.normal(scale=0.03, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on exact max‑convolution.
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, n: int,
                        deadline: float, start_time: float,
                        best_a: np.ndarray, best_val: float,
                        max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule (decrease to 1e‑5)
        prog = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - prog) + 1e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()

        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------
def _expgrad_polish(a0: np.ndarray, n: int,
                    deadline: float, start_time: float,
                    best_a: np.ndarray, best_val: float,
                    max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient updates for the exact max‑convolution."""
    a = a0.copy()
    cur_val = _objective(a)

    eta_start, eta_end = 1.0, 1e-5

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        prog = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - prog) + eta_end * prog

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        new_val = _objective(a)

        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()

        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.018, size=n), 0.0)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray, n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a+(1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Larger‑move simulated‑annealing (final phase)
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   best_a: np.ndarray, best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple annealing with relatively large source→target transfers."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()

        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Run a multi‑size, multi‑phase optimiser under a 1000 s wall‑clock budget.
    Returns a list of non‑negative numbers (∑=1) that minimises
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a few problem sizes; larger sizes usually admit smaller bounds.
    # ------------------------------------------------------------------
    sizes = [8000, 20000]                     # you can add more sizes if time permits
    per_size_budget = (budget_s - 0.5) / len(sizes)

    global_best_val = np.inf
    global_best_seq = None

    for size_idx, n in enumerate(sizes):
        size_start = time.time()
        size_deadline = min(deadline, size_start + per_size_budget)

        # --------------------------------------------------------------
        # Initial population (systematic + random)
        # --------------------------------------------------------------
        best_a = np.full(n, 1.0 / n)            # uniform start
        best_val = _objective(best_a)

        # try to reuse a previous solution if it exists
        try:
            old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            old_len = old.shape[0]
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            val = _objective(a0)
            if val < best_val:
                best_a, best_val = a0, val
        except Exception:
            pass

        # ----- systematic beta grid -----
        logp_vals = np.linspace(-7.0, 2.0, 9)
        logq_vals = np.linspace(-7.0, 2.0, 9)
        for lp in logp_vals:
            for lq in logq_vals:
                cand = _beta_grid_candidate(n, lp, lq)
                val = _objective(cand)
                if val < best_val:
                    best_a, best_val = cand, val

        # ----- random Beta draws -----
        num_beta = max(500, 2000 // (n // 8000))   # fewer for larger n
        for _ in range(num_beta):
            cand = _random_beta_candidate(n, rng)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

        # ----- exponential family -----
        for _ in range(200):
            cand = _exponential_candidate(n, rng)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

        # ----- power‑law family -----
        for p in np.linspace(-4.0, 4.0, 17):
            cand = _powerlaw_candidate(n, p)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

        # ----- edge‑concentrated family -----
        for p in np.linspace(0.5, 5.0, 11):
            cand = _edge_candidate(n, p)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

        # ----- triangular -----
        cand = _triangular_candidate(n)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

        # ----- sparse Dirichlet draws (few non‑zeros) -----
        for k_factor in (0.02, 0.05):      # fraction of non‑zero entries
            k = max(2, int(k_factor * n))
            for _ in range(30):
                cand = _sparse_random_candidate(n, k, rng)
                val = _objective(cand)
                if val < best_val:
                    best_a, best_val = cand, val

        # ----- low‑alpha Dirichlet draws (highly random) -----
        for alpha in (0.005, 0.02, 0.05):
            for _ in range(30):
                cand = rng.dirichlet(np.full(n, alpha))
                val = _objective(cand)
                if val < best_val:
                    best_a, best_val = cand, val

        print(f"[Init] n={n}, best init = {best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase A – Adam optimisation (≈15 % of the size‑budget)
        # ------------------------------------------------------------------
        phase_a_deadline = min(size_deadline,
                               size_start + 0.15 * (size_deadline - size_start))
        best_a, best_val = _adam_optimize(best_a, n,
                                          deadline=phase_a_deadline,
                                          start_time=size_start,
                                          best_a=best_a,
                                          best_val=best_val)
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

        # ------------------------------------------------------------------
        # Phase B – Hill‑climbing (≈45 % of the size‑budget)
        # ------------------------------------------------------------------
        phase_b_deadline = min(size_deadline,
                               size_start + 0.60 * (size_deadline - size_start))
        best_a, best_val = _hillclimb(best_a, n,
                                      deadline=phase_b_deadline,
                                      start_time=size_start,
                                      best_a=best_a,
                                      best_val=best_val)
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

        # ------------------------------------------------------------------
        # Phase C – Sub‑gradient polishing (≈15 % of the size‑budget)
        # ------------------------------------------------------------------
        phase_c_deadline = min(size_deadline,
                               size_start + 0.75 * (size_deadline - size_start))
        best_a, best_val = _subgradient_polish(best_a, n,
                                               deadline=phase_c_deadline,
                                               start_time=size_start,
                                               best_a=best_a,
                                               best_val=best_val,
                                               max_iters=200_000)
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

        # ------------------------------------------------------------------
        # Phase D – Mirror‑descent (exponentiated‑gradient) polishing
        # ------------------------------------------------------------------
        phase_d_deadline = min(size_deadline,
                               size_start + 0.90 * (size_deadline - size_start))
        best_a, best_val = _expgrad_polish(best_a, n,
                                           deadline=phase_d_deadline,
                                           start_time=size_start,
                                           best_a=best_a,
                                           best_val=best_val,
                                           max_iters=150_000)
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

        # ------------------------------------------------------------------
        # Phase E – Final simulated annealing (remaining time for this size)
        # ------------------------------------------------------------------
        if time.time() < size_deadline:
            best_a, best_val = _random_anneal(best_a, n,
                                              deadline=size_deadline,
                                              start_time=size_start,
                                              best_a=best_a,
                                              best_val=best_val,
                                              rng=rng)
            best_a, best_val = _blend_with_uniform(best_a, n, best_val)

        print(f"[Size {n}] final value = {best_val:.6f}")

        # ------------------------------------------------------------------
        # Keep the best across different problem sizes
        # ------------------------------------------------------------------
        if best_val < global_best_val:
            global_best_val = best_val
            global_best_seq = best_a.copy()
            print(f"[Global] new overall best = {global_best_val:.6f} (n={n})")

        # ------------------------------------------------------------------
        # Move to the next size (if any)
        # ------------------------------------------------------------------
        if time.time() >= deadline:
            break   # budget exhausted

    # ----------------------------------------------------------------------
    # Final evaluation with the provided evaluate_sequence()
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"[Final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return plain Python list of floats
    return global_best_seq.tolist()
