"""
Propose a step‐function (non‑negative list of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

which is exactly the value returned by ``evaluate_sequence`` (the ratio is
scale‑invariant, so we may normalise the vector to sum‑to‑one).

The method consists of three phases:

1.  A quick multi‑family random search (monotone power, Beta, exponential
    and simple piecewise‑power shapes) to obtain a good starting point.
2.  A long continuous optimisation of the normalised log‑coefficients
    using a smooth approximation of the “max’’ (soft‑max with a temperature
    that slowly rises) and an Adam‑style update.  The parameters are kept
    on the probability simplex by exponentiating the log‑variables.
3.  The best vector found during the optimisation is returned; a final
    call to ``evaluate_sequence`` is made only for logging.

The whole routine respects the given time budget (default 1000 s) and
prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT based, all operations are O(n log n)).
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear convolution of real vectors x and y using FFT of length ``size``.
    Returns an array of length len(x)+len(y)-1.
    """
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: x.shape[0] + y.shape[0] - 1]


def _full_convolution_self(a: np.ndarray, size: int) -> np.ndarray:
    """Full convolution a*a (used repeatedly)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * a.shape[0] - 1]


def _objective(a: np.ndarray) -> float:
    """Exact surrogate objective: 2·n·max(conv(a,a)) for a normalised (sum=1)."""
    n = a.shape[0]
    b = _full_convolution_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Parametric families – provide diverse starting points.
# ----------------------------------------------------------------------


def _monotone_power_shape(p: float, n: int) -> np.ndarray | None:
    """a_i ∝ (i+1)^p  (i = 0 … n‑1).  Returned vector is normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Beta‑family: a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_shape(gamma: float, n: int) -> np.ndarray | None:
    """Exponential growth a_i ∝ exp(gamma·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(gamma * i)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def _piecewise_power_shape(p1: float, p2: float, cut: int, n: int) -> np.ndarray | None:
    """
    Piecewise power: a_i ∝ (i+1)^p1 for i < cut,
                     a_i ∝ (i+1)^p2 for i ≥ cut.
    Returned vector is normalised.
    """
    if cut <= 0 or cut >= n:
        return None
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.where(i <= cut, i ** p1, i ** p2)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` score.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # small safety margin

    # --------------------------------------------------------------
    # Problem size – 8000 is a comfortable trade‑off between quality
    # and speed for the FFT based optimisation.
    # --------------------------------------------------------------
    n = 8000
    size_fft = _next_pow2(3 * n - 2)              # enough for a * (a * a) style conv

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a possible previously saved solution.
    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = np.inf

    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        x_new = np.linspace(0.0, 1.0, n)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(x_new, x_old, old)
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        if a0_sum > 0:
            a0 /= a0_sum
            val0 = _objective(a0)
            best_seq, best_val = a0, val0
            print("[init] interpolated previous solution – val={:.6f}".format(val0))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad random search over several parametric families.
    # ------------------------------------------------------------------
    def _maybe_update(candidate: np.ndarray, label: str):
        nonlocal best_seq, best_val
        val = _objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print("[search] new best {} → {:.6f}".format(label, val))

    # uniform baseline (always available)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _maybe_update(uniform, "uniform")

    # grid over monotone power shapes
    for p in np.arange(0.5, 10.0, 0.5):
        cand = _monotone_power_shape(p, n)
        if cand is not None:
            _maybe_update(cand, f"pow(p={p:.1f})")
        if time.time() >= deadline - 200:
            break

    # random samples for the same family
    for _ in range(2000):
        p = rng.normal(5.5, 2.0)
        if p <= 0:
            continue
        cand = _monotone_power_shape(p, n)
        if cand is not None:
            _maybe_update(cand, f"pow(p={p:.2f})")
        if time.time() >= deadline - 200:
            break

    # Beta family (occasionally useful)
    for _ in range(800):
        p, q = rng.uniform(0.5, 10.0, size=2)
        cand = _beta_shape(p, q, n)
        if cand is not None:
            _maybe_update(cand, f"beta(p={p:.1f},q={q:.1f})")
        if time.time() >= deadline - 200:
            break

    # Exponential family
    for _ in range(800):
        gamma = rng.uniform(0.01, 5.0)
        cand = _exp_shape(gamma, n)
        if cand is not None:
            _maybe_update(cand, f"exp(g={gamma:.2f})")
        if time.time() >= deadline - 200:
            break

    # Simple piecewise‑power family
    for _ in range(800):
        cut_frac = rng.uniform(0.2, 0.8)
        cut = int(cut_frac * n)
        p1, p2 = rng.uniform(0.5, 10.0, size=2)
        cand = _piecewise_power_shape(p1, p2, cut, n)
        if cand is not None:
            _maybe_update(cand, f"pw(p1={p1:.1f},p2={p2:.1f},cut={cut})")
        if time.time() >= deadline - 200:
            break

    # ------------------------------------------------------------------
    # If nothing succeeded (should never happen) fall back to uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform.copy()
        best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # 2️⃣  Continuous refinement – Adam on log‑probabilities with a
    #     soft‑max approximation of the “max’’ (temperature rises slowly).
    # ------------------------------------------------------------------
    a = best_seq.copy()                     # current iterate (sum = 1)
    theta = np.log(a + 1e-20)               # log‑variables, stay in ℝ
    m = np.zeros_like(theta)               # first moment
    v = np.zeros_like(theta)               # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5                               # initial learning rate
    iteration = 0

    # temperature schedule for the soft‑max
    temp_start = 5.0
    temp_end = 80.0
    temp_factor = (temp_end / temp_start) ** (1.0 / 12000)   # ~12 k iterations → max temp
    temp = temp_start

    while time.time() < deadline:
        # ---------- convolution a*a ----------
        A_fft = np.fft.rfft(a, n=size_fft)
        b_full = np.fft.irfft(A_fft * A_fft, n=size_fft)[: 2 * n - 1]
        max_b = b_full.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print("[optim] iter {:5d}  new best {:.6f}".format(iteration, best_val))

        # ---------- soft‑max weights ----------
        # shift to avoid overflow/underflow
        shift = b_full - max_b
        w = np.exp(temp * shift)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(b_full, 1.0 / b_full.size)
        else:
            w /= w_sum

        # ---------- gradient of the soft‑max objective ----------
        # gradient = 2 * Σ_j w_j * a_{j-i}   (i = 0…n-1)
        w_rev = w[::-1]
        conv_rev = _full_convolution_fft(a, w_rev, size_fft)  # length = 3n‑2
        offset = len(w_rev) - 1          # = 2n‑2
        # take entries offset‑i (i=0…n‑1)
        grad = 2.0 * conv_rev[offset - np.arange(n)]

        # ---------- Adam update on log‑variables ----------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple linear decay of step size
        progress = (time.time() - start_time) / (deadline - start_time)
        lr = lr0 * (1.0 - progress) + 1e-5

        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- map back to the simplex ----------
        max_theta = np.max(theta)
        a = np.exp(theta - max_theta)     # shift for numerical stability
        s = a.sum()
        if s > 0 and np.isfinite(s):
            a /= s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
            theta = np.log(a + 1e-20)
            m.fill(0.0)
            v.fill(0.0)

        # raise the temperature a little each iteration
        temp = min(temp_end, temp * temp_factor)

        # occasional jitter to escape plateaus
        if iteration % 2500 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            theta = np.log(a + 1e-20)
            m.fill(0.0)
            v.fill(0.0)
            print("[optim] jitter applied at iter {:5d}".format(iteration))

        # safety break if we are really close to the deadline
        if time.time() >= deadline - 0.2:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Final reporting – the outer framework will call evaluate_sequence,
    #     we just log the value here for debugging.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print("[final] evaluate_sequence score = {:.6f}".format(final_score))
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
