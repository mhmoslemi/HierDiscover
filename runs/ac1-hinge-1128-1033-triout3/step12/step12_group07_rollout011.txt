"""
Optimiser for the “step‑function attack”.

The objective (for a normalised sequence Σa = 1) is
    L(a) = 2·n·max(conv(a,a)) .
We try to make the self‑convolution as flat as possible.
The programme proceeds as follows

1.  A rich initial population (single Beta families, mixtures of
    three Betas, exponentials, Dirichlet draws and a few hand‑crafted
    shapes) is generated and the best few candidates are kept.

2.  An Adam optimiser works on a smooth log‑sum‑exp approximation of
    the max‑convolution.  It runs on the current best candidate for a
    moderate fraction of the time budget.

3.  An aggressive, simulated‑annealing style hill‑climber is applied
    to the best candidate for the majority of the remaining budget.
    Moves transfer a fraction of the mass from a “source’’ index that
    contributes heavily to the current maximal convolution entry to a
    “target’’ index that contributes little.  Both a linear schedule
    for the step size and an exponential cooling schedule are used.

4.  Two polishing phases – a projected sub‑gradient descent and an
    exponentiated‑gradient (mirror‑descent) step – further refine the
    solution.

5.  A final lightweight annealing with larger moves is performed if
    time permits.

All intermediate vectors are projected onto the probability simplex
(Σa = 1, a_i ≥ 0) after each update.  The final sequence is returned
as a plain list of floats; it will be evaluated by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a))   (assumes Σa = 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def _beta_vector(n: int, logp: float, logq: float) -> np.ndarray:
    """Deterministic Beta shape a_i ∝ i^{p‑1} (n‑i+1)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    p = np.exp(logp)
    q = np.exp(logq)
    a = np.power(i, p - 1.0) * np.power(n - i + 1, q - 1.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Beta shape on a log‑scale."""
    lp = rng.uniform(-9.2, 2.3)
    lq = rng.uniform(-9.2, 2.3)
    return _beta_vector(n, lp, lq)


def _exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Truncated exponential a_i ∝ r^i with r∈(0,1)."""
    log_r = rng.uniform(-8.0, -0.1)
    r = np.exp(log_r)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    a /= a.sum()
    return a


def _mixture_three_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of three independent Beta families."""
    # mixture weights (Dirichlet with concentration 1)
    w = rng.dirichlet(np.ones(3))
    a = np.zeros(n, dtype=np.float64)
    for idx in range(3):
        lp = rng.uniform(-9.2, 2.3)
        lq = rng.uniform(-9.2, 2.3)
        a += w[idx] * _beta_vector(n, lp, lq)
    a /= a.sum()
    return a


def _handcrafted(n: int) -> np.ndarray:
    """A couple of simple textbook shapes."""
    i = np.arange(n, dtype=np.float64)
    # sinusoid
    s1 = np.sin(np.pi * (i + 0.5) / n)
    s1 = np.maximum(s1, 0.0)
    s1 /= s1.sum()
    # cosine‑square
    s2 = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    s2 = np.maximum(s2, 0.0)
    s2 /= s2.sum()
    return s1 if _objective(s1) < _objective(s2) else s2


# ----------------------------------------------------------------------
# Adam optimiser on a smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """Smooth max of conv(a,a) and its gradient (already multiplied by 2·n)."""
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # Convolution of a with reversed weight vector
    p_rev = p[::-1]
    conv_ab = np.fft.irfft(
        np.fft.rfft(a, 1 << (2 * n - 2).bit_length()) *
        np.fft.rfft(p_rev, 1 << (2 * n - 2).bit_length())
    )[: 2 * n - 1]

    segment = conv_ab[n - 1 : 2 * n - 1]          # n entries
    grad = 4.0 * n * segment[::-1]               # factor 2·n already included
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smooth approximation of max‑convolution."""
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)       # work in log‑softmax space
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.001
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 4000

    rng = np.random.default_rng()
    while time.time() < deadline:
        # decode
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule
        prog = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** prog
        eps = eps_start * (eps_end / eps_start) ** prog

        # smooth max and gradient
        _, grad = _smooth_max_grad(a, eps)

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if rng.random() < 0.015:
            w += rng.normal(scale=0.02, size=n)

        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = rng.normal(scale=0.2, size=n)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (source → target moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """Local search with simulated‑annealing acceptance."""
    n = a0.shape[0]
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    rng = np.random.default_rng()
    # schedule parameters
    max_frac_start, max_frac_end = 0.8, 0.001
    temp_start, temp_end = 0.04, 1e-9
    no_improve = 0
    IMPROVE_LIMIT = 80000
    iteration = 0

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index: proportional to contribution
        if total > 0:
            src_probs = contrib / total
            i = rng.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index: favour low contributions
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = rng.choice(n, p=tgt_weights / wsum)
        else:
            # fallback uniform choice different from i
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear time)
        b_new = b.copy()
        b_new[j : j + n] += 2.0 * delta * a
        b_new[i : i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # re‑perturb from the current best
            print(f"[Hill] stagnation, re‑perturb (it={iteration})")
            a = best_a.copy()
            a += rng.normal(scale=0.03, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val

# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    n = a0.shape[0]
    a = a0.copy()
    cur_val = _objective(a)

    rng = np.random.default_rng()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        b = _full_convolution(a)
        max_b = b.max()
        ks = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple diminishing step size
        prog = (time.time() - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - prog) + 1e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < cur_val - 1e-13:
            a, cur_val = a_new, new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to escape flat regions
        if rng.random() < 0.003:
            a = np.maximum(a + rng.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val

# Exponentiated‑gradient polishing
# ----------------------------------------------------------------------
def _expgrad_polish(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) polishing on the exact max."""
    n = a0.shape[0]
    a = a0.copy()
    cur_val = _objective(a)

    rng = np.random.default_rng()
    eta_start, eta_end = 1.0, 1e-5

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        b = _full_convolution(a)
        max_b = b.max()
        ks = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        prog = (time.time() - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - prog) + eta_end * prog

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        new_val = _objective(a)
        if new_val < cur_val - 1e-13:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[ExpG] new best = {best_val:.6f}")

        if rng.random() < 0.002:
            a = np.maximum(a + rng.normal(scale=0.018, size=n), 0.0)
            a /= a.sum()

    return best_a, best_val

def _final_anneal(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  best_a: np.ndarray,
                  best_val: float) -> tuple[np.ndarray, float]:
    """Light‑weight annealing with relatively large source→target transfers."""
    n = a0.shape[0]
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    rng = np.random.default_rng()
    T0 = 0.02

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)            # linear cooling
        max_frac = 0.12 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[j : j + n] += 2.0 * delta * a
        b_new[i : i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # occasional small jitter
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and (tries to) minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initial population (random + systematic)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value = {best_val:.6f}")

    # keep a small elite list
    elite = [(best_val, best_a.copy())]   # list of (val, a)

    def consider(cand: np.ndarray):
        nonlocal best_val, best_a, elite
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Init] new best = {best_val:.6f}")
        # maintain top‑5 elite
        elite.append((val, cand.copy()))
        elite.sort(key=lambda x: x[0])
        if len(elite) > 5:
            elite.pop()

    # ----- systematic Beta grid (coarse) -----
    logp_vals = np.linspace(-8.0, 2.0, 12)
    logq_vals = np.linspace(-8.0, 2.0, 12)
    for lp in logp_vals:
        for lq in logq_vals:
            consider(_beta_vector(n, lp, lq))

    # ----- random Beta draws -----
    for _ in range(3000):
        consider(_random_beta(n, rng))

    # ----- mixtures of three Betas -----
    for _ in range(2500):
        consider(_mixture_three_beta(n, rng))

    # ----- exponential family -----
    for _ in range(800):
        consider(_exponential(n, rng))

    # ----- Dirichlet draws with varied concentration -----
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(20):
            cand = rng.dirichlet(np.full(n, alpha))
            consider(cand)

    # ----- handcrafted candidates -----
    consider(_handcrafted(n))

    print(f"[Init] best after exhaustive init = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈12 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.12 * budget_s
    best_a, best_val = _adam_optimize(
        best_a,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )

    # ------------------------------------------------------------------
    # Phase B – Hill‑climbing (≈55 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.67 * budget_s
    best_a, best_val = _hillclimb(
        best_a,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )

    # ------------------------------------------------------------------
    # Phase C – Sub‑gradient polishing (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.82 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a,
        deadline=phase_c_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=400_000,
    )

    # ------------------------------------------------------------------
    # Phase D – Exponentiated‑gradient polishing (≈10 % of the budget)
    # ------------------------------------------------------------------
    phase_d_deadline = start_time + 0.92 * budget_s
    best_a, best_val = _expgrad_polish(
        best_a,
        deadline=phase_d_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=200_000,
    )

    # ------------------------------------------------------------------
    # Phase E – Final annealing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _final_anneal(
            best_a,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
        )

    # ------------------------------------------------------------------
    # Final evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
