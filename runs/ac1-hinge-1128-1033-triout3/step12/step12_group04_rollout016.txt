def propose_candidate(seed=42, budget_s=1000):

    start = time.time()

    n = 8000

    if 'height_sequence_1' in globals():
        try:
            old = np.asarray(globals()['height_sequence_1'], dtype=float)

            x_old = np.linspace(0., 1., old_len)
            x_new = np.linspace(0., 1., n)

def propose_candidate(...):
    import time, numpy as np
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8000

    # basis
    x = np.linspace(-1, 1, n)
    d = 6  # degree
    # B = np.vstack([x**i for i in range(d+1)]).T
    B = np.empty((n, d+1))
    for i in range(d+1):
        B[:, i] = x**i

    # Adam
    m_c = np.zeros(d+1)
    v_c = np.zeros(d+1)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_c = 0.1

    # smoothing parameter for max
    beta_smooth = 0.5
    beta_max = 200.0
    beta_factor = 1.001

    best_c = None
    best_val = np.inf
    best_a = None

    iteration = 0
    # optionally start from previous best or random init
    # We'll start from uniform (c=0). Optionally add perturbation from height_sequence_1.

    while time.time() < deadline:
        iteration += 1

        # compute a
        log_a = B @ c  # shape (n,)
        max_log = np.max(log_a)
        exp_log = np.exp(log_a - max_log)  # scaling to avoid overflow
        a_raw = exp_log
        a = a_raw / a_raw.sum()  # normalize

        # compute convolution and objective
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv

        if cur_obj < best_val:
            best_val = cur_obj
            best_c = c.copy()
            best_a = a.copy()
            print(f"[iter {iteration:07d}] new best {best_val:.9f} (time {time.time() - start:.1f}s)")

        # compute smoothed weights w
        # use softmax with beta_smooth
        max_c = max_conv
        exp_w = np.exp(beta_smooth * (conv - max_c))
        w = exp_w / exp_w.sum()

        # compute gradient w.r.t a
        a_rev = a[::-1]
        size = _next_pow2(len(a_rev) + len(w) - 1)
        fa_rev = np.fft.rfft(a_rev, size)
        fw = np.fft.rfft(w, size)
        g_full = np.fft.irfft(fa_rev * fw, size)[:len(a_rev) + len(w) - 1]
        start_idx = n - 1
        g_a = 2.0 * g_full[start_idx:start_idx + n]  # shape (n,)

        # gradient w.r.t c via chain rule: ∂obj/∂c = B^T (g_a * a)
        grad_c = B.T @ (g_a * a)

        # Adam update for c
        m_c = beta1 * m_c + (1 - beta1) * grad_c
        v_c = beta2 * v_c + (1 - beta2) * (grad_c ** 2)
        m_hat = m_c / (1 - beta1 ** iteration)
        v_hat = v_c / (1 - beta2 ** iteration)
        c = c - lr_c * m_hat / (np.sqrt(v_hat) + eps)

        # decrease lr over time maybe
        if iteration % 2000 == 0:
            lr_c *= 0.9

        # increase smoothing beta
        beta_smooth = min(beta_max, beta_smooth * beta_factor)

        # optionally random perturb to avoid stuck
        if iteration % 5000 == 0:
            # jitter c a little
            c += np.random.normal(scale=0.01, size=c.shape)
            # reset Adam moments
            m_c.fill(0)
            v_c.fill(0)

    # after loop, construct final best a from best_c
    if best_c is None:
        # fallback to uniform
        best_a = np.full(n, 1.0/n)
        final_score = evaluate_sequence(best_a.tolist())
    else:
        log_a = B @ best_c
        max_log = np.max(log_a)
        a_raw = np.exp(log_a - max_log)
        best_a = a_raw / a_raw.sum()
        final_score = evaluate_sequence(best_a.tolist())
    print("[final] reported score =", final_score)
    return best_a.tolist()

"""
Parametric exponential‑basis optimiser for the L∞ self‑convolution problem.

The idea is to keep the whole vector a of size n (default 8000) positive
by writing it as a normalized exponential of a low‑dimensional polynomial
basis:
        a_i = exp( Σ_j c_j φ_j(i) ) / Σ_i exp( Σ_j c_j φ_j(i) )

The coefficients c_j are the optimisation variables.  Their dimension
(d+1) is small (here d=6) so we can use a fast Adam optimiser.
The gradient of the true objective 2·n·max(a*a) is approximated by the
gradient of a smooth “soft‑max’’ of the convolution:
        w_k ∝ exp(β·(a*a)_k) ,   Σ_k w_k = 1.
For a fixed β the gradient w.r.t. a is
        ∇_a = 2·Conv( a_rev , w )
where a_rev is a reversed copy of a and the convolution is a linear
operation (computed with FFT).  Using the chain‑rule we obtain the gradient
w.r.t. the coefficients c and perform Adam updates.
The smoothing parameter β is increased slowly so that early iterations
see a smooth landscape and later iterations focus on the true
maximum.  Small random jitter of the coefficients and a learning‑rate
decay avoid stagnation.

The routine respects the 1000 s budget, keeps the best sequence seen and
returns it after a final call to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level – no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution x*x (real FFT, length 2·len(x)-1).
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Uses an exponential polynomial parametrisation with Adam optimisation.
    Returns the best sequence (list of floats) found before the time budget
    expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Build a low‑dimensional polynomial basis on the interval [-1, 1].
    # Using monomials is sufficient; degree can be increased if time permits.
    # --------------------------------------------------------------
    d = 6                                 # degree of the polynomial
    x = np.linspace(-1.0, 1.0, n)         # centred coordinate
    B = np.empty((n, d + 1), dtype=float)
    for j in range(d + 1):
        B[:, j] = x ** j                  # φ_j(i) = (scaled i)^j

    # --------------------------------------------------------------
    # Adam optimiser state for the coefficient vector `c`.
    # --------------------------------------------------------------
    c = np.random.normal(scale=0.01, size=d + 1)   # small random start
    m_c = np.zeros_like(c)
    v_c = np.zeros_like(c)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_c = 0.12                               # learning rate for coefficients

    # --------------------------------------------------------------
    # Soft‑max smoothing for the max‑convolution term.
    # --------------------------------------------------------------
    beta_smooth = 0.5          # start with a very smooth approximation
    beta_max = 200.0
    beta_factor = 1.001        # geometric increase each iteration

    # --------------------------------------------------------------
    # Book‑keeping of the best solution seen so far.
    # --------------------------------------------------------------
    best_val = float("inf")
    best_c = None
    best_a = None

    iteration = 0
    # ------------------------------------------------------------------
    # Main optimisation loop – stops automatically when the deadline is
    # reached.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- 1) construct a from the current coefficients -----------------
        log_a = B @ c                              # shape (n,)
        max_log = np.max(log_a)                    # for numerical stability
        a_raw = np.exp(log_a - max_log)            # positive, possibly sparse
        a = a_raw / a_raw.sum()                    # enforce Σ a_i = 1

        # ----- 2) full self‑convolution and exact objective -----------------
        conv = _full_convolution(a)                # length 2n‑1
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv

            best_c = c.copy()

                  f"(time {time.time() - start_time:.1f}s)")

        # ----- 4) soft‑max weights for a smooth gradient --------------------
        #   w_k = exp(β·conv_k) / Σ exp(β·conv_k)
        #   using the same max trick to avoid overflow.
        max_c = max_conv
        exp_w = np.exp(beta_smooth * (conv - max_c))
        w = exp_w / exp_w.sum()                     # shape (2n‑1,)

            v_c.fill(0.0)

    # ------------------------------------------------------------------

        print("[final] evaluate_sequence raised:", exc)
        final_score = float("inf")

    return best_a.tolist()
