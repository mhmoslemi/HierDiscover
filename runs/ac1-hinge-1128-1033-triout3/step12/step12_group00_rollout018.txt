Propose a step‑function sequence that improves the current best upper‑bound
(≈1.508) within a 1000 s budget.

Algorithm
---------
0️⃣  Coarse random sweep (≈5 % of the budget)
    – Uniform, interpolated previous solution (if any) and a large variety
      of parametric families (Beta, Dirichlet, power‑law, exponential,
      binary, sine).  All candidates are normalised to sum = 1 and the
      exact objective 2·n·max(conv(a,a)) is evaluated.  The best few
      candidates are kept.

1️⃣  Multi‑stage smooth optimisation (≈30 % of the budget)
    – L‑BFGS‑B on a log‑sum‑exp (smooth‑max) surrogate.  Several ε‑values
      are used (from 1e‑1 down to 1e‑12).  The optimizer works in log‑it
      space (soft‑max) so the simplex constraints are satisfied automatically.

2️⃣  Adam‑style projected optimisation (≈20 % of the budget)
    – Using the exact sub‑gradient of the true (nonsmooth) objective.
      An Adam update with projection onto the probability simplex is run
      for a few thousand iterations on each of the current best candidates.

3️⃣  Exact sub‑gradient polishing (≈30 % of the budget)
    – Projected sub‑gradient descent with a slowly decaying step‑size,
      limited to a large number of iterations (up to 500 000).  This
      tightens the solution and often reduces the bound further.

4️⃣  Blending & final L‑BFGS (remaining time)
    – Convex combinations with the uniform distribution are examined,
      followed by a short final L‑BFGS run with a tiny ε (1e‑12).

All phases respect the global deadline.  The best sequence found is
returned as a plain list of floats; it will be evaluated with the supplied
``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.  The gradient already includes the factor 2·n.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # we need only the n entries that correspond to the “forward” part
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → probability simplex)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Candidate generators
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ≈1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_powerlaw_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ (i+1)^p with a random exponent p∈[-4,4]."""
    i = np.arange(1, n + 1, dtype=np.float64)
    p = rng.uniform(-4.0, 4.0)
    a = np.power(i, p, where=i > 0, out=np.zeros_like(i))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ exp(λ·i) with λ negative (decay)."""
    lam = rng.uniform(-0.15, -0.001)   # moderate decay
    i = np.arange(n, dtype=np.float64)
    a = np.exp(lam * i)
    s = a.sum()
    a /= s
    return a


def _random_binary_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Bernoulli(p) mask, normalised."""
    p = rng.uniform(0.1, 0.9)
    a = rng.binomial(1, p, size=n).astype(np.float64)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _sine_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1))^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    p = rng.uniform(0.1, 4.0)
    a = np.sin(np.pi * i / (n + 1)) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 1200) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on the smoothed objective."""
    # initialise in log‑space (soft‑max guarantees feasibility)
    x0 = np.log(a0 + 1e-12)

    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    # respect remaining time
    time_left = deadline - time.time()
    if time_left < 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:          # pragma: no cover
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the exact (nonsmooth) objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   lr: float = 0.5,
                   max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Projected Adam optimisation using the exact sub‑gradient."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    for it in range(1, max_iters + 1):
        now = time.time()
        if now >= deadline:
            break

        # exact sub‑gradient of 2·n·max(conv(a,a))
        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = _project_simplex(a)

        cur_val = _objective(a)
        if cur_val + 1e-12 < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[Adam]   new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step‑size decreasing linearly from 0.4 → 1e‑5
        progress = (now - start_time) / (deadline - start_time)
        step = 0.40 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        new_val = _objective(a_new)

        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 500 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combos t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and that minimises `evaluate_sequence`.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – a moderately large n gives a richer space.
    # --------------------------------------------------------------
    n = 20000

    # --------------------------------------------------------------
    # 0️⃣  Coarse initialisation (exact objective)
    # --------------------------------------------------------------
    candidates = []

    # 0.1  Previous solution (if defined)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        candidates.append((val, a0))
        print(f"[Init] interpolated previous solution, value={val:.6f}")
    except Exception:
        pass

    # 0.2  Uniform baseline
    uniform = np.full(n, 1.0 / n)
    candidates.append((_objective(uniform), uniform))

    # 0.3  Random Beta candidates
    for _ in range(400):
        cand = _random_beta_candidate(n, rng)
        candidates.append((_objective(cand), cand))

    # 0.4  Dirichlet draws with various concentrations
    for alpha in (0.02, 0.1, 0.3, 0.7, 1.5, 3.0):
        for _ in range(6):
            cand = rng.dirichlet(np.full(n, alpha))
            candidates.append((_objective(cand), cand))

    # 0.5  Power‑law candidates
    for _ in range(200):
        cand = _random_powerlaw_candidate(n, rng)
        candidates.append((_objective(cand), cand))

    # 0.6  Exponential candidates
    for _ in range(200):
        cand = _random_exponential_candidate(n, rng)
        candidates.append((_objective(cand), cand))

    # 0.7  Binary mask candidates
    for _ in range(100):
        cand = _random_binary_candidate(n, rng)
        candidates.append((_objective(cand), cand))

    # 0.8  Sine‑shape candidates
    for _ in range(150):
        cand = _sine_candidate(n, rng)
        candidates.append((_objective(cand), cand))

    # Keep the best few for the next phase
    candidates.sort(key=lambda t: t[0])
    top_k = 12
    best_a, best_val = candidates[0][1], candidates[0][0]
    print(f"[Init] best after sweep: value={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – L‑BFGS on smooth‑max, decreasing ε schedule
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    eps_schedule = [1e-1, 5e-2, 1e-2, 5e-3, 1e-3,
                    5e-4, 1e-4, 5e-5, 1e-5, 5e-6,
                    1e-6, 5e-7, 1e-7, 5e-8, 1e-8,
                    5e-9, 1e-9, 5e-10, 1e-10, 5e-11,
                    1e-11, 5e-12, 1e-12]

    for eps in eps_schedule:
        if time.time() >= phase_a_deadline:
            break
        for i in range(min(top_k, len(candidates))):
            if time.time() >= phase_a_deadline:
                break
            a_start = candidates[i][1]
            best_a, best_val = _lbfgs_optimize(
                a_start, n, deadline, start_time,
                best_a, best_val, eps=eps, max_iter=1200)
            best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase B – Adam optimisation on the exact objective
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.80 * budget_s
    for i in range(min(top_k, len(candidates))):
        if time.time() >= phase_b_deadline:
            break
        a_start = candidates[i][1]
        best_a, best_val = _adam_optimize(
            a_start, n, deadline, start_time,
            best_a, best_val, lr=0.5, max_iters=3000)
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase C – Exact sub‑gradient polishing (large budget)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.95 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase_c_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=500_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase D – Final short L‑BFGS with a tiny ε
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(
            best_a, n, deadline, start_time,
            best_a, best_val, eps=1e-12, max_iter=800)

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Ensure a plain list of floats is returned
    return best_a.tolist()
