    n = a.shape[0]

    for k in active_idx:
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        idx = np.arange(lo, hi+1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    n = 8000
    best_seq = None
    best_obj = np.inf

    # Use previous solution if exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        if old.shape[0] != n:
            old = np.interp(np.linspace(0,1,n), np.linspace(0,1,old.shape[0]), old)
        old = np.maximum(old, 0.0)
        old = old / old.sum()
        maybe_update(old)  # treat as candidate
        best_seq = old.copy()
        best_obj = _objective(old)
    except Exception:
        pass

    # Determine number of restarts based on time
    restart = 0
    while time.time() < deadline - 5.0:
        # Random init of w
        w = np.random.standard_normal(n) * 0.1  # small random
        # Optionally start from uniform also
        # Set Adam state:
        m = np.zeros_like(w)
        v = np.zeros_like(w)
        beta1 = 0.9
        beta2 = 0.999
        epsilon = 1e-8
        lr = 0.2
        iteration = 0
        # loop iterations while time remains
        while time.time() < deadline - 1.0:
            # compute a via softmax
            max_w = np.max(w)
            exp_w = np.exp(w - max_w)  # shift to avoid overflow
            a = exp_w / exp_w.sum()   # sum=1
            # evaluate cheap objective
            b = _full_convolution(a)
            max_b = b.max()
            cur_obj = 2.0 * n * max_b
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()
                print(f"[restart {restart} iter {iteration}] new best {best_obj:.6f}")

            # pick active lags: top m_lags = maybe 7
            top_m = 7
            if top_m >= b.size:
                active_idx = np.arange(b.size)
            else:
                part = np.argpartition(b, -top_m)[-top_m:]
                # optionally filter by relative tolerance
                max_val = max_b
                mask = b[part] >= max_val * (1 - 1e-6)
                if np.any(mask):
                    active_idx = part[mask]
                else:
                    active_idx = np.array([int(np.argmax(b))])
            # subgradient wrt a
            grad_a = subgrad_active(a, b, active_idx)
            # gradient wrt w
            grad_w = a * (grad_a - np.dot(a, grad_a))

            # Adam update
            iteration += 1
            lr_t = lr * (1.0 / (np.sqrt(iteration)))  # optional decay
            m = beta1 * m + (1 - beta1) * grad_w
            v = beta2 * v + (1 - beta2) * (grad_w ** 2)
            m_hat = m / (1 - beta1 ** iteration)
            v_hat = v / (1 - beta2 ** iteration)
            w = w - lr_t * m_hat / (np.sqrt(v_hat) + epsilon)

        print(f"[final] evaluate_sequence returned {final_score:.6f}")

        print("[final] evaluate_sequence raised", e)
    return best_seq.tolist()

    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv

    n = a.shape[0]
    b = _full_convolve(a, a)
    return float(2.0 * n * b.max())

def _gradient_softmax(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv) with respect to the
    simplex‑restricted vector a.

    The surrogate is  (1/α)·log Σ_k exp(α·b_k).  Its gradient w.r.t. a is
    the convex combination of the individual sub‑gradients of each lag,
    where the combination weights are the (normalised) soft‑max of α·b.
    """

    max_b = b.max()
    w = np.exp(alpha * (b - max_b))
    w_sum = w.sum()

def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a vector on the simplex."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.size)
    return e / s

"""
Optimization of a non‑negative step function for the ``evaluate_sequence`` score.

The algorithm proceeds in three stages:

1️⃣  **Random‑support exploration** – many cheap candidates are generated by
    picking random supports and by drawing dense Dirichlet vectors.  The best
    candidate (according to the cheap proxy 2·n·max(conv)) is kept.

2️⃣  **Soft‑max surrogate optimisation** – the coefficients are represented as
    a soft‑max of an unrestricted vector ``w`` (this guarantees positivity and
    Σa = 1).  We minimise a smooth surrogate of ``max(conv(a,a))``:

        f_α(a) = (1/α)·log Σ_k exp(α·b_k) ,   b = a * a   (full linear convolution)

    The gradient of this surrogate is obtained analytically via a second
    convolution, which makes each iteration O(n log n).  An Adam‑style update
    is applied to ``w``; the temperature ``α`` is slowly increased so that the
    surrogate approaches the true maximum.  Several random restarts are run
    within the time budget and the best sequence found is stored.

3️⃣  **Final evaluation** – the best sequence is submitted to ``evaluate_sequence``
    and returned.

The routine respects the supplied time budget (default 1000 s) and never
performs any file or network I/O.
"""

import time
import numpy as np

# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real 1‑D arrays using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n - 1).bit_length()          # next power of two ≥ n
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


def _objective(a: np.ndarray) -> float:
    """Cheap objective used during the search (Σa = 1)."""
    n = a.shape[0]
    b = _full_convolve(a, a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate of max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Returns ∇_a  (1/α)·log Σ_k exp(α·b_k)  where  b = a*a (full convolution).

    The gradient can be written as a single convolution:
        grad_a = 2 * conv( w , a_rev )[ n-1 : n-1+n ]
    where
        w_k = exp(α·b_k) / Σ_j exp(α·b_j)     (soft‑max weights)
        a_rev = a[::-1]                       (reversed a)
    """
    # soft‑max weights (stable)
    max_b = b.max()
    w = np.exp(alpha * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(b, 1.0 / b.size)
    else:
        w /= w_sum

    # convolution w * a_rev
    a_rev = a[::-1]
    conv = _full_convolve(w, a_rev)          # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * conv[n - 1 : n - 1 + n]      # slice of length n
    return grad

    """Numerically stable soft‑max (produces a vector on the simplex)."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full_like(x, 1.0 / x.size)
    return e / s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin for the final call

    # ------------------------------------------------------------------
    # Problem size – a trade‑off between expressivity and speed.
    # ------------------------------------------------------------------
    n = 8000

    best_seq: np.ndarray | None = None
    best_obj = np.inf

    def maybe_update(candidate: np.ndarray):
        """Update the global best if the cheap objective improves."""
        nonlocal best_seq, best_obj
        val = _objective(candidate)
        if val < best_obj:
            best_obj = val
            best_seq = candidate.copy()
            print(f"[update] new best cheap objective={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Warm‑start from a previously saved solution (if available).
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # Interpolate to the current resolution.
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0 / n)
        else:
            a0 = a0 / a0.sum()
        maybe_update(a0)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random‑support exploration (very cheap candidates).
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    support_fractions = np.array([0.25, 0.35, 0.45, 0.55, 0.65, 0.75])
    support_sizes = (support_fractions * n).astype(int)

    while time.time() < deadline - 5.0:
        # random sparsity level
        k = int(rng.choice(support_sizes))
        k = max(2, min(k, n))

        # (a) uniform weight on a random subset
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0 / k
        maybe_update(a)

        # (b) dense Dirichlet‑drawn vector (occasional)
        if rng.random() < 0.05:
            a_dense = rng.random(n)
            a_dense /= a_dense.sum()
            maybe_update(a_dense)

        # (c) deterministic equally‑spaced support (rare)
        if rng.random() < 0.02:
            k_eq = int(rng.uniform(0.2, 0.8) * n)
            eq_idx = np.linspace(0, n - 1, k_eq, dtype=int)
            a_eq = np.zeros(n, dtype=np.float64)
            a_eq[eq_idx] = 1.0 / k_eq
            maybe_update(a_eq)

    # ------------------------------------------------------------------
    # 2️⃣  Stochastic optimisation using a soft‑max surrogate.
    # ------------------------------------------------------------------
    # hyper‑parameters
    max_restarts = 12
    max_iters_per_restart = 20000
    alpha_start = 0.5
    alpha_max = 8.0
    alpha_inc = 0.001
    lr_initial = 0.2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    restarts_done = 0
    while time.time() < deadline - 5.0 and restarts_done < max_restarts:
        # random initialisation of the unconstrained vector w
        w = rng.standard_normal(n) * 0.1
        m = np.zeros_like(w)
        v = np.zeros_like(w)
        alpha = alpha_start
        it = 0

        while time.time() < deadline - 2.0 and it < max_iters_per_restart:
            # coefficients on the simplex
            a = _softmax(w)

            # auto‑convolution and cheap objective
            b = _full_convolve(a, a)
            cur_obj = 2.0 * n * b.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()
                print(f"[optim r{restarts_done} it{it}] new best cheap objective={best_obj:.6f}")

            # gradient of the soft‑max surrogate w.r.t. a
            grad_a = _gradient_softmax(a, b, alpha)

            # chain rule to obtain ∇_w (soft‑max parameters)
            dot_ag = np.dot(a, grad_a)
            grad_w = a * (grad_a - dot_ag)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_w
            v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            lr = lr_initial * (1.0 / (1.0 + 1e-4 * it))
            w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

            # increase the temperature (α) slowly – the surrogate becomes tighter.
            if alpha < alpha_max:
                alpha = min(alpha_max, alpha + alpha_inc)

            # occasional jitter to escape flat plateaux
            if rng.random() < 0.001 or (it % 5000 == 0 and it > 0):
                w += rng.normal(scale=0.01, size=n)

        restarts_done += 1
        print(f"[restart {restarts_done}] finished after {it} iterations, best cheap obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("[fallback] using uniform sequence")

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
