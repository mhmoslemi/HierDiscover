"""
Propose a step‑function that (empirically) beats the previous best
objective O≈1.5081.  The method works in three simple phases:

1.  Build a diverse pool of initial probability vectors (interpolated
    previous solution, several Beta shapes and Dirichlet draws).
2.  For each start run **entropy‑mirror descent** on the exact max
    convolution entry.  The update is
        a ← a·exp(−η·g)   with   g_i = 2·a_{k−i}  (k = argmax conv),
    followed by renormalisation.  A linearly decaying step‑size η
    and occasional jitter avoid plateaus.
3.  After the best vector is identified a long polishing run with a
    smaller η finishes the optimisation.

All work is done in the log‑domain (multiplicative updates) so the
variables stay non‑negative, and the total runtime is kept under the
given 1000 s budget by checking the wall‑clock after each iteration.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolution (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full length 2·len(x)−1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max‑entry of the self‑convolution
# ----------------------------------------------------------------------
def _mirror_descent(a0: np.ndarray,
                    max_steps: int,
                    eta_start: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform mirror‑descent (multiplicative updates) on the exact maximal
    convolution coefficient.

    Parameters
    ----------
    a0 : np.ndarray
        Starting probability vector (already normalised to sum = 1).
    max_steps : int
        Upper bound on the number of iterations.
    eta_start : float
        Initial step‑size; it decays linearly to zero.
    deadline : float
        Absolute time (seconds since epoch) after which the routine must
        stop – it checks this at every iteration.

    Returns
    -------
    best_a : np.ndarray
        The best vector seen (still normalised).
    best_O : float
        Corresponding objective value  O = 2·n·max(conv) .
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_O = np.inf
    stagnation = 0               # counts iterations without improvement

    for step in range(max_steps):
        # ---- time guard -------------------------------------------------
        if time.time() >= deadline:
            break

        # ---- evaluate the current objective -----------------------------
        conv = _full_convolution(a)
        k = int(conv.argmax())
        max_c = conv[k]
        O = 2.0 * n * max_c               # sum(a)=1, so denominator is 1

        if O < best_O - 1e-12:
            best_O = O
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # ---- sub‑gradient of the active max‑entry -----------------------
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            # derivative of (a*a)[k] w.r.t. a_i is 2·a_{k−i}
            g[idx] = 2.0 * a[k - idx]

        # ---- multiplicative (entropy) update ----------------------------
        eta = eta_start * (1.0 - step / max_steps)   # linear decay
        a = a * np.exp(-eta * g)

        # ---- renormalise (keeps the L1‑norm equal to 1) ---------------
        s = a.sum()
        if s > 0.0:
            a /= s
        else:                       # numerical safety net
            a.fill(1.0 / n)

        # ---- jitter to escape long plateaus -----------------------------
        if stagnation >= 4000:
            jitter = np.random.normal(scale=0.02, size=n)
            a = a * np.exp(jitter)
            a /= a.sum()
            stagnation = 0

    return best_a, best_O
