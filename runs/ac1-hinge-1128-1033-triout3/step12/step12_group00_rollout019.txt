"""
Enhanced step‑function optimiser (≈1000 s budget)

Key ideas
---------
* **Broad initial sweep** – many more Dirichlet / Beta / exponential
  seeds (including the previous solution if present).  The very best
  candidates (top‑k) are kept for the optimisation phases.

* **Two‑stage smooth‑max minimisation** – the non‑smooth objective
  `2·n·max(conv(a,a)) / (∑a)²` is approximated by a log‑sum‑exp
  surrogate.  We run L‑BFGS‑B on this surrogate several times with a
  *slowly* decreasing smoothing parameter ε, starting from a relatively
  large value (≈0.5) and ending at 1e‑12.  This gives a fast descent
  towards a low‑max‑convolution region while remaining fully
  differentiable.

* **Heavy sub‑gradient polishing** – after the smooth phases we switch
  to the true non‑smooth objective and perform a long projected
  sub‑gradient descent.  The stepsize follows a simple linear schedule
  and occasional random jitter is added to escape tiny plateaus.

* **Blend & jitter** – after each major phase we try convex blends with
  the uniform distribution and a mirrored copy of the current best.
  Random jitter (small Gaussian noise) is also examined.

The routine respects the 1000 s wall‑time limit, prints progress
information and finally returns a plain list of non‑negative floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.  The gradient already includes the factor 2·n.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick (stable)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p (gradient of the smooth max)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries that affect a
    grad_a = 4.0 * n * segment[::-1]             # multiply by 2·n already included
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max re‑parameterisation (ensures feasibility)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx=1, x≥0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random families of candidates -----------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑shape (i‑based) on a log‑scale for p and q."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-12.0, 3.0)    # a broader range than before
    logq = rng.uniform(-12.0, 3.0)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-λ·i) with random λ."""
    i = np.arange(n, dtype=np.float64)
    lam = rng.uniform(0.0, 5.0)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 1500) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on the smoothed objective."""
    # initialise in log‑space (soft‑max guarantees feasibility)
    x0 = np.log(a0 + 1e-15)

    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    time_left = deadline - time.time()
    if time_left < 0.5:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a) = 2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        max_idx = np.where(b == max_b)[0]

        # sub‑gradient of the max‑convolution
        grad = np.zeros_like(a)
        for k in max_idx:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear schedule – aggressive early, tiny later
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 5e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 500 == 0 and np.random.rand() < 0.02:
            jitter = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Blending helpers
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Convex blend with uniform distribution."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


def _blend_with_mirror(a: np.ndarray,
                       n: int,
                       cur_best_val: float) -> tuple[np.ndarray, float]:
    """Blend with the mirrored copy of a (helps break symmetry)."""
    a_rev = a[::-1]
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * a_rev
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val
