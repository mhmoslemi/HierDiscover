"""
Hybrid optimiser for the self‑convolution max problem.

The optimiser works in two phases:

1️⃣  A short Adam‑optimisation with a smooth‑max surrogate.  The temperature
    λ grows from a modest value to a large one, so the optimiser first follows
    a smooth objective and then the true sub‑gradient when λ is big.
2️⃣  A targeted hill‑climbing phase.  We repeatedly move a small amount of
    mass from the entry participating most in the current active convolution
    (the shift k* where the max is attained) to a location that does not
    contribute to that shift.  The convolution is updated analytically in O(n)
    time, so millions of moves can be tried within the 1000 s budget.
    The step size is decreased when the search stagnates.

The best sequence found is returned (as a list of non‑negative floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2*len(x)-1, computed via FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k-i}
    (cross‑correlation of weight vector w with a).
    Returned vector has length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(conv: np.ndarray, n: int) -> float:
    """Return 2·n·max(convolution) (sum of a is assumed to be 1)."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise `evaluate_sequence`.  The routine first performs a few Adam
    optimisation runs and then refines the best solution with a fast
    targeted hill‑climbing search.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 8000 works well within the time budget.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise from a previous construction if possible.
    # --------------------------------------------------------------
    try:
        old_seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old_seq.size > 1:
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_seq.size)
            a = np.interp(new_x, old_x, old_seq)
            a = np.maximum(a, 0.0)
        else:
            raise Exception
    except Exception:
        a = np.full(n, 1.0 / n)  # uniform start
        print("[init] using uniform start")

    # Normalise to a probability vector (sum = 1)
    s = a.sum()
    if s <= 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    # --------------------------------------------------------------
    # Global best tracker
    # --------------------------------------------------------------
    conv = _full_convolution(a)
    best_a = a.copy()
    best_val = _objective(conv, n)
    print(f"[global] initial objective = {best_val:.9f}")

    # ------------------------------------------------------------------
    # Adam optimiser (smooth‑max) – a few short runs with restarts
    # ------------------------------------------------------------------
    lambda_start = 5.0
    lambda_end = 8000.0
    adam_lr = 0.045
    b1, b2 = 0.9, 0.999
    adam_eps = 1e-8
    max_adam_iters = 20000          # per run
    max_adam_runs = 4               # we stop early if time runs out

    while time.time() < deadline and max_adam_runs > 0:
        max_adam_runs -= 1
        # initialise logits from the current best (log‑softmax trick)
        eps = 1e-12
        z = np.log(best_a + eps) + np.random.normal(scale=0.03, size=n)

        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t_iter = 0
        stagnation = 0

        while time.time() < deadline and t_iter < max_adam_iters:
            t_iter += 1
            # ---------- convert logits → probability vector ----------
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            a = exp_z / exp_z.sum()

            # ---------- evaluate exact objective ----------
            conv = _full_convolution(a)
            cur_val = _objective(conv, n)

            # ---------- keep global best ----------
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                stagnation = 0
                print(f"[adam iter {t_iter:05d}] new best = {best_val:.9f} "
                      f"(t={time.time() - start_time:.1f}s)")
            else:
                stagnation += 1

            # ---------- temperature schedule ----------
            prog = (time.time() - start_time) / (deadline - start_time)
            lam = lambda_start * (1.0 - prog) + lambda_end * prog

            # ---------- smooth‑max weights ----------
            conv_max = conv.max()
            if lam > 2000.0:
                # exact sub‑gradient: one‑hot at the active index
                kmax = int(np.argmax(conv))
                w = np.zeros_like(conv)
                w[kmax] = 1.0
            else:
                exp_vals = np.exp(lam * (conv - conv_max))
                w = exp_vals / exp_vals.sum()

            # ---------- gradient of the (smooth) max ----------
            g_raw = _cross_corr(w, a)          # shape (n,)
            g = 2.0 * g_raw

            # ---------- back‑prop through the soft‑max ----------
            dot_ag = np.dot(a, g)
            grad_z = a * (g - dot_ag)

            # ---------- Adam update ----------
            m = b1 * m + (1.0 - b1) * grad_z
            v = b2 * v + (1.0 - b2) * (grad_z ** 2)
            m_hat = m / (1.0 - b1 ** t_iter)
            v_hat = v / (1.0 - b2 ** t_iter)
            z -= adam_lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # ---------- occasional noise to break stagnation ----------
            if stagnation > 4000:
                z += np.random.normal(scale=0.05, size=n)
                stagnation = 0
                print(f"[adam reset] added noise at iter {t_iter}")

        # one brief projected sub‑gradient sweep after each Adam run
        a = best_a.copy()
        conv = _full_convolution(a)
        cur_obj = _objective(conv, n)
        step = 0.02
        for _ in range(800):
            if time.time() >= deadline:
                break
            kmax = int(np.argmax(conv))
            # sub‑gradient of the true max
            g = np.zeros_like(a)
            if kmax < n:
                g[:kmax + 1] = 2.0 * a[kmax::-1]
            else:
                offset = kmax - (n - 1)
                L = n - offset
                g[offset:] = 2.0 * a[:L][::-1]

            a_trial = _project_simplex(a - step * g)
            conv_trial = _full_convolution(a_trial)
            obj_trial = _objective(conv_trial, n)

            if obj_trial < cur_obj - 1e-12:
                a = a_trial
                conv = conv_trial
                cur_obj = obj_trial
                best_a = a.copy()
                best_val = cur_obj
                step = min(step * 1.1, 0.5)
                print(f"[subgrad] improved to {best_val:.9f}")
                if best_val < 1.503:
                    break
            else:
                step *= 0.7

        # early exit if target already reached
        if best_val < 1.503:
            break

    # --------------------------------------------------------------
    # Phase 2 – targeted hill‑climbing refinement
    # --------------------------------------------------------------
    a = best_a.copy()
    conv = _full_convolution(a)
    cur_val = _objective(conv, n)
    step_base = 0.02                # initial move size
    fails_since_improve = 0

    while time.time() < deadline:
        # pick the active shift (where the max occurs)
        k_star = int(conv.argmax())

        # range of indices participating in this shift
        i_low = max(0, k_star - (n - 1))
        i_high = min(n - 1, k_star)          # inclusive
        idxs = np.arange(i_low, i_high + 1)

        # product a[i] * a[k_star-i] – the contribution of each pair
        prod = a[idxs] * a[k_star - idxs]
        i = idxs[np.argmax(prod)]            # index that contributes most

        # choose a destination j that does NOT contribute to k_star
        if k_star < n:
            # any j > k_star makes a[k_star-j] = 0
            if k_star + 1 >= n:
                # fallback: a random index
                j = np.random.randint(0, n)
            else:
                j = np.random.randint(k_star + 1, n)
        else:
            # any j < k_star-(n-1) makes a[k_star-j] = 0
            limit = k_star - (n - 1) - 1
            if limit < 0:
                j = np.random.randint(0, n)
            else:
                j = np.random.randint(0, limit + 1)

        # amount to move – never exceed half of the source mass
        delta = min(step_base, 0.5 * a[i])
        if delta <= 0:
            # nothing to move from here, choose another i
            continue

        # ----------------------------------------------------------
        # Compute the updated convolution analytically (O(n))
        # ----------------------------------------------------------
        conv_new = conv.copy()
        # core update: -2*delta*a shifted by i, +2*delta*a shifted by j
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a
        # quadratic δ² terms
        ii = i + i
        conv_new[ii] += delta * delta
        jj = j + j
        conv_new[jj] += delta * delta
        if i != j:
            ij = i + j
            conv_new[ij] -= 2.0 * delta * delta

        new_val = _objective(conv_new, n)

        if new_val < cur_val - 1e-12:
            # accept the move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_val = new_val
            fails_since_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[hill] new best = {best_val:.9f} (t={time.time() - start_time:.1f}s)")
                if best_val < 1.503:
                    break
        else:
            fails_since_improve += 1
            if fails_since_improve >= 3000:
                # reduce the step size when we are stuck
                step_base *= 0.5
                fails_since_improve = 0
                if step_base < 1e-6:
                    break   # nothing more to improve

    # --------------------------------------------------------------
    # Final secure evaluation (the only place we call evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
