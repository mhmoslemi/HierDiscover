import time, numpy as np

def _full_convolution(x):
    n = x.shape[0]
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _mirror_descent(a, deadline, start_obj=None):
    n = a.shape[0]
    best_a = a.copy()
    if start_obj is None:
        conv = _full_convolution(a)
        best_obj = 2.0 * n * conv.max()
    else:
        best_obj = start_obj
    stagn = 0
    step = 0
    # decide max steps based on remaining time (e.g., trying to have at most 200k steps)
    max_possible = 200000
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute conv and objective
        conv = _full_convolution(a)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv
        # update best
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagn = 0
        else:
            stagn += 1
        # gradient of max entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # compute subgradient
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # step size schedule: linear decay from eta0 to eta_end
        eta0, eta_end = 0.6, 0.02
        eta = eta0 * (1.0 - step / max_possible) + eta_end * (step / max_possible)
        # mirror update
        a = a * np.exp(-eta * g)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)
        step += 1
        if stagn >= 5000:
            # jitter
            pert = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + pert, 0.0)
            a = a / a.sum()
            stagn = 0
    return best_a, best_obj

* 3️⃣  Final polishing with a longer mirror‑descent run.
The routine respects a 1000 s wall‑clock budget and returns the best

"""
import time
import numpy as np

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself using FFT (length 2·n‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _max_refine(a: np.ndarray, steps: int = 5000, eta_start: float = 0.45) -> np.ndarray:
    """Long entropy‑mirror descent on the exact max‑entry of a*a."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # simple linear decay of η
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a

def _mirror_descent(a: np.ndarray, deadline: float, init_obj: float = None) -> tuple[np.ndarray, float]:
    """
    Exact entropy‑mirror descent that minimises max(convolution).
    Returns (best_a, best_obj) observed before ``deadline``.
    """
    n = a.shape[0]
    best_a = a.copy()
    if init_obj is None:
        conv = _full_convolution(a)
        best_obj = 2.0 * n * conv.max()
    else:
        best_obj = init_obj
    stagn = 0
    step = 0
    # a generous upper bound on the number of iterations; the schedule
    # will be clipped once ``step / max_iter`` exceeds one.
    max_iter = 200_000
    eta0, eta_end = 0.6, 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(a)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagn = 0
        else:
            stagn += 1

        # index of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            # sub‑gradient of the max entry w.r.t. a
            g[idx] = 2.0 * a[k - idx]

        # schedule η (linear decay, clipped at 1)
        prog = step / max_iter
        if prog > 1.0:
            prog = 1.0
        eta = eta0 * (1.0 - prog) + eta_end * prog

        # mirror (multiplicative) update
        a = a * np.exp(-eta * g)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        step += 1

        # occasional jitter to escape plateaus
        if stagn >= 5_000:
            perturb = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + perturb, 0.0)
            a = a / a.sum()
            stagn = 0

    return best_a, best_obj

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    """
    np.random.seed(seed)
    start = time.time()
    # safety margin for final ``evaluate_sequence`` call
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power‑of‑two makes the FFT cheap.
    # ------------------------------------------------------------------
    n = 16384

    # ------------------------------------------------------------------
    # Keep the best candidate seen so far.
    # ------------------------------------------------------------------
    best_a = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 1️⃣  Initialise from a previously found solution if it exists.
    # ------------------------------------------------------------------
    old_seq = globals().get("height_sequence_1")

        try:
            old = np.asarray(old_seq, dtype=float)
            if old.shape[0] != n:
                # linear interpolation onto the new grid
                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
            conv = _full_convolution(a0)
            obj = 2.0 * n * conv.max()

            print("[init] interpolated old solution → O={:.6f}".format(obj))
        except Exception:
            pass

    # ------------------------------------------------------------------

        s = a.sum()

        if obj < best_obj:

    # ------------------------------------------------------------------
    # 3️⃣  Random Dirichlet draws with several concentration parameters.
    # ------------------------------------------------------------------
    alpha_vals = [0.2, 0.4, 0.6, 0.8, 1.2, 2.0]

                break
            v = np.random.gamma(alpha, 1.0, size=n)
            a = v / v.sum()
            conv = _full_convolution(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj, best_a = obj, a.copy()
                print("[rand] Dirichlet(α={:.2f}) → O={:.6f}".format(alpha, obj))

    # ------------------------------------------------------------------
    # 4️⃣  Exact entropy‑mirror descent starting from the best point.
    # ------------------------------------------------------------------

        if refined_obj < best_obj:
            best_obj, best_a = refined_obj, refined
            print("[mirror] improved to O={:.6f}".format(best_obj))

    # ------------------------------------------------------------------
    # 5️⃣  Final polishing – a longer run of the same mirror descent.
    # ------------------------------------------------------------------

    if time_left > 1.0 and best_a is not None:

        est_step_time = 0.0015

        print("[final] evaluate_sequence raised:", exc)

    return final_seq
