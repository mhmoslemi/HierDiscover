"""
Optimiser for the L∞ self‑convolution objective using
soft‑max weighted gradient descent with multiplicative (exponential)
updates on the probability simplex.
The method starts from the previous solution (if available), then
alternates:
  * compute the full self‑convolution (FFT),
  * build a soft‑max distribution over convolution entries,
  * obtain the gradient of the smoothed max via two FFT‑based convolutions,
  * perform an exponential‑mirror step and renormalise,
  * decay the step size and slowly increase the soft‑max temperature.
Occasional Gaussian jitter prevents stagnation.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution x*x using real FFT (output length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Reference: Duchi et al. (2008) – Algorithm 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        a = np.maximum(a, 0.0)
        total = a.sum()
        if total < 1e-12:

        else:
