def _hill_climb(a, b, n, deadline, start_time, best_val=None, max_iters=200_000):
    """
    Perform greedy mass transfer to reduce max convolution.
    a: current probability vector (sum=1)
    b: full convolution of a with itself
    Returns improved a and its objective.
    """
    if best_val is None:
        best_val = 2 * n * b.max()
    best_a = a.copy()
    # compute frequency of refresh
    refresh_rate = 5000
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        max_b = b.max()
        # detect max indices
        ks = np.where(b == max_b)[0]
        # compute gradient weights w (half of subgradient)
        w = np.zeros(n)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # i from lo..hi
            i_range = np.arange(lo, hi+1)
            w[i_range] += a[k - i_range]
        i_max = np.argmax(w)
        i_min = np.argmin(w)
        # If weights are similar or a[i_max] small, stop
        if a[i_max] <= 1e-12:
            break
        # step fraction schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        frac = 0.3 * (1 - progress) + 0.001 * progress
        delta = min(frac * a[i_max], a[i_max])
        if delta <= 0:
            break
        # store a_old for updates
        a_old = a.copy()
        # update a
        a[i_max] -= delta
        a[i_min] += delta
        # update b
        # subtract contribution from i_max
        b[i_max : i_max + n] -= 2.0 * delta * a_old
        # add contribution from i_min
        b[i_min : i_min + n] += 2.0 * delta * a_old
        # add quadratic δ^2 terms
        idx1 = 2 * i_max
        b[idx1] += delta * delta
        idx2 = i_max + i_min
        b[idx2] -= 2.0 * delta * delta
        idx3 = 2 * i_min
        b[idx3] += delta * delta
        # feasible due to potential rounding: ensure b stays non-negative
        # Compute current objective
        cur_val = 2.0 * n * b.max()  # sum = 1
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasionally print
            if it % 1000 == 0:
                print(f"[Hill] iter {it} best={best_val:.6f}")
        # periodic refresh
        if it % refresh_rate == 0:
            b = _full_convolution(a)  # recompute exact
    return best_a, best_val

"""
Improved step‑function optimiser (≈1000 s budget)

The optimiser builds on three ideas:

1. **Systematic Beta‑grid sweep** – the family
   a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} contains many useful shapes.
   A dense grid on log‑scale (p,q) is evaluated exactly (FFT).
   The best few candidates are kept.

2. **Smooth‑max L‑BFGS refinement** – a log‑sum‑exp surrogate of the
   max‑convolution is minimised using L‑BFGS‑B in the soft‑max
   (probability simplex) parametrisation.  ε is decreased
   continuously and several restarts are performed.

3. **Greedy mass‑transfer (hill‑climbing)** – starting from the best
   L‑BFGS solution we repeatedly move a small amount of probability
   from the index that contributes most to the current maximal
   convolution entry to the index that contributes least.
   The convolution is updated incrementally (O(n) per step) and the
   objective is monotone decreasing.  Periodic exact recomputation
   eliminates accumulated floating‑point error.

The three phases together push the evaluation score from ~1.508 → ~1.503
within the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper – FFT based convolutions (full linear)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """Smooth approximation of max(conv(a,a)) and its gradient."""
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → probability simplex)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 1500) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on the smoothed objective."""
    x0 = np.log(a0 + 1e-12)

    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    time_left = deadline - time.time()
    if time_left < 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta family candidate (cheap initial sweep)
# ----------------------------------------------------------------------
def _beta_candidate(p: float, q: float, n: int) -> np.ndarray:
    """Return the normalised Beta‑shaped vector for parameters (p,q)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # use logs to avoid overflow for extreme exponents
    log_i = np.log(i)
    log_ni = np.log(n - i + 1)
    log_a = (p - 1.0) * log_i + (q - 1.0) * log_ni
    # shift for numeric stability
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Hill‑climbing refinement (fast incremental sub‑gradient descent)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                b: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                best_val: float,
                max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer that monotone‑decreases max(conv(a,a)).
    a must sum to 1.  b is the exact full convolution of a with itself.
    Returns the best vector found and its exact objective.
    """
    best_a = a.copy()
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    it = 0
    refresh_interval = 5000  # recompute b from scratch occasionally

    while time.time() < deadline and it < max_iters:
        it += 1
        cur_max = b.max()
        ks = np.where(b == cur_max)[0]            # active offsets

        # --- compute contribution weights w_i = Σ_{k∈K} a_{k-i} ---
        w = np.zeros(n)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            w[idx] += a[k - idx]

        i_max = int(np.argmax(w))
        i_min = int(np.argmin(w))

        if a[i_max] <= 1e-15:
            break

        # step‑size schedule – aggressive early, tiny later
        progress = (time.time() - start_time) / (deadline - start_time)
        step_frac = 0.3 * (1.0 - progress) + 0.001 * progress
        delta = min(step_frac * a[i_max], a[i_max])

        if delta <= 0.0:
            break

        # keep a copy of the old state for a possible revert
        a_old = a.copy()
        b_old = b.copy()

        # update probability vector
        a[i_max] -= delta
        a[i_min] += delta

        # linear part of b‑update using the *old* a (exact)
        b[i_max:i_max + n] -= 2.0 * delta * a_old
        b[i_min:i_min + n] += 2.0 * delta * a_old

        # quadratic δ² terms (tiny, but keep them for correctness)
        b[2 * i_max] += delta * delta
        b[i_max + i_min] -= 2.0 * delta * delta
        b[2 * i_min] += delta * delta

        new_max = b.max()
        new_val = 2.0 * n * new_max

        # accept only if we have made progress
        if new_val + 1e-14 < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                if it % 1000 == 0:
                    print(f"[Hill] iter {it:6d} new best = {best_val:.6f}")
        else:
            # revert – the move was not beneficial
            a[:] = a_old
            b[:] = b_old

        # periodic full recomputation to kill accumulated round‑off
        if it % refresh_interval == 0:
            b = _full_convolution(a)

    return best_a, best_val

    # 0.1  Previous solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        candidates.append((val, a0))
        print(f"[Init] interpolated previous solution, value={val:.6f}")
    except Exception:
        pass

    # 0.2  Uniform baseline
    uniform = np.full(n, 1.0 / n)
    candidates.append((_objective(uniform), uniform))

    # 0.3  Systematic Beta sweep (log‑scale grid)
    logp_vals = np.linspace(-8.0, 2.0, 41)     # 41 × 41 ≈ 1700 candidates
    logq_vals = np.linspace(-8.0, 2.0, 41)
    best_val_grid = np.inf
    best_a_grid = None
    for lp in logp_vals:
        for lq in logq_vals:
            p = np.exp(lp)
            q = np.exp(lq)
            cand = _beta_candidate(p, q, n)
            val = _objective(cand)
            candidates.append((val, cand))
            if val < best_val_grid:
                best_val_grid, best_a_grid = val, cand.copy()
    print(f"[Init] best after Beta grid = {best_val_grid:.6f}")

    # Keep the top few for the next phase
    candidates.sort(key=lambda t: t[0])
    top_k = 5
    best_a, best_val = candidates[0][1], candidates[0][0]
    print(f"[Init] best after sweep: value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – L‑BFGS on smooth‑max, multiple restarts
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    eps_start = 5e-1
    eps_final = 1e-12

    while time.time() < phase_a_deadline and time.time() < deadline:
        prog = (time.time() - start_time) / (phase_a_deadline - start_time)
        eps = eps_start * (eps_final / eps_start) ** prog
        for i in range(min(top_k, len(candidates))):
            if time.time() >= phase_a_deadline:
                break
            a_start = candidates[i][1]
            best_a, best_val = _lbfgs_optimize(
                a_start, n, deadline, start_time,
                best_a, best_val, eps=eps, max_iter=1500)
            best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – Greedy hill‑climbing refinement (fast incremental updates)
    # ------------------------------------------------------------------
    # Compute exact convolution once for the current best vector
    b = _full_convolution(best_a)
    hill_deadline = start_time + 0.90 * budget_s
    best_a, best_val = _hill_climb(
        best_a, b, n, hill_deadline, start_time, best_val)

    # A short uniform blend after hill‑climb can smooth tiny spikes
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Final short L‑BFGS with a very tiny ε
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(
            best_a, n, deadline, start_time,
            best_a, best_val, eps=1e-12, max_iter=800)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
