def propose_candidate(...):
    import numpy as np, time, math

    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8000

    # Precompute index arrays for mirror descent
    # maybe not needed.

    best_seq = None
    best_obj = np.inf

    def evaluate_obj(a):
         conv = _full_convolution(a)
         return 2.0 * n * conv.max()

    # function to run mirror descent
    def run_refine(init_a, max_steps):
         a = init_a.copy()
         best_local = a.copy()
         best_local_obj = evaluate_obj(a)
         for step in range(max_steps):
             conv = _full_convolution(a)
             k = int(conv.argmax())
             lo = max(0, k-(n-1))
             hi = min(k, n-1)
             g = np.zeros_like(a)
             if lo<=hi:
                 idx = np.arange(lo, hi+1)
                 g[idx] = 2.0 * a[k - idx]
             eta = 0.5 * (1 - step / max_steps) # linearly decay
             a = a * np.exp(-eta * g)
             a_sum = a.sum()
             a = a / a_sum if a_sum>0 else np.full_like(a, 1.0/n)
             # compute obj
             obj = evaluate_obj(a)
             if obj < best_local_obj:
                 best_local_obj = obj
                 best_local = a.copy()
         return best_local, best_local_obj
```

"""
Propose a candidate step‑function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))² .

The algorithm explores many low‑dimensional families of distributions
(e.g. Beta, power‑law, linear) and refines each candidate with
entropy‑mirror descent (exponentiated‑gradient on the exact max‑entry of
the self‑convolution).  A small number of randomly‑generated Dirichlet
vectors are also tried.  The best solution found within the time budget
is returned.

Key ideas
---------
* The objective is scale‑invariant → we normalise every vector to sum = 1.
* The optimal vector makes the linear convolution “as flat as possible”.
  Mirror descent on the max‑convolution entry is a fast way to push the
  distribution toward this flatness.
* By starting the descent from many diverse initial shapes we increase
  the chance of escaping the local minimum found by the previous
  implementation.
* Time‑management guarantees a return before the 1000 s budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution via FFT
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent on the exact max‑entry of the self‑convolution
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 5000,
                eta_start: float = 0.5, eta_end: float = 0.01) -> np.ndarray:
    """
    Perform exponentiated‑gradient (mirror) descent on the max entry of
    the self‑convolution.  The step size η decays linearly from
    ``eta_start`` to ``eta_end``.
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]          # sub‑gradient of the max entry

        # linearly decaying step size
        eta = eta_start * (1.0 - t / steps) + eta_end * (t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Objective (the part that varies with the vector; sum = 1 ⇒ 2·n·max)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Compute 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of initial vectors (all normalised to sum = 1)
# ----------------------------------------------------------------------
def _init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _init_linear_descending(n: int) -> np.ndarray:
    """a_i ∝ (n-i) (descending from left to right)."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _init_quadratic_peak(n: int) -> np.ndarray:
    """a_i ∝ (i+1)*(n-i) – a symmetric hill."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    return a / a.sum()


def _init_quadratic_valley(n: int) -> np.ndarray:
    """U‑shaped: a_i ∝ 1/((i+1)*(n-i)+ε)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / (i * (n - i + 1) + eps)
    return a / a.sum()


def _init_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β)‑like density on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _init_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))**p  (handle large/small p safely)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # work in log‑space to avoid overflow/underflow
    log_a = p * (np.log(i) + np.log(n - i + 1))
    # shift for numerical stability
    log_a -= log_a.max()
    a = np.exp(log_a)
    return a / a.sum()


def _init_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw with given concentration (α)."""
    # Gamma(α,1) samples – for α small the distribution is sparse,
    # for α large it is smooth.
    g = rng.gamma(concentration, 1.0, size=n)
    if g.sum() == 0.0:
        g = np.full(n, 1.0)
    return g / g.sum()


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative sequence that (hopefully) improves on the
    previous best bound (≈1.508).  The routine respects the ~1000 s
    time budget, tries many different initial shapes and refines each
    with entropy‑mirror descent.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a margin for the final call to evaluate_sequence
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – keep 8000 as in the reference implementation.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Prepare a list of initial‑vector generators.
    # ------------------------------------------------------------------
    init_fns = []

    # 1) previous solution, interpolated if needed
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        try:
            old = np.asarray(old_seq, dtype=float)
            if old.size != n:
                # interpolate to required length
                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old.size)
                a = np.interp(x_new, x_old, old)
                a = np.maximum(a, 0.0)
                a /= a.sum() if a.sum() > 0 else np.full_like(a, 1.0 / n)
            else:
                a = np.maximum(old, 0.0)
                a /= a.sum() if a.sum() > 0 else np.full_like(a, 1.0 / n)
            init_fns.append(lambda a=a: a.copy())
            print("[init] added previous solution (interpolated if needed)")
        except Exception as exc:
            print("[init] could not use previous solution:", exc)

    # 2) deterministic families
    init_fns.append(lambda n=n: _init_uniform(n))
    init_fns.append(lambda n=n: _init_linear_descending(n))
    init_fns.append(lambda n=n: _init_quadratic_peak(n))
    init_fns.append(lambda n=n: _init_quadratic_valley(n))

    # 3) beta family (symmetrical α=β)
    for a in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0]:
        init_fns.append(lambda n=n, a=a: _init_beta(n, a, a))

    # 4) power‑law family
    for p in [-1.5, -1.0, -0.7, -0.5, -0.3, 0.0, 0.3, 0.5, 0.7, 1.0, 1.5]:
        init_fns.append(lambda n=n, p=p: _init_power(n, p))

    # 5) random Dirichlet draws (different concentrations)
    for conc in [0.1, 0.3, 0.6, 1.0, 2.0, 5.0]:
        for _ in range(3):                         # three seeds per concentration
            init_fns.append(lambda n=n, c=conc, rng=rng: _init_dirichlet(n, c, rng))

    # ------------------------------------------------------------------
    # Book‑keeping for the best solution
    # ------------------------------------------------------------------
    best_a = None
    best_obj = np.inf
    n_iter_total = 0

    # ------------------------------------------------------------------
    # Helper to run mirror descent for a limited number of steps (or
    # until the global deadline is reached).
    # ------------------------------------------------------------------
    def _refine_until_deadline(start_a: np.ndarray,
                               max_steps: int = 4000) -> tuple[np.ndarray, float]:
        a = start_a.copy()
        # keep a local best for this run
        local_best_a = a.copy()
        local_best_obj = _objective(a, n)

        for step in range(max_steps):
            if time.time() >= deadline:
                break
            conv = _full_convolution(a)
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)

            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                g[idx] = 2.0 * a[k - idx]

            # Linear decay of η (0.5 → 0.01)
            eta = 0.5 * (1.0 - step / max_steps) + 0.01 * (step / max_steps)
            a = a * np.exp(-eta * g)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s

            obj_now = float(2.0 * n * conv.max())   # same as _objective
            if obj_now < local_best_obj:
                local_best_obj = obj_now
                local_best_a = a.copy()
        return local_best_a, local_best_obj

    # ------------------------------------------------------------------
    # Main exploration loop – iterate over initialisers while time permits.
    # ------------------------------------------------------------------
    for init_idx, init_fn in enumerate(init_fns, start=1):
        if time.time() >= deadline:
            break

        # generate a fresh candidate
        try:
            a0 = init_fn()
        except Exception as exc:
            print(f"[skip] initialiser #{init_idx} raised {exc}")
            continue

        # safety – enforce sum=1 (the helpers already do this but we double‑check)
        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        if s0 == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s0

        # quick objective for the raw start (useful for pruning)
        start_obj = _objective(a0, n)

        # allocate a modest number of refinement steps; the exact count may be
        # reduced if we are close to the deadline.
        steps_budget = 3000
        remaining = deadline - time.time()
        if remaining < 5.0:            # not enough time for another refinement
            break

        refined_a, refined_obj = _refine_until_deadline(a0, max_steps=steps_budget)

        n_iter_total += steps_budget
        print(f"[run {init_idx:02d}/{len(init_fns):02d}] "
              f"start O={start_obj:.6f} → refined O={refined_obj:.6f}")

        if refined_obj < best_obj:
            best_obj = refined_obj
            best_a = refined_a.copy()
            print(f"[new best] O={best_obj:.6f} (from init #{init_idx})")

    # ------------------------------------------------------------------
    # Final polishing: a longer mirror‑descent run on the global best.
    # ------------------------------------------------------------------
    if best_a is not None:
        # Use a more generous step budget for polishing (up to 20000 steps
        # or whatever time is left).
        steps_polish = 20000
        time_left = deadline - time.time()
        # Rough estimate: 0.001 s per step ⇒ cap at time_left / 0.001
        max_allowed = int(time_left / 0.001)
        steps_polish = min(steps_polish, max_allowed, 25000)

        print(f"[polish] starting final refinement (≤{steps_polish} steps)")
        polished_a, polished_obj = _refine_until_deadline(best_a, max_steps=steps_polish)
        if polished_obj < best_obj:
            best_obj, best_a = polished_obj, polished_a.copy()
            print(f"[polish] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback – a simple uniform distribution (guaranteed feasible)
        best_a = _init_uniform(n)

    final_seq = best_a.tolist()
    # sanity check – evaluate via the provided wrapper (may be omitted in
    # the final competition run to save a few ms).
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence → {final_score:.6f}   (approx O={best_obj:.6f})")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
