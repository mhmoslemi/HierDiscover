def full_conv_fft(x):
    n = len(x)
    out_len = 2*n-1
    size = 1 << (out_len-1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx*fx, size)[:out_len]
    return conv
```

"""
Propose a non‑negative step function that (approximately) minimizes
``evaluate_sequence``.  The method is a two‑phase search:

1. **Parametric sweep** –  a collection of simple families (power‑law,
   exponential, logistic) and mixtures with the uniform distribution
   are sampled.  Each candidate is normalised to sum = 1 and its exact
   objective   2·n·max(conv) (for sum = 1) is evaluated via an FFT‑based
   convolution.  The best candidate from this coarse scan is retained.

2. **Refinement** –  starting from the best candidate we run a few
   mirror‑descent / exponentiated‑gradient iterations.  The sub‑gradient
   of the max‑convolution is computed analytically, a momentum term is
   added, and the update is performed multiplicatively followed by a
   projection onto the simplex (simple renormalisation).  The loop stops
   before the time budget expires.

The algorithm respects the 1000 s budget, prints progress and the
final score, and returns the best sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines – FFT convolution, objective, sub‑gradient
# ----------------------------------------------------------------------


def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a*a using FFT.
    Returns an array of length 2*len(x)-1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum(a)=1):
        2 * n * max(conv(a,a))
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     max_active: int = 5,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Uses the most active lags (those attaining the maximal value
    within a relative tolerance).
    """
    n = a.shape[0]
    max_val = conv.max()
    # pick up to max_active indices where conv is near the maximum
    if max_active >= conv.size:
        active = np.arange(conv.size)
    else:
        idx = np.argpartition(conv, -max_active)[-max_active:]
        mask = conv[idx] >= max_val * (1.0 - rel_tol)
        if np.any(mask):
            active = idx[mask]
        else:
            active = np.array([int(np.argmax(conv))])

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad

def _monotone_power_shape(p: float, n: int) -> np.ndarray | None:
    """a_i ∝ (i+1)^p, normalised to sum=1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def _exp_shape(alpha: float, n: int) -> np.ndarray:
    """a_i ∝ exp(alpha * i/(n-1)), normalised to sum=1."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(alpha * i)
    a /= a.sum()
    return a


def _logistic_shape(k: float, c: float, n: int) -> np.ndarray:
    """Increasing sigmoid shape, normalised to sum=1."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(-k * (i - c)))
    a /= a.sum()
    return a

    # ------------------------------------------------------------------
    a = best_seq.copy()
    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline - 0.4:
        conv = _full_convolution_fft(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_val:
            best_val = cur_obj
            best_seq = a.copy()
            print(f"[refine] iter {it} new best {best_val:.6f}")

        # sub‑gradient of max(conv)
        grad = _subgradient_max(a, conv, max_active=5)

        # momentum term
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size that decays linearly with elapsed time
        elapsed = time.time() - start
        progress = elapsed / (deadline - start)
        step = 0.15 * (1.0 - progress) + 1e-5   # small positive floor

        # exponentiated update (keeps positivity)
        log_factor = -step * momentum
        # avoid overflow/underflow
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = uniform.copy()
        else:
            a /= s

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.001:
            jitter = np.random.normal(scale=5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            momentum.fill(0.0)

        it += 1

    # ------------------------------------------------------------------
    # 7️⃣  Final evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
