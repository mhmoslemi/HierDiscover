def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8000
    best_seq = None
    best_val = np.inf  # objective from surrogate

    # Helper functions defined globally: full_convolution, objective, project_simplex, subgradient_max
    # Already implement these top-level.

    # Define evaluation function: surrogate objective (2*n*max_b; sum normalized to 1)
    def surrogate(a):
        return _objective(a)  # expects sum=1

    # We'll keep exponent p search as before to have strong starting point:
    best_seq = _monotone_power_shape(5.5, n)
    best_val = surrogate(best_seq)
    print(f"init from power shape p=5.5 objective {best_val}")

    # We'll also evaluate other param families (beta)
    for p in np.arange(0.5, 10.5, 0.5):
        a = _monotone_power_shape(p, n)
        if a is not None:

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"grid power p={p} improved to {val}")

    for p,q in [(5.5,0.5), (6,0.2), (5,0.5), (7,0.3)]:

        if a is not None:
            val = surrogate(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"beta shape p={p} q={q} improved to {val}")

    # Define gradient descent inner function
    def refine(a0, max_iters=2000):
        a = a0.copy()

        beta_mom=0.9
        # step schedule maybe exponential
        for it in range(max_iters):

            # momentum
            m = beta_mom*m + (1-beta_mom)*grad
            # step size: linearly decreasing
            elapsed = time.time() - start_local
            # note: start_local captured earlier
            # step = initial * (1 - elapsed/full_time) + eps

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of size power‑of‑2."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact surrogate objective for a normalised sequence (Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _monotone_power_shape(p: float, n: int) -> np.ndarray | None:
    """
    a_i ∝ (i+1)^p,   i = 0 … n‑1.
    Returns a normalised vector (Σ = 1) or None if the computation fails.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, p)
    total = a.sum()
    if total == 0.0:
        return None
    a /= total
    return a


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Discretised Beta(p,q) density on {0,…,n‑1}:
        a_i ∝ (i+1)^{p‑1}·(n‑i)^{q‑1}.
    Computed in log‑space to avoid overflow.
    """
    if p <= 0 or q <= 0:
        return None
    i = np.arange(1, n + 1, dtype=np.float64)               # i = 1 … n
    j = n - i + 1                                          # n‑i+1, also ≥1
    # work in log‑space
    log_a = (p - 1.0) * np.log(i) + (q - 1.0) * np.log(j)
    log_a -= np.max(log_a)                                 # stabilise
    a = np.exp(log_a)
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 5,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Sub‑gradient of max(b) w.r.t. a.
    ``b`` is the full convolution a*a.
    Only the ``max_active`` largest entries (or those within ``rel_tol``
    of the maximum) contribute to the gradient.
    """
    n = a.shape[0]
    max_val = b.max()
    if max_active >= b.size:
        active = np.arange(b.size)
    else:
        # indices of the biggest entries
        active = np.argpartition(b, -max_active)[-max_active:]
        mask = b[active] >= max_val * (1.0 - rel_tol)
        active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget (seconds) and returns the
    best sequence (list of floats) found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the final eval

    # --------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between expressiveness
    # and speed for the beta‑family search.
    # --------------------------------------------------------------
    n = 8000

    best_seq: np.ndarray | None = None
    best_val = np.inf

    # --------------------------------------------------------------
    # Helper to update the global best using the cheap surrogate objective.
    # --------------------------------------------------------------
    def _maybe_update(cand: np.ndarray, descr: str = "") -> None:
        nonlocal best_seq, best_val
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[update] {descr} -> val={best_val:.6f}")

    # --------------------------------------------------------------
    # 0️⃣  If a previous solution exists, start from its interpolation.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)      # type: ignore[name-defined]
        x_new = np.linspace(0.0, 1.0, n)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        cand = np.interp(x_new, x_old, old)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        _maybe_update(cand, "interpolated old")
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Sweep the monotone power‑law family (coarse grid).
    # --------------------------------------------------------------
    for p in np.arange(0.5, 10.5, 0.5):
        cand = _monotone_power_shape(p, n)
        if cand is not None:
            _maybe_update(cand, f"power p={p:.1f}")
        if time.time() >= deadline - 200.0:
            break

    # --------------------------------------------------------------
    # 2️⃣  Exhaustive (yet cheap) grid search over the Beta‑family.
    # --------------------------------------------------------------
    p_vals = np.arange(1.0, 12.5, 0.25)                     # left‑shape exponent
    q_vals = np.arange(0.05, 3.0, 0.25)                     # right‑tail exponent
    for p in p_vals:
        for q in q_vals:
            if time.time() >= deadline - 150.0:
                break
            cand = _beta_shape(p, q, n)
            if cand is None:
                continue
            _maybe_update(cand, f"beta p={p:.2f} q={q:.2f}")
        if time.time() >= deadline - 150.0:
            break

    # --------------------------------------------------------------
    # 3️⃣  Local random perturbations of the best (p,q) pair.
    # --------------------------------------------------------------
    if best_seq is not None:
        # Extract approximate (p,q) from the best sequence if it came
        # from the beta‑family – we simply reuse the best_seq as a warm‑start.
        rng = np.random.default_rng()
        for _ in range(1200):
            if time.time() >= deadline - 120.0:
                break
            # Random walk in the (p,q) space
            dp = rng.normal(scale=0.2)
            dq = rng.normal(scale=0.2)
            # Bounds to stay in the explored region
            p_new = np.clip(p + dp, 1.0, 12.0)
            q_new = np.clip(q + dq, 0.05, 3.0)
            cand = _beta_shape(p_new, q_new, n)
            if cand is not None:
                _maybe_update(cand, f"beta pert p={p_new:.2f} q={q_new:.2f}")

    # --------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient refinement (mirrored descent).
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)

    a = best_seq.copy()
    iter_no = 0
    # step size schedule: α/√(t+1)
    alpha0 = 0.5

    while time.time() < deadline:
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[optim] iter {iter_no} new best {best_val:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_max(a, conv, max_active=5, rel_tol=1e-6)

        # diminishing step size (projected sub‑gradient)
        step = alpha0 / np.sqrt(iter_no + 1.0)
        a = a - step * grad
        a = _project_simplex(a)

        # occasional jitter to escape flat regions
        if iter_no % 600 == 0 and iter_no > 0:
            jitter = np.random.normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        iter_no += 1
        # paranoid safety‑check against pathological NaNs
        if not np.isfinite(a).all():
            a = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 5️⃣  Final secure evaluation (only for logging – the outer framework
    #      will call ``evaluate_sequence`` itself as well).
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
