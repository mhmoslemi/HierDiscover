def _beta_shape(n, p, q):
    i = np.arange(1, n+1, dtype=np.float64)
    log_a = (p - 1) * np.log(i) + (q - 1) * np.log(n - i + 1)
    # subtract max to avoid overflow
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    a /= s
    return a

   * systematic Beta(p,q) grid,
   * random Beta,
   * Dirichlet draws,
   * mixtures of two Betas.
   Exact objective `2·n·max(conv(a,a))` is evaluated and the best few are kept.

2. **Smooth‑max L‑BFGS** – minimise a log‑sum‑exp surrogate of the max.
   Variables are re‑parameterised through a soft‑max, guaranteeing the simplex.
   The smoothing parameter ε is exponentially decreased during the phase.

3. **Exact sub‑gradient polishing** – projected sub‑gradient descent on the true
   (non‑smooth) objective refines the solution.

4. **Local random mass‑transfer search** – hill‑climbing on the exact max
   using cheap O(n) updates of the convolution; accepts only improving moves.

5. **Final short L‑BFGS** – a last pass with a very tiny ε to polish the best
   candidate.

The routine respects the 1000 s budget, prints progress, and finally returns
the best sequence found (as a plain Python list of floats)."""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper – FFT based convolutions (full linear)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.  The gradient already includes the factor 2·n.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # we need only the n entries that correspond to the “forward” part
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → probability simplex)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Beta‑shape generator (log‑space, numerically safe)
# ----------------------------------------------------------------------
def _beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Return the normalised Beta(p,q) shape on a grid of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # work in log‑space to avoid overflow/underflow
    log_a = (p - 1.0) * np.log(i) + (q - 1.0) * np.log(n - i + 1.0)
    log_a -= np.max(log_a)                # stabilise
    a = np.exp(log_a)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random Beta‑shape (log‑uniform p,q)."""
    logp = rng.uniform(-8.0, 2.0)
    logq = rng.uniform(-8.0, 2.0)
    p = np.exp(logp)
    q = np.exp(logq)
    return _beta_shape(n, p, q)


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 1500) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smoothed objective.
    """
    x0 = np.log(a0 + 1e-12)      # initialise in log‑space

    # ----- objective ----------------------------------------------------
    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    # ----- gradient -----------------------------------------------------
    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)              # chain rule through soft‑max
        return a * (grad_a - dot)

    time_left = deadline - time.time()
    if time_left < 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:                 # pragma: no cover
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum over all maximal diagonals
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule (aggressive early, tiny later)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 400 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Local random mass‑transfer hill‑climbing (exact max)
# ----------------------------------------------------------------------
def _local_search(a_start: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_val: float,
                  rng: np.random.Generator,
                  max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that moves a small amount of mass from one index to
    another and keeps the move only if it strictly reduces the exact objective.
    Uses O(n) updates of the convolution (no full FFT each iteration).
    """
    a = a_start.copy()
    b = _full_convolution(a)                     # length 2n‑1
    cur_val = 2.0 * n * b.max()
    best_a, best_val_local = a.copy(), cur_val

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # pick a source index with probability proportional to its mass
        if a.sum() == 0:
            break
        i = rng.choice(n, p=a)
        if a[i] < 1e-12:
            continue

        # distinct target index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # choose a fraction of the source mass to move
        frac = rng.uniform(0.01, 0.20)                     # 1‑20 % moves
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # keep a snapshot of b to allow cheap revert
        b_old = b.copy()

        # --- update convolution (linear, O(n)) ---------------------------
        # removal from i
        len_i = min(n, len(b) - i)
        b[i:i + len_i] -= 2.0 * delta * a[:len_i]

        # addition to j
        len_j = min(n, len(b) - j)
        b[j:j + len_j] += 2.0 * delta * a[:len_j]

        # --- update a ----------------------------------------------------
        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()

        if new_val + 1e-13 < best_val_local:          # strict improvement
            best_val_local = new_val
            best_a = a.copy()
            # keep the improved a, b as current state
        else:
            # revert
            a[i] += delta
            a[j] -= delta
            b = b_old

    return best_a, best_val_local


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and that minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Problem size – a larger n gives a richer search space.
    n = 20000

    candidates = []

    # ----- 0.1  Interpolated previous solution (if available) -----
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.ndim == 1 and old.size > 0:
            old_len = old.shape[0]
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            val = _objective(a0)
            candidates.append((val, a0))
            print(f"[Init] interpolated previous solution, value={val:.6f}")
    except Exception:
        pass

    # ----- 0.2  Uniform baseline -----
    uniform = np.full(n, 1.0 / n)
    candidates.append((_objective(uniform), uniform))

    # ----- 0.3  Grid of Beta(p,q) shapes (log‑space) -----
    log_grid = np.linspace(-8.0, 2.0, 11)          # 11×11 = 121 points
    for logp in log_grid:
        for logq in log_grid:
            if time.time() >= deadline:
                break
            p = np.exp(logp)
            q = np.exp(logq)
            a = _beta_shape(n, p, q)
            val = _objective(a)
            candidates.append((val, a))

    # ----- 0.4  Random Beta candidates -----
    for _ in range(200):
        if time.time() >= deadline:
            break
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        candidates.append((val, cand))

    # ----- 0.5  Dirichlet draws with a handful of concentration parameters -----
    for alpha in (0.1, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            if time.time() >= deadline:
                break
            # use np.random.dirichlet (may be a bit heavy for huge n, but few draws)
            a = rng.dirichlet(np.full(n, alpha))
            val = _objective(a)
            candidates.append((val, a))

    # ----- 0.6  Mixtures of two Beta distributions -----
    for _ in range(100):
        if time.time() >= deadline:
            break
        p1 = np.exp(rng.uniform(-8.0, 2.0))
        q1 = np.exp(rng.uniform(-8.0, 2.0))
        p2 = np.exp(rng.uniform(-8.0, 2.0))
        q2 = np.exp(rng.uniform(-8.0, 2.0))
        t = rng.random()
        a1 = _beta_shape(n, p1, q1)
        a2 = _beta_shape(n, p2, q2)
        a = t * a1 + (1.0 - t) * a2
        a /= a.sum()
        val = _objective(a)
        candidates.append((val, a))

    # ------------------------------------------------------------------
    # Keep the best few for the next phases
    # ------------------------------------------------------------------
    candidates.sort(key=lambda t: t[0])
    top_k = 8
    best_a, best_val = candidates[0][1], candidates[0][0]
    top_candidates = [c[1] for c in candidates[:top_k]]
    print(f"[Init] best after coarse sweep: value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – L‑BFGS on smooth‑max, multiple restarts
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    eps_start = 1.0
    eps_final = 1e-12

    while time.time() < phase_a_deadline and time.time() < deadline:
        prog = (time.time() - start_time) / (phase_a_deadline - start_time)
        eps = eps_start * (eps_final / eps_start) ** prog
        for a_start in top_candidates:
            if time.time() >= phase_a_deadline:
                break
            best_a, best_val = _lbfgs_optimize(
                a_start, n, deadline, start_time,
                best_a, best_val, eps=eps, max_iter=1200)
            best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient polishing (≈40 % of remaining budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.85 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=300_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Local random mass‑transfer hill climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _local_search(
            best_a, n,
            deadline=deadline,
            start_time=start_time,
            best_val=best_val,
            rng=rng,
            max_iters=200_000,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase D – Final short L‑BFGS with a very tiny ε
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(
            best_a, n, deadline, start_time,
            best_a, best_val, eps=1e-14, max_iter=800)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
