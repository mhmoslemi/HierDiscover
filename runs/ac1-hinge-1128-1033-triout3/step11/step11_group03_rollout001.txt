"""
Optimisation of the max‑convolution bound for a non‑negative step function.

The algorithm works in three stages:

1️⃣  **Initialisation** – generate a handful of diverse seeds:
    * an exponential decay,
    * a linear “ramp’’,
    * a triangular shape,
    * a few Dirichlet draws,
    * and (if available) the previous best solution interpolated to the current length.

2️⃣  **Smooth‑max Adam optimisation** – minimise a differentiable surrogate
    `f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)`.  The surrogate approaches the true
    objective as α grows.  We use Adam with Euclidean projection onto the probability
    simplex (∑a_i = 1, a_i ≥ 0).  The schedule is aggressive:
        α : 1 → 800,
        learning‑rate : 0.2 decaying with 1/√t,
        up to the remaining time‑budget.
    The best vector found during the run is kept.

3️⃣  **Mass‑transfer hill‑climb** – a lightweight, greedy polishing phase.
    Small amounts of probability mass are moved from indices that contribute most
    to the current maximal lag to indices that contribute little.  The move is kept
    only when it reduces the exact objective.
    The phase stops early if the target value `1.503` is reached.

All arithmetic is performed with the sum fixed to 1; because the true score
`2·n·max(a∗a)/sum(a)^2` is invariant under uniform scaling, the final
sequence may be returned as‑is.  The procedure respects the 1000 s time
budget and prints intermediate best values for diagnostics.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_rev(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·∑_i a_i·w_{i+j} for j=0..n‑1.
    Implements the gradient of the smooth‑max surrogate.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    l_w = w.shape[0]
    out_len = n + l_w - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]
    start = n - 1
    return conv[start:start + n]        # length n


# ----------------------------------------------------------------------
# Objective helpers (exact, scale‑invariant)
# ----------------------------------------------------------------------
def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact value 2·n·max(a∗a) given the full convolution."""
    return float(2.0 * n * conv.max())


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (assumed non‑negative, sum>0)."""
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.shape[0])


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (∑x = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx = 1} (Euclidean)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(α·conv_k).  Returns a probability vector."""
    maxc = conv.max()
    exp_vals = np.exp(alpha * (conv - maxc))   # shift for stability
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exp_vals / sum_exp

# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------

                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser of the smooth objective.
    Returns the best probability vector (∑a=1) and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s                         # enforce ∑a = 1

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_conv = _full_convolution(a)
    cur_val = _objective_from_conv(cur_conv, n)   # exact objective = 2·n·max(conv)
    best_a = a.copy()
    best_val = cur_val

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear schedule for α
        prog = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        alpha = alpha_start + (alpha_end - alpha_start) * prog

        # compute soft‑max weights and gradient
        w = _softmax_weights(cur_conv, alpha)      # shape (2n‑1,)
        grad = 2.0 * _conv_rev(a, w)                # shape (n,)

        # Adam update with a slowly decaying learning‑rate
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = (lr0 / np.sqrt(t + 1)) / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        cand_conv = _full_convolution(a_candidate)
        cand_val = _objective_from_conv(cand_conv, n)

        # accept if we do not increase the exact objective
        if cand_val <= cur_val:
            a = a_candidate
            cur_conv = cand_conv
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[Adam] iter {t:7d} α={alpha:6.1f} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            # no improvement – keep the old point
            pass

        # occasional jitter to avoid stagnation
        if t % 6000 == 0:
            a = best_a.copy()
            jitter = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)
            cur_conv = _full_convolution(a)
            cur_val = _objective_from_conv(cur_conv, n)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing (mass‑transfer) for final polishing
# ----------------------------------------------------------------------
def _final_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503,
                     max_no_improve: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Very simple greedy hill‑climb that moves a small amount of mass from a
    “guilty’’ index (large contribution to the maximal lag) to a “light’’ one.
    Stops early if the target value is reached or time runs out.
    """
    b = _full_convolution(a)
    cur_val = _objective_from_conv(b, n)
    best_val = cur_val
    best_a = a.copy()

    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # locate the lag giving the maximal autocorrelation
        k_max = int(np.argmax(b))

        # indices that contribute to this lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        # contribution of each position to the offending lag
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (heavy contributor) – sample proportional to contribution
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-14:
            continue

        # target index (light contributor)
        inv = contrib.max() - contrib + 1e-14
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        # move a small random fraction of the source mass
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-14:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = _objective_from_conv(b_new, n)

        if new_val < cur_val:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            # random jitter around current best to escape a flat region
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = _objective_from_conv(b, n)
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _exponential_candidate(n: int, rate: float = 5.0) -> np.ndarray:
    """Monotonically decreasing exponential shape a_i ∝ exp(-rate·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i / n)
    a /= a.sum()
    return a


def _linear_candidate(n: int) -> np.ndarray:
    """Linear ramp decreasing from 1 to 0."""
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a /= a.sum()
    return a


def _triangular_candidate(n: int) -> np.ndarray:
    """Symmetric triangle – rises then falls."""
    half = n // 2
    inc = np.linspace(1.0, 0.0, half, endpoint=False, dtype=np.float64)
    dec = inc[::-1]
    if n % 2 == 1:
        middle = np.array([0.0], dtype=np.float64)
        a = np.concatenate([inc, middle, dec])
    else:
        a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def _interpolate_previous(prev_seq, n):
    """Resample a previously found sequence to a new length n using linear interpolation."""
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s time
    budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Choose problem sizes (we allocate more time to the largest one)
    # ------------------------------------------------------------------
    # small size for quick scouting, large size for final optimisation
    sizes = [20000, 50000]
    # optionally extend if enough time remains
    remaining = deadline - time.time()
    if remaining > 180.0:
        sizes.append(80000)

    # ------------------------------------------------------------------
    # Retrieve a previously found solution, if the notebook provides it.
    # ------------------------------------------------------------------
    try:
        previous = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
    except Exception:
        previous = None

    global_best_a = None
    global_best_val = np.inf
    global_best_n = None

    # ------------------------------------------------------------------
    # Loop over chosen lengths, always keeping the best so far.
    # ------------------------------------------------------------------
    for n in sizes:
        # --------------------------------------------------------------
        # Build a small pool of seeds for this length
        # --------------------------------------------------------------
        seed_pool = []

        # a) interpolated previous solution (if any)
        if previous is not None:
            a0 = _interpolate_previous(previous, n)
            if a0 is not None:
                seed_pool.append(a0)

        # b) handcrafted parametric families
        seed_pool.append(_exponential_candidate(n, rate=5.0))
        seed_pool.append(_exponential_candidate(n, rate=10.0))
        seed_pool.append(_linear_candidate(n))
        seed_pool.append(_triangular_candidate(n))

        # c) a few random Dirichlet draws
        for _ in range(3):
            seed_pool.append(np.random.dirichlet(np.ones(n)))

        # --------------------------------------------------------------
        # Pick the best starter according to the exact objective.
        # --------------------------------------------------------------
        best_start = None
        best_start_val = np.inf
        for cand in seed_pool:
            val = _objective(cand)
            if val < best_start_val:
                best_start_val = val
                best_start = cand

        print(f"[init] n={n} starter value={best_start_val:.6f}")

        # --------------------------------------------------------------
        # Allocate time for this size: a fraction of the remaining budget.
        # --------------------------------------------------------------
        now = time.time()
        remaining = deadline - now
        if remaining <= 0.0:
            break
        # Keep at most ~70 % of the remaining budget for Adam optimisation;
        # the rest will be reserved for the final hill‑climb.
        opt_deadline = now + min(remaining * 0.7, max(20.0, remaining - 10.0))

        # --------------------------------------------------------------
        # Run the smooth‑max Adam optimiser
        # --------------------------------------------------------------
        a_opt, val_opt = _smooth_adam_optimize(
            best_start,
            n,
            opt_deadline,
            start_time,
            alpha_start=1.0,
            alpha_end=800.0,
            lr0=0.2,
            target_val=1.503,
        )
        print(f"[opt] n={n} refined value={val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_a = a_opt.copy()
            global_best_n = n

        # early exit if we have already met the target
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Final polishing with hill‑climbing (time‑boxed)
    # ------------------------------------------------------------------
    if global_best_a is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            hill_deadline = time.time() + remaining
            global_best_a, global_best_val = _final_hillclimb(
                global_best_a,
                global_best_n,
                deadline=hill_deadline,
                start_time=start_time,
                target_val=1.503,
            )
            print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Fallback in the unlikely event that nothing was found.
    # ------------------------------------------------------------------
    if global_best_a is None:
        n = sizes[0]
        global_best_a = np.full(n, 1.0 / n)
        global_best_val = _objective(global_best_a)

    # ------------------------------------------------------------------
    # 4️⃣  Log the final score using the provided evaluation function.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return the best sequence discovered (as a plain Python list)
    return global_best_a.tolist()
