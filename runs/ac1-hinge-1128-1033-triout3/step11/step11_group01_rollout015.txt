def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    best_a = None
    best_val = float('inf')
    # List of n sizes to try: [2000, 4000, 8000] maybe.
    while time.time() < deadline:
        # choose random n from candidates
        n = np.random.choice([2000, 3000, 4000, 8000])
        # random init: either uniform or Dirichlet with random concentration
        if np.random.rand() < 0.5:
            a = np.full(n, 1.0/n)
        else:
            alpha = np.random.rand()*0.5   # small
            a = np.random.dirichlet(np.full(n, alpha))
        # parameters
        lr = 0.1 * (1 + np.random.rand())  # between 0.1 and 0.2
        lambda_start = np.random.uniform(0.5, 10.0)
        lambda_end = np.random.uniform(500, 3000)
        # run optimization for a limited time: e.g., 20 seconds
        cur_a, cur_val = run_opt(a, n, lr, lambda_start, lambda_end, limit=deadline - time.time())
        if cur_val < best_val:
            best_val = cur_val
            best_a = cur_a.copy()
            print(f"[global] new best val {best_val:.9f} with n={n}")
    # final evaluate
    final_score = evaluate_sequence(best_a.tolist())
    print("[final] final score", final_score)
    return best_a.tolist()

"""
Simulated‑annealing optimiser for the max‑convolution objective.
The algorithm works directly on the probability vector (sum‑to‑one
non‑negative) and updates the full self‑convolution incrementally.
A temperature schedule and a decreasing step‑size allow the search
to escape the uniform stationary point that traps gradient‑based
methods.  The best sequence seen during the allotted budget is
returned.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (real FFT)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv
