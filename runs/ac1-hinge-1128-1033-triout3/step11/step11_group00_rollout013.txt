def _full_convolution(x):
    n = x.shape[0]
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx*fx, size)[:out_len]
    return conv

def _full_convolution_two(x, y):
    n = x.shape[0]; m = y.shape[0];
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

def _max_refine(a, steps=5000, eta_start=0.5):
    a = np.array(a, copy=True)
    n = a.shape[0]
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        g = np.zeros_like(a)
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        if lo <= hi:
            idx = np.arange(lo, hi+1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # step schedule: eta decays as eta_start * (1 - t/steps)**2
        eta = eta_start * (1.0 - t/steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a.fill(1.0/n)
    return a

    n = x.shape[0]

    return conv
