def _hillclimb_refine(a0, n, deadline, start_time, best_a, best_val):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    no_improve = 0
    iteration = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))
        # compute partner array
        idx = np.arange(n)
        partner = np.zeros_like(a)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        partner[mask] = a[k_max - idx[mask]]

        # source: index where partner is max and a>eps
        # If multiple, choose one with largest a*partner
        src_candidates = np.where((a > eps) & (partner > 0))[0]
        if src_candidates.size == 0:
            break
        src_vals = partner[src_candidates] # gradient magnitude
        i_src = src_candidates[np.argmax(src_vals)]

        # target: index where partner is minimal (prefer larger a for stability? not)
        # we can choose among all indices excluding i_src
        tgt_candidates = np.arange(n)
        # Optionally, we can bias towards indices where a is small to maintain simplex
        # but just choose min partner
        target_mask = (tgt_candidates != i_src)
        j_tgt = tgt_candidates[np.argmin(partner + (1 - target_mask)*1e9)]  # ensure not i_src

        # compute second max of b (exclude k_max)
        second_max = np.max(np.delete(b, k_max))

        # compute needed reduction of b_max
        required_delta = (b[k_max] - second_max) / (2 * (partner[i_src] - partner[j_tgt] + eps))
        # limit delta to available mass at src
        delta = min(a[i_src] * 0.5, required_delta * 0.9)  # use fraction

        if delta <= eps:
            # fallback to random small move
            delta = a[i_src] * 0.1 * np.random.rand()
        # apply move
        a[i_src] -= delta
        a[j_tgt] += delta

        # recompute convolution
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[HillRefine {iteration}] new best = {best_val:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1000:
            # random perturbation
            a = best_a.copy()
            a += np.random.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
        iteration += 1

    return best_a, best_val

"""
Multiscale random‑seed search + smooth‑max Adam optimisation + deterministic
hill‑climbing.

The idea is to explore many low‑dimensional initialisations (n≈2000) using a
variety of parametric families (Beta, Dirichlet, sinusoidal, random‑phase).
The best few candidates are interpolated to the full problem size
(n≈8000) and refined with a smooth approximation of the max‑convolution
objective (log‑sum‑exp) using Adam and a λ‑continuation schedule.
Finally a greedy hill‑climber moves mass from indices that dominate the
current maximal convolution entry to indices that contribute little,
with a line‑search that directly targets the largest reduction of the
peak.  The whole pipeline respects the 1000 s budget and returns a list
of non‑negative heights.

The returned sequence should achieve a lower evaluation score than the
previous best (~1.5084) on the provided evaluate_sequence() function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x * x (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1} using the O(n) algorithm."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed generation (small problem size)
# ----------------------------------------------------------------------
def _gen_beta_seeds(n: int):
    """Yield a collection of Beta‑family seeds for a given length."""
    p_vals = np.concatenate(
        [np.linspace(0.1, 5.0, 12), np.geomspace(5.0, 20.0, 5)[1:]]
    )
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s > 0:
                a /= s
                yield a


def _gen_dirichlet_seeds(n: int):
    """Yield Dirichlet draws with a range of concentration parameters."""
    for alpha in [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(2):
            a = np.random.dirichlet(np.full(n, alpha))
            yield a


def _gen_sine_seeds(n: int):
    """Yield sine‑power families a_i ∝ sin(π(i+½)/n)^p."""
    xs = np.arange(n, dtype=np.float64) + 0.5
    base = np.sin(np.pi * xs / n)
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.power(base, p)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        yield a


def _gen_random_phase_seeds(n: int):
    """Create seeds by randomising Fourier phases of a flat spectrum."""
    size = _next_pow2(2 * n - 2)  # length sufficient for linear convolution
    base = np.ones(n) / n
    spec = np.fft.rfft(base, size)
    for _ in range(2):
        phases = np.exp(1j * np.random.uniform(0, 2 * np.pi, spec.shape[0]))
        new_spec = np.abs(spec) * phases
        a = np.fft.irfft(new_spec, size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
            yield a


def _collect_initial_seeds(n_small: int):

    seeds = []

    try:

        old_x = np.linspace(0.0, 1.0, old_len)

        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        seeds.append(a0)
        print("[init] interpolated previous solution (small n)")
    except Exception:

    # b) beta family sweep (coarse)

        seeds.append(a)

        seeds.append(a)

# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + λ‑continuation)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation
        f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
    using Adam and a λ schedule that grows from 0.5 to ~200.
    The variables are projected onto the probability simplex after each step.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.08

    # FFT sizes (pre‑computed)
    size1 = _next_pow2(2 * n - 2)      # for a*a
    size2 = _next_pow2(3 * n - 2)      # for a_rev * w

    # Stagnation detection
    last_improve_iter = 0

    t = 0
    while time.time() < deadline:
        t += 1
        prog = (time.time() - start_time) / (deadline - start_time)
        lam = 0.5 + (200.0 - 0.5) * prog          # λ from 0.5 → 200

        # a*a (full linear convolution)
        fa = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fa * fa, size1)[: 2 * n - 1]
        max_c = conv.max()

        # smooth weights
        exps = np.exp(lam * (conv - max_c))
        w = exps / exps.sum()                     # length 2n‑1, sums to 1

        # gradient = 2 * (a_rev  *  w)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, size2)
        fw = np.fft.rfft(w, size2)
        conv_aw = np.fft.irfft(fa_rev * fw, size2)[: 3 * n - 2]
        grad = 2.0 * conv_aw[n - 1 : n - 1 + n]   # slice of length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * (m_hat / (np.sqrt(v_hat) + eps))
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        cur_val = 2.0 * n * max_c
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            last_improve_iter = t
            print(f"[Smooth t={t:07d}] new best = {best_val:.6f} λ={lam:.1f}")

        # occasional reset if stuck
        if t % 1200 == 0 and (t - last_improve_iter) > 1200:
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = t
            print(f"[Smooth t={t:07d}] stagnation – random perturbation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic hill‑climbing refinement
# ----------------------------------------------------------------------
def _hillclimb_refine(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      max_no_improve: int = 1500) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climber.
    At each step we move mass from the index whose partner value
    a_{k_max‑i} is largest (largest impact on the current maximal
    convolution entry) to the index whose partner is smallest,
    using a line‑search that aims to bring the peak down to the second‑largest.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    no_improve = 0
    iteration = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # partner[i] = a[k_max - i] (0 if out of range)
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        partner = np.zeros_like(a)
        partner[mask] = a[k_max - idx[mask]]

        # source: index with largest partner (and positive mass)
        src_mask = (a > eps) & (partner > 0)
        if not np.any(src_mask):
            break
        i_src = np.argmax(partner * src_mask)  # works because non‑selected entries have partner=0

        # target: index with smallest partner (avoid picking i_src)
        tgt_mask = np.arange(n) != i_src
        j_tgt = np.argmin(partner + (1 - tgt_mask) * 1e9)  # large penalty to stay away

        # second largest convolution value
        b_copy = b.copy()
        b_copy[k_max] = -np.inf
        second_max = b_copy.max()

        # required reduction of the peak
        diff = b[k_max] - second_max
        grad_diff = 2.0 * (partner[i_src] - partner[j_tgt]) + eps
        delta_needed = diff / grad_diff

        # feasible delta (cannot move more mass than we have)
        delta = min(a[i_src] * 0.5, delta_needed * 0.9)
        if delta <= eps:
            # fallback to a small random move
            delta = a[i_src] * 0.1 * np.random.rand()
            if delta <= eps:
                delta = eps

        # apply the transfer (preserves Σa)
        a[i_src] -= delta
        a[j_tgt] += delta

        # recompute convolution (full, still cheap with FFT)
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[HillRefine {iteration:07d}] new best = {best_val:.6f}")
            if best_val <= 1.45:               # early exit if already very good
                break
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # escape a plateau: restore best and add noise
                a = best_a.copy()
                a += np.random.normal(scale=0.015, size=n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                b = _full_convolution(a)
                cur_val = 2.0 * n * b.max()
                no_improve = 0
                print(f"[HillRefine {iteration:07d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that attempts to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a little safety margin
    deadline = start_time + budget_s - 0.5

    # Problem sizes -------------------------------------------------------
    n_small = 2000          # cheap exploration size
    n_big = 8000            # final resolution (the original benchmark size)

    # ------------------------------------------------------------------
    # Phase 1 – cheap exploration on a small problem size
    # ------------------------------------------------------------------
    small_seeds = _collect_initial_seeds(n_small)

    # Evaluate all small seeds, keep the best few
    evaluated = [( _objective(a), a ) for a in small_seeds]
    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    top_small = [a for _, a in evaluated[:top_k]]
    print(f"[Phase1] best small‑n value = {evaluated[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to the full size and smooth‑max optimisation
    # ------------------------------------------------------------------
    # Upscale each top seed by linear interpolation
    big_seeds = []
    new_x = np.linspace(0.0, 1.0, n_big)
    for a_small in top_small:
        old_x = np.linspace(0.0, 1.0, a_small.shape[0])
        a_big = np.interp(new_x, old_x, a_small)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()
        big_seeds.append(a_big)

    # add a few extra diversifiers at full resolution
    big_seeds.append(np.full(n_big, 1.0 / n_big))            # uniform
    for _ in range(2):
        big_seeds.append(np.random.dirichlet(np.ones(n_big)))

    global_best_a = None
    global_best_val = np.inf

    # budget split: ~70 % for smooth optimisation, remainder for hill‑climb
    remaining = deadline - time.time()
    smooth_budget = max(0.0, remaining * 0.70)
    per_seed_budget = smooth_budget / max(1, len(big_seeds))

    for idx, seed_vec in enumerate(big_seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed_budget
        a_opt, val_opt = _smooth_optimize(seed_vec, n_big, local_deadline, start_time)
        print(f"[Smooth #{idx}] best = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
        if global_best_val <= 1.45:          # stop early if already good enough
            break

    # ------------------------------------------------------------------
    # Phase 3 – deterministic hill‑climbing refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and global_best_a is not None:
        global_best_a, global_best_val = _hillclimb_refine(
            global_best_a,
            n_big,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            max_no_improve=1500,
        )
        print(f"[HillRefine] final best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final check (only for logging – not required for the harness)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
