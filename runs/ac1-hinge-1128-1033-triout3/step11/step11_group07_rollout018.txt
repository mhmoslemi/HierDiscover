"""
optimal_step_function.py
------------------------

A self‑contained optimiser for the “step‑function” challenge.

The goal is to minimise

        2·n·max(conv(a, a)) / (sum(a)²)

over non‑negative vectors a (the length n is free to choose).  The
algorithm is deliberately simple, yet it mixes several complementary
search ideas:

* **Power‑law grid** – a dense deterministic sweep over families
  a_i ∝ (i+1)^p·(n−i)^q (including the uniform case p=q=0).  This
  quickly uncovers useful U‑shaped candidates that lower the peak
  of the self‑convolution.

* **Dirichlet seeds** – a handful of random Dirichlet draws with
  various concentration parameters to inject diversity.

* **Adam on a smooth max** – we minimise a log‑sum‑exp surrogate for
  max(conv) in the log‑softmax (a.k.a. simplex) domain.  The
  temperature ε is annealed from 0.5 to 1e‑12 while the learning rate
  decays from 0.5 to 0.005.

* **Projected sub‑gradient polishing** – once the smooth phase has
  stalled we switch to a true sub‑gradient method for the exact
  objective, with a linear step‑size schedule.

* **Uniform blending** – occasional convex combinations with the
  uniform distribution help to smooth isolated spikes.

The whole pipeline respects the 1000 s budget, falls back to a uniform
vector if nothing else succeeds, and finally returns the best sequence
found as a plain list of Python floats (ready for ``evaluate_sequence``).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a (the factor 2·n is already incorporated).
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick for numerical stability
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with the reversed weight vector
    conv_ab = _conv_fft(a, p[::-1])              # length 2n‑1
    # the gradient w.r.t. a is 4·n times the relevant slice (mirrored)
    grad_a = 4.0 * n * conv_ab[n - 1:2 * n - 1][::-1]
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam in the log‑softmax (simplex) space."""
    # work in log‑softmax space so the simplex constraints are automatic
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.003
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 2500

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule (linear in log‑time)
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # grad already contains 2·n

        # chain rule through the soft‑max (log‑softmax) parametrisation
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to help escape plateaus
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.04, size=n)

        # restart if we have been idle for too long
        if idle > MAX_IDLE:
            print("[Adam] restart (idle)")
            w = np.random.randn(n) * 0.4
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution objective)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        ks = np.where(b == max_b)[0]

        # analytic sub‑gradient: sum over all active lags
        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple linear step‑size decay
        prog = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - prog) + 1e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional tiny jitter (helps escape shallow plateaus)
        if it % 600 == 0 and np.random.rand() < 0.025:
            a = np.maximum(a + np.random.normal(scale=0.03, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Convex blending with the uniform distribution (post‑processing)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for several t."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 23):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta family: a_i ∝ (i+1)^{p-1}(n-i)^{q-1} (p,q>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _powerlaw_candidate(n: int, p: float, q: float) -> np.ndarray:
    """
    Power‑law family: a_i ∝ (i+1)^p·(n-i)^q.
    Allows negative exponents to concentrate mass at the edges.

    i = np.arange(1, n + 1, dtype=np.float64)
    left = np.power(i, p)
    right = np.power(n - i + 1, q)
    a = left * right
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a
