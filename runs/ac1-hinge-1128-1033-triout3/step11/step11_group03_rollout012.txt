def full_conv(a):
    n = a.shape[0]
    out_len = 2*n - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

    a = a_init.copy()
    conv = conv_init.copy()
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 1e-12:
        return a, np.inf
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max / (sum_a ** 2)
    best_a = a.copy()
    best_val = cur_val

    # temperature schedule parameters
    T_start = cur_val * 0.02  # maybe 2% of value
    T_end = cur_val * 1e-5

    iter_count = 0

        if now >= deadline:

        r = np.random.rand() * sum_a

"""Simulated‑annealing optimizer for the step‑function attack.

The algorithm starts from the best known construction (or a simple
parametric shape), keeps the total sum of the coefficients constant and
iteratively transfers a small amount of mass from one index to another.
A temperature schedule allows occasional uphill moves, helping the
search escape local minima.  Accepted moves update the full auto‑
convolution using an O(n) incremental formula, avoiding costly FFT
calls.  The best sequence found before the time budget expires is
returned.
"""

# ------------------------------------------------------------
# Convolution utilities (linear full convolution via FFT)
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ------------------------------------------------------------
# Objective (exact, scale‑invariant)
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum a)^2."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:                      # safeguard, matches evaluate_sequence
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ------------------------------------------------------------
# Helpers to generate a starting point
# ------------------------------------------------------------
def _parametric_candidate(n: int) -> np.ndarray:
    """Return a decent hand‑crafted shape (Gaussian/Laplace/power)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Gaussian family (a few sigma values)
    for sigma_factor in (0.18, 0.22, 0.26, 0.30):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        if a.sum() > 0:
            a /= a.sum()
            return a

    # Laplace family
    for scale_factor in (0.12, 0.16, 0.20):
        scale = scale_factor * n
        a = np.exp(-np.abs(i - mid) / scale)
        if a.sum() > 0:
            a /= a.sum()
            return a

    # Power‑law / beta‑like family
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = (i + 1) ** (p - 1.0) * (n - i) ** (p - 1.0)
        if a.sum() > 0:
            a /= a.sum()
            return a

    # fallback: uniform (never used because it is sub‑optimal)
    return np.full(n, 1.0 / n)


def _interpolate(previous: np.ndarray, n: int) -> np.ndarray:
    """Linearly interpolate a previous sequence to a new length."""
    old_len = previous.shape[0]
    new_x = np.linspace(0.0, 1.0, n)
    old_x = np.linspace(0.0, 1.0, old_len)
    a = np.interp(new_x, old_x, previous)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ------------------------------------------------------------
# Simulated annealing core
# ------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated annealing.
    Keeps total sum constant and updates the full convolution
    incrementally (O(n) per iteration).
    Returns the best sequence found and its objective.
    """
    a = a0.copy()
    n = a.shape[0]
    total = a.sum()
    if total < 1e-12:
        a = np.full(n, 1.0 / n)
        total = a.sum()

    conv = _full_convolution(a)
    cur_max = conv.max()
    cur_val = float(2.0 * n * cur_max / (total ** 2))

    best_a = a.copy()
    best_val = cur_val

    # temperature schedule (linear from T_start to T_end)
    T_start = cur_val * 0.02          # a few percent of the current value
    T_end = cur_val * 1e-5

    no_improve = 0
    iter_cnt = 0
    IMPROVE_LIMIT = 20000

        now = time.time()
        if now >= deadline:
            break

        # linearly interpolated temperature
        frac = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        T = (1 - frac) * T_start + frac * T_end

        # ---- pick source proportional to its mass ----
        cum = np.cumsum(a)
        r = np.random.rand() * total
        src = int(np.searchsorted(cum, r))

        if a[src] <= 1e-12:
            continue

        # pick a different target uniformly at random
        tgt = np.random.randint(0, n)
        while tgt == src:
            tgt = np.random.randint(0, n)

        # magnitude of the transfer (up to 50 % of source)
        delta = a[src] * np.random.rand() * 0.5

            continue

        conv_cand = conv.copy()
        # linear part: subtract contribution of src, add contribution of tgt
        conv_cand[src:src + n] -= 2.0 * delta * a
        conv_cand[tgt:tgt + n] += 2.0 * delta * a
        # quadratic correction for the moved mass itself
        conv_cand[src + src] += delta * delta
        conv_cand[tgt + tgt] += delta * delta
        conv_cand[src + tgt] -= 2.0 * delta * delta

        new_max = conv_cand.max()
        new_val = float(2.0 * n * new_max / (total ** 2))

        # ---- Metropolis acceptance ----
        accept = False
        if new_val < cur_val:
            accept = True
        else:

            if T > 0.0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            # commit the move
            a[src] -= delta
            a[tgt] += delta
            conv = conv_cand
            cur_val = new_val
            cur_max = new_max
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SA] iter {iter_cnt:7d} val={best_val:.6f}")
                if best_val <= 1.503:
                    # we have met the target – can stop early
                    break
        else:
            no_improve += 1

        # ---- occasional jitter to leave deep plateaus ----
        if no_improve > IMPROVE_LIMIT:
            # restart from the best seen so far with a tiny random perturbation
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            # renormalise to keep the total sum exactly the same
            s = a.sum()
            if s > 0:
                a *= total / s
            conv = _full_convolution(a)
            cur_max = conv.max()
            cur_val = float(2.0 * n * cur_max / (total ** 2))
            no_improve = 0

        iter_cnt += 1

    return best_a, best_val
