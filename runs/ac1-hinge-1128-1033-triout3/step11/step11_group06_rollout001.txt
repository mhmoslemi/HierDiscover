"""
Propose a candidate step‑function (a probability vector) that minimises
``evaluate_sequence``.  The approach combines

1.  A cheap global scan of the 2‑parameter Beta family (and a few
    Dirichlet draws) to obtain a good initial shape.
2.  Several Adam‑optimisation runs on the smooth log‑sum‑exp
    approximation of the max‑convolution.
    The smoothing parameter β is increased exponentially so that the
    optimiser eventually works on the true nonsmooth objective.
3.  Enforcing symmetry a ↔ a[::-1] after every projection onto the
    simplex – the optimal solution is known to be symmetric.
4.  A short polishing phase with a very large β.

The best sequence found before the time budget expires is returned as a
list of floats.  All helper routines are written at module level (no
closures) and respect the 1000‑second wall‑clock limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex Σx=1, x≥0.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the parameters create an all‑zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _enforce_symmetry(v: np.ndarray) -> np.ndarray:
    """Replace v by (v + v[::-1])/2 – keeps the sum unchanged."""
    return (v + v[::-1]) * 0.5


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient of φ_β(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i).

    Returns (grad, max_conv) where `grad` is the Euclidean gradient
    (same shape as a) and `max_conv` = max(a⋆a).
    """
    n = a.shape[0]

    # ---- full convolution a⋆a (length 2n‑1) -------------------------
    out_len = 2 * n - 1
    size_conv = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[:out_len]

    max_conv = conv.max()

    # ---- soft‑max weights -----------------------------------------
    # shift by max to avoid overflow
    exp_arg = beta * (conv - max_conv)          # ≤ 0
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum                           # Σ_i w_i = 1

    # ---- gradient = 2·(a ⋆_cross w_norm) -------------------------
    # Reversing w_norm gives the kernel for cross‑correlation.
    rev_w = w_norm[::-1]
    out_len_corr = n + out_len - 1                # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1                          # lag 0 in the full correlation
    idx = offset - np.arange(n)                   # indices for j = 0…n‑1
    grad = 2.0 * corr_full[idx]

    return grad, max_conv

            step = step0 * np.sqrt(1 - beta2 ** (polish_iter + 1)) / (1 - beta1 ** (polish_iter + 1))
            grad, cur_max = _gradient_smooth(a, beta)   # β is already huge → true subgradient
            cur_val = 2.0 * n * cur_max
            if cur_val < best_score:
                best_score, best_seq = cur_val, a.copy()
                print(f"[polish] new best = {best_score:.6f} (iter {polish_iter})")
                if best_score <= 1.503:
                    break
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            a_candidate = a - step * m / (np.sqrt(v) + eps)
            a_candidate = _project_simplex(a_candidate)
            a_candidate = _enforce_symmetry(a_candidate)
            a = a_candidate

    # ------------------------------------------------------------------
    # 4️⃣  Final evaluation (the only place we call the provided function)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # return plain Python list as required by the outer harness
    return best_seq.tolist()
```
