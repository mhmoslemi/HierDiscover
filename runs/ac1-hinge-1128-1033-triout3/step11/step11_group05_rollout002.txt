Hybrid search for a non‑negative step‑function that minimizes
   evaluate_sequence(sequence)

Algorithm
---------
1️⃣  Broad exploration:
    •   Several analytic windows (Beta, exponential, Gaussian, power‑sine,
        triangular, Hann, Hamming, Blackman, Kaiser) are sampled with
        random parameters.

    •   Each candidate is normalised to Σa = 1 and the exact objective
        2·n·max(conv(a,a)) is evaluated (fast via FFT).  The best few

    •   Use a smoothed version of the max‑convolution:
            Lα(a) = 2·n·log‑sum‑exp(α·conv(a,a)) / α

# Linear‑algebra utilities (FFT based convolutions)

def _linear_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two 1‑D arrays using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 2).bit_length())      # next power‑of‑two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: n + m - 1]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution of a with itself (length 2·len(a)‑1)."""
    return _linear_convolution(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())

def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    tot = a.sum()
    if tot <= 0.0:
        return None
    a /= tot
    return a


def _exp_shape(c: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a


def _gaussian_shape(sigma: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a


def _power_sine_shape(p: float, n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    a /= a.sum()
    return a


def _triangular_shape(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a


def _hann_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))

    return a


def _hamming_shape(n: int) -> np.ndarray:

    return a


def _blackman_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)

    return a


def _kaiser_shape(beta: float, n: int) -> np.ndarray:

    mid = (n - 1) / 2.0
    arg = (i - mid) / mid                     # in [‑1,1]
    w = special.i0(beta * np.sqrt(1.0 - arg ** 2))
    a = w / w.sum()
    return a

def _grad_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k * conv_k(a,a)  w.r.t. a.
    w has length 2·n‑1 and is assumed to sum to 1.
    """
    n = a.shape[0]
    # reverse w
    w_rev = w[::-1]

    slice_ = full[n - 1 : 2 * n - 1]
    grad = 2.0 * slice_[::-1]                     # reverse to match formula
    return grad

    alpha = 1.0               # smoothing parameter (will increase)
    alpha_max = 4000.0
    lr_base = 0.15            # base learning rate for the exponentiated step
    no_improve = 0
    max_no_improve = 5000

    while time.time() < deadline:
        t_iter += 1

        # --- current convolution (full) ---
        conv = _full_convolution(a)               # length 2n‑1
        cur_val = 2.0 * n * conv.max()

        # record improvement
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[optim] iter {t_iter:5d} new best {best_val:.6f}")
        else:
            no_improve += 1

        # --- smoothed max (log‑sum‑exp) ---
        # increase smoothing gradually
        alpha = min(alpha_max, alpha * 1.012)
        # stable softmax
        shifted = conv - conv.max()
        exp_scaled = np.exp(alpha * shifted)
        w = exp_scaled / exp_scaled.sum()        # weights sum to 1

        # --- gradient of the smoothed objective ---
        grad = _grad_weighted(a, w)

        # --- Adam update (exponentiated) ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = lr_base * m_hat / (np.sqrt(v_hat) + eps)

        # multiplicative update keeps positivity
        a = a * np.exp(-step)
        a_sum = a.sum()
        if a_sum == 0.0 or not np.isfinite(a_sum):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.001 or no_improve >= max_no_improve:
            jitter = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[optim] jitter applied at iter {t_iter}")

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
