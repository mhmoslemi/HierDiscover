def generate_seeds(n, n_seed=2000):
    seeds = []

    # 1) previous solution interpolation
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        if old.ndim == 1 and old.size > 0:
            new_x = np.linspace(0, 1, n)
            old_x = np.linspace(0, 1, old.size)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0)
            a0 /= a0.sum()
            seeds.append(a0)
    except Exception:
        pass

    # 2) Beta family (coarse grid)
    i = np.arange(1, n_seed+1, dtype=float)
    p_vals = np.concatenate([np.geomspace(0.05, 0.5, 6), np.linspace(0.5, 5, 10), np.geomspace(5, 30, 5)])
    q_vals = p_vals.copy()
    for p in p_vals:
        ip = i**(p-1)
        for q in q_vals:
            a_seed = ip * ((n_seed - i + 1) ** (q-1))
            s = a_seed.sum()
            if s == 0:
                continue
            a_seed = a_seed / s
            if n != n_seed:
                new_x = np.linspace(0, 1, n)
                old_x = np.linspace(0, 1, n_seed)
                a_up = np.interp(new_x, old_x, a_seed)
                a_up = np.maximum(a_up, 0)
                a_up /= a_up.sum()
                seeds.append(a_up)
            else:
                seeds.append(a_seed)

    # 3) Gaussian seeds with varying sigma
    center = (n+1)/2
    for sigma_factor in [0.1, 0.15, 0.2, 0.25]:
        sigma = sigma_factor * n
        x = np.arange(1, n+1)
        a_seed = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        a_seed /= a_seed.sum()
        seeds.append(a_seed)

    # 4) Cosine window (Hann)
    x = np.arange(n)
    a_seed = 0.5 - 0.5 * np.cos(2 * np.pi * x / (n - 1))
    a_seed = np.maximum(a_seed, 0)
    a_seed /= a_seed.sum()
    seeds.append(a_seed)

    # 5) Raised cosine (Tukey with alpha=0.5)
    alpha = 0.5
    w = np.ones(n)
    t = np.arange(0, n) / (n - 1)
    mask = t < alpha/2
    w[mask] = 0.5 * (1 + np.cos(np.pi * (2*t[mask]/alpha - 1)))
    mask = t > 1 - alpha/2
    w[mask] = 0.5 * (1 + np.cos(np.pi * (2*t[mask]/alpha - 2/alpha + 1)))
    a_seed = w
    a_seed /= a_seed.sum()
    seeds.append(a_seed)

    # 6) Triangular (symmetric)
    a_seed = np.minimum(np.arange(1, n+1), np.arange(n, 0, -1))
    a_seed = a_seed.astype(float)
    a_seed /= a_seed.sum()
    seeds.append(a_seed)

    # 7) Random Dirichlet draws with concentration 0.5, 1, 2, 5
    alphas = [0.5, 1.0, 2.0, 5.0]
    for a in alphas:
        seeds.append(np.random.dirichlet(np.full(n, a)))
    # 8) Random Dirichlet with random concentration
    for _ in range(3):
        concentration = 10**np.random.uniform(-2, 1) # between 0.01 and 10
        seeds.append(np.random.dirichlet(np.full(n, concentration)))

    return seeds

def _smooth_optimize(a0, n, deadline, start_time, lambda_max=200):
    a = np.maximum(a0, 0.0)
    a /= a.sum()
    best_a = a.copy()
    best_val = _objective(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha_base = 0.07
    t = 0
    size1 = 1 << ((2*n - 2).bit_length())
    size2 = 1 << ((3*n - 3).bit_length())

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1
        prog = (now - start_time) / (deadline - start_time)
        lam = 0.5 + (lambda_max - 0.5) * prog**0.8  # progressive growth

        # compute convolution
        fa = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fa * fa, size1)[:2*n-1]
        max_c = conv.max()

        # compute smooth weights
        exps = np.exp(lam * (conv - max_c))
        w = exps / exps.sum()

        # gradient
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, size2)
        fw = np.fft.rfft(w, size2)
        conv_aw = np.fft.irfft(fa_rev * fw, size2)[:3*n-2]
        grad = 2.0 * conv_aw[n-1:n-1+n]

        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = alpha_base * (1 - prog) + 0.01 * prog
        step = lr * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        cur_val = 2.0 * n * max_c
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth] iter {t:6d} best={best_val:.6f} λ={lam:.2f}")
            if best_val <= 1.503:
                break

        if t % 800 == 0:
            if cur_val >= best_val * 0.9999:
                # perturb
                noise = np.random.normal(scale=0.015, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print("[Smooth] stagnation perturb")
    return best_a, best_val

assistantfinal**Solution Overview**

The algorithm searches for a non‑negative step‑function `a = (a0,…,a_{n‑1})` that minimises  

\[
\mathrm{score}(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2}.
\]

Because the score is invariant under scaling we may restrict ourselves to the
probability simplex `∑a = 1`.  
The optimisation is performed in three stages

1. **Diversified seeding** – a collection of promising shapes is built:
   * interpolation of a previous solution (if any),
   * a few best members of a dense *Beta‑family* sweep,
   * analytic windows (Gaussian, Hann, Tukey, triangular),
   * several Dirichlet draws with different concentration parameters.

2. **Smooth‑max gradient descent** – the non‑smooth `max` is replaced by a
   log‑sum‑exp approximation  
   \[
   f_\lambda(a)=\frac{1}{\lambda}\log\!\sum_k e^{\lambda\,(a*a)_k},
   \]
   whose gradient is a convolution and can be evaluated with FFTs.  
   Adam with a simplex projection is applied while the smoothing parameter
   `λ` is increased from `0.5` to `≈400`.  Each seed receives a fair share of
   the total budget.

3. **Intensive hill‑climbing** – a simulated‑annealing style local search
   moves mass from indices that dominate the current maximal convolution entry
   to indices that contribute little.  The best vector from step 2 serves as
   the starting point and the remaining time budget is dedicated to this
   phase.  Finally a short high‑`λ` smooth‑max refinement is performed.

All steps respect the 1000 s time limit; the best vector found is returned
as a plain list of floats.

"""
Robust optimiser for the step‑function evaluation problem.

The code builds a diverse set of seeds (previous solution, Beta‑family,
Gaussian / cosine / Tukey windows, random Dirichlet draws), refines each of
them with a log‑sum‑exp smooth‑max Adam optimiser (λ grows to ≈400) while
projecting onto the probability simplex, and finally applies an intensive
simulated‑annealing hill‑climber.  The result is a non‑negative sequence of
floats that minimises `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Seed generators
# ----------------------------------------------------------------------
def _interpolate_previous(n: int):
    """If a previous solution `height_sequence_1` exists, interpolate it."""
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.ndim != 1 or old.size == 0:
            raise ValueError
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.size)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        print("[seed] interpolated previous solution")
        return a
    except Exception:
        return None


def _beta_family_topk(n_target: int,
                      n_src: int = 2000,
                      top_k: int = 5) -> list[np.ndarray]:
    """Sweep the Beta family on a cheap grid and return the `top_k` best seeds
       (interpolated to the target length)."""
    i = np.arange(1, n_src + 1, dtype=float)

    # a modest but fairly rich grid
    p_vals = np.concatenate([
        np.geomspace(0.05, 0.5, 5),
        np.linspace(0.5, 5.0, 9),
        np.geomspace(5.0, 30.0, 4)[1:],
    ])
    q_vals = p_vals.copy()

    seeds_src = []
    scores = []

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n_src - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            # cheap objective (using n_src)
            val = _objective(a)
            seeds_src.append(a)
            scores.append(val)

    # keep the best `top_k`
    if not scores:
        return []
    idx = np.argsort(scores)[:top_k]
    selected = [seeds_src[i] for i in idx]

    # interpolate to the final length
    new_x = np.linspace(0.0, 1.0, n_target)
    src_x = np.linspace(0.0, 1.0, n_src)
    upscaled = []
    for a in selected:
        a_up = np.interp(new_x, src_x, a)
        a_up = np.maximum(a_up, 0.0)
        a_up /= a_up.sum()
        upscaled.append(a_up)
    print(f"[seed] {len(upscaled)} Beta‑family seeds (top‑{top_k})")
    return upscaled


def _gaussian_seeds(n: int) -> list[np.ndarray]:
    """Symmetric Gaussian windows with several bandwidths."""
    seeds = []
    centre = (n + 1) / 2.0
    x = np.arange(1, n + 1, dtype=float)
    for sigma_rel in (0.08, 0.12, 0.18, 0.25):
        sigma = sigma_rel * n
        a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        a /= a.sum()
        seeds.append(a)
    print("[seed] Gaussian windows")
    return seeds


def _cosine_seeds(n: int) -> list[np.ndarray]:
    """Hann (cosine) and Tukey windows."""
    seeds = []
    # Hann
    x = np.arange(n, dtype=float)
    hann = 0.5 - 0.5 * np.cos(2 * np.pi * x / (n - 1))
    hann = np.maximum(hann, 0)
    hann /= hann.sum()
    seeds.append(hann)

    # Tukey (alpha = 0.5)
    alpha = 0.5
    t = np.arange(n, dtype=float) / (n - 1)
    w = np.ones_like(t)
    mask = t < alpha / 2
    w[mask] = 0.5 * (1 + np.cos(np.pi * (2 * t[mask] / alpha - 1)))
    mask = t > 1 - alpha / 2
    w[mask] = 0.5 * (1 + np.cos(np.pi * (2 * t[mask] / alpha - 2 / alpha + 1)))
    w = np.maximum(w, 0)
    w /= w.sum()
    seeds.append(w)

    # Triangular (symmetric)
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(float)
    tri /= tri.sum()
    seeds.append(tri)

    print("[seed] cosine / Tukey / triangular windows")
    return seeds


def _dirichlet_seeds(n: int) -> list[np.ndarray]:
    """Various Dirichlet draws (different concentration parameters)."""
    seeds = []
    concentrations = [0.3, 0.5, 1.0, 2.0, 5.0]
    for c in concentrations:
        seeds.append(np.random.dirichlet(np.full(n, c)))
    # a few random concentrations
    for _ in range(3):
        c = 10 ** np.random.uniform(-2, 1)          # between 0.01 and 10
        seeds.append(np.random.dirichlet(np.full(n, c)))
    print("[seed] Dirichlet draws")
    return seeds


def _gather_seeds(n: int) -> list[np.ndarray]:
    """Collect a diverse pool of initial vectors."""
    seeds = []

    # 1) previous solution (if any)
    prev = _interpolate_previous(n)
    if prev is not None:
        seeds.append(prev)

    # 2) best Beta‑family members
    seeds.extend(_beta_family_topk(n, n_src=2000, top_k=5))

    # 3) analytical windows
    seeds.extend(_gaussian_seeds(n))
    seeds.extend(_cosine_seeds(n))

    # 4) random Dirichlet draws
    seeds.extend(_dirichlet_seeds(n))

    # 5) a few pure random draws as a safety net
    for _ in range(3):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # make sure everything is on the simplex
    for i, a in enumerate(seeds):
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.random.dirichlet(np.ones(n))
            s = 1.0
        seeds[i] = a / s

    print(f"[seed] gathered {len(seeds)} initial candidates")
    return seeds


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + λ‑continuation)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     global_start: float,
                     lambda_max: float = 400.0) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective
        f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
    using Adam with simplex projection.
    λ grows from 0.5 to `lambda_max` as the optimisation proceeds.
    """
    a = np.maximum(a0, 0.0)
    a /= a.sum()                     # enforce Σa = 1

    best_a = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha_base = 0.07                # learning‑rate at the start
    t = 0

    # FFT sizes (pre‑computed for the whole run)
    size1 = 1 << ((2 * n - 2).bit_length())       # for a*a
    size2 = 1 << ((3 * n - 3).bit_length())       # for a_rev * w

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        # λ‑continuation (smoothly increasing)
        prog = (now - global_start) / (deadline - global_start)
        lam = 0.5 + (lambda_max - 0.5) * (prog ** 0.8)   # a slightly convex schedule

        # ---------- a*a ----------
        fa = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fa * fa, size1)[:2 * n - 1]
        max_c = conv.max()

        # ---------- smooth weights ----------
        # w_k ∝ exp(λ·(conv_k – max_c))  – numerically stable
        exps = np.exp(lam * (conv - max_c))
        w = exps / exps.sum()            # length 2·n‑1, sum to 1

        # ---------- gradient ----------
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, size2)
        fw = np.fft.rfft(w, size2)
        conv_aw = np.fft.irfft(fa_rev * fw, size2)[:3 * n - 2]
        grad = 2.0 * conv_aw[n - 1 : n - 1 + n]

        # ---------- Adam update ----------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # slowly decay the learning‑rate as λ grows
        lr = alpha_base * (1 - prog) + 0.015 * prog
        step = lr * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:                     # very unlikely, but guard against it
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        # ---------- keep the best ----------
        cur_val = 2.0 * n * max_c          # exact objective for the old a
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth] it {t:6d}  best={best_val:.6f}  λ={lam:.1f}")
            if best_val <= 1.503:        # early exit if we already beat the target
                break

        # ---------- occasional random perturbation ----------
        if t % 800 == 0:
            if cur_val >= best_val * 0.9999:
                noise = np.random.normal(scale=0.015, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print("[Smooth] stagnation – random perturb")

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core, slight tweaks)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that contribute
    little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.70, 0.001   # a slightly more aggressive schedule
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        # indices that feed the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # pick a target that contributes little
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # update the convolution efficiently
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt]     += delta * delta
        b_new[2 * i_src]     += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] it {iteration:7d}  new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # escape from a plateau
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] it {iteration:7d}  stagnation – random perturb")
        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # Problem size – a few ten‑thousands entries give a good trade‑off.
    n = 12000

    # ------------------------------------------------------------------
    # 0️⃣ Gather a diverse set of initial seeds
    # ------------------------------------------------------------------
    seeds = _gather_seeds(n)

    # ------------------------------------------------------------------
    # 1️⃣ Smooth‑max optimisation (roughly half of the total budget)
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_val = np.inf
    remaining = deadline - time.time()
    time_for_smooth = remaining * 0.55
    per_seed = time_for_smooth / max(1, len(seeds))

    print(f"[main] allocated {time_for_smooth:.1f}s for smooth optimisation "
          f"({len(seeds)} seeds, ≈{per_seed:.1f}s each)")

    for i, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed
        a_opt, val_opt = _smooth_optimize(seed_vec, n, local_deadline,
                                          start_time, lambda_max=400.0)
        print(f"[seed {i:02d}] smooth best = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Intensive hill‑climbing on the remaining time slice
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and global_best_a is not None:
        print(f"[main] starting hill‑climb with {remaining:.1f}s left")
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final high‑λ smooth refinement (if any time is left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and global_best_a is not None:
        print(f"[main] final smooth refinement with {remaining:.1f}s left")
        fin_deadline = time.time() + remaining
        global_best_a, global_best_val = _smooth_optimize(
            global_best_a,
            n,
            fin_deadline,
            start_time,
            lambda_max=800.0,          # push λ even higher for a near‑exact max
        )
        print(f"[final smooth] best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Secure final evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
