"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence().  The algorithm consists of three stages:

1️⃣  Diverse initialisation  –  previous solution (if any),
    a U‑shaped “arcsine / Beta(0.5,0.5)” seed,
    a dense sweep of the Beta(p,q) family (allowing p,q<1 for
    strongly U‑shaped curves) and a few random Dirichlet draws.
2️⃣  Smooth‑max optimisation – minimise a smooth approximation of
    the max‑convolution using the log‑sum‑exp surrogate and Adam.
    The smoothing parameter λ is increased from 0.5 to 200 while the
    iterate is repeatedly projected onto the probability simplex.
3️⃣  Intensive hill‑climbing – a simulated‑annealing style local
    search that transfers a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.

All stages respect the 1000 s budget, checking the wall‑clock
regularly.  The best vector found is returned as a plain list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_rev_weight(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·(rev(a) * w) restricted to n entries.
    rev(a) is a reversed copy of a, w has length 2·n‑1.
    The result is the gradient of the smooth‑max surrogate.
    """
    n = a.shape[0]
    # length of rev(a) + w - 1 = n + (2n-1) - 1 = 3n-2
    out_len = 3 * n - 2
    size = 1 << ((out_len - 1).bit_length())
    fa_rev = np.fft.rfft(a[::-1], size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa_rev * fw, size)[:out_len]
    # take entries n-1 … n-1+n-1 (length n)
    return 2.0 * conv[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x≥0, Σx = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Beta‑family candidate generator (allows p,q in (0, 12])
# ----------------------------------------------------------------------
def _beta_candidates(n: int):
    """Yield several Beta(p,q) shaped vectors (including very U‑shaped)."""
    i = np.arange(1, n + 1, dtype=np.float64)  # start at 1 to avoid 0⁰
    # a modest grid that explores both U‑shaped (p,q<1) and bell‑shaped (p,q>1)
    p_vals = np.concatenate([
        np.geomspace(0.01, 0.4, 6),          # strong U‑shapes
        np.linspace(0.5, 3.0, 10),           # moderate shapes
        np.geomspace(3.0, 12.0, 5)[1:],     # heavy tails
    ])
    q_vals = p_vals.copy()
    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            yield a


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + λ continuation)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv) via a log‑sum‑exp surrogate.
    Adam updates are projected onto the simplex after each step.
    λ grows from 0.5 to 200.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.12

    step_counter = 0

    # pre‑compute FFT sizes
    size_conv = 1 << ((2 * n - 2).bit_length())          # for a*a
    size_grad = 1 << ((3 * n - 3).bit_length())          # for rev(a)*w

    while True:
        now = time.time()
        if now >= deadline:
            break
        step_counter += 1

        # λ schedule (smooth -> sharper)
        prog = (now - start_time) / (deadline - start_time)
        lam = 0.5 + (200.0 - 0.5) * prog

        # a*a
        Fa = np.fft.rfft(a, size_conv)
        conv = np.fft.irfft(Fa * Fa, size_conv)[:2 * n - 1]
        max_c = conv.max()

        # smooth weights (softmax)
        exps = np.exp(lam * (conv - max_c))
        w = exps / exps.sum()                     # length 2·n‑1, sums to 1

        # gradient of smooth surrogate
        grad = _conv_rev_weight(a, w)

        # Adam step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** step_counter)
        v_hat = v / (1 - beta2 ** step_counter)
        step = base_lr * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        # keep best (using the *old* max_c, i.e. exact max before the move)
        cur_val = 2.0 * n * max_c
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth {step_counter:6d}] new best = {best_val:.6f} λ={lam:.2f}")

        # occasional random perturbation to escape plateaus
        if step_counter % 1000 == 0:
            if cur_val >= best_val * 0.9999:
                noise = np.random.normal(scale=0.01, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[Smooth {step_counter:6d}] stagnation – perturb")

    return best_a, best_val

# Simulated‑annealing hill climber (same spirit as the reference)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      best_val_target: float = 1.50) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.  Temperature and allowed transfer fraction decay
    linearly with time.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    # annealing schedules
    max_frac_start, max_frac_end = 0.7, 0.001
    temp_start, temp_end = 0.05, 5e-9
    eps = 1e-15
    no_improve = 0
    IMPROVE_LIMIT = 120_000
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # locate the index where the convolution is maximal
        k_max = int(np.argmax(b))

        # the set of i that actually appear in that convolution term:
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose source index (where mass is “dangerous”)
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # compute a convenient target weighting (favor indices that appear rarely)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # update convolution efficiently
        b_new = b.copy()
        # every entry where a[i_src] participates receives +/- 2*delta*a[?]
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        # adjust the pure quadratic terms for the moved mass
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val + 1e-12 < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= best_val_target:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # escape stagnation
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturb")

        it += 1

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (a step‑function) that attempts
    to minimise ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # leave a tiny safety margin

    # ----- problem size -------------------------------------------------
    # A larger n gives a finer discretisation and a lower theoretical bound.
    n = 18000          # fits easily into RAM & FFT budget

    # ----- collect diverse seeds ----------------------------------------
    seeds = []

    # a) previous solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        seeds.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # b) arcsine / Beta(0.5,0.5) shape – a classic U‑shaped candidate
    i = np.arange(1, n + 1, dtype=np.float64)
    arcsine = (i ** (-0.5)) * ((n - i + 1) ** (-0.5))
    arcsine = np.maximum(arcsine, 0.0)
    arcsine /= arcsine.sum()
    seeds.append(arcsine)
    print("[init] arcsine (Beta 0.5,0.5) seed")

    # c) sweep over Beta family (including strong U‑shapes)
    best_beta = None
    best_beta_val = np.inf
    for a in _beta_candidates(n):
        val = _objective(a)
        if val < best_beta_val:
            best_beta_val = val
            best_beta = a
    if best_beta is not None:
        seeds.append(best_beta)
        print(f"[init] best beta‑family seed value = {best_beta_val:.6f}")

    # d) a few random Dirichlet draws
    for _ in range(2):
        seeds.append(np.random.dirichlet(np.ones(n)))
    print(f"[init] total seeds collected: {len(seeds)}")

    # ----- Phase 1: smooth‑max optimisation -------------------------------
    time_for_smooth = (deadline - time.time()) * 0.55
    if time_for_smooth < 1.0:
        time_for_smooth = 0.0
    per_seed = time_for_smooth / max(1, len(seeds))

    global_best_a = None
    global_best_val = np.inf

    for sid, a_seed in enumerate(seeds):
        now = time.time()
        if now >= deadline:
            break
        local_deadline = now + per_seed
        a_opt, val_opt = _smooth_optimize(a_seed, local_deadline, start_time)
        print(f"[seed {sid}] smooth best = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
        # early exit if we already beat the target
        if global_best_val <= 1.503:
            break

    # ----- Phase 2: hill‑climbing ----------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and global_best_a is not None:
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            deadline=deadline,
            start_time=start_time,
            best_val_target=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ----- Final evaluation (for logging) --------------------------------
    try:
        reported = evaluate_sequence(global_best_a.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return plain Python list
    return global_best_a.tolist()
