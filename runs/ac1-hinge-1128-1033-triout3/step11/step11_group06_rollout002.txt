def _gradient_and_conv(a, beta):
    # compute conv using FFT
    n = a.shape[0]
    out_len = 2*n-1
    size = 1 << (out_len-1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    max_conv = conv.max()
    if beta > 20:  # high beta, approximate subgradient
        idx_max = np.argmax(conv)
        idx = idx_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2 * a[idx[mask]]
        return grad, max_conv
    # else compute softmax gradient as earlier
    exp_arg = beta * (conv - max_conv)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum
    rev_w = w_norm[::-1]
    out_len_corr = n + out_len - 1
    size_corr = 1 << (out_len_corr-1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]
    offset = out_len - 1
    idx2 = offset - np.arange(n)
    grad = 2.0 * corr_full[idx2]
    return grad, max_conv
```

uses a smoothed max‑convolution (log‑sum‑exp) at low temperature and
switches to the exact sub‑gradient of the max at high temperature.
Multiple random restarts are performed until the time budget expires.
The best sequence found (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become zero – fallback to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the parameters produce an all‑zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _sin_power_shape(power: float, n: int) -> np.ndarray | None:
    """a_i ∝ sin(iπ/(n+1))**power, normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1)) ** power
    if a.sum() == 0:
        return None
    a /= a.sum()
    return a


def _exp_decay_shape(lam: float, n: int) -> np.ndarray | None:
    """a_i ∝ exp(-lam * i), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    if a.sum() == 0:
        return None
    a /= a.sum()
    return a


def _mixture_gaussian(n: int, rng: np.random.Generator,
                      ncomp: int = 2) -> np.ndarray:
    """
    Sample a mixture of `ncomp` (2‑3) Gaussian bumps, normalised to sum 1.
    """
    a = np.zeros(n, dtype=np.float64)
    weights = rng.random(ncomp)
    weights /= weights.sum()
    idx = np.arange(n, dtype=np.float64)
    for w in weights:
        centre = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.01 * n, 0.15 * n)
        a += w * np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _gradient_and_conv(a: np.ndarray, beta: float,
                       high_beta_thr: float = 20.0) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the smoothed max‑convolution objective and
    the true max convolution value.

    For beta > `high_beta_thr` the soft‑max is essentially a hard max,
    and the gradient reduces to the exact sub‑gradient of the max
    (which is a simple shifted copy of `a`).  For lower beta the full
    log‑sum‑exp gradient is used (cross‑correlation with the soft‑max
    weights).
    """
    n = a.shape[0]
    # ----- full convolution a⋆a (length 2n‑1) -----
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]      # (a⋆a)

    max_conv = conv.max()

    if beta > high_beta_thr:
        # exact sub‑gradient of the max (hard‑max case)
        idx_max = int(conv.argmax())
        # grad_k = 2 * a_{idx_max - k} if the index is inside [0, n)
        idx = idx_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]
        return grad, max_conv

    # ----- soft‑max weights w_i -----
    exp_arg = beta * (conv - max_conv)                # ≤ 0
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum                                # Σ_i w_i = 1

    # ----- gradient = 2·(a ⋆_cross w_norm) -----
    rev_w = w_norm[::-1]                              # reverse for convolution
    out_len_corr = n + out_len - 1                    # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1                              # lag 0 position
    idx = offset - np.arange(n)                       # indices for j = 0…n‑1
    grad = 2.0 * corr_full[idx]

    return grad, max_conv


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1337,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressivity and speed
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Collect a diverse pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1. interpolated previous solution (if present)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.ndim == 1 and old.size > 0:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() > 0:
                a0 /= a0.sum()
                candidates.append(a0)
                print("[init] added interpolated previous solution")
    except Exception:
        pass

    # 2. a few deterministic Beta shapes (coarse grid)
    p_vals = np.geomspace(0.01, 15.0, 9)
    q_vals = p_vals.copy()
    for p in p_vals:
        for q in q_vals:
            cand = _beta_shape(p, q, n)
            if cand is not None:
                candidates.append(cand)

    # 3. random Dirichlet draws
    for _ in range(12):
        cand = rng.dirichlet(np.ones(n))
        candidates.append(cand)

    # 4. mixtures of Gaussians
    for _ in range(12):
        ncomp = rng.integers(2, 4)          # 2 or 3 components
        cand = _mixture_gaussian(n, rng, ncomp)
        candidates.append(cand)

    # 5. sin‑power families
    for _ in range(8):
        power = rng.uniform(0.3, 5.0)
        cand = _sin_power_shape(power, n)
        if cand is not None:
            candidates.append(cand)

    # 6. exponential decays
    for _ in range(8):
        lam = rng.uniform(0.0005, 0.02)     # 0.05% – 2% per step
        cand = _exp_decay_shape(lam, n)
        if cand is not None:
            candidates.append(cand)

    # 7. fallback uniform (guaranteed to exist)
    uniform = np.full(n, 1.0 / n)
    candidates.append(uniform)

    # ------------------------------------------------------------------
    # Optimisation parameters (Adam + temperature schedule)
    # ------------------------------------------------------------------
    alpha = 0.07                # base learning rate for Adam
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    beta_start = 0.5            # smoothing temperature
    beta_max = 250.0
    beta_mul = 1.012            # aggressive increase

    best_score = np.inf
    best_seq = None
    n_iter_total = 0
    seed_index = 0

    # ------------------------------------------------------------------
    # Main optimisation loop over all seeds (early exit when time runs out)
    # ------------------------------------------------------------------
    for seed_vec in candidates:
        if time.time() >= deadline:
            break

        # ensure feasible start (simplex projection)
        a = _project_simplex(seed_vec.copy())
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        beta = beta_start
        iter_local = 0

        # per‑seed iteration budget – we limit to a few thousand steps
        while time.time() < deadline - 0.2:
            # compute gradient and current max‑convolution value
            grad, cur_max = _gradient_and_conv(a, beta)
            cur_val = 2.0 * n * cur_max                # exact objective for sum=1

            # keep the best ever seen
            if cur_val < best_score:
                best_score = cur_val
                best_seq = a.copy()
                print(f"[improve] new best {best_score:.6f} "
                      f"(seed {seed_index}, iter {iter_local}, β={beta:.2f})")
                if best_score <= 1.503:
                    # target reached – we can stop early
                    deadline = time.time() + 0.3   # finish soon
                    break

            # ---- Adam update ----
            iter_local += 1
            n_iter_total += 1

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** iter_local)
            v_hat = v / (1.0 - beta2 ** iter_local)

            step = alpha * m_hat / (np.sqrt(v_hat) + eps)

            # projected Adam step
            a_candidate = a - step
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

            # tiny jitter occasionally to escape flat plateaus
            if rng.random() < 0.001:
                a_candidate += rng.normal(scale=0.001, size=n)
                a_candidate = np.maximum(a_candidate, 0.0)
                a_candidate = _project_simplex(a_candidate)

            a = a_candidate

            # temperature schedule
            beta = min(beta_max, beta * beta_mul)

            # stop condition for this seed (prevent endless loops)
            if iter_local >= 6000:
                break

        seed_index += 1

    # ------------------------------------------------------------------
    # Final secure evaluation (the evaluator caps values to [0,1000] etc.)
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen, but fall back to uniform just in case
        best_seq = uniform

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_seq.tolist()

This implementation expands the search space beyond the previous Run‑of‑the‑mill solution, uses a faster exact sub‑gradient once the smoothing parameter grows large, employs an Adam optimiser with simplex projection, and systematically restarts from a wide variety of seeds. It is designed to stay well within the 1000 s time budget while striving to beat the target bound of 1.503.
