"""
Random-sparse search for step function with low max self‑convolution.
The algorithm draws many binary masks of a given density, assigns equal
mass to the selected indices (so the sum equals 1) and keeps the best
candidate.  Afterwards a simple hill‑climb moves mass from indices that
contribute most to the current maximal convolution coefficient to indices
that contribute little.  The whole search respects the 1000 s budget.
"""

import time
import numpy as np

def _full_convolution(x: np.ndarray) -> np.ndarray:
    ...

def _objective(a: np.ndarray) -> float:
    conv = _full_convolution(a)
    return float(2.0 * a.shape[0] * conv.max())

def _random_sparse_candidate(n: int, density: float) -> np.ndarray:
    m = max(1, int(density * n))
    # pick distinct indices
    idx = np.random.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m   # mass normalises to 1
    return a

def _hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray,float]:
    """
    Simple hill‑climb: move a small fraction of mass from the index that
    participates most in the current maximal convolution coefficient to a
    random low‑contribution index.  Accept only improving moves.
    """
    n = a.shape[0]
    best = a.copy()
    best_val = _objective(best)
    cur = a.copy()
    cur_val = best_val

    while time.time() < deadline:
        # compute convolution once per iteration
        conv = _full_convolution(cur)
        k_max = int(np.argmax(conv))
        # compute contribution of each i to that lag
        contrib = np.zeros(n)
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib[mask] = cur[i_vals[mask]] * cur[j_vals[mask]]
        total = contrib.sum()
        if total == 0.0:
            break
        # source index – weighted by contribution
        src = np.random.choice(n, p=contrib/total)
        if cur[src] < 1e-12:
            continue
        # target index – low contribution (inverse)
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv/inv.sum())
        # move a random fraction up to 50% of src mass
        delta = cur[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue
        new = cur.copy()
        new[src] -= delta
        new[tgt] += delta
        val = _objective(new)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best = new.copy()
                best_val = val
        # early exit if below target
        if best_val <= 1.503:
            break
    return best, best_val

"""
Random‑sparse search for a step function with a small
self‑convolution maximum.

The algorithm works in two phases:

1️⃣  Randomly generate many *sparse* candidates.
    A fixed number `m` of positions are chosen uniformly,
    each receives the same weight `1/m` (so the total sum is 1).
    The convolution is computed with an FFT – this is cheap
    for the lengths we use (8k‑16k).  The best candidate seen
    so far is stored.

2️⃣  A tiny hill‑climbing / mass‑transfer refinement moves a
    fraction of mass from indices that contribute most to the
    currently maximal convolution coefficient to indices that
    contribute little.  Only improving moves are kept.

Both phases respect the 1000 s time budget.
The final sequence is returned as a plain list of floats;
`evaluate_sequence` will then be called by the host.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum of the vector is assumed to be 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(a∗a) for a (non‑negative, sum≈1)."""
    conv = _full_convolution(a)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Random sparse candidate generator
# ----------------------------------------------------------------------
def _random_sparse_candidate(n: int, density: float) -> np.ndarray:
    """
    Return a length‑n vector with `m = round(density·n)` non‑zero entries,
    each equal to 1/m, so that the vector sums to 1.
    """
    m = max(1, int(round(density * n)))
    # pick `m` distinct indices
    idx = np.random.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Light hill‑climbing (mass transfer) for polishing
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very simple hill‑climb that reduces the maximal convolution
    coefficient by moving a small fraction of mass from a “heavy”
    contributor to a “light” one.  Stops when `deadline` is reached
    or the target value 1.503 is achieved.
    """
    n = a.shape[0]
    best = a.copy()
    best_val = _objective(best)
    cur = a.copy()
    cur_val = best_val

    while time.time() < deadline:
        conv = _full_convolution(cur)
        k_max = int(np.argmax(conv))            # lag with maximal coeff.

        # contribution of each index to this lag
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[i[mask]] * cur[j[mask]]
        total = contrib.sum()
        if total == 0.0:
            break

        # source index – preferably a heavy contributor
        src = np.random.choice(n, p=contrib / total)
        if cur[src] < 1e-12:
            continue

        # target index – a light contributor (inverse weighting)
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        # move a random fraction of the source mass (up to 50%)
        delta = cur[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        new = cur.copy()
        new[src] -= delta
        new[tgt] += delta

        new_val = _objective(new)
        if new_val < cur_val:                     # accept only improving moves
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val
                print(f"[Hill] n={n} val={best_val:.6f}")
                if best_val <= 1.503:
                    break

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function sequence that minimises ``evaluate_sequence``.
    The routine spends most of the budget drawing random sparse masks,
    then applies a tiny hill‑climbing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # Problem sizes – we try a few; larger sizes give a finer bound
    sizes = [8000, 12000, 16000]

    global_best_a = None
    global_best_val = np.inf

    # ------------------------------------------------------------------
    # 1️⃣ Random sparse search for each size
    # ------------------------------------------------------------------
    for n in sizes:
        # how much time may we still use for this size?
        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # allocate at most 30 s (or a fraction of remaining time) to random sampling
        rand_deadline = time.time() + min(30.0, remaining * 0.45)

        best_a_local = None
        best_val_local = np.inf

        # a few densities to explore (from 5 % up to 45 %)
        densities = [0.05, 0.08, 0.12, 0.18, 0.25, 0.33, 0.40]

        while time.time() < rand_deadline:
            density = np.random.choice(densities)
            cand = _random_sparse_candidate(n, density)
            val = _objective(cand)
            if val < best_val_local:
                best_val_local = val
                best_a_local = cand
                print(f"[Rand] n={n} dens={density:.2f} m={int(round(density*n))} val={val:.6f}")
                if best_val_local <= 1.503:           # early exit
                    break

        if best_a_local is None:
            continue

        # ------------------------------------------------------------------
        # 2️⃣ Light hill‑climb polishing (time‑boxed)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 2.0:
            hill_deadline = time.time() + min(10.0, remaining * 0.35)
            best_a_local, best_val_local = _hill_climb(best_a_local, hill_deadline)

        # keep the best across sizes
        if best_val_local < global_best_val:
            global_best_val = best_val_local
            global_best_a = best_a_local.copy()
            print(f"[Best] new global best n={n} val={global_best_val:.6f}")

        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if global_best_a is None:
        n = sizes[0]
        global_best_a = np.full(n, 1.0 / n)
        global_best_val = _objective(global_best_a)

    # ------------------------------------------------------------------
    # Final secure evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
