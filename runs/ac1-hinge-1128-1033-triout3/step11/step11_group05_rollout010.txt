The script below searches for a non‑negative step‑function (a probability vector) that minimizes  

\[
\text{score}= \frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

where `a*a` is the full linear convolution of the vector with itself.
The idea is a **two‑phase strategy**:

1. **Exploratory phase** – generate many candidates from expressive parametric families  
   (Beta, exponential, Gaussian, power‑sine, inverse‑polynomial, mixtures, sparse Dirichlet).  
   Every candidate is normalised to sum = 1 and evaluated with a cheap objective
   `2·n·max(conv)`.  The best few candidates are kept.

2. **Refinement phase** – improve the current best candidate by **mirror‑descent** on the
   exact objective.  
   * The sub‑gradient of `max(conv)` is built from the *k* largest convolution entries
     (default k = 30) – this is a smooth approximation of the true sub‑gradient.  
   * An exponential (multiplicative) update keeps the iterate inside the simplex
     (entropic mirror descent).  
   * Momentum (β ≈ 0.9) stabilises the descent, a linearly‑decaying step‑size encourages
     convergence, and occasional Gaussian jitter helps to escape plateaus.  
   * Several independent restarts are performed while time permits.

The whole routine respects the supplied time budget (`budget_s`, default 1000 s).  
All helper functions are defined at top level, there are no closures or lamb‑functions,
and the only external call is the already‑imported `evaluate_sequence`.

```python
"""
Propose a non‑negative step function that (hopefully) beats the previous best
score ≈ 1.5085.  Two phases are used:

1️⃣  Wide random search over many parametric families (Beta, exponential,
    Gaussian, power‑sine, inverse‑polynomial, mixtures, sparse Dirichlet).
    Every candidate is normalised to Σa = 1 and evaluated with the cheap
    objective 2·n·max(conv(a,a)).

2️⃣  Mirror‑descent refinement of the current best candidate.
    • Sub‑gradient of max(conv) is assembled from the 30 largest lags.
    • Exponential (entropic) update + momentum = Adam‑like behaviour.
    • Linear step‑size decay, occasional Gaussian jitter and many restarts
      make the search robust.
    • The loop runs until the time budget is exhausted; the best sequence
      found is finally handed to the provided ``evaluate_sequence`` for the
      official score.

The algorithm runs on a single CPU, uses only numpy (no external solvers)
and respects the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution / objective utilities
# ----------------------------------------------------------------------


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (length 2·len(a)−1) via FFT.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()           # next power of two
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Cheap objective used during the search: 2·n·max(conv(a,a)).
    The vector a is assumed to be already normalised (sum = 1).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     active_k: int = 30,
                     rel_tol: float = 1e-8) -> np.ndarray:
    """
    Return a sub‑gradient of max(conv) w.r.t. a.
    Only the ``active_k`` largest entries of conv are used (with optional
    tolerance) – this yields a smoother direction than a single maximiser.
    """
    n = a.shape[0]
    max_val = conv.max()

    if active_k >= conv.size:
        active = np.arange(conv.size)
    else:
        # indices of the largest entries
        active = np.argpartition(conv, -active_k)[-active_k:]
        mask = conv[active] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            active = np.array([int(np.argmax(conv))])
        else:
            active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Shape families (used in the wide exploration phase)
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """a_i ∝ i^{p‑1}(n‑i+1)^{q‑1},   i = 1,…,n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_shape(c: float, n: int) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(−c·|i−mid|)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a


def _gaussian_shape(sigma: float, n: int) -> np.ndarray:
    """Gaussian centred at the middle."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a


def _power_sine_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1))^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    a /= a.sum()
    return a


def _inverse_poly_shape(alpha: float, n: int) -> np.ndarray:
    """U‑shaped profile a_i ∝ ((i+1)*(n‑i))^{‑α}."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = (i * (n - i + 1.0)) ** (-alpha)
    a /= a.sum()
    return a


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a


def _mixture_beta(p1: float, q1: float,
                  p2: float, q2: float,
                  w: float, n: int) -> np.ndarray:
    """Convex mixture of two Beta shapes."""
    a1 = _beta_shape(p1, q1, n)
    a2 = _beta_shape(p2, q2, n)
    if a1 is None or a2 is None:
        return None
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Mirror‑descent refinement
# ----------------------------------------------------------------------


def _refine_sequence(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) optimisation of the exact
    objective.  Returns the best sequence seen during the run and its
    objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    no_improve = 0
    # parameters
    active_k = 30
    jitter_prob = 0.001
    max_no_improve = 6_000

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution_fft(a)
        cur_val = 2.0 * n * conv.max()

        # keep the best version
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[refine] iter {it:6d}  new best {best_val:.6f}")
        else:
            no_improve += 1

        # sub‑gradient from the largest lags
        grad = _subgradient_max(a, conv, active_k=active_k)

        # momentum (exponential moving average)
        momentum = beta * momentum + (1.0 - beta) * grad

        # step‑size schedule – linear decay from 0.2 to 1e‑5
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = elapsed / total
        step = 0.2 * (1.0 - progress) + 1e-5 * progress

        # exponentiated (multiplicative) update
        log_factor = -step * momentum
        # clip to avoid overflow/underflow
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional random jitter to shake out of plateaus
        if np.random.rand() < jitter_prob or no_improve >= max_no_improve:
            jitter = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            momentum.fill(0.0)
            no_improve = 0

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget, returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressivity and speed.
    # ------------------------------------------------------------------
    n = 8000                       # length of the step function

    best_seq: np.ndarray | None = None
    best_val = np.inf

    def _maybe_update(candidate: np.ndarray):
        """Update the global best if the candidate improves the cheap objective."""
        nonlocal best_seq, best_val
        val = _objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[search] new best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣ Initialise from a previously saved solution (if any)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _maybe_update(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Wide exploration over several parametric families
    # ------------------------------------------------------------------
    # (a) dense Beta grid
    p_grid = np.concatenate(
        [
            np.geomspace(0.001, 0.1, 8, endpoint=False),
            np.geomspace(0.1, 5.0, 12, endpoint=False),
            np.geomspace(5.0, 20.0, 5),
        ]
    )
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is not None:
                _maybe_update(cand)

    # (b) random Beta draws (log‑uniform)
    rng = np.random.default_rng()
    log_min, log_max = np.log(1e-3), np.log(20.0)
    for _ in range(8000):
        p = np.exp(rng.uniform(log_min, log_max))
        q = np.exp(rng.uniform(log_min, log_max))
        cand = _beta_shape(p, q, n)
        if cand is not None:
            _maybe_update(cand)

        if time.time() >= deadline - 200.0:      # reserve ~200 s for refinement
            break

    # (c) exponential family
    for _ in range(3000):
        c = rng.uniform(0.0, 6.0)
        _maybe_update(_exp_shape(c, n))

    # (d) Gaussian family
    for _ in range(3000):
        sigma = rng.uniform(0.5, n / 4.0)
        _maybe_update(_gaussian_shape(sigma, n))

    # (e) power‑sine family
    for _ in range(3000):
        p = rng.uniform(0.1, 10.0)
        _maybe_update(_power_sine_shape(p, n))

    # (f) inverse‑polynomial (U‑shaped) family
    for _ in range(2000):
        alpha = rng.uniform(0.2, 4.0)
        _maybe_update(_inverse_poly_shape(alpha, n))

    # (g) simple triangular window
    _maybe_update(_triangular_shape(n))

    # (h) mixture of two Beta distributions
    for _ in range(2000):
        p1 = np.exp(rng.uniform(log_min, log_max))
        q1 = np.exp(rng.uniform(log_min, log_max))
        p2 = np.exp(rng.uniform(log_min, log_max))
        q2 = np.exp(rng.uniform(log_min, log_max))
        w = rng.random()
        cand = _mixture_beta(p1, q1, p2, q2, w, n)
        if cand is not None:
            _maybe_update(cand)

    # (i) sparse Dirichlet‑like samples (few non‑zero entries)
    for _ in range(1200):
        support = rng.integers(20, 1800)
        idx = rng.choice(n, size=support, replace=False)
        vals = rng.random(support)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = vals
        cand /= cand.sum()
        _maybe_update(cand)

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)
        print("[fallback] uniform start, value=", best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Refinement phase – mirror‑descent optimiser
    # ------------------------------------------------------------------
    # reserve a bit of time for the final official evaluation
    refine_deadline = deadline - 0.2

    # We will perform several independent refinement runs.  The first run starts
    # from the best candidate found so far; subsequent runs start from random
    # Dirichlet draws to explore other basins.
    refinement_runs = 5
    for run in range(refinement_runs):
        if time.time() >= refine_deadline:
            break

        # initialise the sequence for this run
        if run == 0:
            a_start = best_seq.copy()
        else:
            # a fresh random start – Dirichlet with random concentration
            concentration = rng.uniform(0.05, 2.0)
            a_start = rng.gamma(concentration, 1.0, size=n)
            a_start /= a_start.sum()

        a_opt, val_opt = _refine_sequence(a_start,
                                          deadline=refine_deadline,
                                          start_time=start_time)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[overall] improvement after refinement run {run} -> {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final secure evaluation with the provided routine
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:                                   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = float('inf')

    # return the raw list (evaluate_sequence will do its own clipping)
    return best_seq.tolist()
```
