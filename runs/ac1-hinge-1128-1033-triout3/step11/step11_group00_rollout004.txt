The objective is   J(a) = 2·n·max(conv(a,a))   (scale‑invariant, we keep Σa = 1).
We minimise a smooth approximation ϕ_α(a) = 2·n·(log‑sum‑exp(α·conv(a,a))/α)
using Adam on the logits z=log(a) and a schedule that drives the smoothing
parameter α from 1 → 500.  After the Adam phase a short entropy‑mirror‑descent
(polishing) on the exact max‑entry is performed.

Key improvements over the previous baseline:
* α grows to 500 (instead of ≈200) giving a sharper approximation of the true max.
* Learning‑rate decays to 0.001 (instead of 0.004) for finer final steps.
* More aggressive entropy‑mirror polishing (1200 steps, η₀=0.55).
* Jittering is triggered earlier (after 2000 non‑improving steps) to escape
  shallow plateaus.
* If a previous solution (``height_sequence_1``) exists we start from an
  interpolated version, otherwise we try three diverse initialisations
  (Beta(2,2), Beta(0.5,0.5) and a Dirichlet sample) and keep the best.
The routine respects the 1000 s budget, automatically leaving a short safety
margin before calling ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT based full convolutions
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two (possibly different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror‐descent polishing (exact max)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 1200, eta_start: float = 0.55) -> np.ndarray:
    """
    Entropy mirror descent on the *exact* max‑entry of the convolution.
    The update is a ← a·exp(‑η·g) then renormalisation, where g is the
    sub‑gradient of the max‑convolution entry.
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # Sub‑gradient of the selected entry (ordered pairs)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            # each pair (i, k‑i) contributes a[k‑i]; ordered pairs are counted twice
            g[idx] = 2.0 * a[k - idx]
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Helper: generate different initial probability vectors
# ----------------------------------------------------------------------
def _initial_vectors(n: int, seed: int) -> List[np.ndarray]:
    """Return a list of plausible starting probability vectors."""
    rng = np.random.default_rng(seed)

    # 1. Previous solution (if available) – interpolated to current length.
    vecs = []
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
            vecs.append(a)
            print("[init] interpolated previous solution")
    except Exception:
        pass

    # 2. Symmetric Beta(2,2) – gently bell shaped.
    i = np.arange(1, n + 1, dtype=np.float64)
    p = q = 2.0
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    a /= a.sum()
    vecs.append(a)
    print("[init] generated Beta(2,2) start")

    # 3. U‑shaped Beta(0.5,0.5) – more mass at the ends.
    p = q = 0.5
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a /= a.sum()
    vecs.append(a)
    print("[init] generated Beta(0.5,0.5) start")

    # 4. Random Dirichlet (α = 0.3) – sparse, exploratory start.
    a = rng.dirichlet(np.full(n, 0.3))
    vecs.append(a)
    print("[init] generated Dirichlet(0.3) start")

    return vecs


# ----------------------------------------------------------------------
# Core optimiser (Adam on a soft‑max of the convolution)
# ----------------------------------------------------------------------
def _optimise(a0: np.ndarray,
              deadline: float,
              max_alpha: float = 500.0,
              lr_start: float = 0.12,
              lr_end: float = 0.001,
              stagnation_limit: int = 2000) -> np.ndarray:
    """
    Run the Adam optimiser on a given starting probability vector.
    Returns the best vector seen before the deadline.
    """
    n = a0.shape[0]
    eps = 1e-30
    z = np.log(a0 + eps)          # work in log‑domain → positivity automatically

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = None
    best_obj = np.inf
    stagnation = 0
    it = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Learning‑rate and smoothing schedules (both monotone)
        # --------------------------------------------------------------
        now = time.time()
        prog = (now - (deadline - (deadline - start))) / (deadline - (deadline - (deadline - start)))
        # linear decay for LR
        lr = lr_start * (1.0 - prog) + lr_end * prog
        # α grows polynomially (more aggressive than linear)
        alpha = 1.0 + (max_alpha - 1.0) * (prog ** 1.3)

        # --------------------------------------------------------------
        # Recover probability vector a from logits
        # --------------------------------------------------------------
        a = np.exp(z)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # --------------------------------------------------------------
        # Full self‑convolution (needed for soft‑max)
        # --------------------------------------------------------------
        conv = _full_convolution(a)            # length 2n‑1

        # --------------------------------------------------------------
        # Soft‑max of the convolution (stable log‑sum‑exp)
        # --------------------------------------------------------------
        maxc = conv.max()
        exp_scaled = np.exp(alpha * (conv - maxc))
        sumexp = exp_scaled.sum()
        # smooth objective (identical to the one used for Adam)
        obj_smooth = 2.0 * n * (maxc + np.log(sumexp) / alpha)

        # --------------------------------------------------------------
        # Keep exact best value (no smoothing)
        # --------------------------------------------------------------
        if obj_smooth < best_obj - 1e-12:
            # true unsmoothed objective for reporting
            exact_obj = 2.0 * n * conv.max()
            best_obj = exact_obj
            best_a = a.copy()
            stagnation = 0
            print(f"[iter] new best J={best_obj:.6f}  it={it}  t={now-start:.1f}s")
        else:
            stagnation += 1

        # --------------------------------------------------------------
        # Gradient w.r.t. a for the smooth objective
        # (derived in the original code)
        # --------------------------------------------------------------
        rev_a = a[::-1]
        conv_rev = _full_convolution_two(rev_a, exp_scaled / sumexp)
        g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]   # slice length n

        # --------------------------------------------------------------
        # Chain rule through the softmax → gradient w.r.t. logits z
        # --------------------------------------------------------------
        dot = np.dot(g_a, a)
        grad_z = a * (g_a - dot)

        # --------------------------------------------------------------
        # Adam update on z
        # --------------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------------------------------------------------------------
        # Escape from plateaus – add a small random jitter
        # --------------------------------------------------------------
        if stagnation >= stagnation_limit:
            jitter_scale = 0.015
            z += np.random.normal(scale=jitter_scale, size=n)
            stagnation = 0
            print(f"[reset] jitter added at it={it}")

    # ------------------------------------------------------------------
    # Return the best probability vector found (or the final one if none)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.exp(z)
        best_a /= best_a.sum()
    return best_a


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s time
    budget.
    """
    np.random.seed(seed)
    global start
    start = time.time()
    # keep a safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.8

    # --------------------------------------------------------------
    # Problem size – a modest increase (9000) gives a finer discretisation
    # without a prohibitive cost increase.
    # --------------------------------------------------------------
    n = 9000

    # --------------------------------------------------------------
    # Generate a few diverse starting points and optimise each for an
    # equal share of the remaining budget.
    # --------------------------------------------------------------
    init_vecs = _initial_vectors(n, seed)
    remaining = deadline - time.time()
    per_run = remaining / max(1, len(init_vecs))

    best_overall = None
    best_score = np.inf

    for idx, a0 in enumerate(init_vecs):
        run_deadline = time.time() + per_run
        if run_deadline - time.time() < 1.0:
            break   # not enough time for another run
        print(f"\n[run] start optimisation #{idx+1} (budget {per_run:.1f}s)")

        a_opt = _optimise(a0, run_deadline)

        # ----------------------------------------------------------
        # Polishing phase – entropy‑mirror descent on the *exact* max.
        # ----------------------------------------------------------
        a_refined = _max_refine(a_opt, steps=1500, eta_start=0.55)
        conv_ref = _full_convolution(a_refined)
        J_ref = 2.0 * n * conv_ref.max()

        if J_ref < best_score:
            best_score = J_ref
            best_overall = a_refined.copy()
            print(f"[run] new global best J={best_score:.6f}")

    # ------------------------------------------------------------------
    # If, for any reason, we have not produced a candidate, fall back to
    # a simple uniform distribution.
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final evaluation (the wrapper will clamp values to the allowed range)
    # ------------------------------------------------------------------
    final_seq = best_overall.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"\n[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx J={best_score:.6f})")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
```

This implementation follows the same high‑level idea as the baseline but
introduces a more aggressive smoothing schedule (α up to 500), a finer
learning‑rate decay, earlier jittering, several diverse initialisations,
and a longer entropy‑mirror polishing phase.  All operations respect the
1000 s time limit and the returned sequence contains only non‑negative
floats, ready for ``evaluate_sequence``.
