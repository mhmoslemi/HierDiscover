def adam_optimize(a0, n, deadline):
    a = a0.copy()
    eps = 1e-30
    z = np.log(a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr_start, lr_end = 0.12, 0.002  # maybe adjust
    best_a = a.copy()
    best_obj = np.inf
    it = 0
    stagnation = 0
    while time.time() < deadline:
        # compute schedule
        prog = (time.time() - start) / (deadline - start)
        lr = lr_start * (1 - prog) + lr_end * prog
        alpha = 1.0 + 500.0 * (prog ** 1.2)  # bigger α
        # reconstruct a
        a = np.exp(z)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a /= s
        # convolution
        conv = _full_convolution(a)
        maxconv = conv.max()
        obj = 2.0 * n * maxconv
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagnation = 0
            print(f"[adam] new best O={best_obj:.6f} it={it}")
        else:
            stagnation += 1
        # soft-max gradient
        maxc = conv.max()
        exp_scaled = np.exp(alpha * (conv - maxc))
        sumexp = exp_scaled.sum()
        weights = exp_scaled / sumexp
        # gradient via conv_rev of rev(a) with weights
        rev_a = a[::-1]
        conv_rev = _full_convolution_two(rev_a, weights)
        g_a = 4.0 * n * conv_rev[n-1:n-1+n]
        # chain rule
        dot = np.dot(g_a, a)
        grad_z = a * (g_a - dot)
        # Adam update
        it += 1
        m = beta1*m + (1-beta1)*grad_z
        v = beta2*v + (1-beta2)*(grad_z**2)
        m_hat = m/(1 - beta1**it)
        v_hat = v/(1 - beta2**it)
        z = z - lr * m_hat/(np.sqrt(v_hat)+eps_adam)
        # jitter resets
        if stagnation >= 5000:
            z += np.random.normal(scale=0.01, size=n)
            stagnation = 0
            print(f"[restart] jitter at it={it}")
    return best_a, best_obj

"""
Hybrid optimization for the “max self‑convolution” problem.

1. Initialise a probability vector a (sum = 1).  If a previous solution
   (global ``height_sequence_1``) is available it is interpolated;
   otherwise a random Dirichlet sample is used.

2. Phase‑A – Adam on a soft‑max approximation of the max‑convolution.
   The objective is 2·n·max(conv(a,a)).  A temperature α grows from 1 to
   several hundred, turning the smooth soft‑max into the true max.
   Adam with exponential‑moving‑averages updates the logits z = log a.

3. Phase‑B – Entropy‑mirror descent on the *exact* max entry.
   Using the sub‑gradient of max(conv) (only the indices participating
   in the current maximal convolution term) the update
   a_i ← a_i·exp(‑η·g_i) is performed and a is re‑normalised.
   A long run (many thousands of steps) drives the maximal
   convolution value down further.

4. Several independent restarts (different random seeds) are executed
   within the 1000 s budget; the best sequence over all restarts is kept.

The final sequence is returned as a ``list[float]`` and its exact score
is printed via ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (self‑convolution and cross‑convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two different vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent polishing (exact max)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray,
                steps: int = 8000,
                eta_start: float = 0.45,
                deadline: float = None) -> np.ndarray:
    """Mirror‑descent on the exact max‑entry of the self‑convolution."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        if deadline is not None and time.time() >= deadline:
            break
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed (soft‑max) version of the objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   start: float,
                   deadline: float,
                   alpha_max: float = 500.0,
                   lr_start: float = 0.12,
                   lr_end: float = 0.002,
                   jitter_thresh: int = 6000) -> tuple[np.ndarray, float]:
    """
    Adam on a soft‑max approximation of max(conv(a,a)).
    Returns the best vector (sum = 1) found and its exact objective.
    """
    a = a0.copy()
    eps = 1e-30
    z = np.log(a + eps)                # logits – guarantees positivity
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_a = a.copy()
    best_obj = np.inf
    it = 0
    stagnation = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start) / (deadline - start)          # 0 → 1
        lr = lr_start * (1.0 - prog) + lr_end * prog
        # α grows from 1 to alpha_max
        alpha = 1.0 + (alpha_max - 1.0) * (prog ** 1.2)

        # -------- recover a and normalise --------
        a = np.exp(z)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # -------- exact max for reporting --------
        conv = _full_convolution(a)
        maxconv = conv.max()
        obj = 2.0 * n * maxconv
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagnation = 0
            print(f"[adam] new best O={best_obj:.6f} it={it} t={now-start:.1f}s")
        else:
            stagnation += 1

        # -------- soft‑max weights (smooth max) --------
        maxc = conv.max()
        exp_scaled = np.exp(alpha * (conv - maxc))       # stable soft‑max
        sumexp = exp_scaled.sum()
        weights = exp_scaled / sumexp                     # Σ weights = 1

        # -------- gradient w.r.t. a (via convolution) --------
        rev_a = a[::-1]
        conv_rev = _full_convolution_two(rev_a, weights)   # length 3n‑2
        start_idx = n - 1
        g_a = 4.0 * n * conv_rev[start_idx:start_idx + n]   # slice length n

        # chain‑rule through the soft‑max (a = softmax(z))
        dot = np.dot(g_a, a)
        grad_z = a * (g_a - dot)

        # -------- Adam update on logits --------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # -------- jitter if stuck for a while --------
        if stagnation >= jitter_thresh:
            z += np.random.normal(scale=0.02, size=n)
            stagnation = 0
            print(f"[adam] jitter added at it={it}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising
    evaluate_sequence(seq).  The routine respects the 1000 s budget.
    """
    np.random.seed(seed)
    start_all = time.time()
    deadline_all = start_all + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer discretisation and usually
    # a slightly smaller achievable bound.
    # ------------------------------------------------------------------
    n = 12000

    best_seq = None
    best_obj = np.inf
    restart = 0

    # ------------------------------------------------------------------
    # Keep trying independent restarts while time permits.
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline_all:
            break

        # Allocate at most 300 s (or the remaining time) for a single run.
        # Very short remaining time is ignored.
        time_left = deadline_all - now
        if time_left < 30.0:
            break
        run_deadline = now + min(300.0, time_left)

        # --------------------- initialise a -------------------------
        if restart == 0 and 'height_sequence_1' in globals():
            try:
                old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
                old_len = old.shape[0]
                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old_len)
                a0 = np.interp(x_new, x_old, old)
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
                print("[init] interpolated previous solution")
            except Exception:
                # fall back to a random Dirichlet sample
                a0 = np.random.gamma(shape=1.0, scale=1.0, size=n)
                a0 /= a0.sum()
                print("[init] fallback Dirichlet")
        else:
            # random Dirichlet with a concentration drawn from a wide range
            conc = np.random.uniform(0.2, 5.0)
            a0 = np.random.gamma(shape=conc, scale=1.0, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            print(f"[init] random Dirichlet (conc={conc:.2f})")
        # ------------------------------------------------------------------
        # Phase‑A – Adam on a smoothed max‑convolution
        # ------------------------------------------------------------------
        a_adam, obj_adam = _adam_optimize(a0, n,
                                           start=now,
                                           deadline=run_deadline,
                                           alpha_max=500.0,
                                           lr_start=0.12,
                                           lr_end=0.002,
                                           jitter_thresh=6000)

        # ------------------------------------------------------------------
        # Phase‑B – long mirror‑descent polishing on the exact max
        # ------------------------------------------------------------------
        polish_deadline = deadline_all - 0.2   # keep a tiny margin for final eval
        a_refined = _max_refine(a_adam,
                                steps=10000,
                                eta_start=0.45,
                                deadline=polish_deadline)
        conv_ref = _full_convolution(a_refined)
        obj_ref = 2.0 * n * conv_ref.max()

        if obj_ref < best_obj:
            best_obj = obj_ref
            best_seq = a_refined.copy()
            print(f"[global] new overall best O={best_obj:.6f} after restart {restart}")

        restart += 1

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should not happen) use uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
