def subgradient_opt(a0, n, deadline, start_time):
    a = a0.copy()
    a /= a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    iter = 0
    while time.time() < deadline:
        # compute convolution via FFT
        conv = _full_convolution(a)
        k = int(np.argmax(conv))
        # compute gradient (subgradient) = 2 * a shifted such that g[i] = a[k - i] if in range
        g = np.zeros_like(a)
        # compute indices: For i where 0 <= k - i < n => i in [max(0, k - (n - 1)), min(k, n-1)]
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # for i in this range: g[i] = 2 * a[k - i]
        # we can vectorized:
        idx = np.arange(i_low, i_high + 1)
        g[idx] = 2.0 * a[k - idx]

        # step size schedule
        iter += 1
        lr = 0.1 / np.sqrt(iter)

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s == 0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Subgrad {iter}] new best = {best_val:.6f}")
            # if best_val <= target break

    return best_a, best_val

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 5000  # moderate size

    seeds = []

    # previous solution interpolation if exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0)
        a0 /= a0.sum()
        seeds.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # beta family
    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_a is not None:
            seeds.append(beta_a)
            print(f"[init] beta seed value={beta_val:.6f}")
    except Exception as e:
        print("[init] beta fail", e)

    # dirichlet draws with multiple concentration levels
    alphas = [0.5, 1.0, 2.0, 5.0]
    for alpha in alphas:
        for _ in range(2):
            seeds.append(np.random.dirichlet(np.full(n, alpha)))
    print(f"[init] total seeds = {len(seeds)}")

    # allocate time for smooth optimisation
    time_for_smooth = (deadline - time.time()) * 0.80
    if time_for_smooth < 0:
        time_for_smooth = 0
    per_seed = time_for_smooth / max(1, len(seeds))

    best_val = np.inf
    best_a = None

    for i, a_seed in enumerate(seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed
        a_opt, val_opt = _smooth_optimize(a_seed, n, local_deadline, start_time, lambda_max=120.0)
        print(f"[seed {i}] smooth best = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
        if best_val <= 1.503:
            break

    # subgradient refinement
    remaining = deadline - time.time()
    if remaining > 5.0 and best_a is not None:
        sub_deadline = time.time() + remaining * 0.4
        a_refined, val_refined = _subgradient_optimize(best_a, n, sub_deadline, start_time)
        print(f"[subgrad] refined best = {val_refined:.6f}")
        if val_refined < best_val:
            best_val, best_a = val_refined, a_refined.copy()

    # final hillclimb

"""
Optimisation of a non‑negative step‑function to minimise the
``evaluate_sequence`` objective.

The algorithm proceeds in three stages:

1️⃣  **Initial seeds** – a mixture of (i) the previously obtained
    solution (if available, interpolated to the current length),
    (ii) the best member of a dense two‑parameter Beta family,
    (iii) several Dirichlet draws with different concentration
    parameters.  All seeds are projected onto the probability simplex.

2️⃣  **Smooth‑max Adam optimisation** – the non‑differentiable
    ``max(a*a)`` is replaced by a log‑sum‑exp surrogate with a
    continuation schedule ``λ`` that grows from 0.5 to a configurable
    ``λ_max`` (default 120).  Adam updates are projected onto the simplex
    after each step.  The best solution over all seeds is retained.

3️⃣  **Projected sub‑gradient polishing** – from the best smooth
    solution we perform a simple sub‑gradient descent on the exact
    max‑convolution (the gradient of ``max_k (a*a)_k`` is
    ``g_i = 2·a_{k*‑i}`` where ``k*`` is the current maximiser).  A
    diminishing step size and simplex projection ensure convergence.

4️⃣  **Final simulated‑annealing hill‑climb** – an intensive local
    search (identical to the reference implementation) cleans up any
    remaining imperfections.

The whole routine respects a 1000 s time budget; intermediate progress
is printed.  The returned list of floats is the candidate sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1, but works for any sum)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) / (sum(a)^2)."""
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Beta‑family sweep (dense but cheap)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search the beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf

    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + λ continuation)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     lambda_max: float = 120.0) -> tuple[np.ndarray, float]:
    """
    Optimises the evaluation function using a smoothed max objective.
    Adam updates are projected onto the simplex.  ``λ`` grows from 0.5
    to ``lambda_max`` as optimisation proceeds.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # ensure Σa = 1

    best_a = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.07                     # base learning rate
    t = 0

    # FFT sizes (pre‑computed for the whole run)
    size1 = 1 << ((2 * n - 2).bit_length())       # for a*a
    size2 = 1 << ((3 * n - 3).bit_length())       # for a_rev * w

    lambda_start = 0.5

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        # ---------- λ continuation ----------
        prog = (now - start_time) / (deadline - start_time)
        lam = lambda_start + (lambda_max - lambda_start) * prog   # 0.5 → lambda_max

        # ---------- a*a ----------
        fa = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fa * fa, size1)[:2 * n - 1]
        max_c = conv.max()

        # ---------- smooth weights ----------
        exps = np.exp(lam * (conv - max_c))          # shift for stability
        w = exps / exps.sum()                        # sum to 1, length 2·n‑1

        # ---------- gradient ----------
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, size2)
        fw = np.fft.rfft(w, size2)
        conv_aw = np.fft.irfft(fa_rev * fw, size2)[:3 * n - 2]
        grad = 2.0 * conv_aw[n - 1 : n - 1 + n]      # length n

        # ---------- Adam update ----------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        # ---------- keep the best ----------
        cur_val = _objective(a)          # exact objective of the *new* a
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[SmoothOpt {t:7d}] new best = {best_val:.6f} λ={lam:.2f}")
            if best_val <= 1.503:
                break

        # occasional random perturbation to avoid stagnation
        if t % 800 == 0:
            if cur_val >= best_val * 0.9999:
                noise = np.random.normal(scale=0.015, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[SmoothOpt {t:7d}] stagnation – random perturb")

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the true max‑convolution.
    The gradient of max_k (a*a)_k is g_i = 2·a_{k*‑i} (if the index exists).
    A diminishing step size 0.1/√t and simplex projection are used.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        it += 1

        conv = _full_convolution(a)
        k_star = int(np.argmax(conv))

        # build sub‑gradient g
        g = np.zeros_like(a)
        i_low = max(0, k_star - (n - 1))
        i_high = min(k_star, n - 1)
        if i_low <= i_high:
            idx = np.arange(i_low, i_high + 1)
            g[idx] = 2.0 * a[k_star - idx]

        # step size schedule
        lr = 0.1 / math.sqrt(it)

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        cur_val = _objective(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[SubGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= 1.503:
                break

        # occasional tiny random perturbation to shake off plateaus
        if it % 5000 == 0 and cur_val >= best_val * 0.9999:
            a = best_a.copy()
            a += np.random.normal(scale=0.008, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            print(f"[SubGrad {it:7d}] stagnation – perturb")

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = _objective(a)
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # Incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max / (a.sum() ** 2)

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective(a)
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # Problem size – 5000 gives a good speed / resolution trade‑off.
    n = 5000

    # ------------------------------------------------------------------
    # 0️⃣ Gather diverse initial seeds
    # ------------------------------------------------------------------
    seeds = []

    # a) previous solution, if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        seeds.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # b) best from a dense Beta‑family sweep
    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_a is not None:
            seeds.append(beta_a)
            print(f"[init] beta‑family seed (value={beta_val:.6f})")
    except Exception as exc:
        print("[init] beta family failed:", exc)

    # c) random Dirichlet draws with several concentration levels
    alphas = [0.3, 0.7, 1.0, 2.0, 5.0]
    for a in alphas:
        for _ in range(2):
            seeds.append(np.random.dirichlet(np.full(n, a)))
    print(f"[init] generated {len(seeds)} seeds")

    # ------------------------------------------------------------------
    # 1️⃣ Smooth‑max optimisation (most of the time budget)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    time_for_smooth = (deadline - time.time()) * 0.80
    per_seed = time_for_smooth / max(1, len(seeds))

    for i, a_seed in enumerate(seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed
        a_opt, val_opt = _smooth_optimize(
            a_seed, n, local_deadline, start_time, lambda_max=120.0
        )
        print(f"[seed {i}] smooth_opt best = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
        if best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Sub‑gradient polishing on the exact max‑convolution
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and best_a is not None:
        sub_deadline = time.time() + remaining * 0.40
        a_refined, val_refined = _subgradient_optimize(
            best_a, n, sub_deadline, start_time
        )
        print(f"[subgrad] refined best = {val_refined:.6f}")
        if val_refined < best_val:
            best_val, best_a = val_refined, a_refined.copy()

    # ------------------------------------------------------------------
    # 3️⃣ Final intensive hill‑climbing on whatever time is left
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and best_a is not None:
        best_a, best_val = _hillclimb_anneal(
            best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final secure evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
