"""
Improved step‑function optimizer.

The program searches a non‑negative vector a of length n (default 12000)
with Σa = 1 that minimises

        2·n·max(conv(a,a))

where conv is the full linear convolution.  The evaluation function used by
the competition simply computes the same quantity (after clipping the
vector to [0,1000] and forcing non‑negativity), so minimising the
expression above is exactly the goal.

The optimisation proceeds in four phases:

0️⃣  **Diversified initialisation** – many random Beta / Dirichlet draws,
    a few handcrafted shapes (sin, cosine‑square, linear ramp, arcsine‑like),
    and a mixture‑of‑Beta candidate.  The best of these seeds is kept.

1️⃣  **Adam on a smooth‐max surrogate** – we replace the hard max by a
    log‑sum‑exp with a temperature ε that is annealed from 0.5 down to 1e‑12.
    The variables are the logits of a soft‑max, guaranteeing the simplex
    constraints automatically.  A slower learning‑rate/ε schedule and a
    higher idle‑restart limit give the optimiser more chance to escape
    plateaus.

2️⃣  **Fine‑grained hill‑climbing** – repeatedly move a small chunk of mass
    from a “source’’ index that contributes heavily to the current maximal
    convolution entry to a “target’’ index (chosen from the whole vector,
    favouring low values).  The update of the convolution is performed in
    O(n) time with a corrected quadratic term (the previous implementation
    subtracted only δ² instead of 2·δ²).  A simulated‑annealing acceptance
    rule with a very gradual temperature schedule is employed, and full
    recomputation of the convolution is done occasionally to keep numerical
    drift in check.

3️⃣  **Exact sub‑gradient polishing** – projected sub‑gradient descent on the
    true objective (no smoothing).  The step‑size decays linearly from
    3·10⁻¹ to 1·10⁻⁵, with small random jitters to escape flat regions.

4️⃣  **Large‑move simulated annealing** – a final cheap annealing phase that
    makes larger source→target transfers, again using the corrected O(n)
    update.

All phases respect the overall time budget (default 1000 s).  The best
vector found is finally evaluated with the provided `evaluate_sequence`
function and returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random candidate generators
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Beta‑shaped distribution (with random exponents) and normalise."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_mixed_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent Beta‑shaped vectors."""
    a1 = _random_beta_candidate(n, rng)
    a2 = _random_beta_candidate(n, rng)
    w = rng.random()
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


def _random_dirichlet_candidate(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Simple Dirichlet draw with scalar concentration α."""
    a = rng.dirichlet(np.full(n, alpha))
    return a


def _arcsine_candidate(n: int) -> np.ndarray:
    """Arcsine‑like shape a_i ∝ 1/√(i·(n‑i+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    a /= a.sum()
    return a


def _linear_ramp_candidate(n: int) -> np.ndarray:
    """Triangular ramp – high at the ends, low in the centre."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - np.abs(i - (n - 1) / 2) / (n / 2))
    a /= a.sum()
    return a


def _sinusoid_candidate(n: int) -> np.ndarray:
    """Zero‑free sine wave (always positive)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _cosine_square_candidate(n: int) -> np.ndarray:
    """Cosine‑square shape (peaked in the middle)."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of the log‑sum‑exp smooth max."""
    w = np.log(a0 + 1e-12)                     # logits (unconstrained)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-12
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 10000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector (soft‑max)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule learning rate and smoothing
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # already includes factor 2·n

        # chain rule through the soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to break stagnation
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)

        # restart if we have been idle far too long
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing with corrected convolution update
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fine‑grained local search.  In each iteration we move a small
    amount δ from a source index to a target index and accept the move
    according to a simulated‑annealing rule.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules (very gradual)
    max_frac_start, max_frac_end = 0.8, 0.001
    temp_start, temp_end = 0.03, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 100_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source selection: mostly proportional to contribution,
        # with a fallback to pick a high‑mass index.
        if total > 0 and rng.random() < 0.85:
            src_probs = contrib / total
            i = rng.choice(n, p=src_probs)
        else:
            # pick an index with large a (to move mass away from centre)
            src_probs = a / a.sum()
            i = rng.choice(n, p=src_probs)

        if a[i] <= eps:
            continue

        # target selection: choose from the whole vector, favouring low mass.
        # Occasionally we also pick from the mask set (indices that affect k_max).
        if rng.random() < 0.70:
            target_weights = 1.0 / (a + eps)
        else:
            # mask‑restricted but still favour low values
            target_weights = np.zeros_like(a)
            target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0
        wsum = target_weights.sum()
        if wsum > 0:
            j = rng.choice(n, p=target_weights / wsum)
        else:
            # fallback uniform
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast O(n) convolution update – corrected quadratic term (‑2·δ²)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")

        else:
            no_improve += 1

        # every few thousand iterations recompute the convolution exactly
        if iteration % 8000 == 0:
            b = _full_convolution(a)

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation (it={iteration}); re‑perturb")
            a = best_a.copy()
            a += rng.normal(scale=0.03, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The step‑size decays linearly from 0.30 to 1e‑5.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to break flat plateaus
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Final simulated‑annealing with larger moves
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing loop that performs random source→target
    transfers with a relatively large step size.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)            # linear cooling
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        # choose j ≠ i uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast O(n) update (corrected quadratic term)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # tiny jitter to keep diversity
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val


# ----------------------------------------------------------------------
# Blending helper (convex combination with uniform)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a plain list whose sum is 1 and which (tries to) minimise
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # little safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives the optimiser more freedom.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # 0️⃣  Initial random / handcrafted candidates
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # try to re‑use a previously discovered solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # many random Beta draws
    for _ in range(2500):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # mixed Beta draws
    for _ in range(800):
        cand = _random_mixed_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Dirichlet draws with several concentration parameters
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(30):
            cand = _random_dirichlet_candidate(n, rng, alpha)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # hand‑crafted shapes
    for cand_func in (_sinusoid_candidate,
                     _cosine_square_candidate,
                     _linear_ramp_candidate,
                     _arcsine_candidate):
        cand = cand_func(n)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] best after random search = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – Hill‑climbing (≈40 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.70 * budget_s
    best_a, best_val = _hillclimb(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        rng=rng,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Sub‑gradient polishing (remaining time before final anneal)
    # ------------------------------------------------------------------
    # reserve a small slice for the final anneal
    subgrad_deadline = deadline - 30.0   # leave ~30 s for the last phase
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=subgrad_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=500_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase D – Final simulated‑annealing with larger moves
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_anneal(
            best_a, n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            rng=rng,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation (the official score)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
