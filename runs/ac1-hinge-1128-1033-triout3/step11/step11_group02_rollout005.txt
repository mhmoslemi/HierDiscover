"""
Propose a step‑function with a very low evaluation score.

The algorithm has three stages:

1️⃣  **Diverse initial seeds** – the best Beta‑family member, a few
    random Dirichlet draws, a Gaussian‑shaped seed, a triangular seed
    and a sine‑powered seed.  If a previous solution (“height_sequence_1”)
    exists it is interpolated as well.

2️⃣  **Smooth‑max Adam optimisation** – minimises 2·n·max(conv(a,a))
    using a log‑sum‑exp surrogate.  The temperature λ starts at 0.5 and
    grows to ~200, i.e. the surrogate becomes almost the exact max.
    Adam updates are projected onto the probability simplex after each
    step.

3️⃣  **Exact‑max projected sub‑gradient descent** – after the smooth
    phase we switch to the true sub‑gradient (λ → ∞) and continue with
    Adam‑style updates for a final refinement.

All phases respect the global 1000 s budget; the best sequence found
is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x*x (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power‑of‑two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, Σx=1} using the algorithm of
    Wang & Carreira‑Perpinán (2013)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameter‑family seeds
# ----------------------------------------------------------------------
def _beta_family_best(n: int):
    """Search the two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 14.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _gaussian_seed(n: int, sigma: float):
    """Gaussian bell centred in the middle."""
    xs = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
    a /= a.sum()
    return a


def _triangular_seed(n: int):
    """Linear ramp up then down (the classic triangular distribution)."""
    half = n // 2
    inc = np.linspace(0.0, 1.0, half + 1, endpoint=True)
    dec = np.linspace(1.0, 0.0, n - half - 1, endpoint=False)
    a = np.concatenate([inc, dec])
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _sinusoidal_seed(n: int, power: float = 1.0):
    """a_i ∝ sin^{power}(π·(i+1)/(n+1))."""
    xs = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * xs / (n + 1)) ** power
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + λ continuation)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     lam_max: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate of max(conv(a,a)).
    λ grows from 0.5 to ``lam_max`` (default 200).  After each update the
    vector is projected onto the simplex.
    """
    a = np.maximum(a0, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.07
    t = 0

    size_c = 1 << ((2 * n - 2).bit_length())       # for a*a
    size_w = 1 << ((3 * n - 3).bit_length())       # for a_rev * w

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        # λ schedule – smooth early, almost exact later
        prog = (now - start_time) / (deadline - start_time)
        lam = 0.5 + (lam_max - 0.5) * prog

        # ---- a*a ----------------------------------------------------
        fa = np.fft.rfft(a, size_c)
        conv = np.fft.irfft(fa * fa, size_c)[:2 * n - 1]
        max_c = conv.max()

        # ---- smooth weights -----------------------------------------
        exps = np.exp(lam * (conv - max_c))          # ≤1, stable
        w = exps / exps.sum()                        # length 2·n‑1, sum = 1

        # ---- gradient (log‑sum‑exp) ---------------------------------
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, size_w)
        fw = np.fft.rfft(w, size_w)
        conv_aw = np.fft.irfft(fa_rev * fw, size_w)[:3 * n - 2]
        grad = 2.0 * conv_aw[n - 1:n - 1 + n]

        # ---- Adam update --------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = base_lr * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:                     # extreme fallback
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        # ---- keep the best -----------------------------------------
        cur_val = 2.0 * n * max_c
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[Smooth λ={lam:.1f} step={t}] new best = {best_val:.6f}")

        # occasional random shake‑up to escape plateaus
        if t % 800 == 0:
            if cur_val >= best_val * 0.9999:
                noise = np.random.normal(scale=0.015, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[Smooth step={t}] stagnation – perturb")

        # early exit if we have already reached the target
        if best_val <= 1.503:
            break

    return best_a, best_val

# Exact‑max (sub‑gradient) optimiser – λ → ∞
# ----------------------------------------------------------------------
def _exact_max_optimize(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the true max(conv(a,a)).
    The gradient is built from the exact active set (indices where the
    convolution attains its maximum).
    """
    a = np.maximum(a0, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.05
    t = 0

    size_w = 1 << ((3 * n - 3).bit_length())

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        conv = _full_convolution(a)
        max_c = conv.max()
        # active set – all indices within a tiny tolerance of the max
        active = np.where(conv >= max_c - 1e-12)[0]
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(1, active.size)

        # gradient = 2·(a_rev * w) restricted to n central entries
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, size_w)
        fw = np.fft.rfft(w, size_w)
        conv_aw = np.fft.irfft(fa_rev * fw, size_w)[:3 * n - 2]
        grad = 2.0 * conv_aw[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = base_lr * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        cur_val = 2.0 * n * max_c
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[Exact step={t}] new best = {best_val:.6f}")

        # simple stagnation detector
        if t % 1500 == 0:
            # if nothing improved for a while, jitter a little
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)

        if best_val <= 1.503:
            break

    return best_a, best_val
