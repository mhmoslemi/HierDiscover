"""
Multi‑resolution smoothed‑max Adam optimiser with final hill‑climbing.

- Starts from a coarse grid (N=2000) and progressively refines to
  N=4000 → N=8000 using linear interpolation of the current best shape.
- At each resolution a soft‑max (logits) parameterisation guarantees a≥0
  and Σa=1.
- The objective max(a∗a) is approximated by a log‑sum‑exp with a
  temperature λ that grows from a small to a large value; its gradient
  is computed with FFT‑based cross‑correlation and fed to Adam.
- Learning‑rate and λ follow schedules; after a long period without
  improvement a small Gaussian perturbation is added to the logits.
- After the last resolution a cheap stochastic hill‑climb moves small
  amounts of mass between two positions, accepting only improving moves
  (with a tiny acceptance probability for occasional uphill steps).
- The best sequence found during the whole run is returned.

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself using real FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k-i}  for i = 0 … len(a)-1.
    This is a cross‑correlation and is obtained via FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = len(w) + n - 1          # = (2n-1) + n-1 = 3n-2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]

                best_a_global = a.copy()
                improve_counter = 0

            # ----------------------------------------------------------

            # ----------------------------------------------------------

            g = 2.0 * g_raw

            # back‑prop through soft‑max

            grad_z = a * (g - dot_ag)

            # ----------------------------------------------------------
            # Adam learning‑rate schedule
            # ----------------------------------------------------------

            # Adam update

            m_hat = m / (1 - b1 ** it)

            # ----------------------------------------------------------

                noise = np.random.normal(scale=0.03, size=n)
                z += noise
                improve_counter = 0
                print(f"[stage {stage_idx}] reset after {IMPROVE_LIMIT} non‑improving it (t={now - t_start:.1f}s)")

        # remember for next stage
        prev_a = a.copy()
        prev_n = n

    # --------------------------------------------------------------
    # Final stochastic hill‑climbing refinement on the best candidate
    # --------------------------------------------------------------
    a = best_a_global.copy()
    n = a.shape[0]
    hill_deadline = t_start + budget_s - 0.6    # tiny safety margin

    while time.time() < hill_deadline:
        # pick two distinct positions
        i, j = np.random.choice(n, 2, replace=False)
        # move a small fraction of mass from i to j
        if a[i] <= 0:
            continue
        frac = np.random.uniform(0.01, 0.04)    # relative step size
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate
        conv = _full_convolution(a_new)
        cur_val = 2.0 * n * conv.max()

        if cur_val < best_val_global - 1e-12:
            best_val_global = cur_val
            best_a_global = a_new.copy()
            a = a_new
            print(f"[hill] improved to {cur_val:.9f} (t={time.time() - t_start:.1f}s)")
        else:
            # occasional uphill acceptance to escape tiny plateaux
            if np.random.rand() < 0.0005:
                a = a_new

    # ------------------------------------------------------------------
    # Final secure evaluation (only once)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a_global.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                       # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return the best sequence found (as a plain Python list)
    return best_a_global.tolist()

                print(f"[stage {stage_idx}] reset after {IMPROVE_LIMIT} non‑improving its (t={now - t_start:.1f}s)")

        # remember the solution for the next resolution
        prev_a = a.copy()
        prev_n = n

    # ------------------------------------------------------------------
    # Final stochastic hill‑climbing refinement on the best candidate
    # ------------------------------------------------------------------
    a = best_a_global.copy()
    n = a.shape[0]
    hill_deadline = t_start + budget_s - 0.6   # tiny safety margin for the final call

    while time.time() < hill_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0.0:
            continue
        # move a small fraction of mass from i to j
        frac = np.random.uniform(0.01, 0.04)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv = _full_convolution(a_new)
        cur_val = 2.0 * n * conv.max()

        if cur_val < best_val_global - 1e-12:
            best_val_global = cur_val
            best_a_global = a_new.copy()
            a = a_new
            print(f"[hill] improved to {cur_val:.9f} (t={time.time() - t_start:.1f}s)")
        else:
            # very occasional uphill acceptance to avoid tiny plateaux
            if np.random.rand() < 0.0005:
                a = a_new

    # ------------------------------------------------------------------
    # Final secure evaluation (the only place we call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a_global.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return the best sequence discovered (as a plain Python list)
    return best_a_global.tolist()
