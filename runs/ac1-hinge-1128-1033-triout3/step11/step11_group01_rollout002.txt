"""
Uniform‑step optimiser.

The evaluation function is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the full linear convolution.  For any non‑negative
vector `a` the maximum convolution entry is attained at zero lag,
i.e. `max(conv) = Σ_i a_i²`.  By Cauchy‑Schwarz

    Σ_i a_i² ≥ (Σ_i a_i)² / n

with equality iff all entries are equal.  Consequently

    score ≥ 2 * n * (Σ_i a_i)² / (n * (Σ_i a_i)²) = 2

and the lower bound 2 is attained by a uniform distribution.
Therefore the globally optimal step‑function (under the given
constraints) is simply a constant vector.  The function below
returns such a vector of a configurable length (default 8000).  No
iteration or heavy computation is required, guaranteeing that the
budget of 1000 s is never exceeded.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     length: int = 8000,
                     **kwargs) -> list[float]:
    """
    Return a uniform non‑negative step function of the requested length.

    Parameters
    ----------
    seed : int, optional
        Ignored – kept for API compatibility.
    budget_s : int, optional
        Ignored – the algorithm finishes instantly.
    length : int, optional
        Number of steps in the returned sequence.  Any positive integer
        works; we use 8000 by default to match the previous experiments.
    **kwargs
        Additional arguments are ignored.

    Returns
    -------
    List[float]
        A list of ``length`` floats, each equal to ``1/length`` so that
        the sum of the sequence is exactly 1.  This yields the minimal
        possible score of 2 for the evaluation function.
    """
    # The seed is not required because the construction is deterministic,
    # but we keep the signature unchanged.
    _ = seed, budget_s, kwargs   # silence unused‑variable warnings

    # Uniform heights – the optimal solution according to the theoretical bound.
    uniform_value = 1.0 / length
    sequence = [uniform_value] * length

    # A tiny sanity check (optional, harmless for the competition).
    # It guarantees that the sequence respects the non‑negativity and
    # sum‑to‑one requirements.
    total = sum(sequence)
    if total <= 0.0:
        raise RuntimeError("Uniform construction produced a non‑positive sum.")

    # Log progress – useful for debugging on the platform.
    start = time.time()
    print(f"[uniform] generated {length}‑length uniform sequence "
          f"(value={uniform_value:.6g}), sum={total:.6g}, "
          f"time={time.time() - start:.6f}s")
    return sequence
