"""
Multi‑phase optimiser for the step‑function problem.

The algorithm works on the probability simplex (the vector is normalised to
unit ℓ₁‑norm).  It consists of:

0️⃣  a rich initialisation pool (uniform, a few hand‑crafted V‑shapes,
    several Beta‑shapes and many random Dirichlet draws).

1️⃣  an Adam optimiser on a smooth LogSumExp approximation of the
    max‑convolution (ε annealed from 5·10⁻¹ to 10⁻¹²).

2️⃣  an aggressive hill‑climbing / simulated‑annealing phase that moves
    a controlled fraction of the mass from indices that contribute most
    to the current maximal convolution entry to low‑contributing indices.

3️⃣  a projected sub‑gradient polishing phase with a linear step‑size
    schedule that finishes the remaining time budget.

4️⃣  a final “fine‑Adam’’ pass with a tiny learning‑rate and an
    ε‑schedule that reaches 10⁻¹⁴ – this often squeezes the last few
    hundredths out of the objective.

All phases stop as soon as the global 1000 s deadline is reached.  The
best vector seen during the whole run is returned as a plain list of
floats; its ℓ₁‑norm is 1, all entries are ≥0.  The final score is printed
using the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – convolutions, objective, projection, etc.
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max(conv(a,a)) using log‑sum‑exp
    and its gradient (incl. the factor 2·n).
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # gradient: derivative of sum_i a_i a_{k-i} w.r.t. a_j is
    #   a_{k-j} + a_{k-j}  (two symmetric terms) -> 2 * a_{k-j}
    # The smooth max weight p_k distributes this contribution.
    p_rev = p[::-1]                              # reverse for convolution
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # the central n entries of conv_ab correspond to the required sum
    segment = conv_ab[n - 1:2 * n - 1]            # n values
    grad_a = 4.0 * n * segment[::-1]             # factor 2·n already included
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen, but fall back to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parameterised families used for initial candidates
# ----------------------------------------------------------------------
def _beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Discrete Beta‑shape a_i ∝ (i+1)^{p‑1} (n‑i)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _valley_shape(n: int, power: float = 1.0) -> np.ndarray:
    """V‑shaped distribution that is zero in the centre and grows towards the ends."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    v = np.abs(i - centre) ** power
    # add a tiny constant to avoid exact zeros (helps numerical stability)
    v += 1e-12
    v /= v.sum()
    return v


def _sine_valley_shape(n: int, power: float = 1.0) -> np.ndarray:
    """Shifted sine that is low in the middle and large at the borders."""
    i = np.arange(n, dtype=np.float64)
    s = 1.0 - np.sin(np.pi * i / (n - 1))
    s = np.maximum(s, 0.0) ** power
    s += 1e-12
    s /= s.sum()
    return s


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   eps_start: float = 0.5,
                   eps_end: float = 1e-12,
                   lr_start: float = 0.20,
                   lr_end: float = 0.001) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smoothed max‑convolution objective."""
    # work in log‑softmax space to keep positivity and sum‑to‑1 automatically
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 12000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode accounting for softmax invariance
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # linear schedules (in log‑space for the learning‑rate)
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient (already contains 2·n factor)
        _, grad = _smooth_max_and_grad(a, eps)

        # back‑prop through the soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.025, size=n)

        # restart if we have been idle for too long
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """
    Aggressive source→target mass transfer with an annealing acceptance rule.
    The move size (fraction of the source mass) shrinks linearly with
    the remaining budget.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule parameters
    max_frac_start, max_frac_end = 0.85, 0.001
    temp_start, temp_end = 0.04, 1e-12

    no_improve = 0
    IMPROVE_LIMIT = 120000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # locate the current maximal convolution entry (might be several)
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # pick a source proportional to its contribution
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # pick a target that is currently under‑represented in the peak
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback: any other index
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move (fraction of source)
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update – O(n)
        b_new = b.copy()
        # the convolution of a with itself can be updated by adding/subtracting
        # 2·δ·a at the shifted positions and the little quadratic corrections
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # re‑perturb around the current best to escape plateaus
            print(f"[Hill] stagnation at it={iteration}, re‑perturb")
            a = best_a.copy()
            a += np.random.normal(scale=0.04, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    A linear step‑size schedule from 0.30 down to 1e‑5 is used.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear schedule – progress based on elapsed wall‑clock
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-14:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to leave flat regions
        if np.random.rand() < 0.004:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Test convex combinations with the uniform distribution."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


def _final_fine_adam(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float) -> tuple[np.ndarray, float]:
    """
    Very fine Adam pass (ε down to 1e‑14, learning‑rate down to 1e‑5).

    # reuse the generic routine with tighter schedules
    return _adam_optimize(

        best_val=best_val,
        eps_start=1e-6,
        eps_end=1e-14,
        lr_start=0.005,
        lr_end=1e-5,
    )

            best_a=best_a,
            best_val=best_val,

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------

        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # return plain python list
    return best_a.tolist()

2️⃣  an aggressive hill‑climbing / simulated‑annealing phase that moves
    a controlled fraction of the mass from indices that contribute most
    to the current maximal convolution entry to low‑contributing indices.

3️⃣  a projected sub‑gradient polishing phase with a linear step‑size
    schedule that finishes the remaining time budget.

4️⃣  a final “fine‑Adam’’ pass with a tiny learning‑rate and an
    ε‑schedule that reaches 10⁻¹⁴ – this often squeezes the last few
    hundredths out of the objective.

All phases stop as soon as the global 1000 s deadline is reached.  The
best vector seen during the whole run is returned as a plain list of
floats; its ℓ₁‑norm is 1, all entries are ≥0.  The final score is printed
using the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – convolutions, objective, projection, etc.
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max(conv(a,a)) using log‑sum‑exp
    and its gradient (incl. the factor 2·n).
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # gradient of the smooth‑max (the factor 2·n is already folded in)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # central n entries
    grad_a = 4.0 * n * segment[::-1]
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parameterised families used for initial candidates
# ----------------------------------------------------------------------
def _beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Discrete Beta‑shape a_i ∝ (i+1)^{p‑1} (n‑i)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _valley_shape(n: int, power: float = 1.0) -> np.ndarray:
    """V‑shaped distribution that is zero in the centre and grows towards the ends."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    v = np.abs(i - centre) ** power
    v += 1e-12
    v /= v.sum()
    return v


def _sine_valley_shape(n: int, power: float = 1.0) -> np.ndarray:
    """Shifted sine that is low in the middle and high at the borders."""
    i = np.arange(n, dtype=np.float64)
    s = 1.0 - np.sin(np.pi * i / (n - 1))
    s = np.maximum(s, 0.0) ** power
    s += 1e-12
    s /= s.sum()
    return s


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   eps_start: float = 0.5,
                   eps_end: float = 1e-12,
                   lr_start: float = 0.20,
                   lr_end: float = 0.001) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smoothed max‑convolution objective."""
    w = np.log(a0 + 1e-12)               # log‑softmax representation

    beta1, beta2 = 0.90, 0.999
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 12000

    while True:
        now = time.time()
        if now >= deadline:
            break

        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        prog = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** prog
        eps = eps_start * (eps_end / eps_start) ** prog

        _, grad = _smooth_max_and_grad(a, eps)

        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.025, size=n)

        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """
    Aggressive source→target mass transfer with an annealing acceptance rule.
    The move size shrinks linearly with the remaining budget.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.85, 0.001
    temp_start, temp_end = 0.04, 1e-12

    no_improve = 0
    IMPROVE_LIMIT = 120000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation at it={iteration}, re‑perturb")
            a = best_a.copy()
            a += np.random.normal(scale=0.04, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    Linear step‑size schedule from 0.30 down to 1e‑5.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-14:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        if np.random.rand() < 0.004:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Test convex combinations with the uniform distribution."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


def _final_fine_adam(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float) -> tuple[np.ndarray, float]:
    """
    Ultra‑fine Adam pass (ε down to 1e‑14, tiny learning‑rate) to
    extract the last fractional improvement.
    """
    return _adam_optimize(
        a0, n,
        deadline=deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        eps_start=1e-6,
        eps_end=1e-14,
        lr_start=0.005,
        lr_end=1e-5,
    )

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – keep the same dimension as the previous best.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Rich initial pool
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)                # uniform start
    best_val = _objective(best_a)

        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # hand‑crafted shapes
    hand_shapes = [
        _valley_shape(n, power=1.0),
        _valley_shape(n, power=2.0),
        _sine_valley_shape(n, power=1.0),
        _sine_valley_shape(n, power=2.0),
    ]
    for cand in hand_shapes:
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] hand‑crafted shape improved to {best_val:.6f}")

    # random Beta draws
    for _ in range(2500):
        logp = rng.uniform(-9.5, 2.5)
        logq = rng.uniform(-9.5, 2.5)
        p, q = np.exp(logp), np.exp(logq)
        cand = _beta_shape(n, p, q)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Dirichlet draws
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(40):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    print(f"[Init] best after random pool = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam optimisation (≈20 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.20 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        eps_start=0.5,
        eps_end=1e-12,
        lr_start=0.20,
        lr_end=0.001,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase B – Aggressive hill‑climbing (≈55 % of the budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * budget_s
    best_a, best_val = _hillclimb(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase C – Sub‑gradient polishing (remaining time minus a tiny final‑Adam)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    final_adam_budget = max(0.0, 0.10 * remaining)
    polish_deadline = time.time() + remaining - final_adam_budget
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=polish_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=400_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase D – Final ultra‑fine Adam pass (≈10 % of the total budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_val = _final_fine_adam(
            best_a, n,
            deadline=final_deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
