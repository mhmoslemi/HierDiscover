"""
search_candidate.py – Simple global‑parameter search + local refinement.

The algorithm searches over a two‑parameter Beta family

    a_i ∝ i^{p‑1} (n‑i+1)^{q‑1},

which is exactly the family used in the reference implementation.
The maximum of the evaluation function is a smooth function of (p,q), so we
use a cheap Nelder‑Mead optimiser from ``scipy.optimize`` to find a good
pair of parameters.  The optimiser works on the logarithms of the
parameters to keep them strictly positive.

After the best (p,q) is found we perform a short hill‑climbing phase.
Small amounts of mass are moved from indices that contribute heavily to
the current maximal convolution entry to indices that contribute little.
Every move is accepted only if it strictly reduces the score; the step
size is slowly annealed so that the search concentrates on fine adjustments.

All sequences are normalised to sum‑to‑one before the evaluation; the
`evaluate_sequence` function (provided by the harness) therefore returns
`2·n·max(conv)`.  The final sequence is returned as a plain list of
floats and respects the non‑negative, ≤1000 bound enforced by the
evaluation routine.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution x*x (length 2·len(x)-1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())           # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a normalised sequence (sum = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Generate a Beta‑family sequence for given (p,q)
# ----------------------------------------------------------------------
def _beta_sequence(n: int, p: float, q: float) -> np.ndarray:
    """Return the normalised Beta family sequence a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, p - 1.0) * np.power(n - i + 1.0, q - 1.0)
    s = a.sum()
    if s == 0.0:
        # fallback – return uniform (should never happen for positive p,q)
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


# ----------------------------------------------------------------------
# Local hill‑climbing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               deadline: float,
               start_time: float,
               best_val: float,
               best_a: np.ndarray) -> tuple[np.ndarray, float]:
    """Simple hill‑climber that transfers a small amount of mass."""
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = best_val
    # annealing schedule for the maximal transfer fraction
    max_frac_start, max_frac_end = 0.5, 0.001
    while time.time() < deadline:
        # progress based on wall‑clock
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # choose a source index proportional to its contribution to the current max entry
        k_max = int(np.argmax(b))
        # which indices i contribute to conv(k_max)? i must satisfy 0≤i≤n‑1 and 0≤k_max‑i≤n‑1
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        contrib = a[i_low:i_high + 1] * a[k_max - np.arange(i_low, i_high + 1)]
        total_contrib = contrib.sum()
        if total_contrib == 0.0:
            # fall back to a random source
            i_src = np.random.randint(n)
        else:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(np.arange(i_low, i_high + 1), p=src_probs)

        # pick a target different from source (uniformly)
        j_tgt = np.random.randint(n - 1)
        if j_tgt >= i_src:
            j_tgt += 1

        # amount to move (never exceeds max_frac·a[i_src])
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # update a and recompute convolution only when necessary
        a[i_src] -= delta
        a[j_tgt] += delta
        b = _full_convolution(a)
        new_val = 2.0 * n * b.max()
        if new_val < cur_val - 1e-15:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
        else:
            # move was not improving – revert
            a[i_src] += delta
            a[j_tgt] -= delta
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8000 entries gives a good trade‑off between quality
    # and speed.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Simple deterministic seeds (uniform, linear, quadratic …)
    # ------------------------------------------------------------------
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # linear decreasing (p=1, q=2)
    seeds.append(_beta_sequence(n, p=1.0, q=2.0))

    # quadratic decreasing (p=1, q=3)
    seeds.append(_beta_sequence(n, p=1.0, q=3.0))

    # cubic decreasing (p=1, q=4)
    seeds.append(_beta_sequence(n, p=1.0, q=4.0))

    # a few random Dirichlet draws (give them a chance)
    for _ in range(5):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 1️⃣  Global search over (p,q) using Nelder‑Mead on log‑parameters
    # ------------------------------------------------------------------
    # we keep the best sequence found so far
    best_a = None
    best_val = np.inf

    # helper: objective for given log‑parameters (logp, logq)
    def _logpq_obj(logpq):
        p, q = np.exp(logpq)               # enforce positivity
        a = _beta_sequence(n, p, q)
        return _objective(a)

    # initialise the optimiser around the linear decreasing shape
    x0 = np.log(np.array([1.0, 2.0]))      # (log p, log q)
    opt_res = minimize(_logpq_obj,
                       x0,
                       method='Nelder-Mead',
                       options={'maxiter': 150,
                                'fatol': 1e-8,
                                'xatol': 1e-8,
                                'disp': False})
    if opt_res.success:
        p_opt, q_opt = np.exp(opt_res.x)
        a_opt = _beta_sequence(n, p_opt, q_opt)
        val_opt = _objective(a_opt)
        print(f"[NelderMead] opt p={p_opt:.4f}, q={q_opt:.4f} → val={val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()

    # ------------------------------------------------------------------
    # 2️⃣  Evaluate the deterministic seeds (they are cheap)
    # ------------------------------------------------------------------
    for idx, a in enumerate(seeds):
        if time.time() >= deadline:
            break
        val = _objective(a)
        print(f"[seed {idx}] deterministic val = {val:.6f}")
        if val < best_val:
            best_val, best_a = val, a.copy()
        if best_val <= 1.503:               # early exit if we already beat the target
            break

    # ------------------------------------------------------------------
    # 3️⃣  Randomised Beta‑family sweep (quick but broad)
    # ------------------------------------------------------------------
    # We prune the sweep if we run out of time.
    while time.time() < deadline and best_val > 1.503:
        # log‑uniform sampling in a sensible range
        logp = np.random.uniform(-2.0, 2.0)   # p ∈ [e^{-2}, e^{2}] ≈ [0.14, 7.4]
        logq = np.random.uniform(-2.0, 2.0)
        p, q = np.exp(logp), np.exp(logq)
        a = _beta_sequence(n, p, q)
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[rand] p={p:.3f}, q={q:.3f} → val={val:.6f}")
        # stop early once we are comfortably below the target
        if best_val <= 1.48:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final intensive hill‑climbing on the remaining time slice
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the rest of the budget to hill‑climbing
        best_a, best_val = _hillclimb(best_a,
                                      deadline=deadline,
                                      start_time=time.time(),
                                      best_val=best_val,
                                      best_a=best_a)

    # ------------------------------------------------------------------
    # 5️⃣  Final secure evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
