assistantfinal**Solution Overview**

The code below searches for a non‑negative step‑function `a = (a₀,…,a_{n‑1})` that minimises  

\[
\text{evaluate\_sequence}(a)=
\frac{2\,n\;\max\bigl(a*a\bigr)}{\bigl(\sum a\bigr)^{2}}
\]

The problem is scale‑invariant, therefore the algorithm works on the simplex
`∑ a = 1`.  
Our strategy consists of three phases

1. **Multi‑start Adam optimisation** – several different initial shapes
   (Beta, Gaussian, uniform, and the previous best if available) are
   optimised with Adam on a *soft‑max* surrogate of the max‑convolution.
   The smoothing parameter `α` is increased from `1` to `≈400` so that the
   landscape becomes increasingly sharp.

2. **Exact‑max polishing** – the best candidate from phase 1 is refined
   with exponentiated‑gradient mirror descent on the *exact* maximum
   entry of the convolution (`_max_refine`).  Up to 4000 steps are taken,
   decreasing the step size gradually.

3. **Local stochastic improvement** – a cheap hill‑climbing scheme moves a
   small amount of mass from one index to another.  Only moves that
   strictly decrease the exact objective are kept.  This phase explores the
   neighbourhood of the polished solution for a few seconds.

All heavy work (full convolutions) is performed with FFTs (`O(n log n)`);
for `n = 8000` a single convolution costs < 1 ms, so a large number of
iterations fits easily into the 1000 s budget.

The final sequence is returned as a plain list of Python `float`s and is
checked with the provided `evaluate_sequence` wrapper.

```python
"""
Multi‑start Adam + exact‑max mirror descent + stochastic local search
to minimise evaluate_sequence(seq).  Works on the probability simplex
(∑seq = 1) and respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (exact, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two different vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact‑max mirror‑descent polishing (exponentiated gradient on the true max)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray,
                steps: int = 4000,
                eta_start: float = 0.6) -> np.ndarray:
    """
    Entropy‑mirror descent on the exact max entry of the self‑convolution.
    a must be a probability vector (sum = 1).
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of max entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]          # 2 * partner entry
        eta = eta_start * (1.0 - t / steps)    # linear decay
        a = a * np.exp(-eta * g)               # exponentiated update
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Helper – exact objective (sum(a)=1) for a probability vector
# ----------------------------------------------------------------------
def _exact_objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) (the denominator is 1 because sum(a)=1)."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   time_limit: float,
                   start_lr: float = 0.20,
                   end_lr: float = 0.005,
                   alpha_end: float = 400.0,
                   jitter_amp: float = 0.03,
                   stagnation_limit: int = 2000) -> tuple[np.ndarray, float]:
    """
    Perform Adam on the smooth approximation.
    Returns (best_a, best_obj) where best_obj is the *exact* objective.
    """
    eps = 1e-30
    z = np.log(init_a + eps)            # logits, guarantee positivity
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = None
    best_obj = np.inf
    stagnation = 0
    it = 0
    start = time.time()
    deadline = start + time_limit

    while time.time() < deadline:
        # progress fraction (0 → 1)
        prog = (time.time() - start) / (deadline - start)

        # learning‑rate and smoothing schedule
        lr = start_lr * (1.0 - prog) + end_lr * prog
        alpha = 1.0 + (alpha_end - 1.0) * (prog ** 1.3)   # a little slower at start

        # recover a from logits and enforce exact sum‑to‑1
        a = np.exp(z)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # full self‑convolution
        conv = _full_convolution(a)                      # length 2·n‑1

        # soft‑max of the convolution (numerically stable)
        maxc = conv.max()
        exp_scaled = np.exp(alpha * (conv - maxc))
        sumexp = exp_scaled.sum()
        weights = exp_scaled / sumexp                    # Σ weights = 1
        log_sumexp = maxc + np.log(sumexp)

        # smooth surrogate objective (not the exact one)
        smooth_obj = 2.0 * n * (log_sumexp / alpha)

        # keep the exact best seen so far
        exact_obj = 2.0 * n * conv.max()
        if exact_obj < best_obj - 1e-12:
            best_obj = exact_obj
            best_a = a.copy()
            stagnation = 0
            print(f"[adam] new best O={best_obj:.6f}  it={it}  t={time.time()-start:.1f}s")
        else:
            stagnation += 1

        # gradient of smooth objective w.r.t. a
        rev_a = a[::-1]
        conv_rev = _full_convolution_two(rev_a, weights)       # length 3·n‑2
        g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]               # slice length n

        # chain rule through softmax (a = softmax(z))
        dot = np.dot(g_a, a)
        grad_z = a * (g_a - dot)

        # Adam update on logits
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # escape from plateaus
        if stagnation >= stagnation_limit:
            z += np.random.normal(scale=jitter_amp, size=n)
            stagnation = 0
            print(f"[adam] jitter added at it={it}")

    # if nothing was ever stored (should not happen) fall back to init
    if best_a is None:
        best_a = init_a.copy()
        best_obj = _exact_objective(best_a, n)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing around a given solution
# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  time_limit: float) -> tuple[np.ndarray, float]:
    """
    Small random perturbations that keep the sum fixed.
    Only moves that improve the exact objective are accepted.
    """
    start = time.time()
    best_a = a.copy()
    best_obj = _exact_objective(best_a, n)

    # pre‑compute indices to avoid “zero‑probability’’ moves
    idxs = np.arange(n, dtype=int)

    while time.time() - start < time_limit:
        # pick source and destination indices (biased towards larger entries)
        i = np.random.choice(idxs, p=best_a / best_a.sum())
        j = np.random.choice(idxs)

        if i == j or best_a[i] < 1e-12:
            continue

        # move a small fraction of mass from i to j
        delta = 0.02 * best_a[i]          # 2 % of source mass
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # exact objective
        obj = _exact_objective(a_new, n)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new
            # occasional log
            if int(time.time() - start) % 30 == 0:
                print(f"[local] improved to O={best_obj:.6f}")

    return best_a, best_obj
```
