            if cur_val < best_val:
                best_val = cur_val

                    break
        if best_val <= 1.503:
            break

    if best_a is None:
        # fallback uniform
        n = n_candidates[0]
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)

    # final evaluation
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score =", final_score)
    except Exception as e:
        print("[final] evaluation error:", e)

    return best_a.tolist()

"""
Sidon‑set based construction for the step‑function attack.

The evaluation metric is
    2·n·max(a∗a) / (sum(a)²)

For a vector with non‑negative entries that sum to 1,
the quantity max(a∗a) equals the largest entry of the linear
convolution a∗a.  If the set of indices where a_i>0 forms a
*Sidon set* (all unordered sums i+j are distinct) then each index
of the convolution receives at most the contribution

    a_i²               (i=j)   → 1/m²
    a_i·a_j + a_j·a_i (i≠j)   → 2/m²

where m is the number of non‑zero entries.
Thus

    max(a∗a) = 2 / m²   (m≥2)
    objective = 4·n / m².

To beat the target 1.503 we need   m ≥ sqrt(4·n/1.503).
A greedy random construction of a Sidon set quickly finds
sets of size ≈ 2·√n, which is enough for n≈8 000–16 000.

The algorithm:
  * pick a length n,
  * repeatedly generate a random ordering of {0,…,n‑1},
    greedily add a position if none of its unordered sums
    clash with already used sums,
  * keep the best set (largest m) found before the time budget expires,
  * build the step function with weight 1/m on the selected positions,
    evaluate the exact score and return the best sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the exact objective used by the host
# ----------------------------------------------------------------------
def _objective_val(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a (non‑negative)."""
    conv = np.convolve(a, a)          # full linear convolution
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Greedy construction of a Sidon set
# ----------------------------------------------------------------------
def _generate_sidon_set(n: int, rng: np.random.Generator) -> list[int]:
    """
    Random greedy construction of a Sidon set S⊂{0,…,n‑1}.
    The set satisfies i+j≠k+ℓ for any unordered pairs
    {i,j}≠{k,ℓ} (including the case i=j).

    Returns the list of selected indices.
    """
    used_sums = set()      # already occupied unordered sums i+j
    S = []

    # random permutation gives many different candidates
    order = rng.permutation(n)

    for x in order:
        # check whether x would create a duplicate sum
        conflict = False
        for y in S:
            if (x + y) in used_sums:
                conflict = True
                break
        if conflict or (2 * x) in used_sums:
            continue

        # accept x
        for y in S:
            used_sums.add(x + y)
        used_sums.add(2 * x)
        S.append(int(x))

    # a second pass can sometimes squeeze a few more elements
    for x in order:
        if x in S:
            continue
        conflict = False
        for y in S:
            if (x + y) in used_sums:
                conflict = True
                break
        if conflict or (2 * x) in used_sums:
            continue
        for y in S:
            used_sums.add(x + y)
        used_sums.add(2 * x)
        S.append(int(x))

    return S

    Produce a non‑negative step function (list of floats) that aims

    budget and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # a few problem sizes – larger n allows larger Sidon sets

    best_n = None

        m_needed = int(np.ceil(np.sqrt(4.0 * n / target)))

            attempts += 1
            S = _generate_sidon_set(n, rng)
            m = len(S)

            if m < m_needed:          # cannot beat the target with this set
                continue

            # build the candidate sequence (uniform weight on the Sidon positions)
            a = np.zeros(n, dtype=np.float64)
            weight = 1.0 / m
            a[S] = weight

            # exact objective (fast, no need to call evaluate_sequence yet)
            val = _objective_val(a)

            if val < best_val - 1e-12:
                best_val = val
                best_seq = a.copy()
                best_n = n
                print(f"[update] n={n} try={attempts:3d} m={m:3d} val={val:.6f}")

                # stop early if we already beat the required threshold
                if best_val <= target:
                    break

        if best_val <= target:
            break

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (very unlikely) use a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = sizes[0]
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective_val(best_seq)
        best_n = n
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Final logging – the host will also evaluate the sequence
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
```
