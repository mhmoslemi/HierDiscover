"""
Propose a step‑function that (hopefully) improves the current best score.
The algorithm mixes three ideas:

1. **Warm‑start** – use the previously found solution if it exists,
   otherwise start from a smooth Beta(2, 2) shape (zero at the ends,
   peaked in the middle).

2. **Smoothed Adam optimisation** – the objective
   2·n·max(conv(a,a)) is approximated by a soft‑max over the full
   convolution.  The smoothing parameter α is increased from 1 to a few
   hundred, turning the landscape gradually from smooth to the true
   max.  Adam updates are performed on the log‑variables (logits) to
   keep the coefficients non‑negative and automatically enforce the
   L1‑normalisation (the simplex).

3. **Entropy‑mirror refinement** – after the Adam phase a dedicated
   mirror‑descent (exact max‑subgradient) is run for many steps with a
   decreasing step size.  This directly reduces the true maximal
   convolution entry.

The loop runs until the time budget (≈1000 s) is exhausted, logs any
improvement, adds a small random jitter when stagnation is detected,
and finally returns the best sequence found.  The returned list
contains non‑negative floats; the wrapper ``evaluate_sequence`` will
clip and normalise it automatically.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (fast, O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two (different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Mirror‑descent polishing on the exact max entry of the convolution
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray,
                steps: int = 3000,
                eta_start: float = 0.35) -> np.ndarray:
    """
    Entropy‑mirror descent that minimises the maximal entry of
    ``_full_convolution(a)`` while keeping ``a`` on the probability
    simplex.
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # decreasing step size
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a
