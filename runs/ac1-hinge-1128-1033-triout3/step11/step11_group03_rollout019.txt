"""
propose_candidate
=================

Search for a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear convolution.  The function uses a
multi‑stage strategy:

1. **Diverse one‑shot initialisations** – a collection of hand‑crafted
   parametric families (Gaussian, Laplace, power‑law, sinusoidal …) and
   a few random Dirichlet draws.  A cheap exact evaluation selects the
   best starter.

2. **Two complementary smooth optimisers**  
   *a)* a soft‑max surrogate of the ``max`` (the original code) – Adam
   optimisation with a linearly increasing temperature (α).  
   *b)* a least‑squares surrogate that tries to flatten the whole
   convolution to a constant (the ideal “uniform” sum‑distribution).  The
   loss is ``∑ (conv_k – C)^2`` with ``C = 1/(2n‑1)`` and Adam is used as well.

   Both runs are bounded only by the remaining time budget and continuously
   keep the best sequence seen (exact objective value).

3. **Fast hill‑climbing (mass‑transfer)** – small random moves that shift a
   tiny amount of mass from a “guilty” index (large contribution to the
   current maximal convolution entry) to a low‑contribution index.
   The move is accepted only if the exact objective improves;
   occasional jitter prevents stagnation.

All updates stay on the probability simplex (non‑negative, sum = 1) via an
Euclidean projection.  The routine respects a 1000 s wall‑clock budget
(the caller supplies a smaller ``budget_s`` argument).  The final
sequence is returned as a plain list of floats ready for ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
# Convolution utilities (FFT – real input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_rev(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient helper: 2·∑_i a_i·w_{i+j} for j=0..n‑1.
    `a` – current vector (length n)
    `w` – weight vector of length 2·n‑1 (e.g. soft‑max weights or residuals)
    """
    n = a.shape[0]
    a_rev = a[::-1]
    l_w = w.shape[0]
    out_len = n + l_w - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]
    start = n - 1
    return conv[start:start + n]          # length n


# ----------------------------------------------------------------------
# Objective helpers (exact, scale‑invariant)
# ----------------------------------------------------------------------
def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact value 2·n·max(a∗a) given the full convolution."""
    return float(2.0 * n * conv.max())


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (assumed non‑negative, sum>0)."""
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.shape[0])


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σ x = 1} (Euclidean)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max surrogate (used in the original approach)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(α·conv_k)."""
    maxc = conv.max()
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exp_vals / sum_exp


def _smooth_adam_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          alpha_start: float = 5.0,
                          alpha_end: float = 500.0,
                          lr0: float = 0.12,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate.
    Returns the best sequence (sum=1) and its exact objective.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_conv = _full_convolution(a)
    cur_val = _objective_from_conv(cur_conv, n)
    best_a = a.copy()
    best_val = cur_val

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear schedule for α
        prog = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        alpha = alpha_start + (alpha_end - alpha_start) * prog

        w = _softmax_weights(cur_conv, alpha)          # soft‑max over 2n‑1 entries
        grad = 2.0 * _conv_rev(a, w)                    # gradient of the surrogate

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        cand_conv = _full_convolution(a_candidate)
        cand_val = _objective_from_conv(cand_conv, n)

        # accept only if the exact objective does not increase
        if cand_val <= cur_val:
            a = a_candidate
            cur_conv = cand_conv
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[Adam] iter {t:7d} α={alpha:6.1f} val={best_val:.6f}")
        else:
            # keep momentum vectors untouched – continue searching
            pass

        # occasional re‑randomisation to escape flat regions
        if t % 8000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_conv = _full_convolution(a)
            cur_val = _objective_from_conv(cur_conv, n)

    return best_a, best_val


# ----------------------------------------------------------------------
# Least‑squares surrogate (flattens the whole convolution)
# ----------------------------------------------------------------------
def _ls_error_adam_optimize(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           lr0: float = 0.12,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the L² error   ∑ (conv_k – C)²,
    where C = 1/(2n‑1) so that the target distribution would be perfectly flat.
    Returns the best sequence (sum=1) and its exact max‑convolution objective.
    """
    C = 1.0 / (2 * n - 1)           # constant target (the average of conv)

    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_conv = _full_convolution(a)
    cur_val = _objective_from_conv(cur_conv, n)   # exact objective (max‑convolution)
    best_a = a.copy()
    best_val = cur_val

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(a)
        residual = conv - C                     # shape (2n‑1,)
        loss = np.dot(residual, residual)       # scalar L² loss (not used for acceptance)

        # gradient of loss w.r.t a: 2 * conv_rev(a, residual)
        grad = 2.0 * _conv_rev(a, residual)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        cand_conv = _full_convolution(a_candidate)
        cand_val = _objective_from_conv(cand_conv, n)

        if cand_val <= cur_val:
            a = a_candidate
            cur_conv = cand_conv
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[LS‑Adam] iter {t:7d} loss={loss:.6e} val={best_val:.6f}")
        else:
            # keep old point, momentum vectors remain to preserve direction
            pass

        # small random reshuffle every few thousand iterations
        if t % 10000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric families (much richer than the original list)
# ----------------------------------------------------------------------
def _parametric_candidates(n: int):
    """Yield a variety of hand‑crafted shapes."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # ----- Gaussian centred (bell‑shaped) -----
    for sigma_factor in np.linspace(0.12, 0.42, 7):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        a /= a.sum()
        yield a

    # ----- Laplace (double‑exponential) -----
    for scale_factor in np.linspace(0.08, 0.24, 5):
        scale = scale_factor * n
        a = np.exp(-np.abs(i - mid) / scale)
        a /= a.sum()
        yield a

    # ----- Power‑law / Beta‑like (p may be <1 for edge‑heavy) -----
    for p in np.linspace(0.08, 2.2, 12):
        a = (i + 1) ** (p - 1.0) * (n - i) ** (p - 1.0)
        a /= a.sum()
        yield a

    # ----- Cosine‑shaped (U‑shaped to emphasize ends) -----
    a = 1.0 + np.cos(np.pi * i / (n - 1))   # values in [0,2]
    a /= a.sum()
    yield a

    # ----- Sin‑squared (smooth bump) -----
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    a /= a.sum()
    yield a

    # ----- Exponential tilt (monotone) -----
    for lam in np.linspace(-4.0, 4.0, 9):
        a = np.exp(lam * (i - mid) / n)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
            yield a


def _interpolate_previous(prev_seq, n):
    """
    Resize a previously found sequence to length `n` via linear interpolation.
    The result is projected onto the simplex.
    """
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None

# Light‑weight hill‑climb (mass‑transfer) for final polishing
# ----------------------------------------------------------------------
def _final_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that repeatedly moves a small random chunk of mass
    from a high‑contributing index to a low‑contributing one.
    Stops early if `target_val` is reached.
    """
    b = _full_convolution(a)
    cur_val = _objective_from_conv(b, n)
    best_val = cur_val
    best_a = a.copy()

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        k_max = int(np.argmax(b))

        # positions that participate in this lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source: preferably a heavy contributor
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-12:
            continue

        # target: a location with low contribution
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        delta = a[src] * np.random.rand() * 0.45   # up to ~45 % of the source mass
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = _objective_from_conv(b_new, n)

        if new_val < cur_val:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # random jitter around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = _objective_from_conv(b, n)
            no_improve = 0

        iteration += 1

    return best_a, best_val
