"""
propose_candidate
-----------------

Search for a non‑negative step function minimising the provided
``evaluate_sequence``.  The search proceeds in three stages:

1️⃣  Build a set of promising starting vectors:
    – interpolated previous solution (if available),
    – a coarse grid of Beta‑shaped vectors,
    – random Beta draws,
    – Dirichlet draws with various concentration parameters.

2️⃣  Refine each start with a *smoothed* Adam optimiser.
    The original max‑convolution term
        max(a⋆a)
    is replaced by a log‑sum‑exp smoothing
        φ_β(a) = (1/β)·log Σ_k exp(β·(a⋆a)_k) .
    Its gradient is a weighted cross‑correlation, computed efficiently
    by FFTs.  The temperature β is increased slowly (β←β·β_mul) so
    that the smoothed problem approaches the true max‑convolution.
    After each Adam step the iterate is projected onto the
    probability simplex (Σa=1, a≥0).

3️⃣  Keep the best vector seen across all restarts and report it
    to ``evaluate_sequence`` before the time budget expires.

The whole routine is fully self‑contained, uses only NumPy and the
already‑imported ``evaluate_sequence`` and respects the 1000‑second
time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # numerical noise may produce tiny negatives – clip them
    return np.maximum(conv, 0.0)


def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a vector with Σa = 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex Σx=1, x≥0.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1,…,n .
    ``None`` is returned if the parameters give an all‑zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smoothed objective
        φ_β(a) = (1/β)·log Σ_k exp(β·(a⋆a)_k)

    Returns
        grad  – Euclidean gradient (same shape as ``a``)
        mx    – exact max(a⋆a) (useful for the true objective).
    """
    n = a.shape[0]

    # ---- 1. full convolution a⋆a (length 2n‑1) ----
    out_len = 2 * n - 1
    size_conv = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[:out_len]      # (a⋆a)
    mx = conv.max()

    # ---- 2. soft‑max weights w_k = exp(β·(conv_k - mx)) / Σ ----
    # shift by mx to avoid overflow
    exp_arg = beta * (conv - mx)        # ≤ 0
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum                  # Σ w = 1

    # ---- 3. gradient = 2·(a ⋆_cross w_norm) ----
    # cross‑correlation: (a ⋆ w_norm)[j] = Σ_k a_k·w_norm_{k+j}
    rev_w = w_norm[::-1]                # reverse for convolution
    out_len_corr = n + out_len - 1      # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    # Position of lag 0 in ``corr_full``:
    offset = out_len - 1
    # indices for j = 0 … n‑1
    idx = offset - np.arange(n)
    grad = 2.0 * corr_full[idx]

    return grad, mx

                        break

            # Adam update (first & second moments)
            m = adam_beta1 * m + (1.0 - adam_beta1) * grad
            v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

            # bias‑correction
            m_hat = m / (1.0 - adam_beta1 ** iteration)
            v_hat = v / (1.0 - adam_beta2 ** iteration)

            a_candidate = a - step * m_hat / (np.sqrt(v_hat) + adam_eps)

            # projection onto simplex
            a_candidate = _project_simplex(a_candidate)

            # occasional tiny jitter to help escape plateaus
            if np.random.rand() < jitter_prob:
                a_candidate += np.random.normal(scale=jitter_scale, size=n)
                a_candidate = np.maximum(a_candidate, 0.0)
                a_candidate = _project_simplex(a_candidate)

            a = a_candidate
            beta = min(beta_max, beta * beta_mul)

        # end of inner loop – we either ran out of time or hit the target.
        # Continue with another fresh start if time permits.
        if overall_best_score <= 1.503:
            # already good enough – will exit after the outer‑loop check
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation (must be called before time runs out)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return overall_best_seq.tolist()
