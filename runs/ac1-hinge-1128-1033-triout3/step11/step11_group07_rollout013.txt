def _local_search(a, n, deadline, start_time, best_a, best_val):
    cur_a = a.copy()
    cur_val = _objective(cur_a)
    # parameters
    T0 = 0.1 * cur_val
    temp = T0
    # maybe schedule factor
    while time.time() < deadline:
        # compute progress fraction
        progress = (time.time() - start_time) / (deadline - start_time)
        # temperature linear decay
        temp = T0 * (1 - progress)
        # random move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        if cur_a[i] == 0:

        delta = cur_a[i] * rng.random() * 0.5  # up to half of a_i

"""
Propose a low‑score step‑function.

The optimiser works in several stages:

1. **Diverse initialisation** – a large collection of random candidates
   (Beta‑shaped, Dirichlet, exponential) is generated and the best one
   (according to the exact objective 2·n·max(conv(a,a))) is kept.

2. **Smooth Adam optimisation** – a smoothed version of the L∞‑norm
   (log‑sum‑exp) is minimised with Adam on the log‑softmax parameters.
   The smoothing ε is annealed from 0.5 to 1e‑12 and the learning rate
   from 0.5 to 0.005.  Restarts are triggered when the optimiser stalls.

3. **Exact sub‑gradient polishing** – a projected sub‑gradient method on the
   true max‑convolution objective is run until most of the time budget is
   exhausted.

4. **Local stochastic refinement** – a lightweight simulated‑annealing
   loop performs random mass transfers between two coordinates while staying
   on the simplex.  The temperature is decreased linearly.  Occasionally the
   current vector is sorted in non‑increasing order (which tends to reduce
   the convolution peak).

Finally `evaluate_sequence` is called on the best vector found and the
vector is returned as a plain list of floats.

The whole routine runs in ≤ 1000 s on a single CPU and never returns an
empty or invalid list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1 – the factor 2·n is included)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick for numerical stability
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p (gives ∂/∂a of smooth max)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 2·n already included
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max (aggressive schedule)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation in the log‑softmax space on the smoothed objective."""
    n = a0.shape[0]
    # initialise logits (log‑softmax trick)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.005
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 2500

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam]  new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule (linear in log‑time)
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        _, grad = _smooth_max_and_grad(a, eps)   # grad already contains 2·n

        # chain rule through the soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.03:
            w += np.random.normal(scale=0.04, size=n)

        if idle > MAX_IDLE:
            # restart from a fresh random point
            print("[Adam]  restart after idle")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  f(a)=2·n·max(conv(a,a)).
    """
    a = a0.copy()
    cur_val = _objective(a)

    n = a.shape[0]

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum of contributions from all maximal diagonals
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule (more aggressive early)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.45 * (1.0 - progress) + 5e-6 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional noise to kick out of plateaus
        if it % 600 == 0 and np.random.rand() < 0.04:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Small stochastic refinement (simulated annealing)
# ----------------------------------------------------------------------
def _local_refine(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  best_a: np.ndarray,
                  best_val: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer SA on the simplex to squeeze a few more points."""
    a = a0.copy()
    cur_val = _objective(a)
    n = a.shape[0]

    # temperature schedule: start a fraction of the current value
    T0 = max(0.1 * cur_val, 1e-4)
    while True:
        now = time.time()
        if now >= deadline:
            break
        # linear cooling
        progress = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - progress)

        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] == 0.0:
            continue

        # transfer a random fraction of a[i] to a[j]
        delta = a[i] * np.random.rand() * 0.4   # up to 40 % of the source mass
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        cand_val = _objective(a_candidate)

        if cand_val < cur_val - 1e-12:
            a, cur_val = a_candidate, cand_val
            if cand_val < best_val:
                best_val, best_a = cand_val, a_candidate.copy()
                print(f"[Local] new best = {best_val:.6f}")
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(cand_val - cur_val) / max(T, 1e-12)):
                a, cur_val = a_candidate, cand_val

        # occasional sorting (helps aligning large values with small ones)
        if np.random.rand() < 0.015:
            a = np.sort(a)[::-1]   # non‑increasing
            a = _project_simplex(a)   # keep on the simplex
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[Sort]  new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Useful candidate generators
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.0, 2.2)
    logq = rng.uniform(-9.0, 2.2)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a decay factor α>0 and return exp(-α·i) normalised."""
    # sample α on a log scale; α≈log(n)/n gives the known ~8 % improvement
    log_alpha = rng.uniform(-9.0, -2.0)   # α from ~1e‑9 to ~1e‑2
    alpha = np.exp(log_alpha)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the time budget.
    Returns a plain list of floats (its sum is not required to be 1;
    ``evaluate_sequence`` handles scaling automatically).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a richer space, 8000 works well.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise – cheap diverse candidates, keep the best one.
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # 0.1  Try to reuse a previous solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[Init] Interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # 0.2  Random Beta shapes
    for _ in range(150):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] Beta candidate improved, value={best_val:.6f}")

    # 0.3  Dirichlet draws (various concentration parameters)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(6):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val
                print(f"[Init] Dirichlet(alpha={alpha}) improved, value={best_val:.6f}")

    # 0.4  Exponential families (different decay rates)
    for _ in range(80):
        cand = _exponential_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] Exponential candidate improved, value={best_val:.6f}")

    # fallback – uniform distribution
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print(f"[Init] fallback uniform, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈35 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.35 * budget_s
    best_a, best_val = _adam_optimize(
        best_a,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )

    # ------------------------------------------------------------------
    # Phase B – Projected sub‑gradient polishing (≈50 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.85 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a,
        deadline=phase_b_deadline,
        start_time=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        max_iters=400_000,
    )

    # ------------------------------------------------------------------
    # Phase C – Local stochastic refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _local_refine(
            best_a,
            deadline=deadline,
            start_time=phase_b_deadline,
            best_a=best_a,
            best_val=best_val,
        )

    # ------------------------------------------------------------------
    # Final secure evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return a plain list of floats
    return best_a.tolist()
