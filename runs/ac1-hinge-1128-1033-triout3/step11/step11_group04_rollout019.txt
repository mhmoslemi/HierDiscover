"""
Propose a low‑upper‑bound step‑function.

The optimizer works in three stages inside the 1000 s budget:

0️⃣  Warm‑start – the best of a few handcrafted shapes, a uniform
    vector, and many Dirichlet / Beta draws (including the previous
    solution if available).

1️⃣  Smooth‑max Adam – minimise a log‑sum‑exp smooth approximation of
    max conv(a,a).  The smoothing ε is annealed from 0.5 to 1e‑9 and the
    learning‑rate from 0.2 to 0.001.  Variables live in log‑softmax space
    so the ℓ₁‑norm = 1 and non‑negativity are automatic.

2️⃣  Deterministic sub‑gradient descent with back‑tracking line‑search.
    The exact (non‑smooth) objective 2·n·max(conv(a,a)) is reduced by
    repeatedly moving a along the sub‑gradient of the current maximal
    convolution entry, projecting onto the probability simplex after each
    step.  Step‑sizes are adapted automatically; occasional jitter helps
    to escape flat plateaus.

3️⃣  A short “reverse‑mix” polishing – repeatedly blend the current
    candidate with its reversed copy (a←t·a+(1‑t)·rev(a)),
    projecting onto the simplex.  This simple symmetrisation often
    reduces the peak of the convolution a little further.

The best vector found (its ℓ₁‑norm is 1 and all entries ≥ 0) is returned

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10

    q = np.exp(logq)

    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

    max_b = b.max()
    # stability shift
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries (centered)
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max objective."""
    w = np.log(a0 + 1e-12)          # logits
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 5000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits → probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedules
        prog = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** prog
        eps = eps_start * (eps_end / eps_start) ** prog

        # smooth max + gradient
        _, grad = _smooth_max_and_grad(a, eps)   # already contains 2·n

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.02, size=n)

        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# 2️⃣  Exact sub‑gradient descent with line‑search
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_a: np.ndarray,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses a back‑tracking line‑search to adapt the step size.
    """
    a = a0.copy()
    cur_val = _objective(a)

    step = 0.3                     # initial step size
    min_step = 1e-7
    max_step = 0.5
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient of the max entry (or entries) – see paper description
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # try a few step‑size reductions until improvement
        improved = False
        trial_step = step
        for _ in range(6):
            a_new = a - trial_step * grad
            a_new = _project_simplex(a_new)

            new_val = _objective(a_new)
            if new_val < cur_val - 1e-14:          # strict improvement
                a = a_new
                cur_val = new_val
                improved = True
                # enlarge step a little for the next iteration
                step = min(trial_step * 1.2, max_step)
                if new_val < best_val:
                    best_val, best_a = new_val, a.copy()
                    print(f"[SubG] new best = {best_val:.6f}")
                break
            else:
                trial_step *= 0.5

        if not improved:
            # no improvement – shrink the step size
            step = max(step * 0.5, min_step)

        # occasional jitter to escape flat regions
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# 3️⃣  Reverse‑mix polishing (convex combination with a reversed copy)
# ----------------------------------------------------------------------
def _reverse_mix_polish(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly blend the candidate with its reversed copy:
        a ← t·a + (1‑t)·rev(a)
    The blend factor t is annealed from 0.8 down to 0.02.
    After each blend we project onto the simplex.
    """
    a = a0.copy()
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        t = 0.8 * (1.0 - prog) + 0.02 * prog
        cand = t * a + (1.0 - t) * a[::-1]
        cand = _project_simplex(cand)

        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            a = cand
            print(f"[RevMix] new best = {best_val:.6f}")
        else:
            a = cand  # still move even if not better – helps escape plates

    return best_a, best_val


# ----------------------------------------------------------------------
# Small helper – blend with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray, n: int, cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand = _project_simplex(cand)
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a plain list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 8000 works well and was used in the reference code
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Warm‑start
    # --------------------------------------------------------------
    best_a, best_val = _initial_candidates(n, rng)

    # --------------------------------------------------------------
    # 1️⃣  Adam optimisation (≈20 % of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase1_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # 2️⃣  Exact sub‑gradient descent with line‑search (≈60 % of budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _subgradient_descent(
        best_a, n,
        deadline=phase2_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # 3️⃣  Final reverse‑mix polishing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _reverse_mix_polish(
            best_a, n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # return plain Python list
    return best_a.tolist()
