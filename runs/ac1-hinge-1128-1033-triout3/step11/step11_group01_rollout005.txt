    # Project vector v onto simplex {x>=0, sum x = s}
    n = len(v)

    rho = ind[cond][-1]

"""
A simple stochastic optimiser for the step‑function objective.

The routine works with a probability vector (non‑negative entries,
summed to 1).  At each iteration it either

* performs a projected gradient step based on the exact sub‑gradient
  of the current maximal autocorrelation entry, or
* makes a small random transfer of mass between two coordinates.

The step size and a simulated‑annealing temperature decay linearly
with the elapsed time; after a long period without improvement a
Gaussian perturbation is added to escape possible plateaus.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (used for FFT sizes)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Full linear convolution a*a (length 2·n‑1) via FFT
# ----------------------------------------------------------------------
def _full_convolution_max(a: np.ndarray, fft_size: int) -> float:
    """
    Compute the maximum entry of the full self‑convolution of `a`
    using a real FFT of length `fft_size`.
    """
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[: 2 * a.shape[0] - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex {x≥0, sum x = 1}
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex {x_i ≥ 0, Σ x_i = s}.
    The implementation follows Duchi et al. (2008).
    """
    if s <= 0:
        raise ValueError("Simplex radius must be positive")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        if now >= deadline:
            break

        iteration += 1
        prog = (now - start_time) / (deadline - start_time)

        # linearly decaying temperature and step size
        temperature = start_temp * (1.0 - prog) + end_temp * prog
        step_size = start_step * (1.0 - prog) + end_step * prog

        # ----------------------------------------------------------
        # 1) Decide which elementary move to make
        # ----------------------------------------------------------
        if np.random.rand() < prob_gradient:
            # ---- 1a) Projected sub‑gradient descent on the current max ----
            # Compute full convolution once (needed for max location)
            fa = np.fft.rfft(a, fft_size)
            conv = np.fft.irfft(fa * fa, fft_size)[: 2 * n - 1]
            max_b = float(conv.max())
            # indices where the maximum is attained (within a small tolerance)
            max_idx = np.where(conv >= max_b - 1e-12)[0]
            k = int(np.random.choice(max_idx))

            # Gradient of the max entry w.r.t. a:
            # g_i = 2 * a_{k - i}  for indices where 0 ≤ k-i < n, else 0
            grad = np.zeros_like(a)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            # Slice assignment avoids a Python loop
            rev_slice = slice(i_max, i_min - 1, -1)  # reversed view of relevant a
            grad[i_min : i_max + 1] = 2.0 * a[rev_slice]

            # Take a step opposite to the sub‑gradient and project back to simplex
            candidate = _proj_simplex(a - step_size * grad, s=1.0)
        else:
            # ---- 1b) Small random transfer of mass between two coordinates ----
            i, j = np.random.randint(0, n, size=2)
            while i == j:
                j = np.random.randint(0, n)

            # Randomly decide direction of the transfer
            if np.random.rand() < 0.5:
                delta = a[i] * step_size * np.random.rand()
                candidate = a.copy()
                candidate[i] -= delta
                candidate[j] += delta
            else:
                delta = a[j] * step_size * np.random.rand()
                candidate = a.copy()
                candidate[j] -= delta
                candidate[i] += delta

        # ----------------------------------------------------------
        # 2) Evaluate the candidate and decide on acceptance
        # ----------------------------------------------------------
        cand_val = _objective(candidate)
        delta_E = cand_val - cur_val

        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / temperature):
            # Accept the move
            a = candidate
            cur_val = cand_val
            no_improve = 0
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[iter {iteration:07d}] new best = {best_val:.9f} (t={now-start_time:.1f}s)")
        else:
            no_improve += 1

        # ----------------------------------------------------------
        # 3) Occasionally apply a larger perturbation to escape stagnation
        # ----------------------------------------------------------
        if no_improve > max_no_improve:
            noise = np.random.normal(scale=0.02, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _proj_simplex(a, s=1.0)
            cur_val = _objective(a)
            no_improve = 0
            print(f"[reset] large perturbation at iter {iteration}")

    # --------------------------------------------------------------
    # Final secure evaluation (the only place we call evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:           # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return the best sequence found (as a plain Python list)
    return best_a.tolist()
