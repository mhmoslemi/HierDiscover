"""
Propose a low‑value step function.

Algorithm
---------
1. **Seed generation** – a small, diverse set of candidate sequences:
   * the previous solution (if supplied as ``height_sequence_1``);
   * the best member of the 2‑parameter Beta family;
   * a few exponential, sinusoidal and Dirichlet draws;
   * the uniform distribution.

2. **Smooth‑max optimisation** – for each seed we run an Adam optimiser
   on the smooth approximation
        f_λ(a)= (1/λ)·log Σ_k exp(λ·(a∗a)_k)
   with a continuation on λ (0.5 → 80).  The parameters are projected onto
   the probability simplex after each update.  The best sequence from all
   seeds is kept.

3. **Exact‑max sub‑gradient descent** – starting from the best smooth
   solution we minimise the true objective 2·n·max(a∗a) by repeatedly
   moving opposite to the sub‑gradient of the maximal convolution entry,
   with a linearly decaying step size and projection onto the simplex.

4. **Intensive hill‑climbing** – a simulated‑annealing style local search
   (identical to the reference implementation) is run on the remaining
   time budget to polish the result.

All phases respect the 1000 s limit; the final sequence is returned as a
plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Beta‑family sweep (dense but cheap)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray | None, float]:
    """Search the beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf

    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + continuation on λ)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Optimises 2·n·max(conv(a,a)) using a smoothed max objective.
    Adam updates are projected onto the simplex after each step.
    λ grows from 0.5 to λ_max (default 80) as the optimisation proceeds.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # ensure Σa = 1

    best_a = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.07                     # base learning rate
    t = 0

    # FFT sizes (pre‑computed for the whole run)
    size1 = 1 << ((2 * n - 2).bit_length())       # for a*a
    size2 = 1 << ((3 * n - 3).bit_length())       # for a_rev * w

    lam_start = 0.5
    lam_max = 80.0

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        # ---------- λ continuation ----------
        prog = (now - start_time) / (deadline - start_time)
        lam = lam_start + (lam_max - lam_start) * prog

        # ---------- a*a ----------
        fa = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fa * fa, size1)[:2 * n - 1]
        max_c = conv.max()

        # ---------- smooth weights ----------
        # w_k ∝ exp(λ·(conv_k – max_c))  (max_c stabilises the exponent)
        exps = np.exp(lam * (conv - max_c))
        w = exps / exps.sum()            # length 2·n‑1, sum to 1

        # ---------- gradient ----------
        a_rev = a[::-1]
        fw = np.fft.rfft(w, size2)
        conv_aw = np.fft.irfft(np.fft.rfft(a_rev, size2) * fw, size2)[:3 * n - 2]
        grad = 2.0 * conv_aw[n - 1 : n - 1 + n]

        # ---------- Adam update ----------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        # ---------- keep the best ----------
        cur_val = 2.0 * n * max_c          # objective of the *old* a
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SmoothOpt {t:7d}] new best = {best_val:.6f} λ={lam:.2f}")
            if best_val <= 1.503:
                break

        # optional stagnation reset (every 800 steps without improvement)
        if t % 800 == 0:
            if cur_val >= best_val * 0.9999:
                noise = np.random.normal(scale=0.015, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[SmoothOpt {t:7d}] stagnation – random perturb")
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient optimiser
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    2·n·max(conv(a,a)).  Gradient is taken w.r.t. the currently maximal
    convolution entry.  A simple linear step‑size schedule and occasional
    random perturbations are used to escape plateaus.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    # step‑size schedule parameters
    step_start = 0.05
    step_end = 0.0005
    stagnation_counter = 0
    stagnation_limit = 2000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # full convolution and current max
        conv = _full_convolution(a)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        kmax = int(np.argmax(conv))

        # record improvement
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            stagnation_counter = 0
            print(f"[Subgrad] new best = {best_val:.6f}")
        else:
            stagnation_counter += 1

        # sub‑gradient of the max entry
        offsets = kmax - np.arange(n)
        mask = (offsets >= 0) & (offsets < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[offsets[mask]]

        # linear decay of step size
        prog = (now - start_time) / (deadline - start_time)
        lr = step_start * (1 - prog) + step_end * prog

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        # occasional random kick if stuck
        if stagnation_counter >= stagnation_limit:
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            stagnation_counter = 0
            print("[Subgrad] stagnation – random perturb")

    # final check
    final_val = _objective(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # Problem size – 8000 entries give a good trade‑off between quality and speed.
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣ Gather diverse initial seeds
    # ------------------------------------------------------------------
    seeds = []

    # a) previous solution, if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        seeds.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # b) best from a dense Beta‑family sweep
    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_a is not None:
            seeds.append(beta_a)
            print(f"[init] beta‑family seed (value={beta_val:.6f})")
    except Exception as exc:
        print("[init] beta family failed:", exc)

    # c) exponential decay families
    for lam in [0.0002, 0.0005, 0.001, 0.002]:
        a = np.exp(-lam * np.arange(n))
        a += np.exp(-lam * (np.arange(n)[::-1]))
        a /= a.sum()
        seeds.append(a)

    # d) sinusoidal families (raised to powers)
    for p in [0.5, 1.0, 2.0, 3.0]:
        x = np.arange(1, n + 1, dtype=float)
        a = np.sin(np.pi * x / (n + 1)) ** p
        a /= a.sum()
        seeds.append(a)

    # e) Dirichlet draws with various concentration parameters
    for alpha in [0.5, 2.0, 5.0]:
        a = np.random.dirichlet(np.full(n, alpha))
        seeds.append(a)

    # f) uniform distribution (fallback)
    seeds.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # 1️⃣ Smooth‑max optimisation (share the budget roughly equally)
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_val = np.inf

    total_time_remaining = deadline - time.time()
    if total_time_remaining <= 0:
        total_time_remaining = 0.0
    time_for_smooth = total_time_remaining * 0.55
    per_seed = time_for_smooth / max(1, len(seeds))

    for i, a_seed in enumerate(seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed
        a_opt, val_opt = _smooth_optimize(a_seed, n, local_deadline, start_time)
        print(f"[seed {i}] smooth_opt best = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Exact‑max sub‑gradient descent (medium‑range refinement)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and global_best_a is not None:
        subgrad_deadline = time.time() + remaining * 0.30
        a_opt, val_opt = _subgradient_optimize(global_best_a, n, subgrad_deadline, start_time)
        print(f"[subgrad] best = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()

    # ------------------------------------------------------------------
    # 3️⃣ Final intensive hill‑climbing on the remaining slice
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and global_best_a is not None:
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final secure evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # fallback guard – should never happen, but ensure a valid return
    if global_best_a is None:
        global_best_a = np.full(n, 1.0 / n)

    return global_best_a.tolist()
