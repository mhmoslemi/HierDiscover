"""
Hybrid spectral‑initialisation + Simulated‑Annealing optimiser
-------------------------------------------------------------

The optimiser builds a rich set of seed vectors, the most important one
being a *spectral root* of the flat target (the constant 1/(2·n‑1)
self‑convolution).  This seed already yields a max‑convolution close to
the theoretical lower bound 1/(2·n‑1).  From the best seed we run a
simple simulated‑annealing walk directly on the simplex (log‑space
updates, projection by Euclidean simplex projection).  Acceptance is
controlled by a temperature schedule that slowly cools to near‑zero.
Periodically a few projected sub‑gradient steps are performed to
polish the current point.  All heavy arithmetic is carried out with
FFT‑based convolutions (O(n log n)) so that thousands of objective
evaluations easily fit into the 1000 s budget.

The final vector is returned as a plain list of floats, ready to be
fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full self‑convolution (linear, not circular)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_fft(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence (without the sum‑square)."""
    n = a.shape[0]
    b = _conv_self_fft(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random Beta‑shaped candidate (symmetric when p == q)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)      # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Spectral root of the flat target – this is the main novelty
# ----------------------------------------------------------------------
def _spectral_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Construct a candidate whose self‑convolution is close to the flat
    1/(2n‑1) distribution.  We take the square‑root of the DFT of the
    constant target and randomly flip signs of the non‑DC frequencies.
    The resulting time‑domain vector is projected onto the simplex.
    """
    # target flat convolution (length 2n‑1)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # length for efficient FFT (next power of two)
    size = 1 << ((2 * n - 1).bit_length())

    # DFT of the target (real‑valued, but we keep complex for safety)
    T = np.fft.rfft(target, n=size)

    # magnitude of the square‑root spectrum
    mag = np.sqrt(np.abs(T))

    # Random sign pattern for all non‑DC frequencies (keeps the result real)
    signs = np.ones_like(mag)
    # rfft stores DC at index 0, the Nyquist term (if present) at the last index;
    # we treat all other components symmetrically.
    if len(signs) > 1:
        signs[1:] = np.where(rng.random(signs[1:].shape) < 0.5, 1.0, -1.0)

    A = mag * signs

    # Inverse transform back to time domain.
    a_full = np.fft.irfft(A, n=size)
    a = a_full[:n]                       # keep only the first n entries

    # Clip negatives (caused by random sign flips) and renormalise.
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (few iterations, used as a polish)
# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray) -> np.ndarray:
    """One projected sub‑gradient step for the exact max‑convolution objective."""
    n = a.shape[0]
    b = _conv_self_fft(a)
    max_b = b.max()
    idxs = np.where(b == max_b)[0]          # all positions attaining the max

    grad = np.zeros_like(a)
    for k in idxs:
        i = k - np.arange(n)
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]
    # simple diminishing step size (will be scaled outside)
    return grad

    # 1️⃣  Simulated‑Annealing walk on the simplex.
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val

    # Annealing schedule parameters
    T_start = 0.04 * best_val          # initial temperature (≈ 4 % of current score)
    T_end = 1e-6                       # final temperature
    sigma_start = 0.25                 # initial log‑space noise magnitude
    sigma_end = 0.001                  # final log‑space noise magnitude

    idle_counter = 0
    IDLE_LIMIT = 4000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear progress in [0,1]
        prog = (now - start_time) / (deadline - start_time)

        # temperature and noise magnitude schedules
        T = T_start * (1.0 - prog) + T_end * prog
        sigma = sigma_start * (1.0 - prog) + sigma_end * prog

        # propose a new point in log‑space then project back to the simplex
        w = np.log(cur_a + 1e-12)
        w_new = w + sigma * rng.standard_normal(n)
        a_new = np.exp(w_new)
        a_new /= a_new.sum()

        val_new = _objective_fft(a_new)
        delta = val_new - cur_val

        # Metropolis acceptance test
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_a, cur_val = a_new, val_new
            idle_counter = 0
            if val_new < best_val:
                best_a, best_val = a_new.copy(), val_new
                print(f"[SA ]  New global best = {best_val:.6f} (t={now - start_time:.1f}s)")
        else:
            idle_counter += 1

        # If we have been idle for a long time, re‑inject a small random
        # perturbation around the best known point – this helps escaping
        # shallow plateaus.
        if idle_counter > IDLE_LIMIT:
            w = np.log(best_a + 1e-12)
            w_new = w + sigma * rng.standard_normal(n)
            cur_a = np.exp(w_new)
            cur_a /= cur_a.sum()
            cur_val = _objective_fft(cur_a)
            idle_counter = 0
            print(f"[SA ]  Re‑started from best after idle at t={now - start_time:.1f}s")

        # Every few hundred iterations perform a tiny sub‑gradient polish
        # on the current point (only a handful of steps – cheap but often
        # beneficial).
        if int(now * 1000) % 500 == 0:
            # do 5 cheap sub‑gradient steps
            a_tmp = cur_a.copy()
            for _ in range(5):
                grad = _subgradient_step(a_tmp)
                # simple diminishing step (scaled by current temperature)
                step = 0.2 * T + 1e-6
                a_tmp = a_tmp - step * grad
                a_tmp = _project_simplex(a_tmp)
            cand_val = _objective_fft(a_tmp)
            if cand_val < cur_val:
                cur_a, cur_val = a_tmp, cand_val
                if cand_val < best_val:
                    best_a, best_val = a_tmp.copy(), cand_val
                    print(f"[SA ]  Sub‑grad polish improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Final short sub‑gradient polishing (exact objective)
    # --------------------------------------------------------------
    # We run a fixed number of iterations; each iteration costs roughly
    # one FFT, which is cheap at this problem size.
    max_polish_iters = 80_000
    a = best_a.copy()
    cur_val = best_val
    start_polish = time.time()
    for it in range(max_polish_iters):
        now = time.time()
        if now >= deadline:
            break

        grad = _subgradient_step(a)
        # step size linearly decreasing to near‑zero
        prog = (now - start_polish) / (deadline - start_polish)
        step = 0.12 * (1.0 - prog) + 1e-7 * prog
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective_fft(a_new)
        if new_val + 1e-12 < cur_val:
            a, cur_val = a_new, new_val
            if new_val < best_val:
                best_a, best_val = a_new.copy(), new_val
                print(f"[Pol]  New best after polishing = {best_val:.6f}")

        # occasional jitter to avoid tiny stagnation
        if it % 1000 == 0 and rng.random() < 0.03:
            a = np.maximum(a + rng.normal(scale=0.01, size=n), 0.0)
            a = _project_simplex(a)

    # --------------------------------------------------------------
    # 3️⃣  Final secure evaluation and return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Ensure plain list of Python floats
    return best_a.tolist()
