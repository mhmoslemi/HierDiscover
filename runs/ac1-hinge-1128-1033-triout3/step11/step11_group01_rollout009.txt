"""
Hybrid Frank‑Wolfe / exponentiated‑subgradient optimizer.

Phase 1 (small problem, n≈2000) solves a linear programme

    maximise   Σ b_i
    subject to  (f * b)_k ≤ M   for every k,
                b_i ≥ 0,

where f is the current sequence, M = max(f*f) and * is the linear
convolution.  The LP is cheap for n≈2000 and quickly reduces the
peak of the self‑convolution.

The LP solution b is orthogonally scaled to the required sum
√(2n) and mixed with f (mixing weight t≈0.2).  Repeating this step
produces a monotone decreasing sequence of maxima.

Phase 2 interpolates the best small‑n sequence to the target length
n≈8000 and refines it with an exponentiated‑gradient method that
uses the true (non‑smooth) max‑convolution as objective.
The gradient of max(c) w.r.t. a is
    g = 2·cross_corr(w,a)
where w is the uniform distribution over the indices attaining the
maximum of the convolution c = a*a.
A decreasing step‑size η_t = η₀/√t and occasional Gaussian
perturbations avoid stagnation.

Only the final call to ``evaluate_sequence`` is performed at the end,
so the routine respects the 1000 s budget and returns the best
sequence discovered.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (required for efficient FFT length)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·n‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b (both real)."""
    n, m = a.shape[0], b.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k·a_{k‑i} for i=0…n‑1.
    Implemented via FFT (convolution with reversed a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1          # = (2n‑1)+n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _interpolate(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to length ``new_n``."""
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new = np.interp(new_x, old_x, old)
    new = np.maximum(new, 0.0)
    return new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s
    time budget and prints progress information.
    """

    start_time = time.time()
    # reserve a small safety margin for the final secure call
    final_deadline = start_time + budget_s - 0.5

    phase1_deadline = start_time + 0.6 * budget_s

    # Phase 1 – small LP based Frank‑Wolfe iterations
    # ------------------------------------------------------------------
    n_small = 2000                     # problem size for the LP stage
    sum_target_small = math.sqrt(2.0 * n_small)

        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a_small = _interpolate(old, n_small)
        if a_small.sum() == 0:
            raise ValueError
        a_small = a_small * (sum_target_small / a_small.sum())
        print("[phase1] interpolated from previous solution")
    except Exception:
        a_small = np.full(n_small, sum_target_small / n_small)
        print("[phase1] uniform start")

    best_a_small = a_small.copy()
    best_val_small = float('inf')

    # pre‑allocate the inequality matrix for the LP (reuse each iteration)
    rows_small = 2 * n_small - 1
    A_ub = np.empty((rows_small, n_small), dtype=np.float64)

    # mixing weight (kept constant – small enough to guarantee descent)
    t_mix = 0.2

    iter_small = 0
    while time.time() < phase1_deadline:

        # current self‑convolution and objective (objective = 2·n·max)
        conv_f = _full_convolution(a_small)

        cur_val = 2.0 * n_small * cur_max
        if cur_val < best_val_small - 1e-12:
            best_val_small = cur_val
            best_a_small = a_small.copy()
            print(f"[phase1 iter {iter_small:05d}] best = {best_val_small:.9f}")

        # bound M for the linear programme (the current peak)
        M = cur_max

        # build the Toeplitz‐like matrix representing the convolution constraints
        # A_ub[k,i] = a_i  for rows k = i … i+n_small‑1
        for i in range(n_small):
            A_ub[i:i + n_small, i] = a_small

        # linear programme: maximise Σ b_i  ⇔ minimise −Σ b_i
        c = -np.ones(n_small)                     # objective coefficients
        b_ub = np.full(rows_small, M)            # right‑hand side of each inequality
        bounds = [(0.0, None)] * n_small         # b_i ≥ 0

        sum_b = b.sum()

    best_val = float('inf')

        cur_max = conv.max()
        cur_val = 2.0 * n_large * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a_large.copy()
            stagnation_cnt = 0
            print(f"[phase2 iter {iter_large:06d}] new best = {best_val:.9f}")
        else:
            stagnation_cnt += 1

        # indices attaining (or almost attaining) the maximum

        max_idx = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        w[max_idx] = 1.0 / max_idx.size

        # gradient g = 2·cross_corr(w, a)
        grad = 2.0 * _cross_corr(w, a_large)

        # step size decreasing like 1/√t
        eta = eta0 / math.sqrt(iter_large)

        # exponentiated (mirror‑descent) update – guaranteed non‑negative
        a_large = a_large * np.exp(-eta * grad)

        # renormalise to the prescribed sum √(2n)
        total = a_large.sum()
        if total <= 0.0:

            a_large = a_large * np.exp(noise)                     # multiplicative noise

            stagnation_cnt = 0
            print(f"[phase2 reset] added Gaussian noise at iter {iter_large}")

    # Final secure evaluation (only once)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
