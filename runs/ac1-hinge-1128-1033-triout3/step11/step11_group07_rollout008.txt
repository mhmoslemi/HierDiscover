```
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8000
    elite_size = 30
    elites = [] # list of (value, array)

    def add_candidate(a):
        nonlocal elites
        val = _objective(a)
        if len(elites) < elite_size:
            elites.append((val, a))
            elites.sort(key=lambda x: x[0])
        else:
            if val < elites[-1][0]:
                elites[-1] = (val, a)
                elites.sort(key=lambda x: x[0])

    # initial candidates:
    # include uniform, previous solution if exists
    try: 
        tmp = np.asarray(height_sequence_1, dtype=float)
        a0 = tmp.copy()
        # ensure sum=1, clip negative, scale to 1
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        add_candidate(a0)
        print("[Init] previous solution added value=", _objective(a0))
    except Exception:
        pass

    # uniform
    a_uni = np.full(n, 1.0/n)
    add_candidate(a_uni)

    # Random Beta candidates
    for _ in range(200):
        cand = _random_beta_candidate(n, rng)
        add_candidate(cand)

    # Dirichlet for various alphas
    for alpha in (0.01,0.05,0.1,0.2,0.5,1.0,2.0,5.0):
        for _ in range(5):
            cand = rng.dirichlet(np.full(n, alpha))
            add_candidate(cand)

    # Sparse candidates
    for _ in range(200):
        s = rng.integers(2, 61) # support size
        idx = rng.choice(n, size=s, replace=False)
        weights = rng.random(s)
        weights /= weights.sum()
        a = np.zeros(n)
        a[idx] = weights
        add_candidate(a)

    # Symmetric version of some candidates
    # We'll mirror half of elites later.

    # evolution loop
    while time.time() < deadline - 5:  # keep 5 secs for polishing
        # select parents
        # Use tournament selection more likely to pick top elite
        # Weighted by rank: probability ~ 1/popsize
        # For simplicity: random index with distribution 0..elite_size-1 favor low values
        ranks = np.arange(elite_size)
        probs = 1.0/(ranks+1)
        probs /= probs.sum()
        i, j = rng.choice(elite_size, size=2, replace=False, p=probs)
        parent1 = elites[i][1]
        parent2 = elites[j][1]
        # crossover
        t = rng.random()
        child = t * parent1 + (1-t) * parent2
        # mutate:
        if rng.random() < 0.3:
            # log-space Gaussian noise
            log_child = np.log(child + 1e-12)
            log_child += rng.normal(scale=0.1, size=n)
            child = np.exp(log_child)
        # ensure non-neg, projection simplex
        child = np.maximum(child, 0.0)
        child /= child.sum()
        add_candidate(child)

    # At this point best = elites[0]
    best_a = elites[0][1].copy()
    best_val = elites[0][0]
```

```python
def _full_convolution(x):
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _conv_fft(x, y):
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

def _objective(a):
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())
```

```python
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Evolutionary+gradient optimiser for the max‑convolution objective.
    1️⃣ Initialise a diverse elite set of candidates (Beta, Dirichlet,
       sparse random).
    2️⃣ Apply a simple evolutionary algorithm: combine elite members,
       mutate in log‑space and keep the best.
    3️⃣ Refine the current best with Adam on a smooth approximation of the
       L∞‑norm.
    4️⃣ Final polishing with a projected sub‑gradient method that uses
       back‑tracking line‑search.
    Returns a list of non‑negative numbers whose sum is 1.
    """
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    n = 8000
    elite_size = 30
    elites = []  # list of (value, array)

    def add_to_elite(arr):
        nonlocal elites
        val = _objective(arr)
        if len(elites) < elite_size:
            elites.append((val, arr.copy()))
            elites.sort(key=lambda x: x[0])
        else:
            if val < elites[-1][0]:
                elites[-1] = (val, arr.copy())
                elites.sort(key=lambda x: x[0])

    # 0️⃣ use known solution if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        old = np.maximum(old, 0.0)
        if old.sum() == 0:
            old = np.full_like(old, 1.0 / old.shape[0])
        else:
            old /= old.sum()
        add_to_elite(old)
        print("[Init] previous solution added, value={:.6f}".format(elites[0][0]))
    except Exception:
        pass

    # uniform
    add_to_elite(np.full(n, 1.0 / n))

    # 0.1 Beta candidates
    for _ in range(300):
        cand = _random_beta_candidate(n, rng)
        add_to_elite(cand)

    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):

            add_to_elite(cand)

    for _ in range(300):
```

The following programme implements a fresh optimiser for the max‑convolution objective

\[
f(a)=\frac{2\,n\; \max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

The optimisation proceeds in four stages

1. **Diverse initialisation** – a large set of cheap candidates
   (Beta‑shapes, Dirichlet draws, sparse random vectors) is built.
   The `elite` list keeps the `elite_size` best candidates.

2. **Evolutionary recombination** – elite members are repeatedly
   convex‑combined, mutated in log‑space and projected back onto the
   probability simplex.  New offspring replace the worst elite member
   when they improve the objective.

3. **Fine‑grained Adam optimisation** – the best elite is refined by
   Adam on a smooth (log‑sum‑exp) approximation of the `max`.
   The smoothing parameter ε and learning rate are annealed.

4. **Exact sub‑gradient polishing** – a projected sub‑gradient method
   with back‑tracking line‑search finishes the search, guaranteeing
   monotone improvement of the true objective.

All numbers are kept non‑negative and normalised to sum = 1, so the
returned list can be fed directly to the provided `evaluate_sequence`
function.

```python
"""Step‑function optimiser for the max‑convolution bound.

The algorithm consists of

1.  a large, diverse random initialisation (Beta, Dirichlet,
    sparse vectors);
2.  an evolutionary recombination of the best “elite’’ candidates;
3.  Adam optimisation on a smooth surrogate of the max‑norm;
4.  a final exact sub‑gradient polishing with back‑tracking line search.

All vectors are projected onto the probability simplex (sum = 1,
non‑negative) after every modification.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    nx, ny = x.shape[0], y.shape[0]
    out_len = nx + ny - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient
    of 2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)               # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()            # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)          # length 2n‑1
    seg = conv_ab[n - 1:2 * n - 1]          # the n entries that correspond to a*a
    grad_a = 4.0 * n * seg[::-1]           # gradient already contains the factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex Σ x = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random Beta‑shaped candidate (used for cheap initial sweep)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max (more aggressive schedule)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for the smoothed objective.
    The smoothing ε is annealed from 0.5 to 1e‑8 and the learning rate
    from 0.5 down to 0.01.
    """
    # work in log‑softmax space to keep a on the simplex automatically
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.01
    eps_start, eps_end = 0.5, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 2000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam]  new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # grad already contains 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.05, size=n)

        # restart if stalled
        if idle > MAX_IDLE:
            print("[Adam]  restart after idle")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing with back‑tracking line‑search
# ----------------------------------------------------------------------
def _subgradient_line_search_polish(a0: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    start_time: float,
                                    best_a: np.ndarray,
                                    best_val: float,
                                    max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* objective
    f(a)=2·n·max(conv(a,a)).  The step size is chosen by a simple
    back‑tracking line search, and a small log‑space jitter is applied
    occasionally to escape flat plateaus.
    """
    a = a0.copy()
    cur_val = _objective(a)
    if cur_val < best_val:
        best_val, best_a = cur_val, a.copy()

    rng = np.random.default_rng()
    it = 0
    while it < max_iters:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient of max_b (vector of length n)
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # back‑tracking line‑search (heuristic initial step)
        gnorm2 = np.dot(grad, grad)
        if gnorm2 == 0.0:
            break
        step = cur_val / (gnorm2 + 1e-12)
        step = min(step, 1.0)                # avoid huge jumps
        improved = False
        for _ in range(7):                  # at most 7 reductions
            cand = a - step * grad
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < cur_val - 1e-12:
                a = cand
                cur_val = cand_val
                improved = True
                break
            step *= 0.5

        if not improved:
            # jitter to get out of a flat region
            if rng.random() < 0.1:
                w = np.log(a + 1e-12)
                w += rng.normal(scale=0.03, size=n)
                a = np.exp(w)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                cur_val = _objective(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[Polish] new best = {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size (larger n gives a richer search space)
    # --------------------------------------------------------------
    n = 8000
    elite_size = 30
    elites: list[tuple[float, np.ndarray]] = []   # sorted by value (ascending)

    def add_to_elite(arr: np.ndarray):
        """Insert ``arr`` into the elite list if it is among the best."""
        nonlocal elites
        val = _objective(arr)
        if len(elites) < elite_size:
            elites.append((val, arr.copy()))
            elites.sort(key=lambda x: x[0])
        else:
            if val < elites[-1][0]:
                elites[-1] = (val, arr.copy())
                elites.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 0️⃣  Initialise – many cheap candidates, keep the best ones
    # --------------------------------------------------------------
    # 0.1  Re‑use a previously discovered solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old = np.maximum(old, 0.0)
        if old.sum() == 0.0:
            old = np.full_like(old, 1.0 / old.shape[0])
        else:
            old /= old.sum()
        add_to_elite(old)
        print(f"[Init] old solution added, value = {elites[0][0]:.6f}")
    except Exception:
        pass

    # 0.2  Uniform baseline
    add_to_elite(np.full(n, 1.0 / n))

    # 0.3  Random Beta shapes
    for _ in range(300):
        cand = _random_beta_candidate(n, rng)
        add_to_elite(cand)

    # 0.4  Dirichlet draws with many concentrations
    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(6):
            cand = rng.dirichlet(np.full(n, alpha))
            add_to_elite(cand)

    # 0.5  Sparse random vectors (useful for flattening the convolution)
    for _ in range(300):
        support = rng.integers(2, 51)                 # support size 2 … 50
        idx = rng.choice(n, size=support, replace=False)
        weights = rng.random(support)
        weights /= weights.sum()
        a = np.zeros(n)
        a[idx] = weights
        add_to_elite(a)

    best_a = elites[0][1].copy()
    best_val = elites[0][0]
    print(f"[Init] best of initial pool: {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Evolutionary recombination (≈40 % of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.40 * budget_s
    # rank‑based parent selection probabilities
    ranks = np.arange(elite_size)
    parent_prob = 1.0 / (ranks + 1)
    parent_prob /= parent_prob.sum()

    while time.time() < phase1_deadline:
        i, j = rng.choice(elite_size, size=2, replace=False, p=parent_prob)
        p1, p2 = elites[i][1], elites[j][1]
        t = rng.random()
        child = t * p1 + (1.0 - t) * p2

        # log‑space mutation with probability 0.3
        if rng.random() < 0.3:
            log_child = np.log(child + 1e-12)
            log_child += rng.normal(scale=0.07, size=n)
            child = np.exp(log_child)

        child = np.maximum(child, 0.0)
        child /= child.sum()
        add_to_elite(child)

    best_a, best_val = elites[0][1].copy(), elites[0][0]
    print(f"[Phase 1] best after evolution: {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smooth surrogate (≈40 %)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase2_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)
    print(f"[Phase 2] best after Adam: {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Exact sub‑gradient polishing (remaining time)
    # --------------------------------------------------------------
    best_a, best_val = _subgradient_line_search_polish(
        best_a, n,
        deadline=deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=500_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)
    print(f"[Phase 3] final best before secure eval: {best_val:.6f}")

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {reported:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return a plain list of floats
    return best_a.tolist()
```
