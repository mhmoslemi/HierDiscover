        start = k - (n - 1)
        grad[start:] = 2 * a[:n-start][::-1]

def permutation_local_search(a, max_iters, time_deadline):
    best_a = a.copy()
    best_val = evaluate_objective(a)
    n = len(a)
    while time.time() < time_deadline and max_iters>0:
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        # swap values at i and j
        a_sw = best_a.copy()
        a_sw[i], a_sw[j] = a_sw[j], a_sw[i]
        cur_val = evaluate_objective(a_sw)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_sw
        max_iters -= 1
    return best_a, best_val
```

"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence().  The optimisation proceeds in three stages:

1️⃣  A fast global scan over the 2‑parameter Beta family (plus a few
    Dirichlet draws) gives a solid initial distribution.

2️⃣  A long smoothed gradient‑descent phase on the log‑sum‑exp
    approximation of max(a∗a).  The smoothing parameter β is increased
    slowly up to 5 000 and the step size decays as 1/√t.
    The best iterate is kept.

3️⃣  A short exact sub‑gradient refinement (gradient of the true max)
    followed by a tiny random‑swap local search.

All updates respect the simplex Σa=1, a≥0 (Euclidean projection).
The best vector found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the parameters produce an all‑zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the smoothed objective
        φ_β(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i)

    Returns (grad, max_conv) where grad is the Euclidean gradient
    (same shape as a) and max_conv is max(a⋆a) (useful for the exact
    objective value).
    """
    n = a.shape[0]

    # ----- 1. full convolution a⋆a (length 2n‑1) -----
    out_len = 2 * n - 1
    size_conv = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[:out_len]        # (a⋆a)

    max_conv = conv.max()

    # ----- 2. soft‑max weights w_i -----
    exp_arg = beta * (conv - max_conv)            # ≤ 0
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum                            # Σ_i w_i = 1

    # ----- 3. gradient = 2·(a ⋆_cross w_norm) -----
    # Cross‑correlation (a ⋆ w_norm)[j] = Σ_k a_k·w_norm_{k+j}
    rev_w = w_norm[::-1]                          # reverse for convolution
    out_len_corr = n + out_len - 1                # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1                          # position of lag 0 in conv_full
    idx = offset - np.arange(n)                   # indices for j = 0…n‑1
    grad = 2.0 * corr_full[idx]

    return grad, max_conv


def _subgradient_maxconv(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Exact sub‑gradient of max(a⋆a).  Returns (grad, max_val, k) where
    k is the index at which the maximum of the convolution is attained.
    The gradient is the vector g with g_j = 2·a_{k‑j} whenever the index
    is inside the support, otherwise 0.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_val = conv.max()
    k = int(conv.argmax())

    grad = np.zeros_like(a)

    if k <= n - 1:                      # left half of convolution
        # indices 0..k receive values
        grad[:k + 1] = 2.0 * a[k::-1]
    else:                               # right half
        start = k - (n - 1)              # first index that receives a non‑zero entry
        # grad[start:] corresponds to a[:n-start] reversed
        grad[start:] = 2.0 * a[:n - start][::-1]

    return grad, max_val, k


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def _smooth_descent(a_start: np.ndarray,
                    time_budget: float,
                    beta_max: float = 5000.0,
                    beta_mul: float = 1.005,
                    step0: float = 0.3) -> tuple[np.ndarray, float]:
    """Smoothed log‑sum‑exp gradient descent for a limited time budget."""
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(a)
    n = a.shape[0]
    beta = 0.5
    it = 0
    start = time.time()
    while time.time() - start < time_budget:
        step = step0 / np.sqrt(it + 1.0)
        grad, cur_max = _gradient_smooth(a, beta)
        # Gradient of the true objective = 2·n·grad (β‑smooth approximates max)
        a = a - step * grad
        a = _project_simplex(a)

        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # tiny random jitter to escape flat plateaus
        if np.random.rand() < 0.001:
            a += np.random.normal(scale=5e-4, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

        beta = min(beta_max, beta * beta_mul)
        it += 1
    return best_a, best_val


def _subgrad_descent(a_start: np.ndarray,
                     time_budget: float,
                     step0: float = 0.15) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on max(a⋆a)."""
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(a)
    n = a.shape[0]
    it = 0
    start = time.time()
    while time.time() - start < time_budget:
        step = step0 / np.sqrt(it + 1.0)
        grad, max_val, _ = _subgradient_maxconv(a)
        a = a - step * grad
        a = _project_simplex(a)

        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


def _local_swap_search(a_start: np.ndarray,
                       time_budget: float,
                       max_trials: int = 10000) -> tuple[np.ndarray, float]:
    """
    Very cheap local improvement: random swaps of two coordinates.
    Accept a swap only if it strictly improves the exact objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    trials = 0
    while trials < max_trials and time.time() - start < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform swap
        a_sw = a.copy()
        a_sw[i], a_sw[j] = a[j], a[i]
        cur_val = _objective(a_sw)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_sw
            a = a_sw   # keep the improved swap as current state
        trials += 1
    return best_a, best_val
