"""
Propose a high‑quality step function for the given evaluation metric.

Algorithm
---------

1️⃣ **Initialisation** – Scan a variety of parametric families
   (Beta, exponential, linear ramps, Dirichlet draws) on a fixed
   length `n` (8000).  Keep the best vector w.r.t. the exact objective
   `2·n·max(conv(a,a))` (the sequence is kept on the simplex
   Σa=1).

2️⃣ **Refinement** – Optimise the exact objective with a fast
   first‑order method:
   * a smooth approximation of the max‑convolution
     φβ(a)= (1/β)·log Σ exp(β·(a⋆a)ₖ) is used.
   * The gradient of φβ is obtained via FFT‑based cross‑correlation.
   * Adam (adaptive moments) is employed for the projected updates.
   * After each step the iterate is projected back onto the simplex.
   * The smoothing parameter β is increased geometrically (β←β·β_mul)
     up to a large value (≈200), at which φβ≈max.
   * Very occasional random jitter helps to escape tiny plateaus.

3️⃣ **Tracking** – The best sequence seen during the run is stored.
   At the end it is passed to the provided ``evaluate_sequence`` for
   the final (secure) score.

The whole routine respects the 1000 s time budget, checks for the
optional global ``height_sequence_1`` (previous best) and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``x`` with itself (length 2·len(x)‑1)
    using a power‑of‑two FFT for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective = 2·n·max(conv(a,a)) (a lives on the simplex)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    Σx=1, x≥0.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Normalised Beta‑shaped vector: a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_shape(r: float, n: int) -> np.ndarray | None:
    """Normalised geometric (exponential) shape: a_i ∝ (1‑r)·r**i."""
    if not (0.0 < r < 1.0):
        return None
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _grad_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smoothed max‑convolution
        φβ(a) = (1/β)·log Σ exp(β·(a⋆a)_k)

    Returns (grad, max_conv) where ``max_conv`` is the exact
    max‐convolution value (useful for the true objective).
    """
    n = a.shape[0]

    # ---- 1. a⋆a (full) ----
    out_len = 2 * n - 1
    size_conv = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[:out_len]

    max_conv = conv.max()

    # ---- 2. soft‑max weights w_k ----
    exp_arg = beta * (conv - max_conv)               # ≤ 0
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum                               # Σ w = 1

    # ---- 3. gradient = 2·(a ⋆_cross w_norm) ----
    # cross‑correlation via FFT (a * reversed w_norm)
    rev_w = w_norm[::-1]
    out_len_corr = n + out_len - 1                     # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1                               # lag 0 position
    idx = offset - np.arange(n)                        # indices for the gradient
    grad = 2.0 * corr_full[idx]

    return grad, max_conv


def _grad_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient of maxₖ (a⋆a)ₖ and the corresponding max
    value.  If the maximum is attained at several indices,
    their gradients are summed (any convex combination is valid).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_val = conv.max()
    ks = np.where(conv == max_val)[0]        # all arg‑max positions
    grad = np.zeros_like(a)

    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1)            # indices of a participating
        grad[lo:hi + 1] += 2.0 * a[k - j]    # 2·a_{k‑j}
    return grad, max_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine works within the given time
    budget (seconds) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # Problem size – a good trade‑off between expressivity and speed
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Try to reuse a previous solution, if available
    # --------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.size > 1:
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() > 0.0:
                a0 /= a0.sum()
                print("[init] Interpolated previous solution")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Coarse search in simple parametric families
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # helper to evaluate and possibly update the incumbent
    def _maybe_update(cand: np.ndarray):
        nonlocal best_a, best_val
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[scan] new best = {best_val:.6f}")

    # start from a0 if we have it
    if a0 is not None:
        _maybe_update(a0)

    # ----- Beta family (grid + random) -----
    p_grid = np.concatenate(
        [
            np.geomspace(0.001, 0.1, 8, endpoint=False),
            np.geomspace(0.1, 5.0, 12, endpoint=False),
            np.geomspace(5.0, 20.0, 5),
        ]
    )
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is None:
                continue
            _maybe_update(cand)

    # random (p,q) in log‑space
    rng = np.random.default_rng()
    for _ in range(5000):
        logp = rng.uniform(np.log(1e-3), np.log(20.0))
        logq = rng.uniform(np.log(1e-3), np.log(20.0))
        p, q = np.exp(logp), np.exp(logq)
        cand = _beta_shape(p, q, n)
        if cand is None:
            continue
        _maybe_update(cand)
        if time.time() >= deadline - 30.0:      # keep ~30 s for refinement
            break

    # ----- Exponential family (geometric decay) -----
    for _ in range(3000):
        # ε = 1‑r will be sampled log‑uniformly in [1e‑6, 0.2]
        eps = np.exp(rng.uniform(np.log(1e-6), np.log(0.2)))
        r = 1.0 - eps
        cand = _exp_shape(r, n)
        if cand is None:
            continue
        _maybe_update(cand)

    # ----- Linear ramps (simple triangular shapes) -----
    lin_inc = np.arange(1, n + 1, dtype=float)
    cand = lin_inc / lin_inc.sum()
    _maybe_update(cand)

    lin_dec = lin_inc[::-1]
    cand = lin_dec / lin_dec.sum()
    _maybe_update(cand)

    # ----- A few Dirichlet draws for diversity -----
    for _ in range(8):
        cand = rng.dirichlet(np.ones(n))
        _maybe_update(cand)

    # fallback – uniform distribution
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print("[fallback] uniform start, value =", best_val)

    # --------------------------------------------------------------
    # 2️⃣  Refine the incumbent with smoothed Adam descent
    # --------------------------------------------------------------
    a = best_a.copy()               # always kept on the simplex Σa=1
    best_seq = a.copy()
    best_score = best_val

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    t = 0

    # schedule for the smoothing parameter β
    beta = 0.5
    beta_mul = 1.001
    beta_max = 200.0

    # step‑size base (will decay as 1/√t)
    step0 = 0.6

    # target value – we can stop early if we beat it
    target_score = 1.503

    while time.time() < deadline:
        t += 1

        # ---- gradient (smooth at low β, exact when β reached max) ----
        if beta < beta_max:
            grad, cur_max = _grad_smooth(a, beta)
        else:
            # at very high β the smooth gradient is already ~exact,
            # but we switch to the true sub‑gradient for numerical safety.
            grad, cur_max = _grad_max(a)

        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[iter] new best = {best_score:.6f} (t={t}, β={beta:.2f})")
            if best_score <= target_score:
                print("[info] target reached – terminating early")
                break

        # ---- Adam update with projection onto the simplex ----
        step = step0 / np.sqrt(t)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_candidate = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = _project_simplex(a_candidate)

        # occasional tiny random jitter (helps to escape flat plateaus)
        if np.random.rand() < 0.001:
            a_candidate += np.random.normal(scale=0.001, size=n)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

        a = a_candidate

        # increase smoothing temperature
        beta = min(beta_max, beta * beta_mul)

    # --------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return the best sequence discovered within the time budget
    return best_seq.tolist()
