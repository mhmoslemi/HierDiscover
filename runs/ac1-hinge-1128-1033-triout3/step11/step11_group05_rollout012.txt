"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  The method is a two‑phase search:

1️⃣  Broad exploration of a handful of parametric families (Beta,
    asymmetric exponential, Gaussian, power‑sine, triangular and
    sparse Dirichlet draws).  The best *K* candidates (by the cheap
    objective 2·n·max(conv)) are retained.

2️⃣  Each retained candidate is refined by an Adam optimiser that
    minimises a smooth approximation of the max‑convolution:
        f(a) = 2·n·Mα(a) ,  Mα = (1/α)·log∑exp(α·conv(a,a)).
    The smooth weight vector w = softmax(α·conv) gives a well‑defined
    gradient:
        ∇f = 2·n·2·correlate(a, w) .
    Adam’s adaptive step‑size, a schedule for α (starting small,
    ending ≈200) and occasional jitter allow the optimiser to escape
    plateaus.  The search respects the supplied time budget and the
    best sequence found is finally evaluated with ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based – fast for n≈8000)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Full correlation of a with w (length len(a)+len(w)-1).
    Used for the gradient of the smooth‐max objective.
    """
    n = a.shape[0]
    m = w.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    # correlation = ifft(Fa * conj(Fw))
    corr = np.fft.irfft(fa * np.conj(fw), size)[:out_len]
    return corr


def _objective(a: np.ndarray) -> float:
    """
    Cheap proxy: exact value of 2·n·max(conv(a,a)) with Σa = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Parametric families (used in the broad exploration)
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} , i=1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    tot = a.sum()
    if tot <= 0.0:
        return None
    a /= tot
    return a


def _exp_shape(c: float, n: int) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·|i-mid|)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a


def _asym_exp_shape(c: float, n: int) -> np.ndarray:
    """One‑sided exponential a_i ∝ exp(-c·i) (mass concentrated on the left)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-c * i)
    a /= a.sum()
    return a


def _gaussian_shape(sigma: float, n: int) -> np.ndarray:
    """Centered Gaussian."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a


def _power_sine_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1))^p ."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    a /= a.sum()
    return a


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max objective
# ----------------------------------------------------------------------


def _smooth_max_optimize(
    init_a: np.ndarray,
    n: int,
    budget_s: float,
    start_time: float,
    deadline: float,
) -> np.ndarray:
    """
    Refine ``init_a`` (already normalised to sum = 1) using Adam on the
    smoothed max objective.
    Returns the best sequence discovered before ``deadline``.
    """
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Controls the smoothness of the max approximation
    alpha_start = 5.0
    alpha_end = 200.0

    # Learning‑rate schedule (initially 0.2, decays to ~1e‑4)
    lr_start = 0.2
    lr_end = 1e-4

    # Convert to log‑space to keep positivity automatically
    log_x = np.log(np.clip(init_a, 1e-30, None))
    m = np.zeros_like(log_x)
    v = np.zeros_like(log_x)

    best_seq = init_a.copy()
    best_val = _objective(best_seq)

    t = 0
    no_improve = 0
    max_no_imp = 4000

    while time.time() < deadline:
        t += 1
        # step‑size and α schedules (linear in optimisation progress)
        progress = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (1 - progress) + lr_end * progress
        alpha = alpha_start * (1 - progress) + alpha_end * progress

        # reconstruct a from log‑space
        a = np.exp(log_x - np.max(log_x))          # stabilise
        a_sum = a.sum()
        a = a / a_sum

        # full convolution a*a
        conv = _full_convolution(a)

        # smooth‑max weights: softmax(α·conv) – numerically stable
        conv_shift = conv - conv.max()
        w = np.exp(alpha * conv_shift)
        w_sum = w.sum()
        w = w / w_sum

        # gradient of the smooth max:
        #   ∇f = 2·n·2·correlate(a, w)   (the factor 2 comes from d(conv)/da)
        corr = _full_correlation(a, w)
        # slice the part that corresponds to indices 0 … n‑1
        start = w.shape[0] - 1                # = 2·n‑2
        grad = 4.0 * corr[start : start + n]   # 2·n·2 = 4·n, we absorb n later

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        log_x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to break out of plateaus
        if (no_improve >= max_no_imp) or (np.random.rand() < 0.001):
            jitter = np.random.normal(scale=1e-3, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            log_x = np.log(np.clip(a, 1e-30, None))
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # evaluate current objective (exact, cheap proxy)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[adam] iter {t:05d} new best {best_val:.6f}")
        else:
            no_improve += 1

    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence discovered before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin for final eval

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressivity and speed.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Storage for the top‑K candidates emerging from the wide search.
    # ------------------------------------------------------------------
    K = 6
    top_vals = [np.inf] * K
    top_seqs = [None] * K

    def _consider(candidate: np.ndarray):
        """Insert candidate into the top‑K list if it improves the proxy."""
        val = _objective(candidate)
        # find position (if any)
        for i in range(K):
            if val < top_vals[i]:
                # shift down
                for j in range(K - 1, i, -1):
                    top_vals[j] = top_vals[j - 1]
                    top_seqs[j] = top_seqs[j - 1]
                top_vals[i] = val
                top_seqs[i] = candidate.copy()
                print(f"[explore] new top‑{i+1} value={val:.6f}")
                break

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from an older solution if present
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _consider(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Wide exploration across expressive families
    # ------------------------------------------------------------------
    # (a) dense Beta grid
    p_grid = np.concatenate(
        [
            np.geomspace(0.001, 0.1, 8, endpoint=False),
            np.geomspace(0.1, 5.0, 12, endpoint=False),
            np.geomspace(5.0, 20.0, 5),
        ]
    )
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is not None:
                _consider(cand)

    # (b) random Beta draws (log‑uniform)
    rng = np.random.default_rng()
    log_min, log_max = np.log(1e-3), np.log(20.0)
    for _ in range(8000):
        p = np.exp(rng.uniform(log_min, log_max))
        q = np.exp(rng.uniform(log_min, log_max))
        cand = _beta_shape(p, q, n)
        if cand is not None:
            _consider(cand)

        if time.time() >= deadline - 250.0:   # reserve ~250 s for refinement
            break

    # (c) symmetric exponential decay
    for _ in range(1500):
        c = rng.uniform(0.0, 5.0)
        _consider(_exp_shape(c, n))

    # (d) one‑sided exponential (mass concentrated on the left)
    for _ in range(1500):
        c = rng.uniform(0.0, 8.0)
        _consider(_asym_exp_shape(c, n))

    # (e) Gaussian family
    for _ in range(1500):
        sigma = rng.uniform(0.5, n / 5.0)
        _consider(_gaussian_shape(sigma, n))

    # (f) power‑sine family
    for _ in range(1500):
        p = rng.uniform(0.1, 8.0)
        _consider(_power_sine_shape(p, n))

    # (g) simple triangular window
    _consider(_triangular_shape(n))

    # (h) sparse Dirichlet‑like samples (few non‑zero entries)
    for _ in range(1000):
        support = rng.integers(10, 2000)
        idx = rng.choice(n, size=support, replace=False)
        vals = rng.random(support)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = vals
        cand /= cand.sum()
        _consider(cand)

    # fallback – uniform if nothing survived
    if top_seqs[0] is None:
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        _consider(uniform)

    # ------------------------------------------------------------------
    # 2️⃣  Refine the elite candidates with Adam + smooth‑max
    # ------------------------------------------------------------------
    # Split the remaining time roughly equally among the elite candidates.
    time_per_candidate = (deadline - time.time()) / K
    for idx, cand in enumerate(top_seqs):
        if cand is None:
            continue
        sub_deadline = time.time() + time_per_candidate
        print(f"[refine] start Adam on elite #{idx+1} (budget {time_per_candidate:.1f}s)")
        refined = _smooth_max_optimize(cand, n, budget_s, time.time(), sub_deadline)
        # update global best if improved
        val = _objective(refined)
        if val < top_vals[0]:
            top_vals[0] = val
            top_seqs[0] = refined.copy()
            print(f"[global] improved global best to {val:.6f}")

    best_seq = top_seqs[0]
    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
