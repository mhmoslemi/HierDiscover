    sequence = [float(x) for x in sequence]  # after checks
    sequence = [max(0, x) for x in sequence]  # clamp negative to 0

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  The algorithm consists of three phases:

1️⃣  Generate a large pool of candidate shapes (uniform, Beta,
    exponential‑decay, geometric, Gaussian, power‑sine, random Dirichlet
    etc.) on a moderately large dimension (n≈12 k).  All candidates are
    normalised to Σa=1 and the best few are kept.

2️⃣  Starting from the current best candidate run a fast projected Adam
    optimiser on the exact objective
        f(a)=2·n·max(conv(a,a))
    (the sum is fixed to 1 so the denominator can be ignored).
    The sub‑gradient of the max‑convolution is built from the
    *max_active* largest convolution entries (default 20).  After each
    Adam step the vector is projected onto the probability simplex.
    An adaptive step‑size schedule, occasional Gaussian jitter and a
    simple restart‐on‑stagnation strategy are used to escape plateaus.

3️⃣  The finally best sequence is handed to the provided ``evaluate_sequence``
    function and returned.

The routine respects a 1000 s wall‑time budget and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based full convolution, simplex projection, …)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value of 2·n·max(conv(a,a)) assuming Σa = 1.
    Used for cheap comparison during the search.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex Σx=1, x≥0.
    Reference: Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the initial wide exploration
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """a_i ∝ i^{p‑1}(n‑i+1)^{q‑1},  i = 1,…,n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    tot = a.sum()
    if tot <= 0.0:
        return None
    a /= tot
    return a


def _exp_shape(c: float, n: int) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·|i-mid|)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a


def _geom_shape(r: float, n: int) -> np.ndarray:
    """
    One‑sided geometric shape a_i ∝ r^i (truncated to length n) and
    normalised to sum 1.  r∈(0,1).  r≈1 yields an almost uniform vector,
    smaller r gives a fast decay.
    """
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _gaussian_shape(sigma: float, n: int) -> np.ndarray:
    """Gaussian centred at the middle."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a


def _power_sine_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1))^p ."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    a /= a.sum()
    return a


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 20,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Return a sub‑gradient of max(b) w.r.t. a.
    The active set consists of the ``max_active`` largest entries of b,
    optionally filtered by a relative tolerance.
    """
    n = a.shape[0]
    max_val = b.max()

    # pick active indices (allow ties close to the max)
    if max_active >= b.size:
        active = np.arange(b.size)
    else:
        active = np.argpartition(b, -max_active)[-max_active:]
        mask = b[active] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            active = np.array([int(np.argmax(b))])
        else:
            active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad

# ----------------------------------------------------------------------

        nonlocal best_seq, best_val

        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _maybe_update(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

        if cand is not None:

    # (d) Symmetric exponential decay

    # (e) One‑sided geometric (decreasing) shape
    for _ in range(3000):
        r = rng.uniform(0.70, 0.99999)          # r close to 1 → nearly uniform
        cand = _geom_shape(r, n)
        _maybe_update(cand)

    # (f) Gaussian shapes
    for _ in range(2000):
        sigma = rng.uniform(0.5, n / 5.0)
        cand = _gaussian_shape(sigma, n)
        _maybe_update(cand)

    # (g) Power‑sine family
    for _ in range(2000):
        p = rng.uniform(0.1, 8.0)
        cand = _power_sine_shape(p, n)
        _maybe_update(cand)

    cand = _triangular_shape(n)

    # (i) Random Dirichlet draws (allowing very sparse vectors)
    for _ in range(2500):

    if best_seq is None:

        best_val = _objective(best_seq)

    # 2️⃣ Refine the current best using a projected Adam optimiser
    # --------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)          # first‑moment accumulator
    v = np.zeros_like(a)          # second‑moment accumulator
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    no_improve = 0
