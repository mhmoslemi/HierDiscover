"""
Propose a high‑quality step function.

Algorithm
---------
1. **Low‑dimensional global search** – we search over a 4‑parameter
   family of “skewed Beta” shapes
       a_i ∝ (i+1)^{p‑1}·(n‑i)^{q‑1}·exp(r·i)  +  u·Uniform
   using SciPy's differential evolution on a modest size `n_small`.
   This gives a diverse, near‑optimal shape without expensive high‑dim
   optimisation.

2. **Upscaling** – the best parameters are used to build a vector of the
   target length (`n_full = 8000`).  The shape is already normalised to
   sum = 1, so the objective value is comparable across lengths.

3. **Fine‑grained Adam optimisation** – we minimise a smooth
   log‑sum‑exp approximation of the L∞‑norm of the self‑convolution.
   The schedule for the learning rate and the smoothing parameter is
   aggressively annealed, and the optimisation works in log‑softmax
   space so the simplex constraints are satisfied automatically.
   Restarts are performed if progress stalls.

4. **Exact sub‑gradient polishing** – a projected sub‑gradient descent
   on the true objective (max‑convolution) is run until the time budget
   expires.

5. **Uniform blending** – occasional convex combinations with the
   uniform distribution are tested; they often smooth isolated spikes.

The whole pipeline respects a 1000 s wall‑clock budget and returns the
best sequence found.  The code contains no closures or lambdas and
uses only NumPy, SciPy and the standard library.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp weights
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # entries that correspond to the original indices
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for the smoothed objective.
    The smoothing ε is annealed from 0.5 to 1e‑12, learning‑rate from
    0.5 down to 0.005.
    """
    # work in log‑softmax space (automatically respects simplex)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.005
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 2000
    TARGET = 1.503  # stop early if we have already beaten the target

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
            if best_val <= TARGET:
                print("[Adam] target reached, exiting early")
                break
        else:
            idle += 1

        # schedule
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # already includes 2·n

        # chain rule through soft‑max (log‑softmax parameters)
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.05, size=n)

        # restart if stalled
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    """
    a = a0.copy()
    cur_val = _objective(a)
    TARGET = 1.503
    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum of contributions from all maximal diagonals
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule (decreasing)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")
                if best_val <= TARGET:
                    print("[SubG] target reached, exiting early")
                    break

        # tiny random jitter every 500 iterations
        if it % 500 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Uniform blending (helps smooth isolated spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·uniform for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Parameterised skewed‑Beta shape generator
# ----------------------------------------------------------------------
def _generate_skewed_beta(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a normalised vector from parameters:
        params = [p, q, r, u]
    a_i ∝ (i+1)^{p‑1}·(n‑i)^{q‑1}·exp(r·i)  +  u·Uniform
    """
    p, q, r, u = params
    i = np.arange(n, dtype=np.float64)
    # log‑space to avoid overflow
    log_a = (p - 1.0) * np.log(i + 1.0) + (q - 1.0) * np.log(n - i) + r * i
    # stabilise
    log_a = log_a - np.max(log_a)
    a = np.exp(log_a)
    if u > 0.0:
        a = (1.0 - u) * a + u * np.ones_like(a)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a.astype(np.float64)

    # 2️⃣  If we still have no candidate, fall back to a simple Beta draw
    # ------------------------------------------------------------------
    if best_a is None:
        cand = _generate_skewed_beta(np.array([1.0, 1.0, 0.0, 0.0]), n_full)  # uniform
        best_a = cand
        best_val = _objective(cand)
        print(f"[Fallback] uniform candidate, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Phase A – Adam optimisation (≈40 % of total budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.70 * budget_s
    print("[Phase A] Adam optimisation start")
    best_a, best_val = _adam_optimize(
        best_a, n_full,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n_full, best_val)

    # ------------------------------------------------------------------
    # 4️⃣  Phase B – Sub‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Phase B] Sub‑gradient polishing start")
        best_a, best_val = _subgradient_polish(
            best_a, n_full,
            deadline=deadline,
            start_time=phase_a_deadline,
            best_a=best_a,
            best_val=best_val,
        )
        best_a, best_val = _blend_with_uniform(best_a, n_full, best_val)

    # ------------------------------------------------------------------
    # 5️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return a plain list of floats
    return best_a.tolist()
```

This implementation adds a genuine global optimisation step (differential evolution) over a flexible four‑parameter family of skewed Beta shapes, followed by the previously proven Adam‑smooth‑max refinement and exact sub‑gradient polishing.  By allocating a sizeable portion of the time budget to the low‑dimensional global search we increase the chance of escaping the local minima that limited the earlier approach, while the later stages polish the candidate to a high‑precision solution.
