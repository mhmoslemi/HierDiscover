def run_optimize(p0, n, sum_target, start_time, deadline):
    # convert to logits
    eps=1e-12
    z = np.log(p0 + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t_iter=0
    b1=0.9; b2=0.999; adam_eps=1e-8; lr=0.1
    best_p = p0.copy()
    best_val = float('inf')
    stagnated = 0
    IMPROVE_LIMIT = 2000
    while True:
        now = time.time()
        if now >= deadline:
            break
        t_iter+=1
        # compute p and a
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        p = exp_z / exp_z.sum()
        a = p * sum_target
        # compute conv and max
        conv = _full_convolution(a)
        max_conv = conv.max()
        # objective is max_conv (since sum_target**2 = 2n)
        cur_val = max_conv
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_p = p.copy()
            stagnated = 0
            print(f"[{n}] iter {t_iter} new best = {best_val:.9f}")
        else:
            stagnated += 1
        # gradient
        tol = 1e-12
        mask = conv >= max_conv - tol
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum
        # gradient of max w.r.t a
        g_raw = _cross_corr(w, a)
        g = 2.0 * g_raw
        # subgradient wrt logits
        dot_pg = np.dot(p, g)
        grad_z = sum_target * (p * (g - dot_pg))
        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z ** 2)
        m_hat = m / (1 - b1 ** t_iter)
        v_hat = v / (1 - b2 ** t_iter)
        z -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        # stagnation handling: if no improvement for many iterations, break
        if stagnated > IMPROVE_LIMIT:
            break
    # after loop return best distribution scaled to sum_target
    best_a = best_p * sum_target
    # ensure non-neg and clip to bounds implied by evaluate_sequence (0 to 1000)
    best_a = np.clip(best_a, 0, 1000.0)
    return best_a, best_val

"""
Propose a step‑function (non‑negative heights) that minimizes the
evaluation function from the problem statement.

Algorithm
---------
1. Work in the probability simplex: a = p * S, where p≥0, Σp=1 and
   S = sqrt(2·n).  With this normalisation the evaluation reduces to
   `max(a*a)`, i.e. the maximum entry of the self‑convolution.
2. Optimize p by (sub)‑gradient descent on the exact max (not a smooth
   approximation).  The sub‑gradient of `max(a*a)` w.r.t. a is
   `g = 2 * cross_corr(w, a)`, where `w` is the uniform distribution over
   all convolution indices attaining the current maximum.
   The gradient w.r.t. the logits `z` (soft‑max parametrisation of p)
   is `∂F/∂z = S * p ⊙ (g – ⟨p,g⟩)`.
3. Use Adam for adaptive steps and inject Gaussian noise into the logits
   after a long streak without improvement (to escape flat regions).
4. Two‑phase optimisation:
   * a quick optimisation on a small problem (n≈1024) to discover a good
     shape,
   * up‑sample this shape to the final size (n≈8000) and continue the
     Adam‑descent for the remaining time budget.
5. Keep the best sequence seen (according to the exact objective)
   and finally return it.

The routine respects the 1000 s budget, prints progress and finally
evaluates the returned sequence with `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself using a real FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation needed for the sub‑gradient:
        g_i = Σ_k w_k a_{k-i}   (i = 0 … n‑1)

    Implemented via FFT as a convolution of w with the reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = len(w) + n - 1          # (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the step function) that
    attempts to minimise `evaluate_sequence`.  The routine runs for at
    most `budget_s` seconds and prints progress information.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a tiny safety margin for the final secure call.
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    small_n = 1024                     # quick search dimension
    large_n = 8000                     # final dimension (as in the
                                       # previous best solution)
    sum_target_small = math.sqrt(2.0 * small_n)
    sum_target_large = math.sqrt(2.0 * large_n)

    # Adam hyper‑parameters (shared by both phases)
    lr = 0.07
    b1, b2 = 0.9, 0.999
    adam_eps = 1e-8
    stagnation_limit = 2500           # reset after this many non‑improvements
    noise_scale = 0.12                # magnitude of the injected noise

    # ------------------------------------------------------------------
    # Helper: run Adam optimisation for a given size and start point.
    # Returns (best_a, best_val) where best_val = max(a*a) (the true
    # objective after the S‑normalisation).
    # ------------------------------------------------------------------
    def _optimise(p0: np.ndarray,
                  n: int,
                  S: float,
                  phase_deadline: float):
        """Adam optimisation on the simplex, using the exact max‑subgradient."""
        # initialise logits for a soft‑max parametrisation of p
        eps = 1e-12
        z = np.log(p0 + eps)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t_iter = 0

        best_p = p0.copy()
        best_val = float('inf')
        stagnated = 0

        while True:
            now = time.time()
            if now >= phase_deadline:
                break
            t_iter += 1

            # ----- logits → probability vector (sum = 1) -----
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            p = exp_z / exp_z.sum()          # Σ p = 1
            a = p * S                         # scale to the required sum

            # ----- evaluate (exact max of self‑convolution) -----
            conv = _full_convolution(a)       # length 2n‑1
            max_conv = conv.max()
            cur_val = max_conv                # because S² = 2n ⇒ objective = max_conv

            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_p = p.copy()
                stagnated = 0
                print(f"[{n}] iter {t_iter:07d}  new best = {best_val:.9f}")
            else:
                stagnated += 1

            # ----- sub‑gradient of max(a*a) w.r.t. a -----
            # w is the uniform distribution over all indices attaining the max
            tol = 1e-12
            mask = conv >= max_conv - tol
            w = mask.astype(float)
            w_sum = w.sum()
            if w_sum == 0.0:                 # extremely unlikely, but safe
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
                w_sum = 1.0
            w /= w_sum

            # gradient g = ∂max / ∂a
            g_raw = _cross_corr(w, a)        # length n
            g = 2.0 * g_raw

            # ----- chain rule back to logits z -----
            dot_pg = np.dot(p, g)            # scalar ⟨p,g⟩
            grad_z = S * (p * (g - dot_pg))

            # ----- Adam update -----
            m = b1 * m + (1.0 - b1) * grad_z
            v = b2 * v + (1.0 - b2) * (grad_z ** 2)
            m_hat = m / (1.0 - b1 ** t_iter)
            v_hat = v / (1.0 - b2 ** t_iter)
            z -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # ----- escape flat regions -----
            if stagnated > stagnation_limit:
                # Add a modest Gaussian perturbation to the logits.
                noise = np.random.normal(scale=noise_scale, size=n)
                z += noise
                stagnated = 0
                print(f"[{n}] reset: added noise at iter {t_iter}")

        # Return the best (scaled) sequence found in this phase.
        best_a = best_p * S
        best_a = np.clip(best_a, 0.0, 1000.0)   # respect the hard caps of evaluate_sequence
        return best_a, best_val

    # ------------------------------------------------------------------
    # Phase 1 – quick optimisation on a small dimension.
    # ------------------------------------------------------------------
    # Initialise p0 for the small problem.  If a previous construction is
    # available (global variable `height_sequence_1`), down‑sample it;
    # otherwise start from a uniform distribution.
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        old_len = old.shape[0]
        xs_old = np.linspace(0.0, 1.0, old_len)
        xs_small = np.linspace(0.0, 1.0, small_n)
        a_small = np.interp(xs_small, xs_old, old)
        a_small = np.maximum(a_small, 0.0)
        sum_a_small = a_small.sum()
        if sum_a_small < 1e-12:
            p0_small = np.full(small_n, 1.0 / small_n)
        else:
            p0_small = a_small / sum_a_small
        print("[init] interpolated old solution for the small phase")
    except Exception:
        p0_small = np.full(small_n, 1.0 / small_n)
        print("[init] uniform start for the small phase")

    # Allocate roughly 20 % of the time budget to the small phase.
    small_deadline = start_time + budget_s * 0.20
    if small_deadline > deadline:
        small_deadline = deadline

    best_a_small, best_val_small = _optimise(p0_small,
                                             small_n,
                                             sum_target_small,
                                             small_deadline)
    print(f"[phase small] best value = {best_val_small:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to the final dimension and continue optimisation.
    # ------------------------------------------------------------------
    # Interpolate the small‑phase result to the final length.
    xs_small = np.linspace(0.0, 1.0, small_n)
    xs_large = np.linspace(0.0, 1.0, large_n)
    a_large = np.interp(xs_large, xs_small, best_a_small)
    a_large = np.maximum(a_large, 0.0)
    p0_large = a_large / a_large.sum()

    # Run the long optimisation until the overall deadline.
    best_a_large, best_val_large = _optimise(p0_large,
                                             large_n,
                                             sum_target_large,
                                             deadline)
    print(f"[phase large] best value = {best_val_large:.9f}")

    # Pick the better of the two phases (normally the large phase wins).
    if best_val_small < best_val_large:
        final_a = best_a_small
        final_val = best_val_small
    else:
        final_a = best_a_large
        final_val = best_val_large

    # ------------------------------------------------------------------
    # Final secure evaluation (the only place we call `evaluate_sequence`).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(final_a.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.9f}")
    except Exception as exc:           # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_a.tolist()
