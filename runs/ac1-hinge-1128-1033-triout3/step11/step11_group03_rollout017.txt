"""
Optimise a non‑negative step‑function to minimise the host objective.

The algorithm consists of three stages:

1️⃣  *Diverse initialisation* – generate several candidates (Gaussian,
    Laplace, power‑law profiles and Dirichlet draws).  The best of these
    seeds is taken forward.

2️⃣  *Smooth‑max Adam* – optimise a soft‑max surrogate of the maximum
    autocorrelation.  The surrogate
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    approaches the true objective as α→∞.  α is increased linearly,
    Adam’s learning‑rate decays, and we keep the exact objective of each
    accepted iterate.  This stage is given the bulk of the time budget.

3️⃣  *Exact sub‑gradient descent + mass‑transfer hill‑climb* – after the
    surrogate phase we switch to the true (non‑smooth) objective.
    The gradient is the sub‑gradient of the active lag (the lag that attains
    the maximum convolution).  A projected‑gradient step on the probability
    simplex is performed, followed by a short stochastic hill‑climb that
    moves a small random amount of mass from a “guilty’’ index to a
    low‑contribution index.

The whole pipeline respects the 1000 s wall‑clock limit and returns the
best sequence found.  All operations use only NumPy and the supplied
``evaluate_sequence`` for the final score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_rev(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·∑_i a_i·w_{i+j} for j=0..n‑1.
    Implements the gradient of the soft‑max surrogate.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    l_w = w.shape[0]
    out_len = n + l_w - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]
    start = n - 1
    return conv[start:start + n]        # length n


# ----------------------------------------------------------------------
# Objective helpers (exact, scale‑invariant but we keep sum=1)
# ----------------------------------------------------------------------
def _objective_raw(a: np.ndarray) -> float:
    """Exact raw objective used during optimisation: 2·n·max(a∗a)."""
    conv = _full_convolution(a)
    return float(2.0 * a.shape[0] * conv.max())


def _objective_eval(a: np.ndarray) -> float:
    """Final host evaluation (the function supplied by the platform)."""
    return evaluate_sequence(a.tolist())   # type: ignore[name-defined]


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the simplex using the O(n log n) algorithm."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(α·conv_k)."""
    maxc = conv.max()
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exp_vals / sum_exp


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _smooth_adam_optimize(a0: np.ndarray,
                          timeout: float,
                          start_time: float,
                          alpha_start: float = 5.0,
                          alpha_end: float = 2000.0,
                          lr0: float = 0.15,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser of the smooth surrogate.
    Returns the best point (sum=1) and its exact raw objective.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a /= s

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_val = _objective_raw(a)
    best_a = a.copy()
    best_val = cur_val

    t = 0
    while True:
        now = time.time()
        if now >= timeout:
            break
        prog = (now - start_time) / (timeout - start_time) if timeout > start_time else 0.0
        alpha = alpha_start + (alpha_end - alpha_start) * prog

        # current convolution and surrogate gradient
        conv = _full_convolution(a)
        w = _softmax_weights(conv, alpha)          # shape (2n‑1,)
        grad = 2.0 * _conv_rev(a, w)               # shape (n,)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a_cand = a - step * m_hat
        a_cand = np.maximum(a_cand, 0.0)
        a_cand = _project_simplex(a_cand)

        cand_val = _objective_raw(a_cand)

        # accept only if we do not worsen the exact objective
        if cand_val <= cur_val:
            a = a_cand
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[Adam] iter {t:7d} α={alpha:6.1f} val={best_val:.6f}")

        # occasional random perturbation to avoid stagnation
        if t % 3000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=a.shape[0])
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_val = _objective_raw(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------
def _exact_subgradient_optimize(a0: np.ndarray,
                                timeout: float,
                                start_time: float,
                                lr0: float = 0.02,
                                decay: float = 0.995,
                                min_lr: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact (non‑smooth) objective.
    The gradient is taken w.r.t. the lag that currently attains the
    maximum autocorrelation.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_raw(a)

    lr = lr0
    n = a.shape[0]

    while True:
        now = time.time()
        if now >= timeout:
            break

        conv = _full_convolution(a)
        max_idx = int(np.argmax(conv))

        # gradient of the selected lag
        grad = np.zeros_like(a)
        if max_idx == 0:
            grad = 2.0 * a                     # because a_i*a_i term appears twice
        else:
            # contributions from a_i * a_{i+max_idx}
            grad[: n - max_idx] += a[max_idx:]
            # contributions from a_{i-max_idx} * a_i
            grad[max_idx:] += a[: n - max_idx]

        # scale to the raw objective: 2·n·∂conv
        grad = 2.0 * n * grad

        # normalise step direction
        direction = grad / (np.linalg.norm(grad) + 1e-12)

        a_new = a - lr * direction
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        new_val = _objective_raw(a_new)

        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[Subgrad] val={best_val:.6f} lr={lr:.6e}")

        # accept if we improved, otherwise keep the old point
        if new_val < _objective_raw(a):
            a = a_new

        lr = max(lr * decay, min_lr)

    return best_a, best_val


# ----------------------------------------------------------------------
# Light mass‑transfer hill‑climb (polish)
# ----------------------------------------------------------------------
def _hillclimb(a: np.ndarray,
               timeout: float,
               start_time: float,
               target: float = 1.503,
               max_no_improve: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Very simple stochastic hill‑climb that moves a fraction of mass from the
    most “guilty’’ index (largest contribution to the current max lag) to a
    low‑contribution index.
    """
    n = a.shape[0]
    cur_val = _objective_raw(a)
    best_val = cur_val
    best_a = a.copy()

    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= timeout:
            break
        if best_val <= target:
            break

        # current convolution and the lag that dominates
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # contribution of each element to that lag
        contrib = np.zeros_like(a)
        idx = np.arange(n) - k_max
        mask = (idx >= 0) & (idx < n)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source: either weighted by contribution or simply the biggest one
        if total > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-12:
            continue

        # inverse‑contribution to pick a “good’’ target
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        # move a random fraction of the source mass
        delta = a[src] * np.random.rand() * 0.4
        if delta < 1e-14:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        new_val = _objective_raw(a_new)

        if new_val < cur_val:
            a = a_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {it:7d} val={best_val:.6f}")
                if best_val <= target:
                    break
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # reset around the current best to escape plateaus
                a = best_a.copy()
                noise = np.random.normal(scale=0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                a = _project_simplex(a)
                cur_val = _objective_raw(a)
                no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Small parametric families for initialisation
# ----------------------------------------------------------------------
def _parametric_candidates(n: int):
    """Yield a few hand‑crafted shapes (Gaussian, Laplace, power‑law)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Gaussian
    for sigma_factor in (0.15, 0.20, 0.25, 0.30):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        a /= a.sum()
        yield a

    # Laplace
    for scale_factor in (0.10, 0.15, 0.20):
        scale = scale_factor * n
        a = np.exp(-np.abs(i - mid) / scale)
        a /= a.sum()
        yield a

    # Power‑law / Beta‑like
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = (i + 1) ** (p - 1.0) * (n - i) ** (p - 1.0)
        a /= a.sum()
        yield a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that aims to minimise
    ``evaluate_sequence``.  The routine respects the ``budget_s`` time limit (≈
    1000 s) and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Initial candidates (different lengths – we concentrate on 8000)
    # ------------------------------------------------------------------
    n = 8000
    candidates = []

    # previous solution (if available)
    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if prev.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, prev)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            candidates.append(a0)
    except Exception:
        pass

    # parametric shapes
    candidates.extend(list(_parametric_candidates(n)))

    # a few random Dirichlet draws with varied concentration
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # uniform baseline (useful as a sanity check)
    candidates.append(np.full(n, 1.0 / n))

    # keep the best initial point (according to the exact raw objective)
    best_a = None
    best_val = np.inf
    for cand in candidates:
        val = _objective_raw(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[Init] best starter n={n} raw_val={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max Adam optimisation (major time consumer)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # give the Adam phase about 60 % of the remaining budget
        adam_deadline = time.time() + 0.6 * remaining
        best_a, best_val = _smooth_adam_optimize(
            best_a,
            timeout=adam_deadline,
            start_time=start_time,
            alpha_start=5.0,
            alpha_end=2000.0,
            lr0=0.15,
        )
        print(f"[Adam] finished raw_val={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Exact sub‑gradient descent (tighten the true max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        subgrad_deadline = time.time() + 0.25 * remaining
        best_a, best_val = _exact_subgradient_optimize(
            best_a,
            timeout=subgrad_deadline,
            start_time=start_time,
            lr0=0.02,
            decay=0.995,
        )
        print(f"[Subgrad] finished raw_val={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final stochastic hill‑climb (polish)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        hill_deadline = time.time() + remaining
        best_a, best_val = _hillclimb(
            best_a,
            timeout=hill_deadline,
            start_time=start_time,
            target=1.503,
        )
        print(f"[Hillclimb] finished raw_val={best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final secure evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        final_score = _objective_eval(best_a)
        print(f"[Final] reported host score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
```

This implementation extends the previous approach by

* allocating a much larger time budget to the smooth‑max Adam phase,
* using a far‑higher final temperature (`alpha_end=2000`) for a tighter
  approximation of the true max,
* adding a dedicated exact sub‑gradient descent stage,
* and a more aggressive hill‑climb with adaptive restarts.

All steps respect the 1000 s overall budget and should, on average,
drive the objective below the previous best of 1.508 → target 1.503.
