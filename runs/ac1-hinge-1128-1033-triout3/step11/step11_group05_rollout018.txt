    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):

    # Convert all elements to float for consistency

    sequence = [max(0, x) for x in sequence]

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Cheap proxy for evaluate_sequence when sum(a) == 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _maybe_update(a0, best_seq, best_val)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Wide exploration: a few classic families (kept for safety)
    # ------------------------------------------------------------------
    # (a) simple triangular window – a cheap baseline
    i = np.arange(1, n + 1, dtype=np.float64)
    tri = np.minimum(i, n + 1 - i)
    tri /= tri.sum()
    _maybe_update(tri, best_seq, best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Main search: random sparse support (Dirichlet‑like samples)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    # Choose support size generous enough to push the objective below 1.5.
    # Experimentally m ≈ 1500–2500 works well for n = 8000.
    min_m = 1200
    max_m = 2500
    attempts = 0

    while time.time() < deadline:
        m = rng.integers(min_m, max_m + 1)          # support cardinality
        idx = rng.choice(n, size=m, replace=False)
        candidate = np.zeros(n, dtype=np.float64)
        candidate[idx] = 1.0 / m                     # exact sum = 1 (up to FP)
        _maybe_update(candidate, best_seq, best_val)

        attempts += 1
        if attempts % 5000 == 0:
            # loose a bit of time for a quick report
            elapsed = time.time() - start_time
            print(f"[info] {attempts} random supports tried – "
                  f"time {elapsed:.1f}s – best {best_val[0]:.6f}")

        # safety break if we’re close to time limit
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Light local refinement (optional but cheap)
    # ------------------------------------------------------------------
    if best_seq[0] is not None:
        a = best_seq[0].copy()
        # a is on the simplex (sum≈1).  Perform a few coordinate‑wise tweaks.
        for _ in range(2000):
            # pick two distinct indices
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] < 1e-12:
                continue
            # move a tiny epsilon from i to j
            eps = min(a[i], 1e-4) * rng.random()
            a[i] -= eps
            a[j] += eps
            # re‑normalise (to fight floating drift)
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a /= a_sum
            # cheap check
            val = _objective(a)
            if val < best_val[0]:
                best_val[0] = val
                best_seq[0] = a.copy()
                # occasional progress report
                print(f"[refine] improved to {val:.6f}")

            # stop early if time runs out
            if time.time() > deadline - 0.5:
                break

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation (the official score)
    # ------------------------------------------------------------------
    final_seq = best_seq[0]
    if final_seq is None:
        # fallback: uniform distribution
        final_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("[fallback] using uniform distribution")

    try:
        final_score = evaluate_sequence(final_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = float('inf')

    return final_seq.tolist()
