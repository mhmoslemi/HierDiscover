"""
Improvements over the reference implementation
------------------------------------------------
* Uses a larger problem size (n ≈ 10 000) – more degrees of freedom give a
  flatter self‑convolution.
* Wider parametric sweep (beta, exponential, Gaussian, power‑sine,
  triangular and mixtures of two beta families) to obtain a strong initial
  point.
* Refines the candidate with an **Adam‑style exponentiated‑gradient**
  mirror‑descent.  The adaptive learning‑rate per coordinate keeps the
  updates well‑behaved even when the sub‑gradient varies a lot.
* Employs a larger active set (top‑15 lag‑values) and a tighter relative
  tolerance when building the sub‑gradient.
* Adds periodic random jitter and a simple multi‑restart scheme after long
  stagnation periods.
* All steps respect the supplied time‑budget (default 1000 s).  The
  best sequence found so far is continuously tracked and finally fed to
  ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``x`` with itself using FFT.
    Length of the result is ``2*len(x)-1``.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value of ``2·n·max(conv(a,a))`` assuming ``sum(a)=1``.
    Used for quick comparison during the exploratory phase.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex:
        Σ x_i = 1 , x_i ≥ 0
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape families for the wide exploration
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1}  (i = 1,…,n)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_shape(c: float, n: int) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·|i‑mid|)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a


def _gaussian_shape(sigma: float, n: int) -> np.ndarray:
    """Gaussian centred at the centre."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a


def _power_sine_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1))^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    a /= a.sum()
    return a


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a


def _beta_mixture(p1, q1, p2, q2, w, n):
    """Convex mixture of two beta shapes."""
    a1 = _beta_shape(p1, q1, n)
    a2 = _beta_shape(p2, q2, n)
    if a1 is None or a2 is None:
        return None
    a = w * a1 + (1.0 - w) * a2
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 15,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Sub‑gradient of ``max(b)`` w.r.t. ``a``.
    ``b`` is the full self‑convolution of ``a``.  The active set consists
    of the ``max_active`` largest entries of ``b`` (subject to the relative
    tolerance).
    """
    n = a.shape[0]
    max_val = b.max()
    # indices of the (up to) max_active largest entries
    if max_active >= b.size:
        active = np.arange(b.size)
    else:
        active = np.argpartition(b, -max_active)[-max_active:]
        mask = b[active] >= max_val * (1.0 - rel_tol)
        active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution from a[k-idx] for each idx
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad

            no_improve = 0
