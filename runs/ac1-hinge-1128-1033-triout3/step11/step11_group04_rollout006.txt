"""
Propose a non‑negative step‑function (list of heights) that minimizes the
provided ``evaluate_sequence`` objective.

The algorithm follows several complementary stages, each respecting the
global 1000 s wall‑clock budget:

0️⃣  *Exploratory seeding* – a few hand‑crafted shapes and many random
    Beta/Dirichlet draws are evaluated; the best among them becomes the
    seed for the subsequent optimisation.

1️⃣  *Adam on a smooth max* – the non‑smooth objective
    ``2·n·max(conv(a,a))`` is replaced by a log‑sum‑exp (smooth‑max)
    approximation.  Adam is run on the logits of a soft‑max (so the
    simplex constraint Σa=1 is always satisfied).  The smoothing
    parameter ε is annealed from 0.5 down to 1e‑9.

2️⃣  *Aggressive hill‑climbing* – at each step a “source’’ index that
    contributes most to the current maximal convolution entry and a
    “target’’ index with a low contribution are selected.  A small chunk
    of mass is moved from source to target and the move is accepted
    either if it improves the objective or with a Metropolis probability
    that slowly cools.

3️⃣  *L‑BFGS‑B refinement* – the same smooth‑max objective is minimised
    using a quasi‑Newton method.  A continuation on ε (0.5 → 1e‑10) is
    performed; after each stage the best vector (according to the exact
    objective) is kept.

4️⃣  *Final polishing* – a short projected sub‑gradient descent cleans
    up any remaining noise.

All intermediate best values are printed; at the end the final sequence
is verified with ``evaluate_sequence`` and returned as a plain
``list[float]`` whose ℓ₁‑norm is 1.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max (no variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex for the loss L(a)=2·n·smooth_max(a)."""
    w = np.log(a0 + 1e-12)               # work in log‑softmax space

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 8000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule learning rate and smoothing
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # grad already contains 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to avoid stagnation
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)

        # restart if idle for too long
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (fast O(n) updates)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """Local source‑target mass‑transfer hill‑climbing with simulated annealing."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.8, 0.001
    temp_start, temp_end = 0.03, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 90000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index: deterministic roulette proportional to contribution
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index: favour low contributions
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation (it={iteration}), re‑perturb")
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B refinement on the smooth‑max objective
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B optimisation of the smoothed objective
    L(a)=2·n·smooth_max(a) using a continuation on ε.
    """
    w = np.log(a0 + 1e-12)               # logits (no constraint needed)

    # a decreasing schedule for the smoothing parameter ε
    eps_schedule = [5e-1, 1e-1, 5e-2, 1e-2, 5e-3,
                    1e-3, 5e-4, 1e-4, 5e-5, 1e-5,
                    5e-6, 1e-6, 5e-7, 1e-7, 5e-8, 1e-8, 5e-9, 1e-9]

    for eps in eps_schedule:
        now = time.time()
        if now >= deadline:
            break

        # allocate a modest budget for this stage (≈5 % of remaining time)
        remaining = deadline - now
        max_iter = max(1, int(0.04 * remaining))   # a few hundred iterations at most

        res = minimize(fun=_lbfgs_objective,
                       x0=w,
                       args=(n, eps),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter,
                                'disp': False,
                                'gtol': 1e-12})

        w = res.x

        # evaluate the exact objective on the newly‑obtained point
        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[L‑BFGS] eps={eps:.1e} new best = {best_val:.6f}")

    return best_a, best_val


def _lbfgs_objective(w: np.ndarray, n: int, eps: float):
    """
    Returns both loss and gradient (w.r.t. the logits w) for the smoothed
    objective.  The gradient supplied to L‑BFGS‑B is ∂L/∂w.
    """
    # soft‑max to obtain a feasible probability vector
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = a.sum()
    a = a / a_sum

    smooth_val, grad_a = _smooth_max_and_grad(a, eps)   # grad_a already has 2·n factor
    loss = 2.0 * n * smooth_val                         # true objective value for this eps

    # chain rule ∂L/∂w = (∂L/∂a)·∂a/∂w  ;  ∂a/∂w = diag(a) - a aᵀ
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return loss, grad_w


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to break flat regions
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta family candidate
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing loop that performs random source→target
    transfers with a relatively large step size.  It runs only on the
    leftover part of the time budget.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)            # linear cooling
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # tiny jitter to keep diversity
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem size (chosen to balance granularity and speed)
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Initial random / handcrafted candidates
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # try to reuse a previously discovered solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # many random Beta draws
    for _ in range(2000):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Dirichlet draws with a range of concentrations
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(30):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # sinusoidal shape (all positive)
    i = np.arange(n, dtype=np.float64)
    cand = np.sin(np.pi * (i + 0.5) / n)          # zero‑free sine wave
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    # cosine‑square shape (peaks in the middle)
    cand = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    print(f"[Init] best after random search = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam optimisation (≈20 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.20 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase B – Hill‑climbing (≈45 % of the budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.65 * budget_s
    best_a, best_val = _hillclimb(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase C – L‑BFGS‑B refinement (≈20 % of the budget)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.85 * budget_s
    best_a, best_val = _lbfgs_optimize(
        best_a, n,
        deadline=phase_c_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Phase D – Final simulated‑annealing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_anneal(
            best_a, n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            rng=rng,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
