"""
Propose a step‑function that (approximately) minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))²

The algorithm works on the probability simplex (∑a = 1) and exploits the
scale‑invariance of the objective.  The search space is reduced by
enforcing mirror symmetry a[i] = a[n‑1‑i]; only the left half is
optimised (≈4000 variables for n = 8192).  Optimisation proceeds in three
phases:

1. **Adam on a smooth soft‑max approximation** – the convolution max is
   replaced by log‑sum‑exp with an annealed smoothing factor α.
   A learning‑rate schedule and occasional random jitter avoid stagnation.

2. **Entropy‑mirror descent polishing** – after the Adam phase the exact
   max entry of the convolution is minimised by exponentiated gradient
   updates (the same idea as in the reference implementation but with
   many more steps).

3. **Optional second Adam run** – a short fine‑tuning pass with a much
   larger α (≈800) refines the solution.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found (as a plain ``list[float]``).  If a prior
solution ``height_sequence_1`` is available it is symmetrised and used as
the starting point.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of x with y (length len(x)+len(y)‑1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy mirror‑descent polishing (exact max entry)
# ----------------------------------------------------------------------


def _max_refine(a: np.ndarray, steps: int = 3000, eta_start: float = 0.45) -> np.ndarray:
    """
    Perform entropy‑mirror descent on the exact max entry of the
    convolution of ``a`` with itself.  ``a`` is assumed to sum to 1 and
    contain non‑negative entries.
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # gradient of the max entry w.r.t. a (sub‑gradient)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        eta = eta_start * (1.0 - t / steps)   # linear decay
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a.fill(1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The optimisation enforces mirror
    symmetry and uses a combination of Adam on a soft‑max surrogate and
    entropy‑mirror descent on the exact objective.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.8       # small safety margin

    # --------------------------------------------------------------
    # Problem size – 8192 (power of two, symmetric)
    # --------------------------------------------------------------
    n = 8192
    half = n // 2                           # work only on the left half

    # --------------------------------------------------------------
    # Initialise the left half (v).  If a previous solution exists,
    # symmetrise it and use it; otherwise start from a smooth Beta shape.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old = np.maximum(old, 0.0)
        old /= old.sum()
        # symmetrise
        sym = (old + old[::-1]) * 0.5
        v0 = sym[:half]
        # ensure the half sums to 0.5 (full vector will sum to 1)
        v0 = v0 * (0.5 / v0.sum())
        print("[init] symmetrised previous solution")
    except Exception:
        # smooth Beta(2,2) on the half interval
        i = np.arange(1, half + 1, dtype=np.float64)
        p = q = 2.0
        v0 = (i ** (p - 1.0)) * ((half - i + 1) ** (q - 1.0))
        v0 = v0 / v0.sum()
        v0 = v0 * 0.5                     # half‑mass (full vector sums to 1)
        print("[init] generated Beta(2,2) start")

    # work in log‑domain to maintain positivity automatically
    eps = 1e-30
    z = np.log(v0 + eps)

    # --------------------------------------------------------------
    # Adam hyper‑parameters
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v_hat = np.zeros_like(z)

    # learning‑rate schedule (high → low)
    lr_start, lr_end = 0.20, 0.002

    # smoothing parameter α schedule (soft‑max → hard max)
    alpha_max = 800.0

    # bookkeeping
    best_a = None
    best_obj = np.inf
    best_iter = 0
    it = 0
    stagnation = 0

    # --------------------------------------------------------------
    # Main Adam optimisation loop (soft‑max surrogate)
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress fraction (0 → 1)
        prog = (now - start) / (deadline - start)

        # schedules
        lr = lr_start * (1.0 - prog) + lr_end * prog
        alpha = 1.0 + (alpha_max - 1.0) * (prog ** 1.8)

        # ----------------------------------------------------------
        # Recover v from logits and enforce exact half‑mass = 0.5
        # ----------------------------------------------------------
        v = np.exp(z)
        s = v.sum()
        if s == 0.0:
            v.fill(0.5 / half)
        else:
            v = v * (0.5 / s)

        # Build full symmetric vector a
        a = np.concatenate([v, v[::-1]])

        # ----------------------------------------------------------
        # Full self‑convolution (needed for soft‑max)
        # ----------------------------------------------------------
        conv = _full_convolution(a)            # length 2·n‑1

        # ----------------------------------------------------------
        # Soft‑max of the convolution (stable)
        # ----------------------------------------------------------
        maxc = conv.max()
        exp_scaled = np.exp(alpha * (conv - maxc))
        sumexp = exp_scaled.sum()
        weights = exp_scaled / sumexp          # Σ weights = 1
        # smooth approximation of max(conv)
        log_sumexp = maxc + math.log(sumexp)
        obj_smooth = 2.0 * n * (log_sumexp / alpha)

        # ----------------------------------------------------------
        # Keep the best exact objective seen so far
        # ----------------------------------------------------------
        exact_obj = 2.0 * n * maxc
        if exact_obj + 1e-12 < best_obj:
            best_obj = exact_obj
            best_a = a.copy()
            best_iter = it
            stagnation = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}  t={now-start:.1f}s")
        else:
            stagnation += 1

        # ----------------------------------------------------------
        # Gradient of the smooth surrogate w.r.t. a
        # ----------------------------------------------------------
        rev_a = a[::-1]
        conv_rev = _full_convolution_two(rev_a, weights)   # length 3·n‑2
        g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]            # slice length n

        # ----------------------------------------------------------
        # Restrict gradient to the left half (variables v)
        # ----------------------------------------------------------
        grad_v = g_a[:half] + g_a[-half:][::-1]

        # ----------------------------------------------------------
        # Chain rule through the softmax (z = log v)
        # ----------------------------------------------------------
        dot = np.dot(grad_v, v)
        grad_z = v * (grad_v - dot)

        # ----------------------------------------------------------
        # Adam update on logits
        # ----------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        z = z - lr * m_hat / (np.sqrt(v_hat_corr) + eps_adam)

        # ----------------------------------------------------------
        # Simple escape mechanism if we have been stuck for a while
        # ----------------------------------------------------------
        if stagnation >= 2500:
            z += np.random.normal(scale=0.02, size=half)
            stagnation = 0
            print(f"[reset] jitter added at it={it}")

    # ------------------------------------------------------------------
    # First polishing phase (exact max entry) – long mirror descent
    # ------------------------------------------------------------------
    if best_a is not None:
        a_refined = _max_refine(best_a, steps=3000, eta_start=0.45)
        conv_ref = _full_convolution(a_refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = a_refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Optional short second Adam run with very large α (fine‑tuning)
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5.0:
        # initialise from the polished solution
        v = best_a[:half]
        z = np.log(v + eps)
        m = np.zeros_like(z)
        v_hat = np.zeros_like(z)
        it2 = 0
        while time.time() < deadline - 2.0:
            prog = it2 / 3000.0            # arbitrary small schedule
            lr = 0.01 * (1.0 - prog) + 0.001
            alpha = 400.0 + 400.0 * prog   # α from 400 → 800
            # recover v
            v = np.exp(z)
            v = v * (0.5 / v.sum())
            a = np.concatenate([v, v[::-1]])
            conv = _full_convolution(a)
            maxc = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc))
            sumexp = exp_scaled.sum()
            weights = exp_scaled / sumexp
            # gradient
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]
            grad_v = g_a[:half] + g_a[-half:][::-1]
            dot = np.dot(grad_v, v)
            grad_z = v * (grad_v - dot)
            # Adam step
            it2 += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it2)
            v_hat_corr = v_hat / (1.0 - beta2 ** it2)
            z = z - lr * m_hat / (np.sqrt(v_hat_corr) + eps_adam)

            # keep best
            exact_obj = 2.0 * n * maxc
            if exact_obj < best_obj:
                best_obj = exact_obj
                best_a = a.copy()
                print(f"[fine] new best O={best_obj:.6f}  it2={it2}")

    # ------------------------------------------------------------------
    # Final polishing (a few more mirror‑descent steps)
    # ------------------------------------------------------------------
    if best_a is not None:
        a_final = _max_refine(best_a, steps=1500, eta_start=0.40)
    else:
        a_final = np.concatenate([v0, v0[::-1]])

    final_seq = a_final.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (best O≈{best_obj:.6f})")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
