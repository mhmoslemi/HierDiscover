"""
Non‑negative step‑function optimiser (hard‑max subgradient descent).

The optimiser works directly on a probability vector `a` (sum = 1) using a
soft‑max parametrisation `z`.  At each iteration we

  1. compute the full self‑convolution `c = a * a`,
  2. locate the indices where `c` attains (or almost attains) its maximum,
  3. form a weight vector `w` that is uniform on those “active’’ indices,
  4. obtain the exact sub‑gradient of `max(c)` w.r.t. `a` as
        g = 2 · cross_corr(w, a)
     (this follows from ∂(a*a)[k]/∂a = 2·a shifted by k),
  5. back‑propagate through the soft‑max to get a gradient w.r.t. `z`,
  6. apply Adam updates to `z`,
  7. keep the best sequence seen (evaluated with the true `evaluate_sequence`).

A simple stagnation detection adds Gaussian noise to the logits to escape
flat regions.  The routine respects a 1000 s budget and finally returns the
best sequence discovered.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of x with itself (real FFT, length 2·len(x)−1).
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k-i} for i = 0..len(a)-1.
    Uses FFT; returns a real vector of length len(a).

    `w` must be the weight vector of length 2·len(a)−1 (the length of the full
    convolution), while `a` is the probability vector.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]

    # ----------------------------------------------------------
