def propose_candidate(...):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0 # safety

    # problem size: choose n based on budget (maybe 12000)
    n = 12000

    # helper to compute objective O = 2*n*conv.max()
    def obj_of(a):
        conv = _full_convolution(a)
        return 2.0 * n * conv.max()

    best_a = None
    best_obj = np.inf

    # Pre-run: maybe use previous solution if available.
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        old_len = old.shape[0]
        # Interpolate to length n
        x_new = np.linspace(0,1,n)
        x_old = np.linspace(0,1,old_len)
        a0 = np.interp(x_new, x_old, old)
        a0 = np.maximum(a0,0)
        a0 = a0 / a0.sum()
        # run a short polishing on this initial?
        # We'll treat it as one start.
        start_vecs = [a0]
    except Exception:
        start_vecs = []

    # add randomly generated starts
    for _ in range(6): # generate 6 random seeds
        # random choose method
        if np.random.rand() < 0.5:
            # beta shape
            p = np.random.uniform(0.5, 5.0)
            q = np.random.uniform(0.5, 5.0)
            i = np.arange(1, n+1, dtype=np.float64)
            a = (i ** (p-1)) * ((n - i + 1) ** (q -1))
            a = a / a.sum()
        else:
            # Dirichlet: concentration param alpha
            alpha = np.random.uniform(0.1, 5.0)
            g = np.random.gamma(alpha, size=n)
            a = g / g.sum()
        start_vecs.append(a)

    # Now for each start, run Adam for a limited time, keep track of best.
    for init_a in start_vecs:
        # check remaining time
        now = time.time()
        if now >= deadline - 30:  # ensure enough for polish
            break
        # run optimizer for at most 30 seconds
        remaining = deadline - now - 30
        run_time = min(30.0, remaining)
        # run optimization - similar to previous but limited.

        # initialize logits
        eps = 1e-30
        z = np.log(init_a + eps)

        # Adam hyperparameters
        beta1, beta2 = 0.9, 0.999
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        eps_adam = 1e-8

        lr_start, lr_end = 0.2, 0.01
        alpha_max = 400.0

        it = 0
        stagnation = 0
        best_local_a = None
        best_local_obj = np.inf

        loop_start = time.time()
        while True:
            tnow = time.time()
            if tnow - loop_start >= run_time:
                break

            # compute progress fraction relative to run_time
            prog = (tnow - loop_start) / run_time
            # learning rate schedule
            lr = lr_start * (1.0 - prog) + lr_end * prog
            # alpha smoothing grows from 1 to alpha_max
            alpha = 1.0 + (alpha_max - 1.0) * (prog ** 1.5)

            # recover probabilities
            a = np.exp(z)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

            # full convolution
            conv = _full_convolution(a)
            maxc = conv.max()
            # objective exact for evaluation
            exact_obj = 2.0 * n * maxc

            # keep best
            if exact_obj < best_local_obj - 1e-12:
                best_local_obj = exact_obj
                best_local_a = a.copy()
                stagnation = 0
                # print improvement for debugging
                print(f"[start] local new best O={best_local_obj:.6f} it={it}")

            else:
                stagnation += 1

            # compute softmax weights
            maxc_alpha = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc_alpha))
            sumexp = exp_scaled.sum()
            weights = exp_scaled / sumexp

            # gradient via convolution
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            g_a = 4.0 * n * conv_rev[n-1:n-1+n]

            # chain rule through softmax (a = softmax(z))
            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional jitter to escape
            if stagnation >= 2000:
                jitter_scale = 0.02 * (1.0 - prog)
                z += np.random.normal(scale=jitter_scale, size=n)
                stagnation = 0
                print(f"[start] jitter added it={it}")

        # after run, compare to global best
        if best_local_obj is not None and best_local_obj < best_obj:
            best_obj = best_local_obj
            best_a = best_local_a
            print(f"[global] improved to O={best_obj:.6f}")

    # after all starts, polishing on best_a
    if best_a is None:
        # fallback: uniform
        best_a = np.full(n, 1.0/n)
        best_obj = 2.0 * n * _full_convolution(best_a).max()
    # polishing via mirror descent
    polishing_steps = 2000
    best_refined = _max_refine(best_a, steps=polishing_steps, eta_start=0.5)
    conv_ref = _full_convolution(best_refined)
    obj_ref = 2.0 * n * conv_ref.max()
    if obj_ref < best_obj:
        best_obj = obj_ref
        best_a = best_refined
        print(f"[polish] improved to O={best_obj:.6f}")

    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    return final_seq

The objective  

\[
E(a)=\frac{2\,n\;\max\_{k}(a*a)\_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is *scale invariant* – we can safely restrict ourselves to probability
vectors (`sum(a)=1`).  
`E(a)` is the scaled ℓ∞‑norm of the self‑convolution of `a`.  
A smooth surrogate  

\[
\tilde E(a)=2n\;\frac{\log\!\bigl(\sum_k
\exp(\alpha\,(a*a)_k)\bigr)}{\alpha}
\]

approaches the true objective when the smoothing parameter `α` grows.
Optimising `\tilde E` in the log‑domain (`z = log a`) with Adam gives a
fast gradient estimator; a final *entropy‑mirror* descent on the exact
max‑entry refines the solution.

The code below improves the previous implementation by

* **multi‑start** – several very different initial vectors (Beta‑shapes,
  Dirichlet draws, interpolated previous result).
* **stronger smoothing** – `α` is increased up to **500** (instead of 200).
* **larger problem size** – `n = 12000` (still cheap for FFT‑based
  convolutions).
* **aggressive restarts** – after a short period without improvement the
  optimisation is re‑initialised from a fresh random start.
* **long polish** – the exact mirror‑descent refinement is run for more
  iterations and a slightly larger step size.

All heavy work is performed with `O(n log n)` FFTs, therefore the whole
procedure fits well inside the 1000 s budget.

#!/usr/bin/env python3
"""
Propose a step function (non‑negative vector) that minimises
evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))².

The algorithm is a multi‑start Adam optimiser on a soft‑max surrogate of
the max‑convolution, followed by an entropy‑mirror descent polish on the
exact objective.
It works on the probability simplex (sum = 1) – the evaluation function
is scale invariant.
"""

from __future__ import annotations
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a vector with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two different vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror‑descent polishing (exact max entry)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray,
                steps: int = 1500,
                eta_start: float = 0.5) -> np.ndarray:
    """
    Mirror‑descent on the exact max‑entry of the convolution.
    a is assumed to be a probability vector (sum = 1).
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of max‑entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # decreasing step size (linear schedule)
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Helper: generate a random probability vector (Dirichlet or Beta)
# ----------------------------------------------------------------------
def _random_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse initial distribution:
    - with 50 % probability a symmetric Beta(p,p) shape,
    - otherwise a Dirichlet draw with random concentration.
    The result is normalised to sum‑to‑one.
    """
    if rng.random() < 0.5:
        # symmetric Beta(p,p) – p∈[0.5,5] gives flat to strongly‑peaked shapes
        p = rng.uniform(0.5, 5.0)
        i = np.arange(1, n + 1, dtype=np.float64)
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (p - 1.0))
    else:
        # Dirichlet with a random concentration parameter
        concentration = rng.uniform(0.05, 5.0)
        a = rng.gamma(concentration, size=n)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a
