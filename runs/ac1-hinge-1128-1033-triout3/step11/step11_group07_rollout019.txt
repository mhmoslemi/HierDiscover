"""
Improved step‑function optimiser
--------------------------------

The optimiser searches for a non‑negative vector `a` (length `n`) with
∑a = 1 that minimises the value returned by ``evaluate_sequence``:

        2·n·max(conv(a,a))

The algorithm consists of three stages:

1. **Broad initialisation** – many cheap Beta‑shaped candidates (and a
   few Dirichlet draws) are examined and the best one is kept.  An
   optional previous solution can be interpolated.

2. **Precise smooth optimisation** – the objective is smoothed with a
   log‑sum‑exp (soft‑max) and minimised using SciPy’s L‑BFGS‑B
   optimiser (fallback to Adam if SciPy is unavailable).  The
   optimisation works in log‑softmax space so the simplex constraint is
   satisfied automatically.

3. **Exact sub‑gradient polishing** – a projected sub‑gradient method on
   the true max‑convolution is performed for the remaining budget.
   The step‑size decays linearly and occasional jitter helps escape
   plateaus.

After the optimisation the candidate is blended with the uniform
distribution and with a few simple analytic shapes; the best variant is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights (log‑sum‑exp trick)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple Adam optimiser (fallback)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed objective."""
    # work in log‑softmax space (automatically stays on the simplex)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.005
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 2000

    while time.time() < deadline:
        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam]  new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # grad already contains 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.05, size=n)

        # restart if stalled
        if idle > MAX_IDLE:
            print("[Adam]  restart after idle")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val

        return obj, grad_a

    # estimate a reasonable iteration budget from remaining time
    remaining = deadline - time.time()
    # a single L‑BFGS iteration is ≈ O(n log n); we conservatively assume 10 ms
    max_iter = max(1, int(remaining / 0.01))
    max_iter = min(max_iter, 2000)

    res = _opt.minimize(fun, w, method='L-BFGS-B', jac=True,
                        options={'maxiter': max_iter, 'disp': False})
    w_opt = res.x
    w_shift = w_opt - np.max(w_opt)
    a_opt = np.exp(w_shift)
    a_opt /= a_opt.sum()
    return a_opt


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        f(a)=2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum_{k∈Kmax} 2·a_{k-i}
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule (more aggressive at start)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # tiny random jitter every few hundred iterations
        if it % 500 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta family candidate (used for a cheap initial sweep)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # problem size – a larger n allows a richer shape space
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialisation – cheap candidates, keep the best one
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # 0.1  Reuse a previous solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.size > 0:
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.size)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            val = _objective(a0)
            best_a, best_val = a0.copy(), val
            print(f"[Init] Interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # 0.2  Scan many Beta shapes (more samples than the original code)
    for _ in range(800):                     # ≈ 800 evaluations (very cheap)
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] Beta candidate improved, value={best_val:.6f}")

    # 0.3  Dirichlet draws with several concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(20):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val
                print(f"[Init] Dirichlet(alpha={alpha}) improved, value={best_val:.6f}")

    # fallback – should never happen
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print(f"[Init] fallback uniform, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Smooth optimisation (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s
    if _SCIPY_OK:
        best_a = _lbfgs_optimize(best_a, n, phase_a_deadline, start_time)
        best_val = _objective(best_a)
        print(f"[L‑BFGS] after phase A, value={best_val:.6f}")
    else:
        best_a, best_val = _adam_optimize(
            best_a, n,
            deadline=phase_a_deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
        )
        print(f"[Adam] after phase A, value={best_val:.6f}")

    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – Sub‑gradient polishing (≈55 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.85 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        max_iters=800_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Final cheap refinements (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # a short extra L‑BFGS pass if time permits
        leftover_deadline = deadline
        if _SCIPY_OK:
            best_a = _lbfgs_optimize(best_a, n, leftover_deadline, phase_b_deadline)
            best_val = _objective(best_a)
        else:
            best_a, best_val = _adam_optimize(
                best_a, n,
                deadline=leftover_deadline,
                start_time=phase_b_deadline,
                best_a=best_a,
                best_val=best_val,
            )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Extra blends with simple analytic shapes
    # ------------------------------------------------------------------
    # linear decreasing shape (a_i ∝ n‑i)
    lin_shape = np.arange(n, 0, -1, dtype=np.float64)
    lin_shape /= lin_shape.sum()
    # exponential decay (rate chosen so mass is spread)
    exp_shape = np.exp(-np.arange(n) / (n / 10.0))
    exp_shape /= exp_shape.sum()

    for shape in (lin_shape, exp_shape):
        for t in np.linspace(0.0, 1.0, 11):
            cand = t * best_a + (1.0 - t) * shape
            cand /= cand.sum()
            val = _objective(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[Blend] new best after shape blend, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure a plain list of floats is returned
    return best_a.tolist()
