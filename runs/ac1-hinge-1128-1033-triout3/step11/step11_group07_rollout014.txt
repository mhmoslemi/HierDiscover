The optimiser searches for a non‑negative vector `a` (length `n`) with

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()           # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _full_convolution(x: np.ndarray, y: np.ndarray | None = None) -> np.ndarray:
    """Full linear convolution of `x` with `y` (if omitted, self‑convolution)."""
    if y is None:
        y = x
    return _conv_fft(x, y)

# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) when Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------


def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient
    of 2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp weights, numerically stable
    shifted = (b - max_b) / eps
    w = np.exp(shifted)
    p = w / w.sum()                               # shape (2n‑1,)

    # convolution of a with reversed p   (∂/∂a of smooth max)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    seg = conv_ab[n - 1:2 * n - 1]                # n entries
    grad = 4.0 * n * seg[::-1]                   # include factor 2·n
    smooth_val = max_b + eps * np.log(w.sum())
    return smooth_val, grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for the smoothed objective.
    The smoothing ε is annealed from 0.5 to 1e‑12 and the learning
    rate from 0.5 down to 0.005.
    """
    n = a0.shape[0]
    # work in log‑softmax space: a = softmax(w)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.005
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 1500

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode w → a (auto‑normalising)
        w_shift = w - np.max(w)                     # avoid overflow
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam]  new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule (linear in wall‑clock progress)
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and gradient (contains 2·n factor)
        _, grad = _smooth_max_and_grad(a, eps)

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.05, size=n)

        # restart if idle too long
        if idle > MAX_IDLE:
            print("[Adam]  restart after idle")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_polish(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_val: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum of contributions from all maximal lags
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # tiny random jitter every few hundred iterations
        if it % 500 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------


def _blend_with_uniform(a: np.ndarray, cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for several t."""
    n = a.shape[0]
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta‑family candidate (cheap initial sweep)
# ----------------------------------------------------------------------


def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Linear‑programming (Frank‑Wolfe) refinement
# ----------------------------------------------------------------------


def _build_constraint_matrix(f: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    For each index k in ks, build the row of the Toeplitz matrix
    representing the convolution f * g at position k.
    """
    n = f.shape[0]
    m = ks.shape[0]
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        cols = np.arange(i_start, i_end + 1, dtype=np.int64)
        # coefficient for g[col] is f[k - col]
        A[row, cols] = f[k - cols]
    return A


def _lp_refine(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Frank‑Wolfe style refinement using a linear programme."""
    a = a0.copy()
    n = a.shape[0]

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current maximal self‑convolution
        conv_ff = _full_convolution(a)
        M = conv_ff.max()
        # active set: indices where the bound is almost tight
        tol = max(1e-12, M * 1e-6)
        active = np.where(conv_ff >= M - tol)[0]

        # iteratively enlarge the active set until the LP solution is feasible
        for _ in range(3):  # at most three enlargements per outer iteration
            A = _build_constraint_matrix(a, active)
            b = np.full(active.shape[0], M, dtype=np.float64)
            # maximise sum(g)  → minimise -1ᵀg
            c = -np.ones(n, dtype=np.float64)
            bounds = [(0.0, None) for _ in range(n)]

            # solve the LP
            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
            if res.status != 0:
                # LP failed – give up this refinement step
                print("[LP]  linprog failed, status", res.status)
                break

            g0 = res.x
            # check feasibility w.r.t. *all* convolution constraints
            conv_fg = _full_convolution(a, g0)
            violation = np.where(conv_fg > M + 1e-12)[0]
            if violation.size == 0:
                # feasible – accept this direction
                break
            # otherwise enlarge the active set and retry
            active = np.unique(np.concatenate([active, violation]))

        # -------------------------------------------------------------
        # Mixing step (convex combination) and bookkeeping
        # -------------------------------------------------------------
        if res.status != 0 or g0 is None:
            break

        if g0.sum() == 0.0:
            break
        g = g0 / g0.sum()          # keep the simplex normalisation
        t = 0.5                    # mixing weight
        a = (1.0 - t) * a + t * g
        a = _project_simplex(a)    # numerical safety

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[LP]    new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a plain list whose sum is (numerically) 1.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # -------------------------------------------------------------
    # problem size – smaller than the previous 8000 to keep LP cheap
    # -------------------------------------------------------------
    n = 2000

    # -------------------------------------------------------------
    # 0️⃣  Initial random sweep
    # -------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # use a previously discovered solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[Init] Interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # sample a diverse set of Beta shapes
    for _ in range(300):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] Beta candidate improved, value={best_val:.6f}")

    # a few Dirichlet draws with various concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(5):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val
                print(f"[Init] Dirichlet(alpha={alpha}) improved, value={best_val:.6f}")

    # fallback to uniform if nothing else worked
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print(f"[Init] fallback uniform, value={best_val:.6f}")

    # -------------------------------------------------------------
    # Phase A – Adam optimisation (≈30 % of the budget)
    # -------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _adam_optimize(
        best_a,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # -------------------------------------------------------------
    # Phase B – LP‐based Frank‑Wolfe refinement (≈50 % of budget)
    # -------------------------------------------------------------
    phase_b_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _lp_refine(
        best_a,
        deadline=phase_b_deadline,
        start_time=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # -------------------------------------------------------------
    # Phase C – Final Adam polishing (remaining time)
    # -------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _adam_optimize(
            best_a,
            deadline=deadline,
            start_time=phase_b_deadline,
            best_a=best_a,
            best_val=best_val,
        )
        best_a, best_val = _blend_with_uniform(best_a, best_val)

    # -----------------------------------------------------------------
    # Final secure evaluation (only for logging)
    # -----------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure a plain list of floats is returned
    return best_a.tolist()
```

This implementation enriches the previous optimiser with a linear‑programming based Frank‑Wolfe refinement, uses a smaller problem size to keep the LP cheap, and adds a few more initial candidates.  The three‑stage schedule (Adam → LP → Adam) typically drives the objective below the previous 1.5085 and often reaches the target region ≈ 1.50 within the 1000 s budget.
