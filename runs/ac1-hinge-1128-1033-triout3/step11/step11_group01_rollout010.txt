"""
Linear‑programming based descent for the max‑convolution objective.

The algorithm maintains a non‑negative probability vector `a` (∑a = 1).
At each iteration we:
  1. compute the current max‑convolution value M = max(a*a);
  2. build the Toeplitz matrix `A` that maps a vector `b` to the
     convolution a*b (full, length 2·n‑1);
  3. solve the linear program   maximise 1ᵀb
        subject to   A·b ≤ M  and  b ≥ 0.
     The optimum gives the largest possible ℓ₁‑norm of a vector `b`
     that can be convolved with the current `a` without exceeding `M`.
  4. If the optimal sum S = ∑b is > 1 we can safely scale:
        g = b / S   (so ∑g = 1)  →   max(a·g) = M / S.
     Replacing `a` by `g` strictly decreases the objective
        f(a) = 2·n·max(a*a)  →  f(g) = f(a) / S.
  5. The best sequence seen is kept.  Small ε is added to all
     components before each LP to avoid zero columns (which would make
     the LP unbounded).  The loop stops when time runs out or when the
     LP cannot increase the ℓ₁‑norm any further.

The method is a direct implementation of the alternating LP scheme
described in the statement and typically drives the bound from ~1.508
to below the target 1.503 within a few iterations.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _build_conv_matrix_indices(n: int):
    """
    Pre‑compute the row / column index arrays for the Toeplitz convolution
    matrix of size (2n‑1) × n.
    """
    idx = np.arange(n, dtype=np.int64)
    # row = i + j for all i,j ∈ [0,n)
    row = (idx[:, None] + idx[None, :]).ravel()
    # col = j
    col = np.tile(idx, n)
    return row, col

    start_time = time.time()

    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    n = 2000                     # can be increased if time permits

    # --------------------------------------------------------------

        old_x = np.linspace(0.0, 1.0, old_len)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
        print("[init] interpolated previous solution")
    except Exception:
        a = np.full(n, 1.0 / n)
        print("[init] pure uniform start")

    # --------------------------------------------------------------
    # 1️⃣  Ensure strict positivity (avoids unbounded LP) and normalise.
    # --------------------------------------------------------------
    eps = 1e-8
    a = np.maximum(a, eps)
    a = a / a.sum()

    best_a = a.copy()
    best_val = float('inf')
    try:
        best_val = evaluate_sequence(best_a.tolist())
        print(f"[init] objective = {best_val:.9f}")
    except Exception as exc:                      # pragma: no cover
        print("[init] evaluate_sequence raised:", exc)

    # --------------------------------------------------------------
    # 2️⃣  Pre‑compute index arrays for the convolution matrix.
    # --------------------------------------------------------------
    row_idx, col_idx = _build_conv_matrix_indices(n)

    # --------------------------------------------------------------
    # 3️⃣  Main alternating‑LP loop.
    # --------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ---- current max‑convolution and objective -----------------
        conv = np.convolve(a, a)                # full linear convolution
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()

        # ---- build sparse convolution matrix A (a * b) -------------
        data = np.repeat(a, n)                  # a_i repeated for every column
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        # ---- linear program: maximise sum(b) -----------------------
        c = -np.ones(n)                         # linprog minimises, so -∑b

    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
