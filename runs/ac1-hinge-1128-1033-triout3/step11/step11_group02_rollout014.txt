"""
Improved step‑function optimiser.

The algorithm proceeds in three stages:

1️⃣ *Diverse warm‑starts* – a handful of analytically‑motivated
   distributions (uniform, Gaussian, sinusoid) and a quick sweep of the
   two‑parameter Beta family are used to generate a set of seeds.
2️⃣ *Smooth‐max Adam optimisation* – we minimise a smooth approximation of
   the supremum norm of the auto‑convolution:
        f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
   by Adam with projection onto the probability simplex.  λ is
   continued from 0.5 up to 200, giving a very tight approximation to the
   true max.  The optimiser runs as long as the budget permits.
3️⃣ *Fine‑grained simulated‑annealing hill‑climber* – starting from the
   best vector found in (2) we perform targeted mass transfers that
   specifically reduce the convolution entry that attains the current
   maximum.  The annealing temperature decays to 0, guaranteeing a
   monotone improvement once we are close to a local optimum.

All stages respect the 1000 s time budget.  The function returns a plain
list of non‑negative floats whose evaluation by ``evaluate_sequence`` is
expected to beat the previous bound of 1.508417 (target ≤ 1.503)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Initial‑seed generators
# ----------------------------------------------------------------------
def _seed_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _seed_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.ones(n, dtype=np.float64)
        a_sum = n
    a /= a_sum
    return a


def _seed_sinusoid(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Exhaustive sweep of the Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # richer grid than the previous implementation
    p_vals = np.concatenate([
        np.linspace(0.1, 5.0, 40),
        np.geomspace(5.0, 30.0, 8)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + λ‑continuation)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     lam_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth‑max objective  f_λ(a) using Adam.
    λ grows from 0.5 → ``lam_end``.
    The iterate is projected onto the probability simplex after every step.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    best_a = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_alpha = 0.07
    t = 0

    # Pre‑computed FFT size for auto‑convolution
    size1 = 1 << ((2 * n - 2).bit_length())
    size2 = 1 << ((3 * n - 3).bit_length())

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        # λ schedule (quadratic easing gives more time at small λ)
        prog = (now - start_time) / (deadline - start_time)
        lam = 0.5 + (lam_end - 0.5) * (prog ** 0.7)

        # ----- a * a -----
        fa = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fa * fa, size1)[:2 * n - 1]

        # ----- smooth weights (softmax) -----
        max_c = conv.max()
        exps = np.exp(lam * (conv - max_c))   # stabilised
        w = exps / exps.sum()                 # sum to 1, length 2·n‑1

        # ----- gradient of the smooth max -----
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, size2)
        fw = np.fft.rfft(w, size2)
        conv_aw = np.fft.irfft(fa_rev * fw, size2)[:3 * n - 2]
        grad = 2.0 * conv_aw[n - 1 : n - 1 + n]   # length n

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # step size decays with λ to keep updates stable
        step = base_alpha / (lam ** 0.5) * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s

        a = a_new

        # ----- keep the best (exact objective) -----
        cur_val = 2.0 * n * max_c
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth] iter={t:7d} λ={lam:6.2f} best={best_val:.6f}")

            # early exit if we are already under the target
            if best_val <= 1.503:
                break

        # occasional random jitter to escape plateaus
        if t % 1200 == 0:
            if cur_val >= best_val * 0.9999:
                noise = np.random.normal(scale=0.012, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[Smooth] iter={t:7d} – jitter applied")

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (enhanced)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local refinement that repeatedly transfers a small amount of mass
    from indices that dominate the current maximal convolution entry
    to indices that contribute little.  Temperature decays exponentially.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-15

    # annealing schedule
    temp_start, temp_end = 0.03, 1e-11
    max_frac_start, max_frac_end = 0.6, 0.001

    no_improve = 0
    IMPROVE_LIMIT = 200_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = temp_start * (temp_end / temp_start) ** prog
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # index of the convolution peak
        k_max = int(np.argmax(b))

        # indices i that actually contribute to k_max (i + j = k_max)
        i_candidates = np.arange(n)
        j_part = k_max - i_candidates
        mask = (j_part >= 0) & (j_part < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_candidates[mask]] * a[j_part[mask]]
        total_contrib = contrib.sum()

        # source – pick a heavy contributor
        if total_contrib > 0 and np.random.rand() < 0.85:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target – we prefer an index that is *not* involved in the peak
        # (i.e. low contribution to the current max)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback uniform choice
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        # amount of mass to move
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # ----- update convolution efficiently -----
        # When we subtract delta from a[i_src] and add to a[j_tgt],
        # the convolution changes by a simple rank‑2 update.
        b_new = b.copy()
        # contribution of the removed mass
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        # contribution of the added mass
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        # self‑pair adjustments
        b_new[2 * i_src]      += delta * delta
        b_new[2 * j_tgt]      += delta * delta
        b_new[i_src + j_tgt]  -= delta * delta
        b_new[j_tgt + i_src]  -= delta * delta   # same index, kept for symmetry

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ----- Metropolis acceptance -----
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-14:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] iter={iteration:7d} new best={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # occasional restart if stuck for long
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.028, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] iter={iteration:7d} – stagnation reset")

        iteration += 1

    return best_a, best_val

                                          deadline=local_deadline,

                                          lam_end=200.0)
        print(f"[seed {idx}] smooth_opt best={val_opt:.6f}")
        if val_opt < global_best_val:

        if global_best_val <= 1.503:
            break

    # --------------------------------------------------------------
    # 2️⃣ Final hill‑climbing refinement on the remaining time
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.8 and global_best_a is not None:

            global_best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Final verification (purely for logging)
    # --------------------------------------------------------------

        reported = evaluate_sequence(global_best_a.tolist())
        print(f"[final] evaluate_sequence reports {reported:.6f}")
    except Exception as exc:              # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Convert to plain Python list as required by the harness
