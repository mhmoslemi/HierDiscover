def subgradient_descent(a_init, deadline, start_time, best_a, best_val):
    a = a_init.copy()
    n = len(a)
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        val = 2 * n * max_b
        if val < best_val:
            best_val, best_a = val, a.copy()
        # compute subgrad
        ks = np.where(b == max_b)[0]
        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2 * a[idx[mask]]
        # step size schedule
        elapsed = time.time() - start_time
        total = deadline - start_time
        # e.g., eta0 * (1 - elapsed/total)**some
        eta0 = 0.5
        eta = eta0 * (1 - elapsed / total + 1e-12)  # linearly decreasing to near 0
        # update
        a_new = a - eta * grad
        # projection onto simplex
        a_new = _project_simplex(a_new)
        a = a_new
        it += 1

    return best_a, best_val

"""
Propose a step‑function with a low max‑auto‑convolution.

The algorithm combines several stages:

0️⃣  Broad random search (Beta / Dirichlet / mixed‑Beta draws) to obtain a
    decent seed.  If a previous solution (`height_sequence_1`) exists it is
    interpolated and used as a seed.

1️⃣  Adam optimisation on a smooth log‑sum‑exp approximation of the
    max‑convolution.  The smoothing parameter ε is annealed from 0.5 to
    1e‑9 while the learning‑rate decays from 0.2 to 0.001.  The variables
    are the logits of a soft‑max, so the simplex constraint (non‑negative
    and sum = 1) is always satisfied.

2️⃣  Long projected sub‑gradient descent on the exact objective
    `f(a)=2·n·max(conv(a,a))`.  The sub‑gradient is computed analytically
    from the current maximal convolution entry(ies).  A linearly
    decreasing step‑size together with periodic jitter and restart
    prevents stagnation.  This stage runs for the bulk of the time budget.

3️⃣  Aggressive hill‑climbing / simulated‑annealing moves that shift a
    small fraction of mass from “high‑contribution’’ indices to “low‑
    contribution’’ ones.  The schedule of the maximal transferable
    fraction and the temperature are slowly annealed.  The best vector
    found through the whole run is returned.

All phases monitor the wall‑clock and stop as soon as the total budget
(≈ 1000 s) is exhausted.  The final sequence is reported through the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective and gradient utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Smooth approximation of max(conv(a,a)) using log‑sum‑exp
    and its gradient (includes the factor 2·n).
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform vector
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random candidate generators
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Beta‑shaped vector (i‑th entry ∝ i^{p‑1}(n‑i+1)^{q‑1}) and normalise."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-12.0, 4.0)   # allow a wider range than before
    logq = rng.uniform(-12.0, 4.0)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_mixed_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two Beta shapes."""
    w = rng.random()
    a1 = _random_beta_candidate(n, rng)
    a2 = _random_beta_candidate(n, rng)
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation (smooth max)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed objective."""
    w = np.log(a0 + 1e-12)                 # logits for soft‑max

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 6000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode current probabilities
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedules
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth max gradient
        _, grad = _smooth_max_and_grad(a, eps)   # already contains 2·n

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)

        # restart if idle too long
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _proj_subgrad_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          best_a: np.ndarray,
                          best_val: float,
                          eta0: float = 0.5,
                          eta_min: float = 1e-6,
                          restart_after: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Long sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Step size decays linearly from eta0 to eta_min.
    """
    a = a0.copy()
    it = 0
    last_improve = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # compute convolution & max
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            last_improve = it
            print(f"[SubG] new best = {best_val:.6f} (it={it})")

        # sub‑gradient of max(b)
        ks = np.where(b == max_b)[0]
        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        prog = (now - start_time) / (deadline - start_time)
        eta = eta0 * (1.0 - prog) + eta_min * prog

        # projected update
        a_new = a - eta * grad
        a_new = _project_simplex(a_new)

        a = a_new
        it += 1

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

        # restart if stagnated
        if it - last_improve > restart_after:
            print(f"[SubG] restart after {it} it (no improve)")
            a = best_a.copy()
            a = np.maximum(a + np.random.normal(scale=0.03, size=n), 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing (fast O(n) updates)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """Aggressive local search moving mass from a “source’’ to a “target’’."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules – a bit more generous than the original version
    max_frac_start, max_frac_end = 0.9, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of the (first) maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source: proportional to contribution
        if total > 0.0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target: favour indices with low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback: uniform random different index
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation at it={it}, re‑perturb")
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to blend with the uniform distribution (often smooths spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Search convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand = _project_simplex(cand)      # keep on simplex
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a plain list of floats (ℓ₁‑norm = 1) that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # problem size (chosen as a good trade‑off between flexibility and speed)
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Broad random search + optional old solution
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)               # uniform start
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # try to reuse a previously discovered solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous, value={best_val:.6f}")
    except Exception:
        pass

    # many random Beta draws
    for _ in range(2500):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Dirichlet draws with a spread of concentration parameters
    for alpha in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(20):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # mixed‑Beta mixtures (adds extra diversity)
    for _ in range(300):
        cand = _random_mixed_beta(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # a couple of handcrafted sinusoidal shapes
    i = np.arange(n, dtype=np.float64)
    cand = np.sin(np.pi * (i + 0.5) / n)
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    cand = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    print(f"[Init] best after random search = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈20 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.20 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – Long projected sub‑gradient descent (≈55 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.75 * budget_s
    best_a, best_val = _proj_subgrad_descent(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        eta0=0.6,               # a slightly larger initial step size
        eta_min=1e-7,
        restart_after=150_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Aggressive hill‑climbing / simulated annealing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hillclimb(
            best_a, n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # return a plain Python list
    return best_a.tolist()
