def max_gradient_descent(a0, n, deadline, start_time, lr0=0.2, beta1=0.9, beta2=0.999, eps=1e-8, init_T=0.01, final_T=1e-6):
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    # initialize Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    conv = full_convolution(a)
    cur_obj = objective_from_conv(conv, n)
    best_a = a.copy()
    best_obj = cur_obj

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute max index
        k_max = int(np.argmax(conv))
        # construct w as one-hot vector
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        # gradient w.r.t a is 2 * conv_rev(a, w)
        grad = 2.0 * conv_rev(a, w)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning rate schedule: maybe decay as 1/sqrt(t) or constant
        step = lr0 / (np.sqrt(v_hat) + eps)

        # compute temperature
        prog = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        T = init_T * (1 - prog) + final_T * prog  # linear decay

        # propose new a
        a_candidate = a - step * m_hat
        # enforce nonnegative and simplex
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = project_simplex(a_candidate)

        conv_candidate = full_convolution(a_candidate)
        obj_candidate = objective_from_conv(conv_candidate, n)

    cur_conv = full_convolution(cur_a)

        # contributions vector
        idx = k_max - np.arange(len(a))
        mask = (idx >= 0) & (idx < len(a))

        # choose source

    (α≈500) polishes the solution.

All steps respect the 1000 s wall‑clock budget, use only one CPU and keep
the sequence non‑negative (projection onto the simplex guarantees Σa=1).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (real input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_rev(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·∑_i a_i·w_{i+j} for j=0..n‑1.
    Used for the gradient of max_k (a*a)_k.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = n + w.shape[0] - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx = 1} (Euclidean)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, assumes Σa>0)
# ----------------------------------------------------------------------
def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """2·n·max(convolution) – the exact quantity evaluated by evaluate_sequence."""
    return float(2.0 * n * conv.max())


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (non‑negative, Σa>0)."""
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.shape[0])


# ----------------------------------------------------------------------
# Small parametric families for initialisation
# ----------------------------------------------------------------------
def _parametric_candidates(n: int):
    """Yield a modest collection of hand‑crafted shapes."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Gaussian families
    for sigma_factor in (0.15, 0.20, 0.25, 0.30):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        a /= a.sum()
        yield a

    # Laplace families
    for scale_factor in (0.08, 0.12, 0.18):
        scale = scale_factor * n
        a = np.exp(-np.abs(i - mid) / scale)
        a /= a.sum()
        yield a

    # Power‑law / Beta‑like families
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = (i + 1) ** (p - 1.0) * (n - i) ** (p - 1.0)
        if a.sum() > 0:
            a /= a.sum()
            yield a


def _interpolate_previous(prev_seq, n):
    """Resize a previously found solution to length n."""
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        return a0
    except Exception:
        return None


# ----------------------------------------------------------------------
# Max‑gradient optimiser (Adam + simulated annealing)
# ----------------------------------------------------------------------
def _max_gradient_optimize(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           lr0: float = 0.2,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8,
                           init_T: float = 0.02,
                           final_T: float = 1e-6,
                           target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that always follows the *active* lag (the one with
    maximal convolution).  Moves that increase the true objective are
    occasionally accepted with a temperature that linearly cools.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 0 else np.full(n, 1.0 / n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    conv = _full_convolution(a)
    cur_obj = _objective_from_conv(conv, n)
    best_a = a.copy()
    best_obj = cur_obj

    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 5000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # identify the worst lag (the one that defines the current max)
        k_max = int(np.argmax(conv))
        # one‑hot weight for the active lag
        w = np.zeros_like(conv)
        w[k_max] = 1.0

        # gradient of max (sub‑gradient)
        grad = 2.0 * _conv_rev(a, w)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)

        # temperature schedule (linear cooling)
        prog = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        T = init_T * (1.0 - prog) + final_T * prog

        # propose new point and project onto simplex
        a_cand = a - step * m_hat
        a_cand = np.maximum(a_cand, 0.0)
        a_cand = _project_simplex(a_cand)

        conv_cand = _full_convolution(a_cand)
        obj_cand = _objective_from_conv(conv_cand, n)

        # accept if improvement, otherwise with Metropolis probability
        accept = False
        if obj_cand < cur_obj - 1e-12:
            accept = True
        else:
            diff = obj_cand - cur_obj
            if np.random.rand() < math.exp(-diff / max(T, 1e-12)):
                accept = True

        if accept:
            a = a_cand
            conv = conv_cand
            cur_obj = obj_cand
            no_improve = 0
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[MaxGrad] iter {t:6d} obj={best_obj:.6f} T={T:.6e}")
                if best_obj <= target_val:
                    break
        else:
            no_improve += 1
            if no_improve >= IMPROVE_LIMIT:
                # jitter around current best to escape stagnation
                a = best_a.copy()
                noise = np.random.normal(scale=0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                a = _project_simplex(a)
                conv = _full_convolution(a)
                cur_obj = _objective_from_conv(conv, n)
                no_improve = 0

    return best_a, best_obj


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (mass‑transfer)
# ----------------------------------------------------------------------
def _annealing_hillclimb(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer hill‑climb that accepts occasional uphill moves.
    The temperature falls linearly with the remaining time.
    """
    a = a0.copy()
    conv = _full_convolution(a)
    cur_obj = _objective_from_conv(conv, n)
    best_a = a.copy()
    best_obj = cur_obj

    no_improve = 0
    IMPROVE_LIMIT = 3000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # worst lag
        k_max = int(np.argmax(conv))

        # contribution of each index to that lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]

        total = contrib.sum()
        if total > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-12:
            continue

        # pick a low‑contributing target
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        # move a random fraction (up to 50 %) of the source mass
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        conv_new = _full_convolution(a_new)
        obj_new = _objective_from_conv(conv_new, n)

        # temperature schedule (linear)
        prog = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        T = 0.02 * (1.0 - prog) + 1e-6 * prog

        # Metropolis acceptance
        if obj_new < cur_obj or np.random.rand() < math.exp(-(obj_new - cur_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = obj_new
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} obj={best_obj:.6f} T={T:.6e}")
                if best_obj <= target_val:
                    break
        else:
            no_improve += 1
            if no_improve >= IMPROVE_LIMIT:
                # jitter around the current best
                a = best_a.copy()
                noise = np.random.normal(scale=0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                a = _project_simplex(a)
                conv = _full_convolution(a)
                cur_obj = _objective_from_conv(conv, n)
                no_improve = 0

        iteration += 1

    return best_a, best_obj


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (the one from the starter code – kept for final polish)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    maxc = conv.max()
    exp_vals = np.exp(alpha * (conv - maxc))
    s = exp_vals.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exp_vals / s


def _smooth_adam_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          alpha_start: float = 5.0,
                          alpha_end: float = 500.0,
                          lr0: float = 0.1,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam optimiser of the soft‑max surrogate (taken from the reference implementation)."""
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 0 else np.full(n, 1.0 / n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_conv = _full_convolution(a)
    cur_val = _objective_from_conv(cur_conv, n)
    best_a = a.copy()
    best_val = cur_val

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        alpha = alpha_start + (alpha_end - alpha_start) * prog

        w = _softmax_weights(cur_conv, alpha)
        grad = 2.0 * _conv_rev(a, w)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a_cand = a - step * m_hat
        a_cand = np.maximum(a_cand, 0.0)
        a_cand = _project_simplex(a_cand)

        conv_cand = _full_convolution(a_cand)
        val_cand = _objective_from_conv(conv_cand, n)

        if val_cand <= cur_val:
            a = a_cand
            cur_conv = conv_cand
            cur_val = val_cand
            if val_cand < best_val - 1e-12:
                best_val = val_cand
                best_a = a.copy()
                print(f"[SmoothAdam] iter {t:7d} val={best_val:.6f}")
                if best_val <= 1.503:
                    break
        else:
            # no improvement – keep current point (momentum stays)
            pass

        if t % 5000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_conv = _full_convolution(a)
            cur_val = _objective_from_conv(cur_conv, n)

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # leave a safety margin

    # --------------------------------------------------------------
    # Choose a problem size – larger size gives a lower theoretical bound.
    # Use 16384 (2^14) if we have enough time, otherwise fall back to 12000.
    # --------------------------------------------------------------
    n = 16384 if budget_s >= 800 else 12000

    # --------------------------------------------------------------
    # Try to load the previously best known sequence, if it exists.
    # --------------------------------------------------------------
    try:
        previous = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    except Exception:
        previous = None

    # --------------------------------------------------------------
    # Build a pool of initial candidates.
    # --------------------------------------------------------------
    init_cands = list(_parametric_candidates(n))

    # a handful of random Dirichlet draws
    for _ in range(5):
        init_cands.append(np.random.dirichlet(np.ones(n)))

    # interpolate the previous solution, if we have one
    if previous is not None:
        a_prev = _interpolate_previous(previous, n)
        if a_prev is not None:
            init_cands.append(a_prev)

    # --------------------------------------------------------------
    # Evaluate all starters, keep the few best for intensive optimisation.
    # --------------------------------------------------------------
    scored = [( _objective(a), a) for a in init_cands]
    scored.sort(key=lambda p: p[0])
    top_start = [a for _, a in scored[:3]]   # the three most promising

    global_best_a = None
    global_best_val = np.inf

    # --------------------------------------------------------------
    # Refine the top starters with max‑gradient → hill‑climb → (optional) smooth Adam.
    # --------------------------------------------------------------
    for a0 in top_start:
        remaining = deadline - time.time()
        if remaining < 30.0:
            break
        # split the remaining budget for this candidate
        half = remaining * 0.45
        # 1) max‑gradient phase
        a_opt, val_opt = _max_gradient_optimize(
            a0, n,
            deadline=time.time() + half,
            start_time=start_time,
            lr0=0.25,
            init_T=0.02,
            final_T=1e-6)
        # 2) hill‑climb phase (uses the remainder of the time)
        if time.time() < deadline:
            a_opt, val_opt = _annealing_hillclimb(
                a_opt, n,
                deadline=deadline,
                start_time=start_time)

        print(f"[Refine] candidate finished with value {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_a = a_opt.copy()
            print(f"[Global] new best value = {global_best_val:.6f}")

        if global_best_val <= 1.503:
            break

    # --------------------------------------------------------------
    # Random restarts while time permits (quick max‑gradient + hill‑climb)
    # --------------------------------------------------------------
    while time.time() < deadline - 15.0:
        a_rand = np.random.dirichlet(np.ones(n))
        # allocate a short budget (≈15 s) for each fresh restart
        sub_dead = time.time() + 15.0
        a_opt, val_opt = _max_gradient_optimize(
            a_rand, n,
            deadline=sub_dead,
            start_time=start_time,
            lr0=0.25,
            init_T=0.02,
            final_T=1e-6)
        if time.time() < deadline:
            a_opt, val_opt = _annealing_hillclimb(
                a_opt, n,
                deadline=deadline,
                start_time=start_time)
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_a = a_opt.copy()
            print(f"[Global‑rand] improved to {global_best_val:.6f}")
        if global_best_val <= 1.503:
            break

    # --------------------------------------------------------------
    # Final polishing with a high‑alpha smooth‑max Adam run (if time remains).
    # --------------------------------------------------------------
    if global_best_a is not None and time.time() < deadline - 5.0:
        a_final, val_final = _smooth_adam_optimize(
            global_best_a,
            n,
            deadline,
            start_time,
            alpha_start=5.0,
            alpha_end=500.0,
            lr0=0.12)
        if val_final < global_best_val:
            global_best_a = a_final
            global_best_val = val_final
            print(f"[Final‑Adam] polished value = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback to a uniform distribution if nothing was found.
    # --------------------------------------------------------------
    if global_best_a is None:
        global_best_a = np.full(n, 1.0 / n)
        global_best_val = _objective(global_best_a)

    # --------------------------------------------------------------
    # Report the final score (for debugging / logging).
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
