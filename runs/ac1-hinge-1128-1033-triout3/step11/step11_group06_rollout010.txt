def _phi_and_grad(a, beta):
    n = len(a)
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    M = conv.max()
    exp_arg = beta * (conv - M)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum

    # gradient with cross-correlation
    rev_w = w_norm[::-1]
    out_len_corr = n + out_len - 1  # = 3n - 2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]
    offset = out_len - 1
    idx = offset - np.arange(n)
    grad = 2.0 * corr_full[idx]  # shape n

def _phi_grad_u(u, beta):
    # compute softmax of u
    max_u = np.max(u)
    exps = np.exp(u - max_u)
    a = exps / np.sum(exps)
    n = a.shape[0]
    # compute conv via FFT
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    M = conv.max()
    # compute softmax of beta*conv
    exp_arg = beta * (conv - M)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum
    # compute gradient wrt a
    rev_w = w_norm[::-1]
    out_len_corr = n + out_len - 1
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]
    offset = out_len - 1
    idx = offset - np.arange(n)
    grad_a = 2.0 * corr_full[idx]
    # smoothed objective phi
    phi = M + (1.0 / beta) * np.log(w_sum)
    # gradient wrt u
    inner = np.dot(a, grad_a)
    grad_u = a * (grad_a - inner)
    return phi, grad_u, M

def _adam_optimize(u0, deadline):
    n = len(u0)
    u = u0.copy()
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5
    beta_smooth = 0.5
    beta_mul = 1.001
    beta_max = 200.0
    iteration = 0
    best_a = None
    best_score = np.inf

    while True:
        now = time.time()
        if now >= deadline:
            break

        phi, grad_u, max_conv = _phi_and_grad_u(u, beta_smooth)
        # compute actual scoring for current a:
        score = 2.0 * n * max_conv

        if score < best_score:
            best_score = score
            best_a = _softmax(u) # compute a
            # print
            print(f"[opt] iter {iteration} beta {beta_smooth:.3f} score {score:.6f}")

            if best_score <= 1.503:
                # early termination
                break

        # Adam update
        iteration += 1
        lr = lr0 / np.sqrt(iteration)

        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u**2)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)

        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter
        if np.random.rand() < 0.001:
            u += np.random.normal(scale=0.01, size=n)

        # increase smoothing parameter
        beta_smooth = min(beta_max, beta_smooth * beta_mul)

        # continue

    if best_a is None:
        # fallback to softmax(u)
        best_a = _softmax(u)

    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimises
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The ratio is invariant under scaling, so we fix sum(a)=1 and
minimise max(conv(a,a)).  This is a smooth approximation of the
max‑convolution objective using a log‑sum‑exp (soft‑max) smoothing.
The optimisation is performed on the unconstrained logits `u`
with `a = softmax(u)`.  An Adam optimiser with a slowly increasing
smoothing parameter β (β≈0.5→200) drives the distribution towards a
flat autocorrelation, i.e. a small maximum of the self‑convolution.
The routine respects the 1000 s time budget, starts from the previous
solution if available, tries a few random Dirichlet restarts and
keeps the best vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(u: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a vector that sums to 1."""
    max_u = np.max(u)
    e = np.exp(u - max_u)
    return e / e.sum()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of `x` with itself (length 2·len(x)‑1)
    using an FFT of the next power of two.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a vector with sum(a)=1:
        2 * n * max(conv(a,a))
    """
    n = a.shape[0]
    max_conv = _full_convolution(a).max()
    return 2.0 * n * max_conv


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Beta‑shaped probability vector (Σ=1).  Returns None if degenerate."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _phi_and_grad_u(u: np.ndarray, beta: float):
    """
    Smoothed objective φ_β and its gradient w.r.t. the logits `u`.

    Returns
        phi   – scalar value of the smoothed max (≈ max(conv) for large β)
        grad_u – gradient ∂φ/∂u  (same shape as `u`)
        max_conv – the true maximal convolution entry (needed for scoring)
    """
    # --- a = softmax(u) -------------------------------------------------
    a = _softmax(u)
    n = a.shape[0]

    # --- full convolution a⋆a -------------------------------------------
    out_len = 2 * n - 1
    size_conv = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[:out_len]   # (a⋆a)
    max_conv = conv.max()

    # --- soft‑max weights for the smoothing -----------------------------
    # shift by max_conv for numerical stability
    exp_arg = beta * (conv - max_conv)            # ≤ 0
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum                            # Σ_i w_norm_i = 1

    # --- gradient w.r.t. a (∂φ/∂a) --------------------------------------
    # gradient = 2·(a ⋆_cross w_norm)
    rev_w = w_norm[::-1]                          # reverse for correlation
    out_len_corr = n + out_len - 1                # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1
    idx = offset - np.arange(n)                   # indices for j = 0…n‑1
    grad_a = 2.0 * corr_full[idx]                 # shape (n,)

    # --- smoothed objective value ---------------------------------------
    phi = max_conv + (1.0 / beta) * np.log(w_sum)

    # --- chain rule through the softmax (∂φ/∂u) -------------------------
    inner = np.dot(a, grad_a)                     # scalar = Σ a_i * ∂φ/∂a_i
    grad_u = a * (grad_a - inner)

    return phi, grad_u, max_conv


def _adam_optimize(u0: np.ndarray, deadline: float):
    """
    Adam‑based optimisation of the smoothed max‑convolution objective.
    Starts from logits `u0` (logits of a probability vector that sums to 1).

    Returns the best probability vector found (sum = 1) and its exact score.
    """
    n = u0.shape[0]
    u = u0.copy()

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.5

    # smoothing schedule
    beta = 0.5
    beta_mul = 1.001
    beta_max = 200.0

    # state
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    iteration = 0
    best_seq = None
    best_score = np.inf

    while True:
        now = time.time()
        if now >= deadline:
            break

        phi, grad_u, max_conv = _phi_and_grad_u(u, beta)
        score = 2.0 * n * max_conv                # exact objective (sum = 1)

        # keep the best seen so far
        if score < best_score:
            best_score = score
            best_seq = _softmax(u).copy()
            print(f"[opt] iter {iteration:5d}  β={beta:6.3f}  score={score:.6f}")

            # early exit if we already beat the target
            if best_score <= 1.503:
                break

        # ---------- Adam update ----------
        iteration += 1
        lr = lr0 / np.sqrt(iteration)

        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        u = u - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional jitter to escape flat plateaus
        if np.random.rand() < 0.001:
            u += np.random.normal(scale=0.01, size=n)

        # increase smoothing
        beta = min(beta_max, beta * beta_mul)

    if best_seq is None:
        best_seq = _softmax(u)

    return best_seq, best_score


def _initial_candidates(n: int):
    """Generate a small list of diverse starting probability vectors."""
    cand = []

    # 1) previous solution (if available)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
        if a.sum() > 0.0:
            a /= a.sum()
            cand.append(a)
            print("[init] interpolated previous sequence")
    except Exception:
        pass

    # 2) uniform
    cand.append(np.full(n, 1.0 / n))

    # 3) a few Dirichlet draws (dense positive vectors)
    rng = np.random.default_rng()
    for _ in range(3):
        a = rng.dirichlet(np.ones(n))
        cand.append(a)

    # 4) some Beta‑shaped candidates
    for p, q in [(0.5, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        a = _beta_shape(p, q, n)
        if a is not None:
            cand.append(a)

    return cand

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # problem size – a good trade‑off between expressivity and speed
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Build a short list of initial probability vectors
    # ------------------------------------------------------------------
    init_vecs = _initial_candidates(n)

    best_a = None
    best_score = np.inf
    for a in init_vecs:
        val = _objective(a)
        if val < best_score:
            best_score, best_a = val, a.copy()
            print(f"[init] new best init score = {best_score:.6f}")

    if best_a is None:                     # mandatory fallback
        best_a = np.full(n, 1.0 / n)
        best_score = _objective(best_a)
        print("[fallback] uniform start, score =", best_score)

    # ------------------------------------------------------------------
    # 1️⃣  Optimise the best initial vector with Adam + smoothing
    # ------------------------------------------------------------------
    eps = 1e-12
    u0 = np.log(best_a + eps)               # logits of the probability vector
    best_seq, opt_score = _adam_optimize(u0, deadline)

    # ------------------------------------------------------------------
    # 2️⃣  Final secure evaluation (the judge will also do this)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports score = {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
