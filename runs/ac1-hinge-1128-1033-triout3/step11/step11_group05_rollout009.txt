#!/usr/bin/env python
"""Propose a non‑negative sequence that minimizes evaluate_sequence.

The algorithm consists of:

1. Broad parametric exploration (Beta, exponential, Gaussian, power‑sine,
   triangular, sparse random).   Every candidate is normalised to sum 1
   and the best few are remembered.

2. Multi‑start exponentiated‑Adam refinement.   Starting from the global
   best and several jittered copies we run a mirror‑descent optimiser
   that uses the sub‑gradient of max(conv(a,a)).  The update is performed
   in the log‑domain (multiplicative) with Adam‑style first/second moment
   estimates, a linear learning‑rate decay and occasional Gaussian jitter.
   The iterate is projected back onto the probability simplex after each
   step (and optionally symmetrised).

3. The best candidate found is finally evaluated with the provided
   ``evaluate_sequence`` and returned.
"""

import time
import numpy as np

def full_convolution(x: np.ndarray) -> np.ndarray:
    """Return full linear convolution of x with itself using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def objective_val(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)); assumes sum(a)=1."""
    n = a.shape[0]
    b = full_convolution(a)
    return float(2.0 * n * b.max())

def subgradient_max(a: np.ndarray,
                    conv: np.ndarray,
                    max_active: int = 30,
                    rel_tol: float = 1e-6) -> np.ndarray:
    """Return a sub‑gradient of max(conv) w.r.t. a."""
    n = a.shape[0]
    max_val = conv.max()
    if max_active >= conv.size:
        active = np.arange(conv.size)
    else:
        # indices of largest entries
        active = np.argpartition(conv, -max_active)[-max_active:]
        mask = conv[active] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            active = np.array([int(np.argmax(conv))])
        else:
            active = active[mask]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad

# ----- shape families ---------------------------------------------------------

def beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} for i=1,…,n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a

def exp_shape(c: float, n: int) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·|i-mid|)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a

def gaussian_shape(sigma: float, n: int) -> np.ndarray:
    """Gaussian centred at the centre."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a

def power_sine_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1))^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    a /= a.sum()
    return a

def triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises evaluate_sequence.
    The routine respects the time budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # problem size – a compromise between expressive power and speed
    n = 8000

    best_seq: np.ndarray | None = None
    best_obj = np.inf

    def try_update(candidate: np.ndarray, label: str = ""):
        """Update global best if candidate improves the exact objective."""
        nonlocal best_seq, best_obj
        val = objective_val(candidate)
        if val < best_obj:
            best_obj = val
            best_seq = candidate.copy()
            print(f"[{label}] new best = {best_obj:.6f}")

    # ----------------------------------------------------------------------
    # 0️⃣  Initialise from possible previous solution
    # ----------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        cand = np.interp(new_x, old_x, old)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        try_update(cand, "init")
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # 1️⃣  Broad parametric exploration
    # ----------------------------------------------------------------------
    # (a) systematic Beta grid
    p_grid = np.geomspace(0.001, 20.0, 30)
    for p in p_grid:
        for q in p_grid:
            cand = beta_shape(p, q, n)
            if cand is not None:
                try_update(cand, "beta_grid")
        # give time a chance to break early
        if time.time() > deadline - 300:
            break

    rng = np.random.default_rng()
    log_min, log_max = np.log(1e-3), np.log(20.0)
    while time.time() < deadline - 250:
        p = np.exp(rng.uniform(log_min, log_max))
        q = np.exp(rng.uniform(log_min, log_max))
        cand = beta_shape(p, q, n)
        if cand is not None:
            try_update(cand, "beta_rand")
        # occasional fast families
        if rng.random() < 0.02:
            # exponential
            c = rng.uniform(0.0, 5.0)
            try_update(exp_shape(c, n), "exp_rand")
        if rng.random() < 0.02:
            # gaussian
            sigma = rng.uniform(0.5, n / 5.0)
            try_update(gaussian_shape(sigma, n), "gauss_rand")
        if rng.random() < 0.02:
            # power‑sine
            p_param = rng.uniform(0.1, 8.0)
            try_update(power_sine_shape(p_param, n), "psin_rand")

    # (c) deterministic families not yet tried
    try_update(triangular_shape(n), "triangular")
    try_update(exp_shape(1.0, n), "exp_c1")
    try_update(gaussian_shape(n / 10.0, n), "gauss_mid")

    # (d) sparse random vectors (Dirichlet‑like)
    while time.time() < deadline - 150:
        support = rng.integers(10, 1500)
        idx = rng.choice(n, size=support, replace=False)
        vals = rng.random(support)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = vals
        cand /= cand.sum()
        try_update(cand, "sparse")
        # break if too long
        if time.time() > deadline - 150:
            break

    # ----------------------------------------------------------------------
    # If nothing succeeded fall back to uniform
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_obj = objective_val(best_seq)
        print("[fallback] uniform start, obj =", best_obj)

    # ----------------------------------------------------------------------
    # 2️⃣  Multi‑start exponentiated‑Adam refinement
    # ----------------------------------------------------------------------
    # collect a few start points: the best one plus a few jittered copies
    start_points = [best_seq]
    for i in range(4):
        jitter = rng.normal(scale=0.01, size=n)
        seq = np.maximum(best_seq + jitter, 0.0)
        seq /= seq.sum()
        start_points.append(seq)

    # hyper‑parameters for Adam‑style multiplicative updates
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    base_lr = 0.2   # will be annealed linearly to 1e-3

    # loop over each start
    for start_idx, a in enumerate(start_points):
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t_iter = 0
        # allow each start a maximum of 15000 inner iterations, but stop early if out of time
        while time.time() < deadline and t_iter < 15000:
            t_iter += 1
            conv = full_convolution(a)
            cur_max = conv.max()
            cur_obj = 2.0 * n * cur_max
            # update global best
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()
                print(f"[refine {start_idx}] iter {t_iter} new best {best_obj:.6f}")

            # sub‑gradient of max(conv)
            grad = subgradient_max(a, conv, max_active=30, rel_tol=1e-6)

            # Adam moments (first and second)
            m = beta1 * m + (1.0 - beta1) * grad

            # bias‑corrected estimates
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            # linear learning‑rate annealing
            elapsed = time.time() - start_time

``evaluate_sequence``.

# ----------------------------------------------------------------------


def full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

    """
    Exact value of the evaluation objective (ignoring the final scaling)
    when the vector is normalised (sum(a)=1):
        f(a) = 2·n·max(conv(a,a))
    """

    b = full_convolution(a)
    return float(2.0 * n * b.max())

    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a

    """Symmetric exponential decay a_i ∝ exp(-c·|i-mid|)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a

    """Gaussian centred at the middle."""

    return a

    """a_i ∝ sin(π·i/(n+1))^p ."""
    i = np.arange(1, n + 1, dtype=np.float64)

    a /= a.sum()
    return a

    """Symmetric triangular window."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a
