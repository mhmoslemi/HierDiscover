"""
Improved step‑function optimiser (≈1000 s budget)

Summary
-------
1. **Diverse coarse sweep** – many Beta / Dirichlet candidates (including a
   possible interpolated previous solution).  Exact objective
   `2·n·max(conv(a,a))` is evaluated and the best few are kept.

2. **L‑BFGS‑B on a smooth approximation** – the non‑smooth max is replaced
   by a log‑sum‑exp (smooth‑max) with a tiny ε (≈1e‑6).  The variables are
   re‑parameterised as logits `x` → `a = softmax(x)` so the simplex
   constraints are satisfied automatically.  SciPy's L‑BFGS‑B (limited‑memory
   quasi‑Newton) is run from each of the top candidates, with a strict
   time‑budget guard.

3. **Exact sub‑gradient polishing** – a projected sub‑gradient descent on the
   true objective (using the analytic sub‑gradient) cleans up any remaining
   spikes.

4. **Uniform blending & final L‑BFGS** – convex combinations with the
   uniform distribution are tried, then a final short L‑BFGS run with an
   even smaller ε (≈1e‑12) is performed.

The whole pipeline runs under the 1000 s limit, returns a non‑negative
vector whose sum is 1, and is evaluated with the secure `evaluate_sequence`
function before being returned.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper – FFT based convolutions (full linear)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.  The gradient already includes the factor 2·n.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # we need only the n entries that correspond to the “forward” part
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (log‑its → probability simplex)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum of contributions from all maximal diagonals
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule (aggressive early, tiny later)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val + 1e-12 < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 400 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta family candidate (cheap initial sweep)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ~1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eps: float = 1e-6,
                    max_iter: int = 1500) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smoothed objective.
    """
    # initialise in log‑space (soft‑max guarantees feasibility)
    x0 = np.log(a0 + 1e-12)

    # ----- objective ----------------------------------------------------
    def fun(x):
        a = _softmax(x)
        smooth_val, _ = _smooth_max_and_grad(a, eps)
        return 2.0 * n * smooth_val

    # ----- gradient -----------------------------------------------------
    def jac(x):
        a = _softmax(x)
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)
        dot = np.dot(a, grad_a)
        return a * (grad_a - dot)

    # respect the remaining time
    time_left = deadline - time.time()
    if time_left < 1.0:
        return best_a, best_val

    opts = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-12}
    try:
        res = minimize(fun, x0, jac=jac, method="L-BFGS-B", options=opts)
    except Exception as exc:
        # fallback – return without changing best
        print("[L‑BFGS] minimise raised:", exc)
        return best_a, best_val

    a_opt = _softmax(res.x)
    val_opt = _objective(a_opt)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[L‑BFGS] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and that minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderately large n gives a richer space.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # 0️⃣  Coarse initialisation (exact objective)
    # ------------------------------------------------------------------
    candidates = []

    # 0.1  Try to reuse a previously discovered solution (if present)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        candidates.append((val, a0))
        print(f"[Init] interpolated previous solution, value={val:.6f}")
    except Exception:
        pass

    # 0.2  Uniform seed (baseline)
    uniform = np.full(n, 1.0 / n)
    candidates.append((_objective(uniform), uniform))

    # 0.3  Random Beta candidates
    for _ in range(300):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        candidates.append((val, cand))

    # 0.4  Dirichlet draws with a set of concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            candidates.append((val, cand))

    # keep the best few for the next phase
    candidates.sort(key=lambda t: t[0])
    top_k = 5
    best_a, best_val = candidates[0][1], candidates[0][0]
    print(f"[Init] best after sweep: value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – L‑BFGS on smooth‑max, multiple restarts
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    eps_start = 5e-1
    eps_final = 1e-12

    # schedule ε linearly with elapsed time inside the loop
    while time.time() < phase_a_deadline and (time.time() < deadline):
        # progress for ε
        prog = (time.time() - start_time) / (phase_a_deadline - start_time)
        eps = eps_start * (eps_final / eps_start) ** prog
        for i in range(min(top_k, len(candidates))):
            if time.time() >= phase_a_deadline:
                break
            a_start = candidates[i][1]
            best_a, best_val = _lbfgs_optimize(
                a_start, n, deadline, start_time,
                best_a, best_val, eps=eps, max_iter=1200)
            best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient polishing (≈40 % of remaining budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.90 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=300_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Final short L‑BFGS with a very tiny ε
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _lbfgs_optimize(
            best_a, n, deadline, start_time,
            best_a, best_val, eps=1e-12, max_iter=800)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Ensure a plain list of floats is returned
    return best_a.tolist()
