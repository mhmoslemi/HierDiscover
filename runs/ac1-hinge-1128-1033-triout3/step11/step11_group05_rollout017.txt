assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (a step function) that minimises  

\[
F(a)=\frac{2\,n\;\max\bigl(a*a\bigr)}{\bigl(\sum a\bigr)^2},
\]

where `*` denotes the discrete linear convolution.
For a fixed length `n` the value of `F` does not change when all entries are multiplied
by the same constant – we may therefore always keep the entries normalised to
`sum(a)=1`.  The difficult part is to make the maximum entry of the auto‑convolution
as small as possible.

The previous implementation already explored several parametric families and then
performed a mirror‑descent optimisation on the exact objective.
To push the bound further we combine three ideas

1. **Much richer initial pool** – besides the Beta, exponential and Gaussian
   families we also sample raised‑cosine windows, mixtures of a few Gaussians,
   random Dirichlet vectors (concentrations in `[0.1,5]`) and a few crafted
   piece‑wise linear shapes.  All candidates are normalised to sum = 1 and the
   best few are kept.

2. **Improved sub‑gradient descent** – we use a larger active set
   (`max_active=20`) and a tighter relative tolerance when selecting the
   binding lags.  The step size follows a cosine‑annealing schedule and we keep
   a momentum term.  Updates are performed multiplicatively (exponentiated
   gradient) which automatically respects the non‑negativity constraint; after
   each step we project back onto the probability simplex.

3. **Late‑stage simulated‑annealing** – after the deterministic descent has
   converged we run a short stochastic search that randomly perturbs a small
   subset of coordinates.  Moves are accepted with the usual Metropolis rule,
   allowing the algorithm to escape shallow plateaus.

All sections respect the 1000 s budget, leaving a small safety margin before the
final call to `evaluate_sequence`.  The program prints progress information so
that the optimisation process can be followed.

```python
"""
Propose a step‑function (non‑negative sequence) minimising
evaluate_sequence().  The algorithm consists of:

1️⃣  A very wide, diverse initial sampling (Beta, exponential,
    Gaussian, raised‑cosine, mixed Gaussians, Dirichlet,
    piece‑wise linear).  The best few candidates are kept.

2️⃣  An improved mirror‑descent optimiser:
    – larger active set for the sub‑gradient of max‑convolution,
    – cosine‑annealing step size,
    – momentum (β≈0.85),
    – multiplicative updates + Euclidean projection onto the simplex.

3️⃣  A final short simulated‑annealing phase to escape local minima.

The total run time respects the supplied budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (exact, assumes Σa = 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) – the exact evaluate_sequence value
    for a normalised sequence (Σa = 1)."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of shapes for the large‑scale exploratory phase
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """a_i ∝ i^{p‑1} (n‑i+1)^{q‑1},  i = 1,…,n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_shape(c: float, n: int) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(‑c·|i‑mid|)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a


def _gaussian_shape(sigma: float, n: int) -> np.ndarray:
    """Gaussian centred at the centre of the window."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a


def _raised_cosine_shape(alpha: float, n: int) -> np.ndarray:
    """a_i ∝ max(0, cos(π·α·(i‑mid)/(n‑1)))  – a smooth window."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    arg = np.pi * alpha * (i - mid) / (n - 1)
    a = np.cos(arg)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a


def _mixed_gaussians(k: int, n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of k Gaussian bumps with random centres & widths."""
    xs = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for _ in range(k):
        centre = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.5, n / 5.0)
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((xs - centre) / sigma) ** 2)
    a += 1e-12                 # avoid pure zeros
    a /= a.sum()
    return a


def _dirichlet_sample(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α (α<1 → sparse)."""
    a = rng.gamma(alpha, 1.0, size=n)
    a += 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     max_active: int = 20,
                     rel_tol: float = 1e-8) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Active set = up to `max_active` indices whose conv values are within
    `rel_tol` of the global maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    # select candidate indices
    if max_active >= conv.size:
        active = np.arange(conv.size)
    else:
        # get the largest values
        idx_candidates = np.argpartition(conv, -max_active)[-max_active:]
        mask = conv[idx_candidates] >= max_val * (1.0 - rel_tol)
        if np.any(mask):
            active = idx_candidates[mask]
        else:
            active = np.array([int(np.argmax(conv))])

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    returns the best sequence discovered before the deadline.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5         # safety margin for final call

    # Problem size – 8000 offers a good trade‑off between richness and speed.
    n = 8000

    best_seq: np.ndarray | None = None
    best_val = np.inf

    def _maybe_update(cand: np.ndarray, source: str = "") -> None:
        nonlocal best_seq, best_val
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[update] new best {best_val:.6f} (source={source})")

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previously saved solution if present
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _maybe_update(a0, source="interpolated")
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Wide exploration over a very diverse collection of families
    # ------------------------------------------------------------------
    rng = np.random.default_rng()

            if cand is not None:
                _maybe_update(cand, source="beta_grid")
        if time.time() >= deadline - 300.0:
            break

    # (b) random draws from several families
    for _ in range(15000):
        choice = rng.integers(0, 7)
        if choice == 0:          # Beta

            cand = _beta_shape(p, q, n)
        elif choice == 1:        # Exponential
            c = rng.uniform(0.0, 5.0)
            cand = _exp_shape(c, n)
        elif choice == 2:        # Gaussian
            sigma = rng.uniform(0.5, n / 5.0)
            cand = _gaussian_shape(sigma, n)
        elif choice == 3:        # Raised cosine

            cand = _raised_cosine_shape(alpha, n)
        elif choice == 4:        # Mixed Gaussians

        elif choice == 5:        # Dirichlet (sparse)

        else:                    # Piece‑wise linear (triangular + flat top)

        if cand is not None:

        if time.time() >= deadline - 300.0:
            break

    # ------------------------------------------------------------------
    if best_seq is None:

        print("[fallback] uniform start, value=", best_val)

    # ------------------------------------------------------------------

    momentum = np.zeros_like(a)

        # sub‑gradient of the max‑convolution
        grad = _subgradient_max(a, conv, max_active=20, rel_tol=1e-8)

        # momentum update

            a = a / a.sum()
            no_improve = 0

    a = best_seq.copy()
    temperature = 0.01

        cur_val = _objective(a)

            if new_val < best_val:

                best_seq = a.copy()
                print(f"[anneal] improved to {best_val:.6f}")

        # exponential cooling
        temperature *= 0.9995
        if temperature < 1e-7:
            temperature = 1e-7

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
```
