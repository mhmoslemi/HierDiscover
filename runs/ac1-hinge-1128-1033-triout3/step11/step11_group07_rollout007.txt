"""
Improved step‑function optimiser
--------------------------------

The optimiser searches for a non‑negative vector `a` (length `n`,
default 8000) with Σa = 1 that minimises the value returned by
``evaluate_sequence``:

        2·n·max(conv(a,a))   (since the denominator Σa² cancels).

The search consists of four successive stages:

1. **Broad parametric sweep** – a large set of cheap candidates
   (uniform, exponential, power‑law, triangular, mixtures and random
   Beta/Dirichlet draws) is evaluated, the best one is kept.

2. **Adam on a smooth max** – a smooth log‑sum‑exp approximation of the
   L∞‑norm is minimised using Adam in the log‑softmax space.
   The smoothing parameter ε is annealed from 0.5 to 1e‑12 while the
   learning‑rate decays from 0.5 to 0.005.

3. **Exact sub‑gradient polishing** – a projected sub‑gradient method on
   the true objective (max‑convolution) is run until the stage deadline.

4. **Simple hill‑climbing (SA‑style)** – random pairwise mass transfers
   are attempted; steps that improve the exact objective are kept.
   The step size shrinks with the remaining time.

After every stage we blend the current best vector with the uniform
distribution (convex combination) because a little uniform mass often
smooths spikes.

All operations respect the simplex (Σa = 1, a_i ≥ 0) and the total
runtime never exceeds the allotted budget (default 1000 s).  The final
vector is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights (log‑sum‑exp trick)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p  (gradient of max‑conv w.r.t. a)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # extract the part that corresponds to the derivative of each a_i
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   max_idle: int = 3000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth approximation of the exact objective.
    Works in log‑softmax space to stay on the simplex.
    """
    # work in log‑softmax space
    w = np.log(a0 + 1e-16)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.005
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedules
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth max and gradient
        _, grad = _smooth_max_and_grad(a, eps)   # grad already contains 2·n

        # chain rule through softmax
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.05, size=n)

        # restart if stuck for too long
        if idle > max_idle:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution objective)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        init_step: float = 0.8,
                        min_step: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective.
    """
    a = a0.copy()
    cur_val = _objective(a)

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        it += 1

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # analytical sub‑gradient (sum of contributions from all maximal k)
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step size: linear decay from init_step to min_step
        progress = (now - start_time) / (deadline - start_time)
        step = init_step * (1.0 - progress) + min_step * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional tiny jitter to avoid stagnation
        if it % 400 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing (pairwise mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                best_a: np.ndarray,
                best_val: float,
                max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very crude simulated‑annealing / hill‑climbing on the exact objective.
    Randomly moves a fraction of mass from one coordinate to another.
    """
    a = a0.copy()
    cur_val = best_val
    rng = np.random.default_rng(12345)

    it = 0
    while True:
        now = time.time()
        if now >= deadline or it >= max_iters:
            break
        it += 1

        # choose source and destination
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] < 1e-12:
            continue

        # step size shrinks with time
        progress = (now - start_time) / (deadline - start_time)
        frac = (0.3 * (1 - progress) + 0.01 * progress)  # between ~0.3 and ~0.01
        delta = a[i] * rng.uniform(0.0, frac)

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        val = _objective(a_candidate)
        if val < cur_val:
            a = a_candidate
            cur_val = val
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

# Candidate generators
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return a normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exponential_candidate(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i) (lam ≥ 0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _reversed_exponential_candidate(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * (n-1-i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * (n - 1 - i))
    a /= a.sum()
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a /= a.sum()
    return a


def _triangular_candidate(n: int) -> np.ndarray:
    """Symmetric triangle (increasing then decreasing)."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        tri = np.concatenate([inc, inc[::-1]])
    else:
        tri = np.concatenate([inc, [half + 1], inc[::-1]])
    a = tri
    a /= a.sum()
    return a


def _mixture_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of left‑ and right‑hand exponentials."""
    lam1 = rng.uniform(0.0, 3.0)
    lam2 = rng.uniform(0.0, 3.0)
    w = rng.uniform(0.0, 1.0)
    left = _exponential_candidate(n, lam1)
    right = _reversed_exponential_candidate(n, lam2)
    a = w * left + (1.0 - w) * right
    a /= a.sum()
    return a
