"""
Optimiser for the step‑function “attack”.

The approach mixes three ideas that were not fully exploited in the
previous submission:

1️⃣  **Large diverse initialisation** – in addition to the uniform,
    sinusoid and cosine‑square shapes we draw many random Beta,
    Dirichlet, exponential, power‑law and Gaussian profiles.

2️⃣  **Deterministic L‑BFGS on a smoothed objective** – we minimise a
    log‑sum‑exp approximation of the max‑convolution using the
    classic L‑BFGS optimiser from SciPy.  The smoothing parameter
    ε is decreased step‑wise (0.5 → 1e‑9) while warm‑starting the next
    optimisation with the previous solution.  This yields a far more
    accurate gradient than the Adam stage used before.

3️⃣  **Fine‑grained stochastic hill‑climbing** – after the L‑BFGS
    phase we perform a fast O(n) source‑target mass transfer with a
    simulated‑annealing acceptance rule.  The schedule for the step
    size and temperature is slower than the original code, allowing
    many more improvement steps.

Finally we polish the result with a few sub‑gradient steps and
blend the best candidate with a tiny uniform component to smooth any
remaining spikes.

The whole pipeline respects the 1000 s budget and returns a plain
list of non‑negative heights whose ℓ₁‑norm is 1 (the optimiser works
in the probability simplex).  The list is evaluated with the
provided ``evaluate_sequence`` function before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (L‑BFGS).  If SciPy is unavailable we simply skip
# the L‑BFGS stage – the remaining phases are still powerful enough.
# ----------------------------------------------------------------------
try:

    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1).  The factor 2·n is part of the
# official scoring function.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σ a = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient w.r.t. the probability
# vector a.  The returned gradient already includes the outer factor
# 2·n.
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns
        smooth_val  :   max_b + eps·log Σexp((b - max_b)/eps)
        grad_a      :   ∇_a (2·n·smooth_max)
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    Z = weights.sum()
    p = weights / Z                               # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 2·n already included

    smooth_val = max_b + eps * np.log(Z)
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Project onto the probability simplex Σx = 1, x ≥ 0.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta(p,q) shape on a linear grid (log‑uniform p,q)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _random_dirichlet_candidate(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Dirichlet draw with scalar concentration alpha."""
    a = rng.dirichlet(np.full(n, alpha))

    """Exponentially decaying shape a_i ∝ exp(-λ·i)."""
    log_lambda = rng.uniform(-5, 0)          # λ ∈ [1e‑5, 1]

    i = np.arange(n, dtype=np.float64)

    a /= s
    return a


def _power_candidate(n: int, rng: np.random.Generator) -> np.ndarray:

    p = rng.uniform(0.5, 5.0)
    i = np.arange(1, n + 1, dtype=np.float64)

def _gaussian_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Discrete Gaussian (truncated) on the grid."""
    mu = rng.uniform(0, n - 1)
    sigma = rng.uniform(0.05 * n, 0.3 * n)

    a /= a.sum()
    return a


def _sinusoid_candidate(n: int) -> np.ndarray:

    a = np.sin(np.pi * (i + 0.5) / n)          # zero‑free sine wave
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smoothed objective
# ----------------------------------------------------------------------
def _smooth_lbfgs(a0: np.ndarray, eps_schedule, deadline, start_time,
                  best_a, best_val):
    """
    Runs a warm‑started L‑BFGS for each ε in ``eps_schedule``.
    Returns the best (a, value) found.
    """
    if not _SCIPY_OK:
        return best_a, best_val

    n = a0.shape[0]
    logits = np.log(a0 + 1e-12)   # initialise soft‑max logits

    for eps in eps_schedule:
        # objective + gradient for the current ε
        def fun(logits_vec, eps_local):
            w = logits_vec - np.max(logits_vec)
            exp_w = np.exp(w)
            a = exp_w / exp_w.sum()
            smooth_val, grad_a = _smooth_max_and_grad(a, eps_local)
            loss = 2.0 * n * smooth_val
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)
            return loss, grad_logits

        # minimise
        res = minimize(fun, logits, args=(eps,), jac=True,
                       method='L-BFGS-B',
                       options={'maxiter': 300, 'ftol': 1e-12,
                                'gtol': 1e-12, 'disp': False})
        # update logits for the next ε
        logits = res.x

        # evaluate exact objective
        w = logits - np.max(logits)
        exp_w = np.exp(w)
        a = exp_w / exp_w.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS ε={eps:.1e}] new best = {best_val:.6f}")

        # early exit if time exhausted
        if time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Fast hill‑climbing (source→target mass moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly moves a small chunk of mass from a
    “high‑contribution’’ index to a “low‑contribution’’ index,
    with a simulated‑annealing acceptance rule.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-14

    # schedule parameters (slow annealing)
    max_frac_start, max_frac_end = 0.8, 0.001
    temp_start, temp_end = 0.05, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index (probability proportional to its contribution)
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – favour positions with low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # fast incremental update of the full convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # re‑perturb around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Linear step‑size from 0.30 to 1e‑5.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple convex combination with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray, cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try mixtures t·a + (1‑t)·u for a few t‑values."""
    n = a.shape[0]
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a plain list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate size keeps the optimisation cheap.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Diverse initial candidates
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # uniform start
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # attempt to reuse a previously discovered solution if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # a bag of handcrafted shapes
    cand = _sinusoid_candidate(n)
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    cand = _cosine_square_candidate(n)
    val = _objective(cand)
    if val < best_val:
        best_a, best_val = cand, val

    # many random draws from a variety of families
    for _ in range(3000):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(40):
            cand = _random_dirichlet_candidate(n, rng, alpha)
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    for _ in range(800):
        cand = _exp_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    for _ in range(800):
        cand = _power_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    for _ in range(800):
        cand = _gaussian_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] best after random search = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  L‑BFGS on the smoothed max‑convolution (≈30 % of budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    eps_schedule = [0.5, 0.1, 0.01, 0.001, 1e-5, 1e-7, 1e-9]
    best_a, best_val = _smooth_lbfgs(best_a, eps_schedule,
                                     phase1_deadline, start_time,
                                     best_a, best_val)

    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # --------------------------------------------------------------
    # 2️⃣  Aggressive hill‑climbing (≈45 % of budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.75 * budget_s
    best_a, best_val = _hillclimb(best_a,
                                  deadline=phase2_deadline,
                                  start_time=start_time,
                                  best_a=best_a,
                                  best_val=best_val)

    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # --------------------------------------------------------------
    # 3️⃣  Sub‑gradient polishing (remaining time)
    # --------------------------------------------------------------
    best_a, best_val = _subgradient_polish(best_a,
                                           deadline=deadline,
                                           start_time=start_time,
                                           best_a=best_a,
                                           best_val=best_val,
                                           max_iters=400_000)

    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure plain list of floats is returned
    return best_a.tolist()
