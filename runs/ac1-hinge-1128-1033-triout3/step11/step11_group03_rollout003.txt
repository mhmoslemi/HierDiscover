def full_conv(a):
    n = a.shape[0]
    out_len = 2*n-1
    size = 1 << ((out_len-1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:out_len]
    return conv
```

def anneal_optimize(a0, n, deadline):
    a = a0.copy()
    cur_conv = full_conv(a)
    cur_obj = 2.0 * n * cur_conv.max()
    best_a = a.copy()
    best_obj = cur_obj

    start = time.time()
    it = 0
    last_improve_iter = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        it += 1
        # compute temperature
        elapsed = now - start
        progress = elapsed / (deadline - start)
        T = max(1e-6, 0.05 * (0.1 ** progress))  # start 0.05 decays to 0.005? Actually 0.05 * 0.1^p? we want T_end ~1e-6.
        # choose move:
        # compute max lag
        k = int(np.argmax(cur_conv))
        # compute contributions
        # compute array of contributions per index
        # compute i range:
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        contrib = np.zeros_like(a)
        # for i in i_min..i_max
        # Could vectorize:
        idx = np.arange(i_min, i_max+1)
        counterpart = k - idx
        contrib[idx] = a[idx] * a[counterpart]
        total_contrib = contrib.sum()
        # If no contributions (shouldn't happen), choose uniform source
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = idx[np.argmax(contrib[idx])] if total_contrib>0 else np.random.randint(n)

        # target selection:
        # choose index not src, weight bias away from high contributions
        inv_weights = contrib.max() - contrib + 1e-12
        inv_weights[src] = 0.0
        inv_weights_sum = inv_weights.sum()
        if inv_weights_sum > 0 and np.random.rand() < 0.9:
            tgt = np.random.choice(n, p=inv_weights / inv_weights_sum)
        else:
            # random other index
            while True:
                tgt = np.random.randint(n)
                if tgt != src:
                    break

        # delta size:
        max_delta = a[src]
        if max_delta <= 0:
            continue
        # random fraction up to 0.5
        delta = max_delta * np.random.rand() * 0.5

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        # re-normalize sum just in case (should stay 1)
        # but due to floating error, enforce projection to simplex:
        a_new = _project_simplex(a_new)  # maybe use same function

        conv_new = full_conv(a_new)
        obj_new = 2.0 * n * conv_new.max()

        d_obj = obj_new - cur_obj
        accept = False
        if d_obj <= 0:
            accept = True
        else:
            if np.random.rand() < np.exp(-d_obj / T):
                accept = True

        if accept:
            a = a_new
            cur_conv = conv_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                last_improve_iter = it
                print(f"[Anneal] iter {it} obj={best_obj:.6f} T={T:.6e}")

        # occasional perturbation if no improvement for many iterations
        if it - last_improve_iter > 200000:
            # add jitter
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_conv = full_conv(a)
            cur_obj = 2.0 * n * cur_conv.max()
            last_improve_iter = it  # reset

    return best_a, best_obj
```

"""
Simulated‑annealing optimizer for the step‑function attack.

The algorithm:
 - start from a good initial shape (previous best or a smooth Gaussian);
 - repeatedly move a small amount of mass from a “heavy” index (one that
   strongly contributes to the current maximal autocorrelation) to a
   “light” index;
 - accept moves that improve the exact objective, and also sometimes
   accept worsening moves according to a temperature schedule (simulated
   annealing).  This allows escaping local minima that trap the pure
   Adam‑based optimiser used before.
 - after the annealing phase a very short hill‑climbing polishing step
   is applied.
The routine respects the 1000 s budget, prints intermediate progress,
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple objective (exact) – sum is assumed to be 1.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(a∗a) for a sequence a (sum(a)=1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Interpolation of a previous solution (if present)
# ----------------------------------------------------------------------
def _interpolate_previous(prev_seq, n):
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (mass‑transfer moves)
# ----------------------------------------------------------------------
def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Annealing optimizer that moves a small amount of mass from a
    high‑contributing index to a low‑contributing index.
    Returns the best sequence seen and its exact objective.
    """
    a = a0.copy()
    # make sure we are on the simplex (sum = 1)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    cur_conv = _full_convolution(a)
    cur_obj = float(2.0 * n * cur_conv.max())
    best_a = a.copy()
    best_obj = cur_obj

    start = time.time()
    last_improve_iter = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        it += 1

        # --------------------------------------------------------------
        # temperature schedule (geometric decay)
        # --------------------------------------------------------------
        progress = (now - start) / (deadline - start) if deadline > start else 0.0
        # start ~5e-2, end ~5e-6
        T = 5e-2 * (1e-4 ** progress)

        # --------------------------------------------------------------
        # Identify the lag that currently attains the maximum
        # --------------------------------------------------------------
        k_max = int(np.argmax(cur_conv))

        # --------------------------------------------------------------
        # Compute contributions of each index to that lag
        # --------------------------------------------------------------
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        idx = np.arange(i_min, i_max + 1)
        counterpart = k_max - idx
        contrib = np.zeros(n, dtype=float)
        contrib[idx] = a[idx] * a[counterpart]

        total_contrib = contrib.sum()

        # --------------------------------------------------------------
        # Choose a source index (heavy contributor) and a target index
        # --------------------------------------------------------------
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            # fallback – pick a random index that participates in the lag
            if idx.size > 0:
                src = int(np.random.choice(idx))
            else:
                src = np.random.randint(n)

        # target: preferably a location that contributes little to the same lag
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        inv_sum = inv.sum()
        if inv_sum > 0 and np.random.rand() < 0.9:
            tgt = np.random.choice(n, p=inv / inv_sum)
        else:
            while True:
                tgt = np.random.randint(n)
                if tgt != src:
                    break

        # --------------------------------------------------------------
        # Move a random fraction of the source mass to the target
        # --------------------------------------------------------------
        max_delta = a[src]
        if max_delta <= 1e-14:
            continue
        delta = max_delta * np.random.rand() * 0.5   # up to 50 % of source mass

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        # re‑project onto the simplex to avoid drift
        a_new = _project_simplex(a_new)

        # --------------------------------------------------------------
        # Evaluate the exact objective for the new point
        # --------------------------------------------------------------
        conv_new = _full_convolution(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        d_obj = obj_new - cur_obj
        accept = False
        if d_obj <= 0:
            accept = True
        else:
            if np.random.rand() < np.exp(-d_obj / T):
                accept = True

        if accept:
            a = a_new
            cur_conv = conv_new
            cur_obj = obj_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                last_improve_iter = it
                print(f"[Anneal] iter {it:8d} obj={best_obj:.6f} T={T:.2e}")

        # --------------------------------------------------------------
        # If we have not improved for a long while, add a jitter
        # --------------------------------------------------------------
        if it - last_improve_iter > 200_000:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_conv = _full_convolution(a)
            cur_obj = float(2.0 * n * cur_conv.max())
            last_improve_iter = it

    return best_a, best_obj

# Light‑weight hill‑climbing polish (identical to the earlier version)
# ----------------------------------------------------------------------
def _final_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Very simple hill‑climbing that moves a small amount of mass from the
    most “guilty’’ index (largest contribution to the current max lag) to a
    low‑contribution index.  Stops early if the target value is reached.
    """
    b = _full_convolution(a)
    cur_val = float(2.0 * n * b.max())
    best_val = cur_val
    best_a = a.copy()

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # locate the lag giving the maximal autocorrelation
        k_max = int(np.argmax(b))

        # indices that participate in this lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        # contribution of each position to the offending lag
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source: preferably a heavy contributor
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-12:
            continue

        # target: a location that is *not* a heavy contributor
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        # move a small random fraction of the source mass
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = float(2.0 * n * b_new.max())

        if new_val < cur_val:
            # accept move
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # random jitter around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = float(2.0 * n * b.max())
            no_improve = 0

        iteration += 1

    return best_a, best_val

"""
Simulated‑annealing optimizer for the step‑function attack.

The algorithm:
 - start from a good initial shape (previous best or a smooth Gaussian);
 - repeatedly move a small amount of mass from a “heavy’’ index (one that
   strongly contributes to the current maximal autocorrelation) to a
   “light’’ index;
 - accept moves that improve the exact objective, and also sometimes
   accept worsening moves according to a temperature schedule (simulated
   annealing).  This allows escaping local minima that trap the pure
   Adam‑based optimiser used before.
 - after the annealing phase a short hill‑climbing polishing step is
   applied.
The routine respects the 1000 s budget, prints intermediate progress,
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(a∗a) for a sequence a (sum(a)=1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Interpolation of a previously found solution (if present)
# ----------------------------------------------------------------------
def _interpolate_previous(prev_seq, n):
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (mass‑transfer moves)
# ----------------------------------------------------------------------
def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Annealing optimizer that moves a small amount of mass from a
    high‑contributing index to a low‑contributing index.
    Returns the best sequence seen and its exact objective.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    cur_conv = _full_convolution(a)
    cur_obj = float(2.0 * n * cur_conv.max())
    best_a = a.copy()
    best_obj = cur_obj

    start = time.time()
    last_improve_iter = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        it += 1

        # --------------------------------------------------------------
        # temperature schedule (geometric decay)
        # --------------------------------------------------------------
        progress = (now - start) / (deadline - start) if deadline > start else 0.0
        T = 5e-2 * (1e-4 ** progress)      # 0.05 → ≈5·10⁻⁶

        # --------------------------------------------------------------
        # lag attaining the current maximum
        # --------------------------------------------------------------
        k_max = int(np.argmax(cur_conv))

        # --------------------------------------------------------------
        # contributions of each index to that lag
        # --------------------------------------------------------------
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        idx = np.arange(i_min, i_max + 1)
        counterpart = k_max - idx
        contrib = np.zeros(n, dtype=float)
        contrib[idx] = a[idx] * a[counterpart]
        total_contrib = contrib.sum()

        # --------------------------------------------------------------
        # choose source (heavy contributor) and target (light contributor)
        # --------------------------------------------------------------
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.random.choice(idx)) if idx.size else np.random.randint(n)

        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        inv_sum = inv.sum()
        if inv_sum > 0 and np.random.rand() < 0.9:
            tgt = np.random.choice(n, p=inv / inv_sum)
        else:
            while True:
                tgt = np.random.randint(n)
                if tgt != src:
                    break

        # --------------------------------------------------------------
        # move a random fraction of source mass to target
        # --------------------------------------------------------------
        max_delta = a[src]
        if max_delta <= 1e-14:
            continue
        delta = max_delta * np.random.rand() * 0.5    # up to 50 % of source

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        a_new = _project_simplex(a_new)

        # --------------------------------------------------------------
        # evaluate the exact objective for the new point
        # --------------------------------------------------------------
        conv_new = _full_convolution(a_new)
        obj_new = float(2.0 * n * conv_new.max())
        d_obj = obj_new - cur_obj

        accept = False
        if d_obj <= 0:
            accept = True
        else:
            if np.random.rand() < np.exp(-d_obj / T):
                accept = True

        if accept:
            a = a_new
            cur_conv = conv_new
            cur_obj = obj_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                last_improve_iter = it
                print(f"[Anneal] iter {it:8d} obj={best_obj:.6f} T={T:.2e}")

        # --------------------------------------------------------------
        # occasional jitter if stuck
        # --------------------------------------------------------------
        if it - last_improve_iter > 200_000:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_conv = _full_convolution(a)
            cur_obj = float(2.0 * n * cur_conv.max())
            last_improve_iter = it

    return best_a, best_obj
