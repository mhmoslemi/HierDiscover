def _fft_convolve(x,y):
    n = len(x) + len(y) - 1
    size = 1 << (n-1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv

def fft_convolve(x, y):
  n = len(x) + len(y) - 1
  size = 1 << (n - 1).bit_length()
  fx = np.fft.rfft(x, size)
  fy = np.fft.rfft(y, size)
  conv = np.fft.irfft(fx * fy, size)[:n]
  return conv

    the best few are retained.

2️⃣  Two‑stage refinement of the current best candidate:
   • Stage A – a classic sub‑gradient descent that uses an *active set*
     of the largest convolution entries (20 by default).  Momentum
     (β≈0.9) and a linear step‑size schedule keep the iterates stable.
   • Stage B – a smoothed‑max (log‑sum‑exp) descent.  The temperature
     β grows from 0.5 to ≈ 80, turning the soft‑max into the true max.
     The gradient is obtained by a single extra FFT (convolution of the
     soft‑max weights with the reversed current vector).  Multiplicative
     (exponentiated) updates keep the iterate inside the probability
     simplex without an explicit projection.

   Both stages keep track of the exact (non‑smoothed) objective
   ``2·n·max(conv(a,a))`` and store the best encountered sequence.
   Small Gaussian jitter is injected occasionally to escape plateaus.

3️⃣  A final call to ``evaluate_sequence`` is made on the best sequence
    found before the time budget expires.

The routine respects the 1000 s budget, prints progress information,
and never returns an empty or invalid list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    Result length = len(x) + len(y) - 1.
    """
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for self‑convolution of a (full length)."""
    return _fft_convolve(a, a)


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) under the assumption Σa = 1.
    Used for fast comparisons during the search.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Parametric shape families
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}, i = 1,…,n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    tot = a.sum()
    if tot <= 0.0:
        return None
    a /= tot
    return a


def _exp_shape(c: float, n: int) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·|i‑mid|)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a


def _gaussian_shape(sigma: float, n: int) -> np.ndarray:
    """Gaussian centred at the middle."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a


def _power_sine_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1))^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    a /= a.sum()
    return a


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the (unsmoothed) max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     max_active: int = 20,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Return a sub‑gradient of ``max(conv)`` w.r.t. ``a``.
    All entries of ``conv`` equal to the global maximum within ``rel_tol``
    are kept, up to ``max_active`` of them (the most extreme ones).
    """
    n = a.shape[0]
    max_val = conv.max()

    # identify active indices – keep those close to the max
    if max_active >= conv.size:
        active = np.arange(conv.size, dtype=int)
    else:
        # fast argpartition to get top candidates
        cand = np.argpartition(conv, -max_active)[-max_active:]
        mask = conv[cand] >= max_val * (1.0 - rel_tol)
        active = cand[mask]
        if active.size == 0:                       # fall back to the absolute maximum
            active = np.array([int(np.argmax(conv))], dtype=int)

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution 2*a_{k-idx} for i!=j, and a_i for i=j – the factor 2 works for all
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max  (1/β)·log Σ exp(β·conv)  w.r.t. ``a``.
    Returns a vector of length ``len(a)``.
    """
    # stable softmax
    shift = conv.max()
    exp_terms = np.exp(beta * (conv - shift))
    w = exp_terms / exp_terms.sum()                # shape (2n‑1,)

    # gradient = 2 * (w * a_rev)  (full convolution)
    a_rev = a[::-1]
    conv_rev = _fft_convolve(w, a_rev)            # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The function respects the supplied ``budget_s`` (seconds) and returns the
    best sequence discovered before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin for final call

    # ----------------------- problem size -----------------------
    n = 8000                                        # chosen as a good speed/expressivity trade‑off

    best_seq: np.ndarray | None = None
    best_val = np.inf

    # helper to update the current global best
    def _maybe_update(candidate: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[search] new best value = {best_val:.6f}")

    # ----------------------- 0️⃣ Initialise from previous run -----------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)      # type: ignore[name-defined]
        # linearly interpolate onto current resolution
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _maybe_update(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    rng = np.random.default_rng()

    # ----------------------- 1️⃣ Wide exploration -----------------------
    # (a) systematic Beta grid
    p_grid = np.concatenate((
        np.geomspace(0.001, 0.1, 8, endpoint=False),
        np.geomspace(0.1, 5.0, 12, endpoint=False),
        np.geomspace(5.0, 20.0, 5)
    ))
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is not None:
                _maybe_update(cand)

    # (b) many random Beta draws
    log_min, log_max = np.log(1e-3), np.log(20.0)
    for _ in range(15000):
        p = np.exp(rng.uniform(log_min, log_max))
        q = np.exp(rng.uniform(log_min, log_max))
        cand = _beta_shape(p, q, n)
        if cand is not None:
            _maybe_update(cand)

        if time.time() >= deadline - 250.0:   # reserve time for refinement
            break

    # (c) exponential family
    for _ in range(2500):
        c = rng.uniform(0.0, 5.0)
        _maybe_update(_exp_shape(c, n))

    # (d) Gaussian family
    for _ in range(2500):
        sigma = rng.uniform(0.5, n / 5.0)
        _maybe_update(_gaussian_shape(sigma, n))

    # (e) power‑sine family
    for _ in range(2500):
        p = rng.uniform(0.1, 8.0)
        _maybe_update(_power_sine_shape(p, n))

    # (f) triangular window (deterministic)
    _maybe_update(_triangular_shape(n))

    # (g) sparse Dirichlet‑like samples (few non‑zero entries)
    for _ in range(1200):
        support = rng.integers(30, 1500)               # avoid too tiny supports
        idx = rng.choice(n, size=support, replace=False)
        vals = rng.random(support)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = vals
        cand /= cand.sum()
        _maybe_update(cand)

    # (h) pure Dirichlet draws (dense but highly variable)
    for _ in range(1800):
        alpha = rng.uniform(0.05, 2.0)                # concentration
        cand = rng.gamma(alpha, size=n)
        cand /= cand.sum()
        _maybe_update(cand)

    # Fallback – uniform if nothing else succeeded (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)
        print("[fallback] uniform start, value =", best_val)

    # ----------------------- 2️⃣ Refinement – Stage A (sub‑gradient) -----------------------
    a = best_seq.copy()
    m = np.zeros_like(a)               # momentum buffer
    beta_mom = 0.9
    it = 0
    no_improve = 0
    max_no_improve = 5000

    while time.time() < deadline - 150.0:       # keep ~150 s for the second stage
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[stage‑A] iter {it:5d} new best {best_val:.6f}")
        else:
            no_improve += 1

        # sub‑gradient from up to 20 active lags
        grad = _subgradient_max(a, conv, max_active=20, rel_tol=1e-6)

        # momentum update
        m = beta_mom * m + (1.0 - beta_mom) * grad

        # linear step‑size schedule (large at start, tiny at end)
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 150.0)          # relative progress inside stage A
        step = 0.7 * (1.0 - progress) + 1e-5 * progress

        # exponentiated (multiplicative) update – stay on the simplex
        log_factor = -step * m
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)
        a_sum = a.sum()
        if a_sum <= 0.0 or not np.isfinite(a_sum):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= a_sum

        # occasional jitter to shake out of plateaus
        if np.random.rand() < 0.001 or no_improve >= max_no_improve:
            jitter = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            m.fill(0.0)
            no_improve = 0

        it += 1

    # ----------------------- 2️⃣ Refinement – Stage B (soft‑max) -----------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    total_iters = 0
    # temperature schedule parameters
    beta_start = 0.5
    beta_end = 80.0
    max_iters = 6000                     # generous upper bound – will be cut by time

    while time.time() < deadline and total_iters < max_iters:
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[stage‑B] iter {total_iters:5d} new best {best_val:.6f}")

        # annealing temperature
        progress = total_iters / max_iters
        beta = beta_start + (beta_end - beta_start) * progress

        # soft‑max gradient
        grad = _softmax_gradient(a, conv, beta)

        # momentum
        m = beta_mom * m + (1.0 - beta_mom) * grad

        # step‑size (decays roughly as 1/√t)
        step = 0.4 / np.sqrt(total_iters + 1.0)

        log_factor = -step * m
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)
        a_sum = a.sum()
        if a_sum <= 0.0 or not np.isfinite(a_sum):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= a_sum

        # jitter every few hundred steps
        if total_iters % 500 == 0 and total_iters > 0:
            jitter = np.random.normal(scale=0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            m.fill(0.0)

        total_iters += 1

    # ----------------------- 3️⃣ Final secure evaluation -----------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
