"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

The algorithm consists of three stages:

1️⃣  **Broad exploration** – generate candidates from many expressive
    families (asymmetric Beta, one‑sided exponential, power‑law,
    linear‑decrease, mixtures of two Betas, sorted Dirichlet draws).
    All candidates are normalised to Σa = 1 and the best few are kept.

2️⃣  **Refinement** – each of the retained candidates is improved by a
    mirror‑descent optimiser on the exact objective
        F(a) = 2·n·max(conv(a,a))      ( Σa = 1 ),
    using an exponentiated‑gradient update with momentum (β≈0.9),
    a linearly decaying step size, a larger active set for the
    sub‑gradient (max_active = 15) and occasional Gaussian jitter.
    The optimiser runs until the global time budget (≈ 1000 s) is exhausted.

3️⃣  **Final evaluation** – the best sequence obtained is handed to the
    provided ``evaluate_sequence`` and returned.

All helper functions are defined at module level; there are no closures
or filesystem access.  The code respects the supplied time budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Linear‑algebra utilities
# ------------------------------------------------------------

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value of 2·n·max(conv(a,a)) assuming Σa = 1.
    Used for cheap comparison during the search.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    Σx = 1, x ≥ 0.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Shape families for the initial wide exploration
# ------------------------------------------------------------

def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    a_i ∝ i^{p‑1} (n‑i+1)^{q‑1},   i = 1,…,n .
    Asymmetric for p ≠ q.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_decay_shape(c: float, n: int) -> np.ndarray:
    """One‑sided exponential decay a_i ∝ exp(-c·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-c * i)
    a /= a.sum()
    return a


def _power_law_shape(p: float, n: int) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p},   p>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-p)
    a /= a.sum()
    return a


def _linear_decrease_shape(n: int) -> np.ndarray:
    """Deterministic linear decrease a_i ∝ (n-i)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = n + 1 - i
    a /= a.sum()
    return a


def _mixture_beta_shape(p1: float, q1: float,
                       p2: float, q2: float,
                       w: float, n: int) -> np.ndarray | None:
    """
    Convex mixture of two Beta shapes:
        a = w·Beta(p1,q1) + (1‑w)·Beta(p2,q2)
    """
    a1 = _beta_shape(p1, q1, n)
    a2 = _beta_shape(p2, q2, n)
    if a1 is None or a2 is None:
        return None
    a = w * a1 + (1.0 - w) * a2
    a_sum = a.sum()
    if a_sum == 0.0:
        return None
    a /= a_sum
    return a


def _sorted_dirichlet_shape(alpha: float, support: int,
                            n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Sample ``support`` positive numbers from Dirichlet(alpha,…,alpha),
    sort them decreasingly and embed into an array of length ``n``.
    """
    vals = rng.gamma(alpha, 1.0, size=support)
    vals.sort()
    vals = vals[::-1]                     # decreasing
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=support, replace=False)
    a[idx] = vals
    a /= a.sum()
    return a


# ------------------------------------------------------------
# Sub‑gradient for the max‑convolution (active‑set version)
# ------------------------------------------------------------

def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 15,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Return a sub‑gradient of ``max(b)`` w.r.t. ``a``.
    ``b`` is the full convolution a*a.
    """
    n = a.shape[0]
    max_val = b.max()
    # Determine the active set of indices
    if max_active >= b.size:
        active = np.arange(b.size)
    else:
        active = np.argpartition(b, -max_active)[-max_active:]
        mask = b[active] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            active = np.array([int(np.argmax(b))])
        else:
            active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad

def propose_candidate(seed: int = 42,

    # ------------------------------------------------------------------

        if cand is not None:

        if time.time() > deadline - 250.0:

    cand = _linear_decrease_shape(n)

    # --- (f) Mixture of two Betas --------------------------------------------------

        q1 = np.exp(rng.uniform(log_min, log_max))

        print("[fallback] uniform start, value =", best_val)

    # 2️⃣  Refine the current best using a mirror‑descent optimiser

    # note: the exploration already kept the best globally; we will also keep
    #      the next few distinct ones by re‑sampling from families with a looser
    #      acceptance criterion.

            progress = elapsed / per_restart

            s = a.sum()

            else:
                a /= s

                a = np.maximum(a + jitter, 0.0)
                a /= a.sum()
                no_improve = 0
                m.fill(0.0)

            it += 1

        # store the final sequence of this restart (could be useful later)
        candidates.append((_objective(a), a.copy()))

        # update remaining time
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        per_restart = remaining / max(1, top_k - (rank + 1))

    # 3️⃣  Final secure evaluation

        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover

    return best_seq.tolist()
