def smooth_max_grad(a, conv, gamma):
    max_c = conv.max()
    w = np.exp(gamma * (conv - max_c))  # shift for stability
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    a_rev = a[::-1]
    # compute full convolution
    size = 1 << ((len(w) + len(a_rev) - 1) - 1).bit_length()
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv_wa_rev = np.fft.irfft(fw * fa, size)[:len(w) + len(a_rev) - 1]
    grad = 2.0 * conv_wa_rev[n-1 : n-1 + n]
    return grad

def _full_convolution_fft(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    out_len = len(x) + len(y) - 1
    return conv[:out_len]


def _objective(a: np.ndarray) -> float:
    """
    Exact value of 2·n·max(conv(a,a)) assuming Σa = 1.
    Used for cheap internal comparisons (no clipping or scaling).
    """
    n = a.shape[0]
    # size large enough for the convolution a*a
    size = 1 << ((2 * n - 1) - 1).bit_length()
    conv = _full_convolution_fft(a, a, size)
    return float(2.0 * n * conv.max())

# Simple symmetric Beta family (used for initialising)
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    a_i ∝ i^{p‑1} (n‑i+1)^{q‑1},   i = 1,…,n .
    Returned array is normalised to sum = 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a

# ----------------------------------------------------------------------

    s = a.sum()
    if s <= 0.0 or not np.isfinite(s):
        # fallback to uniform distribution
        return np.full_like(a, 1.0 / a.shape[0])
    return a / s
