def _simulated_annealing_phase(a, b, n, cur_val, best_a, best_val,
                               deadline, start_time, rng):
    # define parameters:
    # step fraction schedule: from 0.05 to 0.001
    step_start = 0.05
    step_end = 0.001
    # temperature schedule: start as 0.02*cur_val, end as 0.0005*cur_val
    T_start = 0.02 * cur_val
    T_end = 0.0005 * cur_val
    # counters
    iter_count = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start_time) / (deadline - start_time)
        step_frac = step_start * (1.0 - progress) + step_end * progress
        temp = T_start * (1.0 - progress) + T_end * progress

        # sample source index weighted by a (prefer larger)
        i = rng.choice(n, p=a)
        # sample destination index uniformly (could be same; ensure different)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # delta = step_frac * a[i] * rng.random()
        delta = step_frac * a[i] * rng.random()
        if delta <= 0 or delta > a[i]:
            continue

        # compute new b
        # b_new = b.copy()
        # more efficient: we can compute shift of a once
        # but just copy for clarity
        b_new = b.copy()
        # add/subtract contributions
        # add 2*delta*a to slice j:j+n
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        # delta^2 contributions
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        # due to floating errors, clip to >=0
        # np.maximum(b_new, 0, out=b_new) # in-place

        # compute new max
        max_b_new = b_new.max()
        new_val = 2.0 * n * max_b_new

        # acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probability
            prob = np.exp(-(new_val - cur_val) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # update a and b and cur_val
            a[i] -= delta
            a[j] += delta
            # ensure nonnegative
            # a[i] might become slightly negative due to floating; clip
            if a[i] < 0:
                a[i] = 0.0
            # keep sum exactly 1 (numerical error)
            # after each move sum should remain 1 (within eps)
            # we can renormalize occasionally
            cur_val = new_val
            b = b_new
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA]    new best {best_val:.6f} at iter {iter_count}")

        # occasional global jitter to escape local minima
        if iter_count % 5000 == 0 and rng.random() < 0.1:
            # add small Gaussian noise, reproject onto simplex
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SA] jitter improved to {best_val:.6f}")

        iter_count += 1

    return a, b, cur_val, best_a, best_val

                       deadline, start_time, rng):

    max_iters = 200_000
    a_cur = a.copy()
    while True:
        now = time.time()
        if now >= deadline:
            break

"""
Improved step‑function search
-----------------------------

The optimiser works in three stages:

1. **Initial solution** – we either reuse the previously found sequence
   (global `height_sequence_1`) or draw a few Beta‑shaped candidates and
   keep the best one (objective ≈ 1.5).

2. **Projected sub‑gradient polishing** – a plain sub‑gradient descent on
   the exact L∞‑norm of the self‑convolution, with a linearly decaying
   step‑size.  This quickly drives the maximum of the convolution down.

3. **Simulated‑annealing fine‑tuning** – random pairwise transfers of
   mass between entries are performed.  The resulting change of the
   convolution can be updated in O(n) time, allowing millions of cheap
   moves.  An annealing schedule accepts occasional uphill moves, helping
   the search escape local minima.  The best vector seen during this
   phase is kept.

Finally the sequence is evaluated with the provided ``evaluate_sequence``
function and returned as a plain list of floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # numerical noise may produce tiny negatives – clip.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact objective  (assumes Σa = 1)
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ------------------------------------------------------------
# Random Beta‑shaped candidate generator
# ------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return a normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ≈ 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray,
                       n: int,
                       cur_val: float,
                       best_a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       start_time: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, np.ndarray, float, np.ndarray, float]:
    """Run sub‑gradient descent until *deadline*."""
    a = a0.copy()
    it = 0
    # parameters for step‑size schedule
    step_start = 0.50
    step_end = 1e-5

    while True:
        now = time.time()
        if now >= deadline:
            break

        # compute convolution and objective
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        # locate all maximal positions
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient (sum of contributions from all maximal diagonals)
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size decay
        progress = (now - start_time) / (deadline - start_time)
        step = step_start * (1.0 - progress) + step_end * progress

        # gradient step and projection
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        # evaluate new point
        b_new = _full_convolution(a_new)
        new_val = 2.0 * n * b_new.max()

        # accept improvement
        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SubG] new best = {best_val:.6f} (iter {it})")

        # occasional random jitter to escape plateaux
        if it % 5000 == 0 and rng.random() < 0.03:
            a = np.maximum(a + rng.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

        it += 1

    # final convolution for the returned a
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    return a, b, cur_val, best_a, best_val


# ------------------------------------------------------------
# Simulated‑annealing fine‑tuning (pairwise mass transfers)
# ------------------------------------------------------------
def _annealing_phase(a0: np.ndarray,
                     b0: np.ndarray,
                     n: int,
                     cur_val: float,
                     best_a: np.ndarray,
                     best_val: float,
                     deadline: float,
                     start_time: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, np.ndarray, float, np.ndarray, float]:
    """
    Randomly move a small amount of mass from index i to j.
    The convolution is updated in O(n) time using the analytic formula.
    An annealing schedule permits occasional uphill moves.
    """
    a = a0.copy()
    b = b0.copy()
    iter_cnt = 0

    # annealing schedule parameters
    step_start = 0.05      # max fraction of source mass that may be moved
    step_end = 0.001
    temp_start = 0.02 * cur_val
    temp_end = 0.0002 * cur_val

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        step_frac = step_start * (1.0 - progress) + step_end * progress
        temperature = temp_start * (1.0 - progress) + temp_end * progress

        # select source index proportionally to its weight (large entries move)
        i = rng.choice(n, p=a)
        # select a distinct destination uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # choose a transfer amount
        delta = step_frac * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # ---- fast O(n) update of the convolution ----
        # copy because we may reject the move
        b_new = b.copy()
        # 2·δ·a contributions
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # δ² point contributions
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta
        # numerical safety
        np.maximum(b_new, 0.0, out=b_new)

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept if better, otherwise with Metropolis probability
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # commit the move
            a[i] -= delta
            a[j] += delta
            # keep numerical sanity
            if a[i] < 0.0:
                a[i] = 0.0
            # (mass is conserved by construction)
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {iter_cnt})")

        # occasional global jitter to inject fresh noise
        if iter_cnt % 5000 == 0 and rng.random() < 0.1:
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SA] jitter improved best to {best_val:.6f}")

        iter_cnt += 1

    return a, b, cur_val, best_a, best_val


# ------------------------------------------------------------
# Small polishing: convex combination with the uniform vector
# ------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex mixtures t·a + (1‑t)·u (t∈[0,1]) and keep the best."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand = _project_simplex(cand)          # guarantees Σ=1, ≥0
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length *n* that minimises
    ``evaluate_sequence``.  The routine runs a short sub‑gradient phase
    followed by a simulated‑annealing fine‑tune, all within the given
    time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin to stay inside the allotted budget
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # Choose a problem size.  If a previous solution exists we keep its length.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)     # type: ignore[name-defined]
        n = old.shape[0]
    except Exception:
        n = 8000                     # default size used by the reference code

    # ------------------------------------------------------------------
    # 0️⃣  Initialise – reuse the old solution if possible,
    #     otherwise fall back to a few random Beta draws.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.maximum(old, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / a0.sum()
        print("[Init] Re‑using previous sequence (len = %d)" % n)
    else:
        # cheap stochastic sweep for a decent starter
        a0 = None
        best_val = np.inf
        for _ in range(30):
            cand = _random_beta_candidate(n, rng)
            val = _objective(cand)
            if val < best_val:
                best_val = val
                a0 = cand
        if a0 is None:               # should never happen
            a0 = np.full(n, 1.0 / n)
        print("[Init] Random Beta start, obj = %.6f" % _objective(a0))

    # initialise objective and best‑so‑far record
    b0 = _full_convolution(a0)
    cur_val = 2.0 * n * b0.max()
    best_a = a0.copy()
    best_val = cur_val
    print("[Init] initial objective = %.6f" % cur_val)

    # ------------------------------------------------------------------
    # Phase A – short projected sub‑gradient polishing (≈25 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.25 * budget_s
    a, b, cur_val, best_a, best_val = _subgradient_phase(
        a0, n, cur_val, best_a, best_val,
        deadline=phase_a_deadline,
        start_time=start_time,
        rng=rng
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – simulated‑annealing fine‑tuning (remainder of the budget)
    # ------------------------------------------------------------------
    a, b, cur_val, best_a, best_val = _annealing_phase(
        a, b, n, cur_val, best_a, best_val,
        deadline=deadline,
        start_time=phase_a_deadline,
        rng=rng
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation (the function may clip/scale internally)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[Final] evaluate_sequence reports %.6f" % final_score)
    except Exception as exc:   # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Ensure a plain Python list of floats is returned
    return best_a.tolist()
