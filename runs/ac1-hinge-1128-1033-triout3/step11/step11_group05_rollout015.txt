def compute_active_constraints(a, conv, tol=1e-6):
    max_val = conv.max()
    active = np.where(conv >= max_val * (1 - tol))[0] # indices set
    return active

def solve_lp_step(a, M, active_idx):
    n = len(a)
    # variables b of size n
    # constraints: for each k in active_idx, sum_i a_i * b_{k-i} <= M
    # Represent as matrix A (len(active) x n)
    # Build using Toeplitz? We can fill dense matrix for active count maybe <= 50, okay.
    m = len(active_idx)
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        # indices i where 0 <= k-i < n => i in [max(0, k-(n-1)), min(k, n-1)]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi+1)
        A[row, i_range] = a[k - i_range]
    # Solve LP: maximize sum(b) subject to A @ b <= M, b >= 0
    c = -np.ones(n)  # since linprog minimizes, we use -c for maximize sum
    bounds = [(0, None)] * n
    # Use high-level linprog
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    return b

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  The search consists of three stages:

1️⃣  Wide exploration of many parametric families (Beta, mixtures of two
    Betas, exponential, Gaussian, power‑sine, triangular, sparse
    Dirichlet draws).  All candidates are normalised to a unit L1‑norm
    and the best few are kept.

2️⃣  Mirror‑descent refinement using a sub‑gradient of the max‑convolution.
    The active set contains the 20 largest convolution entries (rather
    than only 5) and a Adam‑style learning‑rate schedule is employed.

3️⃣  A cheap “linear‑programming lift” step: using the current best
    sequence we form a small LP with the most active convolution
    constraints (≤ 30).  Solving the LP (via ``scipy.optimize.linprog``)
    yields a direction that increases the total mass while keeping the
    max‑convolution ≤ the current bound.  A convex combination with a
    decaying step‑size is applied and the loop repeats until the time
    budget is exhausted.

Finally the best sequence found is evaluated with the provided
``evaluate_sequence`` and returned.  The implementation respects the
global 1000 s budget and prints intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx = 1, x ≥ 0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape families for the initial wide exploration
# ----------------------------------------------------------------------
def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """a_i ∝ i^{p‑1}(n‑i+1)^{q‑1},   i = 1,…,n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _mixture_beta_shape(p1: float, q1: float, p2: float, q2: float,
                        lam: float, n: int) -> np.ndarray | None:
    """Mixture λ·Beta(p1,q1) + (1‑λ)·Beta(p2,q2)."""
    a1 = _beta_shape(p1, q1, n)
    a2 = _beta_shape(p2, q2, n)
    if a1 is None or a2 is None:
        return None
    a = lam * a1 + (1.0 - lam) * a2
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_shape(c: float, n: int) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·|i‑mid|)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a


def _gaussian_shape(sigma: float, n: int) -> np.ndarray:
    """Gaussian centred at the middle."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a


def _power_sine_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1))^p ."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    a /= a.sum()
    return a


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient for the max‑convolution (active‑set version)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 20,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Return a sub‑gradient of max(b) w.r.t. a.
    The active set consists of the ``max_active`` largest entries of b,
    filtered by a relative tolerance.
    The returned vector is the *average* of the individual gradients,
    which is a valid sub‑gradient.
    """
    n = a.shape[0]
    max_val = b.max()
    if max_active >= b.size:
        act = np.arange(b.size)
    else:
        act = np.argpartition(b, -max_active)[-max_active:]
        mask = b[act] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            act = np.array([int(np.argmax(b))])
        else:
            act = act[mask]

    grad = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0 / max(1, len(act))   # average over the active set
    return grad


# ----------------------------------------------------------------------
# Linear‑programming refinement step (small active set)
# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray,
             M: float,
             active_idx: np.ndarray,
             max_iter: int = 1) -> np.ndarray | None:
    """
    Solve a tiny LP:
        maximise   Σ b_j
        subject to a * b (convolution) ≤ M   on the specified active indices
                   b_j ≥ 0
    Returns the optimal b (or None if the LP fails).
    """
    # Import lazily – SciPy may not be available in minimal environments.
    try:
        from scipy.optimize import linprog
    except Exception as exc:          # pragma: no cover
        print("[lp] SciPy not available:", exc)
        return None

    n = a.shape[0]
    m = active_idx.shape[0]
    A = np.zeros((m, n), dtype=np.float64)

    for row, k in enumerate(active_idx):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=int)
        A[row, i_range] = a[k - i_range]

    # linprog minimises; we maximise Σ b_i  → minimise -Σ b_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c,
                  A_ub=A,
                  b_ub=np.full(m, M, dtype=np.float64),
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if not res.success:
        return None
    return res.x

def propose_candidate(seed: int = 42,

    returns the best sequence discovered before the deadline.

    deadline = start_time + budget_s - 0.5

    best_val = np.inf

        """Update global best if candidate improves the exact objective."""
        nonlocal best_seq, best_val
        val = _objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[search]{tag} new best={best_val:.6f}")

    # --------------------------------------------------------------
    # 0️⃣  Initialise from a previously saved solution (if any)
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        # interpolate to the current resolution
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _maybe_update(a0, tag=" [init]")
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Wide exploration over several parametric families
    # --------------------------------------------------------------
    # (a) dense grid of Beta shapes
    p_grid = np.concatenate(
        [
            np.geomspace(0.001, 0.1, 8, endpoint=False),
            np.geomspace(0.1, 5.0, 12, endpoint=False),
            np.geomspace(5.0, 20.0, 5),
        ]
    )
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is not None:
                _maybe_update(cand, tag=" [beta grid]")

    # (b) random mixtures of two Betas (adds much flexibility)
    rng = np.random.default_rng()
    for _ in range(4000):
        p1 = np.exp(rng.uniform(np.log(1e-3), np.log(20.0)))
        q1 = np.exp(rng.uniform(np.log(1e-3), np.log(20.0)))
        p2 = np.exp(rng.uniform(np.log(1e-3), np.log(20.0)))
        q2 = np.exp(rng.uniform(np.log(1e-3), np.log(20.0)))
        lam = rng.random()
        cand = _mixture_beta_shape(p1, q1, p2, q2, lam, n)
        if cand is not None:
            _maybe_update(cand, tag=" [beta mix]")

        if time.time() >= deadline - 250.0:   # reserve time for refinement
            break

    # (c) exponential decay family
    for _ in range(3000):
        c = rng.uniform(0.0, 5.0)
        cand = _exp_shape(c, n)
        _maybe_update(cand, tag=" [exp]")

    # (d) Gaussian family
    for _ in range(3000):
        sigma = rng.uniform(0.5, n / 5.0)
        cand = _gaussian_shape(sigma, n)
        _maybe_update(cand, tag=" [gauss]")

    # (e) power‑sine family
    for _ in range(3000):
        p = rng.uniform(0.1, 8.0)
        cand = _power_sine_shape(p, n)
        _maybe_update(cand, tag=" [psine]")

    # (f) simple triangular window
    cand = _triangular_shape(n)
    _maybe_update(cand, tag=" [tri]")

    # (g) sparse Dirichlet‑like samples (few non‑zero entries)
    for _ in range(1200):
        support = rng.integers(10, 2000)
        idx = rng.choice(n, size=support, replace=False)
        vals = rng.random(support)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = vals
        cand /= cand.sum()
        _maybe_update(cand, tag=" [sparse]")

    # fallback – uniform if nothing succeeded (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)
        print("[fallback] uniform start, value=", best_val)

    # --------------------------------------------------------------
    # 2️⃣  Mirror‑descent style refinement (Adam‑like)
    # --------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)               # first moment
    v = np.zeros_like(a)               # second moment
    beta1, beta2 = 0.9, 0.999
    step0 = 0.8
    eps = 1e-8
    it = 0
    no_improve = 0
    max_no_improve = 5000

            best_seq = a.copy()

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        v_hat = v / (1.0 - beta2 ** (it + 1))

        # linear decay of step size
        progress = (time.time() - start_time) / (deadline - start_time)
        step = step0 * (1.0 - progress) + 1e-4 * progress

        log_factor = -step * m_hat / (np.sqrt(v_hat) + eps)
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)

        # re‑normalise onto the simplex
        a = _project_simplex(a)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.001 or no_improve >= max_no_improve:
            jitter = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)
            no_improve = 0
            m.fill(0.0)
            v.fill(0.0)

        it += 1

    # --------------------------------------------------------------
    # 3️⃣  LP‑based refinement (small active set)
    # --------------------------------------------------------------
    a = best_seq.copy()
    lp_iter = 0
    while time.time() < deadline - 0.5:
        conv = _full_convolution(a)
        M = conv.max()

        # pick up to 30 most critical constraints
        K = 30
        if conv.size <= K:
            active_idx = np.arange(conv.size)
        else:
            active_idx = np.argpartition(conv, -K)[-K:]

        b = _lp_step(a, M, active_idx)
        if b is None:
            # fall back to a tiny random jitter and continue
            jitter = np.random.normal(scale=5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)
            continue

        sum_b = b.sum()
        if sum_b <= 0:
            # degenerate solution – jitter instead
            jitter = np.random.normal(scale=5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)
            continue

        g = b / sum_b

        # decreasing convex‑combination weight (larger at start)
        t = max(0.05, 0.5 * (1.0 - lp_iter / 200.0))
        a = (1.0 - t) * a + t * g
        a = _project_simplex(a)

        # track improvement
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[lp] iter {lp_iter:04d} new best {best_val:.6f}")

        lp_iter += 1
        if lp_iter >= 250:          # enough LP steps for the remaining budget
            break

    # --------------------------------------------------------------
    # 4️⃣  Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:           # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
