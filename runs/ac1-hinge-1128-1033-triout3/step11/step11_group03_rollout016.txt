```
def _advanced_hillclimb(a, n, deadline, start_time, target_val=1.503, max_iter=10**6):
    cur = a.copy()
    cur_conv = _full_convolution(cur)
    cur_val = _objective_from_conv(cur_conv, n)
    best = cur.copy()
    best_val = cur_val
    stagnation = 0
    step = 0.2 # initial step size maybe.
    while time.time() < deadline:
        k_max = int(np.argmax(cur_conv))
        # compute gradient as 2*a_rev_shifted
        grad = np.zeros_like(cur)
        for idx_j in range(n):
            idx = k_max - idx_j
            if 0 <= idx < n:
                grad[idx_j] = 2.0 * cur[idx]
        # sometimes adapt step size based on curvature
        eta = 0.05 / (1 + stagnation*0.001)  # decays with stagnation

        a_new = cur - eta * grad
        # enforce nonnegative and project onto simplex
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= a_new_sum

        new_conv = _full_convolution(a_new)
        new_val = _objective_from_conv(new_conv, n)

        if new_val < cur_val - 1e-12:
            cur = a_new
            cur_conv = new_conv
            cur_val = new_val
            stagnation = 0
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
                print(f"[AdvHill] new best = {best_val:.6f}")
                if best_val <= target_val:

        else:
            stagnation += 1

                cur = cur / cur_sum

                cur_val = _objective_from_conv(cur_conv, n)
                stagnation = 0

    return best, best_val
```

```
def _anti_symmetric_initial(n):
    # draw Dirichlet with all ones (uniform random)
    v = np.random.dirichlet(np.ones(n))
    v_sorted = np.sort(v)[::-1]  # descending
    half = n // 2
    a = np.empty(n)
    a[:half] = v_sorted[:half]      # largest half on left
    a[-half:] = v_sorted[-half:][::-1]  # smallest half on right reversed
    if n % 2 == 1:
        a[half] = v_sorted[half]   # middle element
    a = np.maximum(a,0)
    a /= a.sum()
    return a
```

"""
Simple multi‑stage optimiser for the “step‑function attack”.

1️⃣  Initialise a family of plausible shapes (Gaussian, Laplace, power‑law,
    Beta‑like, bimodal, anti‑symmetric Dirichlet …).
2️⃣  Run a short Adam optimiser on a log‑sum‑exp (“soft‑max”) surrogate of the
    max‑convolution.  The surrogate becomes tighter as α grows, while Adam
    supplies a cheap stochastic‑gradient style update that respects the simplex
    (non‑negative, sum‑to‑one) constraint.
3️⃣  Perform a deterministic sub‑gradient descent on the true max (using the
    exact gradient of the offending lag) and fall back to a simple hill‑climb
    that moves mass from a “guilty’’ index to a “benign’’ one.
4️⃣  Keep the best solution seen across several sequence lengths and return it.

The code respects a 1000 s time budget; after that the best sequence discovered
so far is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)-1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_rev(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·∑_i a_i·w_{i+j} for j=0..n‑1.
    Implements the gradient of the soft‑max surrogate.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    l_w = w.shape[0]
    out_len = n + l_w - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]
    start = n - 1
    return conv[start:start + n]        # length n


# ----------------------------------------------------------------------
# 2.  Exact objective (scale‑invariant, sum of a is 1)
# ----------------------------------------------------------------------
def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact value 2·n·max(a∗a) given the full convolution."""
    return float(2.0 * n * conv.max())


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (assumed non‑negative, sum>0)."""
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.shape[0])


# ----------------------------------------------------------------------
# 3.  Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx = 1} (Euclidean)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 4.  Soft‑max surrogate helpers
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Computes normalized soft‑max weights w_k ∝ exp(α·conv_k)."""
    maxc = conv.max()
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exp_vals / sum_exp


# ----------------------------------------------------------------------
# 5.  Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _smooth_adam_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          alpha_start: float = 5.0,
                          alpha_end: float = 200.0,
                          lr0: float = 0.1,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser of the smooth objective.
    Returns the best sequence found (sum=1) and its exact objective.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_conv = _full_convolution(a)
    cur_val = _objective_from_conv(cur_conv, n)
    best_a = a.copy()
    best_val = cur_val

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        alpha = alpha_start + (alpha_end - alpha_start) * prog

        # soft‑max weights and gradient
        w = _softmax_weights(cur_conv, alpha)          # shape (2n‑1,)
        grad = 2.0 * _conv_rev(a, w)                    # shape (n,)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        cand_conv = _full_convolution(a_candidate)
        cand_val = _objective_from_conv(cand_conv, n)

        # accept only if we do not increase the exact objective
        if cand_val <= cur_val:
            a = a_candidate
            cur_conv = cand_conv
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[Adam] iter {t:7d} α={alpha:6.1f} val={best_val:.6f}")
                if best_val <= 1.503:
                    break
        # else: reject, keep old a (momentum stays as is)

        # occasional random perturbation to escape plateaus
        if t % 3000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_conv = _full_convolution(a)
            cur_val = _objective_from_conv(cur_conv, n)

    return best_a, best_val

# 6.  Parametric families (extended)
# ----------------------------------------------------------------------
def _parametric_candidates(n: int):
    """Yield a handful of hand‑crafted shapes (Gaussian / Laplace / power‑law)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Gaussian families – centre location at the centre
    for sigma_factor in (0.10, 0.15, 0.20, 0.25, 0.30):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Laplace (double‑exponential) families
    for scale_factor in (0.08, 0.12, 0.16, 0.20):
        scale = scale_factor * n
        a = np.exp(-np.abs(i - mid) / scale)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Power‑law / Beta‑like families (symmetric)
    for p in (0.2, 0.4, 0.5, 0.8, 1.0, 1.5, 2.0):
        a = (i + 1) ** (p - 1.0) * (n - i) ** (p - 1.0)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Bimodal Gaussians (two peaks near the ends)
    for sigma_factor in (0.03, 0.05):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i) / sigma) ** 2) + np.exp(-0.5 * ((i - (n - 1)) / sigma) ** 2)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Exponential decay (left‑to‑right)
    for lam in (0.0005, 0.001, 0.002):
        a = np.exp(-lam * i)
        if a.sum() > 0:
            a /= a.sum()
            yield a

# 7.  Anti‑symmetric Dirichlet initialiser
# ----------------------------------------------------------------------
def _anti_symmetric_dirichlet(n: int) -> np.ndarray:
    """
    Draw a Dirichlet vector and arrange it so that mirrored entries are
    anti‑correlated (large with small).  This tends to reduce the product
    a_i·a_{n‑1‑i} that dominates the centre of the convolution.
    """
    raw = np.random.dirichlet(np.ones(n))
    vals = np.sort(raw)[::-1]          # descending
    half = n // 2
    a = np.empty(n, dtype=np.float64)
    a[:half] = vals[:half]             # biggest half on the left
    a[-half:] = vals[-half:][::-1]     # smallest half on the right, reversed
    if n % 2 == 1:
        a[half] = vals[half]           # middle element
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def _interpolate_previous(prev_seq, n):

        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)

        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None

# 9.  Advanced hill‑climb (sub‑gradient descent + random jitter)

```python
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        cand_conv = _full_convolution(a_candidate)
        cand_val = _objective_from_conv(cand_conv, n)

        # accept only if we do not increase the exact objective
        if cand_val <= cur_val:
            a = a_candidate
            cur_conv = cand_conv
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[Adam] iter {t:7d} α={alpha:6.1f} val={best_val:.6f}")
                if best_val <= 1.503:
                    break
        # else: reject, keep old a (momentum stays as is)

        # occasional random perturbation to escape plateaus
        if t % 3000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_conv = _full_convolution(a)
            cur_val = _objective_from_conv(cur_conv, n)

    return best_a, best_val


# ----------------------------------------------------------------------
# 6.  Parametric families (extended)
# ----------------------------------------------------------------------
def _parametric_candidates(n: int):
    """Yield a handful of hand‑crafted shapes (Gaussian / Laplace / power‑law)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Gaussian families – centre location at the centre
    for sigma_factor in (0.10, 0.15, 0.20, 0.25, 0.30):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Laplace (double‑exponential) families
    for scale_factor in (0.08, 0.12, 0.16, 0.20):
        scale = scale_factor * n
        a = np.exp(-np.abs(i - mid) / scale)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Power‑law / Beta‑like families (symmetric)
    for p in (0.2, 0.4, 0.5, 0.8, 1.0, 1.5, 2.0):
        a = (i + 1) ** (p - 1.0) * (n - i) ** (p - 1.0)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Bimodal Gaussians (two peaks near the ends)
    for sigma_factor in (0.03, 0.05):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i) / sigma) ** 2) + np.exp(-0.5 * ((i - (n - 1)) / sigma) ** 2)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Exponential decay (left‑to‑right)
    for lam in (0.0005, 0.001, 0.002):
        a = np.exp(-lam * i)
        if a.sum() > 0:
            a /= a.sum()
            yield a


# ----------------------------------------------------------------------
# 7.  Anti‑symmetric Dirichlet initialiser
# ----------------------------------------------------------------------
def _anti_symmetric_dirichlet(n: int) -> np.ndarray:
    """
    Draw a Dirichlet vector and arrange it so that mirrored entries are
    anti‑correlated (large with small).  This tends to reduce the product
    a_i·a_{n‑1‑i} that dominates the centre of the convolution.
    """
    raw = np.random.dirichlet(np.ones(n))
    vals = np.sort(raw)[::-1]          # descending
    half = n // 2
    a = np.empty(n, dtype=np.float64)
    a[:half] = vals[:half]             # biggest half on the left
    a[-half:] = vals[-half:][::-1]     # smallest half on the right, reversed
    if n % 2 == 1:
        a[half] = vals[half]           # middle element
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 8.  Interpolation of a previously found sequence (if present)
# ----------------------------------------------------------------------
def _interpolate_previous(prev_seq, n):
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None


# ----------------------------------------------------------------------
# 9.  Advanced hill‑climb (sub‑gradient descent + random jitter)
# ----------------------------------------------------------------------
def _advanced_hillclimb(a: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        target_val: float = 1.503,
                        max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution combined with a simple
    random‑jitter fallback.  The routine stops early if ``target_val`` is reached
    or when ``deadline`` expires.
    """
    # Current state
    conv = _full_convolution(a)
    cur_val = _objective_from_conv(conv, n)
    best_val = cur_val
    best_a = a.copy()

    stagn = 0
    it = 0
    while it < max_iters:
        now = time.time()
        if now >= deadline:
            break

        # index of the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # sub‑gradient of (a*a)[k_max] : g_j = 2·a_{k_max-j}  (if in range)
        grad = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2.0 * a[idx[mask]]

        # Simple step size (decays slowly when we stall)
        step = 0.15 / (1.0 + 0.001 * stagn)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= s

        conv_new = _full_convolution(a_new)
        val_new = _objective_from_conv(conv_new, n)

        if val_new < cur_val - 1e-12:
            # improvement – accept
            a = a_new
            conv = conv_new
            cur_val = val_new
            stagn = 0
            if val_new < best_val:
                best_val = val_new
                best_a = a.copy()
                print(f"[SGD] iter {it:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            stagn += 1
            # occasional random perturbation to escape a flat region
            if stagn % 5000 == 0:
                a = best_a.copy()
                noise = np.random.normal(scale=0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                a_sum = a.sum()
                a = a / a_sum if a_sum != 0 else np.full_like(a, 1.0 / n)
                conv = _full_convolution(a)
                cur_val = _objective_from_conv(conv, n)
                stagn = 0

        it += 1

    return best_a, best_val

# 10.  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to minimise
    ``evaluate_sequence``.  The routine respects the time budget and returns the
    best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # Sequence lengths that we will try.  Longer sequences give more flexibility
    # but also cost more per iteration.
    sizes = [5000, 8000, 12000, 16000, 24000]

    # Grab the previously reported solution if it exists.
    try:
        previous = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    except Exception:
        previous = None

    global_best_a = None
    global_best_val = np.inf
    global_best_n = None

    # ------------------------------------------------------------------
    # Phase 1 – initialise many candidates and run short Adam optimisations
    # ------------------------------------------------------------------
    for n in sizes:
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        if remaining < 10.0:
            break

        # Allocate roughly a third of the remaining time to this size,
        # but cap it so that we keep some budget for the final polishing.
        size_budget = min(remaining * 0.33, 250.0)

        # --------------------------------------------------------------
        # Assemble a diverse pool of starting points
        # --------------------------------------------------------------
        starters = []

        # a) interpolated previous solution
        if previous is not None:
            a0 = _interpolate_previous(previous, n)
            if a0 is not None:
                starters.append(a0)

        # b) hand‑crafted parametric families
        for cand in _parametric_candidates(n):
            starters.append(cand)

        # c) random Dirichlet + anti‑symmetric variant
        for _ in range(4):
            cand = np.random.dirichlet(np.ones(n))
            starters.append(cand)
            starters.append(_anti_symmetric_dirichlet(n))

        # d) plain uniform baseline
        starters.append(np.full(n, 1.0 / n))

        # Keep the best among the raw starters (no optimisation yet)
        best_local_a = None
        best_local_val = np.inf
        for cand in starters:
            val = _objective(cand)
            if val < best_local_val:
                best_local_val = val
                best_local_a = cand

        # --------------------------------------------------------------
        # Run a few independent Adam optimisations starting from the best
        # local starter (with some added noise to diversify the runs)
        # --------------------------------------------------------------
        runs = 3
        per_run_deadline = time.time() + size_budget / runs
        for run_idx in range(runs):
            if time.time() >= deadline:
                break
            a0 = best_local_a.copy()
            a0 += np.random.normal(scale=0.005, size=n)
            a0 = np.maximum(a0, 0.0)
            a0_sum = a0.sum()
            a0 = a0 / a0_sum if a0_sum != 0 else np.full(n, 1.0 / n)

            a_opt, val_opt = _smooth_adam_optimize(
                a0,
                n,
                per_run_deadline,
                start_time,
                alpha_start=0.5,
                alpha_end=400.0,
                lr0=0.2,
            )
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_a = a_opt.copy()
                global_best_n = n

            if global_best_val <= 1.503:
                break

        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 2 – final polishing with deterministic sub‑gradient descent
    # ------------------------------------------------------------------
    if global_best_a is not None:
        remaining = deadline - time.time()
        if remaining > 2.0:
            polished_a, polished_val = _advanced_hillclimb(
                global_best_a,
                global_best_n,
                deadline=deadline,
                start_time=start_time,
                target_val=1.503,
            )
            if polished_val < global_best_val:
                global_best_a = polished_a
                global_best_val = polished_val
            print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final evaluation (for logging – the host will re‑evaluate anyway)
    # ------------------------------------------------------------------
    if global_best_a is None:
        # fallback to a uniform distribution of the smallest size we tried
        n = sizes[0]
        global_best_a = np.full(n, 1.0 / n)
        global_best_val = _objective(global_best_a)

    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
```
