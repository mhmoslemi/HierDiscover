    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    n = a.shape[0]

        if time.time() >= deadline:
            break
        b = _full_convolution(a)
        max_b = b.max()
        idxs = np.where(b == max_b)[0]
        grad = np.zeros_like(a)
        for k in idxs:

* Optimise the smooth surrogate ϕε(a)=max(conv(a,a))+ε·log∑exp((conv−max)/ε)
  which converges to the true max as ε→0.
* The optimisation is performed in the log‑softmax (probability‑simplex)
  space, yielding an unconstrained problem for L‑BFGS.
* A continuation schedule (ε=1e‑2 → 1e‑12) allows the optimiser to first
  move quickly in a smooth landscape and then fine‑tune the exact max.
* A cheap projected sub‑gradient descent polishes the final vector.
* Several start vectors are examined (including a geometric family with
  analytically good behaviour) and the best is kept.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns smooth max(conv(a,a)) and the gradient of 2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp weights
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    wsum = weights.sum()
    p = weights / wsum

    # gradient w.r.t. a (the factor 2·n is already included)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]

    smooth_val = max_b + eps * np.log(wsum)
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth surrogate (log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_obj_and_grad(u: np.ndarray, n: int, eps: float) -> tuple[float, np.ndarray]:
    """
    Objective and gradient for L‑BFGS when parameters are the logits u.
    The returned value is 2·n·smooth_max and its gradient w.r.t. u.
    """
    w = u - np.max(u)                 # stabilise soft‑max
    a = np.exp(w)
    a /= a.sum()

    # smooth max and gradient w.r.t. a
    smooth_val, grad_a = _smooth_max_and_grad(a, eps)

    # chain rule through the soft‑max (a = softmax(u))
    dot = np.dot(grad_a, a)
    grad_u = a * (grad_a - dot)

    # objective includes the external factor 2·n
    val = 2.0 * n * smooth_val
    return val, grad_u.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, eps: float,
                    max_iter: int = 150) -> np.ndarray:
    """
    Perform a single L‑BFGS run on the smooth surrogate with smoothing ε.
    Returns a probability vector (sum = 1) and its exact objective value.
    """
    u0 = np.log(a0 + 1e-12)          # start in log‑softmax space
    res = minimize(_lbfgs_obj_and_grad, u0,
                   args=(n, eps),
                   method='L-BFGS-B',
                   jac=True,
                   options={'maxiter': max_iter,
                            'ftol': 1e-12,
                            'disp': False})
    u_opt = res.x
    w = u_opt - np.max(u_opt)
    a_opt = np.exp(w)
    a_opt /= a_opt.sum()
    return a_opt


def _lbfgs_continuation(a_start: np.ndarray, n: int,
                        deadline: float) -> np.ndarray:
    """
    Run L‑BFGS through a decreasing ε‑schedule (continuation).
    Stops early if the time budget expires.
    """
    a = a_start.copy()
    eps_schedule = [1e-2, 1e-4, 1e-6, 1e-8, 1e-10, 1e-12]
    for eps in eps_schedule:
        if time.time() >= deadline:
            break
        # fewer iterations for coarse stages, more for fine stages
        iters = 80 if eps >= 1e-6 else 200
        a = _lbfgs_optimize(a, n, eps, max_iter=iters)
    return a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iters: int = 150_000) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient for the active diagonal(s)
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size decay
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-6 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        new_val = _objective(a_new)

        if new_val < best_val:
            best_val, best_a = new_val, a_new.copy()
        if new_val < _objective(a):
            a = a_new

        # occasional tiny jitter to break ties
        if it % 800 == 0 and np.random.rand() < 0.02:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Blend with uniform (helps smooth isolated spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random families for cheap initialisation
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a (p,q) pair on a log‑scale and return the corresponding Beta‑shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ~1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _geometric_candidate(n: int, delta_mul: float) -> np.ndarray:
    """Geometric shape a_i ∝ r^i with r = 1‑δ, δ = delta_mul / n."""
    delta = delta_mul / n
    if delta <= 0.0 or delta >= 1.0:
        return np.full(n, 1.0 / n)
    r = 1.0 - delta
    # Normalisation constant C = δ / (1‑rⁿ)
    denom = 1.0 - r ** n
    if denom <= 0.0:
        return np.full(n, 1.0 / n)
    C = delta / denom
    a = C * (r ** np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function within the given time budget.
    Returns a plain list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size – larger n gives a richer search space
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  initialise – keep a handful of promising seeds
    # --------------------------------------------------------------
    candidates = []

    # 0.1  reuse a previously discovered solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        x_new = np.linspace(0.0, 1.0, n)
        x_old = np.linspace(0.0, 1.0, old_len)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        candidates.append(a)
        print("[Init] interpolated previous solution")
    except Exception:
        pass

    # 0.2  uniform
    candidates.append(np.full(n, 1.0 / n))

    # 0.3  a few geometric families (δ≈1/n is known to be good)
    for mul in (0.5, 1.0, 2.0, 4.0):
        candidates.append(_geometric_candidate(n, mul))

    # 0.4  many cheap random Beta / Dirichlet draws
    for _ in range(25):
        candidates.append(_random_beta_candidate(n, rng))
    for alpha in (0.1, 0.5, 1.0, 2.0, 5.0):
        candidates.append(rng.dirichlet(np.full(n, alpha)))

    # keep only the best few (exact objective)
    scored = [( _objective(c), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    top_inits = [vec for _, vec in scored[:5]]
    best_a = top_inits[0].copy()
    best_val = _objective(best_a)
    print(f"[Init] best of cheap seeds  = {best_val:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop (L‑BFGS continuation + polishing)
    # --------------------------------------------------------------
    for idx, a0 in enumerate(top_inits):
        if time.time() >= deadline:
            break
        print(f"[Stage] start L‑BFGS run #{idx+1}")

        # L‑BFGS continuation (smooth surrogate)
        a_opt = _lbfgs_continuation(a0, n, deadline)

        val_opt = _objective(a_opt)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Stage] new best after L‑BFGS = {best_val:.6f}")

        # short exact sub‑gradient polishing
        polish_deadline = min(deadline, time.time() + 0.05 * (deadline - start_time))
        a_pol = _subgradient_polish(best_a, n, polish_deadline,
                                    start_time=time.time(),
                                    max_iters=60_000)
        val_pol = _objective(a_pol)
        if val_pol < best_val:
            best_val, best_a = val_pol, a_pol.copy()
            print(f"[Stage] new best after polishing = {best_val:.6f}")

        # blend a little with uniform – cheap way to kill stray spikes
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # --------------------------------------------------------------
    # Final fine‑tune (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[Final] extra L‑BFGS fine‑tune")
        best_a = _lbfgs_continuation(best_a, n, deadline)
        best_val = _objective(best_a)

    # --------------------------------------------------------------
    # Secure evaluation and return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Result] final reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Result] evaluate_sequence raised:", exc)

    # ensure a plain list of floats is returned
    return best_a.tolist()
