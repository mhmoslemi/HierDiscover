"""
Step‑function optimizer for `evaluate_sequence`.

The algorithm works with a probability vector a (sum = 1) represented by
logits z (soft‑max).  It repeatedly

  1. computes the self‑convolution a*a,
  2. builds a smooth (or hard) max‑weight vector w over the convolution,
  3. obtains the gradient of   Σ w_k·(a*a)_k   with respect to a
     via a cross‑correlation (FFT‑based),
  4. back‑propagates through the soft‑max,
  5. applies an Adam update to the logits,
  6. injects Gaussian noise after long stagnation.

The total time budget (≈1000 s) is split among several problem sizes.
Each run is given an equal share of the remaining time, the best
solution across all runs is kept and finally returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k·a_{k-i} for i=0..len(a)-1.
    Implemented as convolution of w with the reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = len(w) + n - 1
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _run_opt(
    n: int,
    a0: np.ndarray,
    run_deadline: float,
    global_start: float,
    global_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector of length ``n`` starting from ``a0``.
    Returns the best vector (sum = 1) found and its objective value.
    """
    eps = 1e-12
    z = np.log(a0 + eps)               # logits for the soft‑max
    # Adam state
    lr = 0.03
    b1, b2 = 0.9, 0.999
    adam_eps = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # best‑so‑far bookkeeping
    best_a = a0.copy()
    best_val = float("inf")
    stagnation = 0
    IMPROVE_LIMIT = 3000
    it = 0

    # schedule for the smooth‑max temperature λ
    lambda_start = 0.5
    lambda_end = 3000.0
    switch_lambda = 200.0   # switch to hard max when λ exceeds this

    run_start = time.time()

    while True:
        now = time.time()
        if now >= run_deadline - 0.1:
            break
        it += 1

        # ---- logits → probability vector (soft‑max) ----
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # ---- full self‑convolution & objective ----
        conv = _full_convolution(a)          # length 2n‑1
        conv_max = conv.max()
        cur_val = 2.0 * n * conv_max          # because Σa = 1

        # ---- keep global best for this run ----
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            stagnation = 0
            print(
                f"[n={n:05d} it={it:07d}] new best = {best_val:.9f}"
            )
        else:
            stagnation += 1

        # ---- λ schedule (linear in elapsed fraction of the run) ----
        prog = (now - run_start) / (run_deadline - run_start)
        lam = lambda_start + (lambda_end - lambda_start) * prog

        # ---- weight vector w over convolution entries ----
        if lam < switch_lambda:
            # smooth‑max (softmax) weighting
            exp_vals = np.exp(lam * (conv - conv_max))
            w = exp_vals / exp_vals.sum()
        else:
            # hard max: equal weight over all indices attaining the maximum
            mask = (conv >= conv_max - 1e-12)
            w = mask.astype(float)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(conv) / conv.shape[0]
            else:
                w = w / w_sum

        # ---- gradient of Σ w_k·(a*a)_k w.r.t. a ----
        g_raw = _cross_corr(w, a)    # length n
        g = 2.0 * g_raw               # ∂/∂a of the weighted max‑conv

        # ---- back‑prop through the soft‑max ----
        dot_ag = np.dot(a, g)
        grad_z = a * (g - dot_ag)    # ∂f/∂z

        # ---- Adam update of logits ----
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z ** 2)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ---- stagnation handling (Gaussian perturbation) ----
        if stagnation > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.1, size=n)
            z += noise
            stagnation = 0
            print(f"[n={n:05d} it={it:07d}] reset – added noise")

    return best_a, best_val

    n_list = [4000, 8000, 12000, 16000]

    # ------------------------------------------------------------------

            a0 = np.interp(new_x, old_x, init_seq)
        else:
            a0 = np.full(n, 1.0 / n)

        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        if s0 <= 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / s0

        # ----- run the optimiser for this size -----
        best_a_n, best_val_n = _run_opt(
            n, a0, run_deadline, start_time, deadline
        )

        # ----- update the global best -----
        if best_val_n < global_best_val:
            global_best_val = best_val_n
            global_best_a = best_a_n.copy()
            global_best_n = n
            print(
                f"[global] new best (n={n}) = {global_best_val:.9f}"
            )

        # use the obtained solution as initialization for the next size
        init_seq = best_a_n

    # ------------------------------------------------------------------
    # If there is still time left, perform a short refinement on the
    # current global best.
    # ------------------------------------------------------------------
    leftover = deadline - time.time()
    if global_best_a is not None and leftover > 10.0:
        print("[refine] spending remaining time on final refinement")
        refined_a, refined_val = _run_opt(
            global_best_n,
            global_best_a,
            deadline,
            start_time,
            deadline,
        )
        if refined_val < global_best_val:
            global_best_a = refined_a
            global_best_val = refined_val
            print(f"[refine] improved to {global_best_val:.9f}")

    # ------------------------------------------------------------------
    # Fallback – if everything failed, return a simple uniform vector.
    # ------------------------------------------------------------------
    if global_best_a is None:
        n = 4000
        global_best_a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final secure evaluation (only one call to ``evaluate_sequence``).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
