def _max_subgradient_optimize(a0, n, deadline, start_time, step0=0.5, min_step=1e-8, max_iter=1e9):
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= s
    best_a = a.copy(); best_val = _objective(a)
    step = step0
    iteration=0
    while time.time() < deadline:
        # compute conv and max index
        conv = _full_convolution(a)
        k = int(np.argmax(conv))
        grad = _conv_rev(a, (np.arange(2*n-1)==k).astype(float))
        # attempt step
        improved=False
        while step >= min_step:
            a_candidate = a - step * grad
            a_candidate = np.maximum(a_candidate, 0)
            a_candidate = _project_simplex(a_candidate)
            val = _objective(a_candidate)
            if val < best_val - 1e-12:
                best_val = val
                best_a = a_candidate.copy()
                improved=True
                # optionally increase step a bit after success
                step = min(step*1.2, step0)   # increase back
                break
            else:
                # reduce step and try again
                step *= 0.5
        if not improved:
            # if cannot improve, maybe random perturb
            # perturb and continue
            step = step0
            # add small random noise
            a = a + np.random.normal(scale=1e-3, size=n)
            a = np.maximum(a, 0)
            a = _project_simplex(a)
            continue
        else:
            a = best_a.copy()
        iteration+=1
        if iteration % 1000 == 0:
            print("[SGD] iter", iteration, "best_val", best_val)

    return best_a, best_val

"""
Improved step‑function optimizer.

The strategy is a three‑phase pipeline:

1️⃣  **Diverse initialisation** – Gaussian / Laplace families,
    classic window functions (Hann, Hamming, Blackman, Bartlett,
    Cosine), a few Dirichlet draws and, if available,
    interpolation of the previously‑found best sequence.

2️⃣  **Smooth‑max Adam** – minimise a soft‑max surrogate
    of the maximal autocorrelation.
    The temperature (α) is increased to a very large value
    (α≈2000) so that the gradient eventually concentrates on the
    true max, while Adam’s adaptive steps and projection onto the
    probability simplex keep the iterates stable.

3️⃣  **Exact max‑subgradient descent** – after the surrogate has
    converged we switch to the true sub‑gradient (gradient of the
    maximal lag) and perform a simple projected (Adam‑style)
    descent with a decaying step size.  This phase is able to
    break the tiny plateaus that the smooth surrogate cannot
    resolve.

A final short hill‑climbing polish is applied before the best
sequence is returned.  The whole routine respects a 1000 s
budget and prints intermediate bounds for monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of `x` with itself (length 2·len(x)−1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_rev(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Implements the gradient of Σ_k w_k·(a*a)_k.
    The returned vector has length n and already contains the factor 2.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    l_w = w.shape[0]
    out_len = n + l_w - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]
    start = n - 1
    return conv[start:start + n]        # length n


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx = 1} (Euclidean)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` (scale‑invariant)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    s = a.sum()
    # avoid divide‑by‑zero; the caller guarantees s>0
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Soft‑max surrogate helpers
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(α·conv_k)."""
    maxc = conv.max()
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exp_vals / sum_exp


# ----------------------------------------------------------------------
# Adam optimiser for the smooth surrogate
# ----------------------------------------------------------------------
def _smooth_adam_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          alpha_start: float = 5.0,
                          alpha_end: float = 2000.0,
                          lr0: float = 0.12,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max surrogate; returns best (sum=1) sequence."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_conv = _full_convolution(a)
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear schedule for α
        prog = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        alpha = alpha_start + (alpha_end - alpha_start) * prog

        w = _softmax_weights(cur_conv, alpha)            # shape (2n‑1,)
        grad = _conv_rev(a, w)                           # shape (n,)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        cand_val = _objective(a_candidate)

        # accept only non‑worsening updates
        if cand_val <= cur_val:
            a = a_candidate
            cur_conv = _full_convolution(a)
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[Adam] iter {t:7d} α={alpha:6.1f} val={best_val:.6f}")
                if best_val <= 1.503:
                    break
        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_conv = _full_convolution(a)
            cur_val = _objective(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact max‑subgradient optimiser (projected Adam)
# ----------------------------------------------------------------------
def _max_subgradient_optimize(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              step0: float = 0.5,
                              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam‑style descent using the true sub‑gradient of the peak lag.
    Returns the best sequence found.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        conv = _full_convolution(a)
        k = int(np.argmax(conv))
        # unit weight at the maximal lag
        w = np.zeros(2 * n - 1, dtype=float)
        w[k] = 1.0
        grad = _conv_rev(a, w)               # already contains factor 2

        # Adam‑style adaptive step (decay ~ 1/√t)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        step = step0 / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        cand_val = _objective(a_candidate)

        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_a = a_candidate.copy()
            print(f"[MaxGD] iter {t:7d} val={best_val:.6f}")
            a = a_candidate
            if best_val <= 1.503:
                break
        else:
            # if no improvement, keep the old a (momentum still evolves)
            a = a

        # occasional random jitter (to escape flat regions)
        if t % 8000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Light‑weight hill‑climbing (mass‑transfer) for final polishing
# ----------------------------------------------------------------------
def _final_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Very simple hill‑climbing that moves a tiny amount of mass from the
    most “guilty’’ index to a low‑contribution index."""
    b = _full_convolution(a)
    cur_val = _objective(a)
    best_val = cur_val
    best_a = a.copy()
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-12:
            continue

        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = _objective(a_new)

        if new_val < cur_val:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = _objective(a)
            no_improve = 0

        iteration += 1

    return best_a, best_val

# Small parametric families for initialisation
# ----------------------------------------------------------------------
def _parametric_candidates(n: int):
    """Yield a handful of hand‑crafted shapes (Gaussian, Laplace,
    power‑law, and classic window functions)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Gaussian families
    for sigma_factor in (0.18, 0.22, 0.26, 0.30):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Laplace families
    for scale_factor in (0.12, 0.16, 0.20):
        scale = scale_factor * n
        a = np.exp(-np.abs(i - mid) / scale)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Power‑law / Beta‑like families
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = (i + 1) ** (p - 1.0) * (n - i) ** (p - 1.0)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Classic window functions (Hann, Hamming, Blackman, Bartlett, Cosine)
    # Hann
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    if w.sum() > 0:
        a = w / w.sum()
        yield a
    # Hamming
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    if w.sum() > 0:
        a = w / w.sum()
        yield a
    # Blackman
    w = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    if w.sum() > 0:
        a = w / w.sum()
        yield a
    # Bartlett (triangular)
    w = 1 - np.abs((i - (n - 1) / 2) / ((n - 1) / 2))
    if w.sum() > 0:
        a = w / w.sum()
        yield a
    # Cosine
    w = np.sin(np.pi * i / (n - 1))
    if w.sum() > 0:
        a = w / w.sum()
        yield a

def _interpolate_previous(prev_seq, n):
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None

        global_best_a = np.full(n, 1.0 / n)
        global_best_val = _objective(global_best_a)

    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
