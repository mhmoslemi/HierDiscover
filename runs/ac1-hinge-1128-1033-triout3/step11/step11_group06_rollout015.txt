   max_conv = conv.max()
   if beta > 1e3:
       # subgradient
       ind = np.flatnonzero(conv >= max_conv - 1e-12)
       w = np.zeros_like(conv)
       w[ind] = 1.0 / len(ind)
   else:
       exp_arg = beta * (conv - max_conv)
       w = np.exp(exp_arg)
       w_sum = w.sum()
       w /= w_sum
   # compute gradient (weighted cross-correlation) using FFT
   # Equivalent to convolution of a with reversed w
   rev_w = w[::-1]
   size = 1 << (len(rev_w) + len(a) -1 -1).bit_length()
   fa = np.fft.rfft(a, size)
   fw = np.fft.rfft(rev_w, size)
   corr_full = np.fft.irfft(fa * fw, size)[:len(rev_w)+len(a)-1]
   offset = len(conv) - 1
   idx = offset - np.arange(len(a))
   grad = 2.0 * corr_full[idx]
   return grad, max_conv

def _gradient_weighted(a, beta):
    n = a.shape[0]
    # Determine size for FFT (next power of two for conv length)
    conv_len = 2*n - 1
    size = 1 << (conv_len - 1).bit_length()
    # compute fft of a
    fa = np.fft.rfft(a, size)
    # full convolution
    conv = np.fft.irfft(fa * fa, size)[:conv_len]
    max_conv = conv.max()
    if beta >= 1e5:
        # subgradient
        mask = (conv >= max_conv - 1e-12)
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0/conv_len)
        else:
            w /= w_sum
    else:
        exp_arg = beta * (conv - max_conv)  # <=0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        w /= w_sum
    # cross-correlation gradient:
    rev_w = w[::-1]
    # size for cross-correlation same as before (since rev_w length = conv_len)
    size_corr = size  # because n+conv_len-1 <= size (most likely). Ensure size large enough.
    # For safety, recompute size_corr as next power of two for n + conv_len - 1.
    corr_len = n + conv_len - 1
    size_corr = 1 << (corr_len - 1).bit_length()
    Fa = fa  # if size_corr > size, need recompute Fa with larger size.
    if size_corr != size:
        Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:corr_len]
    offset = conv_len - 1
    idx = offset - np.arange(n)
    grad = 2.0 * corr_full[idx]
    return grad, max_conv

"""
Propose a step‑function (non‑negative vector) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm works in three phases:

1️⃣  **Coarse initialization** – sample a few Beta‑shaped families
    (including U‑shaped cases with p,q<1) and a few Dirichlet draws,
    keeping the best candidate according to the exact objective.

2️⃣  **Smoothed (and eventually sub‑) gradient descent** on the
    infinity‑norm of the self‑convolution.
    The loss ϕβ(a)= (1/β)·log Σ exp(β·(a⋆a)k) is differentiable;
    for large β it becomes the true max.  A Nesterov‑style momentum
    scheme together with Euclidean projection onto the probability
    simplex drives the iterate.  The smoothing parameter β is raised
    slowly from 0.5 up to a huge value (≈10⁶) so the final steps act
    on the exact max‑convolution.

3️⃣  **Final secure evaluation** – the best vector seen is handed to
    ``evaluate_sequence`` and returned as a plain Python list.

All heavy work uses FFTs, O(n log n), and the whole search respects the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()           # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised (sum=1) vector."""
    n = a.shape[0]
    max_conv = _full_convolution(a).max()
    return float(2.0 * n * max_conv)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the parameters lead to an all‑zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _gradient_weighted(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smoothed objective

        φβ(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i),

    and the exact maximal self‑convolution value.
    For very large beta the weights become a (normalised) indicator of the
    maximal entry, i.e. a sub‑gradient of the true max‑convolution.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()

    # ----- 1. full convolution a⋆a -----
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fa * Fa, size)[:out_len]
    max_conv = conv.max()

    # ----- 2. weights (soft‑max or pure max) -----
    if beta >= 1e5:                     # pure sub‑gradient
        mask = conv >= max_conv - 1e-12
        w = mask.astype(float)
        ws = w.sum()
        if ws > 0:
            w /= ws
        else:                           # fallback – uniform
            w = np.full_like(conv, 1.0 / out_len)
    else:
        exp_arg = beta * (conv - max_conv)      # ≤0, safe from overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        w /= w_sum

    # ----- 3. gradient = 2·(a ⋆_cross w_rev) -----
    rev_w = w[::-1]
    out_len_corr = n + out_len - 1
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa_corr = np.fft.rfft(a, size_corr) if size_corr != size else Fa
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa_corr * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1                # position of lag 0 in the full cross‑corr
    idx = offset - np.arange(n)         # indices for j = 0…n‑1
    grad = 2.0 * corr_full[idx]

    return grad, max_conv
