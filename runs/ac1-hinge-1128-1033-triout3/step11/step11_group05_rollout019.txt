def sidon_greedy(n, rng, max_attempts=5):
    # We'll run multiple passes to enlarge set
    best_positions = []
    for attempt in range(max_attempts):
        positions = []
        sums = set()
        # random permutation of indices
        order = rng.permutation(n)
        for idx in order:
            # check sums uniqueness
            # early break optimization: if len(positions) + (n - idx) <= best_len maybe break?
            conflict = False
            if (idx + idx) in sums:
                conflict = True
            else:
                for p in positions:
                    if (idx + p) in sums:
                        conflict = True
                        break
            if not conflict:
                # add
                for p in positions:
                    sums.add(idx + p)
                sums.add(idx + idx)
                positions.append(int(idx))
        if len(positions) > len(best_positions):
            best_positions = positions
        # optionally break if near upper bound
        if len(best_positions) >= int(np.sqrt(4 * n)):
            break
    return best_positions

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of ``x`` with itself (full length 2·len(x)‑1) using FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

    """
    Exact value of 2·n·max(conv(a,a)) assuming sum(a)=1.
    Used only for cheap comparisons during the search.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

    Σx = 1 , x ≥ 0.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (full active set)
# ----------------------------------------------------------------------


def _subgradient_max_full(a: np.ndarray,
                          conv: np.ndarray,
                          rel_tol: float = 1e-12) -> np.ndarray:
    """
    Return a sub‑gradient of ``max(conv)`` w.r.t. ``a``.
    All indices whose convolution value is within ``rel_tol`` of the
    global maximum are regarded as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    # active indices (allow a tiny tolerance for floating point noise)
    active = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution a_{k-j} for each j in [lo,hi]
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Helper to create a random start on the simplex
# ----------------------------------------------------------------------


def _random_simplex(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Draw a point uniformly from the simplex Σx=1, x≥0
    (Dirichlet with all parameters equal to 1).
    """
    x = rng.random(n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    returns the best sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # problem size – a good trade‑off between expressivity and speed
    n = 8000

    best_seq: np.ndarray | None = None
    best_val = np.inf

    def _maybe_update(candidate: np.ndarray):
        """Update the global best if the candidate improves the exact objective."""
        nonlocal best_seq, best_val
        val = _objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[improve] new best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Simple dense starts (Dirichlet draws)
    # ------------------------------------------------------------------
    init_trials = 120
    for _ in range(init_trials):
        cand = _random_simplex(rng, n)
        _maybe_update(cand)
        if time.time() >= deadline - 200.0:   # reserve ~200 s for refinement
            break

    # If for any reason we have not found a candidate, fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)
        print("[fallback] uniform start, value =", best_val)

    # ------------------------------------------------------------------
    # 1️⃣  Mirror‑descent optimisation on the exact objective
    # ------------------------------------------------------------------
    a = best_seq.copy()               # current iterate
    m = np.zeros_like(a)              # momentum accumulator
    beta = 0.9                        # momentum coefficient
    it = 0
    no_improve = 0
    max_no_improve = 8000             # after this many stagnating steps we jitter

    while time.time() < deadline:
        # ----- evaluate current state -----
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[optim] iter {it:5d}  new best {best_val:.6f}")
        else:
            no_improve += 1

        # ----- sub‑gradient from *all* maximisers -----
        grad = _subgradient_max_full(a, conv, rel_tol=1e-12)

        # ----- momentum (exponential moving average) -----
        m = beta * m + (1.0 - beta) * grad

        # ----- step‑size schedule (inverse‑sqrt decay) -----
        step = 0.5 / np.sqrt(it + 1.0)

        # ----- multiplicative (exponentiated) update -----
        log_factor = -step * m
        log_factor = np.clip(log_factor, -50.0, 50.0)   # avoid overflow
        a = a * np.exp(log_factor)

        # ----- renormalise to stay on the simplex -----
        s = a.sum()
        if s > 0.0 and np.isfinite(s):
            a = a / s
        else:
            # fall back to a fresh random point if something went wrong
            a = _random_simplex(rng, n)

        # ----- occasional jitter to escape flat plateaus -----
        if np.random.rand() < 0.001 or no_improve >= max_no_improve:
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            no_improve = 0

        it += 1

    # ------------------------------------------------------------------
    # 2️⃣  Simple pairwise hill‑climbing (mass transfer) as a final polish
    # ------------------------------------------------------------------
    a = best_seq.copy()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a small fraction of mass from i to j
        delta = min(a[i], 0.01) * rng.random()
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv = _full_convolution(a_new)
        val = 2.0 * n * conv.max()

        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            print(f"[swap] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
