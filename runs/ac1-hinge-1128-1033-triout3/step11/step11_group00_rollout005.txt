"""
Propose a candidate step‑function that improves on the previous 1.508200
bound.

Approach
--------
1. Initialise a probability vector `a` (length n=8000) – if a previous
   solution (`height_sequence_1`) exists we interpolate it, otherwise a
   U‑shaped Beta(0.5,0.5) distribution is used.
2. Optimise the logits `z = log(a)` with Adam while keeping `a` on the
   probability simplex (soft‑max of the logits).
3. The first half of the time budget uses a *soft‑max* of the full
   self‑convolution (`conv = a*a`) – this gives a smooth approximation
   of the true max‑entry and yields a useful gradient.
4. In the second half we switch to the *exact* max‑entry gradient
   (sub‑gradient of `max(conv)`).  For the active index `k_max` we have
   `∂max/∂a_i = 2·a_{k_max-i}`; the chain‑rule through the soft‑max of
   the logits gives `∂L/∂z = a·(g_a - <g_a,a>)`.
5. Adam updates are performed with a linearly decaying learning rate.
   If no improvement is seen for many iterations a small random jitter
   is added to the logits to escape plateaus.
6. After the time limit a short entropy‑mirror‑descent polishing
   (`_max_refine`) is applied to the best vector found.
7. The final sequence is returned and optionally evaluated with the
   provided `evaluate_sequence`.

The algorithm respects the 1000 s budget, uses only FFT‑based O(n log n)
convolutions and runs entirely on a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (real inputs)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two different vectors (full length)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror‑descent polishing (exact max entry)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray,
                steps: int = 1500,
                eta_start: float = 0.45) -> np.ndarray:
    """
    Perform mirror‑descent on the exact max‑entry of the convolution.
    The iteration keeps the vector on the probability simplex.
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            # sub‑gradient of max‑entry w.r.t. a (ordered pairs)
            g = np.zeros_like(a)
            g[idx] = 2.0 * a[k - idx]
            eta = eta_start * (1.0 - t / steps)
            a = a * np.exp(-eta * g)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s
    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.8               # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the same size that gave the previous best.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initial distribution (probability vector, sum = 1).
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        # Linear interpolation to the target size
        x_new = np.linspace(0.0, 1.0, n)
        x_old = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(x_new, x_old, old)
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= a0_sum
        print("[init] interpolated previous solution")
    except Exception:
        # a U‑shaped Beta(0.5,0.5) – more mass at the ends, tends to flatten convolution
        i = np.arange(1, n + 1, dtype=np.float64)
        p = q = 0.5
        a0 = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        a0_sum = a0.sum()
        a0 /= a0_sum
        print("[init] generated Beta(0.5,0.5) start")

    # Parameters for the log‑softmax representation
    eps = 1e-30
    z = np.log(a0 + eps)          # logits

    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    lr_start, lr_end = 0.12, 0.003           # learning‑rate schedule
    alpha_max = 500.0                        # final smoothing strength

    best_a = None
    best_obj = np.inf
    best_iter = 0
    stagnation = 0

    it = 0
    # split the time budget in two phases: smooth → exact
    warmup_fraction = 0.55   # proportion of the total time spent with soft‑max

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start) / (deadline - start)   # 0 → 1

        # --------------------------------------------------------------
        # Re‑build probability vector from logits
        # --------------------------------------------------------------
        a = np.exp(z)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # --------------------------------------------------------------
        # Full self‑convolution (needed for both phases)
        # --------------------------------------------------------------
        conv = _full_convolution(a)          # length 2n‑1
        max_val = conv.max()
        obj = 2.0 * n * max_val               # exact objective value

        # --------------------------------------------------------------
        # Keep the best exact solution seen so far
        # --------------------------------------------------------------
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            best_iter = it
            stagnation = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}  t={now-start:.1f}s")
        else:
            stagnation += 1

        # --------------------------------------------------------------
        # Determine which gradient to use (soft‑max or exact max)
        # --------------------------------------------------------------
        if prog < warmup_fraction:
            # ----- Soft‑max (smooth) gradient -----
            # learning‑rate for the soft‑max part
            prog_w = prog / warmup_fraction
            alpha = 1.0 + (alpha_max - 1.0) * (prog_w ** 1.5)

            # stable soft‑max of the convolution
            maxc = max_val
            exp_scaled = np.exp(alpha * (conv - maxc))
            weights = exp_scaled / exp_scaled.sum()          # Σ weights = 1

            # gradient of the smoothed objective:
            # g_a = 4·n·(rev_a * weights)
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)   # length 3n‑2
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]          # slice length n
        else:
            # ----- Exact max gradient (sub‑gradient) -----
            k = int(conv.argmax())
            # Compute g_a[i] = 4·n·a[k-i] whenever the index is valid
            g_a = np.zeros_like(a)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                g_a[idx] = 4.0 * n * a[k - idx]

        # --------------------------------------------------------------
        # Chain rule through the soft‑max (a = softmax(z))
        # --------------------------------------------------------------
        dot = np.dot(g_a, a)
        grad_z = a * (g_a - dot)

        # --------------------------------------------------------------
        # Adam update on logits
        # --------------------------------------------------------------
        it += 1
        lr = lr_start * (1.0 - prog) + lr_end * prog

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------------------------------------------------------------
        # Simple escape mechanism if we have been stuck for a while
        # --------------------------------------------------------------
        if stagnation >= 3000:
            z += np.random.normal(scale=0.02, size=n)
            stagnation = 0
            print(f"[reset] jitter added at it={it}")

    # ------------------------------------------------------------------
    # Polishing phase: entropy‑mirror descent on the exact max entry
    # ------------------------------------------------------------------
    if best_a is not None:
        a_refined = _max_refine(best_a, steps=1800, eta_start=0.45)
        obj_ref = 2.0 * n * _full_convolution(a_refined).max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = a_refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation using the provided wrapper
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else a0.tolist()

        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
