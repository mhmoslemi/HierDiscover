"""
Propose a step‑function that (approximately) minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
1️⃣  Initialise several diverse candidates (uniform, decreasing
    exponentials, reversed exponentials and a few random Dirichlet draws).

2️⃣  Refine each candidate with an Adam optimiser on a *smooth*
    approximation of the max‑convolution:
        φ_β(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i) .
    The gradient is the weighted cross‑correlation described in the
    original code.  β grows slowly from 0.5 to a large value (≈300)
    so that the smoothed problem converges to the true max‑convolution.
    After every Adam step the iterate is projected onto the probability
    simplex (non‑negative, sum = 1).

3️⃣  Keep the best iterate seen over the whole run (objective = 2·n·maxconv).

4️⃣  A short “local‑search” phase tries random small mass transfers
    between two indices; any improvement replaces the best solution.

The whole search stops just before the time budget expires and returns the
best sequence found.  The routine works for any length ``n``; a
moderately large ``n=12000`` gives enough expressive power while staying
fast enough for many thousands of Adam updates within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_correlation(x: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of `x` with `w`:
        (x ⋆_cross w)[j] = Σ_i x_i·w_{i+j}
    Returns the vector of length len(x) (the part needed for the gradient).
    """
    n = x.shape[0]
    out_len = n + len(w) - 1
    size = 1 << (out_len - 1).bit_length()
    Fx = np.fft.rfft(x, size)
    Fw = np.fft.rfft(w[::-1], size)    # reverse w for convolution → cross‑correlation
    corr_full = np.fft.irfft(Fx * Fw, size)[:out_len]
    # centre index for lag 0
    offset = len(w) - 1
    idx = offset - np.arange(n)          # indices for lag 0 … n‑1
    return corr_full[idx]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the probability simplex Σ x = 1, x ≥ 0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective and its gradient
# ----------------------------------------------------------------------


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Smoothed max‑convolution gradient.
    Returns (grad, max_conv) where grad has the same shape as ``a`` and
    ``max_conv`` is the exact max of the (unsmoothed) convolution a⋆a.
    """
    n = a.shape[0]

    # --- 1. full convolution a⋆a ---------------------------------
    out_len = 2 * n - 1
    size_c = 1 << (out_len - 1).bit_length()
    Fa = np.fft.rfft(a, size_c)
    conv = np.fft.irfft(Fa * Fa, size_c)[:out_len]      # (a⋆a)

    max_conv = conv.max()

    # --- 2. soft‑max weights ------------------------------------
    # shift by max to avoid overflow
    exp_arg = beta * (conv - max_conv)    # ≤ 0
    w = np.exp(exp_arg)
    w /= w.sum()                          # Σ_i w_i = 1

    # --- 3. gradient = 2·(a ⋆_cross w) -------------------------
    grad = 2.0 * _cross_correlation(a, w)

    return grad, max_conv


# ----------------------------------------------------------------------
# Objective (exact, for bookkeeping)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final eval

    # --------------------------------------------------------------
    # problem size – a compromise between expressivity and speed
    # --------------------------------------------------------------
    n = 12000

    # --------------------------------------------------------------
    # 0️⃣  Build a list of diverse initial candidates
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    uniform = np.full(n, 1.0 / n)
    init_candidates.append(uniform)

    # decreasing exponentials (a_i ∝ r**i)
    for r in (0.9995, 0.9997, 0.99985, 0.99992):
        a = np.exp(np.arange(n, dtype=np.float64) * np.log(r))
        a /= a.sum()
        init_candidates.append(a)

    # increasing exponentials (reverse of decreasing)
    for r in (0.9995, 0.9997, 0.99985):
        a = np.exp(np.arange(n, dtype=np.float64) * np.log(r))[::-1]
        a /= a.sum()
        init_candidates.append(a)

    # a few random Dirichlet draws for extra diversity
    rng = np.random.default_rng()
    for _ in range(5):
        a = rng.dirichlet(np.ones(n))
        init_candidates.append(a)

    # --------------------------------------------------------------
    # 1️⃣  Adam optimiser on the smoothed objective
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    lr0 = 0.12                     # base learning rate (will be scaled)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-9

    # smoothing schedule
    beta_start = 0.5
    beta_max = 350.0
    beta_mul = 1.0015            # moderate growth – many β‑updates before hitting max

    best_score = np.inf
    best_seq = None

    # We'll loop over the initial candidates; each run gets a share of the time.
    # The number of candidates is modest, so we can afford a few hundred
    # thousand Adam updates in total.
    for idx, a0 in enumerate(init_candidates):
        # stop early if we are out of time
        if time.time() >= deadline:
            break

        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        beta = beta_start
        t_iter = 0

        # keep a local best for this run
        local_best = _objective(a)
        local_seq = a.copy()

        # run until either time runs out or we have performed a large number
        # of iterations (cap to avoid infinite loops on tiny budgets)
        max_iters = 250_000
        while time.time() < deadline and t_iter < max_iters:
            t_iter += 1

            # learning‑rate schedule (Adam style)
            lr = lr0 * np.sqrt(1 - beta2**t_iter) / (1 - beta1**t_iter)

            # gradient of smoothed max‑convolution
            grad, max_conv = _gradient_smooth(a, beta)
            cur_score = 2.0 * n * max_conv

            # keep the globally best solution seen so far
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[update] global best = {best_score:.6f} "
                      f"(iter {t_iter}, init {idx}, β={beta:.2f})")

            # also remember a local best (useful after β saturates)
            if cur_score < local_best:
                local_best, local_seq = cur_score, a.copy()

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1**t_iter)
            v_hat = v / (1.0 - beta2**t_iter)

            # descent step
            a -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto the simplex (Σa=1, a≥0)
            a = _project_simplex(a)

            # increase smoothing parameter
            beta = min(beta_max, beta * beta_mul)

        # after the inner loop, propagate the local best if it beats the global one
        if local_best < best_score:
            best_score, best_seq = local_best, local_seq.copy()
            print(f"[final‑run] new global best from run {idx} = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Local random perturbation (tiny hill‑climbing)
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform (should never happen)
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # small moves: take ε mass from a random index i and add it to j
    # accept if the objective improves.
    a = best_seq.copy()
    cur_score = best_score
    move_eps = 1e-4          # amount of mass to move each trial
    trials = 0
    max_trials = 30_000

    while time.time() < deadline and trials < max_trials:
        trials += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] < move_eps:
            continue

        a_new = a.copy()
        a_new[i] -= move_eps
        a_new[j] += move_eps

        # re‑project only to fix tiny negative due to rounding
        if a_new[i] < 0:
            a_new[i] = 0.0
        a_new = _project_simplex(a_new)

        new_score = _objective(a_new)
        if new_score < cur_score:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[hill‑climb] improve to {best_score:.6f} (trial {trials})")

    # --------------------------------------------------------------
    # 3️⃣  Final secure evaluation and return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as e:   # pragma: no cover
        print("[final] evaluate_sequence raised:", e)

    return best_seq.tolist()
