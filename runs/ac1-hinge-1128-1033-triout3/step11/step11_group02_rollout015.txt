def _smooth_obj_grad(y, lam, n, size1, size2):
    # compute a = softmax(y)
    max_y = np.max(y)
    exp_y = np.exp(y - max_y)
    a = exp_y / np.sum(exp_y)

    # conv a*a
    fa = np.fft.rfft(a, size1)
    conv = np.fft.irfft(fa * fa, size1)[:2*n - 1]

    # compute w as softmax over lam * conv
    max_c = np.max(conv)
    # shift for stability
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp  # shape (2n-1)

    # gradient wrt a
    a_rev = a[::-1]
    fw = np.fft.rfft(w, size2)
    fa_rev = np.fft.rfft(a_rev, size2)
    conv_aw = np.fft.irfft(fa_rev * fw, size2)[:3*n - 2]
    grad_a = 2.0 * conv_aw[n-1: n-1 + n]  # shape n

    # gradient wrt y using softmax Jacobian
    mu = np.dot(grad_a, a)  # scalar
    grad_y = a * (grad_a - mu)

    # objective = (1/lam) * log sum exp(lam*conv) = max_c + (1/lam)*log(sum_exp)
    obj = max_c + np.log(sum_exp) / lam
    return obj, grad_y

def _lbfgs_optimize(y0, lam_seq, n, deadline):
    size1 = 1 << ((2*n - 2).bit_length())
    size2 = 1 << ((3*n - 2).bit_length())
    y = y0.copy()
    best_a = None
    best_val = np.inf
    for lam in lam_seq:
        if time.time() >= deadline:
            break
        # define closure for objective/grad for given lam
        def fun(y_vec):
            obj, grad = _smooth_obj_grad(y_vec, lam, n, size1, size2)
            return obj, grad
        # SciPy minimize expects separate functions; we can use lambda returning fun.
        # but we need to pass gradient separately using Jacobian argument. 
        # We'll use minimize with method='L-BFGS-B', jac=True.
        res = minimize(lambda yy: fun(yy)[0],
                       y,
                       method='L-BFGS-B',
                       jac=lambda yy: fun(yy)[1],
                       options={'maxiter': 200, 'ftol': 1e-12, 'gtol':1e-12})
        y = res.x
        # compute a
        a = np.exp(y - np.max(y))
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L-BFGS lam={lam:.1f}] new best = {best_val:.6f}")
        if best_val <= 1.503:
            break
    return best_a, best_val

"""
Propose a step‑function that beats the previous best (≈1.5084) for the
evaluation function.  

The algorithm proceeds in three phases:

1️⃣  **Diverse seeds** – a refined sweep of the two‑parameter Beta family
    (a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}) plus a few Dirichlet draws and, when
    available, an interpolation of the previous solution.

2️⃣  **Smooth‑max L‑BFGS optimisation** – the nonsmooth objective
    ``2·n·max(conv(a,a))`` is replaced by a smooth log‑sum‑exp
    approximation.  The variables are re‑parameterised with a soft‑max
    (so the simplex constraint is always satisfied).  For a schedule of
    increasing smoothing parameters λ∈{5,10,20,40,80,120,200} a limited‑
    iteration L‑BFGS run (SciPy) is performed; after each λ the exact
    objective is evaluated and the best sequence is kept.

3️⃣  **Intensive hill‑climbing** – the classic simulated‑annealing
    mover from the reference implementation finishes the remaining time,
    starting from the best soft‑max solution.

The whole search respects the 1000 s budget; any remaining time after
phase 2 is given to phase 3.  The returned list of floats is a
non‑negative probability vector (∑a=1) and is fed directly to
``evaluate_sequence``.

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray, size: int = None) -> np.ndarray:
    """Return the full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    if size is None:
        size = 1 << ((out_len - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_conv_two(a: np.ndarray, b: np.ndarray, size: int = None) -> np.ndarray:
    """Full linear convolution of a and b (both 1‑D)."""
    n = a.shape[0] + b.shape[0] - 1
    if size is None:
        size = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:n]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (fallback)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Beta‑family sweep (denser grid than the reference implementation)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> Tuple[np.ndarray, float]:
    """Search the symmetric Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf

    # a finer two‑dimensional grid (≈1500 points – still cheap)
    p_vals = np.concatenate([
        np.linspace(0.15, 4.0, 30),          # fine near the optimum
        np.geomspace(4.0, 20.0, 6)[1:],      # a few larger exponents
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max objective (+ gradient) for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_grad(
    y: np.ndarray,
    lam: float,
    n: int,
    size1: int,
    size2: int,
) -> Tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) of the smoothed max
    f_λ(a) = (1/λ)·log∑_k e^{λ·(a*a)_k},
    where a = softmax(y) (so Σa=1, a_i≥0).
    """
    # ---- soft‑max transform to enforce the simplex ----
    max_y = np.max(y)
    exp_y = np.exp(y - max_y)       # stable
    a = exp_y / exp_y.sum()

    # ---- convolution a*a (full) ----
    conv = _full_convolution(a, size=size1)          # length 2·n‑1
    max_c = conv.max()

    # ---- smooth weights w_k ∝ exp(λ·(conv_k – max_c)) ----
    #   (max_c subtraction prevents overflow)
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp               # Σ w = 1, length 2·n‑1

    # ---- gradient w.r.t. a (same derivation as in the reference code) ----
    a_rev = a[::-1]
    fw = np.fft.rfft(w, size2)
    fa_rev = np.fft.rfft(a_rev, size2)
    conv_aw = np.fft.irfft(fa_rev * fw, size2)[: 3 * n - 2]
    grad_a = 2.0 * conv_aw[n - 1 : n - 1 + n]

    # ---- chain rule through the soft‑max ----
    mu = np.dot(grad_a, a)                # scalar ⟨grad_a , a⟩
    grad_y = a * (grad_a - mu)            # ∂obj/∂y_i

    # ---- exact smoothed objective value ----
    obj = max_c + np.log(sum_exp) / lam    # = (1/λ)·log∑e^{λ·conv}
    return float(obj), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS optimisation over the smoothing schedule
# ----------------------------------------------------------------------
def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> Tuple[np.ndarray, float]:
    """
    Runs a short L‑BFGS optimisation for a sequence of λ values.
    Returns the best (a, value) found before the deadline.
    """
    if minimize is None:      # pragma: no cover
        raise RuntimeError("SciPy not available – L‑BFGS cannot run.")

    # initialise y = log(a) (soft‑max parametrisation)
    eps = 1e-12
    y = np.log(a0 + eps)

    # FFT sizes needed throughout the run (pre‑computed)
    size1 = 1 << ((2 * n - 2).bit_length())       # for a*a
    size2 = 1 << ((3 * n - 2).bit_length())       # for a_rev * w

    # smoothing schedule (more aggressive than the reference)
    lam_seq = [5.0, 10.0, 20.0, 40.0, 80.0, 120.0, 200.0]

    best_a = None
    best_val = np.inf

    for lam in lam_seq:
        now = time.time()
        if now >= deadline:
            break

        # small, time‑bounded L‑BFGS run for the current λ
        max_iter = 70

        def fun(yy):
            return _smooth_obj_grad(yy, lam, n, size1, size2)

        try:
            res = minimize(
                fun,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter, "disp": False},
            )
        except Exception as exc:          # pragma: no cover
            print(f"[L‑BFGS λ={lam}] optimisation failed:", exc)
            continue

        y = res.x
        # obtain the probability vector and evaluate the *exact* objective
        a = np.exp(y - np.max(y))
        a /= a.sum()
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam}] new best = {best_val:.6f}")

        # early exit if we already beat the target
        if best_val <= 1.503:
            break

    # Fallback – if nothing succeeded return the initial seed
    if best_a is None:
        best_a = a0.copy()
        best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core)
# ----------------------------------------------------------------------
def _hillclimb_anneal(

    dominate the current maximal convolution entry to indices that
    contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (where mass is taken from)
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index (where mass is moved to)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # update the full convolution efficiently
        b_new = b.copy()
        b_new[j_tgt : j_tgt + n] += 2.0 * delta * a
        b_new[i_src : i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # random perturbation to escape a plateau
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – random perturb")
        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem size – a few thousand entries are a good trade‑off
    n = 5000

    # ------------------------------------------------------------------
    # 0️⃣ Gather diverse initial seeds
    # ------------------------------------------------------------------
    seeds = []

    # a) previous solution, if present (interpolated to the current length)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        seeds.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # b) best Beta‑family seed (denser sweep than the reference code)
    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_a is not None:
            seeds.append(beta_a)
            print(f"[init] Beta‑family seed (value={beta_val:.6f})")
    except Exception as exc:
        print("[init] Beta‑family sweep failed:", exc)

    # c) a few random Dirichlet draws (different concentration factors)
    for _ in range(4):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 1️⃣ Smooth‑max L‑BFGS optimisation (dominant part of budget)
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_val = np.inf

    time_for_smooth = (deadline - time.time()) * 0.65   # 65 % of remaining time
    if time_for_smooth < 1.0:
        time_for_smooth = 0.0
    per_seed = time_for_smooth / max(1, len(seeds))

    for i, a_seed in enumerate(seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed
        try:
            a_opt, val_opt = _lbfgs_optimize(a_seed, n, local_deadline, start_time)
            print(f"[seed {i}] L‑BFGS best = {val_opt:.6f}")
            if val_opt < global_best_val:
                global_best_val, global_best_a = val_opt, a_opt.copy()
            if global_best_val <= 1.503:
                break
        except Exception as exc:
            print(f"[seed {i}] L‑BFGS error:", exc)

    # ------------------------------------------------------------------
    # 2️⃣ Final intensive hill‑climbing on the remaining time slice
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and global_best_a is not None:
        print(f"[final hillclimb] remaining time ≈ {remaining:.1f}s")
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final secure evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
