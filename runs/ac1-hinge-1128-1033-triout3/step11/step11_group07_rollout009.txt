def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    now = time.time()
    deadline = now + budget_s - 0.5
    n = 12000 # maybe
    # initialization
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)

    # Generate random beta candidates
    for _ in range(300):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # main simulated annealing loop
    a = best_a.copy()
    cur_val = best_val
    temperature = 0.5
    while time.time() < deadline:
        # compute temperature schedule based on remaining time
        progress = (time.time() - now) / (deadline - now)
        temperature = max(1e-4, 0.5 * (1 - progress))

        # propose move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j: continue
        # move fraction of a[i]
        if a[i] == 0: continue
        frac = rng.random() * 0.2  # up to 20%
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non-negative due to rounding
        a_new = _project_simplex(a_new)  # ensure sum=1

        new_val = _objective(a_new)
        d = new_val - cur_val
        if d < 0 or np.random.rand() < np.exp(-d / temperature):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        # maybe occasional large jitter:
        if np.random.rand() < 0.001:
            a = a + rng.normal(scale=0.01, size=n)
            a = _project_simplex(a)

    # final polishing with gradient descent
    a = best_a.copy()
    cur_val = best_val
    # run few subgradient steps (maybe 1000)
    for it in range(2000):
        if time.time() >= deadline: break
        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        step = 0.2 * (1 - it/2000) + 1e-6
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)
        new_val = _objective(a_candidate)
        if new_val < cur_val:
            a = a_candidate
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Polish] new best = {best_val:.6f}")

    # possibly blend with uniform at end
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[Final] reported score = {final_score:.6f}")
    return best_a.tolist()

"""
Propose a low‑max‑convolution step function.

Algorithm
---------

1️⃣  **Diverse seeding** – draw many Beta‑shaped and Dirichlet
    vectors (length = *n*).  Keep the best one w.r.t. the exact
    objective `2·n·max(conv(a,a))`.

2️⃣  **Simulated‑annealing refinement** – starting from the best seed,
    repeatedly move a random amount of mass from one coordinate to
    another (preserving the ℓ₁‑norm).  The move is accepted if the
    objective improves, otherwise with probability 𝑒^{‑Δ / T},
    where the temperature *T* is linearly cooled.  The acceptance
    test uses a full FFT‑based convolution; for n≈8000 this is fast.

3️⃣  **Exact sub‑gradient polishing** – after the SA phase we run a
    projected sub‑gradient descent on the true non‑smooth objective
    (`max(conv)`).  The sub‑gradient is the sum of the two mirrored
    copies of the current vector at all maximising shifts.  A
    linearly decreasing step‑size together with simplex projection
    drives the solution further down.

4️⃣  **Uniform blending** – a short convex‑combination sweep with the
    uniform distribution often smoothes remaining spikes.

5️⃣  **Final secure evaluation** – the resulting sequence is handed to
    the provided ``evaluate_sequence`` for the official score.

The whole routine respects the 1000 s budget and never returns a
negative‑infinity value.  All numbers are non‑negative and the vector
is normalised to sum = 1 (the objective is scaling‑invariant, but the
normalisation keeps the optimisation stable).

"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value returned by ``evaluate_sequence`` for a vector with sum=1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex Σx = 1, x ≥ 0.
    Implements the O(n) algorithm from:
    Wang, Weiran, and Miguel A. Carreira‑Perpiñán. “Projection onto the
    probability simplex: An efficient algorithm with a simple proof,
    and an application.”
    """
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Sample a Beta‑like shape on a grid.
    Parameters (p,q) are drawn on a log‑scale to explore very flat
    and very peaked profiles.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)          # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:                           # safety fallback
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _blend_with_uniform(a: np.ndarray, n: int, cur_best: float) -> tuple[np.ndarray, float]:
    """
    Scan a 21‑point grid of convex combinations t·a + (1‑t)·u.
    ``u`` is the uniform distribution.  Keeps the best mixture.
    """
    u = np.full(n, 1.0 / n)
    best_a, best_val = a, cur_best
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
    return best_a, best_val

# Simulated‑Annealing refinement
# --------------------------------------------------------------

def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Randomly transfer mass between coordinates.
    Acceptance follows a simple exponential schedule.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    while time.time() < deadline:
        # linear temperature schedule (high → low)
        progress = (time.time() - start) / (deadline - start)
        T = max(1e-4, 0.4 * (1.0 - progress))

        # propose a move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or cur[i] == 0.0:
            continue

        # transfer a random fraction of the mass at i to j
        frac = rng.random() * 0.25               # up to 25 %
        delta = cur[i] * frac
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)             # keep Σ=1, non‑negative

        new_val = _objective(cand)
        d = new_val - cur_val
        if d < 0.0 or rng.random() < np.exp(-d / T):
            cur, cur_val = cand, new_val
            if new_val < best_val:
                best, best_val = cand.copy(), new_val
                print(f"[SA]   new best = {best_val:.6f}")

        # occasional global jitter to escape plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=0.02, size=n)
            cur = cur + jitter
            cur = _project_simplex(cur)
            cur_val = _objective(cur)

    return best, best_val

def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    cur_val = _objective(a)
    best_a, best_val = a.copy(), cur_val
    n = a.shape[0]

    it = 0
    while time.time() < deadline:
        it += 1
        b = _full_convolution(a)
        max_b = b.max()
        max_idxs = np.where(b == max_b)[0]

        # sub‑gradient: sum of contributions from all maximising shifts
        grad = np.zeros_like(a)
        for k in max_idxs:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple linear schedule: aggressive early, vanishing later
        elapsed = time.time() - start_time
        total = deadline - start_time
        step = 0.25 * (1.0 - elapsed / total) + 1e-5 * (elapsed / total)

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < cur_val - 1e-12:
            a, cur_val = a_new, new_val
            if new_val < best_val:
                best_a, best_val = a.copy(), new_val
                print(f"[Pol]  new best = {best_val:.6f}")

        # small random perturbation every few hundred iterations
        if it % 500 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.01, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val
