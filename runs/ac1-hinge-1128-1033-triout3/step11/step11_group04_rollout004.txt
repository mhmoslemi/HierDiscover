def propose_candidate(seed=42, budget_s=1000, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 0.2
    rng = np.random.default_rng(seed)
    best_a = None
    best_val = float('inf')
    
    while True:
        now = time.time()
        if now >= deadline - 50: # need at least 50s for a run
            break
        # allocate a fraction of remaining time
        remain = deadline - now
        run_time = max(60.0, remain * 0.5) # allocate half of remaining but at least 60s
        run_deadline = now + run_time
        # pick n: random from set
        n_choices = [8000, 10000, 12000]
        n = rng.choice(n_choices)
        a_run, val_run = run_optimization(n, rng, run_deadline)
        if val_run < best_val:
            best_val = val_run
            best_a = a_run
            print(f"[Overall] New global best value={best_val:.6f} with n={n}")
        # continue loop
    # final evaluation
    if best_a is None:
        best_a = np.full(8000, 1/8000)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] evaluate_sequence result={final_score:.6f}")
    except Exception as e:
        print("[Final] evaluate_sequence error:", e)
    return best_a.tolist()

"""
Propose a non‑negative step‑function (list of heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

within a given time budget (default 1000 s).  

The optimizer combines several ideas:

* **Diverse random seeding** – many Beta/Dirichlet draws, a few handcrafted
  shapes (linear decrease, exponential decay) and an optional interpolation
  of a previously discovered solution.
* **Adam on a smooth‐max** – we optimise a log‑sum‑exp surrogate of the
  max‑convolution, annealing the smoothing parameter ε from 0.5 to 1e‑12
  and the learning‑rate from 0.2 to 0.001.
* **Greedy hill‑climbing** – repeatedly move a chunk of mass from an index
  that contributes most to the current maximal convolution entry to a low‑
  contribution index, accepting moves by a Metropolis rule with a slowly
  decreasing temperature.
* **Projected sub‑gradient polishing** – a deterministic sub‑gradient descent
  on the exact objective with a linear step‑size schedule; occasional jitter
  prevents stagnation.
* **Final simulated‑annealing** – larger random source‑target moves for the
  remaining time.

The whole optimisation is wrapped in a multi‑restart loop: several independent
runs (different random seeds, sometimes different problem sizes) are performed
until the global time budget is exhausted; the best sequence found overall is
returned.  All intermediate objectives are evaluated with the fast FFT‑based
`_objective` function, and a final call to the provided `evaluate_sequence`
verifies the result.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1, otherwise the ratio scales automatically)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and
    the gradient of 2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random candidate factories
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # approx 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _linear_decreasing_candidate(n: int) -> np.ndarray:
    """a_i ∝ (n‑i) (i.e. linear decay from the left)."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


def _exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ r**i with random r ∈ (0.7, 0.99)."""
    r = rng.uniform(0.7, 0.99)
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for the loss L(a)=2·n·smooth_max(a).
    The smoothing ε is annealed from a relatively large value to 1e‑12.
    """
    # work in log‑softmax space – this guarantees feasibility automatically
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 4000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # Decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule learning rate and smoothing
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max gradient (already contains 2·n factor)
        _, grad = _smooth_max_and_grad(a, eps)

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if rng.random() < 0.015:
            w += rng.normal(scale=0.02, size=n)

        # restart if idle for too long
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = rng.normal(size=n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (O(n) incremental updates)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly moves a chunk of mass from a “high‑contribution”
    index to a “low‑contribution” index, with a simulated‑annealing acceptance
    rule.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules (more gradual than before)
    max_frac_start, max_frac_end = 0.85, 0.001
    temp_start, temp_end = 0.03, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 80000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index: proportional to its contribution
        if total > 0:
            src_probs = contrib / total
            i = rng.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index: favour low contributions
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = rng.choice(n, p=tgt_weights / wsum)
        else:
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation (it={iteration}), re‑perturb")
            a = best_a.copy()
            noise = rng.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val

# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The step size decays linearly from 0.30 to 1e‑5.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to break flat regions
        if rng.random() < 0.004:
            a = np.maximum(a + rng.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

# Additional simulated‑annealing phase (large moves)
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing loop that performs random source→target
    transfers with a relatively large step size.  It runs only on the
    leftover part of the time budget.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog)            # linear cooling
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        # choose j != i uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # tiny jitter to keep diversity
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a wall‑clock budget.
    Returns a list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # Global best across all restarts
    global_best_a = None
    global_best_val = float("inf")

    # Keep launching independent runs until the budget is exhausted.
    # Each run receives a fraction of the remaining time.
    while True:
        now = time.time()
        remaining = deadline - now
        if remaining < 80.0:          # not enough time for another full run
            break

        # allocate roughly half of the remaining time to this run
        run_time = max(60.0, remaining * 0.5)
        run_deadline = now + run_time

        # choose a problem size (different n give different search landscapes)
        n = rng.choice([8000, 9600, 12000])

        # execute one optimisation run
        a_run, val_run = _run_single_optimisation(n, rng, run_deadline, now)

        if val_run < global_best_val:
            global_best_val = val_run
            global_best_a = a_run
            print(f"[Overall] New global best = {global_best_val:.6f} (n={n})")

    # fallback – in the unlikely event that nothing was found
    if global_best_a is None:
        global_best_a = np.full(8000, 1.0 / 8000)

    # final verification with the provided evaluator
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure the return type is a plain Python list of floats
    return global_best_a.tolist()
