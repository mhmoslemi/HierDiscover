def lp_step(a):
    n = len(a)
    M = np.max(np.convolve(a,a))
    g = cp.Variable(n, nonneg=True)

    # we can compute for each k the index mapping.

    best_a = a.copy()
    best_val = _objective(a)

    # Precompute size for FFT for convolution of a with itself
    size_conv = 1 << ((2*n - 2).bit_length())
    size_weight = 1 << ((2*n - 2).bit_length())  # w length same

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha0 = 0.2
    alpha_end = 0.005
    lambda_start = 0.5
    lambda_end = 200.0

    t = 0
    while time.time() < deadline:
        now = time.time()
        prog = (now - start_time) / (deadline - start_time)
        lam = lambda_start + (lambda_end - lambda_start) * prog
        alpha = alpha0 + (alpha_end - alpha0) * prog

        # compute conv
        fa = np.fft.rfft(a, size_conv)
        conv_full = np.fft.irfft(fa * fa, size_conv)[:2*n - 1]
        max_c = conv_full.max()

        # smooth weights

        exps = np.exp(lam * (conv_full - max_c))
        w = exps / exps.sum()

"""
Improved search for a step‑function that minimises ``evaluate_sequence``.

The algorithm consists of three stages:

1️⃣ **Diverse start generation** – a few well‑known families (Beta,
   exponential, random Dirichlet) together with an interpolation of a
   possible previous solution are gathered as seeds.

2️⃣ **Smooth‑max optimisation** – we minimise the exact objective
   ``2·n·max(conv(a,a))`` by optimising a smooth approximation
   ``f_λ(a) = (1/λ)·log Σ exp(λ·(a∗a)_k)``.
   The gradient of ``f_λ`` is obtained via two FFT‑based convolutions.
   Adam with a linearly growing temperature `λ` (0.5 → 200) and a
   linearly decaying learning rate (0.12 → 0.005) is used.
   After each step the vector is projected onto the probability simplex
   (Σa = 1, a_i ≥ 0).  The best sequence seen so far is kept.

3️⃣ **Fine‑grained hill‑climbing** – a specialised simulated‑annealing
   that moves mass from indices that contribute most to the current
   maximal convolution entry to indices with the smallest contribution.
   The temperature and the maximal transfer fraction decay to zero.
   Periodic random perturbations avoid stagnation.

All stages respect the 1000 s time budget; the best sequence found is
returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Beta‑family sweep (moderately dense)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} for the best value."""
    best_a = None
    best_val = np.inf

    p_vals = np.concatenate([
        np.linspace(0.1, 5.0, 25),
        np.geomspace(5.0, 30.0, 6)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple exponential families (a_i ∝ exp(-c·i))
# ----------------------------------------------------------------------
def _exp_family_best(n: int) -> tuple[np.ndarray, float]:
    """Try a few exponential decays and pick the best."""
    best_a = None
    best_val = np.inf
    rates = [0.0005, 0.001, 0.002, 0.004, 0.008]
    i = np.arange(n, dtype=np.float64)
    for c in rates:
        a = np.exp(-c * i)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + λ‑continuation)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Optimise 2·n·max(conv(a,a)) via a smoothed max objective.
    λ grows from 0.5 to 200, learning rate decays from 0.12 to 0.005.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= s                                      # Σa = 1

    best_a = a.copy()
    best_val = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # FFT sizes (pre‑computed)
    size1 = 1 << ((2 * n - 2).bit_length())       # for a*a
    size2 = 1 << ((3 * n - 3).bit_length())       # for a_rev * w

    t = 0
    while time.time() < deadline:
        now = time.time()
        prog = (now - start_time) / (deadline - start_time)
        lam = 0.5 + (200.0 - 0.5) * prog          # λ: 0.5 → 200
        alpha = 0.12 + (0.005 - 0.12) * prog       # lr: 0.12 → 0.005
        t += 1

        # ----- a*a -----
        fa = np.fft.rfft(a, size1)
        conv_full = np.fft.irfft(fa * fa, size1)[:2 * n - 1]
        max_c = conv_full.max()

        # ----- smooth weights -----
        exps = np.exp(lam * (conv_full - max_c))   # shift by max for stability
        w = exps / exps.sum()

        # ----- gradient -----
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, size2)
        fw = np.fft.rfft(w, size2)
        conv_aw = np.fft.irfft(fa_rev * fw, size2)[:3 * n - 2]
        grad = 2.0 * conv_aw[n - 1 : n - 1 + n]

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a_new /= s
        a = a_new

        # ----- keep the best -----
        cur_val = 2.0 * n * max_c
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SmoothOpt {t:7d}] new best = {best_val:.6f} λ={lam:.1f}")

        # ---- occasional random perturbation to escape plateaus ----
        if t % 1200 == 0:
            # if we haven't improved in the last 1200 steps, perturb
            if cur_val >= best_val * 0.9999:
                noise = np.random.normal(scale=0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[SmoothOpt {t:7d}] stagnation – random perturb")

    return best_a, best_val

# Intensive hill‑climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,

    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.06, 5e-9

    no_improve = 0
    IMPROVE_LIMIT = 50000          # trigger a restart earlier than before
    iteration = 0

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # ----- pick a (possibly) maximal convolution index -----
        max_b = b.max()
        # take any of the top‑1% entries to avoid being stuck on a single peak
        top_idxs = np.where(b >= max_b * 0.999)[0]
        if top_idxs.size > 1:
            k_max = int(np.random.choice(top_idxs))
        else:
            k_max = int(np.argmax(b))

        # ----- source index (where contribution to k_max is large) -----
        i_range = np.arange(max(0, k_max - (n - 1)), min(n - 1, k_max) + 1)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = int(np.random.choice(i_range, p=src_probs))
        else:
            i_src = int(i_range[np.argmax(contrib)])

        if a[i_src] <= eps:
            continue

        # ----- target index – favour positions that hardly take part in k_max -----
        # weight inversely proportional to their contribution to the current max shift
        tgt_weights = np.zeros_like(a)
        mask = (j_range >= 0) & (j_range < n)
        tgt_weights[i_range[mask]] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = int(np.random.choice(n, p=tgt_weights / wsum))
        else:
            # fall back to a completely random index
            j_tgt = int(np.random.randint(n - 1))
            if j_tgt >= i_src:
                j_tgt += 1

        # ----- propose a transfer -----
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # ----- update convolution efficiently (linear work) -----
        b_new = b.copy()
        # contribution of the moved mass to each entry of the convolution
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        # diagonal corrections
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ---- restart if we have been stuck for a while ----
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # Problem size – a few thousand entries give a good trade‑off.
    n = 6000

    # ------------------------------------------------------------------
    # 0️⃣ Gather diverse initial seeds
    # ------------------------------------------------------------------
    seeds = []

    # a) previous solution, if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        seeds.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # b) best from a dense Beta‑family sweep
    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_a is not None:
            seeds.append(beta_a)
            print(f"[init] beta‑family seed (value={beta_val:.6f})")
    except Exception as exc:
        print("[init] beta family failed:", exc)

    # c) exponential family seeds
    try:
        exp_a, exp_val = _exp_family_best(n)
        if exp_a is not None:
            seeds.append(exp_a)
            print(f"[init] exponential seed (value={exp_val:.6f})")
    except Exception as exc:
        print("[init] exp family failed:", exc)

    # d) a few random Dirichlet draws
    for _ in range(4):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 1️⃣ Smooth‑max optimisation (major portion of the budget)
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_val = np.inf

    time_for_smooth = (deadline - time.time()) * 0.90
    if time_for_smooth <= 0:
        time_for_smooth = 0.0
    per_seed = time_for_smooth / max(1, len(seeds))

    for i, a_seed in enumerate(seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed
        a_opt, val_opt = _smooth_optimize(a_seed, n, local_deadline, start_time)
        print(f"[seed {i}] smooth_opt best = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Final intensive hill‑climbing on the remaining time slice
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and global_best_a is not None:
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final secure evaluation (logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # The harness expects a plain list of floats.
    return global_best_a.tolist()
