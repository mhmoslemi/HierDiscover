"""
propose_candidate
-----------------

A fast, gradient‑based search for a non‑negative step‑function that
minimises ``evaluate_sequence``.

The algorithm works in three stages:

1️⃣  Initialise a candidate from a flexible Beta family (the arcsine
    shape p≈q≈0.5 is usually very strong).

2️⃣  Refine it with an *exponentiated‑gradient* optimiser that uses a
    **smoothed max** of the self‑convolution:
       Lλ(a)= (1/λ)·log Σ exp(λ·(a⋆a)),
    whose gradient is a weighted sum of all anti‑diagonals.
    The weights are a soft‑max with a temperature λ that is slowly
    increased, turning the smooth objective into the true max.
    Updates are performed in the log‑domain (mirror descent w.r.t.
    KL‑divergence) which automatically respects non‑negativity,
    followed by one‑norm normalisation.

3️⃣  Whenever the true objective
       f(a)=2·n·max(a⋆a) / (∑a)²
    improves, the current sequence is stored.  After the allotted time
    the best stored sequence is handed to ``evaluate_sequence``.

The routine respects the supplied time budget (default 1000 s) and
never performs any file or network IO.
"""

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a⋆a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()            # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of x and y via FFT (full length).
    """
    out_len = x.shape[0] + y.shape[0] - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """a_i ∝ i^{p‑1}(n‑i+1)^{q‑1},   i = 1,…,n .  Returns None if degenerate."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed max (soft‑max) of the convolution.
    For a given temperature λ, the weights w_k = exp(λ·conv_k)/∑exp(·).
    The gradient w.r.t. a is
        ∇ = 2·(a_rev ⋆ w)[n‑1 : n‑1+n] .
    """
    n = a.shape[0]
    max_c = conv.max()
    # soft‑max weights (stable)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / exp_term.sum()                     # shape (2n‑1,)

    # reversed a
    a_rev = a[::-1]

    # convolution a_rev * w (full length = 3n‑2)
    grad_full = _conv_fft(a_rev, w)
    # slice the part that corresponds to ∂/∂a_j
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
