"""
Hybrid optimiser for the self‑convolution max problem.

The routine keeps a global best sequence (initialised from the previous
solution if available) and repeatedly launches short Adam‑optimisations
with a smooth‑max objective.  The temperature λ grows from a modest
value to a very large one; when λ is large we switch to the exact
sub‑gradient (a single active index) so the optimiser can focus on the
true maximum.  After each run we perform a cheap projected‑sub‑gradient
refinement that often squeezes a little more out of the current best.
The whole procedure respects the 1000 s budget and finally returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2*len(x)-1, computed via FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k-i}
    (cross‑correlation of weight vector w with a).
    Returned vector has length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = w.shape[0] + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1                         # extract indices n‑1 … n‑1+n‑1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the rho where u_i - (cssv_i / i) > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """
    Compute 2 * n * max(self‑convolution) for a probability vector a.
    """
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()

    optimisations with an increasing smooth‑max temperature, switches
    to the exact max‑gradient once the temperature is large, and finally
    performs a cheap projected sub‑gradient refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 8000 works well with the available budget.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise from a previous construction if possible.
    # --------------------------------------------------------------
    try:
        old_seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old_seq.size > 1:
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_seq.size)
            a0 = np.interp(new_x, old_x, old_seq)
            a0 = np.maximum(a0, 0.0)
            a0_sum = a0.sum()
            if a0_sum < 1e-12:

            else:

    # Global best tracker
    # --------------------------------------------------------------
    best_a = a0.copy()
    best_val = _objective(best_a, n)

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters (tuned for this problem)
    # ------------------------------------------------------------------
    lambda_start = 5.0            # start temperature for smooth‑max
    lambda_end   = 8000.0         # final temperature – very sharp
    adam_lr      = 0.04           # base learning rate

    # --------------------------------------------------------------
    while time.time() < deadline:

        # ------------------------------------------------------------------
        eps = 1e-12
        z = np.log(best_a + eps)                     # log‑softmax parameters
        # a small perturbation helps to escape a possible basin
        z += np.random.normal(scale=0.03, size=n)

        m = np.zeros_like(z)

        t_iter = 0
        stagnation = 0

        while time.time() < deadline:

            if t_iter >= 25000:

            max_z = np.max(z)

            conv = _full_convolution(a)                # length 2n‑1
            conv_max = conv.max()
            cur_val = 2.0 * n * conv_max

                      f"(t={now - start_time:.1f}s)")
            else:
                stagnation += 1

            # ---- 2.4 choose weighting for the max --------------------
            # temperature grows linearly from start to end
            lam = lambda_start * (1.0 - prog) + lambda_end * prog

            # If the temperature is already large we use the exact max
            # (i.e. a one‑hot weight vector) – this gives the true sub‑gradient.
            if lam > 2000.0:
                kmax = int(np.argmax(conv))
                w = np.zeros_like(conv)
                w[kmax] = 1.0
            else:
                # smooth‑max via soft‑max over convolution entries
                exp_vals = np.exp(lam * (conv - conv_max))
                w = exp_vals / exp_vals.sum()

            # ---- 2.5 gradient of the (smooth) max w.r.t. a -------------
            # g = 2 * Σ_k w_k a_{k-i}  =  2 * cross‑correlation(w, a)
            g_raw = _cross_corr(w, a)
            g = 2.0 * g_raw

            # ---- 2.6 back‑prop through the soft‑max ------------------
            dot_ag = np.dot(a, g)          # Σ_i a_i g_i
            grad_z = a * (g - dot_ag)      # ∂f/∂z_i = a_i (g_i - Σ a_j g_j)

            # ---- 2.7 Adam update ---------------------------------------
            m = b1 * m + (1.0 - b1) * grad_z
            v = b2 * v + (1.0 - b2) * (grad_z ** 2)
            m_hat = m / (1.0 - b1 ** t_iter)
            v_hat = v / (1.0 - b2 ** t_iter)
            z -= adam_lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # ---- 2.8 occasional noise to avoid stagnation ---------------
            if stagnation > IMPROVE_LIMIT:
                z += np.random.normal(scale=0.05, size=n)
                stagnation = 0
                print(f"[reset] added Gaussian noise at iter {t_iter}")

        # ------------------------------------------------------------------
        # 3️⃣  After the Adam run perform a cheap projected sub‑gradient sweep
        # ------------------------------------------------------------------
        a = best_a.copy()
        obj = best_val
        step = 0.02                     # initial step size for sub‑gradient moves

        for _ in range(1200):          # limited number of refinement steps
            if time.time() >= deadline:
                break
            conv = _full_convolution(a)
            kmax = int(np.argmax(conv))

            # sub‑gradient of the true max (vector g_i = 2 * a_{kmax-i})
            g = np.zeros_like(a)
            if kmax < n:
                # overlap region is 0 … kmax
                g[:kmax + 1] = 2.0 * a[kmax::-1]
            else:
                # overlap region is kmax-(n-1) … n-1
                offset = kmax - (n - 1)
                L = n - offset
                g[offset:] = 2.0 * a[:L][::-1]

            # trial step and projection onto the simplex
            a_trial = _project_simplex(a - step * g)
            conv_trial = _full_convolution(a_trial)
            obj_trial = 2.0 * n * conv_trial.max()

            if obj_trial < obj - 1e-12:
                a = a_trial
                obj = obj_trial
                best_a = a.copy()
                best_val = obj
                step = min(step * 1.1, 0.5)   # slowly increase step after success
                print(f"[subgrad] improved to {obj:.9f}")
                # early exit if we already beat the target
                if obj < 1.503:
                    break
            else:
                step *= 0.7                   # shrink step if no improvement

        # ------------------------------------------------------------------
        # Stop early if we have already satisfied the target bound
        # ------------------------------------------------------------------
        if best_val < 1.503:
            break

    # --------------------------------------------------------------
    # 4️⃣  Final secure evaluation (the only place we call evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
