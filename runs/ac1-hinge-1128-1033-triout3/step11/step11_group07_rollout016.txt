"""
Improved step‑function optimiser (no closures, no lambdas)
---------------------------------------------------------

The optimiser works in four phases on a reduced problem size (n≈2000) and
finally interpolates the best candidate to the required length (8000).
The main ideas are

* a very broad random initialisation (Beta shapes, Dirichlet draws,
  classic Fourier windows);
* deterministic L‑BFGS optimisation of a smooth log‑sum‑exp
  approximation of the max‑convolution;
* a projected sub‑gradient method with an analytically chosen step size;
* a simple simulated‑annealing local search that swaps mass between two
  positions.

All phases respect the global time budget and keep the best vector seen.
The final vector is interpolated to length 8000, re‑normalised and returned
as a plain list of Python floats.
"""

import time
import numpy as np
import scipy.optimize
import scipy.special

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) for Σa=1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (already multiplied by 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random Beta‑shaped candidate (used for cheap initialisation)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ~1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Classic window functions (all normalised to sum = 1)
# ----------------------------------------------------------------------
def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return w / w.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = scipy.special.i0(beta * np.sqrt(1 - t * t)) / scipy.special.i0(beta)
    return w / w.sum()


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth max objective
# ----------------------------------------------------------------------
def _lbfgs_fun(w: np.ndarray, n: int, eps: float) -> tuple[float, np.ndarray]:
    """Return smooth objective and its gradient w.r.t. the log‑softmax variable w."""
    # soft‑max transform (keeps Σa = 1 automatically)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    smooth_val, grad_a = _smooth_max_and_grad(a, eps)   # grad_a already contains 2·n
    obj = 2.0 * n * smooth_val
    # chain rule through the soft‑max
    grad_w = a * (grad_a - np.dot(grad_a, a))
    return obj, grad_w


def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float) -> tuple[np.ndarray, float]:
    """
    Multi‑stage L‑BFGS optimiser.
    The smoothing parameter ε is decreased geometrically.
    """
    w = np.log(a0 + 1e-12)          # work in log‑softmax space

    # eps schedule: from 5e‑1 down to 1e‑12 (geometric)
    eps_values = np.geomspace(5e-1, 1e-12, num=12)

    for eps in eps_values:
        if time.time() >= deadline:
            break

        # L‑BFGS (HiGHS) – limit the internal max‑iterations to keep time in check
        res = scipy.optimize.minimize(
            _lbfgs_fun,
            w,
            args=(n, eps),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 200, "disp": False},
        )
        w = res.x
        # decode current a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[L‑BFGS]  new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        f(a)=2·n·max(conv(a,a))
    The step size is chosen analytically as α = f / ||g||².
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()

        # mask of maximal indices
        mask = (b == max_b).astype(np.float64)

        # gradient for the exact max: g = 2 * (a *⊛ mask)[reversed]
        conv = _conv_fft(a, mask[::-1])          # length 2n‑1
        segment = conv[n - 1:2 * n - 1]          # n entries
        grad = 2.0 * segment[::-1]              # derivative of max_b

        # adaptive step – solves min_{α≥0} f(a - α·g) ≈ f - α·||g||²
        grad_norm_sq = np.dot(grad, grad) + 1e-20
        step = cur_val / grad_norm_sq * 0.5   # 0.5 is a safety factor

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:          # accept only real improvements
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 500 == 0 and np.random.rand() < 0.04:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing local search (pairwise mass swaps)
# ----------------------------------------------------------------------
def _local_search(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_a: np.ndarray,
                  best_val: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climber: relocate a random fraction of the
    mass from one index to another, accepting with a temperature schedule.
    """
    rng = np.random.default_rng()
    cur_a = a0.copy()
    cur_val = _objective(cur_a)

    temperature_start = best_val * 0.02   # a small relative temperature

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear cooling
        progress = (now - start_time) / (deadline - start_time)
        temperature = temperature_start * (1.0 - progress)

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1                     # ensure i != j

        # transfer a random fraction of the source mass
        max_transfer = cur_a[i] * 0.25 * (1.0 - progress) + 1e-6
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        val = _objective(cand)

        if val < cur_val:
            cur_a, cur_val = cand, val
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[Local] new best = {best_val:.6f}")
        else:
            # Metropolis acceptance
            if temperature > 0 and rng.random() < np.exp(-(val - cur_val) / temperature):
                cur_a, cur_val = cand, val

    return best_a, best_val

# Blend with the uniform distribution (helps to smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float,
                        steps: int = 101) -> tuple[np.ndarray, float]:

        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

        best_a, best_val = _local_search(
            best_a, base_n,
            deadline=deadline,
            start_time=phase_b_deadline,
            best_a=best_a,
            best_val=best_val,
        )
        best_a, best_val = _blend_with_uniform(best_a, base_n, best_val, steps=101)

    # ------------------------------------------------------------------
    # Upsample to the requested length (8000) and final evaluation
    # ------------------------------------------------------------------
    final_n = 8000
    if best_a.shape[0] != final_n:
        new_x = np.linspace(0.0, 1.0, final_n)
        old_x = np.linspace(0.0, 1.0, best_a.shape[0])
        best_a = np.interp(new_x, old_x, best_a)
        best_a = np.maximum(best_a, 0.0)
        best_a /= best_a.sum()

    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
