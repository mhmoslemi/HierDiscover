assistantfinal**Solution Overview**

The function `propose_candidate` searches for a non‑negative step function (a probability vector) that minimises the
`evaluate_sequence` score  

\[
\text{score}= \frac{2\,n\;\max(a*a)}{(\sum a)^2}\; .
\]

Because the score is invariant to scaling we may restrict ourselves to vectors
with `∑a = 1`.  The problem is therefore:

*choose a probability vector `a` that makes the full convolution `a*a`
as flat as possible*.

The algorithm consists of three phases

1. **Coarse initialization** – a quick scan of a flexible 2‑parameter
   Beta family (plus a few Dirichlet draws) yields a decent starting point.
   If a previous solution (`height_sequence_1`) exists it is interpolated and used
   instead.

2. **Long‑running projected gradient descent** –
   The non‑smooth maximum in the objective is replaced by a log‑sum‑exp
   smoothing  

   \[
   \phi_\beta(a)=\frac1\beta\log\!\sum_i
   \exp\!\bigl(\beta\,(a*a)_i\bigr) .
   \]

   Its gradient is a *soft‑cross‑correlation* that can be obtained with three
   FFTs per iteration.  The temperature `β` is increased slowly (β→∞ recovers
   the true max).  After each gradient step the iterate is projected onto the
   probability simplex (`∑a = 1, a≥0`).  A tiny random jitter is occasionally
   added to escape plateaus.  The loop runs until the time budget expires or
   the target score `1.503` is reached.

3. **Final secure evaluation** – the best vector found is passed to the
   provided `evaluate_sequence` function and returned.

The whole search fits easily into the 1000 s budget (the heavy part – the
gradient step – costs only a few milliseconds for a vector of length 8000).

```python
"""
Propose a step function that minimises evaluate_sequence.

The algorithm is:
1. Initialise a probability vector from a short Beta‑family scan (or from the
   previous solution if available).
2. Perform a long projected gradient descent on a smoothed version of the
   objective, increasing the smoothing parameter β gradually.
   The gradient is obtained via FFT‑based full convolution and a soft‑cross‑
   correlation.
3. Return the best vector found (after a final call to evaluate_sequence).
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex Σx=1, x≥0.
    Returned array has the same dtype as v.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero → uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the parameters lead to the zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smoothed objective
        φ_β(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i)

    Returns (grad, max_conv) where `grad` has the same shape as a and
    `max_conv = max(a⋆a)`.
    """
    n = a.shape[0]

    # -------- full convolution a⋆a ----------
    out_len = 2 * n - 1
    size_conv = _next_pow2(out_len)
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[:out_len]   # (a⋆a)

    max_conv = conv.max()

    # -------- soft‑max weights ----------
    # shift by max to avoid overflow
    exp_arg = beta * (conv - max_conv)      # ≤ 0
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:          # numerical safety
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum                  # Σ_i w_i = 1

    # -------- gradient = 2·(a ⋆_cross w_norm) ----------
    rev_w = w_norm[::-1]                     # reverse for cross‑correlation
    out_len_corr = n + out_len - 1            # = 3n‑2
    size_corr = _next_pow2(out_len_corr)

    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1                      # index of lag 0 in the full conv
    idx = offset - np.arange(n)               # indices for j = 0…n‑1
    grad = 2.0 * corr_full[idx]

    return grad, max_conv


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressivity and speed
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previous solution when possible
    # ------------------------------------------------------------------
    a0 = None
    try:
        # height_sequence_1 is expected to be a global variable containing a list
        old = np.asarray(globals()["height_sequence_1"], dtype=float)  # type: ignore
        if old.ndim == 1 and old.size > 0:
            # linearly interpolate to the required length
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            total = a0.sum()
            if total > 0.0:
                a0 /= total
                print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse search in the Beta family (grid + random)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    # deterministic grid (a few hundred candidates – cheap)
    p_grid = np.concatenate(
        [
            np.geomspace(0.001, 0.1, 8, endpoint=False),
            np.geomspace(0.1, 5.0, 12, endpoint=False),
            np.geomspace(5.0, 20.0, 5)
        ]
    )
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is None:
                continue
            val = _objective(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[grid] new best = {best_val:.6f} (p={p:.4g}, q={q:.4g})")

    # random sampling in log‑space for extra coverage
    rng = np.random.default_rng()
    random_trials = 5000
    log_min, log_max = np.log(1e-3), np.log(20.0)
    for _ in range(random_trials):
        p = np.exp(rng.uniform(log_min, log_max))
        q = np.exp(rng.uniform(log_min, log_max))
        cand = _beta_shape(p, q, n)
        if cand is None:
            continue
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand] new best = {best_val:.6f} (p={p:.4g}, q={q:.4g})")
        if time.time() >= deadline - 30.0:   # keep ~30 s for the long descent
            break

    # a few Dirichlet draws for diversity
    for _ in range(6):
        cand = rng.dirichlet(np.ones(n))
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[dir] new best = {best_val:.6f}")

    # fallback – uniform distribution if nothing else succeeded
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print("[fallback] uniform start, value=", best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Long‑running smoothed projected gradient descent
    # ------------------------------------------------------------------
    a = best_a.copy()          # current iterate (always Σa = 1)
    best_seq = a.copy()
    best_score = best_val

    # hyper‑parameters
    step0 = 0.5                # initial step size (will decay as 1/√t)
    beta = 0.5                 # start of smoothing
    beta_max = 200.0           # final β – essentially the true max
    beta_mul = 1.01            # multiplicative increase per iteration
    jitter_prob = 0.005        # occasional small random perturbation
    jitter_scale = 1e-3

    target_score = 1.503       # stop early if we already beat the target
    iteration = 0

    # Pre‑compute FFT sizes – they do not change
    out_len = 2 * n - 1
    size_conv = _next_pow2(out_len)
    out_len_corr = n + out_len - 1          # = 3n‑2
    size_corr = _next_pow2(out_len_corr)

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- gradient of the smoothed objective and current exact max ----
        grad, cur_max = _gradient_smooth(a, beta)
        cur_score = 2.0 * n * cur_max   # exact objective for the current a

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[smooth] new best = {best_score:.6f} (iter {iteration}, β={beta:.2f})")
            if best_score <= target_score:
                print("[smooth] target reached – exiting early")
                break

        # ---- projected gradient step ----
        step = step0 / np.sqrt(iteration + 1.0)
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        # ---- tiny random jitter (helps escape flat plateaus) ----
        if np.random.rand() < jitter_prob:
            a_candidate += np.random.normal(scale=jitter_scale, size=n)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

        a = a_candidate

        # ---- increase smoothing parameter ----
        beta = min(beta_max, beta * beta_mul)

        iteration += 1

        # occasional progress report
        if iteration % 500 == 0:
            print(f"[info] iter={iteration}, β={beta:.2f}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
```
