"""
Optimiser for the “max‑convolution” objective.

The algorithm works in three stages and respects a 1000 s wall‑clock
budget:

1️⃣  **Diverse initial seeds** – a dense sweep of the asymmetric
    Beta‑family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} (p and q independent) plus a
    few Dirichlet draws and, if available, the previous solution.

2️⃣  **Smooth‑max Adam optimisation** – minimise the smooth surrogate
    f_λ(a)= (1/λ)·log∑_k e^{λ·(a∗a)_k} by projected Adam.
    λ is increased from 0.5 to 150 (continuation) and every update is
    projected onto the probability simplex.

3️⃣  **Exact‑max sub‑gradient descent** – after the smooth phase the
    true max‑convolution is tackled directly.  The sub‑gradient of the
    max is known analytically:
       ∇max(a∗a)_i = 2·a_{k*‑i}   with k* = argmax(a∗a).
    A diminishing‑step projected sub‑gradient descent is run, with a
    simple random‑perturbation restart when progress stalls.

4️⃣  **Intensive hill‑climbing (annealing)** – finally a local search
    similar to the reference implementation refines the best sequence.

All steps share the same time budget; the best sequence found is
returned as a plain list of floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 0️⃣  Seed generation
# ----------------------------------------------------------------------
def _beta_family_seeds(n: int, max_keep: int = 8):
    """
    Sweep the asymmetric Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}
    (p and q independent). Return the `max_keep` best normalised seeds.
    """
    i = np.arange(1, n + 1, dtype=np.float64)

    # log‑spaced parameters covering (0.1, 12)
    p_vals = np.concatenate([
        np.geomspace(0.1, 1.0, 7),
        np.geomspace(1.0, 4.0, 7)[1:],
        np.geomspace(4.0, 12.0, 5)[1:],
    ])
    q_vals = p_vals.copy()

    best = []   # list of (value, a)
    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if len(best) < max_keep:
                best.append((val, a.copy()))
                best.sort(key=lambda x: x[0])
            elif val < best[-1][0]:
                best[-1] = (val, a.copy())
                best.sort(key=lambda x: x[0])
    return [a for _, a in best]


# ----------------------------------------------------------------------
# 1️⃣ Smooth‑max Adam optimiser (continuation on λ)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     lambda_max: float = 150.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth max surrogate.
    λ is increased linearly from 0.5 to ``lambda_max``.
    """
    a = np.maximum(a0, 0.0)
    a /= a.sum()                     # enforce Σa = 1

    best_a = a.copy()
    best_val = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.07                     # base learning rate
    t = 0

    # FFT sizes (pre‑computed)
    size1 = 1 << ((2 * n - 2).bit_length())       # for a*a
    size2 = 1 << ((3 * n - 3).bit_length())       # for a_rev * w

    while time.time() < deadline:
        t += 1
        prog = (time.time() - start_time) / (deadline - start_time)
        lam = 0.5 + (lambda_max - 0.5) * prog   # schedule λ∈[0.5, λ_max]

        # ---- a*a ----------------------------------------------------
        fa = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fa * fa, size1)[:2 * n - 1]
        max_c = conv.max()

        # ---- smooth weights -----------------------------------------
        # w_k ∝ exp(λ·(conv_k – max_c))
        exps = np.exp(lam * (conv - max_c))
        w = exps / exps.sum()                     # length 2·n‑1, sum = 1

        # ---- gradient (see paper) ----------------------------------
        a_rev = a[::-1]
        fw = np.fft.rfft(w, size2)
        fa_rev = np.fft.rfft(a_rev, size2)
        conv_aw = np.fft.irfft(fa_rev * fw, size2)[:3 * n - 2]
        grad = 2.0 * conv_aw[n - 1 : n - 1 + n]

        # ---- Adam update -------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        cur_val = 2.0 * n * max_c
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth] it={t:6d}  val={best_val:.6f}  λ={lam:.1f}")

        # occasional random restart to escape plateaus
        if t % 700 == 0 and cur_val >= best_val * 0.9999:
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            print(f"[Smooth] restart at it={t}")

    return best_a, best_val

# 2️⃣ Exact‑max sub‑gradient descent (projected)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         init_step: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max (a∗a).
    Uses the analytic sub‑gradient: g_i = 2·a_{k*‑i} where k* is the
    index of the maximal convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    step0 = init_step
    t = 0
    while time.time() < deadline:
        t += 1
        # full convolution (needed for the max index)
        conv = _full_convolution(a)
        k_max = int(conv.argmax())

        # analytic sub‑gradient
        grad = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]

        # diminishing step size
        eta = step0 / math.sqrt(t)
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        cur_val = _objective(a)
        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SubGrad] it={t:6d}  val={best_val:.6f}")

        # restart when stuck
        if t % 1500 == 0 and cur_val >= best_val * 0.9999:
            a = best_a.copy()
            a += np.random.normal(scale=0.025, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            print(f"[SubGrad] reset at it={t}")

    return best_a, best_val

# 3️⃣ Intensive hill‑climbing (annealing) – unchanged core

def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,

                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local SA / hill‑climbing that moves a small amount of mass from
    indices contributing heavily to the current maximal convolution entry
    to indices that contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)                 # indices of contributors
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose source index (high contribution)
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # choose target index (low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # incremental update of convolution (exact)
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] it={iteration:7d}  new best={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.025, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] it={iteration:7d}  stagnation – perturb")

        iteration += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the final evaluation call.
    deadline = start_time + budget_s - 0.5
    n = 8000                     # problem size (good trade‑off)

    # ------------------------------------------------------------------
    # 0️⃣ Gather seeds
    # ------------------------------------------------------------------
    seeds = []

    # a) previous solution, if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        seeds.append(a0)
        print("[Init] interpolated previous solution")
    except Exception:
        pass

    # b) best from asymmetric Beta‑family sweep
    beta_seeds = _beta_family_seeds(n, max_keep=6)
    seeds.extend(beta_seeds)
    print(f"[Init] added {len(beta_seeds)} Beta‑family seeds")

    # c) random Dirichlet draws
    for _ in range(4):
        seeds.append(np.random.dirichlet(np.ones(n)))
    print(f"[Init] total seeds = {len(seeds)}")

    # ------------------------------------------------------------------
    # 1️⃣ Smooth‑max Adam optimisation (≈60 % of remaining time)
    # ------------------------------------------------------------------
    time_for_smooth = max(0.0, (deadline - time.time()) * 0.60)
    per_seed = time_for_smooth / max(1, len(seeds))

    global_best_a = None
    global_best_val = np.inf

    for i, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed
        a_opt, val_opt = _smooth_optimize(seed_vec, n, local_deadline, start_time,
                                          lambda_max=150.0)
        print(f"[Smooth seed {i}] best={val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
            print(f"[Smooth] global best improved to {global_best_val:.6f}")
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Exact‑max sub‑gradient descent (≈20 % of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and global_best_a is not None:
        sub_deadline = time.time() + remaining * 0.20
        a_sub, val_sub = _subgradient_descent(global_best_a, n,
                                              sub_deadline, start_time,
                                              init_step=0.12)
        if val_sub < global_best_val:
            global_best_val, global_best_a = val_sub, a_sub.copy()
            print(f"[SubGrad] global best now {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final hill‑climbing on the rest of the budget
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and global_best_a is not None:
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[Hillclimb] final best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final secure evaluation (only for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[Final] evaluate_sequence = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
