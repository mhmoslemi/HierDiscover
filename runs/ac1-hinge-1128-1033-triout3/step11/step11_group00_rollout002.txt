            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s
    time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8000 works well and matches the previous best.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a probability vector.
    # ------------------------------------------------------------------
    old_seq = globals().get("height_sequence_1")
    if old_seq is not None:
        try:
            old = np.asarray(old_seq, dtype=float)
            old_len = old.shape[0]
            if old_len != n:
                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old_len)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            print("[init] interpolated previous solution")
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # smooth, centrally‑peaked Beta(2,2) density
        i = np.arange(1, n + 1, dtype=np.float64)
        p = q = 2.0
        a0 = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        a0 = a0 / a0.sum()
        print("[init] generated Beta(2,2) start")

    # Work in the log‑domain to guarantee positivity
    eps = 1e-30
    z = np.log(a0 + eps)

    # ------------------------------------------------------------------
    # Adam hyper‑parameters (Stage 1)
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Learning‑rate schedule (from 0.2 down to 0.008)
    lr_start, lr_end = 0.20, 0.008

    # Soft‑max temperature growth
    max_alpha = 250.0                     # final temperature
    switch_alpha = 120.0                  # when to switch to exact max

    # Stagnation handling
    stagn_counter = 0
    stagn_limit = 2500
    jitter_scale_adam = 0.025
    jitter_scale_mirror = 0.005

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_a = None
    best_obj = np.inf
    best_iter = 0
    it = 0
    mirror_mode = False
    switch_prog = None   # progress fraction when we entered mirror mode

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start) / (deadline - start)      # 0 → 1
        alpha = 1.0 + (max_alpha - 1.0) * (prog ** 1.5)

        # ------------------------------------------------------------------
        # Transition from Adam to entropy‑mirror descent
        # ------------------------------------------------------------------
        if not mirror_mode and alpha >= switch_alpha:
            mirror_mode = True
            switch_prog = prog
            # materialise the probability vector from logits
            a = np.exp(z)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)
            print(f"[mode] entering mirror descent at iter={it}, prog={prog:.3f}")

        # ------------------------------------------------------------------
        # Current probability vector
        # ------------------------------------------------------------------
        if not mirror_mode:
            a = np.exp(z)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        # ------------------------------------------------------------------
        # Full self‑convolution (used for both stages)
        # ------------------------------------------------------------------
        conv = _full_convolution(a)
        obj = 2.0 * n * conv.max()                     # exact objective for this a

        # ------------------------------------------------------------------
        # Record best solution seen so far
        # ------------------------------------------------------------------
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            best_iter = it
            stagn_counter = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}  t={now-start:.1f}s")
        else:
            stagn_counter += 1

        # ------------------------------------------------------------------
        # Gradient / update step
        # ------------------------------------------------------------------
        if not mirror_mode:
            # ---- Stage 1: smoothed Adam on the soft‑max objective ----
            maxc = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc))
            sumexp = exp_scaled.sum()
            weights = exp_scaled / sumexp                     # Σ weights = 1

            # gradient w.r.t. a (see derivation in the previous code)
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]        # slice of length n

            # chain rule through softmax(z)
            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            # Adam update on logits
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = lr_start * (1.0 - prog) + lr_end * prog
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        else:
            # ---- Stage 2: exact entropy‑mirror descent on the max entry ----
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)

            g = np.zeros_like(a)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                g[idx] = 2.0 * a[k - idx]                     # sub‑gradient

            # linear decay of the step size within the mirror phase
            stage_prog = (prog - switch_prog) / (1.0 - switch_prog) if switch_prog else prog
            eta_start, eta_end = 0.5, 0.01
            eta = eta_start * (1.0 - stage_prog) + eta_end * stage_prog

            a = a * np.exp(-eta * g)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

            it += 1

        # ------------------------------------------------------------------
        # Escape from stagnation
        # ------------------------------------------------------------------
        if stagn_counter >= stagn_limit:
            if not mirror_mode:
                z += np.random.normal(scale=jitter_scale_adam, size=n)
                print(f"[reset] jitter added (Adam) at it={it}")
            else:
                perturb = np.random.normal(scale=jitter_scale_mirror, size=n)
                a = np.maximum(a + perturb, 0.0)
                s = a.sum()
                a = a / s if s > 0 else np.full_like(a, 1.0 / n)
                print(f"[reset] jitter added (Mirror) at it={it}")
            stagn_counter = 0

    # ------------------------------------------------------------------
    # Polishing phase – longer mirror descent on the best vector found
    # ------------------------------------------------------------------
    if best_a is not None:
        refined = _max_refine(best_a, steps=5000, eta_start=0.45)
        conv_ref = _full_convolution(refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (the wrapper secures the result)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else a0.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
