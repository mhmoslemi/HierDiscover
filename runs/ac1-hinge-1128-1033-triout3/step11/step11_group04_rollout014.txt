def _random_mixture_candidate(n, rng, K=3):
    # draw K weights from Dirichlet(1)
    weights = rng.dirichlet(np.ones(K))
    # for each component sample p,q parameters log-uniform
    a = np.zeros(n, dtype=np.float64)
    i = np.arange(1, n+1)
    for k in range(K):
        p = np.exp(rng.uniform(-8, 2))
        q = np.exp(rng.uniform(-8, 2))
        comp = (i ** (p-1.0)) * ((n - i + 1) ** (q-1.0))
        a += weights[k] * comp
    a /= a.sum()
    return a

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (∑a)²

using a multi‑stage heuristic:
  * a rich random initialisation (Beta, Dirichlet, mixtures, hand‑crafted)
  * Adam optimisation on a smooth log‑sum‑exp approximation of the max–conv
  * an aggressive hill‑climbing phase that moves mass from the indices that
    contribute most to the current maximal convolution entry to the indices
    that contribute least (with a simulated‑annealing acceptance rule)
  * a final projected sub‑gradient polishing step
All phases respect the 1000 s wall‑clock budget.  The returned list is a
plain Python list of non‑negative floats whose ℓ₁‑norm is 1; it can be passed
directly to the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()    # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """Smooth approximation of max(conv(a,a)) and gradient of 2·n·smooth_max."""
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project ``v`` onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the probability simplex using a smooth max."""
    w = np.log(a0 + 1e-12)          # logits

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 4000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedules
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and gradient
        _, grad = _smooth_max_and_grad(a, eps)      # already has 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)

        # restart if idle too long
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing with simulated‑annealing acceptance
# ----------------------------------------------------------------------
def _enhanced_hillclimb(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float) -> tuple[np.ndarray, float]:
    """Move mass from high‑contributing indices to low‑contributing ones."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    no_improve = 0
    IMPROVE_LIMIT = 60000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        # move size decays, temperature decays
        max_frac = 0.6 * (1.0 - progress) + 0.001 * progress
        T = 0.08 * (1.0 - progress) + 1e-7 * progress

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source – proportional to its contribution
        if total_contrib > 0:
            i = np.random.choice(n, p=contrib / total_contrib)
        else:
            i = int(np.argmax(a))

        # target – favour indices with (almost) zero contribution
        zero_mask = contrib == 0
        if np.any(zero_mask):
            cand = np.where(zero_mask)[0]
            cand = cand[cand != i]
            if cand.size > 0:
                j = np.random.choice(cand)
            else:
                # fallback: random different index
                j = (i + np.random.randint(1, n)) % n
        else:
            # inverse‑contribution weighting
            inv = 1.0 / (contrib + 1e-12)
            inv[i] = 0.0
            inv_sum = inv.sum()
            if inv_sum > 0:
                j = np.random.choice(n, p=inv / inv_sum)
            else:
                j = (i + np.random.randint(1, n)) % n

        # amount of mass to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation, re‑perturb (it={iteration})")
            a = best_a.copy()
            a += np.random.normal(scale=0.03, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < cur_val - 1e-13:
            a, cur_val = a_new, new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to escape flat regions
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple convex blending with the uniform vector
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray, n: int, cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for several t."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random candidates – Beta, Dirichlet and mixtures thereof
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_mixture_candidate(n: int, rng: np.random.Generator, K: int = 3) -> np.ndarray:
    """Mixture of K Beta‑shapes (weights from a Dirichlet)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    weights = rng.dirichlet(np.ones(K))
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        logp = rng.uniform(-8.0, 2.0)
        logq = rng.uniform(-8.0, 2.0)
        p = np.exp(logp)
        q = np.exp(logq)
        comp = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        a += weights[k] * comp
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a
