def hillclimb_anneal2(a, n, deadline, start_time):
    a = a.copy()
    b = full_convolution(a)
    cur_val = 2*n*b.max()
    best_a = a.copy()
    best_val = cur_val
    T0 = 0.1  # starting temperature
    Tf = 1e-6
    while time.time() < deadline:
        # compute progress fraction
        prog = (time.time() - start_time)/(deadline-start_time)
        T = T0 * (Tf/T0)**prog
        # choose source index weighted by a_i * (sum of overlapping contributions)
        # compute contributions for each shift equal to current max?

        src = np.random.choice(n, p=a/a.sum())
        # choose target not equal to src uniformly
        tgt = np.random.choice(n-1)
        if tgt >= src:
            tgt += 1
        # delta = random fraction of a[src]
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue
        # compute b_new quickly
        b_new = b.copy()
        b_new[tgt:tgt+n] += 2.0*delta*a

        b_new[2*tgt] += delta*delta

def generate_beta(p,q,n):
    i = np.arange(1,n+1)
    a = np.power(i, p-1) * np.power(n - i + 1, q-1)
    a_sum = a.sum()
    if a_sum == 0:
        return None
    a = a / a_sum
    return a

def _hillclimb_greedy(a0, n, deadline, start_time, best_a, best_val, target_val=1.503):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a = best_a.copy()
    best_val = best_val
    eps = 1e-12
    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 200000
    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = 0.5 * (1 - prog) + 0.001 * prog  # linear decay
        T = 0.05 * ((1e-8/0.05) ** prog)  # exponential decay to near zero
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        # source: largest contributor
        i_src = int(np.argmax(contrib))
        if a[i_src] <= eps:
            continue
        # target: smallest contributor (excluding src)
        # set very large value for src to exclude
        contrib_excl = contrib.copy()
        contrib_excl[i_src] = np.inf
        j_tgt = int(np.argmin(contrib_excl))
        # delta amount
        delta = a[i_src] * max_frac * np.random.rand()
        if delta <= eps:
            continue
        # update b efficiently (like earlier)
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta
        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[HillG {iteration}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1
        if no_improve > IMPROVE_LIMIT:
            # random perturb to escape
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillG {iteration}] stagnation – perturb")
        iteration += 1
    return best_a, best_val

"""
Propose a step‑function (non‑negative list) that minimises
evaluate_sequence() within a 1000 s budget.

Approach
---------

1️⃣  Generate diverse seeds:
    •  previous solution (if any) interpolated to the working size.
    •  random Beta‑family shapes  a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}
       with p,q drawn from a log‑uniform range 10⁻²…10².
    •  exponential decays  a_i ∝ exp(‑α·i).
    •  a handful of Dirichlet draws.
   All candidates are normalised to a sum of one.

2️⃣  Pick the few (≤3) best seeds (exact objective on the full length).

3️⃣  Refine the best seed with a **smooth‑max optimiser**.
    The nonsmooth max ∥a∗a∥∞ is replaced by
        f_λ(a)= (1/λ)·log Σ_k exp(λ·(a∗a)_k) .
    Gradients are computed via FFT; Adam updates are projected onto the
    probability simplex after each step.  λ is continued from 0.5 up to
    ≈200, while the learning rate decays.

4️⃣  Run a **greedy annealed hill–climber** on the remaining time.
    At each iteration the index that contributes most to the current
    maximal convolution entry is chosen as source, the index with the
    smallest contribution as target, and a fraction of mass is moved.
    Moves are always accepted when they improve the exact objective and
    occasionally accepted according to a temperature schedule to escape
    plateaus.

The whole pipeline respects the global time budget; the best sequence
found is returned as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution, FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x * x (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, Σx = 1} (Euclidean projection)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Beta‑family shape a_i ∝ i^{p‑1} (n‑i+1)^{q‑1}, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # work in log‑space to avoid overflow
    log_a = (p - 1.0) * np.log(i) + (q - 1.0) * np.log(n - i + 1.0)
    # shift for numeric stability
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def _exp_shape(alpha: float, n: int) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    s = a.sum()
    if s == 0.0:
        a[0] = 1.0
        s = 1.0
    return a / s


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + λ continuation)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Optimise 2·n·max(conv(a,a)) using a smoothed max.
    Adam updates are projected onto the simplex after each step.
    λ grows from 0.5 up to ≈200.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # FFT sizes (pre‑computed)
    size1 = 1 << ((2 * n - 2).bit_length())
    size2 = 1 << ((3 * n - 3).bit_length())

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1
        prog = (now - start_time) / (deadline - start_time)

        # λ continuation – push it aggressively towards a large value
        lam = 0.5 + (200.0 - 0.5) * (prog ** 1.5)

        # --------------------------------------------------------------
        # a*a  (full linear convolution)
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fa * fa, size1)[:2 * n - 1]
        max_c = conv.max()

        # --------------------------------------------------------------
        # Smooth weights: w_k ∝ exp(λ·(conv_k – max_c))
        # --------------------------------------------------------------
        exps = np.exp(lam * (conv - max_c))
        w = exps / exps.sum()                     # length 2·n‑1, sums to 1

        # --------------------------------------------------------------
        # Gradient w.r.t a (derived from smooth‑max)
        # --------------------------------------------------------------
        a_rev = a[::-1]
        fw = np.fft.rfft(w, size2)
        fa_rev = np.fft.rfft(a_rev, size2)
        conv_aw = np.fft.irfft(fa_rev * fw, size2)[:3 * n - 2]
        grad = 2.0 * conv_aw[n - 1: n - 1 + n]

        # --------------------------------------------------------------
        # Adam update
        # --------------------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning‑rate schedule (decays exponentially)
        base_lr = 0.07
        lr = base_lr * ((0.001 / base_lr) ** prog)

        step = lr * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:               # fall back to a random simplex point
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s

        a = a_new

        cur_val = 2.0 * n * max_c
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SmoothOpt {t:7d}] new best = {best_val:.6f} λ={lam:.2f}")
            if best_val <= 1.503:
                break

        # occasional random shake‑up to escape plateaus
        if t % 1200 == 0:
            if cur_val >= best_val * 0.9999:
                noise = np.random.normal(scale=0.015, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[SmoothOpt {t:7d}] stagnation – random perturb")

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy / annealed hill‑climber (improved version)
# ----------------------------------------------------------------------
def _hillclimb_greedy(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Transfer a fraction of mass from the index that contributes most to
    the current maximal convolution entry to the index that contributes
    the least.  Accept improvements always; accept worsening moves with
    a temperature schedule.  Small random perturbations restart the walk
    after long stagnation.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12
    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)

        # linear decay of maximal allowed transfer fraction
        max_frac = 0.5 * (1.0 - prog) + 0.001 * prog
        # exponential temperature decay
        T = 0.04 * ((1e-9 / 0.04) ** prog)

        k_max = int(np.argmax(b))

        # contributions of each index to the current maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]

        # source = index with biggest contribution
        i_src = int(np.argmax(contrib))
        if a[i_src] <= eps:
            # nothing to move from this index – skip iteration
            continue

        # target = index with *smallest* contribution (excluding source)
        contrib_excl = contrib.copy()
        contrib_excl[i_src] = np.inf
        j_tgt = int(np.argmin(contrib_excl))

        # amount to move
        delta = a[i_src] * max_frac * np.random.rand()
        if delta <= eps:
            continue

        # fast update of the convolution array (identical to the reference)
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:8d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation recovery
        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:8d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.2

    # Working length (trade‑off between resolution and speed)
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Gather a pool of diverse initial seeds
    # ------------------------------------------------------------------
    candidates = []

    # a) previous solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        candidates.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # b) random Beta‑family shapes (log‑uniform p,q)
    n_search = 2000                 # cheaper evaluation during the wide sweep
    for _ in range(30):
        p = 10 ** np.random.uniform(-2, 2)   # 0.01 … 100
        q = 10 ** np.random.uniform(-2, 2)
        a_small = _beta_shape(p, q, n_search)
        if a_small is None:
            continue
        # upscale to the working size by linear interpolation
        a_big = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, n_search),
                          a_small)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()
        candidates.append(a_big)

    # c) exponential decays
    for _ in range(10):
        alpha = np.random.uniform(0.0, 5.0)
        a_small = _exp_shape(alpha, n_search)
        a_big = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, n_search),
                          a_small)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()
        candidates.append(a_big)

    # d) a few random Dirichlet draws (full size)
    for _ in range(10):
        a = np.random.dirichlet(np.ones(n))
        candidates.append(a)

    # ------------------------------------------------------------------
    # 1️⃣  Pick the best few seeds (exact evaluation on the full size)
    # ------------------------------------------------------------------
    scored = []
    for a in candidates:
        val = _objective(a)
        scored.append((val, a))
    scored.sort(key=lambda x: x[0])
    top_k = 3
    best_a, best_val = scored[0][1].copy(), scored[0][0]
    print(f"[init] best seed value = {best_val:.6f}")

    # keep the top_k distinct candidates for possible secondary runs
    top_candidates = [c for _, c in scored[:top_k]]

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max optimisation (majority of the remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 5.0:
        time_smooth = remaining * 0.70
        smooth_deadline = now + time_smooth
        # optimise the *best* seed for the bulk of the time
        a_opt, val_opt = _smooth_optimize(best_a, n, smooth_deadline, start_time)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
        print(f"[smooth] best after smoothing = {best_val:.6f}")

        # give the other top candidates a brief chance (≈5 % each)
        for i, a_seed in enumerate(top_candidates[1:], start=1):
            if time.time() >= deadline:
                break
            per_seed_dead = time.time() + remaining * 0.05 / (top_k - 1)
            a_opt_i, val_opt_i = _smooth_optimize(a_seed, n, per_seed_dead, start_time)
            print(f"[smooth #{i}] value = {val_opt_i:.6f}")
            if val_opt_i < best_val:
                best_val, best_a = val_opt_i, a_opt_i.copy()

    # ------------------------------------------------------------------
    # 3️⃣  Final intensive greedy hill‑climbing on the rest of the budget
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_a, best_val = _hillclimb_greedy(best_a, n,
                                             deadline, start_time,
                                             best_a, best_val,
                                             target_val=1.503)

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:       # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
