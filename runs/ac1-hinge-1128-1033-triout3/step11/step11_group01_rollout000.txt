def project_to_simplex(v):
    # ensures v_i >=0 and sum v = 1
    # follow algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid optimiser for the step‑function attack.

The algorithm works in two stages for a given length `n`:

1. **Smooth‑max Adam descent** – exactly the method used previously,
   but with a tighter restart schedule.  It quickly reduces the
   smoothed max of the self‑convolution.

2. **Exact‑max sub‑gradient descent** – after the smooth phase the
   true maximum entry of the convolution is taken, the sub‑gradient
   (which is simply a shifted copy of the current vector) is applied,
   and the iterate is projected back onto the probability simplex
   (non‑negative, sum = 1).  This step removes the remaining bias of the
   soft approximation and can improve the objective by a few ‰.

The optimiser is run for several candidate lengths (default
`[8000, 12000, 16000]`).  For each length the current best solution is
interpolated from the global variable `height_sequence_1` if it exists,
otherwise a uniform start is used.  All work respects the 1000 s
budget; the best sequence found across all lengths is finally evaluated
with `evaluate_sequence` and returned.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself (length 2·len(x)‑1)
    using an FFT of size the next power of two.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation needed for the smooth‑max gradient:
    g_i = Σ_k w_k * a_{k‑i}   (i = 0…len(a)‑1).
    Implemented via FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = len(w) + n - 1               # = (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.  Implements the algorithm of
    Duchi et al. (2008) in O(n log n) time.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are ≤0, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_subgradient(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the (unsmoothed) maximum entry of the self‑convolution
    with respect to `a`.  For a given convolution index `k_max`
    the gradient is 2·a shifted by that index.
    """
    n = a.shape[0]
    idx = k_max - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    return g


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that aims to
    minimise `evaluate_sequence`.  The routine respects the time budget
    and uses a hybrid optimisation strategy (smooth‑max Adam + exact‑max
    sub‑gradient) across several candidate lengths.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Candidate lengths – a few increasing sizes give the optimiser more
    # degrees of freedom without blowing up the runtime.
    # ------------------------------------------------------------------
    lengths = [8000, 12000, 16000]

    # Global best across all lengths
    global_best_val = float('inf')
    global_best_seq = None

    # Try to reuse a previously discovered solution if available
    old_seq = globals().get('height_sequence_1', None)

    for n in lengths:
        # --------------------------------------------------------------
        # 1️⃣ Initialise a from the old sequence (interpolation) or uniform
        # --------------------------------------------------------------
        if isinstance(old_seq, (list, np.ndarray)):
            old = np.asarray(old_seq, dtype=float)
            old_len = old.shape[0]
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a = np.interp(new_x, old_x, old)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s
            print(f"[init] interpolated old solution for n={n}")
        else:
            a = np.full(n, 1.0 / n)
            print(f"[init] uniform start for n={n}")

        # --------------------------------------------------------------
        # Helper variables shared between phases
        # --------------------------------------------------------------
        eps = 1e-12
        best_a_n = a.copy()
        best_val_n = float('inf')
        stagnation = 0
        IMPROVE_LIMIT = 3000          # restart after this many non‑improvements

        # --------------------------------------------------------------
        # 2️⃣ Phase 1 – smooth‑max Adam descent
        # --------------------------------------------------------------
        # Logits representation (soft‑max guarantees simplex)
        z = np.log(a + eps)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        b1, b2 = 0.9, 0.999
        adam_eps = 1e-8
        lr = 0.05
        max_iters_phase1 = 8000      # generous upper bound; actual may stop earlier

        for it in range(1, max_iters_phase1 + 1):
            now = time.time()
            if now >= deadline:
                break

            # ---- soft‑max to obtain a valid probability vector ----------
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            a = exp_z / exp_z.sum()

            # ---- convolution & objective --------------------------------
            conv = _full_convolution(a)
            conv_max = conv.max()
            cur_val = 2.0 * n * conv_max

            # ---- record best -------------------------------------------
            if cur_val < best_val_n - 1e-12:
                best_val_n = cur_val
                best_a_n = a.copy()
                stagnation = 0
                print(f"[n={n} phase1 iter {it:05d}] new best = {best_val_n:.9f}")
            else:
                stagnation += 1

            # ---- smooth‑max weighting -----------------------------------
            prog = it / max_iters_phase1
            lam = 5.0 * (1.0 - prog) + 1500.0 * prog   # linear schedule
            # stability shift by current max
            exp_vals = np.exp(lam * (conv - conv_max))
            w = exp_vals / exp_vals.sum()

            # ---- gradient (via cross‑correlation) -----------------------
            g_raw = _cross_corr(w, a)
            g = 2.0 * g_raw

            # ---- back‑prop through soft‑max ----------------------------
            dot_ag = np.dot(a, g)
            grad_z = a * (g - dot_ag)

            # ---- Adam update -------------------------------------------
            m = b1 * m + (1.0 - b1) * grad_z
            v = b2 * v + (1.0 - b2) * (grad_z ** 2)
            m_hat = m / (1.0 - b1 ** it)
            v_hat = v / (1.0 - b2 ** it)
            z -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # ---- stagnation handling -----------------------------------
            if stagnation > IMPROVE_LIMIT:
                # perturb logits to escape a flat region
                z += np.random.normal(scale=0.1, size=n)
                stagnation = 0
                print(f"[n={n} phase1] added Gaussian noise at iter {it}")

        # --------------------------------------------------------------
        # 3️⃣ Phase 2 – exact‑max sub‑gradient descent (projected)
        # --------------------------------------------------------------
        a = best_a_n.copy()                  # start from the best of phase 1
        eta = 0.1                             # initial step size
        stagnation = 0
        max_iters_phase2 = 20000

        for it in range(1, max_iters_phase2 + 1):
            now = time.time()
            if now >= deadline:
                break

            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))
            cur_val = 2.0 * n * conv[k_max]

            # record improvement
            if cur_val < best_val_n - 1e-12:
                best_val_n = cur_val
                best_a_n = a.copy()
                stagnation = 0
                print(f"[n={n} phase2 iter {it:05d}] new best = {best_val_n:.9f}")
            else:
                stagnation += 1

            # gradient of the exact max entry
            g = _max_subgradient(a, k_max)

            # simple projected sub‑gradient step
            a = a - eta * g
            a = _project_to_simplex(a)

            # dynamic step‑size adaptation
            if stagnation > 1500:
                eta *= 0.5
                stagnation = 0
                print(f"[n={n} phase2] step size reduced to {eta:.4g}")

        # --------------------------------------------------------------
        # 4️⃣ Update global best
        # --------------------------------------------------------------
        if best_val_n < global_best_val:
            global_best_val = best_val_n
            global_best_seq = best_a_n.copy()
            print(f"[global] new overall best = {global_best_val:.9f} (n={n})")

        # Early exit if time is almost exhausted
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Final secure evaluation (the only place we call evaluate_sequence)
    # --------------------------------------------------------------
    if global_best_seq is None:          # safety fallback – should never happen
        global_best_seq = np.full(8000, 1.0 / 8000)

    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_seq.tolist()
