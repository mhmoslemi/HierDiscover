"""
Enhanced search for a low‑max convolution step function.

Overview
--------
The evaluation to minimise is

    f(a) = 2·n·max(conv(a,a))   (a ≥ 0, Σ a = 1)

where ``conv`` is the full linear convolution.
For a uniform vector this value is 2, the current best known
solution (the code supplied with the task) reaches ≈1.508.
A perfectly flat convolution would give value ≈1, because the
average of the convolution is 1/(2n‑1).

The new optimiser works in two resolution levels:

* **Coarse phase (n≈2000)** – a fast Adam optimiser that minimises
  a *combined* loss:
      L = smooth_max(conv) + λ·‖conv – μ‖₂²
  where ``smooth_max`` is a log‑sum‑exp approximation of the
  maximum, μ = 1/(2n‑1) is the flat target and λ is increased
  during optimisation.  The loss therefore pulls the whole
  convolution towards a constant while still shrinking the actual
  maximum.

* **Fine phase (n≈8000)** – the best coarse solution is linearly
  interpolated to the target length and refined with a few rounds
  of Adam (smooth‑max only) plus an aggressive hill‑climbing
  routine that moves mass from “peak‑contributing’’ indices to
  “valley‑contributing’’ ones with a simulated‑annealing acceptance
  rule.

All steps respect the simplex constraint (non‑negative, sum = 1)
by operating in log‑softmax space (Adam) or by explicit
Euclidean projection (hill‑climbing).  Throughout the run the
current best sequence and its exact objective are tracked; the
final candidate is returned as a plain list of floats.

The routine respects the 1000 s wall‑clock budget and prints
intermediate progress.
"""

import time
import numpy as np
from scipy import fft

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()           # next power of two
    fx = fft.rfft(x, size)
    conv = fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = fft.rfft(x, size)
    fy = fft.rfft(y, size)
    conv = fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2·n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2·n‑1,)

    # gradient: conv(a, reversed(p)) on the central segment
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2·n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# L2‑flattening loss (‖conv – μ‖₂²) and its gradient
# ----------------------------------------------------------------------
def _l2_flat_grad(a: np.ndarray):
    """
    Gradient of  Σ_i (conv_i - μ)²   (μ = 1/(2·n‑1)).
    The returned gradient already contains the factor 4 (no extra n).
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2·n‑1
    mu = 1.0 / (2 * n - 1)
    d = b - mu                                   # deviation vector
    d_rev = d[::-1]
    conv_ad = _conv_fft(a, d_rev)                # length 2·n‑1
    segment = conv_ad[n - 1:2 * n - 1]            # n entries
    grad = 4.0 * segment[::-1]
    return grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the combined loss
# ----------------------------------------------------------------------
def _adam_optimize_combined(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            best_a: np.ndarray,
                            best_val: float,
                            lam_start: float = 0.0,
                            lam_end: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex for the loss
        L(a) = 2·n·smooth_max(a) + λ·‖conv(a,a) – μ‖₂²

    λ is annealed linearly from ``lam_start`` to ``lam_end``.
    """
    # work in log‑softmax space (logits)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.002          # a bit slower for stability
    eps_start, eps_end = 0.5, 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 6000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam‑C] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule learning rate, smoothing and λ
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress
        lam = lam_start + progress * (lam_end - lam_start)

        # smooth‑max and its gradient
        smooth_val, grad_max = _smooth_max_and_grad(a, eps)   # grad includes 2·n
        grad_l2 = _l2_flat_grad(a)                           # factor 4 already present
        grad = grad_max + lam * grad_l2

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.012:
            w += np.random.normal(scale=0.015, size=n)

        if idle > MAX_IDLE:
            print("[Adam‑C] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing with simulated‑annealing moves
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly moves a small chunk of mass from a
    “high‑contribution’’ index to a “low‑contribution’’ index.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules
    max_frac_start, max_frac_end = 0.6, 0.001
    temp_start, temp_end = 0.025, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 80000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)     # source
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target biased towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback uniform
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # fast O(n) convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # re‑inject a little noise around the current best
            print(f"[Hill] stagnation at it={iteration}, perturb")
            a = best_a.copy()
            a += np.random.normal(scale=0.025, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Utility: up‑sample a probability vector by linear interpolation
# ----------------------------------------------------------------------
def _upsample(vec: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    src_n = vec.shape[0]
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, target_n)
    res = np.interp(x_tgt, x_src, vec)
    res = np.maximum(res, 0.0)
    res /= res.sum()
    return res


# ----------------------------------------------------------------------
# Random beta family (used for initial seeding)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a
