    import time, numpy as np, heapq, random

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

        # interpolate to n if needed
        if a.shape[0] != n:
            new_x = np.linspace(0,1,n)
            old_x = np.linspace(0,1,a.shape[0])
            a = np.interp(new_x, old_x, a)
        a = np.maximum(a,0)
        a = a / a.sum()
    except Exception:
        # fallback to triangular shape
        a = _triangular_shape(n)

    # compute conv
    conv = np.convolve(a,a)
    # heap for max
    heap = [(-v, idx) for idx, v in enumerate(conv)]
    heapq.heapify(heap)

    # functions to get current max
    def get_max():
        while True:
            val, idx = heap[0]
            if -val == conv[idx]:
                return -val, idx
            heapq.heappop(heap)

    best_val = 2 * n * get_max()[0]   # *maybe update after each improvement
    best_seq = a.copy()
    best_obj = best_val

    # parameters
    step_factor = 0.1   # initial relative move size (fraction of source)
    temp = 0.01
    temp_decay = 0.99995  # per iteration
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1
        max_val, L = get_max()
        # generate source index candidate
        i_low = max(0, L - (n - 1))
        i_high = min(n-1, L)
        # pick i randomly but weighted by a_i perhaps
        # sample uniform in range then optionally reject if a_i small
        # attempt up to few tries to find a valid move
        moved = False
        for _ in range(3):
            i = np.random.randint(i_low, i_high+1)
            j = L - i
            if i == j or j < 0 or j >= n:
                continue
            if a[i] <= 0:
                continue
            # direction: move from larger to smaller
            if a[i] > a[j]:
                src, tgt = i, j
            elif a[j] > a[i]:
                src, tgt = j, i
            else:
                continue
            # amount
            src_val = a[src]
            delta = step_factor * src_val
            # ensure not too small
            if delta < 1e-12:
                continue
            # apply update
            # store old conv values for undo
            old_conv = {}
            # update conv at indices
            # compute idx1, idx2, idx3
            idx1 = 2*src
            idx2 = 2*tgt
            idx3 = src + tgt
            # In case idx1 == idx2 or idx3 equals one of them, handle summation
            # We'll compute changes individually
            # Save old conv values
            for idx in (idx1, idx2, idx3):
                old_conv[idx] = conv[idx]
            # apply formulas:
            a_src_before = a[src]
            a_tgt_before = a[tgt]
            # conv updates:
            conv[idx1] += -2*delta*a_src_before + delta*delta
            conv[idx2] += 2*delta*a_tgt_before + delta*delta
            conv[idx3] += 2*delta*(a_src_before - a_tgt_before) - 2*delta*delta

            # push new values onto heap
            heapq.heappush(heap, (-conv[idx1], idx1))
            heapq.heappush(heap, (-conv[idx2], idx2))
            heapq.heappush(heap, (-conv[idx3], idx3))

            # update a
            a[src] -= delta
            a[tgt] += delta

            # get new max
            new_max, _ = get_max()
            new_val = 2 * n * new_max

            # accept if better or maybe with probability
            if new_val < best_obj or (temp>0 and np.random.rand() < np.exp(-(new_val-best_obj)/temp)):
                # accept
                if new_val < best_obj:
                    best_obj = new_val
                    best_seq = a.copy()
                    print(f"[search] iter {iter_count} new best {best_obj:.6f}")
                moved = True
                break
            else:
                # revert
                a[src] += delta
                a[tgt] -= delta
                # restore conv entries
                for idx, old in old_conv.items():
                    conv[idx] = old
                    heapq.heappush(heap, (-old, idx))
                # continue to try other i
                moved = False
        # if no move performed, maybe random jitter
        if not moved:
            # attempt random jitter
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if i == j or a[i] == 0:
                continue
            src, tgt = i, j
            delta = step_factor * a[src]
            # same update procedure but no acceptance check? Keep small jitter maybe.
            # Use same logic with acceptance criteria.
            # Actually we can just random small perturbation without checking; skip for now.
        # update temperature
        temp *= temp_decay
        # optionally shrink step factor
        if iter_count % 2000 == 0:
            step_factor *= 0.9
        # early exit if no improvement for long time
        # We can break loop if time low

    final_score = evaluate_sequence(best_seq.tolist())
    print("[final] objective:", final_score)
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

The search consists of three phases:

1️⃣  Initialise from the previously discovered solution
    ``height_sequence_1`` (or fall back to a triangular window) and
    interpolate it to a moderately large size ``n = 12000``.  The
    sequence is normalised so that ``sum(a) = 1``.

2️⃣  Targeted stochastic local search.
    * At each iteration the index of the current maximum of the
      auto‑convolution ``conv = a * a`` is found (``L = argmax(conv)``).
    * A move consists of transferring a small amount ``δ`` from a larger
      coefficient to a smaller one whose indices satisfy ``i + j = L``
      (or, when ``L`` is even, from the index ``i = L//2`` to a random
      partner).  This directly reduces the offending convolution entry.
    * The convolution is updated **in‑place** by the exact formula
      derived from ``Δconv = 2·(a∗d) + d∗d`` using vectorised slice
      operations, i.e. only ``O(n)`` work per move.
    * The new objective ``2·n·max(conv)`` is computed; the move is kept
      if it improves the best value or (with a tiny probability) if it
      is accepted by a simple temperature schedule.  Otherwise the move
      is undone by applying the opposite transfer.
    * Step size (relative amount ``δ = η·a_src``) and temperature decay
      gradually, allowing fine‑grained optimisation.

3️⃣  After the time budget (``budget_s`` seconds) the best sequence seen
    is evaluated with the provided ``evaluate_sequence`` and returned.

All operations respect the non‑negativity and normalisation constraints,
and the algorithm stays well within the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper for simple triangular shape (fallback)
# ----------------------------------------------------------------------


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# In‑place convolution update for a move “δ from src to tgt”
# ----------------------------------------------------------------------


def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, tgt: int,
                delta: float) -> None:
    """
    Move a non‑negative amount ``delta`` from index ``src`` to ``tgt``.
    The vector ``a`` and its full linear convolution ``conv`` are updated
    in place using the exact analytic expression:
        Δconv = 2·(a∗d) + d∗d   with   d[src] = -δ, d[tgt] = +δ.
    ``a`` is assumed to contain the values *before* the move.
    """
    n = a.shape[0]

    # linear part  (slice updates)
    factor = 2.0 * delta
    conv[src:src + n] -= factor * a          # −2δ·a shifted by src
    conv[tgt:tgt + n] += factor * a          # +2δ·a shifted by tgt

    # quadratic part (three isolated entries)
    conv[2 * src] += delta * delta
    conv[2 * tgt] += delta * delta
    conv[src + tgt] -= 2.0 * delta * delta

    # finally update the coefficients themselves
    a[src] -= delta
    a[tgt] += delta


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence discovered before the deadline.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin for final call

    # ------------------------------------------------------------------
    # Problem size – a trade‑off between expressivity and speed.
    # ------------------------------------------------------------------
    n = 12000                                   # length of the candidate

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from the previously saved solution (if any)
    # ------------------------------------------------------------------
    try:

            x_old = np.linspace(0.0, 1.0, a.shape[0])
            a = np.interp(x_new, x_old, a)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    except Exception:
        a = _triangular_shape(n)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    conv = np.convolve(a, a)                    # length 2·n‑1

    best_val = 2.0 * n * conv.max()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    eta = 0.5           # initial relative step size
    eta_decay = 0.9995
    temp = 0.001        # temperature for occasional uphill moves
    temp_decay = 0.9999

    no_improve = 0

        iteration += 1

        cur_max = conv.max()

        # (b) try a move that directly touches this lag
        moved = False

            lo = max(0, L - (n - 1))

            j = L - i
            if i == j or j < 0 or j >= n:

            # choose direction that reduces the product a_i * a_j
            if a[i] > a[j]:

            elif a[j] > a[i]:

            else:
