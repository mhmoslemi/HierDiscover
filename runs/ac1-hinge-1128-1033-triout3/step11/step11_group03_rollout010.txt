def _subgradient_optimize(a0, n, deadline, start_time, eta0=0.5, eta_decay=0.5):

    a = a0.copy()
    # ensure simplex
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum == 0: a = np.full(n, 1.0/n)
    else: a /= a_sum

    best_a = a.copy()
    best_val = _objective(a)

    t = 0
    while True:
        cur_time = time.time()
        if cur_time >= deadline: break

        # compute convolution
        conv = _full_convolution(a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0*n*conv[max_idx]  # same as _objective

        # compute gradient
        w = np.zeros_like(conv, dtype=float)
        w[max_idx] = 1.0
        grad = 2.0 * _conv_rev(a, w)

        # step size schedule: η = eta0 / sqrt(t+1)
        t += 1
        eta = eta0 / np.sqrt(t)

        # do gradient step (descending)
        a_new = a - eta * grad

        # ensure nonnegative, projection onto simplex
        a_new = np.maximum(a_new, 0)
        a_new = _project_simplex(a_new)

        # evaluate new val
        conv_new = _full_convolution(a_new)
        new_val = 2.0*n*conv_new.max()

        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[Subgrad] iter {t} val={best_val:.6f}")

        # accept new point if better or maybe also accept if not too worse (for exploration)
        if new_val <= cur_val:
            a = a_new
        else:
            # maybe accept with probability? Not needed.
            # leave a unchanged.
            pass

        # occasional random perturbation / restart
        if t % 2000 == 0:
            # take best and perturb
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0)
            a = _project_simplex(a)

        # stop early if best_val <= target (e.g., 1.503)
        if best_val <= 1.503:
            break

    return best_a, best_val

"""
Propose a step‑function sequence that minimizes
      evaluate_sequence(sequence) = 2·n·max(a∗a) / (∑a)²

Algorithm
---------
1.  Build a diverse pool of initial candidates for several lengths n.
    – monotone exponential, power‑law, linear and quadratic decays,
      plus a few symmetric families (Gaussian/Laplace) used previously.
    – Dirichlet draws with several concentration parameters.
    – the previous solution (if present) interpolated to the current size.
2.  Keep the best few candidates (by the exact objective) and
    improve each of them with a **sub‑gradient descent** on the
    exact max‑convolution (the gradient is 2·a shifted by the lag
    where the maximum is attained).  A diminishing step‑size
    η = η0/√t and Euclidean projection onto the probability simplex
    maintain feasibility.
3.  Optionally run a short Adam optimisation on the smooth‑max
    surrogate (α up to 500) for a final fine tune.
4.  Polishing phase – a lightweight hill‑climbing that moves a
    small random amount of mass from a “guilty’’ index (large
    contribution to the current maximal lag) to a “safe’’ index.
5.  Return the best sequence discovered before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_rev(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Implements the gradient of a weighted max‑convolution:
    given weight vector w (length 2·n‑1) returns
        2·∑_i a_i·w_{i+j}  for j=0..n‑1.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    l_w = w.shape[0]
    out_len = n + l_w - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]
    start = n - 1
    return conv[start:start + n]        # length n


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence (assumes sum>0)."""
    conv = _full_convolution(a)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max surrogate (used only for a short final Adam phase)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    maxc = conv.max()
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exp_vals / sum_exp


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _smooth_adam_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          alpha_start: float = 5.0,
                          alpha_end: float = 500.0,
                          lr0: float = 0.05,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_conv = _full_convolution(a)
    cur_val = float(2.0 * n * cur_conv.max())
    best_a = a.copy()
    best_val = cur_val

    t = 0
    while True:
        if time.time() >= deadline:
            break
        prog = (time.time() - start_time) / (deadline - start_time)
        alpha = alpha_start + (alpha_end - alpha_start) * prog

        w = _softmax_weights(cur_conv, alpha)
        grad = 2.0 * _conv_rev(a, w)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a_cand = a - step * m_hat
        a_cand = np.maximum(a_cand, 0.0)
        a_cand = _project_simplex(a_cand)

        cand_conv = _full_convolution(a_cand)
        cand_val = float(2.0 * n * cand_conv.max())

        # keep the best seen so far
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_a = a_cand.copy()
            print(f"[Adam] iter {t:6d} α={alpha:6.1f} val={best_val:.6f}")
            if best_val <= 1.503:
                break

        # accept only if we do not increase the exact objective
        if cand_val <= cur_val:
            a = a_cand
            cur_conv = cand_conv
            cur_val = cand_val

        # occasional random jitter
        if t % 5000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_conv = _full_convolution(a)
            cur_val = float(2.0 * n * cur_conv.max())

    return best_a, best_val

    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    best_a = a.copy()

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        cur_val = float(2.0 * n * conv[k_max])

        # sub‑gradient of the max‑lag (one‑hot weight)
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        grad = 2.0 * _conv_rev(a, w)

        a_cand = np.maximum(a_cand, 0.0)
        a_cand = _project_simplex(a_cand)

        cand_conv = _full_convolution(a_cand)
        cand_val = float(2.0 * n * cand_conv.max())

# ----------------------------------------------------------------------
# Light‑weight hill‑climbing (mass‑transfer) for final polishing

def _final_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    b = _full_convolution(a)
    cur_val = float(2.0 * n * b.max())
    best_val = cur_val
    best_a = a.copy()

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        if time.time() >= deadline:
            break

        k_max = int(np.argmax(b))

        # contributions to the offending lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index – heavy contributor (probabilistic)
        if total > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-12:
            continue

        # target – index with small contribution
        inv = contrib.max() - contrib + 1e-15
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = float(2.0 * n * b_new.max())

        if new_val < cur_val:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = float(2.0 * n * b.max())
            no_improve = 0

        iteration += 1

    return best_a, best_val

def _interpolate_previous(prev_seq, n):
    """Linear interpolation of a previous solution to a new length."""
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None


def _parametric_candidates_extended(n):
    """A richer set of hand‑crafted, monotone families."""
    i = np.arange(n, dtype=np.float64)

    # Exponential / geometric decays
    for lam in (0.5 / n, 1.0 / n, 2.0 / n, 5.0 / n, 10.0 / n):
        a = np.exp(-lam * i)
        s = a.sum()
        if s > 0:
            a /= s
            yield a

    # Power‑law decreasing
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = (i + 1) ** (-p)
        s = a.sum()
        if s > 0:
            a /= s
            yield a

    # Linear decreasing (a_i ∝ n‑i)
    a = (n - i).astype(float)
    s = a.sum()
    if s > 0:
        a /= s
        yield a

    # Quadratic decreasing
    for power in (2, 3):
        a = (n - i) ** power
        s = a.sum()
        if s > 0:
            a /= s
            yield a

    # Shifted linear plateaus (mass concentrated in a prefix)
    for frac in (0.25, 0.33, 0.5):
        a = np.maximum(0.0, frac * n - i)
        s = a.sum()
        if s > 0:
            a /= s
            yield a

    # The symmetric families used previously (Gaussian/Laplace/Power‑law)
    # – they are retained for completeness
    mid = (n - 1) / 2.0
    # Gaussian families
    for sigma_factor in (0.18, 0.22, 0.26, 0.30):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        s = a.sum()
        if s > 0:
            a /= s
            yield a
    # Laplace families
    for scale_factor in (0.12, 0.16, 0.20):
        scale = scale_factor * n
        a = np.exp(-np.abs(i - mid) / scale)
        s = a.sum()
        if s > 0:
            a /= s
            yield a
    # Power‑law symmetric (Beta‑like)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = (i + 1) ** (p - 1.0) * (n - i) ** (p - 1.0)
        s = a.sum()
        if s > 0:
            a /= s
            yield a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # tiny safety margin

    # try to reuse the previously reported sequence if it exists
    try:
        previous = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    except Exception:
        previous = None

    global_best_a = None
    global_best_val = np.inf
    global_best_n = None

    # Explore a few problem sizes (larger sizes give the same theoretical bound,
    # but sometimes the optimizer discovers a better pattern).
    sizes = [8000, 12000, 16000]

    for n in sizes:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # 1️⃣  Gather a large pool of diverse initial candidates
        # ------------------------------------------------------------------
        candidates = []

        # a) previous solution (if any)
        if previous is not None:
            cand = _interpolate_previous(previous, n)
            if cand is not None:
                candidates.append(cand)

        # b) rich set of parametric families (including monotone shapes)
        for cand in _parametric_candidates_extended(n):
            candidates.append(cand)

        # c) Dirichlet draws with various concentration parameters
        for alpha in (0.1, 0.5, 1.0, 2.0, 5.0):
            for _ in range(3):
                cand = np.random.dirichlet(np.full(n, alpha))
                candidates.append(cand)

        # ------------------------------------------------------------------
        # 2️⃣  Evaluate them quickly and keep the best few
        # ------------------------------------------------------------------
        evaluated = [(cand, _objective(cand)) for cand in candidates]
        evaluated.sort(key=lambda x: x[1])
        top_candidates = [c for c, _ in evaluated[:5]]

        # ------------------------------------------------------------------
        # 3️⃣  Refine each top candidate with sub‑gradient descent
        # ------------------------------------------------------------------
        for init_a in top_candidates:
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            # allocate a modest slice of the remaining budget per run
            opt_deadline = time.time() + min(20.0, remaining * 0.3)

            a_opt, val_opt = _subgradient_optimize(init_a, n,
                                                   opt_deadline,
                                                   start_time,
                                                   eta0=0.2)
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_a = a_opt.copy()
                global_best_n = n
                print(f"[global] new best n={n} val={global_best_val:.6f}")

            if global_best_val <= 1.503:
                break

        # ------------------------------------------------------------------
        # 4️⃣  Optional short Adam fine‑tuning on the best of this size
        # ------------------------------------------------------------------
        if global_best_a is not None and global_best_n == n:
            remaining = deadline - time.time()
            if remaining > 5.0:
                opt_deadline = time.time() + min(10.0, remaining * 0.2)
                a_fine, val_fine = _smooth_adam_optimize(
                    global_best_a,
                    n,
                    opt_deadline,
                    start_time,
                    alpha_start=10.0,
                    alpha_end=500.0,
                    lr0=0.05,
                )
                if val_fine < global_best_val:
                    global_best_val = val_fine
                    global_best_a = a_fine.copy()
                    global_best_n = n
                    print(f"[fine‑adam] n={n} val={global_best_val:.6f}")

        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 5️⃣  Final polishing with the hill‑climb phase
    # ------------------------------------------------------------------
    if global_best_a is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            global_best_a, global_best_val = _final_hillclimb(
                global_best_a,
                global_best_n,
                deadline=deadline,
                start_time=start_time,
                target_val=1.503,
            )
            print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_a is None:
        n = sizes[0]
        global_best_a = np.full(n, 1.0 / n)
        global_best_val = _objective(global_best_a)

    # ------------------------------------------------------------------
    # 7️⃣  Final evaluation (the host will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
