def _flatten_optimize(a0, n, deadline, start_time):
    a = a0.copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    # step size schedule
    lr = 0.1
    decay = 0.9999
    while time.time() < deadline:
        # compute conv
        conv = _full_convolution(a)
        mean = 1.0 / (2*n - 1)
        diff = conv - mean
        # compute gradient
        a_rev = a[::-1]
        size = 1 << ((len(diff) + n - 1 - 1).bit_length())
        fft_diff = np.fft.rfft(diff, size)
        fft_a_rev = np.fft.rfft(a_rev, size)
        conv_grad = np.fft.irfft(fft_diff * fft_a_rev, size)[:len(diff)+n-1]
        grad = 4.0 * conv_grad[n-1 : n-1 + n]  # factor 4 per derivative, optional scaling
        
        # Update via gradient descent with projection onto simplex
        a_new = a - lr * grad
        # ensure nonnegative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s == 0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new
        lr *= decay
        cur_val = _objective(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Flatten {time.time():.2f}] new best = {best_val:.6f}")
        # optional early stop if best_val <= 1.45 etc
    return best_a, best_val
```

"""
propose_candidate
-----------------
Search for a non‑negative step‑function that minimises the evaluation
function  ``evaluate_sequence``.
The algorithm is a three‑phase pipeline:

1️⃣  **Diverse seeds** – a dense Beta‑family grid (including extreme
    exponents), a few geometric families, and Dirichlet draws provide
    many starting points.  If a previous solution exists (``height_sequence_1``)
    it is interpolated.

2️⃣  **Smooth‑max optimisation** – minimise a smooth approximation of the
    max‑convolution using Adam with a continuation on the smoothness
    parameter λ (0.5 → 200).  After each Adam step the iterate is projected
    onto the probability simplex.

3️⃣  **Flattening optimiser** – a second gradient step minimises the L₂
    deviation of the full convolution from its average value
    (i.e. makes the convolution as flat as possible).  The gradient is
    obtained analytically and the iterate is again simplex‑projected.

4️⃣  **Intensive hill‑climbing** – a simulated‑annealing style local
    search (the “hill‑climb” from the reference implementation) finishes
    the remaining time.

All phases respect a 1000 s budget; the best sequence found is returned
as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _beta_family_seeds(
    n: int, top_k: int = 5
) -> list[tuple[np.ndarray, float]]:
    """Return the `top_k` Beta‑family candidates (a_i ∝ i^{p‑1}(n‑i+1)^{q‑1})."""
    i = np.arange(1, n + 1, dtype=np.float64)

    # a fairly wide grid – linear + log spacing
    p_vals = np.concatenate(
        [
            np.linspace(0.05, 0.5, 12),
            np.linspace(0.6, 3.0, 12),
            np.geomspace(3.0, 20.0, 8)[1:],
        ]
    )
    q_vals = p_vals.copy()

    best = []  # list of (value, a)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if len(best) < top_k:
                best.append((val, a.copy()))
                best.sort(key=lambda x: x[0])
            elif val < best[-1][0]:
                best[-1] = (val, a.copy())
                best.sort(key=lambda x: x[0])
    return [(a, val) for val, a in best]


def _geometric_family(n: int, r: float) -> np.ndarray:
    """Geometric family a_i ∝ r^{i-1}, normalised to Σ=1."""
    i = np.arange(n, dtype=np.float64)
    a = (r ** i)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _dirichlet_seeds(n: int, alphas: list[float]) -> list[np.ndarray]:
    """Return Dirichlet draws with given concentration parameters."""
    seeds = []
    for a in alphas:
        seeds.append(np.random.dirichlet(np.full(n, a)))
    return seeds


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + λ continuation)
# ----------------------------------------------------------------------
def _smooth_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    lam_max: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Optimises 2·n·max(conv(a,a)) using a smoothed max objective.
    Adam updates are projected onto the simplex after each step.
    λ grows from 0.5 to `lam_max` as the optimisation proceeds.
    """
    a = np.maximum(a0, 0.0)
    a /= a.sum()                     # enforce Σ=1

    best_a = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.07                     # learning rate
    t = 0

    # FFT sizes (pre‑computed)
    size1 = 1 << ((2 * n - 2).bit_length())       # for a*a
    size2 = 1 << ((3 * n - 3).bit_length())       # for a_rev * w

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        # λ schedule
        prog = (now - start_time) / (deadline - start_time)
        lam = 0.5 + (lam_max - 0.5) * prog

        # a*a
        fa = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fa * fa, size1)[: 2 * n - 1]
        max_c = conv.max()

        # smooth weights (soft‑max)
        exps = np.exp(lam * (conv - max_c))
        w = exps / exps.sum()

        # gradient = 2 * (a_rev * w)  (slice the centre part)
        a_rev = a[::-1]
        fw = np.fft.rfft(w, size2)
        fa_rev = np.fft.rfft(a_rev, size2)
        conv_aw = np.fft.irfft(fa_rev * fw, size2)[: 3 * n - 2]
        grad = 2.0 * conv_aw[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        cur_val = 2.0 * n * max_c
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[Smooth] new best = {best_val:.6f} λ={lam:.2f}")

        # occasional random perturbation to escape plateaus
        if t % 900 == 0:
            if cur_val >= best_val * 0.9999:
                noise = np.random.normal(scale=0.015, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print("[Smooth] stagnation – perturb")

    return best_a, best_val


# ----------------------------------------------------------------------
# Flattening optimiser (L2 deviation from the mean)
# ----------------------------------------------------------------------
def _flatten_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on  Σ (conv_k – μ)²  where μ is the uniform mean.
    The goal is to make the convolution as flat as possible,
    which often reduces the max‑convolution as a side effect.
    """
    a = np.maximum(a0, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    # learning‑rate schedule
    lr = 0.12
    decay = 0.9998

    mean_target = 1.0 / (2 * n - 1)

    # pre‑compute FFT size for diff * a_rev
    conv_len = 2 * n - 1
    size = 1 << ((conv_len + n - 2).bit_length())   # (conv_len + n -1) − 1

    while time.time() < deadline:
        # full convolution
        conv = _full_convolution(a)                # length 2n‑1
        diff = conv - mean_target

        # gradient = 4 * (diff * a_rev) (centre slice)
        a_rev = a[::-1]
        fdiff = np.fft.rfft(diff, size)
        farev = np.fft.rfft(a_rev, size)
        g_full = np.fft.irfft(fdiff * farev, size)[: conv_len + n - 1]
        grad = 4.0 * g_full[n - 1 : n - 1 + n]

        # Adam‑style (simple) update with projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        lr *= decay

        cur_val = _objective(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[Flatten] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core)
# ----------------------------------------------------------------------
def _hillclimb_anneal(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_val: float,
    target_val: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # pick source index
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # pick target index (prefer low contributors)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # update convolution efficiently
        b_new = b.copy()
        b_new[j_tgt : j_tgt + n] += 2.0 * delta * a
        b_new[i_src : i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # Problem size – a few thousand entries give a good trade‑off.
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣ Gather diverse initial seeds
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # a) interpolated previous solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        seeds.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # b) best few Beta‑family seeds
    try:
        beta_top = _beta_family_seeds(n, top_k=5)
        for a, val in beta_top:
            seeds.append(a)
            print(f"[init] Beta seed value={val:.6f}")
    except Exception as exc:
        print("[init] beta family sweep failed:", exc)

    # c) geometric family (a_i ∝ r^i) – a handful of r values
    for r in [0.3, 0.45, 0.6, 0.75, 0.9]:
        g = _geometric_family(n, r)
        seeds.append(g)
        print(f"[init] geometric seed r={r}")

    # d) Dirichlet draws with varied concentration
    for a in _dirichlet_seeds(n, alphas=[0.1, 1.0, 5.0]):
        seeds.append(a)
        print("[init] Dirichlet seed added")

    # ------------------------------------------------------------------
    # 1️⃣ Smooth‑max optimisation  (≈60 % of the budget)
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_val = np.inf

    time_for_smooth = (deadline - time.time()) * 0.60
    if time_for_smooth < 0:
        time_for_smooth = 0.0
    per_seed = time_for_smooth / max(1, len(seeds))

    for idx, a_seed in enumerate(seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed
        a_opt, val_opt = _smooth_optimize(
            a_seed, n, local_deadline, start_time, lam_max=200.0
        )
        print(f"[seed {idx}] smooth_opt best = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Flattening optimisation  (≈20 % of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    time_for_flatten = remaining * 0.20
    if time_for_flatten > 2.0 and global_best_a is not None:
        flatten_deadline = time.time() + time_for_flatten
        a_flat, val_flat = _flatten_optimize(
            global_best_a, n, flatten_deadline, start_time
        )
        print(f"[flatten] best = {val_flat:.6f}")
        if val_flat < global_best_val:
            global_best_val, global_best_a = val_flat, a_flat.copy()

    # ------------------------------------------------------------------
    # 3️⃣ Final hill‑climbing on what remains of the budget
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best_a is not None:

            global_best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final secure evaluation (logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return plain Python list of floats
    return global_best_a.tolist()
