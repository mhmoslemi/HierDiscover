"""
Simulated‑annealing + mass‑transfer search for a step‑function with
minimal  max‑convolution value.

The algorithm works as follows:

1️⃣  Build a diverse pool of initial probability vectors (Gaussian,
    Laplace, power‑law, Dirichlet draws, sparse Dirichlet, and a
    possible previous solution).

2️⃣  For each candidate run a short simulated‑annealing (SA) loop.
    In every SA iteration we:
      • compute the convolution a∗a,
      • locate the lag k where the convolution is maximal,
      • compute the contribution a_i·a_{k‑i} of every entry to that lag,
      • move a random fraction of mass from a “heavy” contributor to a
        “light” one,
      • accept the move if it improves the exact objective, otherwise
        with probability exp(−Δ/ T) where the temperature T is linearly
        cooled.

    The SA loop is time‑boxed (≈30 s per start) and the best vector
    encountered is kept.

3️⃣  The globally best vector amongst all runs is polished by the
    hill‑climbing routine used in the reference code (mass‑transfer
    with deterministic improvement).

4️⃣  The final sequence is returned; the host will evaluate it using
    ``evaluate_sequence``.

The implementation respects the 1000 s budget, performs only
numpy‑based operations (FFT for convolutions) and never uses a
filesystem or network.  All numbers are non‑negative and the vector
is projected onto the probability simplex (sum = 1) after every
perturbation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(a∗a) for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parameterised families (hand‑crafted shapes)
# ----------------------------------------------------------------------
def _parametric_candidates(n: int):
    """Yield a few deterministic shapes (Gaussian, Laplace, power‑law)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Gaussian
    for sigma_factor in (0.18, 0.22, 0.26, 0.30):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Laplace
    for scale_factor in (0.12, 0.16, 0.20):
        scale = scale_factor * n
        a = np.exp(-np.abs(i - mid) / scale)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Power‑law / Beta‑like
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = (i + 1) ** (p - 1.0) * (n - i) ** (p - 1.0)
        if a.sum() > 0:
            a /= a.sum()
            yield a


# ----------------------------------------------------------------------
# Mass‑transfer move guided by the current worst lag
# ----------------------------------------------------------------------
def _propose_move(a: np.ndarray, k_max: int):
    """
    Propose a new vector by moving a random fraction of mass from a
    heavy contributor to a light one for the lag ``k_max``.
    Returns the new vector and the indices involved (src, tgt, delta).
    """
    n = a.shape[0]
    contrib = np.zeros(n)
    # range of i for which k_max‑i lies inside [0, n‑1]
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idx_i = np.arange(i_start, i_end + 1)
    idx_j = k_max - idx_i
    contrib[idx_i] = a[idx_i] * a[idx_j]

    total = contrib.sum()
    if total > 0 and np.random.rand() < 0.9:
        src = np.random.choice(n, p=contrib / total)
    else:
        src = int(np.argmax(contrib))

    if a[src] < 1e-12:
        # fallback – choose a random source with non‑zero mass
        nonzero = np.where(a > 1e-12)[0]
        if nonzero.size == 0:
            return a.copy(), src, src, 0.0
        src = np.random.choice(nonzero)

    # target – preferably a light contributor
    inv = contrib.max() - contrib + 1e-12
    inv[src] = 0.0
    if inv.sum() > 0:
        tgt = np.random.choice(n, p=inv / inv.sum())
    else:
        # pick a random different index
        candidates = np.delete(np.arange(n), src)
        tgt = np.random.choice(candidates)

    # move a random fraction (up to 50 %) of the source mass
    delta = a[src] * np.random.rand() * 0.5
    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    return a_new, src, tgt, delta


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (mass‑transfer)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(a0: np.ndarray,
                                  n: int,
                                  run_deadline: float,
                                  start_of_run: float):
    """
    Simple SA that repeatedly moves mass from heavy to light contributors.
    Returns the best vector seen and its exact objective.
    """
    a = a0.copy()
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    # Temperature schedule (linear cooling)
    T_start = 0.02 * cur_val          # roughly 2 % of the current value
    T_end = 1e-6

    total_time = run_deadline - start_of_run
    iteration = 0

    while True:
        now = time.time()
        if now >= run_deadline:
            break

        # linear progress in [0,1]
        prog = (now - start_of_run) / total_time
        T = T_start * (1.0 - prog) + T_end * prog

        # current convolution and worst lag
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # propose a directed move
        a_new, src, tgt, delta = _propose_move(a, k_max)
        new_val = _objective(a_new)

        delta_val = new_val - cur_val
        accept = False
        if delta_val <= 0:
            accept = True
        else:
            # Metropolis acceptance
            if T > 0:
                prob = math.exp(-delta_val / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA] iter {iteration:7d} val={best_val:.6f}")

        # occasional global jitter to avoid stagnation
        if iteration % 20000 == 0 and iteration > 0:
            # blend a bit of a fresh Dirichlet sample
            jitter = np.random.dirichlet(np.ones(n))
            a = 0.9 * a + 0.1 * jitter
            a /= a.sum()
            cur_val = _objective(a)

        iteration += 1

        # early exit if we already beat the target
        if best_val <= 1.503:
            break

    return best_a, best_val

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

        now = time.time()

        tgt = np.random.choice(n, p=inv / inv.sum())

        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = _objective(a_new)

        if new_val < cur_val:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # random jitter around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective(a)
            no_improve = 0

        iteration += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise ``evaluate_sequence`` under a 1000 s time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a tiny safety margin

    # Problem sizes to explore – the reference used 8k/12k/16k.
    sizes = [8000, 12000, 16000]

    # Try to reuse a previously found solution if the variable exists.
    try:
        previous = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    except Exception:
        previous = None

    global_best_a = None
    global_best_val = np.inf
    global_best_n = None

    # ------------------------------------------------------------------
    # 1️⃣  Run many short SA searches from diverse starts
    # ------------------------------------------------------------------
    for n in sizes:
        if time.time() >= deadline:
            break

        # --- build a modest pool of starters ---------------------------------
        starters = []

        # a) previous solution (interpolated if lengths differ)
        if previous is not None:
            try:
                old_len = previous.shape[0]
                x_new = np.linspace(0.0, 1.0, n)
                x_old = np.linspace(0.0, 1.0, old_len)
                a0 = np.interp(x_new, x_old, previous)
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
                starters.append(a0)
                print(f"[init] interpolated previous (n={n})")
            except Exception:
                pass

        # b) a handful of hand‑crafted parametric families
        for cand in _parametric_candidates(n):
            starters.append(cand)

        # c) a few Dirichlet draws (dense)
        for _ in range(3):
            starters.append(np.random.dirichlet(np.ones(n)))

        # d) sparse Dirichlet draws (small concentration -> more spiky)
        for alpha in (0.2, 0.5):
            starters.append(np.random.dirichlet(np.full(n, alpha)))

        # --------------------------------------------------------------
        # Run SA on each starter – each run gets a modest time slice.
        # --------------------------------------------------------------
        for a0 in starters:
            if time.time() >= deadline:
                break

            remaining = deadline - time.time()
            # Allocate at most 30 s or a fraction of the remaining budget.
            run_time = min(30.0, remaining * 0.3)
            run_deadline = time.time() + run_time

            a_opt, val_opt = _simulated_annealing_optimize(
                a0, n, run_deadline, time.time()
            )
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_a = a_opt.copy()
                global_best_n = n
                print(f"[run] n={n} new global best = {global_best_val:.6f}")

            if global_best_val <= 1.503:
                break

        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Final polishing with the hill‑climber
    # ------------------------------------------------------------------
    if global_best_a is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            global_best_a, global_best_val = _final_hillclimb(
                global_best_a,
                global_best_n,
                deadline,
                start_time,
                target_val=1.503,
            )
            print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_a is None:
        n = sizes[0]
        global_best_a = np.full(n, 1.0 / n)
        global_best_val = _objective(global_best_a)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 4️⃣  Log the final score using the host's evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return a plain Python list as required by the harness
    return global_best_a.tolist()
