    n = a.shape[0]

    conv = compute_conv(a)
    maxc = conv.max()
    best_obj = 2 * n * maxc
    best_time = time.time()
    stagnation = 0
    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute conv
        conv = compute_conv(a)
        maxc = conv.max()
        k = int(conv.argmax())
        # compute gradient g
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi+1)
            g[idx] = 2.0 * a[k - idx]
        dot = np.dot(g, a)
        # step size
        prog = (now - best_time) / (deadline - best_time)  # Not ideal; just use relative progress overall.
        # Actually we can use it / max_iter
        eta = eta_start * max(0.0, 1.0 - it / max_iter)

        a = a * np.exp(-eta * (g - dot))
        a_sum = a.sum()
        if a_sum == 0.0:

        else:
            a /= a_sum
        # evaluate objective
        conv = compute_conv(a)
        maxc = conv.max()
        obj = 2 * n * maxc
        if obj < best_obj:

import time
import numpy as np
import math

def _full_self_convolution(a, size=None):
    n = a.shape[0]
    out_len = 2 * n - 1
    if size is None:
        size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def _conv_max(a, size):
    conv = _full_self_convolution(a, size)
    return conv.max()

def _initial_beta(n, p, q):
    i = np.arange(1, n+1, dtype=np.float64)
    # compute log form for numerical stability
    log_a = (p-1) * np.log(i) + (q-1) * np.log(n - i + 1)
    a = np.exp(log_a - np.max(log_a))  # shift for stability
    a = a / a.sum()
    return a

def _interpolate(old_seq, new_len):
    old = np.asarray(old_seq, dtype=float)
    old_len = old.shape[0]
    x_new = np.linspace(0.0, 1.0, new_len)
    x_old = np.linspace(0.0, 1.0, old_len)
    new = np.interp(x_new, x_old, old)
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s == 0.0:
        new = np.full(new_len, 1.0 / new_len)
    else:
        new /= s
    return new

def _mirror_descent(a, size, deadline, eta_start=0.5,
                    max_no_improve=5000, jitter_scale=0.02):
    n = a.shape[0]
    best = a.copy()
    conv = _full_self_convolution(a, size)
    best_obj = 2.0 * n * conv.max()
    best_time = time.time()
    stagnation = 0
    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute convolution and max index
        conv = _full_self_convolution(a, size)
        maxc = conv.max()
        k = int(conv.argmax())
        # gradient g (subgradient at max entry)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # enforce sum constraint via projection (entropy mirror)
        dot = np.dot(g, a)
        prog = (now - best_time) / (deadline - best_time)
        eta = eta_start * max(0.0, 1.0 - prog)  # decay linearly
        a = a * np.exp(-eta * (g - dot))
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
        # evaluate objective after update
        conv = _full_self_convolution(a, size)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best = a.copy()
            stagnation = 0

        else:
            stagnation += 1
        if stagnation >= max_no_improve:
            # add jitter to escape flat region
            a += np.random.normal(scale=jitter_scale, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            stagnation = 0
            print(f"[mirror] jitter at iter={it}")
        it += 1
    return best, best_obj

"""
Search for a non‑negative step‑function minimizing

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))²

The algorithm proceeds in three stages:

1. **Broad initialisation** –  a handful of symmetric Beta(p,p) families,
   a few Gaussian‑like shapes, the uniform distribution and (if available)
   the previously discovered solution are generated and the best one is
   selected according to the exact objective (no soft‑max).

2. **Entropy‑mirror descent on the exact max‑entry** –  starting from the
   best candidate we minimise the true ``max(conv)`` on the probability
   simplex.  The sub‑gradient for a given maximum‑index *k* is
   ``g_i = 2·a_{k‑i}`` (zero when the index is out of range).  An
   exponential‑update (mirror descent with Kullback‑Leibler geometry) is
   performed and the step‑size decays linearly with the remaining time.
   If progress stalls, a small random jitter is added to escape flat
   regions.

3. **Final evaluation** –  the resulting vector is handed to the provided
   ``evaluate_sequence`` wrapper and returned.

The whole routine respects the 1000 s budget (a tiny safety margin is
kept for the final call).  It works for the default length **n = 8000**
used by the competition, but the code is completely generic.
"""

# ----------------------------------------------------------------------
# FFT‑based linear convolution (full, self‑convolution)
# ----------------------------------------------------------------------
def _full_self_convolution(a: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Returns the full linear convolution of ``a`` with itself.
    The length of the result is ``2·len(a)‑1``.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, n=fft_size)
    conv = np.fft.irfft(fa * fa, n=fft_size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper to build symmetric Beta(p,p) vectors on the simplex
# ----------------------------------------------------------------------
def _beta_symmetric(n: int, p: float) -> np.ndarray:
    """
    Returns a symmetric Beta(p,p) density sampled on ``n`` points and
    normalised to sum to 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    # log‑space to avoid overflow for extreme p
    log_a = (p - 1.0) * np.log(i) + (p - 1.0) * np.log(n - i + 1.0)
    # shift for numerical stability
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Helper to build a (truncated) Gaussian‑like shape
# ----------------------------------------------------------------------
def _gaussian_shape(n: int, sigma_frac: float) -> np.ndarray:
    """
    Returns a centred Gaussian profile with standard deviation
    ``sigma_frac * n`` (truncated to the interval) and normalised.
    """
    xs = np.arange(n, dtype=np.float64) - (n - 1.0) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * (xs / sigma) ** 2)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Simple linear interpolation for re‑using a previously found sequence
# ----------------------------------------------------------------------
def _interpolate(old_seq, new_len: int) -> np.ndarray:
    old = np.asarray(old_seq, dtype=float)
    old_len = old.shape[0]
    x_new = np.linspace(0.0, 1.0, new_len)
    x_old = np.linspace(0.0, 1.0, old_len)
    new = np.interp(x_new, x_old, old)
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s == 0.0:
        new = np.full(new_len, 1.0 / new_len)
    else:
        new /= s
    return new


# ----------------------------------------------------------------------
# Entropy‑mirror descent that directly minimises the exact max‑entry
# ----------------------------------------------------------------------
def _mirror_descent_max(
    a: np.ndarray,
    fft_size: int,
    deadline: float,
    start_time: float,
    eta_start: float = 0.6,
    max_no_improve: int = 3000,
    jitter_scale: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Mirrors the update ``a ← a * exp(‑η·(g‑<g,a>))`` where ``g`` is the
    sub‑gradient of the max‑convolution entry.
    Returns the best vector seen and its objective value.
    """
    n = a.shape[0]
    best = a.copy()
    conv = _full_self_convolution(a, fft_size)
    best_obj = 2.0 * n * conv.max()
    stagnation = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # --- current convolution and max‑index ---------------------------------
        conv = _full_self_convolution(a, fft_size)
        maxc = conv.max()
        k = int(conv.argmax())

        # --- sub‑gradient of the max entry ------------------------------------
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]

        dot = np.dot(g, a)

        # --- step‑size (linear decay with remaining time) --------------------
        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * max(0.0, 1.0 - progress)

        # --- mirror‑descent update --------------------------------------------
        a = a * np.exp(-eta * (g - dot))
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # --- evaluate new objective -------------------------------------------
        conv = _full_self_convolution(a, fft_size)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best = a.copy()
            stagnation = 0
            print(f"[mirror] it={it:5d}  O={obj:.6f}  t={now - start_time:.1f}s")
        else:
            stagnation += 1

        # --- escape from flat regions -----------------------------------------
        if stagnation >= max_no_improve:
            a += np.random.normal(scale=jitter_scale, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            stagnation = 0
            print(f"[mirror] jitter added at it={it}")

        it += 1

    return best, best_obj
