"""
Propose a highly‑optimised step‑function sequence.

The algorithm works in three stages:

1️⃣ **Coarse initialisation** – a dense scan of the two‑parameter Beta family
   (plus a few Dirichlet draws) yields a good starting point on the simplex
   Σa = 1, a≥0.

2️⃣ **Long‑run stochastic mirror/Adam optimisation** – we minimise the
   smoothed objective
       φ_β(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i)
   where “⋆” is the ordinary polynomial product.  The gradient of φ_β is
   obtained cheaply via FFTs.  An Adam‑style adaptive step together with a
   simplex‑projection (Euclidean) drives the iterates.  The smoothing parameter
   β is increased multiplicatively each iteration, so the optimisation
   gradually approaches the true max‑convolution objective.  Random jitter
   and an occasional mirror‑descent (exponential‑weights) step help escape
   flat plateaus.

3️⃣ **Final secure evaluation** – the best vector encountered is returned and
   evaluated with the supplied ``evaluate_sequence`` routine.

The routine respects the 1000 s time budget, keeps a safety margin and prints
progress information.  All helper functions are top‑level (no closures)
as required by the competition framework.
"""

import time
import numpy as np

# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of `x` with itself using an FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()     # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None when the parameters generate an all‑zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smoothed max‑convolution objective
        φ_β(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i)

    Returns (grad, max_conv) where ``grad`` is the Euclidean gradient
    (same shape as ``a``) and ``max_conv`` is the exact max‑convolution
    value (needed for the real objective 2·n·max_conv).
    """
    n = a.shape[0]

    # ---- 1. full convolution a⋆a (length 2n‑1) ----
    out_len = 2 * n - 1
    size_conv = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[:out_len]       # (a⋆a)

    max_conv = conv.max()

    # ---- 2. soft‑max weights w_i (log‑sum‑exp smoothing) ----
    # shift by max for numeric stability
    exp_arg = beta * (conv - max_conv)                     # ≤ 0
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum                                      # Σ_i w_i = 1

    # ---- 3. gradient = 2·(a ⋆_cross w_norm) ----
    # cross‑correlation (a ⋆ w_norm)[j] = Σ_k a_k·w_norm_{k+j}
    rev_w = w_norm[::-1]                                    # reverse for convolution
    out_len_corr = n + out_len - 1                          # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1                                    # position of lag 0
    idx = offset - np.arange(n)                             # indices for j = 0…n‑1
    grad = 2.0 * corr_full[idx]

    return grad, max_conv
