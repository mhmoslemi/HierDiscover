def smooth_obj_grad(x, lam, size_a, size_b, n):
    # compute a = softmax(x)
    exps = np.exp(x - np.max(x))  # stable
    a = exps / exps.sum()   # shape (n,)
    # compute convolution a*a
    fa = np.fft.rfft(a, size_a)
    conv = np.fft.irfft(fa * fa, size_a)[:2*n-1]
    max_c = conv.max()
    # compute stable softmax weights w
    shifted = lam * (conv - max_c)
    # exp might underflow; but shifted <=0, fine.
    exp_shift = np.exp(shifted)
    sum_exp = exp_shift.sum()
    w = exp_shift / sum_exp
    # smooth objective: 2*n * (max_c + (1/lam) * np.log(sum_exp))
    smooth_val = max_c + (1.0/lam) * np.log(sum_exp)
    f_val = 2.0 * n * smooth_val
    # gradient wrt a:
    # compute conv_aw = a_rev * w (full conv length 3n-2)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, size_b)
    fw = np.fft.rfft(w, size_b)
    conv_aw = np.fft.irfft(fa_rev * fw, size_b)[:3*n-2]
    grad_a = 2.0 * conv_aw[n-1:n-1+n]   # length n
    # chain rule: grad_x = a * (grad_a - np.dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    return f_val, grad_x

def _lbfgs_optimize(a0, n, lambdas, deadline):
    # compute FFT sizes
    size_a = 1 << ((2*n - 2).bit_length())
    size_b = 1 << ((3*n - 3).bit_length())
    # initial x
    eps = 1e-12
    x = np.log(np.maximum(a0, eps))
    best_a = a0.copy()
    best_val = _objective(a0)
    for lam in lambdas:
        if time.time() >= deadline:
            break
        # define max iterations per lambda, maybe 50
        maxiter = 40
        # Use SciPy minimize if available
        try:
            from scipy.optimize import minimize
        except Exception:
            # fallback: break
            break
        # Here call minimize
        res = minimize(smooth_obj_grad, x, args=(lam, size_a, size_b, n),
                       method='L-BFGS-B', jac=True,
                       options={'maxiter': maxiter, 'disp': False})
        # update x to result
        x = res.x
        # compute a from x
        exps = np.exp(x - np.max(x))
        a = exps / exps.sum()
        # compute exact objective
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L-BFGS lam={lam}] new best {best_val:.6f}")
        # Early exit if target met
        if best_val <= 1.503:
            break
    return best_a, best_val

def _linear_program_step(a, n, M, active_idxs):
    # returns g vector or None
    # build matrix A rows for each k in active_idxs
    rows = []
    for k in active_idxs:
        row = np.zeros(n)  # but building many rows leads to memory O(len(active)*n) acceptable for small len.
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        # for i in i_min..i_max:
        #   j = k - i
        # row[j] = a[i]
        # using vectorized assignment:
        idxs = np.arange(i_min, i_max+1)
        js = k - idxs
        row[js] = a[idxs]
        rows.append(row)
    A = np.vstack(rows)  # shape (m, n)
    b = np.full(len(active_idxs), M)
    c = -np.ones(n)  # maximize sum(g) => minimize -sum
    # bounds: (0, None)
    # Solve
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        g0 = res.x
        if g0.sum() > 0:
            return g0
    return None

def _smooth_obj_grad(x, lam, size_a, size_b, n):
    # x: real vector
    # compute a from softmax
    max_x = np.max(x)
    exps = np.exp(x - max_x)
    a = exps / exps.sum()
    # convolution a*a
    fa = np.fft.rfft(a, size_a)
    conv = np.fft.irfft(fa * fa, size_a)[:2*n-1]
    max_c = conv.max()
    shifted = lam * (conv - max_c)
    # compute exp of shifted
    exp_shift = np.exp(shifted)
    sum_exp = exp_shift.sum()
    # smooth value
    smooth_val = max_c + (1.0/lam) * np.log(sum_exp)
    f_val = 2.0 * n * smooth_val
    # compute w
    w = exp_shift / sum_exp
    # compute gradient wrt a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, size_b)
    fw = np.fft.rfft(w, size_b)
    conv_aw = np.fft.irfft(fa_rev * fw, size_b)[:3*n-2]
    grad_a = 2.0 * conv_aw[n-1:n-1+n]
    # chain rule for x
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    return f_val, grad_x

def _lbfgs_optimize(a0, n, lambdas, local_deadline):
    size_a = 1 << ((2*n - 2).bit_length())
    size_b = 1 << ((3*n - 3).bit_length())
    eps = 1e-15
    x0 = np.log(np.maximum(a0, eps))
    best_a = a0.copy()
    best_val = _objective(a0)
    # ensure we have scipy
    if not HAS_SCIPY:
        return best_a, best_val
    for lam in lambdas:
        if time.time() >= local_deadline:
            break
        # SciPy minimize
        res = scipy.optimize.minimize(_smooth_obj_grad,
                                      x0,
                                      args=(lam, size_a, size_b, n),
                                      method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': 50, 'disp': False})
        x0 = res.x
        # compute candidate a
        exps = np.exp(x0 - np.max(x0))
        a = exps / exps.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L-BFGS lam={lam:.1f}] new best = {best_val:.6f}")
        if best_val <= 1.503:
            break
    return best_a, best_val

def _run_lp(f, n, M, active_idxs):
    # build A matrix and solve LP
    # f: array shape (n,)
    # M: target bound
    # active_idxs: list/array of ints
    # Returns g0 vector or None
    # Ensure SciPy present
    if not HAS_SCIPY:
        return None
    m = len(active_idxs)
    if m == 0:
        return None
    A = np.zeros((m, n))
    for row_idx, k in enumerate(active_idxs):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals
        A[row_idx, j_vals] = f[i_vals]
    b = np.full(m, M)
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.success:
        g0 = res.x
        if g0.sum() > 0:
            return g0
    return None

"""
search for a step‑function that minimises the provided evaluation function.

The algorithm combines three ideas:

1️⃣  **Low‑dimensional initialisation** – a few hand‑crafted seeds
    (previous solution, a Beta‑family sweep, a simple symmetric tent,
    and a handful of Dirichlet draws).

2️⃣  **Projected L‑BFGS on a smooth max** – the original objective
    2·n·max(conv) is approximated by

        f_λ(a) = 2·n·[max(conv) + (1/λ)·log Σ_k e^{λ(conv_k–max)}]

    The gradient ∇_a f_λ is obtained analytically via an FFT‑based
    convolution.  Optimisation is performed on the *soft‑max*
    parametrisation a = softmax(x) with the L‑BFGS‑B routine from
    SciPy.  A continuation on λ (0.5 → 40) drives the solution toward
    the true max.

3️⃣  **Linear‑programming refinement** – after the L‑BFGS stage we
    improve the current candidate by solving the LP described in the
    problem statement:

        maximise   Σ_j g_j
        subject to (a * g)_k ≤ M    for k in the (small) active set
                   g_j ≥ 0,

    where M = max(a*a).  The LP yields a direction g that reduces the
    maximal convolution entry; a convex combination a←(1−t)a+t g
    (t=0.2) is taken.  The step is repeated a few times.

4️⃣  **Final hill‑climbing** – a fast simulated‑annealing hill‑climber
    (identical to the reference implementation) polishes the result.

All stages respect the 1000 s budget; the best sequence found is
returned as a plain list of floats.  The code works even if SciPy is
unavailable (it falls back to the original smooth‑optimiser + hill‑climb
pipeline).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS and LP)
# ----------------------------------------------------------------------
try:
    import scipy.optimize            # type: ignore
    _HAS_SCIPY = True
except Exception:                     # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray, size: int | None = None) -> np.ndarray:
    """Return the full convolution x * x (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    if size is None:
        size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_grad(x: np.ndarray,
                     lam: float,
                     size_a: int,
                     size_b: int,
                     n: int) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of the objective and its gradient w.r.t. the
    soft‑max parameters x (a = softmax(x)).  Returns (value, grad_x).
    """
    # ---- a = softmax(x) ----
    mx = np.max(x)
    e = np.exp(x - mx)
    a = e / e.sum()                     # Σa = 1, a_i ≥ 0

    # ---- convolution a*a ----
    fa = np.fft.rfft(a, size_a)
    conv = np.fft.irfft(fa * fa, size_a)[: 2 * n - 1]
    max_c = conv.max()

    # ---- smooth weights (softmax over λ·(conv−max_c)) ----
    shifted = lam * (conv - max_c)               # ≤ 0
    exp_shift = np.exp(shifted)
    sum_exp = exp_shift.sum()
    w = exp_shift / sum_exp                      # Σ w = 1

    # ---- smooth objective (exact max + log‑sum‑exp term) ----
    smooth_val = max_c + (1.0 / lam) * np.log(sum_exp)
    f_val = 2.0 * n * smooth_val

    # ---- gradient wrt a ----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, size_b)
    fw = np.fft.rfft(w, size_b)
    conv_aw = np.fft.irfft(fa_rev * fw, size_b)[: 3 * n - 2]
    grad_a = 2.0 * conv_aw[n - 1 : n - 1 + n]    # length n

    # ---- chain rule: ∂f/∂x = a ⊙ (grad_a − (grad_a·a)) ----
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    return f_val, grad_x


# ----------------------------------------------------------------------
# L‑BFGS optimiser (smooth max)
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    lambdas: list[float],
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Runs a short L‑BFGS optimisation for each λ in *lambdas*.
    Returns the best a (Σa=1) found and its exact objective.
    """
    if not _HAS_SCIPY:
        # fallback – simply return the seed
        return a0.copy(), _objective(a0)

    size_a = 1 << ((2 * n - 2).bit_length())
    size_b = 1 << ((3 * n - 3).bit_length())

    eps = 1e-15
    x = np.log(np.maximum(a0, eps))   # initialise soft‑max parameters

    best_a = a0.copy()
    best_val = _objective(a0)

    for lam in lambdas:
        if time.time() >= deadline:
            break

        res = scipy.optimize.minimize(
            _smooth_obj_grad,
            x,
            args=(lam, size_a, size_b, n),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 50, "disp": False},
        )
        x = res.x
        # reconstruct a and evaluate the true objective
        exps = np.exp(x - np.max(x))
        a = exps / exps.sum()
        val = _objective(a)
        if val < best_val - 1e-12:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new best = {best_val:.6f}")
            if best_val <= 1.503:
                break
    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (single LP step)
# ----------------------------------------------------------------------
def _run_lp(f: np.ndarray,
            n: int,
            M: float,
            active_idxs: np.ndarray) -> np.ndarray | None:
    """
    Solve  maximise Σ g_j  s.t.  (f * g)_k ≤ M  for k in *active_idxs*,
    g_j ≥ 0.  Returns the raw solution g (not normalised) or None.
    """
    if not _HAS_SCIPY:
        return None

    m = len(active_idxs)
    if m == 0:
        return None

    # Build the constraint matrix A (shape m × n)
    A = np.zeros((m, n))
    for row, k in enumerate(active_idxs):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals
        A[row, j_vals] = f[i_vals]

    b = np.full(m, M)
    c = -np.ones(n)                     # maximise Σ g_j  → minimise −Σ g_j
    bounds = [(0.0, None)] * n

    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                    bounds=bounds, method="highs")
    except Exception:                  # pragma: no cover
        return None

    if res.success and res.x.sum() > 0.0:
        return res.x
    return None


def _lp_improve(a: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 10) -> tuple[np.ndarray, float]:
    """
    Repeatedly solves the LP described above and mixes the returned
    direction into the current a.  Stops when no improvement is seen
    or the time budget expires.
    """
    best_a = a.copy()
    best_val = _objective(best_a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(best_a)
        M = conv.max()
        # active set: entries within 0.1 % of the maximum (capped to 15)
        tol = max(1e-12, 0.001 * M)
        active = np.where(conv >= M - tol)[0]
        if active.size > 15:
            # keep the 15 largest entries
            order = np.argsort(-conv[active])
            active = active[order[:15]]

        g0 = _run_lp(best_a, n, M, active)
        if g0 is None:
            break

        g = g0 / g0.sum()               # normalise to Σg = 1
        t = 0.2                         # mixing factor
        new_a = (1.0 - t) * best_a + t * g
        new_val = _objective(new_a)

        if new_val < best_val - 1e-12:
            best_a, best_val = new_a, new_val
            print(f"[LP step {it}] new best = {best_val:.6f}")
            if best_val <= 1.503:
                break
        else:
            break                        # no progress → stop

    return best_a, best_val


# ----------------------------------------------------------------------
# Beta‑family sweep (cheap dense search)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray | None, float]:
    """Search the beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf

    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple symmetric “tent” shape (another heuristic seed)
# ----------------------------------------------------------------------
def _tent_shape(n: int) -> np.ndarray:
    """Symmetric triangular (tent) distribution, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)            # product grows towards the centre
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a tiny margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # Problem dimension – 2000 gives a good trade‑off between quality
    # and speed while still allowing the theoretical constant ≈1.5.
    n = 2000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise a diverse set of seeds
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # a) previous solution (if any)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        seeds.append(a0)
        print("[init] interpolated previous solution")
    except Exception:                                 # pragma: no cover
        pass

    # b) best Beta‑family seed
    beta_a, beta_val = _beta_family_best(n)
    if beta_a is not None:
        seeds.append(beta_a)
        print(f"[init] beta‑family seed (value={beta_val:.6f})")

    # c) symmetric tent shape
    tent_a = _tent_shape(n)
    seeds.append(tent_a)
    print("[init] tent‑shape seed")

    # d) a few random Dirichlet draws
    for _ in range(4):
        seeds.append(np.random.dirichlet(np.ones(n)))
    print(f"[init] {len(seeds)} seeds prepared")

    # ------------------------------------------------------------------
    # 1️⃣  Smooth L‑BFGS optimisation (≈70 % of remaining time)
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_val = np.inf

    remaining = deadline - time.time()
    time_for_smooth = remaining * 0.70
    per_seed = time_for_smooth / max(1, len(seeds))

    # continuation schedule for λ
    lambda_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]

    for i, seed_a in enumerate(seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed
        a_opt, val_opt = _lbfgs_optimize(seed_a, n, lambda_schedule, local_deadline)
        print(f"[seed {i}] L‑BFGS best = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Linear‑programming refinement (≈20 % of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and global_best_a is not None:
        global_best_a, global_best_val = _lp_improve(
            global_best_a, n, deadline, max_iters=12
        )
        print(f"[LP] best value after refinement = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final intensive hill‑climbing (≈remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and global_best_a is not None:
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:                                       # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
