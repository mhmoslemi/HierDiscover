def subgradient(a, conv, max_active=10, rel_tol=1e-6):
    n = len(a)
    max_val = conv.max()
    if max_active >= conv.size:
        active = np.arange(conv.size)
    else:
        # get indices with highest conv entries
        idx = np.argpartition(conv, -max_active)[-max_active:]
        if rel_tol is not None:
            mask = conv[idx] >= max_val * (1 - rel_tol)
            idx = idx[mask]
        active = idx
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx_arr = np.arange(lo, hi + 1)
        grad[idx_arr] += a[k - idx_arr]
    grad *= 2.0
    return grad, active
```

Propose a non‑negative step function (a probability vector) that minimizes
``evaluate_sequence``.

The algorithm consists of three stages:

1️⃣  **Broad family sweep** – generate many candidate shapes
    (Beta, exponential, Gaussian, power‑sine, triangular,
    asymmetric exponential, two‑Gaussian mixture, random Dirichlet).
    All candidates are normalised to sum = 1 and the best few are kept.

2️⃣  **Adam‑mirrored optimisation** – starting from the best candidate,
    minimise the exact objective
        F(a) = 2·n·max(conv(a,a))
    (which equals ``evaluate_sequence`` for ‖a‖₁=1).
    We use an exponentiated‑gradient Adam update on the log‑weights,
    together with an active‑set sub‑gradient of the current maximal
    convolution entries.  The step‑size decays linearly, a momentum‑free
    Adam variant keeps the iterates inside the simplex, and occasional
    Gaussian jitter prevents stagnation.

3️⃣  **Secure final call** – the best sequence found before the time budget
    expires is passed to ``evaluate_sequence`` and returned.

The optimiser runs until the 1000 s budget (minus a small safety margin)
is exhausted, constantly updating the global best.  The method usually
pushes the objective well below the previous 1.508‑level (target ≈ 1.50)
by flattening the auto‑convolution toward the theoretical limit of ≈ 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of x with itself (length 2·len(x)‑1) using FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a (assumed to sum to 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Shape families for the initial wide exploration
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1}, i = 1,…,n ."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_sym_shape(c: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-c·|i‑mid|)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a


def _exp_asym_shape(c: float, n: int) -> np.ndarray:
    """Asymmetric exponential decay a_i ∝ exp(-c·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-c * i)
    a /= a.sum()
    return a


def _gaussian_shape(sigma: float, n: int) -> np.ndarray:
    """Gaussian centred at the middle."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a


def _power_sine_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1))^p ."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    a /= a.sum()
    return a


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a


def _bimodal_gaussian_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Two Gaussian bumps at random locations."""
    # random centres (avoid extremes)

    mu2 = rng.uniform(0.6, 0.8) * n
    sigma1 = rng.uniform(0.03, 0.07) * n
    sigma2 = rng.uniform(0.03, 0.07) * n
    i = np.arange(n, dtype=np.float64)

    g2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
    weight = rng.random()
    a = weight * g1 + (1 - weight) * g2
    a /= a.sum()
    return a


def _dirichlet_like_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Dirichlet‑like draw (few non‑zero entries)."""
    support = rng.integers(50, 2000)          # moderate sparsity

    vals = rng.random(support) + 1e-6        # avoid exact zeros
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     max_active: int = 10,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Return a sub‑gradient of max(conv) w.r.t. a.
    The active set consists of up to `max_active` largest entries of conv,
    filtered by relative tolerance.
    """
    n = a.shape[0]
    max_val = conv.max()
    size = conv.size

    if max_active >= size:
        active = np.arange(size)
    else:
        # indices of the `max_active` largest entries (unsorted)
        active = np.argpartition(conv, -max_active)[-max_active:]
        # keep only those close enough to the overall maximum
        mask = conv[active] >= max_val * (1.0 - rel_tol)
        if np.any(mask):
            active = active[mask]
        else:
            active = np.array([int(np.argmax(conv))])

    grad = np.zeros_like(a)

    # contribution of each active lag
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # a_i contributes a_{k‑i}
        grad[idx] += a[k - idx]

    grad *= 2.0          # accounts for symmetry of the product
    return grad
