def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def interpolate_seq(seq, new_n):
    old_n = len(seq)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    # ensure positivity
    new_seq = np.maximum(new_seq, 0)
    # normalize sum to 1
    s = new_seq.sum()
    if s > 0:
        new_seq = new_seq / s
    else:
        new_seq = np.full(new_n, 1.0/new_n)
    return new_seq

    # ------------------------------------------------------------------
    for start_a in init_candidates:
        if time.time() >= deadline:
            break
        restarts_done += 1

        # Allocate a time slice for this restart.
        time_left = deadline - time.time()
        # Keep at least 5 s for each future restart.
        min_per_restart = 5.0
        max_this_restart = max(min_per_restart,
                               time_left - min_per_restart * (total_restarts - restarts_done))
        restart_deadline = time.time() + max_this_restart

        # Initialise Adam on the logits (log‑softmax).
        eps = 1e-12
        z = np.log(start_a + eps)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        adam_eps = 1e-8
        lr = 0.06                     # a slightly higher learning rate than before
        t_iter = 0
        stagnation = 0
        IMPROVE_LIMIT = 3000

        # Smooth‑max temperature schedule (exponential growth).
        lam_min, lam_max = 5.0, 3000.0

        while True:
            if time.time() >= restart_deadline:
                break
            t_iter += 1

            # 1️⃣  Soft‑max → a (ensures a ≥ 0, Σa = 1).
            a = _softmax(z)

            # 2️⃣  Exact objective (using the true max of the convolution).
            conv = _full_convolution(a)
            conv_max = conv.max()
            cur_val = 2.0 * n * conv_max       # denominator = 1 because Σa = 1

            # Record global best.
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                stagnation = 0
                elapsed = time.time() - start_time
                print(f"[restart {restarts_done}] new global best = {best_val:.9f} "
                      f"after {t_iter} iters ({elapsed:.1f}s)")
            else:
                stagnation += 1

            # 3️⃣  Smooth‑max weight over the convolution entries.
            prog = (time.time() - start_time) / (deadline - start_time)
            lam = lam_min * ((lam_max / lam_min) ** prog)   # exponential schedule
            exp_vals = np.exp(lam * (conv - conv_max))      # max entry gets exp(0)=1
            w = exp_vals / exp_vals.sum()                  # shape (2n‑1,)

            # 4️⃣  Gradient of the smooth‑max w.r.t. a.
            g_raw = _cross_corr(w, a)      # length n
            g = 2.0 * g_raw

            # 5️⃣  Back‑prop through the soft‑max.
            dot_ag = np.dot(a, g)
            grad_z = a * (g - dot_ag)

            # 6️⃣  Adam update.
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            z -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # 7️⃣  Stagnation handling – add a little noise to logits.
            if stagnation > IMPROVE_LIMIT:
                noise = np.random.normal(scale=0.05, size=n)
                z += noise
                stagnation = 0
                print(f"[restart {restarts_done}] added noise at iter {t_iter}")

        # ------------------------------------------------------------------
        # Optional short local refinement using a true sub‑gradient of max.
        # ------------------------------------------------------------------
        if best_a is not None and time.time() < deadline:
            a_local = best_a.copy()
            sub_iters = 500
            eta0 = 0.02
            for k in range(sub_iters):
                if time.time() >= deadline:
                    break
                conv = _full_convolution(a_local)
                idx_max = int(np.argmax(conv))

                # Build sub‑gradient of max_b w.r.t. a.
                g_sub = np.zeros_like(a_local)
                lo = max(0, idx_max - (n - 1))
                hi = min(idx_max, n - 1)
                i_range = np.arange(lo, hi + 1)
                j_range = idx_max - i_range
                g_sub[i_range] = 2.0 * a_local[j_range]

                # Decreasing step size.
                eta = eta0 / np.sqrt(k + 1)
                a_tmp = a_local - eta * g_sub
                a_local = _project_simplex(a_tmp)

                cur_val = _objective(a_local)
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a_local.copy()
                    elapsed = time.time() - start_time
                    print(f"[local] improved global best = {best_val:.9f} "
                          f"after {k+1} sub‑iters ({elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # Final evaluation using the official routine and return.
    # ------------------------------------------------------------------
    if best_a is None:
        # Fallback: uniform distribution (should never happen).
        best_a = np.full(n, 1.0 / n)
        print("[fallback] returning uniform sequence")

    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
