"""
Propose a candidate step‑function that (hopefully) beats the previous
best bound 1.5082.

The algorithm consists of three stages

1. **Warm start** –  if a previous solution ``height_sequence_1`` exists,
   it is interpolated to the working size.  Otherwise a random Beta‑shaped
   density (p,q drawn uniformly) is used.

2. **Symmetric Adam optimisation** –  we minimise a smooth
   log‑sum‑exp approximation of the ℓ∞‑norm of the self‑convolution.
   Positivity is enforced by working in the log‑domain, the sequence is
   projected onto the probability simplex each step and we explicitly
   enforce mirror symmetry (a[i]=a[n‑1‑i]) on the logits.  The smoothing
   parameter α is increased from 1 to about 400 during the run and the
   learning‑rate decays from 0.15 to 0.004.  Stagnation is broken by a
   tiny symmetric jitter.

3. **Exact mirror‑descent polishing** –  starting from the best vector
   found, an entropy‑mirror descent on the *exact* maximal convolution
   entry (the true objective) finishes the optimisation.

All heavy work (full convolutions) is performed with FFTs (O(n log n)).
The routine respects the 1000 s budget, leaves a small safety margin for
the final ``evaluate_sequence`` call and returns the best sequence it has
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two (different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent polishing (exact max entry)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 1500, eta_start: float = 0.45) -> np.ndarray:
    """Mirror‑descent on the true max‑entry of the autocorrelation."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # Sub‑gradient of max‑entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Helper: generate a random Beta‑shaped start vector
# ----------------------------------------------------------------------
def _random_beta_start(n: int) -> np.ndarray:
    """Return a probability vector of length n with random Beta(p,q) shape."""
    # draw shape parameters on a log‑scale to explore both peaked and
    # edge‑heavy densities
    p = 10 ** np.random.uniform(-0.7, 0.7)   # roughly 0.2 … 5
    q = 10 ** np.random.uniform(-0.7, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    # guard against infinities when p or q < 1 (division by zero)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1e12)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a /= a_sum
    return a
