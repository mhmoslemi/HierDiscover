def simplex_projection(v):
    """Project vector v onto probability simplex (sum=1, >=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8000

    # Initialize a
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        old_len = old.shape[0]
        # interpolate to n points
        x_old = np.linspace(0, 1, old_len)
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total
        print("[init] interpolated previous solution to length", n)
    except Exception:
        a = np.full(n, 1.0 / n)
        print("[init] uniform start", n)

    # Helper functions: conv, obj, proj.
    def full_conv(x):
        n_ = x.shape[0]
        out_len = 2 * n_ - 1
        size = 1 << (out_len - 1).bit_length()  # next power of 2
        fx = np.fft.rfft(x, size)
        conv = np.fft.irfft(fx * fx, size)[:out_len]
        return conv

    def objective(x):
        c = full_conv(x)
        return 2.0 * n * c.max()
    
    # Initial evaluation
    cur_obj = objective(a)
    best_a = a.copy()
    best_val = cur_obj
    print("[init] initial objective:", best_val)

    # stochastic subgradient descent parameters
    step = 0.1  # initial step size
    step_min = 0.001
    decay = 0.9995
    stagn = 0
    max_stagn = 2000

    while time.time() < deadline:
        # compute convolution & max index
        conv = full_conv(a)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        # subgradient
        g = np.zeros_like(a)
        # only indices where counterpart exists
        # indices i where 0 <= max_idx - i < n
        # that is i from max_idx - (n-1) to max_idx
        i_low = max(0, max_idx - (n - 1))
        i_high = min(n - 1, max_idx)
        # for i in this interval: g[i] = 2 * a[max_idx - i]
        # compute using vectorized slice
        slice_i = slice(i_low, i_high + 1)
        # counterpart indexes
        counterpart = max_idx - np.arange(i_low, i_high + 1)
        g[slice_i] = 2.0 * a[counterpart]

        # gradient step
        v = a - step * g
        # projection onto simplex
        # efficient projection
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        if not np.any(cond):
            theta = 0.0
        else:
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        a_new = np.maximum(v - theta, 0.0)

        # evaluate new objective
        new_obj = 2.0 * n * full_conv(a_new).max()
        # accept if improvement else maybe keep old?
        if new_obj < cur_obj:
            a = a_new
            cur_obj = new_obj
            stagn = 0
            if new_obj < best_val:
                best_val = new_obj
                best_a = a.copy()
                print(f"[iter] new best {best_val:.9f} time {time.time()-start:.1f}s")
        else:
            # occasionally accept worse move to escape, with prob
            if np.random.rand() < 0.001:
                a = a_new
                cur_obj = new_obj
            stagn += 1

        # step size decay
        step = max(step * decay, step_min)

        # stagnation handling

            cur_obj = objective(a)
            stagn = 0

    # final evaluation
    final_score = evaluate_sequence(best_a.tolist())
    print("[final] reported", final_score)
    return best_a.tolist()

"""
Simple sub‑gradient optimiser for the L∞ self‑convolution problem.

The algorithm works directly on the L∞ objective:
    minimize  2·n·max_k (a*a)_k   under   a_i ≥ 0 , Σ a_i = 1

At each iteration we
    1. compute the full convolution a*a (FFT, O(n log n)),
    2. locate the index k_max where the convolution attains its maximum,
    3. form a sub‑gradient of the max‑entry w.r.t. a:
           g_i = 2·a_{k_max‑i}   (if the index is in‑range, else 0),
    4. take a step a ← a – η·g and project back onto the probability simplex
       (non‑negative entries that sum to 1),
    5. keep the best sequence seen so far.

The step‑size η decays geometrically; when the optimisation stalls we
inject a small random perturbation to escape flat regions.
The whole routine respects the given time budget (default 1000 s) and
returns the best sequence found.  The final score is obtained by a
single call to ``evaluate_sequence``.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT size)."""
    return 1 << (x - 1).bit_length()

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*x (real FFT, length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    # reference:  https://arxiv.org/abs/1101.6081   (Algorithm 1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    start_time = time.time()

    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------

        a = np.maximum(a, 0.0)
        total = a.sum()
        if total < 1e-12:
