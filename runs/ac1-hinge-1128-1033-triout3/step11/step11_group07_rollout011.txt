def _beta_shape(n, p, q):
    i = np.arange(1, n+1, dtype=np.float64)
    a = np.exp((p-1.0) * np.log(i) + (q-1.0) * np.log(n - i + 1.0))
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= s
    return a

def _adam_optimize_custom(a0, n, deadline, start_time, best_a, best_val):
    w = np.log(a0 + 1e-12)
    beta1,beta2=0.9,0.999
    m=np.zeros_like(w); v=np.zeros_like(w); t=0

    while time.time() < deadline:
        # decode
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        lr = 0.5 * (0.01 ** progress)  # exponentially decay from 0.5 to 0.5*0.01=0.005
        eps = 0.5 * (1e-12/0.5) ** progress  # hmm. Actually we want start=0.5 end=1e-12.
        eps = 0.5 * ( (1e-12) / 0.5 ) ** progress
        # compute smooth gradient
        _, grad = _smooth_max_and_grad(a, eps) # includes 2*n factor
        # gradient w.r.t w
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)
        t +=1

def _subgradient_polish_custom(a0, n, deadline, start_time, best_a, best_val, max_iters=800_000):
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        # find indices where near max
        eps_mask = 1e-12
        mask = b >= max_b - eps_mask
        # compute gradient via summation over these maxima
        grad = np.zeros_like(a)
        idxs = np.where(mask)[0]
        for k in idxs:
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            grad[valid] += 2.0 * a[idx[valid]]
        # step size
        progress = (now - start_time) / (deadline - start_time)
        step = 0.2 * (1.0 - progress) + 1e-6
        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        new_val = _objective(a_new)
        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best {best_val:.6f}")
        # jitter occasionally
        if it % 2000 == 0 and np.random.rand() < 0.05:
            a = np.maximum(a + np.random.normal(scale=0.01, size=n), 0)
            a = _project_simplex(a)
    return best_a, best_val

"""
step‑function optimiser – 1000 s budget
-------------------------------------

The optimiser searches a non‑negative vector `a` (length `n`) with
∑a = 1 that minimises the evaluation function

        eval(a) = 2·n·max(conv(a,a))

The strategy consists of three parts:

1. **Exploratory initialization** – a large variety of analytical
   candidates (symmetric / asymmetric Beta shapes, a triangular “tent”
   shape, random Dirichlet draws and a few random Beta draws) is
   examined and the best one is kept.

2. **Smooth‑approximation Adam optimisation** – we minimise a
   smoothed version of the L∞‑norm of the convolution using Adam on
   the log‑softmax (which automatically enforces the simplex
   constraints).  The smoothing parameter ε is annealed from 0.5 to
   1e‑12; restarts are disabled to let the optimiser fully exploit the
   allocated time.

3. **Exact sub‑gradient polishing** – a projected sub‑gradient method
   is run on the true objective (max‑convolution).  The sub‑gradient
   is obtained efficiently via a second FFT, and the step size decays
   linearly.  Random “jitters’’ are injected occasionally to escape
   flat plateaus.

Finally the solution is blended with the uniform distribution (which
helps to smooth isolated spikes) and evaluated with the provided
`evaluate_sequence` function.

All helper functions are defined at top level, there are no closures
or lambdas, and only the standard library together with NumPy is used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (full linear convolution)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)               # length 2n‑1
    max_b = b.max()
    # soft‑max weights (log‑sum‑exp trick)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()            # shape (2n‑1,)

    # convolution of a with reversed p (the same pattern as the smooth gradient)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)          # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]      # n entries
    grad_a = 4.0 * n * segment[::-1]       # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families of candidate shapes
# ----------------------------------------------------------------------
def _beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Symmetric Beta(p,q) shape on a discrete grid, normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # use logs to avoid overflow for extreme exponents
    log_a = (p - 1.0) * np.log(i) + (q - 1.0) * np.log(n - i + 1.0)
    a = np.exp(log_a)
    s = a.sum()
    if s == 0.0 or np.isinf(s):
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _tent_shape(n: int) -> np.ndarray:
    """Triangular (tent) shape, symmetric, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n - i + 1.0)
    a /= a.sum()
    return a


def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Beta‑shaped candidate (p,q sampled on a log‑scale)."""
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    return _beta_shape(n, p, q)


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam on the smooth approximation (no restarts)."""
    n = a0.shape[0]
    # work in log‑softmax space – the simplex is satisfied automatically
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    while time.time() < deadline:
        # decode logits
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[Adam]  new best = {best_val:.6f}")

        # schedule (progress ∈ [0,1])
        progress = (time.time() - start_time) / (deadline - start_time)
        lr = 0.5 * (0.01 ** progress)                              # 0.5 → 0.005
        eps = 0.5 * ((1e-12) / 0.5) ** progress                     # 0.5 → 1e‑12

        # smoothed gradient
        _, grad = _smooth_max_and_grad(a, eps)                     # already contains 2·n

        # chain rule through the soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to help escape plateaus
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.03, size=n)

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    n = a0.shape[0]
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # current convolution and mask of maximal entries
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(np.float64)   # weight 1 for each max entry

        # exact sub‑gradient (same pattern as smooth case, but mask is binary)
        mask_rev = mask[::-1]
        conv_ab = _conv_fft(a, mask_rev)                # length 3n‑2, we need the middle n entries
        segment = conv_ab[n - 1:2 * n - 1]
        grad = 4.0 * n * segment[::-1]                  # includes factor 2·n

        # linear step‑size schedule (larger at start, tiny near the end)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.25 * (1.0 - progress) + 5e-6

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # small random jitter every few thousand iterations
        if it % 2500 == 0 and np.random.rand() < 0.04:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple convex blending with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try mixtures t·a + (1‑t)·u for a few t‑values."""
    n = a.shape[0]
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient polishing (≈55 % of total budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.92 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a,
        deadline=phase_b_deadline,
        start_time=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        max_iters=800_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ------------------------------------------------------------------
    # Phase C – Final quick Adam run with very small epsilon (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _adam_optimize(
            best_a,
            deadline=deadline,
            start_time=phase_b_deadline,
            best_a=best_a,
            best_val=best_val,
        )
        best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
