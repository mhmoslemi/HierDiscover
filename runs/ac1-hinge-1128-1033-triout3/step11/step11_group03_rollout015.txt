"""
Simple projected sub‑gradient descent on the true
max‑convolution objective, followed by a lightweight
mass‑transfer hill‑climb.  The method works directly with
the ‖a∗a‖∞ term (no soft‑max surrogate) and therefore
can push the solution a bit beyond the previous Adam‑based
search.  It respects the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def _objective_normalized(a: np.ndarray) -> float:
    """Objective value 2·n·max(a∗a) for a already normalised to sum=1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # based on the algorithm of Wang & Carreira‑Perpiñán (2013)
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_max_lag(a: np.ndarray, lag: int) -> np.ndarray:
    """Sub‑gradient of max‑lag term wrt a (2·a shifted)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if lag < n:
        # indices 0 .. lag are affected
        g[:lag + 1] = 2.0 * a[lag::-1]
    else:
        start = lag - (n - 1)   # first index that receives a contribution
        length = 2 * n - 1 - lag   # how many entries are involved
        # a[:length] are paired with g[start: start+length] in reversed order
        g[start:] = 2.0 * a[:length][::-1]
    return g

def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 4000,
                         step0: float = 0.3,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    Returns the best (normalized) vector and its objective.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)               # ensure sum = 1
    cur_val = _objective_normalized(a)
    best_a = a.copy()
    best_val = cur_val

    for t in range(1, max_iters + 1):
        now = time.time()
        if now >= deadline:
            break

        # compute convolution once per iteration
        conv = _full_convolution(a)
        lag_max = int(np.argmax(conv))
        # sub‑gradient for the active lag
        g = _gradient_max_lag(a, lag_max)

        # step size schedule – classic 1/√t decay
        step = step0 / np.sqrt(t)

        a_candidate = a - step * g
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        cand_val = _objective_normalized(a_candidate)

        # accept the move if it does not increase the exact objective
        if cand_val <= cur_val:
            a = a_candidate
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[subgrad] iter {t:6d} lag={lag_max:5d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            # occasionally allow a tiny increase to escape plateaus
            if np.random.rand() < 0.001:
                a = a_candidate
                cur_val = cand_val

        # periodic random jitter to avoid stagnation
        if t % 1000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_val = _objective_normalized(a)

    return best_a, best_val

def _final_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Same hill‑climb as in the reference solution – mass transfer
    from a heavy contributor of the worst lag to a light one.
    """
    b = _full_convolution(a)
    cur_val = _objective_normalized(a)
    best_val = cur_val
    best_a = a.copy()

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-12:
            continue

        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = _objective_normalized(a_new)

        if new_val < cur_val:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = _objective_normalized(a)
            no_improve = 0

        iteration += 1

    return best_a, best_val

def _interpolate_previous(prev_seq, n):
    """Resize a previously found sequence to length n using linear interpolation."""
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None

"""
Simple projected sub‑gradient descent on the exact max‑convolution
objective, followed by a lightweight hill‑climb.  The optimiser works
directly with the true ‖a∗a‖∞ term (no soft‑max surrogate) and
therefore can push the evaluation below the previous 1.5084 barrier.
The routine respects a 1000 s time budget, uses a few handcrafted
initialisations, many random Dirichlet draws and the best solution so
far (if available) as starting points.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, real inputs)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a, length = 2·len(a)-1 (FFT implementation)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (exact, assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_normalized(a: np.ndarray) -> float:
    """Returns 2·n·max(a∗a) for a with sum = 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx = 1} (Euclidean)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the active convolution lag
# ----------------------------------------------------------------------
def _gradient_max_lag(a: np.ndarray, lag: int) -> np.ndarray:
    """Sub‑gradient of the term max_k (a∗a)_k at the given lag."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if lag < n:                               # peak lies in the left half
        g[: lag + 1] = 2.0 * a[lag::-1]
    else:                                     # peak in the right half
        start = lag - (n - 1)                 # first index that receives a contribution
        # a[start:] are the values paired (reversed) with g[start:].
        g[start:] = 2.0 * a[start:][::-1]
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 4000,
                         step0: float = 0.3,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent.
    Returns the best (sum = 1) vector and its exact objective.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                     # enforce Σa = 1
    cur_val = _objective_normalized(a)

    best_a = a.copy()
    best_val = cur_val

    for t in range(1, max_iters + 1):
        now = time.time()
        if now >= deadline:
            break

        # current convolution and the lag attaining the maximum
        conv = _full_convolution(a)
        lag_max = int(np.argmax(conv))

        # sub‑gradient for that lag
        g = _gradient_max_lag(a, lag_max)

        # classic 1/√t step‑size decay
        step = step0 / np.sqrt(t)

        a_candidate = a - step * g
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        cand_val = _objective_normalized(a_candidate)

        # accept only if we do not increase the exact objective
        if cand_val <= cur_val:
            a = a_candidate
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[subgrad] iter {t:6d} lag={lag_max:5d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            # very occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = a_candidate
                cur_val = cand_val

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_val = _objective_normalized(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Light‑weight hill‑climbing (mass‑transfer) – same as the reference version
# ----------------------------------------------------------------------
def _final_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Polishes a candidate by moving small mass from heavy to light contributors."""
    b = _full_convolution(a)
    cur_val = _objective_normalized(a)
    best_val = cur_val
    best_a = a.copy()

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-12:
            continue

        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = _objective_normalized(a_new)

        if new_val < cur_val:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = _objective_normalized(a)
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to resize a previously found sequence (if it exists)
# ----------------------------------------------------------------------
def _interpolate_previous(prev_seq, n):
    """Linearly interpolate an older solution to a new length."""
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The method combines a few deterministic initial shapes, many random
    Dirichlet starts and a projected sub‑gradient descent on the exact
    max‑convolution objective, finishing with a mass‑transfer hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Problem size – 8000 works well and matches the previous best.
    n = 8000

    # Try to load the previously reported solution.
    try:
        previous = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    except Exception:
        previous = None

    # ------------------------------------------------------------------
    # Gather a diverse set of initial candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # a) uniform distribution (baseline)
    init_candidates.append(np.full(n, 1.0 / n))

    # b) interpolated previous solution, if available
    if previous is not None:
        a0 = _interpolate_previous(previous, n)
        if a0 is not None:
            init_candidates.append(a0)

    # c) handcrafted shapes
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Gaussian bell
    sigma = 0.15 * n
    gauss = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    gauss /= gauss.sum()
    init_candidates.append(gauss)

    # Exponential decay (right‑hand side heavy)
    decay = np.exp(-i / (0.25 * n))
    decay /= decay.sum()
    init_candidates.append(decay)

    # Power‑law (beta‑like) shape
    p = 1.5
    power = (i + 1) ** (p - 1) * (n - i) ** (p - 1)
    power /= power.sum()
    init_candidates.append(power)

    # d) several random Dirichlet draws (concentration 0.7 → mildly sparse)
    for _ in range(8):
        init_candidates.append(np.random.dirichlet(np.full(n, 0.7)))

    global_best_a = None
    global_best_val = np.inf

    # ------------------------------------------------------------------
    # 1️⃣  Sub‑gradient optimisation from each starter
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(init_candidates):
        remaining = deadline - time.time()
        if remaining < 2.0:
            break

        # Allocate at most 25 s (or a third of the remaining budget) per run
        run_deadline = time.time() + min(25.0, remaining * 0.33)

        a_opt, val_opt = _subgradient_descent(
            a0,
            n,
            run_deadline,
            start_time,
            max_iters=5000,
            step0=0.25,
            target_val=1.503,
        )
        print(f"[run] start {idx:02d} refined val={val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_a = a_opt.copy()

        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Polishing with the reference hill‑climb
    # ------------------------------------------------------------------
    if global_best_a is None:
        global_best_a = np.full(n, 1.0 / n)
        global_best_val = _objective_normalized(global_best_a)

    remaining = deadline - time.time()
    if remaining > 2.0:
        global_best_a, global_best_val = _final_hillclimb(
            global_best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final evaluation (only for logging – the host calls evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
