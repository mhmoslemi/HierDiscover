    cur_conv = _full_convolution(cur_a)
    cur_val = _objective_from_conv(cur_conv, n)
    best_a = cur_a.copy()
    best_val = cur_val

    it = 0
    while time.time() < deadline:
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(0.001, (1-frac) * start_temp)  # gradually decrease
        # propose move
        # Determine src idx weighted by contribution to max conv
        k_max = np.argmax(cur_conv)
        # contributions: a[i] * a[k_max - i] (if valid)
        # Choose src weighted by contribution magnitude
        idx_srcs = np.arange(n)
        contrib_src = np.zeros_like(cur_a)
        # compute contributions quickly: for each i, if 0 <= k_max - i < n: a[i] * a[paired]
        # vectorized:
        j = k_max - idx_srcs
        mask = (j >= 0) & (j < n)
        contrib_src[mask] = cur_a[mask] * cur_a[j[mask]]
        total_contrib = contrib_src.sum()
        if total_contrib == 0 or np.random.rand() < 0.2:
            # random src
            src = np.random.randint(n)
        else:
            src = np.random.choice(n, p=contrib_src/total_contrib)
        # choose tgt randomly as index not equal src
        tgt = np.random.randint(n-1)
        if tgt >= src:
            tgt += 1

        # delta
        delta = cur_a[src] * np.random.rand() * 0.5  # move up to half of src mass
        if delta < 1e-12:
            continue

        cand_a = cur_a.copy()
        cand_a[src] -= delta
        cand_a[tgt] += delta

        # compute new conv and val
        cand_conv = _full_convolution(cand_a)
        cand_val = _objective_from_conv(cand_conv, n)

        if cand_val < cur_val:
            accept = True
        else:
            # maybe accept uphill
            prob = math.exp(-(cand_val - cur_val) / T) if T > 0 else 0.0
            accept = np.random.rand() < prob

        if accept:
            cur_a = cand_a
            cur_conv = cand_conv
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[SA] iter {it} val={best_val:.6f}")

        it += 1

        # optional random restart if no improvement for many iterations
        # keep track.

    return best_a, best_val

def _simulated_annealing(a0, n, deadline, start_time):
    a = a0.copy()
    a_sum = a.sum()  # should be 1, but may not.
    # ensure normalized? Not necessary for objective, but keep sum may be constant to avoid scaling.
    # We'll keep sum constant by moving mass only.
    cur_conv = _full_convolution(a)
    cur_val = _objective_from_conv(cur_conv, n)
    best_a = a.copy()
    best_val = cur_val

    max_iter = 0
    no_improve = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # temperature linear schedule
        frac = (now - start_time) / (deadline - start_time)
        T = max(1e-6, (1 - frac) * 0.05)  # start at 0.05 maybe

        # select src weighted by current values (probability proportional)
        src = np.random.choice(n, p=a / a.sum())
        # ensure src has some mass
        if a[src] < 1e-12:
            continue

        # target random not src
        tgt = np.random.randint(n-1)
        if tgt >= src:
            tgt += 1

        # propose delta
        delta = a[src] * np.random.rand() * 0.5
        cand_a = a.copy()
        cand_a[src] -= delta
        cand_a[tgt] += delta

        # compute conv and objective
        cand_conv = _full_convolution(cand_a)

        accept = False
        if cand_val <= cur_val:

        else:
            # uphill acceptance
            if T > 0:
                prob = math.exp(-(cand_val - cur_val) / T)
                if np.random.rand() < prob:

        if accept:
            a = cand_a
            cur_conv = cand_conv

            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()

                # optional early break if close to target
                if best_val <= 1.503:
                    return best_a, best_val

        if max_iter % 20000 == 0:

"""
Simulated‑annealing + light hill‑climbing search for a low‑peak
step‑function.  The goal is to minimise

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear convolution of the non‑negative sequence `a`
with itself.  The algorithm works inside the 1000 s budget that the
evaluation harness gives us.

Strategy
========
1.  Pick a problem size `n` (we try a few sizes, e.g. 8000 – 16000).  
2.  Initialise several candidate sequences:
    * the previous best solution (if present) interpolated to the new length,
    * the uniform distribution,
    * a handful of random Dirichlet draws.
3.  For each initialisation run a short simulated‑annealing loop:
    * at each iteration a small amount of mass is moved from a
      “heavy’’ index to a random index,
    * the exact objective (`2·n·max(conv) / sum²`) is computed with an FFT‑based
      convolution,
    * downhill moves are always accepted; uphill moves are accepted
      with probability `exp(-(Δ) / T)` where the temperature `T`
      decreases linearly to zero.
    * the best sequence seen so far is kept globally.
4.  After the annealing stage a lightweight hill‑climbing phase
    (the same routine that was used previously) polishes the best
    candidate until time runs out.
5.  The best sequence is returned.  All numbers are non‑negative,
    the sum is kept equal to 1 (the objective is scale‑invariant).

The implementation uses only NumPy and the standard library, stays
well within the CPU time limit and prints a few progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)-1) via FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale invariant – sum is kept = 1)
# ----------------------------------------------------------------------
def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact value 2·n·max(a∗a) given the full convolution."""
    return float(2.0 * n * conv.max())


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (assumed non‑negative, sum>0)."""
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.shape[0])


# ----------------------------------------------------------------------
# Helper: Interpolate a previous solution to a new length
# ----------------------------------------------------------------------
def _interpolate_previous(prev_seq, n):
    """Resize `prev_seq` to length `n` with linear interpolation."""
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         init_temp: float = 0.08,
                         min_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Simple SA on the exact objective.
    Returns the best sequence seen and its exact objective.
    The move transfers a random fraction of the mass of a heavy index
    to a random target index.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s                                 # keep sum = 1

    cur_conv = _full_convolution(a)
    cur_val = _objective_from_conv(cur_conv, n)

    best_a = a.copy()
    best_val = cur_val

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear cooling schedule
        frac = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        T = max(min_temp, (1.0 - frac) * init_temp)

        # --- propose a move -------------------------------------------------
        # source index – choose proportionally to current weight (heavy indices)
        src = np.random.choice(n, p=a / a.sum())
        if a[src] < 1e-12:
            continue

        # target index – any index different from src
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        # amount of mass to move (up to 50 % of the source)
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        cand_a = a.copy()
        cand_a[src] -= delta
        cand_a[tgt] += delta

        # --- evaluate -------------------------------------------------------
        cand_conv = _full_convolution(cand_a)
        cand_val = _objective_from_conv(cand_conv, n)

        # --- acceptance ------------------------------------------------------
        accept = False
        if cand_val <= cur_val:
            accept = True
        else:
            # uphill move accepted with Metropolis probability
            prob = math.exp(-(cand_val - cur_val) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand_a
            cur_conv = cand_conv
            cur_val = cand_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SA] iter {it:7d} val={best_val:.6f}")
                if best_val <= 1.503:
                    return best_a, best_val

        # occasional jitter to avoid stagnation
        if it % 100_000 == 0 and it > 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            cur_conv = _full_convolution(a)
            cur_val = _objective_from_conv(cur_conv, n)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Light hill‑climbing (mass‑transfer) – same as the previous implementation
# ----------------------------------------------------------------------
def _final_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that moves a small amount of mass from the index
    contributing most to the current maximal lag to a low‑contributing index.
    Stops early if the target value is reached.
    """
    b = _full_convolution(a)
    cur_val = _objective_from_conv(b, n)
    best_val = cur_val
    best_a = a.copy()

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # locate lag giving the maximal autocorrelation
        k_max = int(np.argmax(b))

        # indices that participate in this lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        # contribution of each position to the offending lag
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source: heavy contributor
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-12:
            continue

        # target: a location that is not a heavy contributor
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        # move a small random fraction of the source mass
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = _objective_from_conv(b_new, n)

        if new_val < cur_val:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # random jitter around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective_from_conv(b, n)
            no_improve = 0

        iteration += 1

    return best_a, best_val
