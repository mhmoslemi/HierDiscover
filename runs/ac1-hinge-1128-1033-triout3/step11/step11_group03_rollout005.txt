"""
Beta‑distribution search for a near‑flat sum convolution.
...
"""
import time, math, numpy as np

# convolution
def _full_convolution(x):
    n = x.shape[0]
    out_len = 2*n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _objective(a):
    conv = _full_convolution(a)
    maxb = conv.max()
    n = a.shape[0]
    # sum may not be exactly 1 after rounding, keep exact
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * maxb / (s * s))

def _beta_candidate(n, alpha, beta, eps=0.5):
    i = np.arange(n, dtype=np.float64)
    a = ((i + eps) ** (alpha - 1.0)) * ((n - 1 - i + eps) ** (beta - 1.0))
    total = a.sum()
    if total == 0.0:
        return None
    return a / total

def _sin_candidate(n, gamma):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** gamma
    s = a.sum()
    if s == 0.0:
        return None
    return a / s

# hill climbing mass‑transfer (same as earlier version)
def _final_hillclimb(a, n, deadline, start_time, target_val=1.503):
    b = _full_convolution(a)
    cur_val = _objective(a)
    best_val = cur_val
    best_a = a.copy()
    no_improve = 0
    IMPROVE_LIMIT = 50000
    iteration = 0
    while True:
        if time.time() >= deadline:
            break
        # find lag with maximal convolution
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(contrib))
        if a[src] < 1e-12:
            continue
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue
        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        b_new = _full_convolution(a_new)
        new_val = _objective(a_new)
        if new_val < cur_val:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1
            if no_improve > IMPROVE_LIMIT:
                a = best_a.copy()
                noise = np.random.normal(scale=0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                a_sum = a.sum()
                if a_sum > 0:
                    a /= a_sum
                b = _full_convolution(a)
                cur_val = _objective(a)
                no_improve = 0
        iteration += 1
    return best_a, best_val

"""
Beta‑distribution‑based search for a flat self‑convolution.

The objective is
    2·n·max(conv(a,a)) / (sum(a)^2)

which, after normalising sum(a)=1, is simply 2·n·max(conv(a,a)).
For a probability distribution a on {0,…,n‑1}, conv(a,a) is the distribution
of the sum of two independent draws from a.  The ideal situation is a uniform
sum‑distribution (max≈1/(2n) ⇒ objective≈1).  The continuous arcsine density
has exactly this property, and its discrete analogue is a Beta‑distribution with
shape parameters α=β≈½.  The code enumerates a small grid of Beta shapes
(and a few sine‑power families) for several problem sizes, picks the best
candidate, and then runs a cheap mass‑transfer hill‑climb to polish the result.
The whole procedure respects the 1000 s budget and returns a list of non‑negative
heights that beats the target bound of 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (FFT, real‑valued)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(a*a) / (sum(a)^2)."""
    conv = _full_convolution(a)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _beta_candidate(n: int, alpha: float, beta: float, eps: float = 0.5) -> np.ndarray | None:
    """Discrete Beta(α,β) on {0,…,n‑1}, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + eps) ** (alpha - 1.0)) * ((n - 1 - i + eps) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def _sin_candidate(n: int, gamma: float) -> np.ndarray | None:
    """a_i ∝ sin(π·(i+0.5)/n)^γ, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** gamma
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


# ----------------------------------------------------------------------
# Light‑weight hill‑climb (mass‑transfer) for polishing
# ----------------------------------------------------------------------
def _final_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Simple hill‑climb that moves a small amount of mass from a “heavy’’ index
    to a “light’’ one.  Stops early if the target value is reached."""
    b = _full_convolution(a)
    cur_val = _objective(a)
    best_val = cur_val
    best_a = a.copy()

    no_improve = 0
    IMPROVE_LIMIT = 50_000
    iteration = 0

    while True:
        if time.time() >= deadline:
            break

        # lag giving the current maximum
        k_max = int(np.argmax(b))

        # indices that contribute to this lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source: preferably a large contributor
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-12:
            continue

        # target: a location that is *not* a heavy contributor
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        # move a random fraction of the source mass
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = _objective(a_new)

        if new_val < cur_val:
            # accept move
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1
            if no_improve > IMPROVE_LIMIT:
                # jitter around the current best to escape a plateau
                a = best_a.copy()
                noise = np.random.normal(scale=0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                s = a.sum()
                if s > 0:
                    a /= s
                b = _full_convolution(a)
                cur_val = _objective(a)
                no_improve = 0

        iteration += 1

    return best_a, best_val
