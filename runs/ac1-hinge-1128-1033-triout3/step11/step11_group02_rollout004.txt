"""
Optimisation of a non‑negative step function for the given evaluation
metric.

The search consists of three stages:

1️⃣  Initialise from several sensible seeds (previous solution, a
    Beta‑family sweep and a few Dirichlet draws).

2️⃣  A long‑run *smooth‑max* optimiser that works on a differentiable
    approximation of the objective.  The gradient is obtained via FFT
    and the update uses an exponentiated‑gradient (mirror‑descent) step
    with a decaying learning rate.  The smoothness parameter λ is
    increased from 0.5 up to 200; after λ is large enough we switch to
    the exact sub‑gradient of the max‑entry (a cheap O(n) operation).

3️⃣  An intensive simulated‑annealing hill‑climber (the same implementation
    as in the reference code) to perform a final local refinement.

All stages respect the 1000 s budget.  The best sequence found is
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x * x (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Beta‑family sweep (dense but cheap)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray | None, float]:
    """Search the beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a: np.ndarray | None = None
    best_val = np.inf

    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient of a single convolution entry (sub‑gradient of max)
# ----------------------------------------------------------------------
def _gradient_max_conv(a: np.ndarray, k: int, n: int) -> np.ndarray:
    """
    Gradient of (a*a)_k with respect to a.
    Returns g_i = a_{k-i} (if in range) + a_{i-k} (if in range).
    """
    g = np.zeros_like(a)

    # term a_{k-i}
    i_start1 = max(0, k - (n - 1))
    i_end1 = min(k, n - 1)
    if i_start1 <= i_end1:
        idx1 = np.arange(i_start1, i_end1 + 1)
        g[idx1] += a[k - idx1]

    # term a_{i-k}
    i_start2 = max(k, 0)
    i_end2 = min(k + n - 1, n - 1)
    if i_start2 <= i_end2:
        idx2 = np.arange(i_start2, i_end2 + 1)
        g[idx2] += a[idx2 - k]

    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient smooth‑max optimiser
# ----------------------------------------------------------------------
def _smooth_exp_optimize(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Optimises 2·n·max(conv(a,a)) using a smooth‑max surrogate.
    An exponentiated‑gradient (mirror descent) step is applied.
    λ is increased from 0.5 to 200; after λ≈200 a sub‑gradient of the
    true max entry is used.
    """
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= a_sum                     # enforce Σa = 1

    best_a = a.copy()
    best_val = _objective(a)

    size = 1 << ((2 * n - 2).bit_length())    # FFT size for convolutions
    switch_iter = 2000          # after this we stop smoothing
    lambda_start = 0.5
    lambda_end = 200.0
    eta0 = 0.12                 # base learning rate for exponentiated update

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        # ----- full convolution a*a -----
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
        max_c = conv.max()
        k_max = int(np.argmax(conv))
        cur_val = 2.0 * n * max_c

        # keep the best seen so far
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SmoothExp {t:7d}] new best = {best_val:.6f}")

        # ----- gradient computation -----
        if t < switch_iter:
            # smooth‑max (soft‑max) weights
            prog = t / switch_iter
            lam = lambda_start + (lambda_end - lambda_start) * prog
            exps = np.exp(lam * (conv - max_c))
            w = exps / exps.sum()                     # length 2·n‑1
            # gradient = 2·conv(a_rev, w)
            a_rev = a[::-1]
            fw = np.fft.rfft(w, size)
            fa_rev = np.fft.rfft(a_rev, size)
            conv_aw = np.fft.irfft(fa_rev * fw, size)[: 2 * n - 1]
            grad = 2.0 * conv_aw[n - 1 : n - 1 + n]    # shape n
        else:
            # exact sub‑gradient of the max entry
            grad = 2.0 * _gradient_max_conv(a, k_max, n)

        # ----- exponentiated‑gradient update -----
        eta = eta0 / math.sqrt(t)                     # decay ∝ 1/√t
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        # occasional stochastic perturbation to escape plateaus
        if t % 8000 == 0 and cur_val >= best_val * 0.9999:
            print(f"[SmoothExp {t:7d}] stagnation – random perturb")
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val
