"""
Improved minimiser for the step‑function objective.

Approach
--------
1️⃣  Initialise a candidate from a small grid of Beta‑shaped
    distributions and a few Dirichlet draws – cheap O(n log n) work.

2️⃣  Optimise the *exact* objective 2·n·max(a∗a) while keeping
    a on the probability simplex (∑a=1, a≥0).  Instead of a
    smoothed log‑sum‑exp we use a true sub‑gradient of the max
    (the derivative of the active convolution entry).  This
    converges much faster: each iteration needs only one FFT for
    the convolution and a linear‑time construction of the gradient.

3️⃣  A simple decaying step size (∝1/√t) and an occasional tiny
    random jitter avoid plateaus.  The best vector seen so far is
    stored and finally evaluated with the provided ``evaluate_sequence``.

The routine respects the 1000 s time‑budget, prints progress and
returns the best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow_two(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(a∗a) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex Σx = 1, x ≥ 0.
    Returns a new array (the input is not modified).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the parameters give an all‑zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Compute a sub‑gradient of   φ(a) = max_i (a∗a)_i   and also the value.
    The sub‑gradient for an active index k is 2·a shifted by k.
    If several indices are active we average their contributions.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_val = conv.max()
    # indices where the maximum is attained (exact equality is fine)
    max_idx = np.flatnonzero(conv == max_val)

    grad = np.zeros_like(a)
    # for each active shift k add the shifted copy of a
    for k in max_idx:
        # j runs over all positions where a_{k-j} is defined
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # slice of positions in a that contribute
        j_range = np.arange(j_start, j_end + 1, dtype=np.int64)
        a_idx = k - j_range                     # corresponding a‑indices
        grad[j_range] += a[a_idx]

    # average if there were several active shifts
    grad *= 2.0 / max_idx.size
    return grad, max_val

        p = np.exp(rng.uniform(log_min, log_max))
        q = np.exp(rng.uniform(log_min, log_max))
        cand = _beta_shape(p, q, n)
        if cand is None:
            continue
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand] new best = {best_val:.6f} (p={p:.4g}, q={q:.4g})")
        if time.time() >= deadline - 30.0:   # reserve ~30 s for refinement
            break

    # a few Dirichlet draws for extra diversity
    for _ in range(8):
        cand = rng.dirichlet(np.ones(n))
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[dir] new best = {best_val:.6f}")

    # fallback – uniform distribution
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print("[fallback] uniform start, value=", best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent on the exact max‑convolution objective
    # ------------------------------------------------------------------
    a = best_a.copy()
    best_seq = a.copy()
    best_score = best_val

    step0 = 0.8            # initial step size
    jitter_prob = 0.001
    jitter_scale = 0.001
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # compute sub‑gradient of max(conv) and current max
        grad, cur_max = _subgradient_max(a)
        cur_val = 2.0 * n * cur_max       # exact objective (sum(a)=1)

        if cur_val < best_score:
            best_score = cur_val
            best_seq = a.copy()
            print(f"[subgrad] new best = {best_score:.6f} (iter {iteration})")
            if best_score <= 1.503:      # target reached – stop early
                break

        # step size decays like 1/√t
        step = step0 / np.sqrt(iteration + 1.0)

        # projected sub‑gradient step
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        # occasional tiny random jitter to escape flat regions
        if np.random.rand() < jitter_prob:
            a_candidate += np.random.normal(scale=jitter_scale, size=n)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

        a = a_candidate
        iteration += 1

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
