Pseudo:

def proj_simplex(v):
    # Project onto {x >=0, sum x = 1}
    if v.sum() == 1 and np.all(v >=0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Sub‑gradient optimiser for the step‑function attack.

The algorithm works directly on the true objective
    J(a) = 2·n·max_k (a * a)[k]   (with Σ a = 1, a_i ≥ 0)
instead of a smooth approximation.  For a given index k* that attains the
max‑convolution we have the exact sub‑gradient

    ∂J/∂a_i = 2·a_{k*‑i}   (if 0 ≤ k*‑i < n, else 0).

We keep the vector a on the probability simplex using the classic
 Euclidean projection onto {x ≥ 0, Σx = 1}.  A simple diminishing
step‑size (1/√t) rule together with occasional random restarts provides
enough exploration to beat the previous 1.508‑score; the code also scans a
few parametric families (geometric, power‑law and random Dirichlet) to
seed the optimisation with strong starting points.

The routine respects a 1000 s budget, prints progress and finally returns
the best sequence found (as a list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (real FFT, length 2·len(a)-1).
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
        { x | x_i ≥ 0 , Σ x_i = 1 }.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety; fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1            # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact objective value: 2·n·max(convolution).  Assumes Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


def _subgradient(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of max‑convolution at the chosen index k_max.
    """
    n = a.shape[0]
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k_max - idx]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds,
    keeping the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give a tiny safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a compromise between speed and quality.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Helper to evaluate quickly and keep the global best.
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # uniform start
    best_val = _objective(best_a)

    print(f"[init] uniform start => val = {best_val:.9f}")

    # --------------------------------------------------------------
    # 1️⃣  Scan a few parametric families to obtain a good seed.
    # --------------------------------------------------------------
    # geometric (exponential) families
    for r in (0.990, 0.995, 0.997, 0.998, 0.999, 0.9995):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[seed] geom r={r:.4f} => val = {val:.9f}")

    # power‑law families a_i ∝ (i+1)^{-p}
    for p in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0):
        a = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[seed] power p={p:.2f} => val = {val:.9f}")

    # a few random Dirichlet draws
    for _ in range(20):
        a = np.random.gamma(1.0, 1.0, size=n)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[seed] random Dirichlet => val = {val:.9f}")

    # --------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent on the true objective.
    # --------------------------------------------------------------
    a = best_a.copy()
    lr0 = 0.5                     # initial step‑size multiplier
    stagnation = 0                # counts iterations without improvement
    IMPAT = 3000                  # after this many nothing, add noise

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        it += 1
        # ---- compute convection and objective ----
        conv = _full_convolution(a)
        conv_max = conv.max()
        cur_val = 2.0 * n * conv_max

        # ---- keep global best ----
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            stagnation = 0
            print(f"[iter {it:06d}] new best = {best_val:.9f} "
                  f"(t={now - start_time:.1f}s)")
        else:
            stagnation += 1

        # ---- sub‑gradient for a randomly chosen maximal index ----
        # (if several positions share the maximum we pick one uniformly)
        max_idxs = np.nonzero(conv >= conv_max - 1e-12)[0]
        k_star = np.random.choice(max_idxs)

        g = _subgradient(a, k_star)

        # ---- diminishing step‑size (simple 1/√t rule) ----
        step = lr0 / np.sqrt(it)

        # ---- projected gradient step onto the simplex ----
        a_next = _proj_simplex(a - step * g)

        a = a_next

        # ---- stagnation handling: occasional random perturbation ----
        if stagnation > IMPAT:
            noise = np.random.normal(scale=0.02, size=n)
            a = _proj_simplex(a + noise)
            stagnation = 0
            print(f"[reset] added noise at iter {it}")

    # --------------------------------------------------------------
    # 3️⃣  Final evaluation (the only place we call evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
