```python
def _full_convolution(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow_two(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
```

assistantfinal**Solution Overview**

The evaluation metric  

\[
E(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is invariant under scaling of the vector `a`.  
Therefore we may restrict ourselves to the simplex  

\[
\Delta=\{a\in\mathbb R_{\ge0}^n\;|\;\sum_i a_i=1\}.
\]

The difficult part is the `max` in the numerator.  
We replace it by the smooth log‑sum‑exp (soft‑max)

\[
\phi_\lambda(a)=\max_k (a*a)_k
   +\frac{1}{\lambda}\log\!\Bigl(\sum_k e^{\lambda\bigl((a*a)_k-\max_k(a*a)_k\bigr)}\Bigr)
\]

which converges to the true maximum as \(\lambda\to\infty\).  
For a fixed \(\lambda\) the function \(\phi_\lambda\) is differentiable and its
gradient can be written as a convolution, so it can be evaluated with two FFTs.

The optimisation proceeds in three phases

1. **Diverse seeds** – beta‑family sweep, a few Dirichlet draws and (if present)
   the previous solution.
2. **Continuation + L‑BFGS‑B** – starting with a small \(\lambda\) we minimise
   \(\phi_\lambda\) with the bound‑constrained quasi‑Newton method `L‑BFGS‑B`.
   After each stage we increase \(\lambda\) (2·4·8·16·32·64·128) and warm‑start
   the next optimisation, thereby tightening the smooth approximation.
3. **Final intensive hill‑climber** – the (already supplied) simulated‑annealing
   local search is applied to the best vector obtained so far.

All work respects the 1000 s budget, and the best sequence found is returned.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Robust optimisation of a non‑negative step function.

The algorithm consists of
  1) many diverse initial seeds,
  2) a continuation of a smooth‑max objective solved with L‑BFGS‑B
     (exact gradient via two FFTs),
  3) a final simulated‑annealing hill‑climber.

The whole pipeline respects the 1000 s time budget and returns a list of
floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# optional, SciPy is part of the allowed libraries
try:
    from scipy.optimize import minimize
except Exception:               # pragma: no cover
    minimize = None

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow_two(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_rev_weight(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of a_rev (reversed a) with weight vector w.
    Returns the full linear convolution (length len(a)+len(w)-1).
    """
    a_rev = a[::-1]
    size = _next_pow_two(len(a_rev) + len(w) - 1)
    fa_rev = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa_rev * fw, size)
    return conv[: len(a_rev) + len(w) - 1]


# ----------------------------------------------------------------------
# Exact objective (independent of scaling)
# ----------------------------------------------------------------------


def _objective_exact(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a on the simplex."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max objective (soft‑max of the convolution)
# ----------------------------------------------------------------------


def _smooth_max_obj(z: np.ndarray, lam: float):
    """
    Soft‑max of the convolution, expressed on the real parameters z
    where a = softmax(z).  Returns (value, gradient w.r.t. z).
    """
    # soft‑max mapping (shift for numerical stability)
    z_shift = z - np.max(z)
    a = np.exp(z_shift)
    a = a / a.sum()

    n = a.shape[0]

    # ----- convolution a * a ------------------------------------------------
    size1 = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(a, size1)
    conv = np.fft.irfft(fa * fa, size1)[: 2 * n - 1]

    max_c = conv.max()
    # log‑sum‑exp of the shifted values – never overflows
    scaled = np.exp(lam * (conv - max_c))
    sum_scaled = scaled.sum()
    f_val = max_c + np.log(sum_scaled) / lam

    # ----- gradient w.r.t. a -------------------------------------------------
    w = scaled / sum_scaled                         # probabilities, sum=1
    conv_aw = _conv_rev_weight(a, w)                # length 3n‑2
    grad_a = 2.0 * conv_aw[n - 1 : n - 1 + n]       # slice of length n

    # ----- gradient w.r.t. the soft‑max parameters -------------------------
    # d a_i / d z_j = a_i (δ_{ij} - a_j)
    inner = np.dot(a, grad_a)                       # scalar
    grad_z = a * (grad_a - inner)

    return f_val, grad_z


# ----------------------------------------------------------------------
# Simple parametric seed: beta‑family sweep
# ----------------------------------------------------------------------


def _beta_family_best(n: int):
    """Return the best member of a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)

    p_vals = np.concatenate(
        [
            np.linspace(0.1, 3.0, 30),
            np.geomspace(3.0, 12.0, 7)[1:],
        ]
    )
    q_vals = p_vals.copy()
    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective_exact(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS optimisation with continuation on λ
# ----------------------------------------------------------------------


def _optimize_from_seed(a0: np.ndarray,
                        lam_schedule,
                        max_iter_per_lambda,
                        deadline,
                        start_time):
    """
    Run a continuation of smooth‑max optimisation.
    a0  – initial feasible point (on the simplex)
    lam_schedule – increasing list of λ values
    max_iter_per_lambda – L‑BFGS‑B iteration budget per λ
    """
    eps = 1e-12
    a0 = np.maximum(a0, eps)
    a0 = a0 / a0.sum()
    z = np.log(a0)        # soft‑max parameters

    best_a = a0.copy()
    best_val = _objective_exact(best_a)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break

        if minimize is None:                     # fall‑back: simple gradient steps
            for _ in range(max_iter_per_lambda):
                f, g = _smooth_max_obj(z, lam)
                step = 0.05 * g
                z = z - step
        else:
            # L‑BFGS‑B on the smooth objective
            res = minimize(
                fun=lambda zz: _smooth_max_obj(zz, lam),
                x0=z,
                jac=True,
                method="L-BFGS-B",
                options={"maxiter": max_iter_per_lambda,
                         "disp": False}
            )
            z = res.x

        # evaluate the *exact* objective for the current iterate
        a = np.exp(z - np.max(z))
        a = a / a.sum()
        val = _objective_exact(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:5.1f}] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climber (unchanged core)
# ----------------------------------------------------------------------


def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Local stochastic optimisation – same code as the reference implementation."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# ----------------------------------------------------------------------
```
