"""
Propose a candidate step‑function (non‑negative list of heights) that
minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))².

The algorithm is a multi‑restart projected Adam optimiser on a smooth
approximation of the max‑convolution, followed by a long entropy‑mirror
descent polishing (exact max).  After polishing a few cheap permutation
heuristics (sorted, interleaved, random) are tried – the best ordering
is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two (different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror descent polishing on the exact max‑entry
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 5000, eta_start: float = 0.55) -> np.ndarray:
    """Entropy‑mirror descent on the exact max‑entry of the self‑convolution."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of the maximal entry
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # decreasing step size
        eta = eta_start * (1.0 - t / steps)
        a *= np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a.fill(1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Helper: simple projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    # implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán (2013) – "Projection onto the probability simplex"
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s time
    budget and performs several random restarts, a final mirror‑descent
    polishing and lightweight permutation heuristics.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.8

    # --------------------------------------------------------------
    # Problem size – 8000 works well and is the size of the previous best
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Helper to initialise a start vector
    # --------------------------------------------------------------
    def _initial_vector(kind: str):
        if kind == "old" and "height_sequence_1" in globals():
            old = np.asarray(height_sequence_1, dtype=float)
            old_len = old.shape[0]
            x_new = np.linspace(0.0, 1.0, n)
            x_old = np.linspace(0.0, 1.0, old_len)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a.fill(1.0 / n)
            else:
                a /= s
            return a
        elif kind == "dirichlet":
            a = np.random.dirichlet(np.ones(n))
            return a
        elif kind == "sorted_dirichlet":
            a = np.random.dirichlet(np.ones(n))
            a.sort()
            a = a[::-1]                     # decreasing
            return a
        else:
            # fallback: smooth symmetric beta(2,2) (same as reference)
            i = np.arange(1, n + 1, dtype=np.float64)
            p = q = 2.0
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            a = a / a.sum()
            return a

    # --------------------------------------------------------------
    # Optimisation core – Adam on a smooth soft‑max of the convolution
    # --------------------------------------------------------------
    def _run_adam(init_a: np.ndarray,
                  max_iters: int,
                  alpha_max: float,
                  lr_start: float,
                  lr_end: float,
                  jitter_after: int = 1500) -> tuple[np.ndarray, float]:
        """
        Run a short Adam optimisation starting from `init_a`.  Returns the
        best vector (as a numpy array) found and its exact objective
        (2·n·maxconv).  The parameters control learning‑rate and smoothing.
        """
        eps = 1e-30
        z = np.log(init_a + eps)                # work in log‑domain
        m = np.zeros_like(z)
        v = np.zeros_like(z)

        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8

        best_a = None
        best_obj = np.inf
        stagnation = 0

        it = 0
        while it < max_iters and time.time() < deadline:
            # ----- learning‑rate & smoothing schedule (per‑iteration) -----
            prog = it / max_iters
            lr = lr_start * (1.0 - prog) + lr_end * prog
            # α grows non‑linearly – a slightly concave schedule works well
            alpha = 1.0 + (alpha_max - 1.0) * (prog ** 1.3)

            # ----- recover a from logits and enforce simplex -----
            a = np.exp(z)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

            # ----- full self‑convolution (needed for soft‑max) -----
            conv = _full_convolution(a)                # length 2n‑1

            # ----- smooth “soft‑max’’ of the convolution -----
            maxc = conv.max()
            exp_scaled = np.exp(alpha * (conv - maxc))
            sumexp = exp_scaled.sum()
            weights = exp_scaled / sumexp               # Σ weights = 1
            log_sumexp = maxc + np.log(sumexp)
            obj_approx = 2.0 * n * (log_sumexp / alpha)   # smoothed proxy

            # ----- keep exact best so far -----
            exact_obj = 2.0 * n * conv.max()
            if exact_obj < best_obj - 1e-12:
                best_obj = exact_obj
                best_a = a.copy()
                stagnation = 0
                print(f"[restart] new local best O={best_obj:.6f} it={it}")
            else:
                stagnation += 1

            # ----- gradient of the soft‑max w.r.t. a -----
            rev_a = a[::-1]
            conv_rev = _full_convolution_two(rev_a, weights)   # length 3n‑2
            g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]           # slice length n

            # ----- chain rule through softmax (a = softmax(z)) -----
            dot = np.dot(g_a, a)
            grad_z = a * (g_a - dot)

            # ----- Adam update on the logits -----
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ----- occasional jitter to escape plateaus -----
            if stagnation >= jitter_after:
                z += np.random.normal(scale=0.03, size=n)
                stagnation = 0
                print(f"[restart] jitter added at it={it}")

        # return the best found in this restart
        if best_a is None:
            best_a = np.exp(z)
            best_a /= best_a.sum()
            best_obj = 2.0 * n * _full_convolution(best_a).max()
        return best_a, best_obj

    # --------------------------------------------------------------
    # Run a few restarts (the first one re‑uses any known solution)
    # --------------------------------------------------------------
    restarts = []
    # 1) try to use the previous construction if available
    restarts.append(("old", 0))
    # 2‑4) three random initialisations
    restarts.extend([("dirichlet", i + 1) for i in range(3)])

    global_best_a = None
    global_best_obj = np.inf

    # parameters for the Adam phase – these are fairly aggressive
    max_iters_per_restart = 2500            # enough iterations for a good local minima
    alpha_max = 2500.0
    lr_start, lr_end = 0.12, 0.004

    for kind, seed_offset in restarts:
        if time.time() >= deadline:
            break
        np.random.seed(seed + seed_offset)
        init_a = _initial_vector(kind)
        best_a, best_obj = _run_adam(init_a,
                                     max_iters=max_iters_per_restart,
                                     alpha_max=alpha_max,
                                     lr_start=lr_start,
                                     lr_end=lr_end,
                                     jitter_after=1200)
        if best_obj < global_best_obj:
            global_best_obj = best_obj
            global_best_a = best_a.copy()
            print(f"[global] improved to O={global_best_obj:.6f} from restart {kind}")

    # --------------------------------------------------------------
    # Polishing phase: entropy‑mirror descent on the exact max entry
    # --------------------------------------------------------------
    if global_best_a is not None:
        a_refined = _max_refine(global_best_a,
                                steps=5000,
                                eta_start=0.58)
        conv_ref = _full_convolution(a_refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < global_best_obj:
            global_best_obj = obj_ref
            global_best_a = a_refined.copy()
            print(f"[polish] improved to O={global_best_obj:.6f}")

    # --------------------------------------------------------------
    # Light permutation heuristics (sorted, interleaved, random)
    # --------------------------------------------------------------
    def _interleaved(v: np.ndarray) -> np.ndarray:
        """Place the largest values alternately at the two ends."""
        v_sorted = np.sort(v)[::-1]          # descending
        res = np.empty_like(v_sorted)
        left, right = 0, len(v_sorted) - 1
        for i, val in enumerate(v_sorted):
            if i % 2 == 0:
                res[left] = val
                left += 1
            else:
                res[right] = val
                right -= 1
        return res

    candidates = []
    # original ordering from the optimiser
    candidates.append(global_best_a)
    # pure descending order
    candidates.append(np.sort(global_best_a)[::-1])
    # interleaved ends
    candidates.append(_interleaved(global_best_a))
    # a few random shuffles
    for _ in range(2):
        perm = np.random.permutation(n)
        candidates.append(global_best_a[perm])

    # evaluate the cheap candidates (exact objective, no rounding)
    best_seq = global_best_a
    best_val = global_best_obj
    for cand in candidates:
        conv = _full_convolution(cand)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[permute] better O={best_val:.6f}")

    # --------------------------------------------------------------
    # Final secure evaluation using the provided wrapper
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx O={best_val:.6f})")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
