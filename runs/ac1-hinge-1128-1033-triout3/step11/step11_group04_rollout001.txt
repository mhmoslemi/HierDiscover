def _opt_hillclimb(a0, n, deadline, start_time, best_a, best_val):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    # parameters
    # maybe use a schedule for max_frac: start high (0.8) decreasing.
    # We'll also compute contributions to max as before: contributions = a[i] * a[idx].

    iter = 0
    no_improve = 0
    IMPROVE_LIMIT = 50000  # param

    while time.time() < deadline:
        # compute max index
        k_max = int(np.argmax(b))
        # compute contributions array
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        # choose source i
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = np.argmax(a)

        # choose target j
        # compute low contribution weights
        # similar to earlier: target weight = 1/(contrib+eps) for those in mask
        eps = 1e-12
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # random other index
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # compute max possible delta: a[i] (cannot go negative)
        max_delta = a[i]  # moving all from source

        if max_delta <= 0:
            continue

        # compute d array for linear part:
        d = np.zeros_like(b)
        d[i:i+n] -= 2.0 * a
        d[j:j+n] += 2.0 * a

        # Precompute shape: store indices for quadratic terms
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j

        # define function f(delta)
        def f(delta):
            # compute max of b + delta * d + delta**2 * c
            # compute temporary array = b + delta*d
            tmp = b + delta * d
            tmp[idx2i] += delta * delta
            tmp[idx2j] += delta * delta
            tmp[idxij] -= delta * delta
            # return max
            return tmp.max()
        # Use ternary search
        lo = 0.0
        hi = max_delta
        # tolerance (e.g., 1e-12 or relative)
        for _ in range(30):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = f(m1)
            f2 = f(m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        delta_opt = (lo + hi) / 2.0
        new_max_val = f(delta_opt)
        new_val = 2.0 * n * new_max_val
        if new_val < cur_val - 1e-15:

            a[i] -= delta_opt
            a[j] += delta_opt

            b[idx2j] += delta_opt * delta_opt
            b[idxij] -= delta_opt * delta_opt
            cur_val = new_val

"""
Improved optimizer for the step‑function attack.

The algorithm proceeds in four phases:

0️⃣  Random seeding: many Beta / Dirichlet draws and a few handcrafted
    shapes (uniform, sinusoid, cosine‑square).  The best vector found
    becomes the seed.

1️⃣  Adam optimisation on a smooth log‑sum‑exp approximation of the
    maximal auto‑correlation (the same as the original code).  This
    quickly drives the solution into a good basin.

2️⃣  **Optimal hill‑climbing** – a new local‑search method.
    In each iteration a “source’’ index that contributes most to the
    current maximal convolution entry and a “target’’ index with low
    contribution are selected.  The exact one‑dimensional minimisation
    of the new maximal convolution value (including the quadratic
    terms) is performed by ternary search, yielding the best possible
    mass transfer `delta` for that pair.  The move is accepted if it
    decreases the objective (or with a tiny Metropolis probability on
    a slowly cooling schedule).

3️⃣  Deterministic sub‑gradient polishing (as before) to clean up any
    remaining spurious spikes.

All phases respect the global 1000 s wall‑clock budget and keep the
best‑ever sequence.  The final result is returned as a plain list of
floats (ℓ₁‑norm = 1, all entries ≥ 0).  The score is verified with
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x*x (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2·n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2·n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2·n‑1
    seg = conv_ab[n - 1:2 * n - 1]                # n entries
    grad_a = 4.0 * n * seg[::-1]                 # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max (no variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam on the log‑softmax parametrisation of a."""
    w = np.log(a0 + 1e-12)          # logits

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 8000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits → probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        _, grad = _smooth_max_and_grad(a, eps)   # already includes 2·n
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)                # chain rule through soft‑max

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)

        # restart on long stagnation
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Optimal hill‑climbing (deterministic delta via ternary search)
# ----------------------------------------------------------------------
def _opt_hillclimb(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Local search that, for each move, computes the *exact* optimal
    amount of mass to transfer from a source index to a target index.
    The optimal `delta` minimises the maximal convolution entry
    (including the quadratic terms) via a cheap ternary search.
    """
    a = a0.copy()
    b = _full_convolution(a)                     # length 2·n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    no_improve = 0
    IMPROVE_LIMIT = 60000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # index of the current maximal convolution element
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index (probability proportional to contribution)
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – favour low contributions
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # linear part of the convolution change for this pair
        d = np.zeros_like(b)
        d[i:i + n] -= 2.0 * a
        d[j:j + n] += 2.0 * a

        # quadratic‑term locations
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j

        # function returning the maximal convolution entry after moving delta
        def max_after(delta: float) -> float:
            tmp = b + delta * d
            # add the three quadratic corrections
            tmp[idx2i] += delta * delta
            tmp[idx2j] += delta * delta
            tmp[idxij] -= delta * delta
            return tmp.max()

        # ternary search on delta ∈ [0, a[i]]
        lo, hi = 0.0, a[i]
        for _ in range(30):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = max_after(m1)
            f2 = max_after(m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        delta_opt = (lo + hi) / 2.0
        new_max = max_after(delta_opt)
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            # occasional Metropolis acceptance with a very low temperature
            prog = (now - start_time) / (deadline - start_time)
            T = 0.02 * (0.001 ** prog)
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # apply the move (incremental update of b)
            a[i] -= delta_opt
            a[j] += delta_opt
            b = b + delta_opt * d
            b[idx2i] += delta_opt * delta_opt
            b[idx2j] += delta_opt * delta_opt
            b[idxij] -= delta_opt * delta_opt
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[OptHill] new best = {best_val:.6f}")

        else:
            no_improve += 1

        # stagnation handling – perturb around the best found so far
        if no_improve > IMPROVE_LIMIT:
            print(f"[OptHill] stagnation, re‑perturb (it={iteration})")
            a = best_a.copy()
            a += np.random.normal(scale=0.015, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val

# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combos t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta family candidate
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Beta‑shaped distribution (log‑uniform parameters)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a
