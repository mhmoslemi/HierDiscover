The search works in three stages:

1. **Smart initialisation** – a collection of parametric families

   together with a few random Dirichlet draws are evaluated and the
   best one is kept as a starting point.

2. **Projected smooth‑max descent** – we minimise a smooth approximation
   of the ‑max of the convolution:
   ```
   f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
   ```
   The gradient is obtained analytically via FFT and we use a
   back‑tracking line‑search with Euclidean projection onto the
   probability simplex.  The temperature ‘α’ is increased linearly

   entry and move a small amount of probability mass from the most
   “guilty’’ index (largest contribution to that lag) to a “safe’’

   analytically to guarantee a decrease of the exact objective.
   A short random jitter is added if progress stalls.

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (real inputs)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself, length 2·len(x)‑1."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_rev(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient helper: computes 2·∑_i a_i·w_{i+j} for j=0..n‑1.
    Implements the gradient of the soft‑max surrogate.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    l_w = w.shape[0]
    out_len = n + l_w - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(a*a) (sum is assumed >0)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(α·conv_k)."""
    maxc = conv.max()
    exp_vals = np.exp(alpha * (conv - maxc))
    s = exp_vals.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx = 1} (Euclidean)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric families for initial guesses (more diverse than the
# reference implementation)
# ----------------------------------------------------------------------
def _parametric_candidates(n: int):
    """Yield a selection of hand‑crafted shapes."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Uniform (always useful as a baseline)
    a = np.full(n, 1.0 / n)
    yield a

    # Power‑law / Beta families (α,β ranging independently)
    for α, β in ((0.5, 0.5), (0.6, 0.6), (0.8, 0.8),
                 (1.0, 2.0), (2.0, 1.0), (1.5, 2.5), (2.5, 1.5), (3.0, 3.0)):
        a = (i + 1) ** (α - 1.0) * (n - i) ** (β - 1.0)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Exponential / Laplace families
    for lam in (0.0005, 0.001, 0.002, 0.004, 0.008):
        a = np.exp(-lam * i)
        a /= a.sum()
        yield a

    # Symmetric sine‑square shape (looks like a discrete “bump”)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    a /= a.sum()
    yield a

    # Cosine‑square (complementary shape)
    a = np.cos(np.pi * (i + 0.5) / n) ** 2
    a /= a.sum()
    yield a

    # Mixed Gaussian/Laplace (different σ)
    for sigma_factor in (0.12, 0.18, 0.24):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        a /= a.sum()
        yield a

# Small stochastic restarts (Dirichlet draws with different concentration)
# ----------------------------------------------------------------------
def _random_dirichlet_candidates(n: int, rng: np.random.Generator):
    """Yield a few random probability vectors."""
    for conc in (0.2, 0.5, 1.0, 2.0):
        cand = rng.dirichlet(np.full(n, conc))
        yield cand


# ----------------------------------------------------------------------
# Smooth‑max projected gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------
def _smoothmax_descent(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       alpha_start: float = 5.0,
                       alpha_end: float = 2000.0,
                       init_step: float = 0.5,
                       step_decay: float = 0.5,
                       min_step: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth‑max surrogate.
    Returns the best sequence seen.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # enforce simplex

    best_a = a.copy()
    best_val = _objective(a)

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear schedule for temperature α
        prog = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        alpha = alpha_start + (alpha_end - alpha_start) * prog

        conv = _full_convolution(a)
        w = _softmax_weights(conv, alpha)
        grad = 2.0 * _conv_rev(a, w)                 # shape (n,)

        # --- back‑tracking line search ---------------------------------
        step = init_step
        while step >= min_step:
            a_cand = a - step * grad
            a_cand = np.maximum(a_cand, 0.0)
            a_cand = _project_simplex(a_cand)
            cand_val = _objective(a_cand)
            if cand_val <= best_val:          # monotone acceptance
                break
            step *= step_decay

        if step < min_step:                  # no progress – add a small jitter
            a_cand = a + np.random.normal(scale=0.001, size=a.shape)
            a_cand = np.maximum(a_cand, 0.0)
            a_cand = _project_simplex(a_cand)
            cand_val = _objective(a_cand)

        a = a_cand
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_a = a.copy()
            print(f"[Smooth] iter {t:7d} α={alpha:6.1f} step={step:.2e} val={best_val:.6f}")

            # early stop if we are already below the target
            if best_val <= 1.503:
                break

        t += 1

        # occasional random reset around the current best to escape plateaus
        if t % 4000 == 0:
            a = best_a + np.random.normal(scale=0.005 * (1.0 - prog), size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic hill‑climb that moves mass from the largest contribution
# to the maximum‑lag towards a position that does not worsen that lag.
# ----------------------------------------------------------------------
def _polish_hillclimb(a: np.ndarray,
                      deadline: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simple deterministic hill‑climb with analytically chosen δ.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    # pre‑compute index sets for each lag to speed up contribution look‑ups
    # (list of arrays of valid i for each k)
    idx_for_k = [np.arange(max(0, k - (n - 1)), min(n - 1, k) + 1) for k in range(2 * n - 1)]

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val <= target_val:
            best_a = a.copy()
            best_val = cur_val
            break

        k_max = int(np.argmax(conv))

        # contributions for this lag
        idx = idx_for_k[k_max]
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() == 0.0:            # degenerate – break
            break

        # source index – a heavy contributor
        src = idx[np.argmax(contrib)]
        # target index – ideally with the smallest contribution to this lag
        # (avoid idx that would increase the lag)
        # we look for any position not participating in k_max
        not_in_lag = np.setdiff1d(np.arange(n), idx, assume_unique=True)
        if not_in_lag.size == 0:
            # fallback: choose the index with smallest a value overall
            tgt = int(np.argmin(a))
        else:
            tgt = int(not_in_lag[np.argmin(a[not_in_lag])])

        if a[src] < 1e-12:                 # nothing to move
            break

        # Choose δ so that the contribution of src to the offending lag
        # is reduced to the level of the current minimum contribution.
        # Contribution of src is a[src] * a[k_max - src] (if partner exists)
        partner = k_max - src
        partner_val = a[partner] if 0 <= partner < n else 0.0
        src_contrib = a[src] * partner_val

        # Minimal contribution we would like to attain (use the current min)
        min_contrib = contrib.min()
        # Desired new a[src] to achieve src_contrib = min_contrib:
        # (a[src] - δ) * partner_val = min_contrib   →   δ = a[src] - min_contrib / partner_val
        # Guard against division by zero.
        if partner_val > 0:
            δ = a[src] - min_contrib / partner_val
            δ = max(0.0, min(δ, a[src] * 0.5))   # move at most half the mass
        else:
            # if partner is zero, any reduction only helps via self‑square;
            # just move a modest fraction
            δ = a[src] * 0.25

        if δ < 1e-12:
            # fall back to a tiny random move
            δ = a[src] * np.random.rand() * 0.01

        # apply the move
        a_new = a.copy()
        a_new[src] -= δ
        a_new[tgt] += δ

        new_val = _objective(a_new)
        if new_val < cur_val:
            a = a_new
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {it:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        # if no improvement, try a small random jitter (prevent stagnation)
        elif it % 2000 == 0:
            jitter = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

        it += 1

    return best_a, best_val

def _interpolate_previous(prev_seq, n):
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None
