"""
Multi‑phase optimizer for the step‑function “max‑convolution” problem.

The algorithm combines several complementary heuristics:

0️⃣  Randomised “beta/Dirichlet” seeds and a few handcrafted forms (uniform,
    sinusoidal, triangular …).  The best of these seeds seeds the main loop.

1️⃣  Adam on a smooth log‑sum‑exp approximation of the max‑convolution.
    The smoothing ε is annealed to ~1e‑9 and the learning‑rate is annealed
    from 0.2 → 1e‑3.  Variables are stored as logits of a soft‑max so the
    ℓ₁‑simplex (Σa=1, a≥0) is respected automatically.

2️⃣  Aggressive hill‑climbing: repeatedly move a random fraction of the mass
    from a “source” index (large contribution to the current maximal
    convolution entry) to a “target” index (small contribution).  A
    Metropolis acceptance rule with a cooling temperature lets the search
    escape plateaus.

3️⃣  Linear‑programming refinement (optional): given the current shape
    `f`, solve a small LP that finds a new distribution `g` with the same
    ℓ₁‑norm but respects the same maximal convolution bound on a handful
    of “active” constraints (the largest convolution entries).  A convex
    combination `f←(1−α)f+αg` typically reduces the peak.

4️⃣  Exact sub‑gradient polishing of the true objective
    `F(a)=2·n·max(conv(a,a))`.  The step‑size decays linearly from 0.30 to
    1e‑5.

5️⃣  Final simulated‑annealing moves with larger step‑sizes to use any
    remaining time budget.

All phases respect the 1000 s wall‑clock limit.  The best sequence found
is returned as a plain list of floats (ℓ₁‑norm ≈ 1, non‑negative), which
can directly be passed to the supplied `evaluate_sequence` function.
"""

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the 2·n factor)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.  The gradient already includes the outer
    factor 2·n.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for the loss L(a)=2·n·smooth_max(a).
    The smoothing ε is annealed from a relatively large value to 1e‑9.
    """
    n = a0.shape[0]
    # work in log‑softmax space (logits)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 8000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule learning rate and smoothing
        progress = (now - (deadline - (deadline - (deadline - (deadline - (deadline - deadline))))) ) / (deadline - (deadline - (deadline - (deadline - (deadline - deadline)))))
        # The above messy expression is a trick to avoid using a global start_time.
        # We'll recompute progress using the elapsed time from the moment this function was entered.
        # Since we don't have start_time here, we'll estimate progress from idle: not perfect but sufficient.
        # A simpler approach is to use a linear decay based on time remaining.
        remaining = max(deadline - now, 1e-12)
        total = max(deadline - (deadline - 1e-12), 1e-12)  # approx 0 just to avoid zero division
        prog = 1.0 - remaining / (deadline - (deadline - 1.0))  # placeholder

        # Simpler schedule (ignoring exact start_time):
        lr = lr_start * (lr_end / lr_start) ** (idle / MAX_IDLE)
        eps = eps_start * (eps_end / eps_start) ** (idle / MAX_IDLE)

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # grad already contains 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to avoid stagnation
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.02, size=n)

        # restart if idle too long
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing with simulated‑annealing acceptance
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               deadline: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly moves a small chunk of mass from a
    “high‑contribution’’ index to a “low‑contribution’’ index, with a
    simulated‑annealing acceptance rule.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules (more gradual than the original code)
    max_frac_start, max_frac_end = 0.8, 0.001
    temp_start, temp_end = 0.03, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 90000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - (deadline - (deadline - (deadline - (deadline - deadline))))) / (deadline - (deadline - (deadline - deadline)))
        # Use a simple linear schedule based on remaining time
        remaining = max(deadline - now, 1e-12)
        total = max(deadline - (deadline - 1.0), 1e-12)
        prog = 1.0 - remaining / total

        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index: proportional to its contribution
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index: favour low contributions (inverse weighting)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill] stagnation (it={iteration}), re‑perturb")
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The step size decays linearly from 0.30 to 1e‑5.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        # we estimate progress from iteration count
        prog = it / max_iters
        step = 0.30 * (1.0 - prog) + 1e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to break flat regions
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray, cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    n = a.shape[0]
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta family candidate
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Deterministic handcrafted candidates (triangular, cosine‑square, etc.)
# ----------------------------------------------------------------------
def _handcrafted_candidates(n: int) -> list[np.ndarray]:
    """Return a list of a few deterministic shapes."""
    cand_list = []

    # triangular (increase then decrease)
    i = np.arange(n, dtype=np.float64)
    tri = np.minimum(i, n - 1 - i) + 1
    tri = tri.astype(float)
    tri /= tri.sum()
    cand_list.append(tri)

    # sinusoid (positive half‑wave)
    sin = np.sin(np.pi * (i + 0.5) / n)
    sin = np.maximum(sin, 0.0)
    sin /= sin.sum()
    cand_list.append(sin)

    # cosine‑square (peaked in the middle)
    cos2 = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    cos2 = np.maximum(cos2, 0.0)
    cos2 /= cos2.sum()
    cand_list.append(cos2)

    # Beta(0.5,0.5) – the classic arcsine shape
    p = q = 0.5
    i1 = np.arange(1, n + 1, dtype=np.float64)
    beta05 = (i1 ** (p - 1.0)) * ((n - i1 + 1) ** (q - 1.0))
    beta05 /= beta05.sum()
    cand_list.append(beta05)

    return cand_list


# ----------------------------------------------------------------------
# LP refinement phase (optional, requires CVXPY)
# ----------------------------------------------------------------------
def _lp_refine(f0: np.ndarray,
               deadline: float,
               best_a: np.ndarray,
               best_val: float,
               active_set_size: int = 20,
               mix_alpha: float = 0.2,
               max_iterations: int = 15) -> tuple[np.ndarray, float]:
    """
    Given a distribution f, solve a small LP that seeks another distribution g
    (still on the simplex) with the same maximum convolution bound on a set
    of active constraints (the largest entries of conv(f,f)).  The new shape
    is mixed with the old one: f←(1−α)·f+α·g.
    """
    if not _CVXPY_AVAILABLE:
        print("[LP] CVXPY not available – skipping LP refinement.")
        return best_a, best_val

    n = f0.shape[0]
    f = f0.copy()

    for it in range(max_iterations):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(f)
        M = b.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val, best_a = cur_val, f.copy()
            print(f"[LP] new best after iteration {it} = {best_val:.6f}")

        # select active indices (largest entries of b)
        # we take the top `active_set_size` entries regardless of bitness
        topk = min(active_set_size, b.size)
        active_idx = np.argpartition(-b, topk - 1)[:topk]

        # Build the constraint matrix B where (B @ g)_k = Σ_i f_i g_{k-i}
        rows = len(active_idx)
        B = np.zeros((rows, n))
        f_arr = f  # alias for speed
        for row, k in enumerate(active_idx):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            j = np.arange(j_start, j_end + 1, dtype=int)
            B[row, j] = f_arr[k - j]   # coefficient of g_j is f_{k-j}

        # Define and solve the LP:
        g = cp.Variable(n, nonneg=True)
        constraints = [B @ g <= M * np.ones(rows), cp.sum(g) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, verbose=False, warm_start=True, max_iters=1000)
        except Exception as e:  # pragma: no cover
            print("[LP] solver exception:", e)
            break

        if g.value is None:
            print("[LP] solver failed to produce a solution – aborting LP phase.")
            break

        g_vec = np.maximum(np.array(g.value, dtype=float), 0.0)
        g_vec = _project_simplex(g_vec)   # ensure exact sum=1

        # convex combination
        f = (1.0 - mix_alpha) * f + mix_alpha * g_vec
        f = _project_simplex(f)  # numerical safety

    return best_a, best_val


# ----------------------------------------------------------------------
# Final simulated‑annealing phase (large moves)
# ----------------------------------------------------------------------
def _random_anneal(a0: np.ndarray,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing loop that performs random source→target
    transfers with a relatively large step size.  It runs only on the
    leftover part of the time budget.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    T0 = 0.02

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - (deadline - (deadline - deadline))) / (deadline - (deadline - 1.0))
        T = T0 * (1.0 - prog)            # linear cooling
        max_frac = 0.15 * (1.0 - prog) + 0.001 * prog

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        # choose j != i uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # fast convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")

        # occasional jitter to keep diversity
        if rng.random() < 0.01:
            a = np.maximum(a + rng.normal(scale=0.008, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point (required by the competition harness)
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is (approximately) 1 and that (tries to)
    minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # problem size – we keep 8000 as a sweet spot between fidelity and speed.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initial random / handcrafted candidates
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value = {best_val:.6f}")

    # try to reuse a previously discovered solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value = {best_val:.6f}")
    except Exception:
        pass

    # many random Beta draws
    for _ in range(3000):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Dirichlet draws with a range of concentrations
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(48):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # handcrafted shapes
    for cand in _handcrafted_candidates(n):
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] best after random / handcrafted search = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimiser on a smooth approx. (≈15 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.15 * budget_s
    best_a, best_val = _adam_optimize(
        best_a,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ------------------------------------------------------------------
    # Phase B – Aggressive hill‑climbing (≈30 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _hillclimb(
        best_a,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ------------------------------------------------------------------
    # Phase C – LP refinement (≈15 % of budget, optional)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.60 * budget_s
    best_a, best_val = _lp_refine(
        best_a,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        active_set_size=15,
        mix_alpha=0.25,
        max_iterations=12,
    )
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ------------------------------------------------------------------
    # Phase D – Sub‑gradient polishing (≈25 % of the remaining time)
    # ------------------------------------------------------------------
    phase_d_deadline = start_time + 0.90 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a,
        deadline=phase_d_deadline,
        best_a=best_a,
        best_val=best_val,
        max_iters=400_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ------------------------------------------------------------------
    # Phase E – Final simulated‑annealing with larger moves (remaining)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_anneal(
            best_a,
            deadline=deadline,
            best_a=best_a,
            best_val=best_val,
            rng=rng,
        )
        best_a, best_val = _blend_with_uniform(best_a, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # return a plain list of floats
    return best_a.tolist()
