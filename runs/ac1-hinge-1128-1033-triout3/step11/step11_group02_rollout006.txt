"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm
---------
1. **Initial seeding** – generate a few candidate shapes:
   * interpolate a previous solution (if present),
   * exhaustively sweep the two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1},
   * a handful of random Dirichlet draws.
   The best seed (by the exact objective 2·n·max(conv(a,a))) is kept.

2. **Linear‑programming refinement** – given a current candidate f,
   * compute M = max(conv(f,f)).
   * Solve the linear program      max Σ g_i   s.t.   conv(f,g) ≤ M ,  g ≥ ε .
     The constraints are linear because conv(f,g) is a linear map in g.
   * If the LP yields Σ g_i  > 1, scale g to unit ℓ₁‑norm and mix:
        f ← (1‑t)·f + t·g   (t starts at 0.5, is halved until an improvement
        in the objective is observed).
   * Repeat the LP–mix step until no improvement or the time budget is
     exhausted.

3. **Local polishing** – run the simulated‑annealing hill‑climber from the
   reference implementation on the remaining time.

The whole procedure respects the 1000 s budget and returns a plain list of
floats that can be fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x * x (length 2·len(x)‑1)."""
    return np.convolve(x, x, mode='full')


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (used to keep a strictly positive)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Euclidean projection of v onto {x≥eps, Σx=1}."""
    v = np.maximum(v, eps)
    # Standard simplex projection then shift to respect eps
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > eps
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, eps)
    w /= w.sum()            # ensure exact sum = 1
    return w


# ----------------------------------------------------------------------
# Beta‑family sweep – a slightly richer parameter grid than the reference
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray | None, float]:
    """Search the Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf

    # grids: very small exponents (U‑shaped), moderate, and heavy tails
    p_vals = np.concatenate([
        np.geomspace(0.01, 0.5, 12, endpoint=False),   # 0.01 … 0.44
        np.linspace(0.5, 3.0, 25, endpoint=False),    # 0.5 … 2.9
        np.geomspace(3.0, 30.0, 10)                   # 3 … 30
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Build the (2n‑1)×n convolution matrix of a fixed vector f:
#   (conv(f,g))[k] = Σ_i f_i · g_{k‑i}
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the dense matrix A with rows = conv(f,·) (shape (2n‑1, n))."""
    n = f.shape[0]
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.empty(n, dtype=f.dtype)
    row[:] = 0.0
    row[0] = f[0]
    return toeplitz(col, row)


# ----------------------------------------------------------------------
# Linear‑programming refinement step
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               max_iters: int,
               deadline: float,
               eps: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Perform up to ``max_iters`` LP‑based refinement steps.
    Returns the best sequence and its objective value.
    """
    n = f.shape[0]
    best_a = f.copy()
    best_val = _objective(best_a)

    while max_iters > 0 and time.time() < deadline:
        # 1) compute current maximum
        conv_ff = _full_convolution(best_a)
        M = conv_ff.max()

        # 2) build LP: max Σ g_i   s.t.   A·g ≤ M ,  g ≥ eps
        A = _conv_matrix(best_a)                     # (2n‑1) × n
        b = np.full(A.shape[0], M * (1.0 + 1e-12))   # tiny slack for numerics
        c = -np.ones(n)                              # maximise Σg ⇔ minimise -Σg
        bounds = [(eps, 1000.0) for _ in range(n)]

        lp_res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                         method='highs', options={'presolve': True})

        if not lp_res.success:
            # LP failed – fall back to exiting the refinement loop
            print("[LP] Linprog failed, status:", lp_res.message)
            break

        g0 = lp_res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1.0 + 1e-12:
            # No increase in ℓ₁‑norm → cannot improve further
            print("[LP] No ℓ₁‑norm improvement (Σg = {:.6f})".format(sum_g0))
            break

        # Scale to unit sum
        g = g0 / sum_g0

        # 3) Mixing – try several step‑sizes t
        improvement = False
        t = 0.5
        for _ in range(6):      # at most six halvings
            a_candidate = (1.0 - t) * best_a + t * g
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()      # enforce simplex
            val_cand = _objective(a_candidate)

            if val_cand < best_val - 1e-12:
                best_a = a_candidate
                best_val = val_cand
                improvement = True
                print(f"[LP] improved to {best_val:.6f} (t={t:.3f})")
                break
            else:
                t *= 0.5

        if not improvement:
            # mixing did not yield a better objective → stop
            print("[LP] Mixing failed to improve; exiting refinement.")
            break

        max_iters -= 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing phase – identical to the reference implementation
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # Update convolution efficiently
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val
