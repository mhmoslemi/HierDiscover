"""
Propose a step‑function sequence that minimises the float value returned by
``evaluate_sequence``.

Algorithm
---------
1. **Small‑scale exploration** – work first with a modest length (n≈2000)
   where many cheap iterations are possible.
   * Initialise from several random Dirichlet draws and from the
     previously‑found solution (if any).
   * Run a pure *sub‑gradient* descent on  `max_k (a*a)[k]` :
        • each iteration computes the convolution,
        • identifies the lag `k_max` where it is largest,
        • the sub‑gradient of the max‑convolution is
          `g_j = 2·a_{k_max-j}` (zero if the index is out of range),
        • a projected‑simplex step with a decaying step‑size is taken.
   * Keep the best (lowest objective) shape found.

2. **Upsample** – linearly interpolate the best small‑scale shape to a
   larger size (n≈8000).  Re‑normalise to a probability vector.

3. **Large‑scale refinement** – run the same sub‑gradient optimiser on the
   up‑sampled vector.  After a few thousand iterations add occasional
   random jitter to escape plateaus.

4. **Final polishing** – a fast hill‑climbing “mass‑transfer’’ phase:
   repeatedly move a small random fraction of mass from a heavy
   contributor to a light contributor for the lag that currently attains
   the maximal convolution value.  Accept only improving moves; after a
   streak without improvement inject a small Gaussian perturbation.

The whole procedure respects the 1000 s budget and returns the best
sequence discovered.  All numbers are non‑negative and the shape is
projected onto the probability simplex, guaranteeing a finite result
from ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)-1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective helper (exact, scale‑invariant)
# ----------------------------------------------------------------------
def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact value 2·n·max(a∗a) given the full convolution."""
    return float(2.0 * n * conv.max())


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (assumed non‑negative, sum>0)."""
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.shape[0])


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx = 1} (Euclidean)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Returns the sub‑gradient of  max_k (a*a)[k]  and the index where the
    maximum is attained.
    g_j = 2 * a_{k_max-j}  (zero when the index is out of range).
    """
    n = a.shape[0]
    k = int(np.argmax(conv))

    # indices i = k - j
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[idx[mask]]
    return grad, k


# ----------------------------------------------------------------------
# Sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          gamma0: float = 0.3,
                          decay: str = "sqrt") -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on  max_k (a*a)[k].
    Returns the best sequence (sum=1) and its exact objective.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    cur_conv = _full_convolution(a)
    cur_val = _objective_from_conv(cur_conv, n)

    best_a = a.copy()
    best_val = cur_val

    t = 0
    while True:
        if time.time() >= deadline:
            break
        t += 1

        # sub‑gradient at the current maximal lag
        grad, _ = _max_conv_gradient(a, cur_conv)

        # decaying step size
        if decay == "sqrt":
            step = gamma0 / math.sqrt(t)
        elif decay == "log":
            step = gamma0 / (1.0 + math.log(t + 1.0))
        else:  # constant
            step = gamma0

        # gradient step + projection
        a_candidate = a - step * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        cand_conv = _full_convolution(a_candidate)
        cand_val = _objective_from_conv(cand_conv, n)

        # accept only non‑worsening moves
        if cand_val <= cur_val + 1e-12:
            a = a_candidate
            cur_conv = cand_conv
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[SubGrad] iter {t:7d} val={best_val:.6f}")
                # early stop if we have already reached the target region
                if best_val <= 1.51:
                    break
        else:
            # occasional tiny acceptance to provide stochasticity
            if np.random.rand() < 0.001:
                a = a_candidate
                cur_conv = cand_conv
                cur_val = cand_val

        # random jitter every few thousand iterations to shake off plateaus
        if t % 3000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_conv = _full_convolution(a)
            cur_val = _objective_from_conv(cur_conv, n)

    return best_a, best_val

def _parametric_candidates(n: int):

    i = np.arange(n, dtype=np.float64)

    # Gaussian families
    for sigma_factor in (0.18, 0.22, 0.26, 0.30):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Laplace families
    for scale_factor in (0.12, 0.16, 0.20):
        scale = scale_factor * n
        a = np.exp(-np.abs(i - mid) / scale)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Power‑law / Beta‑like families
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = (i + 1) ** (p - 1.0) * (n - i) ** (p - 1.0)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Sine‑squared families (often produce flatter autocorrelation)
    for p in (0.5, 1.0, 2.0, 4.0):
        a = np.sin(np.pi * (i + 0.5) / n) ** p
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            a /= a.sum()
            yield a


# ----------------------------------------------------------------------
# Interpolation of a previously found sequence (if present)
# ----------------------------------------------------------------------
def _interpolate_previous(old_seq, n):
    try:
        old = np.asarray(old_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None


# ----------------------------------------------------------------------
# Light‑weight hill‑climbing (mass‑transfer) for final polishing
# ----------------------------------------------------------------------
def _final_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that moves a fraction of mass from a heavy
    contributor (to the current maximal lag) to a light contributor.
    Stops early if the target value is reached.
    """
    b = _full_convolution(a)
    cur_val = _objective_from_conv(b, n)
    best_val = cur_val
    best_a = a.copy()

    no_improve = 0
    IMPROVE_LIMIT = 140_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # locate the lag achieving the maximal autocorrelation
        k_max = int(np.argmax(b))

        # contributors to this lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # pick a source index (prefer heavy contributors)
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-14:
            continue

        # pick a target index (avoid the current heavy contributor)
        inv = contrib.max() - contrib + 1e-14
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        # move a random fraction (up to half) of the source mass
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-14:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = _objective_from_conv(b_new, n)

        if new_val < cur_val - 1e-15:
            # accept the move
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # random jitter around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = _objective_from_conv(b, n)
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # Initial problem size for cheap exploration
    n_small = 2000
    global_best_a = None
    global_best_val = np.inf
    global_best_n = None

    # ------------------------------------------------------------------
    # 1️⃣  Small‑scale exploration (many cheap iterations)
    # ------------------------------------------------------------------
    # possibly reuse a previously reported solution
    previous_seq = None
    if 'height_sequence_1' in globals():
        try:
            previous_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            previous_seq = None

    for restart in range(12):
        if time.time() >= deadline:
            break

        # initialise candidate
        if restart == 0 and previous_seq is not None:
            a0 = _interpolate_previous(previous_seq, n_small)
            if a0 is None:
                a0 = np.random.dirichlet(np.ones(n_small))
        elif restart % 3 == 0:
            # draw from a handful of parametric families
            cand_gen = _parametric_candidates(n_small)
            try:
                a0 = next(cand_gen)
            except StopIteration:
                a0 = np.random.dirichlet(np.ones(n_small))
        else:
            a0 = np.random.dirichlet(np.ones(n_small))

        a_opt, val_opt = _subgradient_optimize(
            a0,
            n_small,
            deadline,
            start_time,
            gamma0=0.3,
            decay="sqrt",
        )

        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_a = a_opt.copy()
            global_best_n = n_small
            print(f"[Phase‑small] new best n={n_small} val={global_best_val:.6f}")

        # early exit if we already beat the target region
        if global_best_val <= 1.505:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Upsample and refine at a larger, more realistic size
    # ------------------------------------------------------------------
    target_n = 8000
    if global_best_a is not None:
        a_large = _interpolate_previous(global_best_a, target_n)
        a_large = np.maximum(a_large, 0.0)
        a_large /= a_large.sum()
        a_opt_large, val_opt_large = _subgradient_optimize(
            a_large,
            target_n,
            deadline,
            start_time,
            gamma0=0.24,
            decay="sqrt",
        )
        if val_opt_large < global_best_val:
            global_best_val = val_opt_large
            global_best_a = a_opt_large.copy()
            global_best_n = target_n
            print(f"[Phase‑large] refined n={target_n} val={global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Additional random restarts at the large size (time‑boxed)
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        n = target_n
        a0 = np.random.dirichlet(np.ones(n))
        a_opt, val_opt = _subgradient_optimize(
            a0,
            n,
            deadline,
            start_time,
            gamma0=0.2,
            decay="sqrt",
        )
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_a = a_opt.copy()
            global_best_n = n
            print(f"[Random‑large] new best val={global_best_val:.6f}")
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing with hill‑climbing
    # ------------------------------------------------------------------
    if global_best_a is not None:
        remaining = deadline - time.time()
        if remaining > 0.6:
            global_best_a, global_best_val = _final_hillclimb(
                global_best_a,
                global_best_n,
                deadline=deadline,
                start_time=start_time,
                target_val=1.503,
            )
            print(f"[Final‑hill] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Return the best sequence (evaluate_sequence is called by the host)
    # ------------------------------------------------------------------
    if global_best_a is None:
        # fallback: uniform distribution of a moderate size
        n = target_n
        global_best_a = np.full(n, 1.0 / n)
        global_best_val = _objective(global_best_a)

    # optional logging of the final score
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
