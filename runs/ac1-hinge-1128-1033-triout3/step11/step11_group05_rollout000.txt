"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Initialise a probability vector a on the simplex (length n = 8000).
   If a previous solution ``height_sequence_1`` exists it is interpolated;
   otherwise we start from a uniform distribution.

2. Optimise a via gradient descent on a *smooth* surrogate of the
   objective  f(a)=2·n·max(conv(a,a)).
   The surrogate is the log‑sum‑exp of the convolution entries:
        g_α(a) = (1/α)·log Σ_k exp(α·b_k) ,  b = conv(a,a).
   For α → ∞, g_α → max(b).  Using a moderate α gives a differentiable
   objective whose gradient is a convex combination of the sub‑gradients
   of the individual convolution entries.

   • a is parametrised as a soft‑max of real logits y, guaranteeing
     positivity and Σa = 1.
   • The gradient w.r.t. y is obtained analytically:
        ∂g/∂a = Σ_k w_k ∂b_k/∂a ,   w_k ∝ exp(α·b_k)
        ∂b_k/∂a_i = a_{k-i} + a_{k-i}  (both directions) → 
        grad_a = 2·conv(w , a_rev)[n‑1 : n‑1+n]   (a_rev = a[::-1])
   • Adam updates are applied to y.
   • α is annealed from a small value (≈5) to a larger one (≈500) so that
     the optimisation first walks on a smooth landscape and later focuses
     on the true max.

3. Throughout the run the exact (non‑smoothed) objective
   2·n·max(conv(a,a)) is evaluated; the best sequence encountered is kept.

4. After the time budget expires the best sequence is handed to
   ``evaluate_sequence`` for the final (secure) score and returned.

The whole routine respects the supplied budget (default 1000 s) and uses
only NumPy – no external solvers or I/O.  The optimiser is fully
vectorised and runs in ≈ few milliseconds per iteration, allowing a
large number of updates within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of the two 1‑D real vectors using FFT.
    Result length = len(x) + len(y) - 1.
    """
    m = len(x) + len(y) - 1
    size = 1 << (m - 1).bit_length()        # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:m]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """
    Exact evaluation of 2·n·max(conv(a,a)) assuming Σa = 1.
    Used for quick comparisons (avoids the final security checks).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a, a)
    return float(2.0 * n * b.max())

# Core optimiser
# ----------------------------------------------------------------------

    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    # ------------------------------

    try:

            a0 /= a0.sum()
        else:
            a0 = old

    except Exception:

        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        print("[init] uniform start")

    # Logits for the soft‑max parametrisation
    y = np.log(a0 + 1e-12)

    v = np.zeros_like(y)

    # --------------------------------------------------------------

    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_seq.tolist()
