"""
Hybrid optimizer for the “max‑convolution” objective.

The objective for a non‑negative probability vector `a` (sum(a)=1) is

    F(a) = 2·n·max(conv(a,a))

where `conv` is the full linear convolution.
The function is non‑convex and has many plateaus.  We use a two‑stage
strategy:

1.  A smooth Adam optimisation that minimises a soft‑max
    approximation of the max‑entry.  The smoothing parameter α is
    increased from 1 up to a large value (≈800) while the learning
    rate is annealed.

2.  An entropic mirror‑descent refinement that works on the *exact*
    max‑entry.  In each step we move probability mass away from the
    indices that dominate the current maximum (the top‑k entries of the
    convolution), using a multiplicative update
    a_i ← a_i·exp(‑η·g_i) and re‑normalising.  The step‑size η decays
    linearly to 0.

Both phases run as long as the supplied time budget allows; the best
sequence found so far is stored and finally returned.  The routine
respects the 1000 s budget, prints a few progress messages and works
without any file or network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full convolutions (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two (possibly different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropic mirror‑descent refinement (exact max, top‑k sub‑gradient)
# ----------------------------------------------------------------------
def _max_refine_multi(a: np.ndarray,
                      top_k: int = 5,
                      eta_start: float = 0.45,
                      time_deadline: float = None) -> np.ndarray:
    """
    Entropic mirror descent on the exact max‑entry of `conv(a,a)`.
    In each iteration we sum the sub‑gradients of the `top_k` largest
    convolution entries, then perform a multiplicative update.
    The step‑size η decays linearly from `eta_start` to 0.
    The loop stops when the time budget ends (if `time_deadline` is set)
    or after a default of 20000 iterations.
    """
    n = a.shape[0]
    a = a.copy()
    steps = 20000
    t = 0
    while True:
        if time_deadline is not None and time.time() >= time_deadline:
            break
        if t >= steps:
            break
        conv = _full_convolution(a)
        # indices of the largest entries
        if top_k == 1:
            ks = [int(conv.argmax())]
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        # accumulate sub‑gradient of the selected entries
        g = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            if lo <= hi:
                idx = np.arange(lo, hi + 1, dtype=int)
                g[idx] += 2.0 * a[k - idx]
        # linearly decaying step size
        prog = t / steps
        eta = eta_start * (1.0 - prog)
        # multiplicative (entropy‑mirror) update
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a.fill(1.0 / n)
        else:
            a /= s
        t += 1
    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,

    """

    # --------------------------------------------------------------

    v = np.zeros_like(z)

    best_a = None

        maxc = conv.max()

        # smooth approximation of the true objective (sum(a)=1)
        obj_smooth = 2.0 * n * (maxc + np.log(sumexp) / alpha)

        # Exact objective for reporting
        exact_obj = 2.0 * n * conv.max()

        if exact_obj < best_obj - 1e-12:

            best_a = a.copy()
            best_iter = it
            stagnation = 0

        else:

        # Gradient of the soft‑max w.r.t. a
        rev_a = a[::-1]
        conv_rev = _full_convolution_two(rev_a, weights)   # length 3n‑2
        g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]

        # Chain rule through the softmax (a = softmax(z))
        dot = np.dot(g_a, a)
        grad_z = a * (g_a - dot)

        # Adam update on the logits
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Simple escape mechanism if we have been stuck for a while
        if stagnation >= 3000:
            z += np.random.normal(scale=0.02, size=n)
            stagnation = 0
            print(f"[reset] jitter added at it={it}")

    # Mirror‑descent polishing phase (exact max, top‑k sub‑gradient)
    # --------------------------------------------------------------
    if best_a is not None:

        refine_deadline = deadline
        print("[polish] start exact‑max mirror descent")

                                      top_k=5,
                                      eta_start=0.45,
                                      time_deadline=refine_deadline)
        conv_ref = _full_convolution(a_refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = a_refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # --------------------------------------------------------------
    # Final secure evaluation using the provided wrapper
    # --------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else a0.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
