def _gradient_max(a: np.ndarray) -> tuple[np.ndarray, float]:

    n = a.shape[0]

    out_len = 2*n - 1

    size = 1 << (out_len - 1).bit_length()

    fa = np.fft.rfft(a, size)

    conv = np.fft.irfft(fa * fa, size)[:out_len]

    max_conv = conv.max()

    # mask for max indices

    mask = (conv == max_conv)  # could be exact equality for floats; may have rounding diff; use tolerance.

    # Use tolerance to include near max values:

    eps = max_conv * 1e-12

    mask = conv >= max_conv - eps

    w = mask.astype(float)

    w_sum = w.sum()

    if w_sum > 0:

        w = w / w_sum

    else:

        w = np.zeros_like(conv)

    # Compute gradient:

    rev_w = w[::-1]   (conv length = 2n -1)

    # compute correlation length as n + out_len -1 = n + (2n-1) -1 = 3n -2.

    # Use sizecorr = next pow2.

    out_len_corr = n + out_len - 1

    size_corr = 1 << (out_len_corr - 1).bit_length()

    Fa = np.fft.rfft(a, size_corr)

    Fw = np.fft.rfft(rev_w, size_corr)

    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1

    idx = offset - np.arange(n)

    grad = 2.0 * corr_full[idx]

    return grad, max_conv

"""
Fast, multi‑start minimisation of the step‑function evaluation metric.

The algorithm works in three phases:

1️⃣  Build a modest pool of candidates (Beta‑shaped families,
    Dirichlet draws and an interpolation of the previous best solution).
    The exact objective `2·n·max(a⋆a)` (sum(a)=1) is evaluated with an
    FFT‑based convolution and the best few candidates are kept.

2️⃣  Refine the current best candidate with a projected (onto the simplex)
    gradient descent.  The gradient is first obtained from a
    log‑sum‑exp smoothing of the convolution (parameter β).  β is
    increased quickly, then the *exact* sub‑gradient of the max‑convolution
    is used.  Momentum and a decaying step‑size accelerate convergence.

3️⃣  The best sequence found is passed to the supplied ``evaluate_sequence``
    function (the final, security‑checked evaluation) and returned.

All numbers are kept non‑negative and normalised to a sum of 1, which is
scale‑invariant for the evaluation function.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(a⋆a)
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the parameters would give an all‑zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smoothed objective
        φ_β(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i)

    Returns (grad, max_conv) where ``grad`` is the Euclidean gradient
    (same shape as ``a``) and ``max_conv`` = max(a⋆a).
    """
    n = a.shape[0]
    out_len = 2 * n - 1

    # ----- 1. full convolution a⋆a -----
    size_conv = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[:out_len]
    max_conv = conv.max()

    # ----- 2. soft‑max weights -----
    exp_arg = beta * (conv - max_conv)          # ≤ 0, shift to avoid overflow
    w = np.exp(exp_arg)
    w /= w.sum()                                # Σ w_i = 1

    # ----- 3. gradient = 2·(a ⋆_cross w) -----
    rev_w = w[::-1]                             # reverse for the cross‑correlation
    out_len_corr = n + out_len - 1               # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1
    idx = offset - np.arange(n)                 # indices for j = 0…n‑1
    grad = 2.0 * corr_full[idx]

    return grad, max_conv


def _gradient_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient of the (unsmoothed) max‑convolution objective.
    Returns (grad, max_conv) where ``grad`` corresponds to the sub‑gradient
    at a point attaining the maximal convolution value.
    """
    n = a.shape[0]
    out_len = 2 * n - 1

    # ----- 1. full convolution -----
    size_conv = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[:out_len]
    max_conv = conv.max()

    # ----- 2. hard‑max weights -----
    eps = max_conv * 1e-12
    mask = conv >= max_conv - eps
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum

    # ----- 3. gradient (same formula as in the smoothed case) -----
    rev_w = w[::-1]
    out_len_corr = n + out_len - 1
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1
    idx = offset - np.arange(n)
    grad = 2.0 * corr_full[idx]

    return grad, max_conv

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the
    supplied ``evaluate_sequence`` metric.  Returns the best sequence
    discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a short margin for the final ``evaluate_sequence`` call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 8000 provides a good trade‑off between expressivity
    # and speed (the reference solution also used 8000).
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # 0a – Interpolate the previously best known solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.ndim == 1 and old.size > 0:
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() > 0:
                a0 /= a0.sum()
                candidates.append(a0)
                print("[init] interpolated previous best solution")
    except Exception:
        pass

    # 0b – Deterministic Beta grid (coarse)
    p_grid = np.geomspace(0.2, 5.0, 9)      # 9 values → 81 combos
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is not None:
                candidates.append(cand)

    # 0c – Random Beta draws (log‑uniform in (0.1,10))
    rng = np.random.default_rng()
    for _ in range(1500):
        p = np.exp(rng.uniform(np.log(0.1), np.log(10.0)))
        q = np.exp(rng.uniform(np.log(0.1), np.log(10.0)))
        cand = _beta_shape(p, q, n)
        if cand is not None:
            candidates.append(cand)

    # 0d – A few Dirichlet samples for diversity
    for _ in range(8):
        cand = rng.dirichlet(np.ones(n))
        candidates.append(cand)

    # 0e – Uniform distribution (fallback)
    candidates.append(np.full(n, 1.0 / n))

    # --------------------------------------------------------------
    # 1️⃣  Evaluate all candidates, keep the top few
    # --------------------------------------------------------------
    evaluated = []
    best_val = np.inf
    for a in candidates:
        val = _objective(a)          # 2·n·max_conv (sum(a)=1)
        evaluated.append((val, a))
        if val < best_val:
            best_val = val
    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [c[1].copy() for c in evaluated[:top_k]]
    best_seq = top_candidates[0].copy()
    best_score = best_val
    print(f"[init] pool size={len(candidates)}, best initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Refine the current best candidate with projected gradient descent
    # --------------------------------------------------------------
    # Hyper‑parameters (tuned for speed / quality)
    step0 = 0.7                 # initial step magnitude
    beta = 0.5                  # start of smoothing
    beta_mul = 1.01             # geometric increase of β
    beta_max = 800.0            # switch to exact sub‑gradient when reached
    mu = 0.9                    # momentum coefficient
    momentum = np.zeros_like(best_seq)

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # 2a – gradient (smooth → exact when β is large enough)
        if beta < beta_max:
            grad, cur_max = _gradient_smooth(best_seq, beta)
            beta = min(beta_max, beta * beta_mul)
        else:
            grad, cur_max = _gradient_max(best_seq)

        cur_val = 2.0 * n * cur_max   # exact objective for the current a

        # keep the best ever seen
        if cur_val < best_score:
            best_score = cur_val
            best_seq = best_seq.copy()
            print(f"[refine] new best = {best_score:.6f} (iter {iteration}, β={beta:.2f})")
            # early bail‑out if we have already crossed the target threshold
            if best_score <= 1.503:
                break

        # 2b – momentum update and projected step
        momentum = mu * momentum + grad
        step = step0 / np.sqrt(iteration + 1.0)
        a_next = best_seq - step * momentum
        a_next = _project_simplex(a_next)

        # occasional tiny random jitter to shake off flat plateaus
        if np.random.rand() < 0.001:
            a_next += np.random.normal(scale=0.001, size=n)
            a_next = np.maximum(a_next, 0.0)
            a_next = _project_simplex(a_next)

        best_seq = a_next
        iteration += 1

    # --------------------------------------------------------------
    # 3️⃣  Final secure evaluation (the only place the provided function is called)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = float('inf')

    return best_seq.tolist()
