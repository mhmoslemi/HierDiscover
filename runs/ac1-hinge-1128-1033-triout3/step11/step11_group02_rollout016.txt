def _mse_optimize(a0, n, deadline, start_time):
    a = a0.copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    # target value per entry:
    target = 1.0 / (2 * n - 1)
    best_a = a.copy()
    # compute initial objective
    conv = _full_convolution(a)
    max_b = conv.max()
    best_val = 2.0 * n * max_b
    # adam variables
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1,beta2,eps = 0.9,0.999,1e-8
    alpha = 0.01 # learning rate
    t=0
    while time.time() < deadline:
        t+=1
        conv = _full_convolution(a)
        e = conv - target
        # compute gradient: grad = 4 * conv_rev(e, a_rev)?? Let's compute.
        a_rev = a[::-1]
        # use FFT for convolution e * a_rev
        size2 = 1 << ((len(e)+len(a_rev)-2).bit_length())
        fe = np.fft.rfft(e, size2)
        fa_rev = np.fft.rfft(a_rev, size2)
        conv_ew = np.fft.irfft(fe * fa_rev, size2)[:len(a)+len(e)-1] # length n+2n-2-1=3n-2?
        # conv_ew length is len(e)+len(a_rev)-1 = (2n-1)+(n)-1 = 3n-2
        grad = 4.0 * conv_ew[n-1:n-1+n]   # slice length n
        # adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1-beta1**t)
        v_hat = v / (1-beta2**t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # projection
        a_new = np.maximum(a_new,0.0)
        s = a_new.sum()
        if s == 0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new
        # compute new objective
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[MSEopt {t}] new best = {best_val:.6f}")
        # optional break if below target
        if best_val <= 1.4:
            break
    return best_a, best_val
```

    i = np.arange(n)

def _window_blackman(n):
    i = np.arange(n)
    a = 0.42 - 0.5*np.cos(2*np.pi*i/(n-1)) + 0.08*np.cos(4*np.pi*i/(n-1))

    return a

    i = np.arange(n)
    a = np.minimum(i+1, n-i)
    return a.astype(float)

"""
Propose a step‑function minimizing `evaluate_sequence`.

The search proceeds in three phases:

1️⃣  **Diverse initial seeds** – we reuse a previous solution (if any),
    a brief Beta‑family sweep, a collection of classic window shapes
    (Hann, Hamming, Blackman, Gaussian, sinⁿ, triangular) and a few
    random Dirichlet draws.  All seeds are projected onto the simplex
    (Σa = 1, a_i ≥ 0).

2️⃣  **Smooth‑max optimisation** – we minimise a smooth approximation of
    the max‑convolution using Adam and a λ‑continuation (λ grows from 0.5
    to a large value).  After each gradient step the iterate is projected
    back onto the simplex.

3️⃣  **Constancy optimiser** – we then try to make the convolution as
    flat as possible by minimising the L2 distance to the theoretical
    constant value (1/(2n‑1)).  This also uses Adam with simplex projection.

4️⃣  **Intensive hill‑climbing** – a simulated‑annealing hill‑climber
    refines the best candidate found so far.

All phases respect the 1000 s time limit.  The best sequence (a list of
non‑negative floats) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x * x (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two ≥ out_len
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Beta‑family sweep (dense but cheap)
# ----------------------------------------------------------------------
def _beta_family_best(n: int):
    """Search the beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Classic window families (return a list of candidate arrays)
# ----------------------------------------------------------------------
def _window_seeds(n: int):
    """Generate a collection of classic non‑negative windows."""
    i = np.arange(n, dtype=np.float64)
    seeds = []

    # Hann (sin²) – also covers sin^p with p=2
    seeds.append(np.sin(np.pi * i / (n - 1)) ** 2)

    # Hamming
    seeds.append(0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1)))

    # Blackman
    seeds.append(
        0.42
        - 0.5 * np.cos(2 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    )

    # Triangle (Bartlett)
    seeds.append(np.minimum(i + 1, n - i))

    # Gaussian windows with various width parameters
    for sigma_frac in (0.04, 0.08, 0.12, 0.20):
        mu = (n - 1) / 2.0
        sigma = sigma_frac * n
        seeds.append(np.exp(-0.5 * ((i - mu) / sigma) ** 2))

    # sin^p windows (p = 1…6)
    for p in range(1, 7):
        seeds.append(np.sin(np.pi * i / (n - 1)) ** p)

    # exponential decay
    for alpha in (0.0008, 0.0016, 0.0032):
        seeds.append(np.exp(-alpha * i))

    # normalise all of them
    normed = []
    for a in seeds:
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            continue
        a /= a.sum()
        normed.append(a)
    return normed


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + λ‑continuation)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Optimises 2·n·max(conv(a,a)) using a smoothed max
    (log‑sum‑exp) objective.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_alpha = 0.07
    t = 0

    # FFT sizes (pre‑computed)
    size_conv = 1 << ((2 * n - 2).bit_length())           # for a*a
    size_grad = 1 << ((3 * n - 3).bit_length())           # for a_rev * w

    lambda_max = 200.0

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        # -------- λ continuation ----------
        prog = (now - start_time) / (deadline - start_time)
        lam = 0.5 + (lambda_max - 0.5) * prog

        # -------- compute a*a ----------
        fa = np.fft.rfft(a, size_conv)
        conv = np.fft.irfft(fa * fa, size_conv)[: 2 * n - 1]
        max_c = conv.max()

        # -------- smooth weights ----------
        exps = np.exp(lam * (conv - max_c))
        w = exps / exps.sum()                     # length 2·n‑1, sum 1

        # -------- gradient ----------
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, size_grad)
        fw = np.fft.rfft(w, size_grad)
        conv_aw = np.fft.irfft(fa_rev * fw, size_grad)[: 3 * n - 2]
        grad = 2.0 * conv_aw[n - 1 : n - 1 + n]   # (∂/∂a) of smooth‑max

        # -------- Adam update ----------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning rate is gently decayed when λ grows large
        alpha = base_alpha / (1 + lam / 40.0)

        step = alpha * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        # keep the best (actual max, not the smooth proxy)
        cur_val = 2.0 * n * max_c
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SmoothOpt {t:7d}] new best = {best_val:.6f} λ={lam:.2f}")
            if best_val <= 1.40:          # early stop if very good
                break

        # occasional random perturbation if stuck
        if t % 800 == 0:
            if cur_val >= best_val * 0.9999:
                noise = np.random.normal(scale=0.015, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[SmoothOpt {t:7d}] stagnation – random reset")
    return best_a, best_val

# Constancy optimiser – minimise L2 deviation from flat convolution
# ----------------------------------------------------------------------
def _mse_optimize(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise ∑_k (conv_k - avg)² where avg = 1/(2n‑1).
    Uses Adam and simplex projection.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    target = 1.0 / (2 * n - 1)      # desired flat value
    size_grad = 1 << ((3 * n - 2).bit_length())

    best_a = a.copy()
    conv = _full_convolution(a)
    best_val = 2.0 * n * conv.max()

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_alpha = 0.02
    t = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        # convolution and error
        conv = _full_convolution(a)
        err = conv - target                               # length 2n‑1

        # gradient of squared error: 4 * (err * a_rev) sliced
        a_rev = a[::-1]
        f_err = np.fft.rfft(err, size_grad)
        f_rev = np.fft.rfft(a_rev, size_grad)
        conv_ew = np.fft.irfft(f_err * f_rev, size_grad)[: 3 * n - 2]
        grad = 4.0 * conv_ew[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        alpha = base_alpha / (1 + t / 2000.0)             # decay a little
        step = alpha * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        # evaluate real objective
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[MSEOpt {t:7d}] new best = {best_val:.6f}")
            if best_val <= 1.30:     # aggressive early stop
                break
    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core, slight tweaks)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.40) -> tuple[np.ndarray, float]:
    """
    Local search that moves a little mass from indices dominating the
    maximal convolution entry towards indices with little contribution.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule parameters (more aggressive than the reference)
    max_frac_start, max_frac_end = 0.6, 0.002
    temp_start, temp_end = 0.06, 5e-10

    no_improve = 0
    IMPROVE_LIMIT = 200_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        # indices that contribute to the current maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – weighted random choice among contributors
        if total_contrib > 0 and np.random.rand() < 0.85:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index – prefers indices that affect the max little
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # update convolution efficiently
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # escape from a plateau
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – random perturb")
        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries give a good trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣ Gather diverse initial seeds
    # ------------------------------------------------------------------
    seeds = []

    # a) previous solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)    # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 /= a0.sum()
        seeds.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # b) best from a dense Beta‑family sweep
    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_a is not None:
            seeds.append(beta_a)
            print(f"[init] beta‑family seed (value={beta_val:.6f})")
    except Exception as exc:
        print("[init] beta family failed:", exc)

    # c) classic windows
    try:
        window_seeds = _window_seeds(n)
        seeds.extend(window_seeds)
        print(f"[init] added {len(window_seeds)} classic‑window seeds")
    except Exception as exc:
        print("[init] window seeds failed:", exc)

    # d) a few random Dirichlet draws
    for _ in range(4):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # ensure we have at least one seed
    if not seeds:
        seeds.append(np.random.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 1️⃣ Smooth‑max optimisation (approximately 60 % of the time budget)
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_val = np.inf

    time_for_smooth = (deadline - time.time()) * 0.60
    if time_for_smooth <= 0:
        time_for_smooth = 0.0
    per_seed = time_for_smooth / max(1, len(seeds))

    for idx, a_seed in enumerate(seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed
        a_opt, val_opt = _smooth_optimize(a_seed, n, local_deadline, start_time)
        print(f"[seed {idx}] smooth_opt best = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
        if global_best_val <= 1.40:                      # early exit if already very good
            break

    # ------------------------------------------------------------------
    # 2️⃣ Constancy optimiser (≈ 25 % of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and global_best_a is not None:
        const_deadline = time.time() + remaining * 0.25
        a_const, val_const = _mse_optimize(global_best_a, n, const_deadline, start_time)
        print(f"[MSE phase] best value = {val_const:.6f}")
        if val_const < global_best_val:
            global_best_val, global_best_a = val_const, a_const.copy()

    # ------------------------------------------------------------------
    # 3️⃣ Final intensive hill‑climbing on the rest of the budget
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and global_best_a is not None:
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.40,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final secure evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
