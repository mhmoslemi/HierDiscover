"""
Stochastic smoothed‑gradient descent for the min‑max convolution problem.

The goal is to minimise the evaluation function

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

which, after projecting the candidate onto the probability simplex
(∑a = 1, a ≥ 0), reduces to 2·n·max(conv(a,a)).  The algorithm proceeds as
follows:

1️⃣  Initialise from several cheap families (uniform, Beta‑shapes,
    Dirichlet draws) and keep the best start.
2️⃣  Perform many iterations of projected gradient descent on the
    log‑sum‑exp smoothing of the max‑convolution.
    * β (soft‑max temperature) is increased *very* quickly (×1.01 per
      iteration) so that the smooth objective rapidly approaches the true
      max.
    * The step size decays as 1/√t, and a tiny random jitter is added
      occasionally.
    * After a long period without improvement the current iterate is
      perturbed around the best found so far (a light “restart”).
3️⃣  The best vector encountered is finally handed to the supplied
    ``evaluate_sequence`` and returned.

The whole search respects the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of ``x`` with itself using an FFT of power‑of‑2 size."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _max_convolution(x: np.ndarray) -> float:
    """Return max(conv(x,x)) using the FFT routine above."""
    return float(_full_convolution(x).max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = ind[mask][-1] - 1
    theta = cssv[mask][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Beta‑shaped density on {1,…,n} (unnormalised).  Returns None if all zero."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp smoothing of max(conv(a,a)).
    Returns (grad, max_conv) where ``grad`` has the same shape as ``a``.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()

    # 1️⃣  a⋆a (full linear convolution)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]       # (a⋆a)

    max_conv = conv.max()

    # 2️⃣  Soft‑max weights w_i = exp(beta·(conv_i‑max))/∑exp(...)
    exp_arg = beta * (conv - max_conv)                # ≤ 0
    w = np.exp(exp_arg)
    w /= w.sum()                                      # Σ w_i = 1

    # 3️⃣  Gradient = 2·(a ⋆_cross w)  (cross‑correlation of a and w)
    rev_w = w[::-1]                                   # reversed for convolution
    out_len_corr = n + out_len - 1                     # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1                               # position of lag 0
    idx = offset - np.arange(n)                        # indices for j = 0,…,n‑1
    grad = 2.0 * corr_full[idx]

    return grad, float(max_conv)
