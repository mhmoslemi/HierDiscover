"""
Improved step‑function optimiser.

The optimiser works with a log‑parameter vector `z` (soft‑max guarantees a ≥ 0,
Σa = 1).  At each iteration we

  1. build the *exact* max‑convolution (instead of a smooth‑max)
  2. compute the true sub‑gradient of the max‑convolution w.r.t. the
     probabilities,
  3. back‑propagate through the soft‑max,
  4. perform an Adam update,
  5. keep the best sequence seen so far (evaluated with the real
     `evaluate_sequence`).

During the first half of the time budget we follow the original smooth‑max
schedule (λ grows from 5 to 1500) to give a gentle warm‑up.  In the second
half we switch to the *hard* max (a one‑hot weight vector) which provides a
precise gradient of the objective and often yields a slight improvement.
If the optimiser stalls for a while we inject Gaussian noise into the logits
to escape flat regions.

The routine respects the 1000 s budget, prints progress information and
finally returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (FFT size)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Full linear convolution a*a (length 2·n‑1) via FFT
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear full convolution of x with itself (real FFT)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Cross‑correlation needed for the max‑gradient:
#   g_i = Σ_k w_k * a_{k‑i}
# ----------------------------------------------------------------------
def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k-i} for i=0..len(a)-1.
    Uses FFT; returns a real vector of length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = len(w) + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1                     # extract indices n‑1 … n‑1+n‑1
    return conv[start:start + n]
