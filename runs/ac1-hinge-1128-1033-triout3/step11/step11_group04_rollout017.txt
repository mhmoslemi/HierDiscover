    # Convert to list
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Protect against negative numbers

"""
Optimiser for the step‑function “max‑convolution” objective.

The goal is to minimise   2·n·max(conv(a,a))/(sum a)^2 .
All entries of a are non‑negative and are projected onto the
probability simplex (so Σa = 1).  The optimiser consists of
four loosely coupled phases that have proved useful in practice:

0️⃣  Diverse initialisation – many Dirichlet / Beta draws,
    a few deterministic shapes and a simple entropy‑based
    refinement.

1️⃣  Adam on a smooth approximation of the max‑convolution.
    The smoothing parameter ε is annealed from 0.5 to 10⁻⁹;
    the learning‑rate is annealed from 0.25 to 0.001.
    Variables live in the log‑softmax (logits) space, which
    automatically enforces the simplex constraint.

2️⃣  Aggressive hill‑climbing – in each step we look at the
    current maximal convolution entry, pick a “source” index
    proportional to its contribution and a “target’’ index
    with a low contribution, then move a controlled chunk of
    mass from source to target.  Acceptance follows a
    simulated‑annealing rule.

3️⃣  Entropy‑gradient polishing – the convolution of a with its
    reverse is a probability distribution (it sums to 1).  By
    maximising its Shannon entropy we encourage the distribution
    to become flat, which directly reduces the ℓ∞‑norm.
    A projected gradient ascent with a line‑search step‑size is
    performed while the wall‑clock permits.

All phases monitor the remaining wall‑time and stop as soon
as the 1000 s budget is exhausted.  The best sequence encountered
throughout the run is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex Σx = 1, x ≥ 0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam with log‑softmax parametrisation and exponential ε‑schedule."""
    n = a0.shape[0]
    # work in logits (log‑softmax) – this guarantees a valid simplex point
    w = np.log(a0 + 1e-16)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.25, 0.001
    eps_start, eps_end = 0.5, 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 5000

    while time.time() < deadline:
        # decode logits
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedules
        prog = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** prog
        eps = eps_start * (eps_end / eps_start) ** prog

        # smooth‑max and its gradient (already contains the factor 2·n)
        _, grad = _smooth_max_and_grad(a, eps)

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to avoid dead‑ends
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.015, size=n)

        # aggressive restart when stuck
        if idle > MAX_IDLE:
            print("[Adam] restart – idle")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max(conv(a,a)) together with gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed weight vector
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes 2·n factor
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Hill‑climbing on the true objective (exact max)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """
    Local search moving a chunk of mass from a high‑contribution
    index to a low‑contribution index, accepting moves with a
    simple simulated‑annealing rule.
    """
    n = a0.shape[0]
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # annealing schedules
    max_frac_start, max_frac_end = 0.9, 0.002
    temp_start, temp_end = 0.05, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iteration = 0

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # offset with the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        if total > 0.0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target: prefer indices with low contribution to the same offset
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback: random distinct index
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # re‑inject a little noise around the current best
            print(f"[Hill] stagnation at it={iteration}, re‑perturb")
            a = best_a.copy()
            a += np.random.normal(scale=0.025, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Entropy‑gradient polishing (flattens the convolution distribution)
# ----------------------------------------------------------------------
def _entropy_polish(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient ascent on the Shannon entropy of the
    convolution distribution.  The entropy is
        H(c) = - Σ c_k log c_k ,   where c = conv(a,a)
    Maximising H drives c towards a flat distribution,
    which reduces its ℓ∞‑norm.
    """
    n = a0.shape[0]
    a = a0.copy()
    cur_val = _objective(a)

    while time.time() < deadline:
        # current convolution
        c = _full_convolution(a)               # length 2n‑1, sums to 1
        # avoid log(0)
        safe_c = np.maximum(c, 1e-20)
        weight = - (np.log(safe_c) + 1.0)       # derivative of -c log c

        # gradient of entropy w.r.t. a  (see derivation in analysis)
        grad = -2.0 * _conv_fft(a, weight[::-1])   # shape (2n‑1)
        grad = grad[n - 1:2 * n - 1]               # keep n entries

        # simple line‑search step‑size (decrease as we approach the deadline)
        prog = (time.time() - start_time) / (deadline - start_time)
        step = 0.4 * (1.0 - prog) + 1e-5 * prog

        a_new = a + step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        # accept if the objective improves (i.e. max‑convolution drops)
        if new_val < cur_val:
            a, cur_val = a_new, new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Ent] new best = {best_val:.6f}")

        # occasional jitter to escape flat regions
        if np.random.rand() < 0.004:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to generate a few deterministic candidates
# ----------------------------------------------------------------------
def _deterministic_candidates(n: int) -> list[np.ndarray]:
    """
    Return a short list of manually crafted shape candidates.
    All are normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)

    cand_list = []

    # (a) linear decreasing shape (triangular)
    triangular = (n - i).astype(np.float64)
    triangular /= triangular.sum()
    cand_list.append(triangular)

    # (b) quadratic decreasing
    quad = ((n - i) ** 2).astype(np.float64)
    quad /= quad.sum()
    cand_list.append(quad)

    # (c) exponential decay (several rates)
    for base in (0.9995, 0.999, 0.995):
        exp_decay = base ** i
        exp_decay /= exp_decay.sum()
        cand_list.append(exp_decay)

    # (d) a simple “U‑shape”: small at ends, larger in the middle
    u_shape = np.sin(np.pi * (i + 0.5) / n) + 0.1
    u_shape = np.maximum(u_shape, 0.0)
    u_shape /= u_shape.sum()
    cand_list.append(u_shape)

    return cand_list

        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()


def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Draw parameters (p,q) on a log‑scale and return a symmetric Beta‑shaped
    probability vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a
