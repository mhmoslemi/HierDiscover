def _full_convolution_fft(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)

"""
Propose a non‑negative step function that minimizes the provided
``evaluate_sequence``.  The algorithm works as follows:

1️⃣  Initialise a smooth, symmetric shape (triangular or an interpolated
    previous solution) on a moderately large grid (default n≈8000).

2️⃣  Run a *soft‑max mirror descent* on the exact objective
    f(a)=2·n·max(conv(a,a)) :

   –  each iteration evaluates the full self‑convolution via a single
      FFT (complexity O(N log N));
   –  a temperature α turns the max into a smooth “log‑sum‑exp”
      (soft‑max) which yields a dense sub‑gradient;
   –  the sub‑gradient is obtained analytically by another FFT
      (convolution with the soft‑max weights);
   –  an exponential‑gradient update (multiplicative) together with a
      momentum term keeps the iterate inside the probability simplex;
   –  the step size decays linearly from 0.5 to 1e‑4, while α grows
      slowly (0.5 → 100) so that the optimisation first sees a smooth
      landscape and later refines the true max‑objective;
   –  occasional Gaussian jitter and momentum reset help to escape
      plateaus.

3️⃣  Keep the best iterate (in terms of the *exact* max‑convolution)
    witnessed during the search.

4️⃣  Return the best sequence; a single call to ``evaluate_sequence``
    reports the final score.

The whole routine respects a user supplied time budget (default 1000 s)
and never uses file or network I/O.  All helper routines are defined
at module level (no closures)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities – all based on real FFT for speed
# ----------------------------------------------------------------------


def _full_convolution_fft(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """
    Return the full linear convolution of x and y (length len(x)+len(y)-1)
    using a real FFT of length ``size`` (must be >= len(x)+len(y)-1).
    """
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: x.shape[0] + y.shape[0] - 1]


def _grad_from_weights(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the gradient of the smoothed objective
        f_s(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    where w_k = exp(α·(a*a)_k) / Σ exp(...).

    The exact gradient is   ∇f_s = 2· Σ_k w_k·a_{k‑i}.
    It can be obtained by a convolution of the reversed weight vector
    with ``a``.
    """
    n = a.shape[0]
    w_rev = w[::-1]                                 # length 2n‑1
    # full convolution of w_rev (len 2n‑1) with a (len n) → length 3n‑2
    conv_full = _full_convolution_fft(w_rev, a, size)
    # indices that correspond to the required slice:
    #   grad_i = 2·conv_full[2n‑2‑i]   (i = 0…n‑1)
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * conv_full[idx]


# ----------------------------------------------------------------------
# Tiny helper – numerical stability for log‑sum‑exp
# ----------------------------------------------------------------------


def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return the soft‑max weights w_k = exp(α·b_k) / Σ exp(α·b_k)
    in a numerically stable way (subtract max before exponentiation).
    """
    b_max = b.max()
    # shift for stability
    exp_term = np.exp(alpha * (b - b_max))
    sum_exp = exp_term.sum()
    # avoid division by zero (should never happen)
    if sum_exp == 0.0:
        return np.full_like(b, 1.0 / b.shape[0])
    return exp_term / sum_exp


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The method follows the plan described in the
    module docstring and returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a compromise between flexibility and speed.
    # --------------------------------------------------------------
    n = 8000                     # feel free to increase if time permits

    # --------------------------------------------------------------
    # 0️⃣  Initialise from a previous solution if available,
    #     otherwise use a symmetric triangular window.
    # --------------------------------------------------------------
    old_seq = globals().get("height_sequence_1")
    if isinstance(old_seq, (list, np.ndarray)):
        old_arr = np.asarray(old_seq, dtype=float)
        # linear interpolation to the current resolution
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_arr.size)
        a = np.interp(new_x, old_x, old_arr)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0 or not np.isfinite(a_sum):
            a = np.full(n, 1.0 / n, dtype=float)
        else:
            a /= a_sum
        print("[init] interpolated previous solution")
    else:
        # symmetric triangular shape – simple, already fairly good
        i = np.arange(1, n + 1, dtype=float)
        a = np.minimum(i, n + 1 - i)
        a /= a.sum()
        print("[init] triangular start")

    # --------------------------------------------------------------
    # Pre‑compute the FFT size (next power‑of‑two ≥ 3·n‑2)
    # --------------------------------------------------------------
    conv_len = 3 * n - 2
    fft_size = 1 << ((conv_len - 1).bit_length())   # next power of two

    # --------------------------------------------------------------
    # Keep the best sequence witnessed during the search
    # --------------------------------------------------------------
    # cheap objective (exact max‑convolution) – used for comparison only
    def _cheap_objective(vec: np.ndarray) -> float:
        fa = np.fft.rfft(vec, fft_size)
        conv = np.fft.irfft(fa * fa, n=fft_size)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    best_seq = a.copy()
    best_val = _cheap_objective(best_seq)
    print(f"[search] initial cheap value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------------
    beta_mom = 0.9               # momentum coefficient
    m = np.zeros_like(a)         # momentum accumulator
    it = 0
    no_improve = 0
    max_no_improve = 5000        # after this many stale steps we jitter
    jitter_prob = 0.001

    # temperature schedule for the soft‑max (α)
    alpha_start = 0.5
    alpha_end = 120.0
    # step‑size schedule (0.5 → 1e‑4)
    step_start = 0.5
    step_end = 1e-4

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ---- self‑convolution and current objective ----
        fa = np.fft.rfft(a, fft_size)
        conv_self = np.fft.irfft(fa * fa, n=fft_size)[: 2 * n - 1]
        max_conv = conv_self.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[optim] iter {it:06d} new best {best_val:.6f}")
        else:
            no_improve += 1

        # ---- temperature (α) for the smooth max ----
        progress = (time.time() - start_time) / (deadline - start_time)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        # ---- soft‑max weights w_k ----
        w = _softmax_weights(conv_self, alpha)

        # ---- gradient of the smooth objective ----
        grad = _grad_from_weights(a, w, fft_size)

        # ---- momentum update ----
        m = beta_mom * m + (1.0 - beta_mom) * grad

        # ---- step‑size schedule ----
        step = step_start + (step_end - step_start) * progress

        # ---- (multiplicative) exponentiated‑gradient update ----
        log_factor = -step * m
        # keep the exponent in a safe range to avoid overflow/underflow
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)

        # ---- renormalise onto the simplex (sum = 1) ----
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=float)
        else:
            a = a / s

        # ---- occasional jitter to escape flat regions ----
        if np.random.rand() < jitter_prob or no_improve >= max_no_improve:
            jitter = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            m.fill(0.0)
            no_improve = 0
            print(f"[optim] iter {it:06d} jitter applied")

        it += 1

    # --------------------------------------------------------------
    # 3️⃣  Final evaluation of the best candidate
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
