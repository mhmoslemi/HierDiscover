"""
Propose a step‑function that (approximately) solves

    minimise  2·n·max(conv(a,a)) / (sum a)^2   with a_i ≥ 0 ,  sum a = 1 .

The algorithm consists of four stages:

0️⃣  Initialise a large number of random Beta / Dirichlet candidates and keep the best.
1️⃣  Successively minimise a smooth‑approximation of the objective with
    L‑BFGS‑B on the soft‑max (logits) parametrisation.
    The smoothing parameter ε is decreased geometrically (0.5 → 1e‑9).
2️⃣  A short projected sub‑gradient polishing phase on the true (non‑smooth) objective.
3️⃣  Blend the final candidate with the uniform distribution (helps to smooth isolated spikes)
    and return the best sequence found.
All stages respect the 1000 s wall‑clock budget.
"""

import time
import numpy as np
import sys

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # Log‑sum‑exp stabilisation
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # (2n‑1,) – soft max over sums

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries (the “middle” part)
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n

    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Soft‑max (logits → probability simplex)
# ----------------------------------------------------------------------
def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector that sums to 1."""
    w_shift = w - np.max(w)
    exp_w = np.exp(w_shift)
    return exp_w / exp_w.sum()


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth objective (logits space)
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    eps_schedule: list[float],
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float) -> tuple[np.ndarray, float]:
    """
    Runs a sequence of L‑BFGS‑B optimisations on the smooth approximation.
    Each stage uses a smaller smoothing ε so that the optimisation gets
    progressively more accurate.
    """
    try:
        import scipy.optimize as opt
    except Exception:               # pragma: no cover
        print("[LBFGS] SciPy not available – skipping this stage.")
        return best_a, best_val

    # initialise logits from the current best vector
    w = np.log(a0 + 1e-12)

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec: np.ndarray, n=n, eps=eps):
            a = _softmax(w_vec)
            smooth_val, grad_a = _smooth_max_and_grad(a, eps)   # grad_a already has 2·n factor
            obj = 2.0 * n * smooth_val
            # chain‑rule to logits
            dot = np.dot(grad_a, a)
            grad_w = a * (grad_a - dot)
            return obj, grad_w

        # L‑BFGS‑B parameters – a modest iteration budget for each ε
        maxiter = 200
        try:
            res = opt.minimize(fun_and_grad,
                               w,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-12})
        except Exception as exc:       # pragma: no cover
            print(f"[LBFGS] optimisation failed at ε={eps:.1e}: {exc}")
            continue

        w = res.x
        a = _softmax(w)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[LBFGS ε={eps:.1e}] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The step size decays linearly from 3e‑1 to 1e‑5.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        # indices where the maximum is attained (allow a tiny tolerance)
        k_maxes = np.where(np.abs(b - max_b) < 1e-15)[0]

        # sub‑gradient of 2·n·max_b
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule (t ∈ [0,1])
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-13:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to escape flat regions
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta family candidate (used for the initial random stage)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a
