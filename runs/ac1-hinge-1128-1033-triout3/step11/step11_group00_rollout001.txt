"""
Propose a candidate step function that (hopefully) beats the previous best
(≈ 1.5082).  

The method combines three ideas:

1. **Coarse parametric search** – on a small problem (n≈2400) we test a
   handful of symmetric Beta‑shapes, discrete Gaussians and cosine‑power
   shapes.  The best of them is interpolated to the final size.

2. **Adam on a smooth approximation of the max‑convolution** – we optimise
   the “soft‑max’’ of the self‑convolution in the log‑domain (so the
   variables stay non‑negative) while the smoothing parameter α is
   slowly increased from 1 to a few hundred.  This is the same principle
   as the reference code but with a larger problem size (≈ 12000) and
   tuned schedules.

3. **Exact mirror‑descent polishing** – after the Adam phase we run an
   entropy‑mirror‑descent (multiplicative updates) on the exact max entry
   of the convolution.  This short deterministic refinement often yields
   a few‑percent improvement.

The total wall‑time is limited to ``budget_s`` seconds (default 1000 s).
All time checks leave a small safety margin before the final call to
``evaluate_sequence``.  The routine returns a list of non‑negative floats
summed to a positive value (the scale does not matter because the
objective is scale‑invariant)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (exact linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two (possibly different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Mirror‑descent polishing (exact max entry)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray,
                steps: int = 1500,
                eta_start: float = 0.5) -> np.ndarray:
    """
    Entropy‑mirror descent on the exact max‑entry of the convolution.
    The updates are multiplicative (a_i ← a_i·exp(‑η·g_i)) and a is
    renormalised to stay on the probability simplex.
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of max‑entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # decreasing step size
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Helper: evaluate exact objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _exact_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) (the denominator is 1 because sum(a)=1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())

        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------------------------------------------------------------
        # Simple escape mechanism if we have been stuck for a while
        # --------------------------------------------------------------
        if stagnation >= 2500:
            # add a small random jitter to the logits
            z += np.random.normal(scale=0.03, size=n_target)
            stagnation = 0
            print(f"[reset] jitter added at it={it}")

        # Early exit if we already beat the target (1.503) comfortably
        if best_obj is not None and best_obj <= 1.503:
            print(f"[early] target reached (O={best_obj:.6f}) at iteration {it}")
            break

    # ------------------------------------------------------------------
    # 4. Polishing phase: exact max‑entry mirror‑descent
    # ------------------------------------------------------------------
    if best_a is not None:
        a_refined = _max_refine(best_a, steps=2200, eta_start=0.45)
        obj_ref = _exact_obj(a_refined)
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = a_refined
            print(f"[polish] improved to O={best_obj:.6f}")

        # A very short random‑perturbation neighbourhood search (optional)
        # This sometimes squeezes a few extra hundredths out of the result.
        neigh_iters = 800
        for _ in range(neigh_iters):
            # jitter in log‑space, keep sum = 1
            z_tmp = np.log(best_a + eps) + np.random.normal(scale=0.015,
                                                            size=n_target)
            a_tmp = np.exp(z_tmp)
            a_tmp /= a_tmp.sum()
            obj_tmp = _exact_obj(a_tmp)
            if obj_tmp < best_obj:
                best_obj = obj_tmp
                best_a = a_tmp
                print(f"[neigh] improvement O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the provided security wrapper
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else a0.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx O={best_obj:.6f})")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
