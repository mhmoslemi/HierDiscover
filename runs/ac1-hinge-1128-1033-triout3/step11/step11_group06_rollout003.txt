def _gradient_smooth(a, beta):
    n = a.shape[0]
    conv = _full_convolution_fft(a)
    max_conv = conv.max()
    # compute w
    exp_arg = beta * (conv - max_conv)
    w = np.exp(exp_arg)  # safe
    w_sum = w.sum()
    w_norm = w / w_sum
    # compute gradient: 2 * cross-correlation of a with w_norm reversed.
    rev_w = w_norm[::-1]
    out_len_corr = n + (2*n-1) - 1  # 3n-2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)

    offset = (2*n-1) - 1  # out_len - 1
    idx = offset - np.arange(n)
    grad = 2.0 * corr_full[idx]  # shape n
    return grad, max_conv

"""
Propose a near‑optimal non‑negative step function.

The search works in three phases (all run within the given time budget):

1️⃣  **Initialisation** – try a few Beta‑family shapes, random Dirichlet draws,
    and (if present) the previously discovered sequence `height_sequence_1`.
    The best among them is kept.

2️⃣  **Smoothed gradient descent** – minimise the log‑sum‑exp smoothing of the
    max‑convolution objective.
    The temperature parameter `β` starts small and is multiplied each iteration
    (up to a very large value), so the smoothed gradient approaches the true
    sub‑gradient of the max‑convolution.  After each gradient step the vector is
    projected onto the probability simplex.

3️⃣  **Fine‑tuning** – two complementary local‑search stages are applied:
    *Adam* sub‑gradient descent (exact max‑convolution) and a simple
    “mass‑swap” Monte‑Carlo (simulated‑annealing style) that moves a tiny
    amount of probability mass from one coordinate to another.

The best sequence encountered during the whole run is finally evaluated with
the provided ``evaluate_sequence`` function and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _objective_val(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the parameters would give an all‑zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smoothed objective
        φ_β(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i)

    Returns (grad, max_conv) where `grad` has the same shape as `a`
    and `max_conv` is max(a⋆a) (useful for the exact objective).
    """
    n = a.shape[0]

    # ----- 1. full convolution a⋆a (length 2n‑1) -----
    out_len = 2 * n - 1
    conv = _full_convolution_fft(a)            # (a⋆a)
    max_conv = conv.max()

    # ----- 2. soft‑max weights w_i -----
    exp_arg = beta * (conv - max_conv)         # ≤ 0, numerically stable
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum                         # Σ_i w_i = 1

    # ----- 3. gradient = 2·(a ⋆_cross w_norm) -----
    rev_w = w_norm[::-1]                       # reverse for correlation
    out_len_corr = n + out_len - 1             # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1                       # position of lag 0 in full corr
    idx = offset - np.arange(n)                # indices for j = 0…n‑1
    grad = 2.0 * corr_full[idx]

    return grad, float(max_conv)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # problem size – a trade‑off between expressivity and speed
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previously found construction (if it exists)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old.size > 0:
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() > 0:
                a0 /= a0.sum()
                best_a = a0.copy()
                best_val = _objective_val(best_a)
                print("[init] interpolated previous solution, value =", best_val)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse scan of the two‑parameter Beta family
    # ------------------------------------------------------------------
    def _maybe_update(candidate: np.ndarray):
        nonlocal best_a, best_val
        if candidate is None:
            return
        cand = candidate.astype(float)
        if cand.sum() == 0:
            return
        cand /= cand.sum()
        val = _objective_val(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            print(f"[grid] new best = {best_val:.6f}")

    # deterministic grid (moderately dense)
    p_grid = np.concatenate((
        np.geomspace(0.01, 0.1, 6, endpoint=False),
        np.geomspace(0.1, 5.0, 15, endpoint=False),
        np.geomspace(5.0, 30.0, 7)
    ))
    q_grid = p_grid.copy()

    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            _maybe_update(cand)
        if time.time() >= deadline - 300:
            break

    # random draws in (p, q)-space (log‑uniform)
    log_min, log_max = np.log(1e-3), np.log(30.0)
    for _ in range(6000):
        p = np.exp(rng.uniform(log_min, log_max))
        q = np.exp(rng.uniform(log_min, log_max))
        cand = _beta_shape(p, q, n)
        _maybe_update(cand)
        if time.time() >= deadline - 300:
            break

    # a few Dirichlet draws with various concentration parameters
    for concentration in (0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            cand = rng.dirichlet(np.full(n, concentration))
            _maybe_update(cand)
            if time.time() >= deadline - 300:
                break

    # fallback – uniform distribution
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective_val(best_a)
        print("[fallback] uniform start, value =", best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent (log‑sum‑exp smoothing)
    # ------------------------------------------------------------------
    a = best_a.copy()
    step0 = 0.6                     # initial step size
    beta = 0.5                      # start temperature
    beta_max = 5000.0
    beta_mul = 1.003                # multiplicative increase per iteration
    iteration = 0

    while True:
        now = time.time()
        # reserve about 200 s for the later fine‑tuning phases
        if now >= deadline - 200:
            break

        step = step0 / np.sqrt(iteration + 1.0)

        grad, cur_max = _gradient_smooth(a, beta)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[smooth] new best = {best_val:.6f} (iter {iteration}, β={beta:.2f})")
            if best_val <= 1.503:
                break

        # projected gradient step
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        # occasional tiny jitter to escape flat plateaus
        if rng.random() < 0.001:
            a_candidate += rng.normal(scale=0.001, size=n)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

        a = a_candidate
        beta = min(beta_max, beta * beta_mul)
        iteration += 1

    # ------------------------------------------------------------------
    # 3️⃣  Fine‑tuning: Adam sub‑gradient descent
    # ------------------------------------------------------------------
    a = best_a.copy()
    lr = 0.12
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    iteration = 0

    while True:
        now = time.time()
        # keep ~60 s for the final Monte‑Carlo stage
        if now >= deadline - 60:
            break

        conv = _full_convolution_fft(a)
        max_idx = int(conv.argmax())
        max_val = float(conv.max())
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] new best = {best_val:.6f} (iter {iteration})")
            if best_val <= 1.503:
                break

        # sub‑gradient of max‑convolution (2·a shifted by max_idx)
        grad = np.zeros_like(a)
        j_start = max(0, max_idx - (n - 1))
        j_end = min(n - 1, max_idx)
        if j_start <= j_end:
            idxs = max_idx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] = 2.0 * a[idxs]

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))
        step = lr / np.sqrt(iteration + 1)
        a_candidate = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = _project_simplex(a_candidate)

        a = a_candidate
        iteration += 1

    # ------------------------------------------------------------------
    # 4️⃣  Local Monte‑Carlo (mass‑swap) refinement
    # ------------------------------------------------------------------
    a = best_a.copy()
    iter_mc = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if a[i] <= 1e-12:
            continue

        # amount to transfer – slowly shrinking with time
        time_frac = (now - start_time) / budget_s
        max_transfer = min(0.005 * (1.0 - time_frac), a[i])
        delta = rng.uniform(0.0, max_transfer)
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # exact objective for the proposal
        val = _objective_val(new_a)
        if val < best_val:
            best_val = val
            best_a = new_a.copy()
            a = new_a.copy()
            print(f"[swap] new best = {best_val:.6f} (iter {iter_mc})")
        else:
            # simple Metropolis acceptance to keep diversity
            temp = 0.001 * (1.0 - time_frac)
            if temp > 0 and rng.random() < np.exp(-(val - best_val) / temp):
                a = new_a
        iter_mc += 1

    # ------------------------------------------------------------------
    # 5️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
