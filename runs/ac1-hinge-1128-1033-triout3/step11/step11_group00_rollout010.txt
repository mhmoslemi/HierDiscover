def build_sidon_set(n, attempts, seed):
    best_set = []
    rng = np.random.default_rng(seed)
    candidates = np.arange(n)
    for att in range(attempts):
        rng.shuffle(candidates)
        used = np.zeros(2*n, dtype=bool)
        current = []
        for i in candidates:
            # compute sums with current set
            possible = True
            # start with self sum
            if used[i + i]:
                possible = False
            else:
                # test sums with existing elements
                for j in current:
                    s = i + j
                    if used[s]:
                        possible = False
                        break
            if possible:
                # add
                current.append(i)
                used[i + i] = True
                for j in current[:-1]:  # all prior elements
                    used[i + j] = True
        if len(current) > len(best_set):
            best_set = current[:]
    return np.array(best_set)
```

The implementation builds on the successful “soft‑max + Adam” approach

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions (full linear)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a vector with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two (possibly different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

# Entropy‑mirror descent polishing on the *exact* max entry
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 5000, eta_start: float = 0.5) -> np.ndarray:
    """
    Perform entropy‑mirror descent on the exact maximum of ``a*a``.
    ``a`` is assumed to be a probability vector (sum == 1, non‑negative).
    """
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())                       # index of the worst entry
        # sub‑gradient of the max‑entry w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]

        # decreasing stepsize
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a

            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ----- occasional random jitter to escape plateaus -----
            if stagnation >= 2500:
                z += np.random.normal(scale=0.015, size=n)
                stagnation = 0
                print(f"[restart {restart_counter}] jitter added at it={it}")

        # -----------------------------------------------------------------
        # Polishing phase for the best candidate of this restart
        # -----------------------------------------------------------------
        if best_a is not None:
            a_refined = _max_refine(best_a, steps=5000, eta_start=0.45)
            obj_ref = _exact_obj(a_refined)
            if obj_ref < best_obj:
                best_obj = obj_ref
                best_a = a_refined
                print(f"[polish] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else np.full(n, 1.0 / n).tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(recorded O≈{best_obj:.6f})")
    except Exception as exc:                 # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
```
