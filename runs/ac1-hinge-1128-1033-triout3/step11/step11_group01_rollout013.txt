"""
Non‑negative step‑function optimiser (sparse‑init + Adam).

The optimiser works in two phases.
1️⃣  A sparse support is built (greedy / random) and an equal‑weight
    distribution on that support is evaluated.  This gives a good
    starting point that already beats the uniform “triangular” baseline.
2️⃣  From the best seed we run a short Adam‑optimisation on the
    log‑probabilities restricted to the current support.  A smooth‑max
    (soft‑max over the self‑convolution) provides a differentiable proxy
    for the `max` in the objective, the temperature λ is increased
    linearly up to a large value (≈10 000) so that the gradient focuses
    on the true peak.  Stagnation is broken by adding Gaussian noise to
    the logits.

The routine respects the 1000 s budget, prints progress and finally
returns the best sequence found (as a plain list of floats).  The final
score is obtained with the official `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full self‑convolution of a real vector using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation needed for the gradient:
        g_i = Σ_k w_k * a_{k-i}
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = len(w) + n - 1            # = (2n‑1)+n‑1 = 3n‑2
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    start = n - 1                       # slice the central part
    return conv[start:start + n]


def _score_of(a: np.ndarray) -> float:
    """Objective value 2·n·max(conv) (sum(a) is normalised to 1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Greedy construction of a low‑collision support set
# ----------------------------------------------------------------------
def _build_sparse_support(n: int,
                          target_size: int,
                          max_collisions: int = 5) -> list[int]:
    """
    Greedy construction of a set of indices S ⊂ {0,…,n‑1}
    such that every ordered pair (i,j) with i,j∈S produces a sum
    that occurs at most `max_collisions` times.
    Returns the list of selected indices.
    """
    sum_counts = np.zeros(2 * n - 1, dtype=np.int32)
    chosen = []
    order = np.random.permutation(n)

    for idx in order:
        # test if adding `idx` would violate the collision bound
        ok = True
        # collisions with already chosen elements (ordered pairs give +2)
        for p in chosen:
            s = idx + p
            if sum_counts[s] + 2 > max_collisions:
                ok = False
                break
        if not ok:
            continue
        # self‑pair (i,i) contributes only once
        if sum_counts[idx + idx] + 1 > max_collisions:
            continue

        # accept the index
        for p in chosen:
            s = idx + p
            sum_counts[s] += 2
        sum_counts[idx + idx] += 1
        chosen.append(idx)
        if len(chosen) >= target_size:
            break

    return chosen

        else:
            stagnation_cnt += 1

        lam = lambda_start * (1.0 - prog) + lambda_end * prog
        # shift by max for numeric stability
        exp_vals = np.exp(lam * (conv - conv_max))

        v_hat = v / (1.0 - b2 ** t_iter)
