def _lp_optimize_for_b(a: np.ndarray, M: float) -> np.ndarray:
    n = a.shape[0]
    # Build A_ub rows for constraints conv(a,b) <= M
    # Because we need each row k: sum_j a_{k-j} * b_j <= M
    # Use sparse SciPy matrix for efficiency.
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])  # coefficient a_{k - j}
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    # b_ub vector:
    b_ub = np.full(2 * n - 1, M)
    # Objective maximize sum(b) => minimize -1^T b
    c = -np.ones(n)
    # bounds
    bounds = [(0, None)] * n
    # Use scipy.optimize.linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return a itself
        return a.copy()
    b_opt = res.x
    return b_opt

"""
Iterated Linear‑Program + Hill‑Climbing optimizer for the step‑function attack.

The routine works in two stages:

1️⃣  **LP‑refinement** – on a modest size (n≈1000) we repeatedly:
    * compute M = max(conv(a,a)),
    * solve the linear programme   maximise   Σ b_j   subject to   (a⋆b)_k ≤ M,
      b≥0   (the convolution (a⋆b) is linear in b),
    * normalise the obtained b to a probability vector,
    * mix it with the current a (convex combination) and re‑project onto the simplex.
    Each LP pushes mass into coordinates where the current a is tiny, thus decreasing
    the maximal convolution entry.

2️⃣  **Fast hill‑climbing** – after a few LP steps we switch to an O(n) local search
    that moves a small “chunk’’ of mass from a source index (large contribution to the
    maximal convolution entry) toward a target index (small contribution).  Simulated‑
    annealing acceptance allows occasional uphill moves.

Both phases respect the non‑negativity and simplex constraints, so the final vector
can be fed directly to ``evaluate_sequence``.  The whole search is bounded by the
given wall‑clock budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑program that, for a given a, finds the largest‑ℓ₁ vector b
# such that conv(a,b) ≤ M (entry‑wise).  Returns the raw b (not normalised).
# ----------------------------------------------------------------------
def _lp_optimize_for_b(a: np.ndarray, M: float) -> np.ndarray:
    """Solve  max Σ b_j   s.t.   (a⋆b)_k ≤ M  (for all k),  b ≥ 0."""
    n = a.shape[0]
    # Build a sparse matrix representing the convolution constraints
    # (a⋆b)_k = Σ_j a_{k-j} b_j  (indices out of range give 0).
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])          # coefficient a_{k-j}
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Maximise Σ b_j  → minimise -Σ b_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    # Use the high‑performance HiGHS solver (available via SciPy ≥1.6)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        # Fallback: return the current a (feasible because a itself satisfies the constraints)
        return a.copy()
    return res.x


# ----------------------------------------------------------------------
# Fast O(n) hill‑climbing (source → target mass transfer)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """Local search moving small chunks of mass from a “high’’ index to a “low’’ one."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12
    no_improve = 0
    IMPROVE_LIMIT = 50000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        # gradually shrink moves and temperature
        max_frac = 0.6 * (1.0 - prog) + 0.001 * prog
        T = 0.02 * (1.0 - prog) + 1e-9 * prog

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # compute contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source – pick proportionally to contribution
        if total > 0.0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target – favour positions with small contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # any different index
            j = (i + 1) % n

        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional re‑perturbation on stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a non‑negative step‑function under a fixed time budget.
    Returns a plain list of floats (its ℓ₁‑norm equals 1); ``evaluate_sequence``
    can be called on the result to obtain the official score.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size – a moderate dimension keeps the LP cheap
    # ------------------------------------------------------------------
    n = 1024

    # ------------------------------------------------------------------
    # 0️⃣  Random initialisation (uniform + a few Dirichlet draws)
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n)                # uniform simplex point
    best_a = a.copy()
    best_val = _objective(best_a)
    print(f"[Init] uniform start, value={best_val:.6f}")

    # a handful of Dirichlet candidates with various concentrations
    for alpha in (0.02, 0.05, 0.1, 0.3, 0.7, 1.0):
        for _ in range(30):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[Init] Dirichlet α={alpha:.2f} better = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  LP‑refinement loop (takes the bulk of the budget)
    # ------------------------------------------------------------------
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline - 10.0:            # keep ~10 s for final hill‑climb
            break
        # compute the current maximal convolution entry
        conv_a = _full_convolution(best_a)
        M = conv_a.max()

        # solve the linear programme for the partner vector b
        b_vec = _lp_optimize_for_b(best_a, M)

        # normalise b to stay on the simplex
        if b_vec.sum() == 0.0:
            # fall back to uniform if LP fails dramatically
            b_vec = np.full(n, 1.0 / n)
        else:
            b_vec = b_vec / b_vec.sum()

        # convex mixing – decreasing step size as we iterate
        t = 0.6 * (0.9 ** iteration) + 0.02   # starts around 0.6, decays
        a_new = (1.0 - t) * best_a + t * b_vec
        a_new = _project_simplex(a_new)

        cur_val = _objective(a_new)
        if cur_val < best_val:
            best_val, best_a = cur_val, a_new.copy()
            print(f"[LP] iter {iteration:3d} new best = {best_val:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 2️⃣  Final fast hill‑climbing refinement (uses the remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb(best_a, n,
                                  deadline=deadline,
                                  start_time=start_time,
                                  best_a=best_a,
                                  best_val=best_val)

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation (just for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
