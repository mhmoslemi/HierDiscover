def _lbfgs_objective(w, eps, n):
    a = np.exp(w - np.max(w))
    a = a / a.sum()
    smooth_val, grad_a = _smooth_max_and_grad(a, eps)
    val = 2.0 * n * smooth_val
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return val, grad_w

"""
Improved step‑function optimiser (≈ 1000 s budget)

The optimiser proceeds in the following stages:

1. **Broad random initialisation** – many β‑shaped and Dirichlet
   candidates are sampled; the best one (lowest exact objective)
   is kept.  This replaces the tiny 120‑sample sweep of the previous
   version.

2. **Continuation L‑BFGS on a smooth surrogate** – the objective
   ``2·n·max(conv(a,a))`` is approximated by a log‑sum‑exp (smooth‑max)
   with a temperature ε.  We minimise the smooth surrogate in the
   log‑softmax (logits) space using SciPy's L‑BFGS‑B optimiser.
   ε is annealed from 5 e‑1 down to 1 e‑12 (a geometric schedule);
   after each ε stage we keep the best vector and optionally perform
   a small random jitter + a short restart to escape shallow minima.

3. **Projected sub‑gradient polishing** – the true non‑smooth objective
   is polished with a simple sub‑gradient descent that projects onto
   the probability simplex after each step.

4. **Uniform blending** – a final convex combination with the uniform
   distribution is tried to smooth tiny spikes that may survive the
   polishing phase.

All stages respect the 1000 s wall‑time limit; when the budget is
exhausted the current best vector is returned.  The returned list
contains non‑negative numbers (the simplex projection guarantees this)
and can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective ( Σa = 1 is not required – the ratio is scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2·n‑1
    max_b = b.max()

    # soft‑max weights (log‑sum‑exp trick)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2·n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2·n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries centred
    grad_a = 4.0 * n * segment[::-1]             # already contains 2·n factor

    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Blend with the uniform distribution (helps to smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta family candidate (cheap initial sweep)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# L‑BFGS objective in log‑softmax (logits) space
# ----------------------------------------------------------------------
def _lbfgs_objective(w: np.ndarray, eps: float, n: int) -> tuple[float, np.ndarray]:
    """
    Returns (value, grad) of the smoothed objective 2·n·smooth_max
    with respect to the logits w (log‑softmax parametrisation).
    """
    # decode logits to a probability vector on the simplex
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    smooth_val, grad_a = _smooth_max_and_grad(a, eps)   # grad_a already 2·n·…
    val = 2.0 * n * smooth_val

    # chain rule through the soft‑max mapping
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return val, grad_w


# ----------------------------------------------------------------------
# Simple sub‑gradient step (exact max)
# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray,
                      step: float) -> tuple[np.ndarray, float]:
    """
    One projected sub‑gradient update for the exact objective.
    Returns the new vector and the current max‑convolution value.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    max_idx = np.where(b == max_b)[0]

    grad = np.zeros_like(a)
    for k in max_idx:
        # indices i with 0 ≤ i < n and 0 ≤ k‑i < n
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_range = np.arange(i_min, i_max + 1, dtype=np.int64)
        j = k - i_range
        grad[i_range] += 2.0 * a[j]

    a_new = a - step * grad
    a_new = _project_simplex(a_new)
    return a_new, max_b
