"""
Propose a candidate step‑function that (hopefully) beats the current best
≈ 1.5082.  The strategy is:

1.  Work on the probability simplex (∑aᵢ = 1) – the objective is
    scale‑invariant.
2.  Generate a diverse set of “prototype’’ sequences (Beta‑shapes,
    exponentials, √‑differences, interpolated previous solution, etc.).
    The best prototype is selected by a fast FFT‑based evaluation
    (2·n·max(conv)).
3.  Starting from that prototype run a simple entropy‑mirror‑descent
    (sub‑gradient) optimisation on the exact max‑convolution.  The update
    is a = a·exp(‑η·g) with a decreasing step‑size η.  The sub‑gradient
    g is the (averaged) gradient of the active convolution entry/entries.
4.  Occasional jitter is added if progress stalls.
5.  The best sequence found is finally checked with the provided
    ``evaluate_sequence`` wrapper and returned.

All heavy work (full self‑convolution) is performed with an FFT of
length power‑of‑two, i.e. O(n log n).  With n = 8000 the whole search
comfortably fits into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself using an FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Fast objective for a probability vector (sum==1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) – the exact value of evaluate_sequence
    when the vector is normalised to sum 1."""
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution entry (averaged over all active ks)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Compute the sub‑gradient of max(conv) with respect to a (entropy
    mirror‑descent style).  If several entries attain the maximum they are
    averaged."""
    n = a.shape[0]
    max_val = conv.max()
    # indices where the maximum is attained
    ks = np.where(conv == max_val)[0]
    g = np.zeros_like(a)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # i runs from lo … hi, contributing a[k−i] to the gradient
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] += 2.0 * a[k - idx]
    # average if more than one active constraint
    if len(ks) > 1:
        g /= float(len(ks))
    return g


# ----------------------------------------------------------------------
# Simple helper to generate a Beta‑shaped probability vector
# ----------------------------------------------------------------------
def _beta_shape(p: float, q: float, n: int) -> np.ndarray:
    """Return a normalised Beta(p,q) density on {0,…,n‑1} (discrete)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Exponential‑tail shape
# ----------------------------------------------------------------------
def _exp_shape(lam: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# sqrt‑difference shape  (discrete derivative of sqrt)
# ----------------------------------------------------------------------
def _sqrt_diff_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a /= total
    return a

    stagnation = 0
    best_iter = 0

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # full convolution and exact objective
        conv = _full_convolution(a)
        cur_obj = 2.0 * n * conv.max()

        # keep track of the best seen during refinement
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = it
            stagnation = 0
            print(f"[refine] iter {it:5d} new best O={best_obj:.6f}")
        else:
            stagnation += 1

        # sub‑gradient of the max entry (averaged if several are active)
        g = _subgradient_max(a, conv)

        # step‑size schedule (linear decay)
        eta = eta_start * (1.0 - it / max_iters) + eta_end * (it / max_iters)

        # entropy‑mirror update (guarantees positivity)
        a = a * np.exp(-eta * g)

        # renormalise to stay on the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional jitter if we are stuck
        if stagnation >= 1000:
            jitter = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            stagnation = 0
            print(f"[jitter] iter={it} added random noise")

    # ------------------------------------------------------------------
    # Final security‑checked evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(best O recorded = {best_obj:.6f})")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
