def _optimize_l_bfgs(a0, lam, deadline, start_time):
    n = a0.shape[0]
    # transform to z param
    eps = 1e-12
    z0 = np.log(np.maximum(a0, eps))
    def fun(z):
        a = np.exp(z)
        s = a.sum()
        a = a / s
        conv = _full_convolution(a)
        max_c = conv.max()
        # smooth approximated: use log-sum-exp
        # return smooth = (max_c) considered? Actually we can use max_c directly as objective for L-BFGS; it's not differentiable.
        # Instead use smooth approx:
        lam_local = lam
        exps = np.exp(lam_local * (conv - max_c))
        smooth_val = max_c + (1/lam_local) * np.log(np.sum(exps))
        # gradient: compute w = exps / sum exps
        w = exps / np.sum(exps)
        # compute gradient wrt a:
        size2 = 1 << ((3 * n - 3).bit_length())
        fa_rev = np.fft.rfft(a[::-1], size2)
        fw = np.fft.rfft(w, size2)
        conv_aw = np.fft.irfft(fa_rev * fw, size2)[:3 * n - 2]
        grad_a = 2.0 * conv_aw[n-1:n-1+n]
        # chain rule
        grad_z = a * (grad_a - np.dot(grad_a, a))
        return smooth_val, grad_z
    # call scipy minimize with jac=True
    res = minimize(fun, z0, method='L-BFGS-B', jac=True, options={'maxiter': 200, 'disp': False})
    # get a
    a_opt = np.exp(res.x)
    a_opt = a_opt / a_opt.sum()
    return a_opt, res.fun  # smoothval but we compute actual objective later

"""
Improved search for the step‑function that minimises
``evaluate_sequence`` (the objective 2·n·max(conv)/sum²).

Main ideas
-----------

1️⃣  **Soft‑max (log‑sum‑exp) surrogate** – we minimise a smooth
    approximation of the max‑convolution using L‑BFGS‑B on the
    *softmax* (log‑sum‑exp) with a large λ (≈80).  The variables are
    kept on the simplex by a softmax‐parameterisation `a = softmax(z)`.

2️⃣  **Multi‑seed strategy** – we start the optimisation from a
    diverse set of initial points (uniform, interpolated previous
    result, a few Dirichlet draws with different concentration and the
    best Beta‑family sweep).

3️⃣  **Final hill‑climbing** – the best candidate is refined with the
    intensive annealing hill‑climber from the reference implementation.

All phases respect the 1000 s budget; the best sequence found is
returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (for logging only)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)).  Assumes Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Beta‑family sweep (cheap dense exploration)
# ----------------------------------------------------------------------
def _beta_family_best(n: int):
    """Search the beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf

    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. *z*
# (the softmax parameters).  The gradient w.r.t. *a* is
#   grad_a = 2· (a_rev * w)
# where w is the softmax weight on the convolution entries.
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(z: np.ndarray, n: int, lam: float):
    """
    Returns (value, grad) for the smoothed max objective
    f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)  with a = softmax(z).
    """
    # ---- a = softmax(z) ------------------------------------------------
    max_z = np.max(z)
    expz = np.exp(z - max_z)          # stable
    a = expz / expz.sum()             # Σa = 1, a ≥ 0

    # ---- convolution a*a ------------------------------------------------
    size1 = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, size1)
    conv = np.fft.irfft(fa * fa, size1)[: 2 * n - 1]
    max_c = conv.max()

    # ---- softmax weights on the convolution ---------------------------
    exps = np.exp(lam * (conv - max_c))          # already shifted by max_c
    w = exps / exps.sum()                        # length 2·n‑1, sum to 1

    # ---- gradient ∂f/∂a -------------------------------------------------
    size2 = 1 << ((3 * n - 3).bit_length())      # for a_rev * w
    fa_rev = np.fft.rfft(a[::-1], size2)
    fw = np.fft.rfft(w, size2)
    conv_aw = np.fft.irfft(fa_rev * fw, size2)[: 3 * n - 2]
    grad_a = 2.0 * conv_aw[n - 1 : n - 1 + n]    # slice of length n

    # ---- chain rule to get ∂f/∂z (softmax Jacobian) --------------------
    dot = np.dot(grad_a, a)                      # scalar
    grad_z = a * (grad_a - dot)                  # shape (n,)

    # ---- smooth objective value (log‑sum‑exp) -------------------------
    smooth_val = max_c + (1.0 / lam) * np.log(np.sum(exps))

    return smooth_val, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the softmax parameters (no closures)
# ----------------------------------------------------------------------
def _run_l_bfgs(a0: np.ndarray,
                lam: float,
                max_iter: int,
                n: int,
                deadline: float) -> np.ndarray:
    """
    Optimises the smoothed max using L‑BFGS‑B.
    Returns a new probability vector (Σ=1, a_i≥0).
    """
    # initialise z = log(a0) (softmax trick)
    eps = 1e-14
    z0 = np.log(np.maximum(a0, eps))

    # objective wrapper for scipy (expects (fun, grad) together)
    def fun(z):
        return _smooth_obj_and_grad(z, n, lam)

    # run the optimiser
    res = minimize(fun,
                   z0,
                   method='L-BFGS-B',
                   jac=True,
                   options={'maxiter': max_iter, 'disp': False})

    # recover a from the optimiser's final z
    max_z = np.max(res.x)
    expz = np.exp(res.x - max_z)
    a_opt = expz / expz.sum()
    return a_opt


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        # indices that contribute to the current maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # pick a target index that contributes little
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # update the convolution efficiently (see original code)
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

            seeds.append(beta_a)
            print(f"[init] beta‑family seed (val={beta_val:.6f})")
    except Exception as exc:
        print("[init] beta family failed:", exc)

    # d) Random Dirichlet draws with assorted concentration parameters
    for alpha in [0.3, 0.7, 1.5, 3.0, 6.0]:
        seeds.append(np.random.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # 1️⃣ L‑BFGS optimisation on the smoothed‑max surrogate
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    time_for_lbgfs = (deadline - time.time()) * 0.55   # roughly 55 % of remaining budget
    per_seed = time_for_lbgfs / max(1, len(seeds))

    for i, a_start in enumerate(seeds):
        if time.time() >= deadline:
            break
        # a quick low‑λ pass to get into a good basin
        low_lambda = 5.0
        a_mid = _run_l_bfgs(a_start, lam=low_lambda, max_iter=120, n=n, deadline=deadline)

        # a tighter high‑λ refinement
        high_lambda = 80.0
        a_opt = _run_l_bfgs(a_mid, lam=high_lambda, max_iter=260, n=n, deadline=deadline)

        cur_val = evaluate_sequence(a_opt.tolist())
        print(f"[L‑BFGS seed {i}] value = {cur_val:.6f}")
        if cur_val < best_val:
            best_val, best_a = cur_val, a_opt.copy()
        if best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Final intensive hill‑climbing (remaining time slice)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and best_a is not None:
        best_a, best_val = _hillclimb_anneal(
            best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final secure evaluation (only for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return the raw list – the harness will apply the security checks.
    return best_a.tolist()
