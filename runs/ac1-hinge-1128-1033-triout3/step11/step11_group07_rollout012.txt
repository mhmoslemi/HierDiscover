"""
Improved step‑function optimiser for the L∞ auto‑convolution objective.

The algorithm proceeds in four stages:

1. **Extensive random seeding** – a large pool of candidates from several
   parametric families (Beta, power‑law, exponential, logistic, Dirichlet)
   is generated and the best one (according to the exact objective) is kept.
   This pool is an order of magnitude larger than the previous implementation.

2. **Fast Adam‑style optimisation** – the smooth log‑sum‑exp approximation
   of the max‑convolution is minimised on the probability simplex.
   The smoothing parameter ε and the learning‑rate are annealed geometrically,
   and a simple jitter strategy is used to escape plateaus.  The optimizer
   works in log‑softmax coordinates so the simplex constraints are satisfied
   automatically.

3. **Exact sub‑gradient polishing** – after the smooth phase a projected
   sub‑gradient descent on the true (non‑smooth) objective is performed.
   The gradient is obtained via a convolution with a binary mask that marks
   the active (maximal) convolution entries, which makes each iteration cheap.

4. **Uniform blending** – a final convex combination with the uniform
   distribution is tried to smooth possible spikes.

The code respects the 1000 s budget, automatically stops when time runs out,
and returns a plain list of non‑negative floats whose sum is 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution primitives (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient (includes 2·n factor)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp weights (stable)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # gradient = 2·n * 2 * (conv of a with reversed p)
    # the factor 4·n appears because each convolution entry contributes twice
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 3n‑2
    # extract the part that corresponds to the derivative w.r.t. a
    seg = conv_ab[n - 1:2 * n - 1]                # n entries
    grad = 4.0 * n * seg[::-1]                   # already contains 2·n

    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max (aggressive schedule)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for the smoothed objective.
    The smoothing ε is annealed from 0.5 to 1e‑12 and the learning
    rate from 0.5 down to 0.005.
    """
    # log‑softmax parametrisation
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.005
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 1200   # restart earlier than the previous version

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam]  new best = {best_val:.6f}")
        else:
            idle += 1

        # scheduling
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max gradient
        _, grad = _smooth_max_and_grad(a, eps)

        # chain rule through the soft‑max (as in standard softmax‑Adam)
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.015:
            w += np.random.normal(scale=0.04, size=n)

        # restart if idle for too long
        if idle > MAX_IDLE:
            print("[Adam]  restart after idle")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (vectorised via convolution)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a))
    using a convolution‑based (vectorised) sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        active = (b == max_b).astype(float)      # binary mask, length 2n‑1
        # normalize the mask to obtain a convex combination of active gradients
        active /= active.sum()

        # sub‑gradient = 2 * conv(active, a) restricted to the n entries that matter
        # (the same extraction trick as in the smooth case)
        active_rev = active[::-1]
        conv_ag = _conv_fft(a, active_rev)          # length 3n‑2
        seg = conv_ag[n - 1:2 * n - 1]               # n entries
        grad = 2.0 * seg[::-1]                      # factor 2 from derivative

        # linear step‑size schedule (decays to a tiny value)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - progress) + 1e-6 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # tiny random jitter every few hundred steps
        if it % 400 == 0 and np.random.rand() < 0.025:
            a = np.clip(a + np.random.normal(scale=0.015, size=n), 0.0, None)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 23):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generators -------------------------------------------------
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑shaped distribution (i^p‑1 (n‑i+1)^q‑1) on the simplex."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # sample p<1, q>1 more often (decreasing shapes), but allow any sign
    logp = rng.uniform(-6.5, 2.0)
    logq = rng.uniform(-6.5, 2.0)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_powerlaw_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (decreasing power‑law)."""
    p = rng.uniform(0.2, 1.3)          # exponent, 0 → uniform, >1 → very steep
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a /= a.sum()
    return a


def _random_exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ exp(-λ·i)."""
    lam = rng.uniform(0.0, 0.0012)    # 0 → uniform, larger → steeper decay
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def _random_logistic_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Logistic decreasing shape."""
    steep = rng.uniform(0.008, 0.04)   # controls how fast it drops
    shift = (n - 1) / 2
    x = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    a /= a.sum()
    return a


def _random_dirichlet_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet sample with a random concentration."""
    alpha = rng.choice([0.02, 0.05, 0.1, 0.3, 0.7, 1.5, 3.0])
    a = rng.dirichlet(np.full(n, alpha))
    return a


# ----------------------------------------------------------------------
# Main entry point ----------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a bit larger than before (more degrees of freedom)
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise – massive cheap search for a good seed
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)   # uniform as a fallback
    best_val = _objective(best_a)

    # 0.1  If a previous solution is available, interpolate it
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        if val < best_val:
            best_a, best_val = a0, val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # 0.2  Prepare a list of generator functions
    generators = [
        lambda: _random_beta_candidate(n, rng),
        lambda: _random_powerlaw_candidate(n, rng),
        lambda: _random_exponential_candidate(n, rng),
        lambda: _random_logistic_candidate(n, rng),
        lambda: _random_dirichlet_candidate(n, rng),
    ]

    # 0.3  Exhaustive random sweep (roughly 3000 candidates)
    MAX_INIT = 3000
    for _ in range(MAX_INIT):
        if time.time() >= deadline:
            break
        cand = rng.choice(generators)()
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] candidate improved, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – aggressive Adam optimisation (≈30 % of remaining budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * (deadline - start_time)
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – sub‑gradient polishing (≈50 % of remaining budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=400_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – a few quick Adam fine‑tunes with fresh restarts
    # ------------------------------------------------------------------
    while time.time() + 8.0 < deadline:   # leave a few seconds for final eval
        # start from a jittered copy of the current best
        jitter = best_a + np.random.normal(scale=0.01, size=n)
        jitter = np.clip(jitter, 0.0, None)
        jitter = _project_simplex(jitter)

        short_dead = time.time() + min(7.0, (deadline - time.time()) * 0.2)
        best_a, best_val = _adam_optimize(
            jitter, n,
            deadline=short_dead,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # return a plain list of floats
    return best_a.tolist()
