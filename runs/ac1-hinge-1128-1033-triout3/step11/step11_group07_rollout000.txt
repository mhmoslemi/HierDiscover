def propose_move(a, rng, move_frac=0.1):
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    while j == i:
        j = rng.integers(0, n-1)
    # amount to move
    max_move = move_frac * a[i]
    # ensure some positive
    if max_move <= 1e-12:
        return None  # skip
    delta = rng.random() * max_move
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return a_new

The algorithm uses a *simulated‑annealing* (SA) search that starts from the best
known construction (`height_sequence_1` if it exists, otherwise a random Beta
shape).  
At each step a small amount of mass is moved from one index to another,
the objective  

\[
F(a)=\frac{2\,n\; \max (a*a)}{(\sum a)^2}
\]

is evaluated by an FFT‑based convolution ( `O(n log n)` ), and the move is
accepted either because it improves the score or with a temperature‑dependent
probability.  
The temperature and the step size decay linearly with the elapsed time, so
the search is aggressive at the beginning and fine‑grained near the end.
Occasional global “jitter’’ (adding a tiny Gaussian noise and re‑projecting on
the probability simplex) helps to escape plateaus.

All heavy work is done inside a simple loop; with `n = 8000` a single
iteration costs roughly a millisecond, leaving plenty of time for tens of
thousands of SA steps within the 1000 s budget.

The final candidate is checked with the provided ``evaluate_sequence`` to
guarantee a valid result.

```python
"""
Simulated‑Annealing optimiser for the step‑function problem.

The routine starts from the best known sequence (global variable
`height_sequence_1`) or from a random Beta‑shaped seed, then performs a
temperature‑driven random walk that moves a small amount of mass from one
position to another while keeping the total sum equal to one.  The objective
(2·n·max(conv(a,a))/(sum a)^2) is evaluated with an FFT‑based full
convolution.  Temperature and step size are linearly annealed; occasional
global jitter adds diversity and helps escape local minima.

The method is pure NumPy, uses no closures or lambdas, and respects the
1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution of a vector with itself (FFT)
# ----------------------------------------------------------------------
def _full_self_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the full linear convolution a * a using FFT of length `size`.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, size)
    conv = np.fft.ifft(fa * fa, size)          # complex but real‑valued
    # irfft would give the same result and be a bit faster; keep explicit
    # to avoid any hidden complex‑to‑real conversion subtleties.
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum of a is kept = 1 -> denominator = 1)
# ----------------------------------------------------------------------
def _objective_fast(a: np.ndarray, n: int, size: int) -> float:
    """
    Exact value of the objective 2·n·max(conv(a,a)) / (sum a)^2.
    The sum is expected to be 1, but the formula works for any positive sum.
    """
    # convolution via FFT
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, sum x = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random Beta‑shaped candidate (simple, cheap initialisation)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a normalised vector whose shape follows a Beta(p,q) distribution."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # approx 1e‑4 … 10 on log‑scale
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

# Main entry point required by the platform
# ----------------------------------------------------------------------

    """
    Simulated‑annealing optimiser.
    Returns a non‑negative list whose sum is 1 and that (hopefully) minimises
    ``evaluate_sequence``.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Determine problem size and initialise a starting vector
    # ------------------------------------------------------------------
    try:
        # use the solution discovered by the previous run, if available
        init_seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init_seq.shape[0]
        a = np.maximum(init_seq, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        print(f"[Init] Loaded height_sequence_1 (n={n})")
    except Exception:
        # fallback: a random Beta candidate of the default size
        n = 8000
        a = _random_beta_candidate(n, rng)
        print(f"[Init] Random Beta seed (n={n})")

    # pre‑compute FFT size (next power of two)
    size = 1 << ((2 * n - 1) - 1).bit_length()

    # current and best values
    cur_val = _objective_fast(a, n, size)
    best_val = cur_val
    best_a = a.copy()
    print(f"[Init] Initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Simulated annealing parameters
    # ------------------------------------------------------------------
    # Temperature schedule (linear from T_start to T_end)
    T_start = 5.0e-2
    T_end = 1.0e-9
    # Step‑size schedule (fraction of the source mass that can be moved)
    step_start = 0.4   # up to 40 % of a[i] at the very beginning
    step_end = 0.001   # very fine later

    iteration = 0
    idle_iters = 0
    MAX_IDLE = 200_000   # if we see no improvement for a long time, jitter

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress ∈ [0,1]
        progress = (now - start_time) / (deadline - start_time)

        # linearly annealed temperature and step size
        T = T_start * (1.0 - progress) + T_end * progress
        cur_step = step_start * (1.0 - progress) + step_end * progress

        # --------------------------------------------------------------
        # 2.1 pick two distinct indices and propose a move
        # --------------------------------------------------------------
        i = rng.integers(0, n)
        if a[i] <= 1e-15:
            # nothing to move from this index – skip this iteration
            continue
        # ensure j ≠ i
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # amount to transfer (bounded by cur_step * a[i])
        max_delta = cur_step * a[i]
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # apply tentative move (sum stays exactly 1)
        a[i] -= delta
        a[j] += delta

        # --------------------------------------------------------------
        # 2.2 evaluate new candidate
        # --------------------------------------------------------------
        new_val = _objective_fast(a, n, size)

        # --------------------------------------------------------------
        # 2.3 acceptance test
        # --------------------------------------------------------------
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis rule
            if T > 0 and rng.random() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            cur_val = new_val
            idle_iters = 0
            # record best ever seen
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA] iter {iteration:>7d}  new best = {best_val:.6f}")
        else:
            # reject: revert the move
            a[i] += delta
            a[j] -= delta
            idle_iters += 1

        iteration += 1

        # --------------------------------------------------------------
        # 2.4 occasional global jitter to shake out of plateaus
        # --------------------------------------------------------------
        if iteration % 5000 == 0:
            # add tiny Gaussian noise, re‑project onto simplex
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_val = _objective_fast(a, n, size)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[SA] jitter improved best = {best_val:.6f}")

        # --------------------------------------------------------------
        # 2.5 long idle → random restart (helps when we are stuck)
        # --------------------------------------------------------------
        if idle_iters >= MAX_IDLE:
            a = _random_beta_candidate(n, rng)
            a = _project_simplex(a)            # guarantees sum = 1
            cur_val = _objective_fast(a, n, size)
            print(f"[SA] idle restart at iter {iteration}, value = {cur_val:.6f}")
            idle_iters = 0

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation (required by the platform)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] secure score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return a plain Python list of floats
    return best_a.tolist()
```
