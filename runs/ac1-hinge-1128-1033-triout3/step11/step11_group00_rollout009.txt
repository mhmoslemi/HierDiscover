"""
Propose a candidate step function that (approximately) minimizes
evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))².

The algorithm consists of three phases:

1. **Warm‑start** – build a probability vector of length *n* (default 20 000)
   either by interpolating the previous best solution (global ``height_sequence_1``)
   or from a smooth Beta(2,2) shape.

2. **Soft‑max Adam optimisation** – minimise a smooth log‑sum‑exp surrogate
   of the max‑convolution entry.  The surrogate smoothness parameter α is
   annealed from 1 to ≈800, while the Adam learning‑rate is annealed from
   0.20 to 0.005.  A simple jitter mechanism avoids long stagnation periods.

3. **Exact‑max mirror‑descent polishing** – starting from the best vector
   found in phase 2, perform many (default 5000) entropy‑mirror‑descent
   steps on the true objective ``max(conv)``.  The step size η decays linearly.

All operations use FFT‑based full linear convolutions (O(n log n)).  The
procedure respects the 1000 s global time budget; the final sequence is
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (reuse from the reference implementation)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two (possibly different‑length) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Entropy‑mirror‑descent polishing on the exact max entry
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 5000, eta_start: float = 0.55) -> np.ndarray:
    """
    Perform ``steps`` mirror‑descent updates on the exact max‑convolution entry.
    The update is a multiplicative‑weights step: a ← a·exp(-η·g) with g the
    sub‑gradient of the max entry.  The step size η decays linearly.
    """
    n = a.shape[0]
    a = np.copy(a)
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # sub‑gradient of the max entry w.r.t a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            # contribution 2·a_{k-idx} (see derivation in the reference code)
            g[idx] = 2.0 * a[k - idx]
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that aims to minimise
    ``evaluate_sequence``.  The routine respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.8   # safety margin for final call

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more degrees of freedom while keeping
    # FFT size modest (next power‑of‑two stays 2¹⁶ = 65536 for n ≤ 30 000).
    # ------------------------------------------------------------------
    n = 20000

    # ------------------------------------------------------------------
    # Initialise a (probability vector).  Interpolate a previous solution if
    # available; otherwise fall back to a centred Beta(2,2) density.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        x_new = np.linspace(0.0, 1.0, n)
        x_old = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(x_new, x_old, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum() if a0.sum() > 0 else 1.0 / n
        print("[init] interpolated previous solution (n={})".format(n))
    except Exception:
        i = np.arange(1, n + 1, dtype=np.float64)
        p = q = 2.0
        a0 = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        a0 = a0 / a0.sum()
        print("[init] generated Beta(2,2) start (n={})".format(n))

    # Work in the log‑domain to enforce a_i > 0 automatically.
    eps = 1e-30
    z = np.log(a0 + eps)

    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Learning‑rate schedule (from 0.20 down to 0.005)
    lr_start, lr_end = 0.20, 0.005
    # Smoothing (soft‑max) parameter schedule (α from 1 to ≈800)
    alpha_start, alpha_end = 1.0, 800.0

    best_a = None
    best_obj = np.inf
    best_iter = 0
    stagnation = 0
    it = 0

    # ------------------------------------------------------------------
    # Main Adam optimisation loop (soft‑max surrogate of the max‑convolution)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        now = time.time()
        prog = (now - start) / (deadline - start)          # 0 → 1

        # schedule learning rate and smoothing parameter
        lr = lr_start * (1.0 - prog) + lr_end * prog
        alpha = alpha_start + (alpha_end - alpha_start) * (prog ** 1.3)

        # --------------------------------------------------------------
        # Recover a from logits and re‑normalise to the simplex
        # --------------------------------------------------------------
        a = np.exp(z)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # --------------------------------------------------------------
        # Full self‑convolution (used for both the soft‑max and exact obj)
        # --------------------------------------------------------------
        conv = _full_convolution(a)               # length 2·n‑1

        # --------------------------------------------------------------
        # Soft‑max of the convolution (stable log‑sum‑exp)
        # --------------------------------------------------------------
        maxc = conv.max()
        exp_scaled = np.exp(alpha * (conv - maxc))   # α·(conv−maxc) is safe
        sumexp = exp_scaled.sum()
        weights = exp_scaled / sumexp                # Σ weights = 1
        # smooth approximation of 2·n·max(conv)
        obj_smooth = 2.0 * n * (maxc + np.log(sumexp) / alpha)

        # --------------------------------------------------------------
        # Track the best exact objective seen so far
        # --------------------------------------------------------------
        exact_obj = 2.0 * n * conv.max()
        if exact_obj < best_obj - 1e-12:
            best_obj = exact_obj
            best_a = a.copy()
            best_iter = it
            stagnation = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}  t={now-start:.1f}s")
        else:
            stagnation += 1

        # --------------------------------------------------------------
        # Gradient of the soft‑max surrogate w.r.t. a
        #   ∂/∂a_i Σ_k w_k (a * a)_k = 4·n·(rev_a ⋆ w)[i]
        # --------------------------------------------------------------
        rev_a = a[::-1]
        conv_rev = _full_convolution_two(rev_a, weights)   # length 3·n‑2
        g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]           # slice of length n

        # --------------------------------------------------------------
        # Chain rule through the softmax (a = softmax(z))
        # --------------------------------------------------------------
        dot = np.dot(g_a, a)
        grad_z = a * (g_a - dot)

        # --------------------------------------------------------------
        # Adam update on the logits
        # --------------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------------------------------------------------------------
        # Simple jitter to escape plateaus
        # --------------------------------------------------------------
        if stagnation >= 5000:
            z += np.random.normal(scale=0.015, size=n)
            stagnation = 0
            print(f"[reset] jitter added at it={it}")

    # ------------------------------------------------------------------
    # Polishing phase – entropy‑mirror descent on the exact max entry
    # ------------------------------------------------------------------
    if best_a is not None:
        print("[polish] start entropy‑mirror refinement")
        a_refined = _max_refine(best_a, steps=5000, eta_start=0.55)
        conv_ref = _full_convolution(a_refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = a_refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation using the provided wrapper
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else a0.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} "
              f"(approx O={best_obj:.6f})")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
