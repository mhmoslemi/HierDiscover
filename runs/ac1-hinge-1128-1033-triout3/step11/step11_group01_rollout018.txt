def proj_simplex(v):
    # Projects onto simplex {x: x>=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    """Projection of vector v onto the probability simplex."""
    # shift to handle negatives
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1

"""
Greedy Sidon‑like construction.

The evaluation function is  2·n·max(a*a) / (sum(a))² .
For a non‑negative vector a with Σa = 1 the denominator is fixed, therefore we
must make the maximal entry of the self‑convolution as small as possible.
If the indices of the non‑zero entries form a set in which each possible sum
(i+j) is used at most L times, then for equal weights w = 1/k (k = #indices)

    max(a*a) ≤ 2·L / k²

and the objective becomes  2·n·max ≤ 4·n·L / k² .
For L=2 the theoretical capacity is k≈250 for n=8000, which yields an
objective ≈1.0 – well below the target 1.508.

The algorithm builds such a set greedily:
* keep a counter `sum_counts[s]` for every possible sum s∈[0,2·n‑2];
* repeatedly draw a random index i and insert it iff all new sums
  (i+i and i+existing) would keep `sum_counts` ≤ L;
* stop after a configurable number of consecutive failures.

We try several values of L (1,2,3) and many random restarts, keep the best
vector (zeros everywhere, weight 1/k on the selected positions) and finally
return it.  The whole search respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex {y ≥ 0, Σy = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    # Duchi et al. (2008) algorithm
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Greedy construction of a set with bounded sum multiplicities
# ----------------------------------------------------------------------
def _greedy_set(n: int, L: int, max_failures: int = 20000) -> list[int]:
    """
    Build a set S ⊂ {0,…,n‑1} such that every sum i+j (including i=i) is used
    at most L times.  Returns the list of selected indices.
    """
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
    positions: list[int] = []
    failures = 0

    # random order of candidates – we reshuffle when exhausted
    candidates = np.arange(n, dtype=np.int32)
    np.random.shuffle(candidates)
    ptr = 0

    while failures < max_failures:
        if ptr >= n:                     # start a new permutation
            np.random.shuffle(candidates)
            ptr = 0
        i = int(candidates[ptr])
        ptr += 1

        # check the self‑sum
        if sum_counts[i + i] >= L:
            failures += 1
            continue

        if positions:
            # sums with already chosen positions
            cur = i + np.array(positions, dtype=np.int32)
            if np.any(sum_counts[cur] >= L):
                failures += 1
                continue

        # all checks passed → accept i
        positions.append(i)
        sum_counts[i + i] += 1
        if positions[:-1]:
            cur = i + np.array(positions[:-1], dtype=np.int32)
            sum_counts[cur] += 1

        # reset failure counter after a successful insertion
        failures = 0

    return positions

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Generate a non‑negative step function (list of floats) that minimises
    `evaluate_sequence`.  The method greedily constructs a sparse vector with
    a bounded number of collisions between index sums (L = 1,2,3).  The best
    vector found within the 1000 s time budget is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    n = 8000                       # problem size used in previous runs
    best_a = np.full(n, 1.0 / n)   # uniform start (objective = 2.0)
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate quickly (uses the provided function)
    # ------------------------------------------------------------------
    def _eval_vec(a_vec: np.ndarray) -> float:
        return evaluate_sequence(a_vec.tolist())

    # ------------------------------------------------------------------
    # Try a few random restarts with different L (collision limits)
    # ------------------------------------------------------------------
    L_values = [1, 2, 3]                       # max allowed uses per sum
    attempts_per_L = 8                         # how many independent tries

    for L in L_values:
        for attempt in range(attempts_per_L):
            now = time.time()
            if now >= deadline:
                break

            # greedy construction; allow many failures to approach capacity
            positions = _greedy_set(n, L, max_failures=30000)

            k = len(positions)
            if k < 2:
                continue

            # equal weight on the selected positions
            a_vec = np.zeros(n, dtype=np.float64)
            a_vec[positions] = 1.0 / k

            cur_val = _eval_vec(a_vec)

            if cur_val < best_val:
                best_val = cur_val
                best_a = a_vec.copy()
                print(f"[L={L} attempt={attempt}] new best = {best_val:.9f} "
                      f"with k={k}")

    # ------------------------------------------------------------------
    # Optional local refinement: simple sub‑gradient steps from the best
    # candidate found above (helps if the greedy set is not optimal).
    # ------------------------------------------------------------------
    def _refine(a_start: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
        a = a_start.copy()
        best_local = a.copy()
        best_local_val = _eval_vec(a)

        it = 0
        while time.time() - start_time < time_limit:
            it += 1
            conv = _full_convolution(a)
            kmax = int(np.argmax(conv))
            max_val = conv[kmax]

            # sub‑gradient of the objective (sum is fixed = 1)
            idx = np.arange(n)
            rev_idx = kmax - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            rev = np.zeros_like(a)
            rev[mask] = a[rev_idx[mask]]

            # step size decreases with the iteration count
            step = 0.5 / np.sqrt(it)
            a = a - step * rev
            a = _project_simplex(a)

            cur_val = 2.0 * n * max_val
            if cur_val < best_local_val - 1e-12:
                best_local_val = cur_val
                best_local = a.copy()
                # occasional progress report
                elapsed = time.time() - start_time
                print(f"[refine iter={it}] val={best_local_val:.9f} "
                      f"t={elapsed:.1f}s")
        return best_local, best_local_val

    # give the refinement a few seconds if time permits
    remaining = deadline - time.time()
    if remaining > 5.0 and best_val > 1.6:      # only if we still have room
        refined_a, refined_val = _refine(best_a, remaining * 0.6)
        if refined_val < best_val:
            best_val = refined_val
            best_a = refined_a.copy()
            print(f"[refine] improved to {best_val:.9f}")

    # ------------------------------------------------------------------
    # Final secure evaluation (the only place we call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.9f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
