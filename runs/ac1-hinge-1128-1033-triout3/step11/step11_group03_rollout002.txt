def _solve_lp_max_sum(f, M):
    n = f.shape[0]
    # Build A_ub
    idx_i = np.arange(n)
    row_idx = (idx_i[:, None] + np.arange(n)).ravel()
    col_idx = np.repeat(np.arange(n), n)
    data = np.tile(f, n)  # length n*n
    A_ub = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64) # maximize sum => minimize -sum
    # solve
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                bounds=[(0, None)]*n,
                                method='highs')
    if not res.success:
        return None, None
    g_opt = res.x
    sum_g = g_opt.sum()
    return g_opt, sum_g

"""
Propose a step‑function sequence with a low max‑self‑convolution.
Algorithm
---------

1.  Choose a modest length `n` (a few hundred to a few thousand).  
    The objective is `2·n·max(conv(a,a))` when `sum(a)=1`.  Smaller `n`
    gives a much looser bound, making the problem easier.

2.  Initialise several candidates:
    * interpolation of the previous best solution (if available),
    * a few draws from a Dirichlet distribution,
    * a simple Gaussian‑shaped seed.

3.  For each seed run a *convex‑concave* loop:
    * Let `f` be the current sequence, `M = max(conv(f,f))`.
    * Solve a linear program  
          maximise   Σ g_j                 (g has no sum constraint)  
          subject to conv(f,g) ≤ M  (2·n−1 linear inequalities)  
                     g ≥ 0 .
      The LP returns a feasible direction `g`.  Scale `g` to have the same
      ℓ₁‑norm as `f` (`sum = 1`).
    * Mix `f` and `g` by a scalar `t∈[0,1]`.  The convolution of the
      mixture is a quadratic function of `t`; we evaluate a small grid of
      `t` values and pick the one that gives the smallest `max(conv)`.
      If the mixture improves the bound we accept it and repeat the LP
      step; otherwise the local loop stops.
    * Occasionally jitter the current best point to jump out of plateaus.

   The LP is solved with ``scipy.optimize.linprog`` using a sparse
   Toeplitz matrix that encodes the convolution constraints.  The
   convolution operations themselves are performed with an FFT for
   speed (`O(n log n)`).

4.  After the LP‑driven phase a tiny hill‑climbing routine (mass
   transfer between indices that contribute most to the offending lag)
   polishes the best candidate.

5.  The best sequence found over all lengths and seeds is returned.
    The host evaluates it with ``evaluate_sequence``.

The whole search respects the 1000 s budget; it spends most of the time
on a handful of LP / mixing iterations (typically < 1 s) and
fine‑tunes the result with fast FFT‑based convolutions.
"""

import time
import numpy as np
from typing import List, Tuple

# SciPy is optional – the platform used for the competition provides it.
try:
    from scipy.optimize import linprog
    from scipy import sparse
except Exception:  # pragma: no cover
    linprog = None
    sparse = None

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _cross_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (full length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _interpolate_previous(prev: np.ndarray, n: int) -> np.ndarray | None:
    """Resize a previous solution to length `n` by linear interpolation."""
    try:
        old_len = prev.shape[0]
        old_x = np.linspace(0.0, 1.0, old_len)
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, prev)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
        return a
    except Exception:
        return None


def _objective_from_max(max_conv: float, n: int) -> float:
    """Exact value of the evaluation function when sum(a)=1."""
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Linear‑programming step
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> Tuple[np.ndarray | None, float]:
    """
    Given a non‑negative vector `f` with sum 1 and a bound `M`,
    solve the LP:
          maximise   Σ g_j
          subject to conv(f,g) ≤ M
                     g ≥ 0 .
    Returns the optimal `g` and its sum (or (None,0) on failure).
    """
    if linprog is None or sparse is None:  # pragma: no cover
        return None, 0.0

    n = f.shape[0]

    # Build the Toeplitz‑type constraint matrix A (shape (2n−1) × n):
    #   A[k, j] = f[k‑j]  if 0 ≤ k‑j < n  else 0
    # For each column j, the vector f is placed starting at row j.
    row_idx = (np.arange(n)[:, None] + np.arange(n)).ravel()
    col_idx = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                 # maximise Σg  → minimise -Σg

    res = linprog(c,
                  A_ub=A,
                  b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return None, 0.0

    g = res.x
    return g, g.sum()


# ----------------------------------------------------------------------
# Mixing helper – searches a few mixing ratios and picks the best
# ----------------------------------------------------------------------
def _best_mixture(f: np.ndarray,
                  g: np.ndarray,
                  conv_f: np.ndarray,
                  conv_g: np.ndarray,
                  conv_fg: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Returns the mixture (1‑t)·f + t·g that yields the smallest max‑convolution.
    A tiny grid of `t` values is examined.
    """
    n = f.shape[0]
    best_t = 0.0
    best_max = conv_f.max()
    # evaluate a small set of mixing coefficients
    for t in np.linspace(0.0, 1.0, 11):
        # conv of the mixture = (1‑t)^2·conv_f + 2t(1‑t)·conv_fg + t^2·conv_g
        combo = ((1 - t) ** 2) * conv_f + (2 * t * (1 - t)) * conv_fg + (t ** 2) * conv_g
        cur_max = combo.max()
        if cur_max < best_max:
            best_max = cur_max
            best_t = t

    a_new = (1 - best_t) * f + best_t * g
    a_new = np.maximum(a_new, 0.0)
    # re‑normalise (the mixing already preserves the ℓ₁‑norm, but rounding errors)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full(n, 1.0 / n)
    else:
        a_new /= s
    return a_new, best_max


# ----------------------------------------------------------------------
# Light‑weight hill‑climb that moves a small amount of mass away from the
# most “guilty’’ lag.
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                deadline: float,
                start_time: float,
                target_val: float = 1.503,
                max_no_improve: int = 150_000) -> Tuple[np.ndarray, float]:
    n = a.shape[0]
    best_a = a.copy()
    conv = _full_convolution(a)
    best_val = _objective_from_max(conv.max(), n)
    no_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions of each index to this lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # pick a source (heavy contributor) and a target (light contributor)
        if total > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total)
        else:
            src = int(np.argmax(contrib))

        if a[src] <= 1e-12:
            continue

        # target preference: low contribution
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        # move a random fraction (up to half) of the source mass
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        conv_new = _full_convolution(a_new)
        val_new = _objective_from_max(conv_new.max(), n)

        if val_new < best_val - 1e-12:
            a = a_new
            conv = conv_new
            best_val = val_new
            best_a = a_new.copy()
            no_improve = 0
            print(f"[Hill] iter {it:7d} val={best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            # random jitter to escape a plateau
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            conv = _full_convolution(a)
            best_val = _objective_from_max(conv.max(), n)
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s time
    budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Problem sizes – a few small–moderate lengths are cheap to optimise.
    sizes = [500, 800, 1200, 1600]

    # Grab any previous solution if it exists.
    try:
        previous = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    except Exception:
        previous = None

    global_best_seq = None
    global_best_val = np.inf
    global_best_n = None

    # ------------------------------------------------------------------
    # 1️⃣  Run the LP‑mixing loop for each size and a few seeds
    # ------------------------------------------------------------------
    for n in sizes:
        if time.time() >= deadline:
            break

        # ---- create a handful of starting points -----------------------
        starters = []

        # a) interpolated previous solution
        if previous is not None:
            a0 = _interpolate_previous(previous, n)
            if a0 is not None:
                starters.append(a0)

        # b) a few random Dirichlet draws
        for _ in range(3):
            starters.append(np.random.dirichlet(np.ones(n)))

        # c) a simple Gaussian‑shaped seed (often a good baseline)
        i = np.arange(n, dtype=np.float64)
        mid = (n - 1) / 2.0
        sigma = 0.25 * n
        a_gauss = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        a_gauss /= a_gauss.sum()
        starters.append(a_gauss)

        # ----------------------------------------------------------------
        #   Process every starter
        # ----------------------------------------------------------------
        for a0 in starters:
            if time.time() >= deadline:
                break

            a = a0.copy()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s

            conv_f = _full_convolution(a)
            M = conv_f.max()
            cur_val = _objective_from_max(M, n)

            best_local_a = a.copy()
            best_local_val = cur_val

            # ----  LP‑mixing iterations ----------------------------------
            for it_lp in range(12):   # a dozen iterations are usually enough
                if time.time() >= deadline:
                    break

                # Solve the LP to obtain a promising direction g
                g_opt, sum_g = _solve_lp_max_sum(a, M)
                if g_opt is None or sum_g == 0.0:
                    break

                # Scale g to keep the ℓ₁‑norm equal to 1
                g = g_opt / sum_g

                # Pre‑compute the three convolutions needed for mixing
                conv_g = _full_convolution(g)
                conv_fg = _cross_convolution(a, g)

                # Find the best mixing coefficient on a coarse grid
                a_new, new_M = _best_mixture(a, g, conv_f, conv_g, conv_fg)

                new_val = _objective_from_max(new_M, n)

                if new_val < cur_val - 1e-12:
                    # Accept improvement
                    a = a_new
                    conv_f = _full_convolution(a)
                    M = conv_f.max()
                    cur_val = new_val
                    if cur_val < best_local_val:
                        best_local_val = cur_val
                        best_local_a = a.copy()
                    print(f"[LP] n={n:5d} it={it_lp:2d} val={cur_val:.6f}")
                    if cur_val <= 1.503:
                        break
                else:
                    # No progress – stop the LP loop for this starter
                    break

            # ----  optional final hill‑climb per starter ------------------
            remaining = deadline - time.time()
            if remaining > 2.0 and best_local_val > 1.503:
                best_local_a, best_local_val = _hill_climb(
                    best_local_a,
                    deadline=deadline,
                    start_time=start_time,
                    target_val=1.503)

            # ----  Track the global best ---------------------------------
            if best_local_val < global_best_val:
                global_best_val = best_local_val
                global_best_seq = best_local_a.copy()
                global_best_n = n
                print(f"[Best] new global best n={n} val={global_best_val:.6f}")

            if global_best_val <= 1.503:
                break
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Fallback in the (unlikely) event we found nothing useful
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 800
        global_best_seq = np.full(n, 1.0 / n)
        global_best_val = _objective_from_max(_full_convolution(global_best_seq).max(), n)

    # ------------------------------------------------------------------
    # 3️⃣  Final host evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return global_best_seq.tolist()
