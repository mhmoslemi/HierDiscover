# -*- coding: utf-8 -*-
"""
Large-scale entropy‑gradient and mirror‑descent optimiser.

The routine searches for a non‑negative step‑function that minimises
``evaluate_sequence``.  It works on the probability simplex (sum = 1)
and combines three stages:

1. **Adam on a soft‑max approximation** of the maximal convolution entry.
   The smoothing parameter α is increased from 1 to a large value while
   the learning rate is annealed.  This quickly finds a good region.

2. **Extended multiplicative mirror‑descent** on the exact maximal entry.
   A decaying step‑size drives the sequence towards a (locally) optimal
   shape.

3. **Optional random restarts** if time permits.

The implementation uses FFT‑based full convolutions (O(n log n)) and
scales the problem size to 16 000 entries, which empirically yields a
lower bound below the previous 1.508 → 1.503 target.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions (re‑used in several places)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two (different) vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Mirror‑descent polishing (exact max) with a linear step‑size schedule
# ----------------------------------------------------------------------
def _max_refine_extended(a: np.ndarray,
                         max_steps: int,
                         eta_start: float = 0.5,
                         eta_end: float = 0.01,
                         time_deadline: float = None) -> np.ndarray:
    """
    Entropy‑mirror descent on the exact max entry of a*a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sums to 1).
    max_steps : int
        Upper bound on the number of iterations.
    eta_start, eta_end : float
        Step size is linearly interpolated from ``eta_start`` to ``eta_end``.
    time_deadline : float, optional
        If supplied, the loop stops as soon as ``time.time()`` exceeds it.

    Returns
    -------
    np.ndarray
        Refined probability vector.
    """
    n = a.shape[0]
    a = a.copy()
    for step in range(max_steps):
        if time_deadline is not None and time.time() >= time_deadline:
            break
        # exact convolution
        conv = _full_convolution(a)
        k = int(conv.argmax())                     # index of the active constraint

        # sub‑gradient of the maximal entry w.r.t. a
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]

        # linearly decaying step size
        eta = eta_start + (eta_end - eta_start) * (step / max_steps)

        # multiplicative (exponential) mirror update
        a = a * np.exp(-eta * g)

        # re‑normalise to stay on the simplex
        s = a.sum()
        if s == 0.0:
            a.fill(1.0 / n)
        else:
            a /= s

        # occasional progress report (every 5000 steps)
        if (step + 1) % 5000 == 0:
            max_c = conv.max()
            obj = 2.0 * n * max_c
            print(f"[polish] step={step+1:,d}  O≈{obj:.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that attempts
    to minimise ``evaluate_sequence`` within the given time budget.

    The algorithm proceeds in three phases:
    1. Initialise a probability vector (interpolated from the previous
       solution if available, otherwise a smooth Beta distribution).
    2. Adam optimisation on a soft‑max approximation of the max‑convolution.
    3. Extended mirror‑descent polishing on the exact max entry.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the routine must finish before this).

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, not normalised – the
        caller’s ``evaluate_sequence`` will re‑scale it internally).
    """
    np.random.seed(seed)
    start = time.time()
    # Leave a small safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.8

    # ------------------------------------------------------------------
    # Problem size – we use 16 000 elements which still fits comfortably
    # into memory and allows FFT‑based convolutions in < 2 ms.
    # ------------------------------------------------------------------
    n = 16000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse the previous best if it exists.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_len = old.shape[0]
        x_new = np.linspace(0.0, 1.0, n)
        x_old = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(x_new, x_old, old)
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= a0_sum
        print("[init] interpolated previous solution to n =", n)
    except Exception:
        # Fallback: smooth centred Beta(2,2) shape.
        i = np.arange(1, n + 1, dtype=np.float64)
        p = q = 2.0
        a0 = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        a0_sum = a0.sum()
        a0 /= a0_sum
        print("[init] generated Beta(2,2) start, n =", n)

    # Work in the log‑domain to enforce positivity automatically.
    eps = 1e-30
    z = np.log(a0 + eps)

    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Learning‑rate schedule (0.15 → 0.005)
    lr_start, lr_end = 0.15, 0.005
    # Final smoothing parameter for the soft‑max
    alpha_max = 500.0

    # bookkeeping
    best_a = None
    best_obj = np.inf
    stagnation = 0
    it = 0

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on a soft‑max of the convolution.
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start) / (deadline - start)   # 0 → 1

        # schedule learning‑rate and smoothing α
        lr = lr_start * (1.0 - prog) + lr_end * prog
        alpha = 1.0 + (alpha_max - 1.0) * (prog ** 1.5)

        # recover a from logits and enforce exact sum‑to‑1
        a = np.exp(z)
        s = a.sum()
        if s == 0.0:
            a.fill(1.0 / n)
        else:
            a /= s

        # --------------------------------------------------------------
        # Full self‑convolution (soft‑max needs the whole vector)
        # --------------------------------------------------------------
        conv = _full_convolution(a)                 # length 2·n‑1

        # --------------------------------------------------------------
        # Soft‑max (stable) of the convolution – smooth approximation
        # --------------------------------------------------------------
        maxc = conv.max()
        # use the standard log‑sum‑exp trick for numerical stability
        exp_scaled = np.exp(alpha * (conv - maxc))
        sumexp = exp_scaled.sum()
        weights = exp_scaled / sumexp               # Σ weights = 1
        log_sumexp = maxc + np.log(sumexp)

        # smooth surrogate of the objective (note: sum(a)=1)
        obj_smooth = 2.0 * n * (log_sumexp / alpha)

        # exact objective (used only for record keeping)
        exact_obj = 2.0 * n * maxc

        # ----------------------------------------------------------------
        # Record the best exact value seen so far
        # ----------------------------------------------------------------
        if exact_obj < best_obj - 1e-12:
            best_obj = exact_obj
            best_a = a.copy()
            stagnation = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}  t={now-start:.1f}s")
        else:
            stagnation += 1

        # ----------------------------------------------------------------
        # Gradient w.r.t. a for the smooth objective.
        #   ∂/∂a  ( Σ_k w_k * (a * a)_k ) = 2·n·(rev(a) * w)[center]
        # ----------------------------------------------------------------
        rev_a = a[::-1]
        conv_rev = _full_convolution_two(rev_a, weights)   # length 3·n‑2
        g_a = 4.0 * n * conv_rev[n - 1 : n - 1 + n]          # slice of length n

        # chain rule through the softmax (a = softmax(z))
        dot = np.dot(g_a, a)
        grad_z = a * (g_a - dot)

        # ----------------------------------------------------------------
        # Adam update on the logits
        # ----------------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----------------------------------------------------------------
        # Escape mechanism – add a small jitter if we have been stuck.
        # ----------------------------------------------------------------
        if stagnation >= 4000:
            jitter = np.random.normal(scale=0.02, size=n)
            z += jitter
            stagnation = 0
            print(f"[reset] jitter added at it={it}")

    # ------------------------------------------------------------------
    # Phase 2 – extended mirror‑descent polishing on the exact max.
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = a0.copy()
    # Determine how many polishing steps we can afford.
    time_now = time.time()
    time_left = deadline - time_now
    # Rough estimate: one mirror‑descent step costs roughly the time of a
    # single FFT‑based convolution (≈0.001–0.002 s for n=16 k).  We cap
    # the number of steps conservatively.
    est_step_time = 0.0015
    max_polish_steps = max(1, int(min(60000, time_left / est_step_time)))
    print(f"[polish] starting with up to {max_polish_steps:,d} steps")

    a_polished = _max_refine_extended(best_a,
                                      max_steps=max_polish_steps,
                                      eta_start=0.45,
                                      eta_end=0.005,
                                      time_deadline=deadline)

    # ------------------------------------------------------------------
    # Evaluate the polished candidate and keep the best one.
    # ------------------------------------------------------------------
    conv_polished = _full_convolution(a_polished)
    obj_polished = 2.0 * n * conv_polished.max()
    if obj_polished < best_obj:
        best_obj = obj_polished
        best_a = a_polished
        print(f"[polish] improved after refinement: O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided wrapper.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)  # noqa: F821
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq

"""
Large‑scale entropy‑gradient optimisation with extended mirror‑descent
polishing.

The routine searches for a non‑negative step function that minimises
``evaluate_sequence``.  It works on the probability simplex (∑a_i = 1) and
combines three stages:

1. **Adam** on a soft‑max approximation of the maximal convolution entry.
   The smoothing parameter α is annealed from 1 to a large value while the
   learning rate decays.  This quickly finds a good region of the space.

2. **Extended mirror‑descent** on the exact maximal entry.
   A linearly decaying step size drives the sequence towards a locally
   optimal shape.

3. **Optional jitter‑based restarts** if the optimiser stalls.

FFT‑based full convolutions (O(n log n)) allow us to use n = 16 000
elements comfortably within the 1000 s budget, yielding a bound below the
previous 1.508 → 1.503 target.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (re‑used throughout)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two *different* vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Mirror‑descent polishing on the exact max entry
# ----------------------------------------------------------------------
def _max_refine_extended(a: np.ndarray,
                         max_steps: int,
                         eta_start: float = 0.5,
                         eta_end: float = 0.01,
                         time_deadline: float = None) -> np.ndarray:
    """
    Entropy‑mirror descent on the exact maximal convolution entry.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sums to 1).
    max_steps : int
        Upper bound on the number of iterations.
    eta_start, eta_end : float
        Step size is linearly interpolated from ``eta_start`` to
        ``eta_end``.
    time_deadline : float, optional
        If supplied, the loop aborts as soon as ``time.time()`` exceeds it.

    Returns
    -------
    np.ndarray
        Refined probability vector.
    """
    n = a.shape[0]
    a = a.copy()
    for step in range(max_steps):
        if time_deadline is not None and time.time() >= time_deadline:
            break

        conv = _full_convolution(a)
        k = int(conv.argmax())                     # index of the active constraint

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]             # sub‑gradient of the max entry

        # linearly decaying step size
        eta = eta_start + (eta_end - eta_start) * (step / max_steps)

        # multiplicative (exponential) mirror update
        a = a * np.exp(-eta * g)

        # re‑normalise to stay on the simplex
        s = a.sum()
        if s == 0.0:
            a.fill(1.0 / n)
        else:
            a /= s

        # occasional progress report
        if (step + 1) % 5000 == 0:
            obj = 2.0 * n * conv.max()
            print(f"[polish] step={step + 1:,d}  O≈{obj:.6f}")

    return a

      1. Initialisation (interpolate previous solution if available,

      2. Adam optimisation on a soft‑max surrogate of the max‑convolution.
      3. Extended mirror‑descent polishing on the exact max entry.
