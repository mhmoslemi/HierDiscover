"""
Propose a new candidate step‑function.

Strategy
--------
1.  **Coarse search (fast, low‑dimensional).**  
    Scan a small Beta‑family on a moderate size (n≈2000) and a few
    Dirichlet draws.  Keep the best vector.

2.  **Upsample** the coarse optimum to a larger size (n≈8000) – the
    evaluation function is scale‑invariant, so the shape is what
    matters.

3.  **Smoothed gradient descent** on the exact objective.  The
    non‑smooth max‑convolution term is replaced by a log‑sum‑exp
    smoothing φ_β.  An Adam‑style update is used for faster
    convergence; β is increased exponentially until a large bound,
    turning the smooth problem into the original one.

4.  **Local random perturbations** (hill‑climbing) while time permits.

All steps respect the simplex (Σa=1, a≥0) which makes the output
compatible with ``evaluate_sequence``.  The routine runs for up to
the given time budget (default 1000 s) and returns the best sequence
found.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()               # next power of two
    fx = rfft(x, size)
    conv = irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a⋆a) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Return a normalized Beta‑shaped vector, or None if degenerate."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    return a / total


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smoothed max‑convolution objective
        φ_β(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i)

    Returns (grad, max_conv) where grad has shape a.shape.
    """
    n = a.shape[0]

    # ---- 1. full convolution a⋆a (length 2n‑1) ----
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:out_len]       # (a⋆a)

    max_conv = conv.max()

    # ---- 2. soft‑max weights ----
    exp_arg = beta * (conv - max_conv)          # ≤ 0, safe
    w = np.exp(exp_arg)
    w /= w.sum()                                 # Σ w_i = 1

    # ---- 3. gradient = 2·(a ⋆_cross w) ----
    rev_w = w[::-1]                              # reverse for cross‑correlation
    out_len_corr = n + out_len - 1               # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = rfft(a, size_corr)
    Fw = rfft(rev_w, size_corr)
    corr_full = irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1                         # lag 0 position
    idx = offset - np.arange(n)                  # indices for j = 0…n‑1
    grad = 2.0 * corr_full[idx]

    return grad, max_conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence discovered before
    the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # keep a safety margin

    # ------------------------------------------------------------------
    # 0️⃣  Problem dimensions
    # ------------------------------------------------------------------
    n_fine = 8000          # size used for the final refinement
    n_coarse = 2000        # cheap exploration size

    # ------------------------------------------------------------------
    # 1️⃣  Coarse search on a smaller problem
    # ------------------------------------------------------------------
    best_coarse = None
    best_val = np.inf

    # try to reuse a previously found construction if available
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_n = old.shape[0]
        x_new = np.linspace(0.0, 1.0, n_coarse)
        x_old = np.linspace(0.0, 1.0, old_n)
        a0 = np.interp(x_new, x_old, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0.0:
            a0 /= a0.sum()
            print("[init] interpolated previous solution (coarse)")
    except Exception:
        pass

    # a single deterministic start (interpolated previous or uniform)
    if a0 is not None:
        cand = a0
        val = _objective(cand)
        best_coarse, best_val = cand, val
        print(f"[init] start from interpolated solution, value={val:.6f}")

    # ----- grid over Beta family (very coarse) -----
    p_vals = np.geomspace(0.01, 10.0, 12)
    q_vals = p_vals.copy()
    for p in p_vals:
        for q in q_vals:
            cand = _beta_shape(p, q, n_coarse)
            if cand is None:
                continue
            val = _objective(cand)
            if val < best_val:
                best_val, best_coarse = val, cand
                print(f"[grid] new best = {best_val:.6f} (p={p:.3g}, q={q:.3g})")

    # ----- random beta draws -----
    rng = np.random.default_rng()
    for _ in range(1200):
        p = np.exp(rng.uniform(np.log(1e-3), np.log(20.0)))
        q = np.exp(rng.uniform(np.log(1e-3), np.log(20.0)))
        cand = _beta_shape(p, q, n_coarse)
        if cand is None:
            continue
        val = _objective(cand)
        if val < best_val:
            best_val, best_coarse = val, cand
            print(f"[rand] new best = {best_val:.6f} (p={p:.3g}, q={q:.3g})")

    # ----- a few Dirichlet draws for diversity -----
    for _ in range(8):
        cand = rng.dirichlet(np.ones(n_coarse))
        val = _objective(cand)
        if val < best_val:
            best_val, best_coarse = val, cand
            print(f"[dir] new best = {best_val:.6f}")

    # fallback
    if best_coarse is None:
        best_coarse = np.full(n_coarse, 1.0 / n_coarse)
        best_val = _objective(best_coarse)
        print("[fallback] uniform coarse start", best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Upsample to the fine resolution
    # ------------------------------------------------------------------
    x_fine = np.linspace(0.0, 1.0, n_fine)
    x_coarse = np.linspace(0.0, 1.0, n_coarse)
    a = np.interp(x_fine, x_coarse, best_coarse)
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # enforce the simplex

    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[upsample] objective = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smoothed gradient descent (Adam‑style)
    # ------------------------------------------------------------------
    beta = 0.5
    beta_max = 5000.0
    beta_mul = 1.01                # accelerate the approach to the true max
    step0 = 0.8                    # initial step size (scaled later)
    eps = 1e-8                     # Adam epsilon
    beta1, beta2 = 0.9, 0.999      # Adam momentum parameters

    m = np.zeros_like(a)           # first moment
    v = np.zeros_like(a)           # second moment
    t = 1                          # iteration counter (1‑based for bias correction)

    # schedule for occasional jitter (helps to escape plateaus)
    jitter_prob = 0.002
    jitter_scale = 5e-4

    while True:
        now = time.time()
        if now >= deadline:
            break

        # step size decays as 1/√t (Adam's own adaptation already
        # reduces it, but we keep an outer decay as well)
        step = step0 / np.sqrt(t)

        # gradient of the smoothed objective and current max‑conv
        grad, cur_max = _gradient_smooth(a, beta)
        cur_val = 2.0 * n_fine * cur_max

        # record the best point seen so far
        if cur_val < best_score:
            best_score = cur_val
            best_seq = a.copy()
            print(f"[grad] new best = {best_score:.6f} (iter {t}, β={beta:.2f})")
            # early exit if the target is already beaten
            if best_score <= 1.503:
                break

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_candidate = a - step * (m_hat / (np.sqrt(v_hat) + eps))
        a_candidate = _project_simplex(a_candidate)

        # occasional tiny random perturbation
        if np.random.rand() < jitter_prob:
            a_candidate += np.random.normal(scale=jitter_scale, size=n_fine)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

        a = a_candidate

        # increase the smoothing parameter
        beta = min(beta_max, beta * beta_mul)

        t += 1

    # ------------------------------------------------------------------
    # 4️⃣  Local random hill‑climbing (still inside the time budget)
    # ------------------------------------------------------------------
    epsilon = 2e-3                     # initial amount of mass to move
    while True:
        now = time.time()
        if now >= deadline:
            break
        # pick two distinct indices
        i, j = rng.choice(n_fine, size=2, replace=False)
        if a[i] < epsilon:
            continue
        delta = epsilon * rng.random()
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # (projection not needed – sum stays 1 and we keep non‑negativity)
        cur_val = _objective(a_new)
        if cur_val < best_score:
            best_score = cur_val
            best_seq = a_new.copy()
            a = a_new
            print(f"[local] improvement to {best_score:.6f}")
            # shrink the step size a bit after a success
            epsilon = max(epsilon * 0.95, 1e-5)
        else:
            # slowly decay epsilon even on failure to fine‑tune later
            epsilon = max(epsilon * 0.997, 1e-5)

    # ------------------------------------------------------------------
    # 5️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
