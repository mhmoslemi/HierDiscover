"""
Step‑function optimizer for `evaluate_sequence`.

The algorithm consists of four stages that fit into the 1000 s budget:

0) **Seed selection** – start from the previous construction
   (global variable `height_sequence_1`) if it exists, otherwise fall back
   to a quick sweep over several parametric families and a few random
   Dirichlet draws.

1) **Smooth‑max exponentiated gradient** – minimise the quantity
   `2·n·max(conv(a,a))` (the sum of the coefficients is fixed to 1).
   We replace the hard `max` by a soft‑max with temperature β and perform
   an exponentiated‑gradient update

        a ← a·exp(‑η·∇F),

   where the gradient ∇F of the smoothed objective is computed *exactly*
   by one FFT‑based convolution per iteration:

        w_i ∝ exp(β·(b_i – max(b)))        (soft‑max weights)
        ∇F = 2·conv(w, a[::-1])           (slice length n)

   The step‑size η decays from 0.15 to 5·10⁻⁴, while β grows from 1 to 30,
   gradually focusing the descent onto the true peak.

2) **Deterministic greedy hill‑climbing** – after the smooth phase we
   fine‑tune the solution by moving a small fraction of mass from the
   index that contributes most to the current peak to the index that
   contributes the least.  The transfer fraction anneals from 0.5 down to
   0.001.  Only strictly improving moves are accepted; after a long
   stagnation a tiny Gaussian jitter is added.

3) **Uniform mixing** – finally we test convex combinations with a tiny
   amount of the uniform vector; this occasionally reduces the final
   peak further.

All phases respect the overall time limit, keep the vector non‑negative,
renormalise after each modification and retain the best vector ever seen.
The function returns the best list of floats (Σa = 1) found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors `x` and `y`."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a (full convolution)."""
    return _full_convolution(a, a)


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _self_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Phase‑0 – quick families + random draws
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(arr):
        nonlocal best_a, best_val
        a = np.asarray(arr, dtype=np.float64)
        if a.shape != (n,):
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        v = _objective(a, n)
        if v < best_val:
            best_a, best_val = a.copy(), v

    rng = np.random.default_rng(12345)
    idx = np.arange(n)

    # deterministic families
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # linear U‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # quadratic decreasing
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # random Dirichlet draws (cheap)
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑1 – smooth‑max exponentiated gradient descent
# ----------------------------------------------------------------------
def _smooth_expgrad_phase(a0: np.ndarray, n: int,
                          start: float, deadline: float,
                          best_a: np.ndarray, best_val: float,
                          target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smoothed version of max(conv(a,a)).
    Uses soft‑max weights with temperature β that is increased over time.
    """
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 150_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # (1) evaluate current objective
        b = _self_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        # (2) record improvement
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SmoothEG {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # (3) compute soft‑max weights
        beta_start, beta_end = 1.0, 30.0
        progress = (now - start) / (deadline - start)
        beta = beta_start + (beta_end - beta_start) * progress

        # subtract max for numerical stability
        w = np.exp(beta * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:                     # extreme underflow – fallback to hard max
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # (4) gradient of the weighted sum = 2 * conv(w, a_rev)
        a_rev = cur[::-1]
        grad_full = _full_convolution(w, a_rev)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]                # slice of length n

        # (5) step‑size schedule (linear decay)
        eta = (1.0 - progress) * 0.15 + progress * 0.0005

        # (6) exponentiated update (keeps positivity)
        cur = cur * np.exp(-eta * grad)

        # (7) renormalise to Σa = 1
        total = cur.sum()
        if total <= 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # (8) occasional jitter if stuck
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.006, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[SmoothEG {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑2 – deterministic greedy hill‑climbing
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the index that contributes most to the
    current maximal convolution entry to the index that contributes the
    least.  The transferred fraction anneals from 0.5 → 0.001.
    """
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 250_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _self_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # annealing of the transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate index k where the peak occurs
        k = int(np.argmax(b))

        # build the vector of contributions of each a_i to the peak b_k
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]                #  a_{k-i}
        rev *= 2.0                              # sub‑gradient for the peak

        # source = index with largest contribution
        i = int(np.argmax(rev))

        # target = index with smallest contribution (ignore source)
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            # source exhausted – occasional jitter
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.004, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        # amount to move (random fraction of the allowed maximum)
        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # trial move
        cur[i] -= delta
        cur[j] += delta

        # evaluate trial
        b_new = _self_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # accept – the outer loop will record improvement on next iteration
            pass
        else:
            # reject – revert
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a′ = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[phase3] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check through the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
