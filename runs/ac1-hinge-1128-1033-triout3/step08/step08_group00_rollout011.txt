def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step function minimizing evaluate_sequence.
    Approach:
    1. Initialise with a set of parametric families (Beta, Gaussian, cosine, sine).
    2. Run Adam gradient descent on the smooth max objective with variance penalty.
    3. Finish with a simulated‑annealing hill‑climbing local search.
    The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine n; set to 12000 for balance.
    n = 12000

    # Try to get previous solution if available
    a0 = None
    if 'height_sequence_1' in globals():
        try:
            old = np.asarray(globals()['height_sequence_1'], dtype=float)
            old_len = old.shape[0]
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            print("[init] interpolated previous solution")
        except Exception:
            a0 = None

    best_a = None
    best_val = np.inf

    # evaluate a0 if exists
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print("[init] start from interpolated solution, val=%.6f" % best_val)

    # Phase 0: parametric families

    # Beta family sweep
    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_val < best_val:
            best_a, best_val = beta_a, beta_val
            print("[init] beta family improved val=%.6f" % best_val)
    except Exception as e:
        print("[init] beta family failed:", e)

    # Gaussian family sweep
    try:
        i = np.arange(n)
        # centre at middle
        mu = (n - 1) / 2.0
        # sigma range
        sigma_vals = np.geomspace(n/30.0, n/2.0, num=12)
        for sigma in sigma_vals:
            a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum
            val = _objective(a)
            if val < best_val:
                best_a, best_val = a.copy(), val
                print("[init] gaussian sigma=%.1f improved val=%.6f" % (sigma, best_val))
    except Exception as e:
        print("[init] gaussian sweep failed:", e)

    # Cosine (raised cosine) shape
    try:
        a = 1.0 - np.cos(2 * np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a.copy(), val
            print("[init] cosine shape improved val=%.6f" % best_val)
    except Exception as e:
        print("[init] cosine shape failed:", e)

    # Sine half-wave shape
    try:
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a.copy(), val
            print("[init] sine shape improved val=%.6f" % best_val)
    except Exception as e:
        print("[init] sine shape failed:", e)

    # Random Dirichlet seeds
    for _ in range(3):
        a = np.random.dirichlet(np.ones(n))
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a, val
            print("[init] random dirichlet improved val=%.6f" % best_val)

    # Ensure we have a start
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] random start, val=%.6f" % best_val)

    # Phase A: Adam
    phase_a_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # Phase B: Hill climbing
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=phase_a_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # Final evaluation
    try:
        final_val = evaluate_sequence(best_a.tolist())
        print("[final] evaluate_sequence score = %.6f" % final_val)
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()

"""
Search for a non‑negative step function that minimises the objective
``evaluate_sequence``.  The algorithm proceeds in three stages:

1️⃣  **Initialisation** – a cheap sweep over several parametric families
    (Beta, Gaussian, cosine, sine) plus a few random Dirichlet draws.
    The best of these is kept as a starting point.

2️⃣  **Adam optimisation** – minimise a smooth surrogate consisting of a
    smooth approximation of the max‑convolution term plus a variance
    penalty.  The variables are parametrised by logits (soft‑max) so that
    the simplex constraint Σa=1 is satisfied automatically.

3️⃣  **Local refinement** – (a) simulated‑annealing hill‑climbing that
    moves a small amount of mass from “heavy” to “light” indices, and
    (b) a short sub‑gradient descent on the exact max‑convolution term.

All steps stop as soon as the global time budget (default 1000 s) is
exhausted, leaving a half‑second safety margin for the final call to
``evaluate_sequence``.  The returned list is a plain Python list of
non‑negative numbers whose sum is 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution helpers
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) for Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) of the convolution and its gradient.
# Returns smooth value of max(conv) and ∂(2·n·smooth_max)/∂a.
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Variance of the auto‑convolution and its gradient
# ----------------------------------------------------------------------
def _var_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # gradient: (4/m) * cross‑correlation of a with v
    conv_ab = _conv_fft(a, v[::-1])              # length 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad = (4.0 / m) * segment[::-1]
    return var, grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate (max + variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises
        L(a) = 2·n·smooth_max(a) + λ·Var(conv(a,a)),
    where λ is annealed from a large value to zero.
    """
    # Initialise logits (soft‑max parametrisation)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.18, 0.001
    eps_start, eps_end = 0.6, 1e-8

    # variance‑penalty schedule
    lam_start = 9.0 * n
    lam_end = 0.0

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 12_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- convert logits → simplex‑constrained a -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ----- evaluate exact objective (for book‑keeping) -----
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # ----- schedules -----
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        lam = lam_start * (1.0 - progress) + lam_end * progress

        # ----- smooth max and its gradient -----
        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)

        # ----- variance penalty and gradient -----
        var_val, grad_var = _var_and_grad(a)

        # total gradient of the loss (we minimise it)
        grad_a = grad_smooth + lam * grad_var

        # chain rule through soft‑max (gradient wrt logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape shallow wells
        if np.random.rand() < 0.007:
            w += np.random.normal(scale=0.02, size=n)

        # restart if stagnating too long
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.12
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly transfers a small amount of mass from a
    heavily‑contributing index to a weakly‑contributing one, accepting
    moves according to a cooling schedule.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules
    max_frac_start, max_frac_end = 0.55, 0.001
    temp_start, temp_end = 0.018, 1e-8

    no_improve = 0
    IMPROVE_LIMIT = 120_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # ----- locate the current worst convolution entry -----
        k_max = int(np.argmax(b))

        # contribution of each a[i] to this entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (probability proportional to contribution)
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index (biased towards low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # restart if stuck for too long
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution term
# ----------------------------------------------------------------------
def _subgradient_refine(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Very short sub‑gradient walk on the exact objective
        f(a) = 2·n·max(conv(a,a)),
    with projection onto the simplex after each step.
    """
    a = a0.copy()
    cur_val = _objective(a)
    step = 0.12                # initial step size
    decay = 0.9995             # geometric decay
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # exact gradient of max(b) w.r.t. a:
        #   ∂b_kmax/∂a_i = a_{k_max-i} (if in range) + a_{i-(k_max)} (same term)
        # which reduces to 2*a_{k_max-i} for interior points.
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 4.0 * n * a[idx[mask]]          # factor 2·n already included

        # gradient descent step + projection
        a = a - step * grad
        a = _project_simplex(a)

        # evaluate
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[SubG {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        # decay step size
        step *= decay
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric sweep of the two‑parameter Beta family (cheap)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # coarse grid – enough to get a decent start
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 5.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under the given time budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 12000 entries give a good trade‑off between
    # expressivity and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Try to initialise from a previously found solution, if available.
    # ------------------------------------------------------------------
    a0 = None
    if 'height_sequence_1' in globals():
        try:
            old = np.asarray(globals()['height_sequence_1'], dtype=float)
            old_len = old.shape[0]
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            print("[init] interpolated previous solution")
        except Exception:
            a0 = None

    best_a = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweeps + random Dirichlet seeds
    # ------------------------------------------------------------------
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    # Beta family sweep
    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_val < best_val:
            best_a, best_val = beta_a, beta_val
            print(f"[init] beta‑family improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # Gaussian family sweep
    try:
        i = np.arange(n)
        mu = (n - 1) / 2.0
        sigma_vals = np.geomspace(n / 30.0, n / 2.0, num=12)
        for sigma in sigma_vals:
            a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_a, best_val = a.copy(), val
                print(f"[init] gaussian σ={sigma:.1f} improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] Gaussian sweep failed:", exc)

    # Raised cosine shape
    try:
        a = 1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a.copy(), val
            print(f"[init] cosine shape improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] cosine shape failed:", exc)

    # Half‑sine shape
    try:
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a.copy(), val
            print(f"[init] sine shape improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] sine shape failed:", exc)

    # A few random Dirichlet draws
    for _ in range(3):
        a = np.random.dirichlet(np.ones(n))
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[init] random Dirichlet improved value={best_val:.6f}")

    # Fallback – pure random start
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print(f"[init] random start, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈45 % of total budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=phase_a_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # Phase C – Very short sub‑gradient refinement on the exact max
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 0.2  # keep a tiny margin for the final call
    if subgrad_deadline > time.time():
        best_a, best_val = _subgradient_refine(best_a, n,
                                               deadline=subgrad_deadline,
                                               start_time=time.time(),
                                               best_a=best_a,
                                               best_val=best_val,
                                               target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
