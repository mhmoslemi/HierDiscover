def _hill_climbing(a, cur_val, deadline):
    n = len(a)
    rng = np.random.default_rng()
    cur_a = a.copy()
    cur_val = cur_val

    while time.time() < deadline:
        # only transfer moves (no swaps)
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_a[i] == 0:
            continue
        delta = cur_a[i] * 0.01 * rng.random()
        cur_a[i] -= delta
        cur_a[j] += delta

        new_val = _eval_fast(cur_a)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            # keep changes, optionally print
            # print(f"[hill] new best = {cur_val:.6f}")
        else:
            cur_a[i] += delta
            cur_a[j] -= delta

    return cur_a, cur_val

"""
Propose a step‑function that improves the current upper bound (≈1.5113)
by:
  1. Exploring a few low‑dimensional parametric families (V‑shapes,
     edge‑blocks and mixtures with the uniform vector).
  2. Running many cheap projected exponential‑gradient descents on the
     exact max‑convolution (sub‑gradient of the ℓ∞‑norm).  Each run
     starts from a random point on the simplex (Dirichlet(1)) or from
     the best solution found so far.
  3. Finishing with a very small hill‑climbing phase (tiny mass moves).

All heavy operations use an FFT‑based convolution (_fft_convolve_self)
which runs in O(n log n).  The total wall‑time is limited to the given
budget (≈1000 s) with a 1.5 s safety margin for the final official
evaluation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Numerical noise can produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence() for a vector that is already
    ℓ₁‑normalised (sum == 1).  Returns the exact objective value.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    # sum_a == 1 by construction, but we keep the formula for safety.
    return float(2.0 * n * max_b)


def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic re‑ordering that puts large values at the two ends,
    then alternates small/large.  Used as a cheap heuristic after a few
    optimisation steps.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    out = np.empty_like(a)
    left, right = 0, n - 1
    i = 0
    while i < n:
        out[i] = s[right]
        right -= 1
        i += 1
        if i == n:
            break
        out[i] = s[left]
        left += 1
        i += 1
    return out


def _make_uniform(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def _make_v_shape(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """
    V‑shaped distribution (large at the ends, small at the centre):
        a_i ∝ (|i – (n‑1)/2| + eps) ** p
    """
    i = np.arange(n)
    dist = np.abs(i - (n - 1) / 2.0)
    a = (dist + eps) ** p
    a /= a.sum()
    return a


def _make_edge_block(n: int, L: int) -> np.ndarray:
    """
    Two equal blocks of width L at the ends, zero elsewhere.
    The total mass is 1 (0.5 in each block).
    """
    a = np.zeros(n)
    if L > 0:
        val = 1.0 / (2 * L)
        a[:L] = val
        a[-L:] = val
    return a


def _make_combo(n: int, p: float, alpha: float) -> np.ndarray:
    """Convex combination of a V‑shape (parameter p) and the uniform vector."""
    a_v = _make_v_shape(n, p)
    a_u = _make_uniform(n)
    a = alpha * a_v + (1.0 - alpha) * a_u
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}
    (Duchi et al. 2008)
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform
        return _make_uniform(len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exponential‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                         max_iter: int,
                         eta0: float,
                         decay: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Sub‑gradient descent on 𝑚𝑎𝑥(a∗a) using an exponentiated update
    (multiplicative mirror descent on the simplex).  The step size
    η decays geometrically.
    """
    a = a0.copy()
    n = a.shape[0]
    eta = eta0
    best_a = a.copy()
    best_val = _eval_fast(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # 1‑D convolution and index of the current maximum
        conv = _fft_convolve_self(a)
        k = int(conv.argmax())

        # Gradient of the active max entry (2·a_{k‑i} where defined)
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # Exponentiated update  a_i ← a_i · exp(-η·grad_i)
        a = a * np.exp(-eta * grad)

        # Re‑normalise to stay on the simplex
        s = a.sum()
        if s <= 0 or np.isnan(s) or np.isinf(s):
            a = _make_uniform(n)
            s = 1.0
        a = a / s

        # Evaluate
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[EGD] iter {t:4d}  best = {best_val:.6f}")

        # Decay step size
        eta *= decay

    return best_a, best_val

# Tiny hill‑climbing (mass transfers) – final polishing
# ----------------------------------------------------------------------
def _final_hill_climb(a: np.ndarray,
                      cur_val: float,
                      deadline: float) -> (np.ndarray, float):
    """Very cheap local search that only accepts strictly improving moves."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_a[i] <= 0.0:
            continue
        # move up to 2 % of the source mass
        delta = cur_a[i] * 0.02 * rng.random()
        cur_a[i] -= delta
        cur_a[j] += delta
        val = _eval_fast(cur_a)
        if val + 1e-12 < cur_val:
            cur_val = val
        else:
            # revert
            cur_a[i] += delta
            cur_a[j] -= delta

    return cur_a, cur_val

            if time.time() >= deadline - 300.0:
                break
            a = _make_combo(n, p, alpha)
            val = _eval_fast(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[family] combo p={p:.2f} α={alpha:.2f} → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – many cheap exponential‑gradient descents from random starts
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    max_iter_per_run = 300          # enough for a decent descent, cheap
    eta_start = 0.5
    decay = 0.985

    run = 0
    while time.time() < deadline - 30.0:
        run += 1
        # Occasionally start from the current best to polish it
        if run % 8 == 0:
            a_start = best_a.copy()
        else:
            # Dirichlet(1) → uniform on the simplex
            a_start = rng.gamma(1.0, 1.0, size=n)
            a_start /= a_start.sum()

        # run the optimiser
        a_candidate, val_candidate = _exp_gradient_descent(
            a_start,
            max_iter=max_iter_per_run,
            eta0=eta_start,
            decay=decay,
            deadline=deadline - 30.0
        )

        if val_candidate < best_val:
            best_val, best_a = val_candidate, a_candidate.copy()
            print(f"[run {run:3d}] new global best = {best_val:.6f}")

        # Occasionally enforce the “ends‑large” heuristic – it can
        # improve the shape without breaking the simplex constraint.
        if run % 15 == 0:
            a_ordered = _ends_large_reorder(best_a)
            a_ordered /= a_ordered.sum()
            val_ord = _eval_fast(a_ordered)
            if val_ord < best_val:
                best_val, best_a = val_ord, a_ordered.copy()
                print(f"[reorder] after run {run} → {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final tiny hill‑climbing refinement (swap‑free)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final hill‑climb] polishing")
        best_a, best_val = _final_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # Official evaluation (security‑checked)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    # Return a plain Python list – the required format.
    return best_a.tolist()
