"""
Smooth‑max exponential‑gradient descent for the self‑convolution L∞ problem.

The algorithm works on a probability vector `a` (non‑negative, sum = 1).
At each iteration we compute the full self‑convolution `c = a*a`
and replace the hard `max` in the objective by a smooth
approximation  `λ·log Σ exp(c/λ)`.  The gradient of this
approximation is a weighted sum of the gradients of the
individual convolution entries, which can be obtained with a
single FFT‑based convolution.  After a gradient step we
project back to the simplex by an exponential‑mirroring update.
A slowly decaying smoothing parameter `λ` moves the optimisation
from a “soft‑max” regime (many peaks are reduced simultaneously)
to a true `max` regime.  Occasionally we apply an anti‑symmetric
reordering (pairing large and small values) and finally perform
a lightweight hill‑climbing refinement.

The routine respects the 1000 s time budget, calls
`evaluate_sequence` only for logging the current best value,
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (useful for zero‑padding FFTs)."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a real vector with itself (length 2·len‑1) using FFT.
    All values are forced to be non‑negative (tiny negative artefacts are clipped).
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)[k] with respect to a.
    g_i = 2· Σ_k w_k·a_{k-i}  (out‑of‑range indices give 0).
    Computed via a single FFT‑based convolution.
    The weight vector w must have length (2·len(a)‑1).
    """
    n = a.shape[0]
    # reverse the weight vector and convolve with a
    w_rev = w[::-1]
    size = _next_pow2(n + w_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv_aw = np.fft.irfft(fa * fw, size)[: n + w_rev.shape[0] - 1]

    # slice the middle n entries (see _gradient_selected_peaks)
    seg = conv_aw[n - 1 : 2 * n - 1]
    # reverse because of the reversal of w
    g = 2.0 * seg[::-1]
    return g


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest entries with the smallest ones.
    This rearrangement (a, reversed(a)) is known to lower the central
    autocorrelation and often improves the objective.
    """
    n = a.shape[0]
    s = np.sort(a)              # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]        # large → left side
        b[-1 - i] = s[i]        # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Light‑weight hill‑climbing refinement (calls evaluate_sequence)
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Random swaps / tiny mass transfers, keeping only strict improvements."""
    cur_a = a.copy()
    cur_val = best_val
    n = cur_a.shape[0]
    rng = np.random.default_rng()

    # Helper to evaluate (calls the provided evaluate_sequence)
    def _eval(v):
        return evaluate_sequence(v.tolist())

    while time.time() < deadline:
        # Choose operation: swap (20 %) or mass transfer (80 %)
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] < 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Projection onto the simplex (non‑negative, sum=1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        # Evaluate – we are cheap here because hill‑climb runs only after
        # the main optimisation phase.
        try:
            val = _eval(new_a)
        except Exception:
            continue

        if val < cur_val - 1e-12:      # strict improvement required
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
