def _correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1) - 1).bit_length()
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:len(w) + len(a_rev) - 1]
    start = len(a) - 1
    return conv[start:start + len(a)]
```

"""
Step‑function optimizer for `evaluate_sequence`.

Goal
----
Find a non‑negative vector a (Σa = 1) that minimises
        f(a) = 2·n·max(conv(a,a)),
where `conv` is the full linear convolution of a with itself.
The scale‑invariance of the original evaluation function allows us to
work with the normalised sum = 1.

Method
------
The optimizer combines three stages that together explore the space
more aggressively than the previous baseline.

1. **Phase‑0 – quick parametric sweep**  
   A handful of hand‑crafted families (uniform, power‑law, exponentials,
   sines, random Dirichlet draws) are evaluated and the best is kept as
   the starting point.

2. **Phase‑1 – soft‑max Adam optimisation**  
   The objective contains a `max` which is non‑differentiable.
   We replace it by a smoothed soft‑max:
        L_β(a) = (1/β)·log Σ_k exp(β·c_k) ,   c = conv(a,a).
   Its gradient can be written in closed form and is computed via one
   FFT‑based convolution.  The parameters are the logits `z` with
        a = softmax(z) .
   An Adam optimiser with a linearly decreasing step size and a
   temperature β that grows from 1 to 3000 gradually focuses the
   optimisation onto the true maximum.  The best vector encountered
   (according to the true max‑based objective) is stored.

3. **Phase‑2 – deterministic greedy hill‑climb**  
   A fast local search moves a small fraction of the mass from the
   index that contributes most to the current peak to the index that
   contributes least.  The transfer fraction anneals from 50 % down to
   0.1 % and only improving moves are kept.  Periodic random jitter
   helps escape plateaus.

4. **Phase‑3 – tiny uniform mixing**  
   Finally a few convex combinations with the uniform vector are tried;
   this can occasionally shave off the last bit of the peak.

All phases respect the overall time budget (default 1000 s) and print
progress information.  The returned list of floats sums to 1 and is
ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the valid correlation of w (length 2·n‑1) with a (length n):
        r[i] = Σ_j w[i+j]·a[j]    for i = 0 … n‑1
    Implemented by an FFT convolution of w with the reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = _next_pow2(len(w) + len(a_rev) - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:len(w) + len(a_rev) - 1]
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Phase‑0 – cheap parametric families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.shape != (n,):
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(12345)

    idx = np.arange(n)
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # U‑shape linear
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # a few random Dirichlet draws
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑1 – soft‑max Adam optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   start: float,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective
        Lβ(a) = (1/β)·log Σ exp(β·conv(a,a)),
    where β grows from 1 to β_max (≈3000).
    The vector is parametrised by logits `z` with a = softmax(z).
    """
    # Initialise logits from the starting probability vector
    eps = 1e-30
    z = np.log(np.maximum(a0, eps))

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    epsadam = 1e-8

    it = 0
    no_improve = 0
    IMPATIENT = 50000

    # temperature schedule parameters
    beta_start = 1.0
    beta_end = 3000.0

    # learning‑rate schedule (linear decay)
    lr_start = 0.05
    lr_end = 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- compute a from logits (softmax) ----
        maxz = np.max(z)
        expz = np.exp(z - maxz)
        a = expz / expz.sum()

        # ---- evaluate true objective (max‑based) ----
        conv_full = _full_convolution(a)
        cur_val = 2.0 * n * conv_full.max()
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:8d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---- soft‑max weights for the surrogate ----
        progress = (now - start) / (deadline - start)
        beta = beta_start * (1.0 - progress) + beta_end * progress

        # stable softmax over the convolution entries
        max_c = np.max(beta * conv_full)
        w = np.exp(beta * conv_full - max_c)
        w /= w.sum()                     # w shape = (2n‑1,)

        # ---- gradient of surrogate w.r.t a ----
        # ∇_a Σ w_k·c_k = 2 · correlate_valid(w, a)
        grad_a = 2.0 * _correlate_valid(w, a)

        # ---- gradient w.r.t logits (softmax Jacobian) ----
        dot = np.dot(a, grad_a)
        dz = a * (grad_a - dot)

        # ---- Adam update ----
        it += 1
        lr = lr_start * (1.0 - progress) + lr_end * progress
        m = beta1 * m + (1.0 - beta1) * dz
        v = beta2 * v + (1.0 - beta2) * (dz * dz)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # ---- occasional jitter if stuck ----
        if no_improve > IMPATIENT:
            # add small Gaussian noise to logits, re‑normalise via softmax implicitly
            z += np.random.normal(scale=0.2, size=n)
            no_improve = 0
            print(f"[Adam {it:8d}] jitter after stagnation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑2 – deterministic greedy hill‑climbing (same as the baseline)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray,
                      n: int,
                      start: float,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer hill‑climb, identical to the baseline."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:9d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # annealing of transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # peak index
        k = int(np.argmax(b))

        # reverse‑relevant slice of a (a_{k-i})
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        # most and least contributing indices
        i = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                cur += np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:9d}] jitter after stagnation")
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        new_val = 2.0 * n * _full_convolution(cur).max()
        if new_val < cur_val - 1e-15:
            # improvement – keep the move
            pass
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val

# Phase‑3 – tiny uniform mixing
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a small amount of the uniform vector."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
