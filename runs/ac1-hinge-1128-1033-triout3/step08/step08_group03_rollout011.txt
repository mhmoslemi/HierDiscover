def _weighted_gradient(a, w):
    n = a.shape[0]
    # w is length 2n-1
    w_rev = w[::-1]
    size = _next_pow2(n + len(w_rev) - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[: n + len(w_rev) - 1]
    seg = conv[n-1:2*n-1]
    g = seg[::-1]
    return g

def _exp_gradient_optimized(a_start, deadline):

    rel_tol = 0.02  # start

    rel_tol_decay = (rel_tol - min_rel_tol) / 4000  # gradually reduce

    prev_val = best_val
    while time.time() < deadline - 5:
        conv = _full_convolution(a)  # length 2n-1
        max_conv = conv.max()
        # adjust tolerance gradually
        cur_rel_tol = max(min_rel_tol, rel_tol - it * rel_tol_decay)

        thresh = max_conv * (1 - cur_rel_tol)
        idxs = np.where(conv >= thresh)[0]
        # ensure at least few peaks
        if idxs.size < 5:
            idxs = np.argpartition(-conv, 4)[:5]
        # construct weight vector w
        w = np.zeros_like(conv)
        # use actual conv values as weights for selected peaks
        w[idxs] = conv[idxs]  # could be simply 1.0 if want uniform

        g = _weighted_gradient(a, w)

        # step size schedule
        alpha = alpha0 * (decay_factor ** (it // 2000))

        delta = -alpha * g
        # optional scaling normalisation to keep updates moderate
        # Clip delta
        delta = np.clip(delta, -20., 20.)
        a = a * np.exp(delta)

        # enforce positivity and renormalize
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # Logging
        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val

def _hill_climb_refine(a, best_val, deadline, n):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    iter = 0
    while time.time() < deadline:
        # choose operation: swap or transfer
        if rng.random() < 0.5:
            # large transfer
            i = rng.integers(0, n)
            if cur_a[i] < 1e-12: continue
            # decide direction: move proportion
            delta = cur_a[i] * rng.uniform(0.01, 0.1)   # up to 10% of mass
            j = rng.integers(0, n)
            if i == j: continue
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j: continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        # projection
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()

        val = evaluate_sequence(new_a.tolist())

            cur_a = new_a
            cur_val = val
            print(f"[refine {iter}] new best {cur_val:.6f}")
        iter += 1
    return cur_a, cur_val

"""
Non‑negative step‑function optimiser.

The objective of ``evaluate_sequence`` is
    2·n·max(conv(a,a)) / (sum(a)²)          (n = len(a))

Since the denominator is fixed to 1 after normalisation, minimising the
maximum entry of the self‑convolution is the whole problem.
The algorithm proceeds in three phases:

1. **Initialisation** – take the historic construction if available,
   otherwise generate a random Dirichlet vector and normalise it.

2. **Monotone exponential‑gradient descent** – at each iteration
   the full self‑convolution is obtained with a single FFT.
   All peaks whose value is within a (gradually shrinking) relative
   tolerance of the current maximum are selected, weighted by their
   heights and used to build a sub‑gradient (also via FFT).  A line‑search
   chooses a step‑size that strictly reduces the maximal convolution
   entry, guaranteeing monotone progress.  Every 1000 steps an
   anti‑symmetric re‑ordering (pairing largest and smallest values)
   is tried, which provably cannot increase any autocorrelation entry.

3. **Hill‑climbing refinement** – random swaps and tiny mass transfers
   are accepted only when they strictly improve the cheap (FFT‑based)
   evaluation.  This stage finishes the optimisation.

All heavy work uses cheap FFT‑based evaluation; the expensive
``evaluate_sequence`` is called only at the beginning (for logging) and
once at the end to produce the final score.  The routine respects the
1000 s time budget and returns the best list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (useful for zero‑padding FFTs)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a real vector with itself (length 2·len‑1)
    using a single FFT.  Result is clipped to non‑negative values.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_{k} w_k·(a*a)_k w.r.t. ``a``.
    The computation is performed by a single FFT:
        g_i = Σ_k w_k·a_{k‑i}
    (indices outside the vector are treated as zero).
    """
    n = a.shape[0]
    # w has length 2·n‑1 (full convolution length)
    w_rev = w[::-1]
    size = _next_pow2(n + w_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[: n + w_rev.shape[0] - 1]
    # middle segment corresponds to the required sums
    seg = conv[n - 1: 2 * n - 1]
    g = seg[::-1]
    return g


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # largest → left side
        b[-1 - i] = s[i]               # smallest → right side
    if n % 2:
        b[half] = s[half]
    return b


def _cheap_eval(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective without the heavy ``np.convolve``.
    Returns ``2·n·max(conv) / (sum(a)²)`` using an FFT‑based convolution.
    """
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _full_convolution(a)
    max_conv = conv.max()
    n = a.shape[0]
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Phase 1 – monotone exponential‑gradient descent
# ----------------------------------------------------------------------


def _monotone_exp_gradient(a_start: np.ndarray, deadline: float):
    """
    Exponential mirror‑descent with a back‑tracking line search ensuring
    a strict reduction of the maximal convolution entry at every step.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _cheap_eval(best_a)

    it = 0
    cur_alpha = 0.5                     # initial step size for the line search
    # tolerance schedule: start wide, finish tight
    rel_tol_start = 0.02
    rel_tol_end = 5e-4
    # estimate number of iterations we might afford (very rough)
    max_iters = 10000

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[gd iter {it}] new best {best_val:.6f}")

        # --------------------------------------------------------------
        # adapt the relative tolerance (shrink to focus on the very top)
        # --------------------------------------------------------------
        rel_tol = rel_tol_start * (rel_tol_end / rel_tol_start) ** (it / max_iters)

        thresh = max_conv * (1.0 - rel_tol)
        idxs = np.where(conv >= thresh)[0]

        # guarantee a few peaks (helps when the maximum is isolated)
        if idxs.size < 5:
            idxs = np.argpartition(-conv, 4)[:5]

        # build a weight vector – use the actual peak heights as weights
        w = np.zeros_like(conv)
        w[idxs] = conv[idxs]

        # gradient of the weighted sum of selected peaks
        g = _weighted_gradient(a, w)

        # --------------------------------------------------------------
        # line search – shrink alpha until the maximal convolution drops
        # --------------------------------------------------------------
        alpha = cur_alpha
        while alpha > 1e-8:
            delta = -alpha * g
            delta = np.clip(delta, -20.0, 20.0)   # keep exp() in a safe range
            a_new = a * np.exp(delta)
            a_new = np.maximum(a_new, 1e-15)
            a_new /= a_new.sum()

            conv_new = _full_convolution(a_new)
            max_new = conv_new.max()
            if max_new < max_conv - 1e-14:      # strict improvement
                a = a_new
                cur_alpha = alpha * 0.9         # gently shrink step for future
                break
            else:
                alpha *= 0.5                     # try a smaller step

        else:
            # no improvement found – decrease the global step size
            cur_alpha *= 0.5
            if cur_alpha < 1e-8:
                # nothing more can be done
                break

        # --------------------------------------------------------------
        # occasional anti‑symmetric re‑ordering (never hurts the max)
        # --------------------------------------------------------------
        if it % 1000 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _cheap_eval(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                a = a_ord.copy()
                print(f"[gd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – simple hill‑climbing refinement
# ----------------------------------------------------------------------


def _hill_climb_refine(a: np.ndarray, cur_val: float, deadline: float) -> tuple:
    """
    Random local moves (swaps and tiny mass transfers) accepted only
    when they strictly improve the cheap objective.
    """
    n = a.shape[0]
    cur_a = a.copy()
    rng = np.random.default_rng()

    iter_cnt = 0
    while time.time() < deadline:
        # decide move type
        if rng.random() < 0.4:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            # move up to 5 % of the source mass
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _cheap_eval(new_a)
        if new_val < cur_val - 1e-12:      # strict improvement required
            cur_a = new_a
            cur_val = new_val
            print(f"[refine {iter_cnt}] new best {cur_val:.6f}")

        iter_cnt += 1

    return cur_a, cur_val
