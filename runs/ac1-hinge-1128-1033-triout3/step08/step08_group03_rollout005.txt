"""
gradient_smt: Smoothed‑max exponential‑gradient descent for the
self‑convolution objective.

The evaluation function is scale‑invariant:
    2·n·max(a∗a)/(∑a)²  ==  2·n·max(a∗a)  when the coefficients are
    normalised to sum = 1.

We therefore keep the vector on the probability simplex and minimise
the L∞‑norm of its (full) autocorrelation.  A smoothed max
(log‑sum‑exp) is used as a differentiable surrogate; its gradient is a
single FFT‑based convolution.  The step is performed in the
multiplicative (exponential‑gradient) mirror‑descent scheme, which
preserves non‑negativity.  The temperature parameter β is increased
during the run, so the surrogate gradually concentrates on the true
maximum.  Periodic anti‑symmetric re‑ordering and a final hill‑climb
refinement improve the bound further.

The routine respects a 1000 s budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding in FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(v: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (length 2·len‑1)."""
    n = v.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fv = np.fft.rfft(v, size)
    conv = np.fft.irfft(fv * fv, size)[:out_len]
    # Clip tiny negatives caused by floating‑point noise.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                      # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]                # large → left side
        b[-1 - i] = s[i]                # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _gradient_from_weight(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  ( 1/β·log∑_k exp(β·(a∗a)_k) )   using one FFT.

    The gradient is the correlation of ``a`` with the weight vector ``w``
    (w sums to 1).  Implemented via convolution with reversed ``w``.
    """
    n = a.shape[0]          # length of a
    # w has length 2·n‑1
    w_rev = w[::-1]
    size = _next_pow2(n + w_rev.shape[0] - 1)   # ≥ 3·n‑2
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[: n + w_rev.shape[0] - 1]
    # Index where the gradient entries start (2·n‑2 … n‑1)
    start = 2 * n - 2
    grad = conv[start - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Smoothed‑max exponential‑gradient descent
# ----------------------------------------------------------------------
def _smoothed_exponential_descent(a_start: np.ndarray,
                                 deadline: float,
                                 report_interval: int = 100) -> tuple[np.ndarray, float]:
    """
    Perform a large number of cheap iterations that minimise the
    L∞‑norm of ``a∗a`` using a soft‑max surrogate.

    Parameters
    ----------
    a_start : np.ndarray
        Initial probability vector (sum = 1, non‑negative).
    deadline : float
        Absolute time (seconds since epoch) after which the routine must stop.
    report_interval : int
        How many iterations between full calls to ``evaluate_sequence``.

    Returns
    -------
    best_a, best_val
        The best vector seen (as a NumPy array) and its exact evaluation
        via ``evaluate_sequence``.
    """
    n = a_start.shape[0]
    a = a_start.copy()

    # initial evaluation (fast approximated value)
    conv = _full_convolution(a)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    print(f"[gd start] approx value ≈ {best_val:.6f}")

    # hyper‑parameters
    alpha0 = 0.20                # initial step size (multiplicative)
    beta = 5.0                   # start temperature for soft‑max
    beta_max = 800.0
    beta_factor = 1.07           # geometric increase per epoch
    epoch_len = 250              # iterations between temperature updates
    it = 0

    while time.time() < deadline - 5.0:
        # ----- 1. self‑convolution & soft‑max weights -----
        conv = _full_convolution(a)          # length 2·n‑1
        max_c = conv.max()
        # soft‑max (subtract max for numerical stability)
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fall back to a uniform distribution if overflow occurs
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- 2. gradient of the smoothed objective -----
        grad = _gradient_from_weight(a, w)

        # ----- 3. step‑size schedule (slow decay) -----
        alpha = alpha0 / (1.0 + it * 1e-4)

        # ----- 4. multiplicative (mirror‑descent) update -----
        a = a * np.exp(-alpha * grad)

        # ----- 5. re‑project onto the simplex -----
        a = np.maximum(a, 1e-15)   # keep positivity, avoid division by zero
        a /= a.sum()

        # ----- 6. occasional re‑ordering (helps spread mass) -----
        if (it + 1) % 1000 == 0:
            a = _anti_symmetric_reorder(a)

        # ----- 7. logging & best‑track -----
        if (it % report_interval) == 0:
            # fast approximate value
            cur_val = 2.0 * n * conv.max()
            # exact evaluation every few hundred iterations (still cheap)
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[gd iter {it:5d}] new best = {best_val:.6f}")
            elif cur_val < best_val:
                # keep the best even if we haven't called evaluate_sequence yet
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd iter {it:5d}] approx best = {best_val:.6f}")

        # ----- 8. temperature schedule -----
        if (it + 1) % epoch_len == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    # final exact evaluation of the best vector found
    final_val = evaluate_sequence(best_a.tolist())
    print(f"[gd finished] iterations={it}, best = {final_val:.6f}")
    return best_a, final_val


# ----------------------------------------------------------------------
# Light‑weight hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray,
                       current_val: float,
                       deadline: float,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Random local moves (swaps or small mass transfers) accepting only
    strict improvements.  Stops when time runs out.
    """
    cur_a = a.copy()
    cur_val = current_val
    rng = np.random.default_rng()
    move_counter = 0

    while time.time() < deadline - 0.2:
        move_counter += 1
        if rng.random() < 0.2:                        # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                         # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.04 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val + 1e-12 < cur_val:                     # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine {move_counter}] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.

    The algorithm consists of:
        1. Initialise from the historic construction if available,
           otherwise random exponential data.
        2. Smoothed‑max exponential‑gradient descent (≈ 80 % of budget).
        3. Light hill‑climbing refinement (remaining time).
    The best sequence found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand coefficients is a good trade‑off
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise (use the previously discovered construction if present)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = rng.exponential(scale=1.0, size=n)

    a0 = np.clip(a0, 0.0, None)
    s = a0.sum()
    if s < 1e-12:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 /= s

    # ------------------------------------------------------------------
    # Phase 0 – a couple of cheap random restarts (helps avoid a bad start)
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    for _ in range(3):
        if time.time() >= deadline - 200:
            break
        cand = rng.exponential(scale=1.0, size=n)
        cand /= cand.sum()
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            print(f"[rand init] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed‑max exponential‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        gd_a, gd_val = _smoothed_exponential_descent(best_a, deadline)
        if gd_val < best_val:
            best_val = gd_val
            best_a = gd_a.copy()
            print(f"[phase 1] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final check and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
