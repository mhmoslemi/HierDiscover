def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5

    try:
        old = np.asarray(height_sequence_1, dtype=float)   # previous best
        n0 = old.shape[0]
    except Exception:
        old = None
        n0 = None

    # Choose n: maybe set n = min(2000, n0) if old present else 2000.
    if n0 is not None:
        n = min(2000, n0)
        if n < n0:
            # downsample by averaging groups.
            factor = n0 // n
            old_down = old[:factor*n].reshape(n, factor).mean(axis=1)
            f0 = old_down
        else:
            f0 = old.copy()
    else:
        n = 2000
        f0 = np.full(n, 1.0/n)   # uniform in sum=1 then scale.

    # Ensure nonnegative
    f0 = np.maximum(f0, 0.0)
    # Scale to target sum sqrt(2*n)
    target_sum = math.sqrt(2 * n)
    f = f0 * (target_sum / max(f0.sum(), 1e-12))

    best_f = f.copy()
    best_M = np.convolve(best_f, best_f).max()
    print(f"[init] n={n}, initial max={best_M:.6f}")

    # main loop
    it = 0
    while time.time() < deadline - 1:
        # compute M
        conv = np.convolve(f, f)
        M = conv.max()
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            print(f"[iter {it}] new best M={best_M:.6f}")

        # Prepare A matrix
        # Build convolution matrix using toeplitz
        c = np.concatenate([f, np.zeros(n-1)])
        r = np.concatenate([f[0:1], np.zeros(n-1)])
        A = scipy.linalg.toeplitz(c, r)  # shape (2n-1, n)
        # max convex constraints M * 1
        b_ub = np.full(2*n-1, M)
        # Linear program maximize sum(g)
        c_obj = -np.ones(n)  # minimize negative sum
        res = scipy.optimize.linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                                    method='highs')
        if not res.success:
            print(f"[iter {it}] LP failed: {res.message}")
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            print(f"[iter {it}] LP returned zero sum")
            break
        scale = target_sum / sum_g0
        g = g0 * scale

        # line search for t
        t = 0.5
        while t > 1e-4:
            f_new = (1 - t) * f + t * g
            conv_new = np.convolve(f_new, f_new)
            M_new = conv_new.max()
            if M_new <= M * (1 + 1e-12):   # nonincrease
                f = f_new
                break
            t *= 0.5
        # if no acceptable t, break
        if t <= 1e-4:
            # can't improve
            print(f"[iter {it}] cannot find improving step")
            break

        it += 1

    # May run some final hill climbing via random small moves.
    cur_f = best_f.copy()
    cur_M = best_M
    # simple random local search
    while time.time() < deadline:
        # random index move small delta
        i, j = rng.integers(0,n,size=2)
        if i == j or cur_f[i] <= 0:
            continue
        delta = cur_f[i] * 0.1 * rng.random()
        new_f = cur_f.copy()
        new_f[i] -= delta
        new_f[j] += delta
        # maintain sum automatically (since total same)
        # compute max
        new_M = np.convolve(new_f, new_f).max()
        if new_M < cur_M:
            cur_f = new_f
            cur_M = new_M
            print(f"[local] new M {cur_M:.6f}")
        # also compare to best_f
        if cur_M < best_M:
            best_M = cur_M
            best_f = cur_f.copy()
    # final evaluation
    final_seq = best_f.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence score {final_score:.6f}")
    except Exception as e:
        print("[final] eval error", e)

    return final_seq

"""
Propose a step‑function that minimizes ``evaluate_sequence`` using a
linear‑programming / mirror‑descent scheme.

The algorithm works on a fixed length ``n`` (default 2000).  Starting from a
feasible non‑negative vector with the required ℓ₁‑norm
√(2 n) it repeats:

1. Compute M = max(a * a) .
2. Build the dense convolution matrix A such that (a * x) = A·x.
3. Solve the LP  maximise 1ᵀx  subject to   A·x ≤ M ,  x ≥ 0 .
   The optimum is an extreme point that concentrates the mass where it does
   not increase the peak of the convolution.
4. Rescale the LP solution to the target sum √(2 n) and move the current
   vector towards it by a line‑search on the step size t∈(0,1].
   The move is accepted only if the new maximal self‑convolution does not
   increase.
5. Occasionally apply the “anti‑symmetric” re‑ordering heuristic that
   pushes large entries to the left and small entries to the right –
   this often flattens the convolution peak.
6. When the time budget is almost exhausted, perform a cheap random‑mass‑
   transfer hill‑climb to polish the result.

The LP is solved with ``scipy.optimize.linprog`` (HiGHS) which is fast for
the moderate dimensions used here (≈ 2000 variables, ≈ 4000 constraints).
All vectors are kept on the simplex (ℓ₁‑norm = √(2 n)), therefore the
objective of ``evaluate_sequence`` reduces to ``max(a * a)`` and can be
checked cheaply with a fast FFT‑based convolution.

The function respects the 1000 s budget, prints progress and finally
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a using FFT (fast for many calls)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()           # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may create tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place large values left, small values right – a cheap peak‑flattening heuristic."""
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[-1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]       # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense (2n‑1) × n matrix A such that (f * x) = A·x.
    Uses Toeplitz structure; for n≈2000 this costs ≈64 MiB of RAM.
    """
    n = f.shape[0]
    c = np.concatenate([f, np.zeros(n - 1)])
    r = np.concatenate([f[:1], np.zeros(n - 1)])
    # toeplitz returns shape (len(c), len(r)) = (2n‑1, n)
    return toeplitz(c, r)

    Returns a plain Python list of floats.

    start = time.time()
