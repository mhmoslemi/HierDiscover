def _weighted_gradient(a,w):
    n = len(a)
    L = len(w) # 2n-1

    # compute convolution a * w (full)
    size = _next_pow2(n + L - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    conv_fwd = np.fft.irfft(fa * fw, size)[: n + L - 1]

    # compute convolution a * w_rev (full)
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, size)
    conv_rev_full = np.fft.irfft(fa * fw_rev, size)[: n + L - 1]

    # extract first n entries of conv_fwd
    term1 = conv_fwd[:n]

    # extract seg for reverse part, as per earlier logic
    seg = conv_rev_full[n-1:2*n-1][::-1]
    term2 = seg

    grad = term1 + term2
    return grad

def _softmax_gradient_descent(a_start, deadline, beta=5, alpha0=0.5, decay=0.98):
    a = a_start.copy()
    best = a.copy()
    best_val = evaluate_sequence(best.tolist())
    t = time.time()
    iteration = 0
    while time.time() < deadline - 5:
        # compute self-convolution
        conv = _full_convolution(a)
        max_conv = conv.max()
        # compute weight vector (softmax)
        # w = exp(beta * (conv - max_conv))  # max weight 1, others <1
        # optionally normalize
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # compute gradient
        grad = _weighted_gradient(a, w)

        # step size schedule
        alpha = alpha0 * (decay ** (iteration // 2000))

        # multiplicative update
        delta = -alpha * grad
        delta = np.clip(delta, -20, 20)  # to avoid overflow
        a = a * np.exp(delta)

        # project onto simplex
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # logging
        if iteration % 200 == 0:
            cur_val = 2 * len(a) * max(conv) / (np.sum(a) **2) # same as evaluate but using conv we have
            # Actually max(conv) is max_conv.
            cur_val = 2 * len(a) * max_conv / (np.sum(a)**2)
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[gd iter {iteration}] new best approx = {best_val:.6f}")

            # occasional reordering
            if iteration % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                conv_ord = _full_convolution(a_ord)
                max_conv_ord = conv_ord.max()
                val_ord = 2 * len(a) * max_conv_ord / (np.sum(a_ord) **2)
                if val_ord < best_val:
                    best_val = val_ord
                    best = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] improved approx = {best_val:.6f}")

        iteration += 1

    # final evaluation with evaluate_sequence
    final_val = evaluate_sequence(best.tolist())
    print(f"[gd final] best evaluated = {final_val:.6f}")
    return best, final_val

"""
Propose a non‑negative step function minimizing `evaluate_sequence`.

Algorithm Overview
------------------
* **Initialization** – Build a heavy‑tailed Pareto distribution, normalise it to
  sum = 1, and (if available) interpolate the historic construction
  `height_sequence_1` to the target length.

* **Soft‑max weighted mirror descent** – At each iteration the full
  self‑convolution `c = a * a` is computed (FFT, O(n log n)).  A
  soft‑max weight `w_k ∝ exp(β·(c_k − max(c)))` focuses the gradient on the
  currently largest peaks while still pulling the whole spectrum flat.
  The exact gradient of the weighted sum `Σ w_k·c_k` with respect to `a`
  is obtained by two FFT‑based convolutions, giving a cheap O(n log n) update.
  An exponential (mirror) update `a ← a·exp(−α·g)` keeps the vector on the
  probability simplex and guarantees non‑negativity.

* **Dynamic schedule** – `β` grows slowly (sharper focus on the maximum) and
  `α` decays geometrically, allowing aggressive early moves and fine tuning
  later.  Every 1000 iterations the vector is anti‑symmetrically reordered
  (largest ↔ smallest) which provably reduces the central autocorrelation.

* **Simulated‑annealing hill‑climbing** – After the gradient phase a lightweight
  stochastic search (random swaps & small mass transfers) with a temperature
  schedule explores nearby configurations and escapes shallow minima.

* **Final evaluation** – The best candidate found is reported via the provided
  `evaluate_sequence` function.

The whole routine respects a 1000 s budget, performs only a few hundred

# Basic utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    # Numerical noise may introduce tiny negatives – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    return _convolve_fft(a, a)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Gradient of a weighted sum of convolution entries
# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute  g_i = Σ_k w_k·(a_{k‑i} + a_{i‑k})   (out‑of‑range terms = 0).

    The computation uses two FFT‑based convolutions:
        term1 = (a * w)[:n]                     = Σ_k w_k·a_{i‑k}
        term2 = (a * w[::-1])[n‑1:2n‑1][::-1]   = Σ_k w_k·a_{k‑i}
    Returns the vector g of length n.
    """
    n = a.shape[0]
    L = w.shape[0]                      # = 2·n‑1
    size = _next_pow2(n + L - 1)

    # FFT of the current coefficient vector
    fa = np.fft.rfft(a, size)

    # ----- term1 : Σ_k w_k a_{i‑k}  (first n entries of a*w) -----
    fw = np.fft.rfft(w, size)
    conv_fwd = np.fft.irfft(fa * fw, size)[: n + L - 1]   # length 3·n‑2
    term1 = conv_fwd[:n]

    # ----- term2 : Σ_k w_k a_{k‑i}  (reverse‑convolution trick) -----
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, size)
    conv_rev_full = np.fft.irfft(fa * fw_rev, size)[: n + L - 1]
    # slice n‑1 … 2n‑1 (inclusive of n‑1, exclusive of 2n‑1) and reverse
    term2 = conv_rev_full[n - 1: 2 * n - 1][::-1]

    return term1 + term2

def _objective_fast(a: np.ndarray) -> float:

    n = a.shape[0]

    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:

    return 2.0 * n * max_conv / (sum_a ** 2)

            new_a[j] += delta

        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[softmax descent] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – final hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline, n)

    # --------------------------------------------------------------
    # Final evaluation & return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
