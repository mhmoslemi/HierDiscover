"""
Propose a non‑negative step‑function that (approximately) minimises
``evaluate_sequence``.

Main ideas
----------

1. **Smooth surrogate + L‑BFGS** – we minimise a log‑sum‑exp approximation
   of the max‑convolution.  A fine schedule of ε (1e‑1 → 1e‑14) and many
   random restarts gives a good basin of attraction.

2. **Projected sub‑gradient descent** on the exact max‑convolution.
   The sub‑gradient for the peak index is simply the reversed vector.
   An exponential‑mirror‑update (EG) is used; the step size is slowly
   decayed.

3. **Aggressive pairwise mass‑transfer hill‑climbing**.
   The pair (i,j) that contributes most to the current peak is found;
   a fraction of mass is taken either from the larger of the two
   (single‑donor move) or from *both* (double‑donor move) and transferred
   to a low‑mass location.  An annealing schedule accepts occasional
   uphill moves.

4. **Final cheap polishing** – random swaps and tiny transfers, only
   strict improvements are kept.

All stages respect a global 1000 s budget (a 5 s safety margin is left
for the final call to ``evaluate_sequence``).  The best vector found is
returned as a plain Python list.

The helper ``_fast_evaluate`` works on vectors that already sum to 1,
returning the simplified objective `2·n·max(conv)`.  The final call to
the official ``evaluate_sequence`` normalises the vector again and checks
the safety constraints.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None          # smoothing parameter ε
_GLOBAL_N = None            # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient wrt the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max -----
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- convolution -----
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max -----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient wrt a -----
    # p is the softmax over (conv - max_c)/ε  → probabilities for each lag
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()

    # derivative of each conv entry w.r.t. a is a convolution with the
    # reversed vector of p (see derivation in the previous solution)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment that corresponds to ∂conv/∂a (the middle part)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]     # reverse to align with a_i order

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical fallback – uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors that
    already sum to 1.  Returns   2·n·max(conv) .
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)      # denominator Σ a_i = 1, squared = 1


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places the largest values on the left side,
    the smallest on the right side.  Often reduces the peak of the
    self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector is kept on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.03 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s budget, prints
    progress messages and finally returns the best vector (as a plain
    Python list).
    """
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine problem size n
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be present from a previous run
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    print(f"[setup] problem size n = {n}")

    # --------------------------------------------------------------
    # Initialise a small portfolio of diverse candidates
    # --------------------------------------------------------------
    init_candidates = []

    # uniform distribution
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # Dirichlet with α = 1 (pure random)
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))

    # Dirichlet with α = 0.2 (more spiky)
    init_candidates.append(rng.gamma(shape=0.2, scale=1.0, size=n))

    # sine‑shaped (smooth variation)
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # anti‑symmetric reorder of the uniform vector (often a good start)
    init_candidates.append(_anti_symmetric_reorder(init_candidates[0].copy()))

    # normalise everything onto the simplex
    init_candidates = [c / c.sum() for c in init_candidates]

    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – aggressive L‑BFGS on the smoothed surrogate
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-1), np.log10(1e-14), num=24)
    w = np.log(best_a + 1e-12)           # initialise soft‑max parameters
    max_restarts = 12

    for restart in range(max_restarts):
        if time.time() >= deadline - 200.0:
            break
        print(f"[L‑BFGS restart {restart + 1}/{max_restarts}]")
        if restart % 4 == 0:
            # occasional full random restart of the soft‑max parameters
            w = rng.normal(size=n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 200.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 2500,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x
            # back‑transform to probabilities
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[L‑BFGS eps={eps:.1e}] new best = {best_val:.6f}")

        # Occasionally apply the anti‑symmetric reorder – often a cheap
        # improvement when the iterates become too symmetric.
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val - 1e-9:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – projected sub‑gradient (exponential mirror) descent
    # --------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.6          # initial step size
    decay = 0.9997        # geometric decay
    min_alpha = 0.008
    it = 0
    while time.time() < deadline - 150.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a: g_i = a_{max_k - i}  (if the index is valid)
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]

        # exponential mirror‑descent update (guarantees non‑negativity)
        alpha = max(alpha0 * (decay ** (it // 300)), min_alpha)
        a = a * np.exp(-alpha * g)

        # project onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 300 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = a.copy()
                print(f"[MD it {it}] new best = {best_val:.6f}")

            # occasional anti‑symmetric reordering
            if it % 1200 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val - 1e-9:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[MD reorder] improved to {best_val:.6f}")

        it += 1

    # --------------------------------------------------------------
    # Stage 3 – aggressive pairwise mass‑transfer hill‑climbing with SA
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.025                # initial temperature
    cooling = 0.99993
    move_fraction = 0.15     # start with relatively large moves

    while time.time() < deadline - 30.0:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        # indices participating in the maximal convolution entry
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        contrib = cur_a[idx] * cur_a[jdx]

        # pair (i,j) with the largest contribution to the peak
        pair_idx = np.argmax(contrib)
        i = idx[pair_idx]
        j = jdx[pair_idx]

        # choose the donor (the larger of the two) and a recipient in the
        # low‑mass tail (5 % smallest entries)
        donor = i if cur_a[i] >= cur_a[j] else j
        low_mass_idx = np.where(cur_a < np.percentile(cur_a, 5))[0]
        if low_mass_idx.size == 0:
            recipient = rng.integers(0, n)
        else:
            recipient = rng.choice(low_mass_idx)

        # Two possible move types:
        if rng.random() < 0.6:
            # **single‑donor**: move a fraction from donor to recipient
            delta = cur_a[donor] * move_fraction * rng.random()
            new_a = cur_a.copy()
            new_a[donor] -= delta
            new_a[recipient] += delta
        else:
            # **double‑donor**: take a fraction from *both* members of the pair
            # and deposit it on a low–mass location.
            delta = min(cur_a[i], cur_a[j]) * (move_fraction * 0.5) * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] -= delta
            new_a[recipient] += 2 * delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)

        # Simulated‑annealing acceptance
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[pairhill] improved to {best_val:.6f}")

        # temperature & move‑size schedules
        T = max(T * cooling, 1e-7)
        # slowly shrink the move fraction for finer adjustments
        move_fraction = max(move_fraction * 0.9999, 0.02)

        # Occasionally shuffle the whole vector with the anti‑symmetric trick
        if rng.random() < 0.003:
            cur_a = _anti_symmetric_reorder(cur_a)
            cur_val = _fast_evaluate(cur_a)
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[pairhill reorder] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – final cheap polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # Final safety check with the official evaluator
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        # If the official function raises, we still return our best guess.
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
