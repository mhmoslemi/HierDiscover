Define function softmax weights for convolution:

```
def softmax_weights(b, lam):
    maxb = b.max()
    # Because b-maxb <= 0, exp(lam*(b-maxb)) stable
    w = np.exp(lam * (b - maxb))
    w_sum = w.sum()
    if w_sum == 0:
        # fallback to uniform weight at max index
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
        return w
    return w / w_sum
```

Define gradient via correlation using FFT:

```
def gradient_from_weights(w, a):
    # Compute g_i = 2 * sum_k w_k * a_{k-i} (zero if out of bounds)
    # This is the 'valid' correlation of w with a.
    # Using FFT for speed.
    n = a.shape[0]
    len_w = w.shape[0]  # = 2n-1
    size = 1 << (len_w + n - 1).bit_length()
    # FFT of w
    W = np.fft.rfft(w, size)
    # FFT of reversed a
    A_rev = np.fft.rfft(a[::-1], size)
    conv = np.fft.irfft(W * A_rev, size)[:len_w + n - 1]
    # extract valid part: indices n-1 ... n-1 + n -1 = 2n-2 inclusive (size n)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g
```

def smoothed_expgrad(a0, n, start_time, deadline, best_a, best_val,
                     lam_start=200.0, lam_end=5.0, eta_start=0.8, eta_end=0.02,
                     target_val=1.503):
    cur = a0.copy()
    no_improve = 0
    IMPATIENT = 200_000
    rng = np.random.default_rng(123456)
    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute conv and objective
        b = full_convolve(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SmoothedExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule parameters
        progress = (now - start_time) / (deadline - start_time)
        lam = lam_start * (1.0 - progress) + lam_end * progress
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # compute softmax weights
        w = np.exp(lam * (b - maxb))
        w_sum = w.sum()
        # avoid underflow if all zeros - unlikely
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

        # compute gradient
        g = gradient_from_weights(w, cur)

        # exponentiated gradient update
        # stabilise exponent:
        g_shift = g - g.max()
        cur = cur * np.exp(-eta * g_shift)
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # jitter if stagnated
        if no_improve > IMPATIENT:
            # add small Gaussian noise, re-normalise
            noise = rng.normal(scale=0.005, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[SmoothedExpGrad {it:7d}] jitter after stagnation")

        it += 1
    return best_a, best_val

def _initial_candidate_extended(n):
    # similar to earlier but add Beta and arcsine-like families.
    best_a = np.full(n, 1.0 / n)
    best_val = objective(best_a)

    def try_update(a):
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n: return
        if np.any(a < 0): return
        s = a.sum()
        if s <= 0: return
        a = a / s
        val = objective(a)
        nonlocal best_a, best_val
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(12345)

    idx = np.arange(n)
    # deterministic families
    try_update(np.arange(n, 0, -1, dtype=float))
    try_update(np.maximum(idx + 1, n - idx))
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # power-law U-shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # Beta distributions with parameters <1 (U-shape)
    for a_param, b_param in [(0.3,0.3), (0.2,0.2), (0.1,0.1), (0.4,0.6), (0.6,0.4)]:
        x = (np.arange(n, dtype=float) + 0.5) / n
        a = (x ** (a_param - 1)) * ((1 - x) ** (b_param - 1))
        try_update(a)

    # arcsine-like shape: a_i = 1 / sqrt((i+0.5)*(n-0.5 - i))
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    try_update(a)

    # random Dirichlet draws
    for _ in range(20):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val

def _greedy_hillclimb(a_start, n, start, deadline, best_a, best_val, target_val=1.503):
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001
    rng = np.random.default_rng(987654)
    while True:
        now = time.time()
        if now >= deadline:
            break
        b = full_convolve(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate maximal convolution entry
        k = int(np.argmax(b))

        # compute rev contributions
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]
        # source index i: biggest contribution
        i = int(np.argmax(rev))
        # target index j: smallest contribution (different from i)
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = rng.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        # amount to move
        delta = cur[i] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        # recompute objective after move
        b_new = full_convolve(cur)
        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-15:
            # accept improvement
            # keep cur and continue (cur_val updated)
            pass
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val
```

"""
Propose a near‑optimal non‑negative step function for the
`evaluate_sequence` benchmark.

Algorithm
---------

1. **Problem size** – we use `n = 10000`.  The objective is
   f(a)=2·n·max(conv(a,a)) under the simplex constraint Σa = 1,
   a_i ≥ 0.

2. **Rich initial sweep** – a handful of deterministic families
   (uniform, powers, exponentials, Beta‑shapes, arcsine‑like) and a
   few random Dirichlet draws are evaluated; the best vector is kept.

3. **Smoothed exponentiated‑gradient descent** – instead of using the
   sub‑gradient of a single maximal convolution entry, we form a
   soft‑max weight `w` over all convolution entries
   (`w_k ∝ exp(λ·(b_k‑max(b))`).  The gradient of the smoothed max is

        g_i = 2·∑_k w_k·a_{k‑i}      (valid correlation of w with a)

   and we update multiplicatively
        a ← a·exp(‑η·g)  ,   a ← a/Σa .
   The temperature λ and the step‑size η are annealed from
   (λ,η)≈(200,0.2) down to (5,0.001).  A jitter is added after long
   stagnation.

4. **Deterministic greedy hill‑climb** – repeatedly transfer a small
   fraction of the mass from the index that contributes most to the
   current peak to the index that contributes the least.  The transfer
   fraction decays from 50 % to 0.1 % while the algorithm runs.

5. **Random local search** – short phase of Gaussian perturbations of the
   current best vector, accepting any improvement.

6. **Final uniform mixing** – test a few convex combinations with the
   uniform vector; sometimes this reduces the peak further.

All phases respect the overall 1000 s budget, keep the sequence
non‑negative and normalised, and store the best vector found.
"""

import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)

# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)               # uniform
    best_val = _objective(best_a, n)

    def try_update(x):
        nonlocal best_a, best_val
        a = np.asarray(x, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(12345)

    idx = np.arange(n)

    # deterministic families
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # U‑shape linear
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # Beta‑type U‑shapes (parameters < 1 concentrate mass at the ends)
    for a_par, b_par in [(0.3, 0.3), (0.2, 0.2), (0.1, 0.1),
                         (0.4, 0.6), (0.6, 0.4)]:
        x = (np.arange(n, dtype=float) + 0.5) / n
        a = (x ** (a_par - 1)) * ((1 - x) ** (b_par - 1))
        try_update(a)

    # arcsine‑like shape : a_i ∝ 1/√((i+½)(n‑½‑i))
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    try_update(a)

    # a few random Dirichlet draws (quick)
    for _ in range(20):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent (Phase A)
# ----------------------------------------------------------------------
def _smoothed_expgrad(a0: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient using a soft‑max over all convolution entries."""
    cur = a0.copy()
    rng = np.random.default_rng(987654)
    no_improve = 0
    IMPATIENT = 200_000
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SmoothedExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # annealing schedules
        progress = (now - start) / (deadline - start)
        lam = 200.0 * (1.0 - progress) + 5.0 * progress          # temperature
        eta = 0.20 * (1.0 - progress) + 0.001 * progress          # step size

        # soft‑max weights (stable)
        w = np.exp(lam * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

        # gradient = 2 * (valid correlation of w with a)
        # compute via FFT for speed
        len_w = w.shape[0]               # = 2n‑1
        size = 1 << (len_w + n - 1).bit_length()
        W = np.fft.rfft(w, size)
        A_rev = np.fft.rfft(cur[::-1], size)
        conv = np.fft.irfft(W * A_rev, size)[:len_w + n - 1]
        start_idx = n - 1
        g = 2.0 * conv[start_idx:start_idx + n]   # length n

        # exponentiated‑gradient multiplicative update
        g_shift = g - g.max()               # shift to keep exponents moderate
        cur = cur * np.exp(-eta * g_shift)
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # occasional jitter when stuck
        if no_improve > IMPATIENT:
            noise = rng.normal(scale=0.005, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[SmoothedExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climbing (Phase B)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Move a fraction of mass from the worst to the best contributing indices."""
    cur = a_start.copy()
    rng = np.random.default_rng(123987)
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # anneal transferred fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # index where convolution attains its maximum
        k = int(np.argmax(b))

        # contributions a_{k‑i} for this peak (zero outside support)
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        rev[idx] = cur[k - idx]

        # source = index with largest contribution, target = smallest (different)
        i = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = rng.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        # amount to transfer (random fraction of the allowed maximum)
        delta = cur[i] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        # evaluate new objective
        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-15:
            # improvement accepted – keep the move
            pass
        else:
            # revert move
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Random local perturbations (Phase C‑a)
# ----------------------------------------------------------------------
def _random_local_search(a_start: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Add Gaussian noise and keep any improvement (very short phase)."""
    cur = a_start.copy()
    rng = np.random.default_rng(424242)
    it = 0

    while time.time() < deadline:
        noise = rng.normal(scale=0.001, size=n)
        cand = np.maximum(cur + noise, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        val = _objective(cand, n)
        if val < best_val - 1e-15:
            best_val, best_a, cur = val, cand.copy(), cand.copy()
            print(f"[RandomSearch {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Final tiny uniform mixing (Phase C‑b)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combinations a′ = (1‑α)a + α·uniform for a few α."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under the given time budget.
    Returns the best list of floats found (Σa = 1) which minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – a few ten‑thousands strike a good balance.
    # ------------------------------------------------------------------
    n = 10000

    # ------------------------------------------------------------------
    # Initialise – try to reuse a previously discovered vector.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)

    # ------------------------------------------------------------------
    # Split the budget among the phases
    # ------------------------------------------------------------------
    total = deadline - start_time
    t1 = start_time + 0.40 * total   # Phase A
    t2 = t1 + 0.35 * total           # Phase B
    t3 = t2 + 0.15 * total           # Phase C‑a (random search)
    # remaining time (≈10 %) for final mixing

    # ------------------------------------------------------------------
    # Phase A – smoothed exponentiated‑gradient descent
    # ------------------------------------------------------------------
    best_a, best_val = _smoothed_expgrad(
        a0, n,
        start=start_time,
        deadline=t1,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )

    # ------------------------------------------------------------------
    # Phase B – deterministic greedy hill‑climb
    # ------------------------------------------------------------------
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        start=t1,
        deadline=t2,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )

    # ------------------------------------------------------------------
    # Phase C‑a – short random local search
    # ------------------------------------------------------------------
    best_a, best_val = _random_local_search(
        best_a, n,
        start=t2,
        deadline=t3,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )

    # ------------------------------------------------------------------
    # Phase C‑b – final uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification through the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
