Implementation of projection onto simplex:

def simplex_projection(v):
    # ensures nonnegative and sum=1
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Non‑negative step‑function optimizer for `evaluate_sequence`.

The algorithm consists of three stages that fit comfortably inside the
1000 s time budget:

1. **Quick families** – generate a few parametric candidates
   (uniform, power‑laws, exponentials, Dirichlet draws) and keep the
   best one.  This gives a solid starting point.

2. **Smoothed gradient descent** – minimise the exact objective
   `2·n·max(conv(a,a))` while keeping Σa = 1.
   Instead of the nonsmooth max we use a softmax
   (log‑sum‑exp) surrogate with a gradually increasing temperature
   parameter `γ`.  The gradient of the surrogate can be written as a
   convolution, which we evaluate with FFTs (O(n log n)).  After each
   additive step we project back onto the probability simplex to
   enforce non‑negativity and unit sum.  The step‑size and `γ`
   decay linearly, allowing large exploratory steps early and fine
   tuning later.  When no improvement is seen for many iterations a
   small random jitter is added to escape stagnation.

3. **Deterministic greedy hill‑climb** – a fast local search that
   repeatedly moves a fraction of mass from the index that contributes
   most to the current peak to the index that contributes the least.
   Only moves that strictly improve the exact objective are kept.
   This phase polishes the solution obtained by the gradient descent.

Finally the best vector found is returned (as a plain list of floats).
All operations respect the time limit and the required non‑negativity.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]          # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of given size (output length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv


def _conv_rev_fft(w: np.ndarray, a_rev: np.ndarray, size: int) -> np.ndarray:
    """Convolution of w (length 2n‑1) with a_rev (length n) using FFT."""
    fw = np.fft.rfft(w, size)
    fr = np.fft.rfft(a_rev, size)
    conv_full = np.fft.irfft(fw * fr, size)[: len(w) + len(a_rev) - 1]
    # The gradient slice we need is the central n entries:
    n = a_rev.shape[0]
    start = n - 1
    return conv_full[start : start + n]   # length n


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a vector with Σa = 1."""
    return float(2.0 * n * conv.max())


def _objective(a: np.ndarray, n: int) -> float:
    """Convenient wrapper that computes the objective from a."""
    conv = _full_convolution_fft(a, _next_pow2(2 * n - 1))
    return _objective_from_conv(conv, n)


# ----------------------------------------------------------------------
# Phase‑0 – quick families (same idea as the reference implementation)
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)                 # uniform
    best_val = _objective(best_a, n)

    def _try_update(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(12345)

    idx = np.arange(n, dtype=float)

    # deterministic families
    _try_update(np.arange(n, 0, -1, dtype=float))                           # decreasing
    _try_update(np.maximum(idx + 1, n - idx))                              # U‑shape linear
    _try_update((np.arange(n, 0, -1, dtype=float) ** 2))                   # decreasing quadratic
    _try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))            # half‑wave sine
    _try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))        # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p) + (n - i + 1) ** (-p)
        _try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try_update(a)

    # random Dirichlet draws (quick)
    for _ in range(12):
        _try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑1 – smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a_start: np.ndarray,
                             n: int,
                             deadline: float,
                             best_a: np.ndarray,
                             best_val: float,
                             target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth approximation
        φγ(a) = (1/γ)·log Σ_k exp(γ·conv_k(a))
    of the max‑convolution.
    The gradient can be written as a convolution and is computed via FFT.
    """
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 25000

    # FFT sizes (chosen once for the whole phase)
    size_full = _next_pow2(2 * n - 1)          # for a*a
    size_big = _next_pow2(3 * n - 2)           # for softmax * a_rev

    # Hyper‑parameters (linear schedules)
    gamma_start, gamma_end = 5.0, 1000.0
    step_start, step_end = 0.50, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear progress in [0,1]
        progress = (now - (deadline - (deadline - time.time()))) / (deadline - (deadline - (deadline - time.time())))  # dummy to avoid division by zero
        # simpler: use elapsed since start of this phase
        # but we don't have phase start timestamp here – compute it from iteration count
        # Instead we use a decaying schedule based on iteration count:
        progress = min(it / 200_000.0, 1.0)   # after ~200k iters we are at full progress

        gamma = gamma_start + (gamma_end - gamma_start) * progress
        step = step_start + (step_end - step_start) * progress

        # ---- compute convolution of cur with itself ----
        conv = _full_convolution_fft(cur, size_full)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        # ---- softmax weights for the smooth max ----
        # Shift by max to avoid overflow
        shift = max_conv
        exp_vals = np.exp(gamma * (conv - shift))
        w = exp_vals / exp_vals.sum()          # length 2n‑1, sums to 1

        # ---- gradient of the surrogate ----
        grad = 2.0 * _conv_rev_fft(w, cur[::-1], size_big)   # length n

        # ---- take a step and project back onto the simplex ----
        cand = cur - step * grad
        cand = _simplex_projection(cand)

        # ---- evaluate candidate ----
        conv_cand = _full_convolution_fft(cand, size_full)
        cand_val = 2.0 * n * conv_cand.max()

        # ---- keep the best overall ----
        if cand_val < best_val - 1e-12:
            best_val, best_a = cand_val, cand.copy()
            no_improve = 0
            print(f"[GD {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---- accept candidate if it improves the current state ----
        if cand_val < cur_val:
            cur = cand

        # ---- occasional jitter to escape stagnation ----
        if no_improve >= IMPATIENT:
            noise = np.random.normal(scale=0.005, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur = _simplex_projection(cur)
            no_improve = 0
            print(f"[GD {it:7d}] jitter after stagnation")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑2 – deterministic greedy hill‑climb (same idea as the reference)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climbing on the exact objective.
    Moves a (decaying) fraction of the weight from the index that
    contributes most to the current maximal convolution entry to the
    index that contributes the least.
    """
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.45, 0.001   # a little more aggressive than the reference

    # FFT size for the exact convolution (re‑used many times)
    size_full = _next_pow2(2 * n - 1)

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current convolution and objective
        conv = _full_convolution_fft(cur, size_full)
        cur_val = 2.0 * n * conv.max()

        # record improvement
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # anneal transfer fraction
        progress = min(it / 250_000.0, 1.0)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # location of the maximal convolution entry
        k = int(np.argmax(conv))

        # slice of a that contributes to this entry (a_{k-i})
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        rev[idx] = cur[k - idx]                     # a_{k‑i}

        # source index = biggest contribution
        i = int(np.argmax(rev))

        # target index = smallest contribution (excluding source)
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            # source exhausted; add jitter if we are stuck
            if no_improve >= IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur = _simplex_projection(cur)
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            it += 1
            continue

        # amount to move (fraction of source weight)
        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            it += 1
            continue

        # trial move
        cur[i] -= delta
        cur[j] += delta

        # evaluate new objective
        conv_new = _full_convolution_fft(cur, size_full)
        new_val = 2.0 * n * conv_new.max()

        if new_val < cur_val - 1e-15:
            # keep move
            cur_val = new_val
        else:
            # revert move
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑3 – tiny uniform mixing (may shave off the last fraction)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test convex combinations with the uniform vector."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _simplex_projection(cand)      # ensure exact sum = 1
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under the given time budget.
    Returns the best list of floats found (Σa = 1) which minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety buffer (0.2 s) for the final verification call
    deadline = start_time + budget_s - 0.2

    # --------------------------------------------------------------
    # Problem size – 8000 entries provides a good balance.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Phase‑0 – initialise from a previous solution if available,
    #           otherwise fall back to a quick parametric sweep.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 = _simplex_projection(a0)
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase‑1 – smoothed gradient descent (majority of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.55 * (deadline - start_time)  # ~55 % of the total time
    best_a, best_val = _smooth_gradient_descent(
        best_a, n,
        deadline=phase1_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )
    print(f"[Phase‑1] best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase‑2 – deterministic greedy hill‑climbing (fine‑tuning)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.90 * (deadline - start_time)  # another ~35 % of time
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        deadline=phase2_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )
    print(f"[Phase‑2] best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase‑3 – final tiny uniform mixing
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Phase‑3] best after mixing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final verification through the provided secure wrapper
    # --------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
