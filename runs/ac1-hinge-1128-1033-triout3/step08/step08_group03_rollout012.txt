"""
Propose a non‑negative step function that minimizes `evaluate_sequence`.

Algorithm
---------
1. **Problem scaling** – we keep the L¹‑norm fixed to
   √(2·n).  With that normalisation the evaluation function becomes
   simply the maximal value of the full self‑convolution, i.e.
   `evaluate_sequence(a) = max(conv(a,a))`.  Minimising the bound is
   therefore equivalent to flattening the convolution.

2. **Initial construction** – draw an exponential vector,
   normalise it to the target sum and apply an *anti‑symmetric*
   re‑ordering (largest values opposite the smallest ones).  This
   guarantees that the central product `a_i·a_{n‑1‑i}` is as small as
   possible.

3. **Focused exponential‑gradient descent** – at every iteration
   we compute the full convolution, locate the *exact* maximal peak
   (only the entries attaining the current maximum) and compute the
   sub‑gradient of the sum of those peaks w.r.t. the coefficients.
   The coefficients are updated multiplicatively (`a ← a·exp(-α·g)`)
   and projected back onto the simplex of the prescribed L¹‑norm.
   The step size α starts relatively large (0.8) and decays
   geometrically; a cheap back‑tracking cut‑back is performed if a
   tentative update would increase the current maximum.

4. **Periodic anti‑symmetric re‑ordering** – after any successful
   improvement we re‑order the vector again (large ↔ small) and keep
   the better of the two.

5. **Hill‑climbing refinement** – a lightweight stochastic local
   search (random swaps inside each half and tiny mass transfers)
   is run until the time budget expires.  Only strictly improving
   moves are kept.

The whole search runs within the 1000 s budget, constantly logging
progress and finally returning the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (length 2·len‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values opposite the smallest values.
    After the operation a[i]·a[n‑1‑i] is as small as possible.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    # left half gets the largest values, right half the smallest
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]              # middle element stays put
    return b


def _gradient_selected_peaks(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Gradient of the sum of the convolution peaks whose indices are in `idxs`.

    The derivative of   Σ_{k∈idxs} (a * a)[k]  w.r.t. a_i  equals
    Σ_{k∈idxs} a_{k‑i}   (indices outside the range contribute 0).
    The computation is performed via a single FFT.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1

    # indicator vector for the selected peaks
    p = np.zeros(conv_len, dtype=float)
    p[idxs] = 1.0

    # convolution a * reverse(p)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the middle n entries correspond to the required sums
    seg = conv_ap[n - 1: 2 * n - 1]
    g = seg[::-1]       # reverse because p was reversed
    return g


def _project_to_sum(a: np.ndarray, target: float) -> np.ndarray:
    """
    Project a non‑negative vector onto the simplex {x≥0, Σx = target}.
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        # fall back to uniform distribution
        return np.full_like(a, target / a.shape[0])
    return a * (target / s)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that
    minimises `evaluate_sequence` under the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # a small safety margin

    # --------------------------------------------------------------
    # Problem size – a few‑ten‑thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 15000                                   # feel free to adjust
    target_sum = math.sqrt(2.0 * n)             # sum = √(2n)

    # --------------------------------------------------------------
    # Initialise – reuse a known construction if available, otherwise draw a fresh one.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)    # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the new grid
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = rng.exponential(scale=1.0, size=n)

    a = np.maximum(a, 0.0)
    a = _project_to_sum(a, target_sum)
    a = _anti_symmetric_reorder(a)

    # --------------------------------------------------------------
    # Initial evaluation (the objective reduces to the maximal convolution value)
    # --------------------------------------------------------------
    conv = _full_convolution(a)
    best_val = conv.max()
    best_a = a.copy()
    print(f"[init] n={n}, current max = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – focused exponential‑gradient descent
    # --------------------------------------------------------------
    it = 0
    alpha0 = 0.8            # initial step size
    decay = 0.996           # exponential decay factor
    min_alpha = 1e-4

    while time.time() < deadline - 30.0:     # reserve time for the final refinement
        # full self‑convolution
        conv = _full_convolution(a)
        max_val = conv.max()

        # keep the best ever seen
        if max_val < best_val:
            best_val = max_val
            best_a = a.copy()
            print(f"[gd  iter {it}] new best max = {best_val:.6f}")

        # indices attaining the current maximum (tight constraints)
        idxs = np.where(conv >= max_val - 1e-12)[0]

        g = _gradient_selected_peaks(a, idxs)

            except Exception:
                pass

    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    half = n // 2

        if rng.random() < 0.25:          # swap two positions *inside* the same half

                i = rng.integers(0, half)
                j = rng.integers(0, half)

                j = rng.integers(half, n)

            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                            # small mass transfer between two arbitrary locations
            i = rng.integers(0, n)
            j = rng.integers(0, n)

                continue

            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta

        # re‑project onto the simplex (keeps the L¹‑norm fixed)
        cand = _project_to_sum(cand, target_sum)

        # evaluate the candidate (max of the convolution)
        conv_cand = _full_convolution(cand)

    try:

        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluation raised an exception:", exc)

    return final_a.tolist()
