def _adam_basis_phase(n, d, deadline, best_a=None, best_val=float('inf'), target_val=1.503):
    start = time.time()
    # compute basis
    i = np.arange(n, dtype=np.float64)
    Phi = np.empty((n, d), dtype=np.float64)
    Phi[:,0] = 1.0
    for k in range(1,d):
        Phi[:,k] = np.cos(np.pi * k * i / (n-1))
    # Initialize w, m, v.
    w = np.random.randn(d) * 0.1
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 5000
    # Schedules
    lr_start, lr_end = 0.15, 0.003
    eps_start, eps_end = 0.1, 1e-9

    while time.time() < deadline:
        # compute a
        h = Phi.dot(w)
        h_shift = h - np.max(h)
        a = np.exp(h_shift)
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1.0/n)
            sum_a = 1.0
        a = a / sum_a

        # compute convolution and objective
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print("[AdamBasis] new best {:.6f}".format(best_val))
            if best_val <= target_val:
                # early stop
                break
        else:
            no_improve += 1

        # schedule
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # compute smooth gradient
        shifted = (b - max_b) / eps
        weights = np.exp(shifted - np.max(shifted))   # avoid overflow, subtract max for safety
        p = weights / weights.sum()

        # gradient w.r.t a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        seg = conv_ab[n-1:2*n-1]
        grad_a = 4.0 * n * seg[::-1]

        # gradient w.r.t h
        dot = np.dot(grad_a, a)
        grad_h = a * (grad_a - dot)

        # gradient w.r.t w
        grad_w = Phi.T.dot(grad_h)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.02, size=d)

        # restart if stagnant
        if no_improve > IMPROVE_LIMIT:
            print("[AdamBasis] restart")
            w = np.random.randn(d) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

    return best_a, best_val

"""Optimiser for the step‑function objective 2·n·max(conv(a,a))/sum(a)^2.

The main idea
--------------
* Work on the probability simplex (Σ a_i = 1) – the scale of a does not
  affect the objective.
* Parametrise the vector a by a low‑dimensional Fourier basis and optimise
  the basis coefficients with Adam on a smooth log‑sum‑exp approximation of
  the max‑convolution.
* Refine the obtained candidate with the exact, element‑wise hill‑climbing
  and projected sub‑gradient methods from the reference implementation.
* Occasionally mix a tiny amount of uniform mass – this often reduces the
  peak further.
* Run the whole pipeline on a sequence of increasing sizes (2000 → 4000 → 8000)
  using the solution from the previous size as warm‑start; this gives a
  good trade‑off between runtime and solution quality.

All helper functions are top‑level (no closures, no lambdas) and the only
external call is ``evaluate_sequence`` which is supplied by the harness.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``a`` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a normalised to sum 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w ≥ 0, Σ w = 1}."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Basis construction (bias + cosine/sine Fourier modes)
# ----------------------------------------------------------------------
def _build_basis(n: int, K: int) -> np.ndarray:
    """Return an (n × (1+2K)) basis matrix: [1, cos, sin] modes."""
    i = np.arange(n, dtype=np.float64)
    d = 1 + 2 * K
    Phi = np.empty((n, d), dtype=np.float64)
    Phi[:, 0] = 1.0                                 # bias
    for k in range(1, K + 1):
        Phi[:, 2 * k - 1] = np.cos(2.0 * np.pi * k * i / n)
        Phi[:, 2 * k] = np.sin(2.0 * np.pi * k * i / n)
    return Phi


# ----------------------------------------------------------------------
# Adam optimisation on a low‑dimensional basis (smooth max‑conv)
# ----------------------------------------------------------------------
def _adam_basis_phase(n: int,
                      K: int,
                      deadline: float,
                      w0: np.ndarray | None = None,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Optimise the vector a = softmax(Phi @ w) on the smooth log‑sum‑exp
    approximation of max(conv(a,a)) using Adam.
    Returns the best a (sum = 1) and its exact objective value.
    """
    start = time.time()
    Phi = _build_basis(n, K)                       # (n, d)
    d = Phi.shape[1]

    if w0 is None:
        w = np.random.randn(d) * 0.1
    else:
        w = w0.copy()

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.90, 0.999
    t = 0
    best_a = None
    best_val = float('inf')
    no_improve = 0
    IMPROVE_LIMIT = 5000

    # schedules
    lr_start, lr_end = 0.15, 0.003
    eps_start, eps_end = 0.1, 1e-9

    while time.time() < deadline:
        # --- a on simplex (soft‑max) ---
        h = Phi @ w                                 # (n,)
        h_shift = h - np.max(h)                     # improve numerical stability
        a = np.exp(h_shift)
        a_sum = a.sum()
        if a_sum == 0.0:                            # fall back to uniform
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # --- objective (exact) ---
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[AdamBasis] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # --- schedules (linear in log‑time) ---
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # --- smooth max via log‑sum‑exp ---
        shifted = (b - max_b) / eps
        # avoid overflow by subtracting max before exp
        shift_max = np.max(shifted)
        weights = np.exp(shifted - shift_max)
        p = weights / weights.sum()                  # soft‑max over convolution positions

        # --- gradient w.r.t a (smooth objective) ---
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]           # the part that corresponds to a‑self terms
        grad_a = 4.0 * n * segment[::-1]             # ∂/∂a  2·n·∑_k p_k (a*a)_k

        # --- chain rule through the soft‑max (log‑softmax) ---
        dot = np.dot(grad_a, a)
        grad_h = a * (grad_a - dot)                  # ∂L/∂h

        # --- gradient w.r.t basis weights ---
        grad_w = Phi.T @ grad_h

        # --- Adam update ---
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.02, size=d)

        # restart if stuck for too long
        if no_improve > IMPROVE_LIMIT:
            print("[AdamBasis] restart after stagnation")
            w = np.random.randn(d) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

    # sanity: if nothing improved, fall back to uniform
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective_value(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – proportional to contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – preferably where moving mass hurts the max less
        if np.random.rand() < 0.35:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill     ] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and restart from the global best so far
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill     ] perturbing after stagnation")

        it += 1
    return best_a


# ----------------------------------------------------------------------
# Phase C – Exact sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray,
                       n: int,
                       start: float,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n            # initial step size (roughly 0.5·average(a))
    eta = eta0
    decay = 0.9995
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD      ] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step & projection onto the simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # decay step size slowly
        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # shake a little and restart step‑size
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD      ] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small post‑processing: mixing with uniform
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few convex combinations (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

    """

    TARGET = 1.503

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    sizes = [2000, 4000, 8000]

    for n in sizes:

            break
        print(f"\n=== Optimising size n={n} ===")
        remaining = deadline - time.time()
        if remaining < 30.0:          # not enough time for a full run
            break

        # --------------------------------------------------------------
        # Warm‑start: either interpolate from previous best or start uniform.
        # --------------------------------------------------------------
        if global_best_a is not None:
            old_n = len(global_best_a)
            old_x = np.linspace(0.0, 1.0, old_n)
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, global_best_a)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)

        # --------------------------------------------------------------
        # Phase A – Adam on a low‑dimensional Fourier basis.
        # --------------------------------------------------------------
        phase_a_deadline = time.time() + remaining * 0.30
        K = max(5, min(30, n // 250))          # number of cosine/sine pairs
        # initialise w0 by fitting log‑a0 in the basis (least‑squares)
        Phi = _build_basis(n, K)
        try:

        except Exception:                     # fallback to random init

        a_candidate, cand_val = _adam_basis_phase(

        if cand_val < global_best_val:
