def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of size next power of two.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # tiny negative values caused by rounding are removed
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Exact copy of evaluate_sequence for an already‑normalised vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (non‑negative, sum = 1). """
    # implementation of the well‑known algorithm by Wang & Carreira‑Perpinán
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_of_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution entry with respect to a.
    For a given shift k_max (the index where convolution attains its maximum),
    the gradient component j is 2 * a_{k_max - j} if the index is inside the
    array, otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    idx = k_max - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad[mask] = 2.0 * a[idx[mask]]
    return grad


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Perform a single projected sub‑gradient step that tries to reduce
    max(conv(a,a)).
    """
    conv = _fft_convolve_self(a)
    k_max = int(np.argmax(conv))
    grad_a = _gradient_of_max(a, k_max)
    # move opposite to the gradient
    a_new = a - step * grad_a
    # keep the vector feasible (non‑negative and sum == 1)
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_simplex(a_new)
    return a_new


def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic “large‑at‑ends” reorder.
    The sorted values are placed alternatingly at the left‑most and right‑most
    free positions, i.e. large – small – second large – second small – …
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a

# Core optimisation phases
# ----------------------------------------------------------------------
def _run_subgradient(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Run a few hundred projected sub‑gradient steps.
    The step‑size decreases geometrically.
    """
    a = a0.copy()
    step = 0.5
    while time.time() < deadline:
        a = _subgradient_step(a, step)
        step *= 0.995          # slow decay, keeps the process stable
        if step < 1e-7:
            break
    return a


def _simulated_annealing(a_start: np.ndarray,
                         cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Long‑run SA that mixes swaps and tiny mass transfers.
    Temperature is decreased slowly (factor 0.99994) so that
    the algorithm has enough time to cross barriers.
    """
    n = a_start.shape[0]
    rng = np.random.default_rng()
    a = a_start.copy()
    best_a = a.copy()
    best_val = cur_val

    # Initial temperature is a small fraction of the current value
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99994
    it = 0

    while time.time() < deadline:
        if rng.random() < 0.35:               # pure position swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                a[i], a[j] = a[j], a[i]
        else:                                 # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-12:
                delta = a[i] * (0.01 + 0.09 * rng.random())   # 1‑9 % of source mass
                a[i] -= delta
                a[j] += delta

        new_val = _eval_fast(a)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            # revert the move
            if rng.random() < 0.35:   # we performed a swap
                a[i], a[j] = a[j], a[i]
            else:
                a[i] += delta
                a[j] -= delta

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1

    return best_a, best_val


def _hill_climb(a_start: np.ndarray,
                cur_val: float,
                deadline: float) -> (np.ndarray, float):
    """
    Very cheap refinement that accepts any strictly improving move.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.3:                     # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # tiny transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-12:
                continue
            delta = a[i] * 0.015 * rng.random()    # ≤1.5 % of source mass
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta

        val = _eval_fast(cand)
        if val + 1e-12 < best_val:
            a = cand
            best_val = val
            print(f"[refine] new best = {best_val:.6f}")

    return a, best_val

            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using fresh uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)          # ℓ₁‑normalise to 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent (several short runs)
    # ------------------------------------------------------------------
    phase1_deadline = deadline - 400.0   # keep ~6 min for later phases
    while time.time() < phase1_deadline:
        # small random perturbation before each run
        a_candidate = best_a + 1e-3 * np.random.rand(n)
        a_candidate = _project_simplex(a_candidate)

        a_candidate = _run_subgradient(a_candidate, phase1_deadline)
        cur_val = _eval_fast(a_candidate)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_candidate.copy()
            print(f"[phase1] improved to {best_val:.6f}")

        # apply the “ends‑large” re‑ordering, which often decorrelates large
        # entries and helps the next optimisation round
        a_ord = _ends_large_reorder(best_a)
        a_ord = _project_simplex(a_ord)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – very long simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("[phase2] simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val, deadline - 140.0)

    # ------------------------------------------------------------------
    # Phase 3 – final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[phase3] final hill‑climbing")
        best_a, best_val = _hill_climb(best_a, best_val, deadline)

    # Final official evaluation (uses the provided security‑checked function)
    # ------------------------------------------------------------------
    try:

        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    # Return a plain Python list as required by the framework
    return best_a.tolist()
