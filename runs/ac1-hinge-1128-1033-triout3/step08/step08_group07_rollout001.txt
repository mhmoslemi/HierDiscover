    n = a.shape[0]
    size = _next_pow2(2*n-1)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2*n-1]
    conv = np.maximum(conv, 0)  # clip negative
    return conv

"""
A lightweight exponentiated‑gradient optimizer for the step‑function
minimisation problem.

The algorithm works in four stages within a 1000 s wall‑time budget:

1. **Exponentiated‑gradient descent (EG)** – the sequence is represented by
   soft‑max parameters `w`.  At each iteration we compute the convolution
   `c = a * a`, locate the index of its maximal entry, obtain the
   sub‑gradient of the max‑convolution w.r.t. `a`, transform it to a
   gradient w.r.t. `w` and perform a diminishing‑step‑size update.
   The soft‑max projection guarantees non‑negativity and ℓ₁‑normalisation.
   This stage runs for the bulk of the time (≈ 800 s) and drives the
   solution close to a local optimum.

2. **Deterministic “ends‑large’’ re‑ordering** – after EG we try a simple
   heuristic that places the largest values at the two ends of the array,
   which often reduces the peak of the self‑convolution.

3. **Simulated‑annealing on swaps + tiny mass transfers** – a classic SA
   that accepts worsening moves with a temperature proportional to the
   current objective.  It relaxes the solution slightly and can escape
   shallow basins.

4. **Final hill‑climbing refinement** – greedy swaps and tiny mass moves
   accept only strict improvements and polish the best found candidate.

All heavy work (convolution, gradients) uses an FFT‑based O(n log n)
implementation.  The code never creates closures or uses lambdas,
conforms to the required `propose_candidate` signature and prints progress
whenever a new best value is discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to ``x``."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns the array of length ``2*len(a) - 1``.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Clip tiny negatives that arise from round‑off errors
    np.maximum(conv, 0, out=conv)
    return conv


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    w_shift = w - np.max(w)
    e = np.exp(w_shift)
    return e / np.sum(e)


def _eval_fast(a: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` for a normalised vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic heuristic: place the biggest values at the two ends,
    then alternately pick the smallest remaining value.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)            # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


# ----------------------------------------------------------------------
# Simulated annealing (swap + tiny mass moves)
# ----------------------------------------------------------------------


def _simulated_annealing_mass(a: np.ndarray,
                              cur_val: float,
                              deadline: float) -> (np.ndarray, float):
    """
    Simulated annealing that alternates between a random swap and a tiny
    transfer of mass.  ``a`` is modified in‑place.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # Initial temperature is a small fraction of the current value
    T = max(cur_val * 0.02, 1e-7)
    decay = 0.99993

    it = 0
    while time.time() < deadline:
        # Choose move type
        if rng.random() < 0.3:                       # -- swap --
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            move = ("swap", i, j)
        else:                                         # -- tiny transfer --
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-14:
                continue
            delta = a[i] * 0.03 * rng.random()        # ≤ 3 % of source mass
            a[i] -= delta
            a[j] += delta
            move = ("transfer", i, j, delta)

        new_val = _eval_fast(a)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            # Accept move
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] iter {it:06d} new best = {best_val:.6f}")
        else:
            # Reject – revert
            if move[0] == "swap":
                i, j = move[1], move[2]
                a[i], a[j] = a[j], a[i]
            else:
                i, j, delta = move[1], move[2], move[3]
                a[i] += delta
                a[j] -= delta

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing (strictly improving moves only)
# ----------------------------------------------------------------------


def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Greedy local search that accepts any strictly improving swap or transfer."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                        # tiny transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-14:
                continue
            delta = cur_a[i] * 0.02 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] iter {it:06d} new best = {cur_val:.6f}")

        it += 1

    return cur_a, cur_val
