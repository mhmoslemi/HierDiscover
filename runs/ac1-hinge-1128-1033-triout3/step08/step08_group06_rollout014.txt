def max_convolution(a):
    n=len(a)
    size = 1 << ((2*n-1)-1).bit_length() # next pow2
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:2*n-1]
    return conv.max()

def balanced_reorder(a):
    n = len(a)
    indices = list(range(n))
    # recursively assign positions
    order = []

    def assign(lo, hi):
        if lo > hi:
            return
        mid = (lo+hi)//2
        order.append(mid)
        assign(lo, mid-1)
        assign(mid+1, hi)
    assign(0, n-1)
    # now order list of indices in which to place descending values
    # fill with sorted descending a
    sorted_desc = np.sort(a)[::-1]
    b = np.empty(n, dtype=float)
    for idx, val in zip(order, sorted_desc):
        b[idx] = val
    return b

def simulated_annealing(a_init, deadline):

   a = a_init.copy()
   best = a.copy()
   best_val = objective(best)
   n = a.shape[0]
   rng = np.random.default_rng(12345) # but we want deterministic maybe seed derived from global.

   T = 1.0  # initial temperature maybe set adaptively based on best_val.

   iteration=0

   while time.time() < deadline:

        iteration+=1

        move_type = rng.random()
        if move_type < 0.2:
            # swap
            i,j = rng.integers(0,n, size=2)
            if i==j: continue
            a_new = a.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]
        else:
            # mass transfer
            i,j = rng.integers(0,n,size=2)
            if i==j: continue
            if a[i] <= 1e-15: continue
            delta = a[i] * rng.random() * 0.1  # up to 10% of mass
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

        # ensure non-negative and renormalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s ==0: continue
        a_new = a_new / s

        val = objective(a_new)
        cur_val = objective(a)  # compute current? we can maintain cur_val variable.

        # Accept condition:
        if val < cur_val:
            accept = True
        else:
            # annealing probability based on T
            accept = np.exp(-(val - cur_val)/max(T, 1e-12)) > rng.random()
        if accept:
            a = a_new
            cur_val = val

        if val < best_val:
            best_val = val
            best = a_new.copy()
            print(f"[anneal iter {iteration}] new best = {best_val:.6f}")

        # temperature decay
        if iteration % 1000 == 0:
            T *= 0.99  # slowly decay

        # occasional reorder
        if iteration % 5000 == 0:
            a_ord = balanced_reorder(a)
            a_ord = a_ord / a_ord.sum()
            val_ord = objective(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best = a_ord.copy()
                a = a_ord.copy()
                cur_val = val_ord
                print(f"[balanced reorder iter {iteration}] improved to {best_val:.6f}")

   return best, best_val

"""
propose_candidate
-----------------

Search for a non‑negative step function that minimises
``evaluate_sequence``.  The algorithm works with a modest
vector size (default 1024) so that many cheap trial moves can be
tried within the 1000 s budget.

* 0.  Fast objective: 2·n·max(conv(a,a)) / (sum a)²,
   where the convolution is computed by FFT.
* 1.  Initialise a diverse pool of candidates (uniform,
   exponential, power‑law, geometric) and keep the best.
* 2.  A simple sub‑gradient descent on the true L∞‑objective is
   performed (mirrored exponentiated update).
* 3.  Long simulated‑annealing phase with three move types:
   - swap two entries,
   - transfer a small amount of mass from i → j,
   - occasionally apply a *balanced* re‑ordering that spreads
     the largest values as far apart as possible.
* 4.  After each successful move the vector is projected back onto
   the probability simplex (non‑negative, unit sum).
* 5.  The best vector encountered is returned as a plain Python
   list of floats.

The code is pure NumPy + standard library, uses no file or network
IO, and respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return max_{k} (a * a)[k] using FFT.
    a is assumed to be a 1‑D non‑negative array.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # tiny negative artefacts from floating point are ignored
    return float(conv.max())


def _objective(a: np.ndarray) -> float:
    """
    Compute the exact value that ``evaluate_sequence`` would return,
    but much faster (no clipping, no NaN checks – the caller ensures
    a is well‑behaved).
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    max_conv = _max_convolution_fft(a)
    n = a.shape[0]
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Largest values on the left, smallest on the right (previous heuristic)."""
    s = np.sort(a)                # ascending
    n = a.shape[0]
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]         # large → left side
        res[n - 1 - i] = s[i]         # small → right side
    if n % 2:
        res[half] = s[half]
    return res


def _balanced_reorder(a: np.ndarray) -> np.ndarray:
    """
    Spread the largest values as far apart as possible.
    The indices are visited in a binary‑search order (mid, left‑mid,
    right‑mid, …) and the sorted values are placed accordingly.
    """
    n = a.shape[0]
    order = []

    def _rec(lo: int, hi: int):
        if lo > hi:
            return
        mid = (lo + hi) // 2
        order.append(mid)
        _rec(lo, mid - 1)
        _rec(mid + 1, hi)

    _rec(0, n - 1)

    sorted_desc = np.sort(a)[::-1]          # largest → smallest
    b = np.empty_like(a)
    for idx, val in zip(order, sorted_desc):
        b[idx] = val
    return b


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------
def _subgradient_descent(a_start: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Exponentiated mirror‑descent that reduces the *active* convolution peaks.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    alpha0 = 0.8          # initial step size factor
    rel_tol = 0.05        # initial active‑set tolerance (percentage)

    while time.time() < deadline - 5.0:
        # full convolution via FFT (fast)
        conv = np.fft.irfft(np.fft.rfft(a, _next_pow2(2 * n - 1)) ** 2,
                            _next_pow2(2 * n - 1))[: 2 * n - 1]

        max_conv = conv.max()
        # tighten tolerance gradually
        if it and it % 20000 == 0:
            rel_tol = max(rel_tol * 0.5, 1e-5)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]   # indices where conv is near its max

        # compute sub‑gradient for the L∞ norm
        mask = np.zeros(2 * n - 1, dtype=np.float64)
        mask[active] = 1.0
        mask_rev = mask[::-1]

        size = _next_pow2(n + mask_rev.shape[0] - 1)
        fa = np.fft.rfft(a, size)
        fm = np.fft.rfft(mask_rev, size)
        conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

        g = conv_full[(2 * n - 2) - np.arange(n)]   # sub‑gradient vector

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative update
        a *= np.exp(-alpha * g)

        # project onto the probability simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional logging and re‑ordering
        if it % 500 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f}")

            if it % 5000 == 0:
                # try the balanced re‑ordering – it often improves the peak
                a_bal = _balanced_reorder(a)
                a_bal /= a_bal.sum()
                val_bal = _objective(a_bal)
                if val_bal < best_val:
                    best_val = val_bal
                    best_a = a_bal.copy()
                    a = a_bal.copy()
                    print(f"[sgd balanced reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Random local moves (swap, mass transfer, occasional re‑ordering)
    with an annealing schedule.
    """
    cur_a = a_start.copy()
    cur_val = _objective(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    n = cur_a.shape[0]
    rng = np.random.default_rng(123456)
    T = 0.02                     # initial temperature
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- propose a move -------------------------------------------------
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # ----- re‑project onto the simplex ------------------------------------
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        # ----- evaluate --------------------------------------------------------
        val = _objective(new_a)

        # ----- Metropolis acceptance -------------------------------------------
        if val < cur_val:
            accept = True
        else:
            T_cur = max(T, 1e-12)
            accept = np.exp(-(val - cur_val) / T_cur) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val - 1e-12:
                best_val = val
                best_a = new_a.copy()
                print(f"[anneal iter {it}] new best = {best_val:.6f}")

        # ----- temperature decay ------------------------------------------------
        if it % 1000 == 0:
            T *= 0.9995

        # ----- occasional re‑ordering -----------------------------------------
        if it % 5000 == 0:
            a_bal = _balanced_reorder(cur_a)
            a_bal /= a_bal.sum()
            val_bal = _objective(a_bal)
            if val_bal < best_val:
                best_val = val_bal
                best_a = a_bal.copy()
                cur_a = a_bal.copy()
                cur_val = val_bal
                print(f"[anneal balanced reorder] improved to {best_val:.6f}")

            a_asym = _anti_symmetric_reorder(cur_a)
            a_asym /= a_asym.sum()
            val_asym = _objective(a_asym)
            if val_asym < best_val:
                best_val = val_asym
                best_a = a_asym.copy()
                cur_a = a_asym.copy()
                cur_val = val_asym
                print(f"[anneal anti‑sym reorder] improved to {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to build a few structured candidates (stage 0)
# ----------------------------------------------------------------------
def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (unnormalised)."""
    a = (np.arange(1, n + 1) ** (-p)).astype(np.float64)
    a /= a.sum()
    return a


def _geometric_candidate(lam: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a

# Main entry point
# ----------------------------------------------------------------------

    for _ in range(4):
