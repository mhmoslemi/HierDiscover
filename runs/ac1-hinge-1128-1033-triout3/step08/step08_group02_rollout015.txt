def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2
    # choose n
    n = 8000 # maybe increase if time permits
    # load previous if exists else generate init
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        # Interpolate if old length differs
        if old.shape[0] != n:
            x_old = np.linspace(0, 1, old.shape[0])
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, old)
        else:
            a = old.copy()
        a = np.maximum(a, 0)
        if a.sum() == 0:
            a = np.full(n, 1/n)
        else:
            a /= a.sum()
    except Exception:
        a, _ = _initial_candidate(n)
    best_a = a.copy()
    best_val = _objective(best_a, n)
    print(f"[init] best_val = {best_val:.6f}")
    # parameters
    lambda_start = 5.0
    lambda_end = 2000.0
    eta_start = 0.5
    eta_end = 0.001
    # iteration
    it = 0
    no_improve = 0
    while time.time() < deadline:
        progress = (time.time() - start_time) / (deadline - start_time)
        lam = lambda_start * (lambda_end / lambda_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress   # exponential decay
        # compute convolution a*a
        conv = _full_convolution(a)
        # compute smoothed max weight
        # use stable softmax
        maxc = conv.max()
        exp_terms = np.exp(lam * (conv - maxc))
        w = exp_terms / exp_terms.sum()
        # compute c vector
        a_rev = a[::-1]
        # compute conv_rev = a_rev * w
        # size for FFT
        len_c = len(a_rev) + len(w) - 1
        size = 1 << (len_c - 1).bit_length()
        # FFTs
        fa_rev = np.fft.rfft(a_rev, size)
        fw = np.fft.rfft(w, size)
        conv_c_full = np.fft.irfft(fa_rev * fw, size)[:len_c]
        c = conv_c_full[n-1:n-1+n]
        grad = c + a
        # exponentiated gradient update
        a = a * np.exp(-eta * grad)
        # renorm
        total = a.sum()
        if total == 0:
            a = np.full(n, 1/n)
        else:
            a /= total
        # evaluate objective (exact)
        cur_val = 2.0 * n * conv.max()
        # keep best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[iter {it:07d}] new best = {best_val:.6f} (lam {lam:.2f} eta {eta:.5f})")
            # early break if reached target
            if best_val <= 1.503:
                break
        else:
            no_improve += 1
        # jitter if stuck
        if no_improve > 2000:
            # small random perturbation preserving sum
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            no_improve = 0
            print(f"[iter {it}] jitter")
        it += 1
    # final mixing with uniform maybe
    best_a, best_val = _mix_with_uniform(best_a, n)
    # final evaluation compute evaluate_sequence
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence error:", e)
    return best_a.tolist()

    cssv = np.cumsum(u) - 1

        return np.full_like(v, 1.0/n)

    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimises

        f(a) = 2·n·max(conv(a,a)) / (sum a)²

The algorithm consists of:

1.  **Initial families** – a quick sweep over several deterministic
    and random families (uniform, powers, exponentials, U‑shapes,
    Dirichlet draws).  The best vector found is kept.

2.  **Smoothed‑max gradient descent** – we replace the non‑smooth
    max by a soft‑max (temperature λ).  For a given λ the gradient of
    the smooth objective is

            g = 2·c,   cⱼ = Σᵢ aᵢ·w_{i+j}

    where w is the soft‑max weight of the convolution entries.
    The gradient is computed by a single FFT‑based convolution of the
    reversed vector with w.  An additive step is taken and the result
    is projected back onto the probability simplex.  λ and the step
    size η are increased / decreased exponentially, giving a coarse
    global search that gradually focuses on the true maximum.

3.  **Deterministic greedy hill‑climb** – a short post‑processing phase
    identical to the one used in the previous solution, moving a small
    fraction of mass from the “worst” to the “best” index while
    guaranteeing improvement.

4.  **Uniform mixing** – a tiny convex combination with the uniform
    vector is tried (α∈{0.001,…,0.020}) which occasionally reduces the
    peak further.

The code respects the 1000 s budget, prints intermediate progress
and finally returns the best list of floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of reversed a with weight vector w; returns the centre n entries."""
    n = a_rev.shape[0]
    len_c = n + w.shape[0] - 1
    size = 1 << (len_c - 1).bit_length()
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    full = np.fft.irfft(fa * fw, size)[:len_c]
    return full[n - 1:n - 1 + n]


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1 , x ≥ 0)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):               # all entries are non‑positive
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                # last positive index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families – fast initial sweep
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)                # uniform
    best_val = 2.0 * n * _full_convolution(best_a).max()

    def try_update(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = 2.0 * n * _full_convolution(a).max()
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(12345)

    idx = np.arange(n)
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # linear U‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # random Dirichlet draws (quick)
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑B – deterministic greedy hill‑climb (same as previous code)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while time.time() < deadline:
        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k = int(np.argmax(b))
        # reverse‑relevant slice of a (a_{k‑i})
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        i = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it}] jitter after stagnation")
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # keep move
            pass
        else:
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val

def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, 2.0 * n * _full_convolution(a).max()
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = 2.0 * n * _full_convolution(cand).max()
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under the given time budget.
    Returns the best list of floats found (Σa = 1) which minimises
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # Problem size – 8000 entries give a good compromise
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from a previous construction if available; otherwise use families
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = 2.0 * n * _full_convolution(best_a).max()
    print(f"[init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – smoothed‑max gradient descent
    # ------------------------------------------------------------------
    lam_start = 50.0          # start temperature (soft‑max sharpness)
    lam_end = 2000.0          # final temperature
    eta_start = 15.0          # large first step size
    eta_end = 0.2             # final step size
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    while time.time() < deadline:
        progress = (time.time() - start_time) / (deadline - start_time)
        lam = lam_start * (lam_end / lam_start) ** progress   # exponential increase
        eta = eta_start * (eta_end / eta_start) ** progress   # exponential decay

        conv = _full_convolution(best_a)
        cur_obj = 2.0 * n * conv.max()

        # Record improvement
        if cur_obj < best_val - 1e-12:
            best_val, best_a = cur_obj, best_a.copy()
            no_improve = 0
            print(f"[GD {it:7d}] best = {best_val:.6f}  λ={lam:.1f} η={eta:.3f}")
            if best_val <= 1.503:
                break
        else:
            no_improve += 1

        # ----- gradient of the smoothed max -----
        # soft‑max weights
        maxc = conv.max()
        w = np.exp(lam * (conv - maxc))
        w /= w.sum()

        a_rev = best_a[::-1]
        c = _conv_rev_weight(a_rev, w)          # c has length n
        grad = 2.0 * c

        # ----- additive step + projection onto simplex -----
        y = best_a - eta * grad
        best_a = _simplex_projection(y)

        # occasional jitter if we are stuck for a while
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.004, size=n)
            best_a = np.maximum(best_a + noise, 0.0)
            best_a /= best_a.sum()
            no_improve = 0
            print(f"[GD {it}] jitter after stagnation")
        it += 1

    # ------------------------------------------------------------------
    # Phase B – deterministic greedy hill‑climb (shortened)
    # ------------------------------------------------------------------
    phase_b_deadline = deadline
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        start=start_time,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )

    # ------------------------------------------------------------------
    # Phase C – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification through the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
