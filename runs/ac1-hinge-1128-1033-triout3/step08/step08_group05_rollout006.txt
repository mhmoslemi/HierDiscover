def symmetric_edge_reorder(a):
    # sort descending, then place largest at both edges alternating
    n = len(a)
    sorted_vals = np.sort(a)[::-1]   # descending
    new_a = np.zeros_like(a)
    left = 0
    right = n-1
    idx = 0
    while left <= right:
        # assign to left
        new_a[left] = sorted_vals[idx]
        idx += 1
        if left == right:
            break
        new_a[right] = sorted_vals[idx]
        idx += 1
        left += 1
        right -= 1
    # ensure sum=1
    new_a /= new_a.sum()
    return new_a

Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Algorithm overview
-----------------
1. **Rich initialisation** – generate a diverse pool of candidates:
   uniform, previous solution (if any), exponential/U‑shaped families,
   polynomial families, and a few Dirichlet draws.

2. **Smooth surrogate optimisation** – minimise a log‑sum‑exp approximation
   of the max‑convolution with L‑BFGS on the soft‑max parametrisation.
   The smoothing parameter ε is decreased on a logarithmic schedule.
   Several random restarts are performed.

3. **Exact mirror‑descent** – use the sub‑gradient of the exact
   max‑convolution together with an exponential (mirror) update and
   simplex projection.

4. **Annealed pairwise hill‑climbing** – repeatedly move a small
   fraction of mass from the pair that contributes most to the current
   peak towards a low‑mass index, accepting uphill moves with a
   temperature schedule.

5. **Cheap final polishing** – stochastic swaps / small mass transfers
   that accept only strict improvements.

All stages respect the 1000 s time budget; the best sequence found is
returned as a plain Python list and finally evaluated with the official
``evaluate_sequence`` for safety.

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None  # smoothing parameter ε
_GLOBAL_N = None    # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Remove tiny negative artefacts caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient wrt a -----
    weights = np.exp(shifted)
    p = weights / weights.sum()      # probability distribution over lag indices

    # derivative of conv_k w.r.t. a is a convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]    # reverse to align with a_i order

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors on the simplex
    (Σ a_i = 1).  Returns 2·n·max(conv) (the denominator Σ a_i² = 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Shape generators for diverse initial candidates
# ----------------------------------------------------------------------
def _shape_exponential(n: int, beta: float) -> np.ndarray:
    """U‑shaped exponential decay from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-beta * d)
    return a / a.sum()


def _shape_polynomial(n: int, p: float) -> np.ndarray:
    """U‑shaped power‑law decay from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)
    a = (d + 1.0) ** (-p)
    return a / a.sum()


def _shape_linear(n: int) -> np.ndarray:
    """Linear decrease from the edges to the centre."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)
    a = (n - d).astype(float)
    return a / a.sum()


def _shape_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw (α‑parameter)."""
    a = rng.gamma(shape=alpha, scale=1.0, size=n)
    return a / a.sum()


def _symmetric_edge_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values symmetrically on both edges,
    the next largest values next‑to‑the‑edges, and so on.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)[::-1]          # descending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    pos = 0
    while left <= right:
        new_a[left] = sorted_vals[pos]
        pos += 1
        if left == right:
            break
        new_a[right] = sorted_vals[pos]
        pos += 1
        left += 1
        right -= 1
    # Re‑normalise to guard against numerical drift
    new_a /= new_a.sum()
    return new_a


# ----------------------------------------------------------------------
# Very cheap final hill‑climber (strict improvements only)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climber that accepts only strict improvements."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Simple simplex projection (non‑negative + normalisation)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list before the
    allotted time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for final check

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # 1. uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # 2. previous solution (if available)
    if has_old:
        old_norm = np.maximum(old, 0.0)
        s = old_norm.sum()
        if s < 1e-12:
            old_norm = np.full(n, 1.0 / n)
        else:
            old_norm = old_norm / s
        init_candidates.append(old_norm)

    # 3. exponential family (edge‑decay)
    for beta in np.logspace(-5, -0.5, num=12):
        init_candidates.append(_shape_exponential(n, beta))

    # 4. polynomial family (edge‑power‑law)
    for p in np.linspace(0.2, 5.0, num=12):
        init_candidates.append(_shape_polynomial(n, p))

    # 5. linear edge shape
    init_candidates.append(_shape_linear(n))

    # 6. a few Dirichlet draws
    for alpha in (0.3, 0.7, 1.5, 3.0):
        init_candidates.append(_shape_dirichlet(n, alpha, rng))

    # ------------------------------------------------------------------
    # Evaluate the pool and keep the best
    # ------------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s
        val = _fast_evaluate(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
        # occasional symmetric edge reorder to see if it helps immediately
        if idx < 6:
            cand2 = _symmetric_edge_reorder(cand)
            val2 = _fast_evaluate(cand2)
            if val2 < best_val:
                best_val = val2
                best_a = cand2.copy()
        print(f"[init {idx}] value = {val:.6f}")

    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on smooth surrogate
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-1), np.log10(1e-14), num=25)
    w = np.log(best_a + 1e-12)  # soft‑max parameters

    max_restarts = 12
    for restart in range(max_restarts):
        if time.time() >= deadline - 150:
            break
        if restart % 3 == 0:
            w = rng.normal(size=n) * 1.0   # fresh random start

        for eps in eps_schedule:
            if time.time() >= deadline - 150:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            # back‑transform to probabilities
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _fast_evaluate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[L‑BFGS restart {restart+1} eps {eps:.1e}] new best = {best_val:.6f}")

        # symmetric edge reordering after each restart
        a_ord = _symmetric_edge_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – refined mirror descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    base_alpha = 0.5
    decay = 0.9999
    min_alpha = 0.005

    while time.time() < deadline - 100:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # Sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]          # derivative of a_i a_{k-i} w.r.t. a_i

        # Exponential (mirror) update
        alpha = max(base_alpha * (decay ** (it // 200)), min_alpha)
        a = a * np.exp(-alpha * g)

        # Project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        if it % 200 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[MD it {it}] new best = {best_val:.6f}")
            # occasional symmetric reorder
            if it % 1000 == 0:
                a_ord = _symmetric_edge_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[MD reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – targeted pairwise mass‑transfer hill‑climbing with annealing
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.02                     # initial temperature
    cooling = 0.99995

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        # indices that actually participate in the maximal entry
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        contrib = cur_a[idx] * cur_a[jdx]

        # pair that contributes most to the peak
        pair_idx = np.argmax(contrib)
        i = idx[pair_idx]
        j = jdx[pair_idx]

        donor = i if cur_a[i] >= cur_a[j] else j

        # pick a low‑mass recipient (bottom 5 % of entries)
        low_mass = np.where(cur_a < np.percentile(cur_a, 5))[0]
        if low_mass.size == 0:
            recipient = rng.integers(0, n)
        else:
            recipient = rng.choice(low_mass)

        delta = cur_a[donor] * 0.03 * rng.random()   # up to 3 % of donor mass
        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[recipient] += delta

        # projection onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[pairhill] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-7)

    # ------------------------------------------------------------------
    # Phase 4 – final cheap polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
