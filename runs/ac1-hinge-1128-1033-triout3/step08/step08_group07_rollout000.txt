"""
Hybrid Linear‑Program + Stochastic optimiser for the step‑function problem.

The search works in three phases:

1.  **LP improvement** –  Starting from a strictly positive distribution
    a (ℓ₁‑normalised to 1) we repeatedly solve the linear programme

           max   Σ b_i
           s.t.  (a * b)_k ≤ M   for all k = 0,…,2n‑2
                 b_i ≥ 0

    where M = max(a*a).  The LP pushes mass into positions where a is
    small, decreasing the next max‑convolution.  The new vector is
    blended with the old one (convex combination) and re‑normalised.
    A tiny “ends‑large” re‑ordering is applied every few iterations.

2.  **Short simulated‑annealing** –  With the improved vector we run a
    classic Metropolis SA that swaps indices and moves a small fraction
    of the mass between two positions.  This can escape shallow local
    minima left by the LP phase.

3.  **Cheap hill‑climbing** –  Finally we perform deterministic
    improving moves (swap or tiny mass transfer) until the time budget
    runs out.

All heavy work (convolutions, max‑values) is performed with an FFT‑based
routine ` _fft_convolve_self`.  The code respects the 1000 s wall‑time
budget, prints any new best value and returns the best sequence (as a
plain Python list) found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import linprog
from typing import List

# -------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ∗ a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # round‑off may produce tiny negatives
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for a vector that is already
    ℓ₁‑normalised (∑a = 1).  Returns 2·n·max(conv) .
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    # denominator is 1 (∑a = 1) → objective = 2·n·max_b
    return float(2.0 * n * max_b)


# -------------------------------------------------------------
# Simple deterministic re‑ordering: large values to the ends
# -------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values at the two extremes, then alternate."""
    n = a.shape[0]
    sorted_vals = np.sort(a)                     # ascending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    pos = 0
    while pos < n:
        new_a[pos] = sorted_vals[right]          # large
        right -= 1
        pos += 1
        if pos >= n:
            break
        new_a[pos] = sorted_vals[left]           # small
        left += 1
        pos += 1
    return new_a


# -------------------------------------------------------------
# Main optimisation entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence discovered before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the official evaluation
    deadline = start_time + budget_s - 1.0
    print(f"[init] budget={budget_s}s, deadline≈{deadline:.2f}")

    # -----------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # -----------------------------------------------------------------
    n = 2500                     # you may tune this up/down
    eps_min = 1e-12             # avoid exact zeros (LP would become unbounded)

    # -----------------------------------------------------------------
    # Initialise a – try to reuse a previously discovered construction.
    # -----------------------------------------------------------------
    try:
        # `height_sequence_1` may be present in the global namespace.
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required size
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
            print("[init] loaded previous construction (interpolated)")
        else:
            a = old.copy()
            print("[init] loaded previous construction (exact size)")
    except Exception:
        # fresh random start – guarantee strict positivity
        a = np.random.rand(n) + eps_min
        print("[init] fresh random start")

    # enforce strict positivity and ℓ₁‑normalisation
    a = np.maximum(a, eps_min)
    a /= np.sum(a)

    best_a = a.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial objective = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Phase 1 – LP based improvement
    # -----------------------------------------------------------------
    max_lp_iters = 12                     # enough for a noticeable drop
    t = 0.5                               # convex‑combination step size
    lp_start = time.time()
    for lp_iter in range(1, max_lp_iters + 1):
        if time.time() >= deadline - 30.0:
            print("[LP] time left insufficient, breaking")
            break

        # current bound M = max(a*a)
        conv_a = _fft_convolve_self(a)
        M = np.max(conv_a)

        # build the (2n‑1) × n matrix A with rows = shifted copies of a
        rows = 2 * n - 1
        A = np.empty((rows, n), dtype=np.float64)
        A.fill(0.0)
        for i in range(n):
            A[i:i + n, i] = a                     # column i = a shifted down

        # solve   max Σ b_i   s.t.   A @ b ≤ M·1 ,  b ≥ 0
        c = -np.ones(n, dtype=np.float64)         # linprog minimises c·x
        b_ub = M * np.ones(rows, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        # free the (large) matrix early
        del A

        if not res.success:
            print(f"[LP] iteration {lp_iter} – linprog failed: {res.message}")
            break

        b = res.x
        sum_b = np.sum(b)
        if sum_b <= eps_min:
            print(f"[LP] iter {lp_iter} – sum(b) too small, abort")
            break

        # normalise b (keep total mass = 1) and convex‑combine
        b_norm = b / sum_b
        a_candidate = (1.0 - t) * a + t * b_norm
        a_candidate = np.maximum(a_candidate, eps_min)
        a_candidate /= np.sum(a_candidate)

        cand_val = _eval_fast(a_candidate)
        print(f"[LP] iter {lp_iter:2d}  val={cand_val:.6f}  t={t:.3f}")

        # accept if we got improvement, otherwise shrink step size
        if cand_val + 1e-12 < best_val:
            best_val = cand_val
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"   -> new best (LP) = {best_val:.6f}")
        else:
            t *= 0.5
            if t < 1e-3:
                print("[LP] step size too small, terminating LP phase")
                break

        # occasional heuristic re‑ordering (every 4 LP steps)
        if lp_iter % 4 == 0:
            a = _ends_large_reorder(a)
            a /= np.sum(a)
            val_re = _eval_fast(a)
            if val_re < best_val:
                best_val, best_a = val_re, a.copy()
                print(f"[LP reorder] improved to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Phase 2 – brief simulated annealing (swap + tiny mass moves)
    # -----------------------------------------------------------------
    if time.time() < deadline - 15.0:
        print("[SA] starting short simulated annealing")
        cur_a = best_a.copy()
        cur_val = best_val
        rng = np.random.default_rng()
        T = max(cur_val * 0.02, 1e-7)
        decay = 0.9997
        sa_iters = 0
        while time.time() < deadline - 5.0:
            sa_iters += 1
            if rng.random() < 0.5:
                # swap two positions
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
            else:
                # tiny mass transfer
                i, j = rng.integers(0, n, size=2)
                if i != j and cur_a[i] > eps_min:
                    delta = cur_a[i] * 0.05 * rng.random()
                    cur_a[i] -= delta
                    cur_a[j] += delta

            new_val = _eval_fast(cur_a)
            delta_val = new_val - cur_val
            accept = False
            if delta_val < 0:
                accept = True
            else:
                if rng.random() < np.exp(-delta_val / T):
                    accept = True

            if accept:
                cur_val = new_val
                if cur_val + 1e-12 < best_val:
                    best_val = cur_val
                    best_a = cur_a.copy()
                    print(f"[SA] iter {sa_iters:6d} new best = {best_val:.6f}")
            else:
                # revert the move
                if rng.random() < 0.5:
                    cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
                else:
                    cur_a[i] += delta
                    cur_a[j] -= delta

            T = max(T * decay, 1e-12)

    # -----------------------------------------------------------------
    # Phase 3 – final cheap hill‑climbing (only improving moves)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        print("[Hill] final greedy refinement")
        cur_a = best_a.copy()
        cur_val = best_val
        rng = np.random.default_rng()
        while time.time() < deadline:
            # try a random swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                trial = cur_a.copy()
                trial[i], trial[j] = trial[j], trial[i]
                val = _eval_fast(trial)
                if val + 1e-12 < cur_val:
                    cur_a, cur_val = trial, val
                    best_a, best_val = cur_a.copy(), cur_val
                    print(f"[Hill] swap improved -> {best_val:.6f}")
                    continue

            # try a tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and cur_a[i] > eps_min:
                delta = cur_a[i] * 0.02 * rng.random()
                trial = cur_a.copy()
                trial[i] -= delta
                trial[j] += delta
                val = _eval_fast(trial)
                if val + 1e-12 < cur_val:
                    cur_a, cur_val = trial, val
                    best_a, best_val = cur_a.copy(), cur_val
                    print(f"[Hill] transfer improved -> {best_val:.6f}")

    # -----------------------------------------------------------------
    # Final official evaluation (using the provided security‑checked function)
    # -----------------------------------------------------------------
    try:

    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    # Return the pure Python list as required by the platform
    return best_a.tolist()
