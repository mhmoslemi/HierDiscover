def project_to_simplex(v):
    # v is a non-negative vector; we project onto simplex sum=1.
    # Implementation: sort descending, find rho, compute theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, len(v)+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def grad_from_weights(w, a):
    n = a.shape[0]
    len_w = w.shape[0]
    size = 1 << ((len_w + n - 1).bit_length())

"""
Soft‑max gradient descent + greedy polishing for the
convolution‑peak minimisation problem.

The objective (after normalising Σa = 1) is
        f(a) = 2·n·max_k (a * a)_k .
A smooth approximation of the max is
        L_γ(a) = (1/γ)·log Σ_k exp(γ·b_k) ,   b = a∗a .
∇L_γ = 2·(w * a)  where
        w_k = exp(γ·b_k) / Σ_j exp(γ·b_j) .
The algorithm iteratively:
   1. normalises a to the probability simplex,
   2. computes b = a∗a (FFT, O(n log n)),
   3. builds the soft‑max weights w,
   4. obtains the gradient by another FFT‑based convolution,
   5. applies Adam (multiplicative‑gradient‑like) updates,
   6. projects back onto the simplex.

When the schedule for γ grows the loss focuses on the true maximum,
driving the peak down.  After the continuous phase a short deterministic
greedy hill‑climb (the same idea used in the reference code) is run to
fine‑tune the result, followed by a tiny uniform mixing step.

The whole routine respects a 1000 s wall‑time budget and returns the best
vector found (as a plain ``list[float]``) which can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    #  reference: Duchi et al., “Efficient Projections onto the L1‑Ball”
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Full linear convolution via FFT (real‑valued)
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Returns the linear convolution a * b (full length) using FFT.
    Both inputs are real, non‑negative vectors.
    """
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:n]
    return conv

# ----------------------------------------------------------------------
# Gradient of the soft‑max loss L_γ (see docstring)
# ----------------------------------------------------------------------
def _softmax_grad(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Given the weight vector w (length 2·n‑1, Σw = 1) and the current a
    (length n), return ∇L_γ = 2·(w ⋆ a) where ⋆ denotes the linear
    cross‑correlation needed for the gradient.
    """
    # The required expression is   g_i = 2·∑_k w_k·a_{k-i}
    # which can be obtained as
    #   g = 2·conv(w_rev, a)[offset - np.arange(n)]
    w_rev = w[::-1]
    conv = _conv_full(w_rev, a)          # length = len(w)+n‑1
    offset = len(w) - 1                  # index of zero‑lag term
    n = a.shape[0]
    grad = 2.0 * conv[offset - np.arange(n)]
    return grad

# ----------------------------------------------------------------------
# Soft‑max based optimisation (continuous phase)
# ----------------------------------------------------------------------
def _smooth_opt(a0: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """
    Runs a soft‑max approximation of the peak with an Adam optimiser.
    The learning‑rate schedule and the soft‑max temperature γ are
    annealed so that the optimisation focuses increasingly on the true
    maximum.
    """
    a = a0.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0

    # schedule parameters
    max_gamma = 200.0          # final temperature
    init_lr = 0.08
    min_lr = 0.0005

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- convolution of a with itself (full) ----
        b = _conv_full(a, a)                 # length 2·n‑1
        maxb = b.max()
        cur_obj = 2.0 * n * maxb
        if cur_obj < best_val - 1e-12:
            best_val, best_a = cur_obj, a.copy()
            print(f"[Smooth  ] new best = {best_val:.6f}")

        # ---- temperature (γ) annealing ----
        progress = (now - start_time) / (deadline - start_time)
        gamma = 1.0 + progress * (max_gamma - 1.0)

        # ---- soft‑max weights (stable version) ----
        # subtract the max to avoid overflow
        shift = b.max()
        exp_terms = np.exp(gamma * (b - shift))
        w = exp_terms / exp_terms.sum()      # Σ w = 1

        # ---- gradient of L_γ ----
        g = _softmax_grad(w, a)

        # ---- Adam update (gradient descent) ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate decay
        lr = init_lr * (1.0 - progress) + min_lr * progress

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ---- keep a a probability vector ----
        a = _project_simplex(a)

    return best_a, best_val

# ----------------------------------------------------------------------
# Deterministic greedy hill‑climb (fine‑graining)
# ----------------------------------------------------------------------
def _hill_climb(a_start: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Moves a small fraction of mass from the index that contributes most
    to the current peak to the index that contributes least.
    """
    a = a_start.copy()
    no_improve = 0
    IMPATIENT = 250_000
    max_frac_start, max_frac_end = 0.4, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _conv_full(a, a)
        maxb = b.max()
        cur_obj = 2.0 * n * maxb

        if cur_obj < best_val - 1e-12:
            best_val, best_a = cur_obj, a.copy()
            no_improve = 0
            print(f"[HillCl  ] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # anneal transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # peak index
        k = int(np.argmax(b))

        # contributions to that peak (a_{k‑i})
        contrib = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib[idx] = a[k - idx]          # a_{k‑i}
        # source = biggest contributor, target = smallest (different)
        src = int(np.argmax(contrib))
        tmp = contrib.copy()
        tmp[src] = np.inf
        tgt = int(np.argmin(tmp))

        if a[src] <= 1e-15:
            if no_improve > IMPATIENT:
                # random jitter to escape stagnation
                noise = np.random.normal(scale=0.005, size=n)
                a = np.maximum(a + noise, 0.0)
                a = _project_simplex(a)
                no_improve = 0
                print("[HillCl  ] jitter")
            continue

        delta = a[src] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        a[src] -= delta
        a[tgt] += delta

        # evaluate the tentative move
        b_new = _conv_full(a, a)
        new_obj = 2.0 * n * b_new.max()
        if new_obj < cur_obj - 1e-14:
            # accept
            cur_obj = new_obj
        else:
            # revert
            a[src] += delta
            a[tgt] -= delta

    return best_a, best_val

# ----------------------------------------------------------------------
# Tiny uniform mixing (often helps the final peak)
# ----------------------------------------------------------------------
def _mix_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, 2.0 * n * _conv_full(a, a).max()
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_simplex(cand)
        val = 2.0 * n * _conv_full(cand, cand).max()
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
