"""
propose_candidate.py

A multi‑stage optimizer for the step‑function evaluation problem.
The pipeline:

0️⃣  Initialise a diverse pool of candidates (uniform, random
    exponentials, power‑law, sinusoidal, and mixtures).  Each candidate
    is put into the “anti‑symmetric” order (largest coefficients on the
    left, smallest on the right) which provably minimises the centre
    convolution peak.

1️⃣  Sub‑gradient descent on the true L∞‑objective.  The update is a
    mirror‑descent step with an exponentially weighted multiplicative
    rule.  After every few thousand iterations the vector is re‑ordered
    anti‑symmetrically to keep the active set well balanced.

2️⃣  Smooth surrogate optimisation.  The max‑convolution is approximated
    by a log‑sum‑exp (LSE) smoothing; we minimise the LSE using L‑BFGS‑B
    on the soft‑max parametrisation.  A geometric ε‑schedule from 5e‑3
    down to 1e‑12 is employed, with several random restarts and a short
    post‑L‑BFGS sub‑gradient refinement.

3️⃣  Final simulated‑annealing hill‑climbing.  Random swaps and
    mass‑transfer moves are tried; occasional uphill moves are accepted
    according to a cooling schedule.  After each move the vector is
    projected back onto the probability simplex.

All stages respect the overall time budget (default 1000 s) and keep
track of the best sequence seen so far.  The returned list of floats
contains non‑negative numbers whose sum is 1, which is the domain
required by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)        # kill tiny negatives
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Return the anti‑symmetric ordering of ``a``:
    largest values on the left, smallest on the right.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]       # large → left side
        res[n - 1 - i] = s[i]       # small → right side
    if n % 2:
        res[half] = s[half]
    return res


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)     (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # self‑convolution
    conv = _full_convolution(a)
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t a
    weights = np.exp(shifted)
    p = weights / weights.sum()               # soft‑max over convolution indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries give derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]              # ∂L/∂a_i

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} a_{k‑i} using a single FFT.
    ``active_idx`` contains convolution indices (0 … 2n‑2) belonging
    to the active set.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent_multi(a_start: np.ndarray, deadline: float,
                               reorder_interval: int = 1000):
    """
    Mirror‑descent on the true L∞‑objective.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.8               # initial multiplicative step‑size factor
    rel_tol = 0.07             # start with fairly tight active‑set tolerance

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten tolerance gradually
        if it > 0 and it % 30000 == 0:
            rel_tol = max(rel_tol * 0.6, 1e-5)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]   # near‑max peaks

        # sub‑gradient
        g = _active_set_gradient(a, active)

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative update
        a = a * np.exp(-alpha * g)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # periodic anti‑symmetric re‑ordering (helps keep the peak low)
        if it % reorder_interval == 0:
            a = _anti_symmetric_reorder(a)

        # logging
        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")
        it += 1

    return best_a, best_val

# Simulated‑annealing based hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    """
    Random local moves (swap or mass transfer) with an annealing schedule.
    Accepts improving moves always; uphill moves are accepted with a
    temperature that decays exponentially.
    """
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.015                     # initial temperature
    it = 0

    while time.time() < deadline:
        it += 1
        # propose a move
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                        # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.07 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        # optional anti‑symmetric re‑ordering (keeps the search in a good subspace)
        if rng.random() < 0.05:
            new_a = _anti_symmetric_reorder(new_a)

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            accept = True
        else:
            T = T0 * (0.9993 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val_local:.6f}")

    return best_a, best_val_local

def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """
    Symmetric power‑law shape: a_i ∝ (dist_to_end+1)^(-p).
    """
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


def _exponential_candidate(lam: float, n: int) -> np.ndarray:
    """
    One‑sided exponential decay, then symmetrised.
    """
    x = np.arange(n)
    a = np.exp(-lam * x)
    a += a[::-1]          # make it symmetric
    a /= a.sum()
    return a


def _sinusoid_candidate(n: int) -> np.ndarray:
    """
    Mildly peaked sinusoid (always positive) and normalised.
    """
    x = np.arange(n)
    a = np.sin(np.pi * (x + 0.5) / n) + 0.1   # shift to avoid zeros
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

        # Perturb current best (soft‑max parameters) for a fresh start
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.35, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[restart {restarts}] L‑BFGS improved to {best_val:.6f}")

        # one more short sub‑gradient refinement (few thousand iters)
        if time.time() < deadline - 180:
            a_ref, val_ref = _subgradient_descent_multi(best_a, deadline, reorder_interval=1200)
            if val_ref < best_val:
                best_val, best_a = val_ref, a_ref.copy()
                print(f"[restart {restarts}] post‑L‑BFGS sub‑grad improved to {best_val:.6f}")

        # anti‑symmetric sweep
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[restart {restarts}] reorder improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_simulated_anneal(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list of floats
    return best_a.tolist()
