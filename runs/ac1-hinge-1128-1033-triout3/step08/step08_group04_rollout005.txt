def _adam_optimize(a0, deadline, max_iters=100000):
    n = len(a0)
    a = a0.copy()
    w = np.log(a + 1e-12) # log probabilities
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    best_a = a.copy()
    best_val = _fast_evaluate(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        # compute a from w
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        # conv and max index
        conv = _full_convolution(a)
        max_val = conv.max()
        if max_val * 2 * n < best_val - 1e-12: # improvement
            best_val = float(2.0 * n * max_val)
            best_a = a.copy()
            print(f"[adam] iter {t} new best = {best_val:.6f}")
        # gradient wrt a
        m_idx = int(conv.argmax())
        # compute gradient vector g_a
        g_a = np.zeros_like(a)
        i_start = max(0, m_idx - (n - 1))
        i_end = min(n - 1, m_idx)
        idx = np.arange(i_start, i_end + 1)
        g_a[idx] = a[m_idx - idx]   # a_{m-i}
        # optionally multiply by 2 for derivative exactness but can be baked into lr.
        # convert to gradient wrt w
        dot = np.dot(g_a, a)
        grad_w = a * (g_a - dot)

        # Adam updates
        t += 1
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)
        # optional: re-normalize w? Not needed.
    return best_a, best_val
```

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # generate candidates
    init_candidates = []
    # uniform
    init_candidates.append(np.full(n, 1.0/n))
    if has_old:
        init_candidates.append(old / old.sum())
    # random Dirichlet (alpha = 1)
    init_candidates.append(rng.gamma(1.0, 1.0, size=n))
    init_candidates.append(rng.gamma(0.3, 1.0, size=n))

    for lam in np.logspace(-4, -0.5, 5):

        a /= a.sum()
        init_candidates.append(a)
        a_sym = np.exp(-lam*np.abs(np.arange(n) - (n-1)/2))
        a_sym /= a_sym.sum()
        init_candidates.append(a_sym)

"""
Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Initial seeding** – a large, diverse pool of candidates
   (uniform, previous best, Dirichlet draws, exponential /
   symmetric‑exponential, Gaussian, Beta‑shaped families).  The cheapest
   evaluator ``_fast_evaluate`` (exact 2·n·max(conv) for a normalized
   vector) selects the best start.

2. **Exact Adam on the true max‑convolution** – the objective is
   ``F(a)=2·n·max_k (a*a)[k]``.  For the current maximal lag *m* the
   sub‑gradient w.r.t. ``a`` is
   ``g_i = a_{m-i}`` (zero when the index is out of range).  We optimise
   the *soft‑max* parameters ``w_i`` with Adam:
   ``a_i = exp(w_i)/∑_j exp(w_j)``.
   The gradient w.r.t. ``w`` follows from the chain rule:
   ``∇_w F = a * (g - ⟨g,a⟩)``.
   Several short Adam runs (restarts with random initialisations) are
   performed until the time budget expires.

3. **Local random moves (simulated‑annealing style)** – occasional
   single‑index mass transfers and short block swaps are accepted
   according to a geometric cooling schedule.  This helps to escape any
   remaining shallow plateaux.

4. **Final cheap hill‑climbing** – random swaps / mass‑transfer moves
   that accept only strict improvements.

All phases respect the 1000 s budget.  The best vector is finally
checked with the official ``evaluate_sequence`` and returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    The result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # tiny negative rounding noise → 0
    np.maximum(conv, 0, out=conv)
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the O(n) algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors already
    normalised to sum = 1.  Returns 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places the largest values to the left side and the
    smallest to the right side, often flattening the peak of the
    convolution.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[n - 1 - i]          # large → left
        out[n - 1 - i] = s[i]          # small → right
    if n % 2:
        out[half] = s[half]
    return out


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector stays on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def _adam_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the exact objective 2·n·max(conv) using the
    sub‑gradient of the max‑convolution.  The parameters are re‑parametrised
    with soft‑max variables w_i (unconstrained).  Returns the best vector
    seen and its objective value.
    """
    n = a0.shape[0]
    # initialise soft‑max parameters
    w = np.log(a0 + 1e-12)

    m = np.zeros_like(w)          # first moment
    v = np.zeros_like(w)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.08                    # initial learning rate
    best_a = None
    best_val = float('inf')
    t = 0

    while time.time() < deadline:
        # ----- obtain probability vector from w -----
        w_shift = w - np.max(w)          # for numerical stability
        a = np.exp(w_shift)
        a /= a.sum()

        # ----- evaluate objective -----
        conv = _full_convolution(a)
        max_val = conv.max()
        cur_val = float(2.0 * n * max_val)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter {t:05d} new best = {best_val:.6f}")

        # ----- sub‑gradient of max(conv) w.r.t. a -----
        m_idx = int(conv.argmax())
        g = np.zeros_like(a)                     # g_i = a_{m-i}
        i_start = max(0, m_idx - (n - 1))
        i_end = min(n - 1, m_idx)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = a[m_idx - idx]

        # ----- chain rule to obtain gradient w.r.t. w -----
        dot = np.dot(g, a)
        grad_w = a * (g - dot)

        # ----- Adam update on w -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # optional decay of the learning rate
        lr = lr0 * (0.99 ** (t // 2000))

        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        # sanity: if w explodes, re‑initialise it from current a
        if np.isnan(w).any() or np.isinf(w).any():
            w = np.log(a + 1e-12)
            m.fill(0.0)
            v.fill(0.0)

    # if Adam never improved (unlikely) fall back to the start point
    if best_a is None:
        best_a = a0.copy()
        best_val = _fast_evaluate(best_a)

    return best_a, best_val
