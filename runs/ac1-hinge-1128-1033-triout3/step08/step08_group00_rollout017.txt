import numpy as np
import time

def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = 2*n - 1
    size = 1 << (m - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:m]
    return conv

def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # All are zero or negative; return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns smooth max of b = conv(a,a) and gradient of 2*n*smooth_max wrt a.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()  # shape (2n-1)
    # gradient:   ∂/∂a (a * (p * a) mirrored)
    # Equivalent to convolution of a with reversed p
    p_rev = p[::-1]
    conv_ap = np.fft.rfft(a, 1 << ((2*n-1)-1).bit_length())
    # but easier: use np.convolve
    # We'll compute with np.fft for speed:
    size = 1 << ((a.shape[0] + p_rev.shape[0] - 1) - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[:2*n-1]
    # segment of length n centred:
    segment = conv[n-1:2*n-1]   # indices n-1..2n-2 inclusive
    grad = 4.0 * n * segment[::-1]  # include factor 2n
    smooth = max_b + eps * np.log(weights.sum())
    return smooth, grad

def _var_and_grad(a: np.ndarray):
    """
    Variance of b = conv(a,a) and its gradient (no scaling).
    """
    n = a.shape[0]
    b = _full_convolution(a)
    m = 2*n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v*v)
    # gradient: (4/m) * cross-correlation of a with v
    # conv(a, v_rev)
    v_rev = v[::-1]
    size = 1 << ((a.shape[0] + v_rev.shape[0] - 1) - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fv = np.fft.rfft(v_rev, size)
    conv = np.fft.irfft(fa * fv, size)[:a.shape[0] + v_rev.shape[0] - 1]
    segment = conv[n-1:2*n-1]
    grad = (4.0 / m) * segment[::-1]
    return var, grad

def _adam_optimize(a0, n, deadline, start_time, best_a, best_val,
                   target_val=1.503, max_idle=20000):
    # Logistic parameterization using log( a + eps )
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.25, 0.001
    eps_start, eps_end = 0.8, 1e-9
    lam_start = 10.0 * n  # stronger variance penalty
    lam_end = 0.0

    m = np.zeros_like(w)
    v_vec = np.zeros_like(w)
    t = 0
    idle = 0
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # map w -> a via softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        lam = lam_start * (1.0 - progress) + lam_end * progress

        # smooth max gradient
        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)

        # variance gradient
        var_val, grad_var = _var_and_grad(a)

        grad_a = grad_smooth + lam * grad_var

        # gradient wrt logits w
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturb
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.02, size=n)

        # restart if idle
        if idle > max_idle:
            print(f"[Adam {iteration:7d}] restart after {idle} idle")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v_vec.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val

def _deterministic_hill_climb(a0, n, deadline, start_time,
                             best_a, best_val, target_val=1.503):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.02, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate current maximal convolution index
        k_max = int(np.argmax(b))

        # compute contributions c_i = a_i * a_{k_max-i}
        c = np.zeros_like(a)
        if k_max < n:
            # i = 0..k_max
            c[:k_max+1] = a[:k_max+1] * a[k_max::-1]
        else:
            start = k_max - (n - 1)
            length = n - start
            c[start:] = a[start:] * a[:length][::-1]

        # source index with largest contribution (if zero, break)
        i = int(np.argmax(c))
        if c[i] <= eps:
            # No contribution; cannot reduce further.
            break

        # target indices with zero (or minimal) contribution
        zero_mask = c <= eps
        zero_idxs = np.where(zero_mask)[0]
        # ensure we have at least one target different from source
        if zero_idxs.size > 1:
            # choose random among zero indices excluding i if present
            if zero_idxs.size == 1:
                j = zero_idxs[0]
                if j == i:
                    # fallback to argmin
                    j = int(np.argmin(c + (np.arange(n) == i) * np.inf))
            else:
                # exclude source if present
                if i in zero_idxs:
                    zero_idxs = zero_idxs[zero_idxs != i]
                if zero_idxs.size == 0:
                    # fallback
                    j = int(np.argmin(c + (np.arange(n) == i) * np.inf))
                else:
                    j = int(np.random.choice(zero_idxs))
        else:
            # no zero contribution indices, pick minimal contribution
            j = int(np.argmin(c + (np.arange(n) == i) * np.inf))

        if i == j:
            # can't move to same; pick another minimal not i
            candidates = np.arange(n)
            candidates = candidates[candidates != i]
            j = int(np.random.choice(candidates))

        if a[i] <= eps:
            # source empty, skip
            continue

        # amount to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of b (copy for evaluation)
        b_new = b.copy()
        # update slices
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb a from best solution
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

def _beta_family_best(n):
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.2, 3.0, 15),
                             np.geomspace(3.0, 10.0, 8)[1:]])
    q_vals = p_vals.copy()
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        for q in q_vals:
            a = (idx ** (p - 1.0)) * ((n - idx + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val

def _gaussian_best(n):
    best_a = None
    best_val = np.inf
    center = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    sigma_vals = np.linspace(0.05 * n, 0.5 * n, 12)
    for sigma in sigma_vals:
        a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val

def _sin_shape_best(n):
    a = np.sin(np.pi * (np.arange(1, n+1) / (n+1)))
    s = a.sum()
    a /= s
    val = _objective(a)
    return a, val

def propose_candidate(seed: int = 42,

      2. Adam optimisation on a smooth approximation of the objective,

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # Problem size
    # ------------------------------------------------------------------
    n = 8000  # a compromise between flexibility and speed

    # ------------------------------------------------------------------
    # Initial solution (try to reuse earlier best if present)
    # ------------------------------------------------------------------

"""
Optimisation of a non‑negative step function for ``evaluate_sequence``.

The search proceeds in three stages:

1️⃣ **Initialisation** – a fast sweep over several parametric families
   (Beta, Gaussian, sinusoid) and a handful of Dirichlet draws.
   The best candidate from this cheap phase seeds the optimiser.

2️⃣ **Adam optimisation** – minimise a smooth surrogate of the true objective
   using a log‑softmax (simplex) parametrisation.  The loss is
   ``2·n·smooth_max(conv(a,a)) + λ·Var(conv(a,a))``; the variance term
   encourages a flat convolution.  Learning‑rate, smoothing and the variance
   weight are annealed throughout the allotted time.

3️⃣ **Deterministic hill‑climbing** – iteratively moves a small amount of
   mass from the index that contributes most to the current maximal
   convolution entry to an index with (almost) zero contribution.
   The move size and a Metropolis temperature are both annealed,
   and occasional perturbations escape stagnation.

The whole routine respects the 1000 s budget (0.5 s safety margin).  The
final sequence (a probability vector) is returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors x and y using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                       # length 2n‑1
    max_b = b.max()

    # soft‑max weights (log‑sum‑exp trick)
    shifted = (b - max_b) / eps
    w = np.exp(shifted)
    p = w / w.sum()                                 # shape (2n‑1,)

    # gradient via convolution of a with reversed p
    p_rev = p[::-1]
    conv = _conv_fft(a, p_rev)                     # length 3n‑2
    segment = conv[n - 1:2 * n - 1]                # n entries centred
    grad = 4.0 * n * segment[::-1]                 # contains the factor 2·n

    smooth = max_b + eps * np.log(w.sum())
    return smooth, grad


# ----------------------------------------------------------------------
# Variance of the convolution and its gradient
# ----------------------------------------------------------------------
def _var_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns  var(b) = mean((b‑avg)²)  and  ∂var/∂a.
    Here b = a*a and avg = 1/(2n‑1) because Σb = (Σa)² = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)          # length 2n‑1
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # gradient: (4/m) * cross‑correlation of a with v
    v_rev = v[::-1]
    conv = _conv_fft(a, v_rev)        # length 3n‑2
    segment = conv[n - 1:2 * n - 1]   # n entries
    grad = (4.0 / m) * segment[::-1]
    return var, grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth max + variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503,
                   max_idle: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises
        L(a) = 2·n·smooth_max(a) + λ·Var(conv(a,a)),
    where λ is linearly annealed from λ_start to 0.
    The simplex constraint Σa = 1 is handled via a log‑softmax
    parametrisation.
    """
    # logits for soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.25, 0.001
    eps_start, eps_end = 0.8, 1e-9
    lam_start = 10.0 * n       # stronger variance penalty
    lam_end = 0.0

    m = np.zeros_like(w)
    v_vec = np.zeros_like(w)
    t = 0
    idle = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---------- simplex ↔ logits ----------
        w_shift = w - np.max(w)           # numerical stability
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # ---------- schedules ----------
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        lam = lam_start * (1.0 - progress) + lam_end * progress

        # ---------- smooth‑max ----------
        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)

        # ---------- variance ----------
        var_val, grad_var = _var_and_grad(a)

        # total gradient of the *loss* L (we minimise L)
        grad_a = grad_smooth + lam * grad_var

        # chain rule through the soft‑max (gradient w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation to avoid lock‑in
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.02, size=n)

        # restart on stagnation
        if idle > max_idle:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v_vec.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (simulated‑annealing flavour)
# ----------------------------------------------------------------------
def _deterministic_hill_climb(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              best_a: np.ndarray,
                              best_val: float,
                              target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly moves a small amount of mass from the index
    that contributes most to the current maximal convolution entry to an index
    with (almost) zero contribution.  The transfer fraction and temperature are
    annealed to zero.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules
    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.02, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to this entry
        c = np.zeros_like(a)
        if k_max < n:
            # i = 0 … k_max
            c[: k_max + 1] = a[: k_max + 1] * a[k_max :: -1]
        else:
            # i = k_max-(n-1) … n-1
            start = k_max - (n - 1)
            length = n - start
            c[start:] = a[start:] * a[:length][::-1]

        # source index (largest contribution)
        i = int(np.argmax(c))
        if c[i] <= eps:                      # nothing contributes any more
            break

        # target index (prefer zero‑contribution locations)
        zero_mask = c <= eps
        zero_idxs = np.where(zero_mask)[0]
        if zero_idxs.size > 1:
            # exclude source if it also happens to be zero (unlikely)
            if i in zero_idxs:
                zero_idxs = zero_idxs[zero_idxs != i]
            if zero_idxs.size == 0:
                j = int(np.argmin(c + (np.arange(n) == i) * np.inf))
            else:
                j = int(np.random.choice(zero_idxs))
        else:
            # fall back to minimal contribution (excluding i)
            j = int(np.argmin(c + (np.arange(n) == i) * np.inf))

        if i == j:
            # pick any different index
            cand = np.arange(n)
            cand = cand[cand != i]
            j = int(np.random.choice(cand))

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – random perturbation around current best
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple parametric families (cheap sweeps)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a, best_val = None, np.inf
    p_vals = np.concatenate([np.linspace(0.2, 3.0, 15),
                             np.geomspace(3.0, 10.0, 8)[1:]])
    q_vals = p_vals.copy()
    idx = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (idx ** (p - 1.0)) * ((n - idx + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _gaussian_best(n: int) -> tuple[np.ndarray, float]:
    """Gaussian‑shaped candidates a_i ∝ exp(-(i‑c)²/(2σ²))."""
    best_a, best_val = None, np.inf
    c = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    sigma_vals = np.linspace(0.05 * n, 0.5 * n, 12)

    for sigma in sigma_vals:
        a = np.exp(-0.5 * ((i - c) / sigma) ** 2)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


def _sinusoid_best(n: int) -> tuple[np.ndarray, float]:
    """Single sinusoid shape a_i ∝ sin(π·(i+1)/(n+1))."""
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a /= a.sum()
    return a, _objective(a)

        best_val = _objective(best_a)
        print("[init] random start, value=", best_val)

    # --------------------------------------------------------------
    # Phase A – Adam optimisation (≈45 % of total budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # --------------------------------------------------------------
    # Phase B – Deterministic hill‑climbing (remaining budget)
    # --------------------------------------------------------------
    best_a, best_val = _deterministic_hill_climb(best_a, n,
                                                 deadline=deadline,
                                                 start_time=phase_a_deadline,
                                                 best_a=best_a,
                                                 best_val=best_val,
                                                 target_val=1.503)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
