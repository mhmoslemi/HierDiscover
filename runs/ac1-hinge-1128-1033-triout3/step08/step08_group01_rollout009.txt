    perturbations avoid stagnation.

"""
Proposes a non‑negative step function (vector) that minimises

    2·n·max(conv(a, a)) / sum(a)²

where conv is the full linear convolution.  The algorithm works in three
stages within the 1000 s time budget:

1. **Fast parametric sweep** – Randomly generate vectors from several
   families (Gaussian, generalized Gaussian, exponential, power‑law,
   sine, cosine bump, …).  Each candidate is normalised to sum = 1 and
   evaluated using an FFT‑based convolution.  The best candidate from
   the first ~25 % of the budget seeds the later stages.

2. **Targeted hill‑climbing** – Repeatedly move a small fraction of
   mass from an index that heavily contributes to the current maximal
   convolution entry to an index that contributes little.  The
   convolution is updated incrementally in O(n) per move, and only
   improving moves are kept.

3. **Projected sub‑gradient descent** – Optimises the exact objective
   by following a sub‑gradient of the maximal convolution entry and
   projecting back onto the probability simplex.  The step size decays
   slowly and occasional random perturbations escape stagnation.

A final tiny mixing with the uniform distribution is performed because
adding a little uniform mass often reduces the peak further.

All heavy work uses NumPy (FFT for convolutions).  The routine respects
the wall‑clock budget and returns a list of non‑negative floats (the
step heights).  The evaluation harness will call `evaluate_sequence`
on the returned list.
"""

import time
import math
import random
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return full linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def objective_normed(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum == 1)."""
    n = a.shape[0]
    max_b = full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {w ≥ 0, Σ w = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric shape generators
# ----------------------------------------------------------------------
def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    return np.exp(-((i - mid) ** 2) / (2.0 * sigma * sigma))


def shape_gen_gaussian(n: int, sigma: float, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    return np.exp(- (np.abs(i - mid) / sigma) ** p)


def shape_exponential(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return r ** i


def shape_sym_exponential(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return r ** i + r ** (n - 1 - i)


def shape_powerlaw(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def shape_sym_powerlaw(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p) + (n - i + 1) ** (-p)


def shape_half_sine(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    return np.sin(np.pi * i / (n + 1))


def shape_cosine_bump(n: int, width: float, power: float) -> np.ndarray:
    """Cosine bump centred in the middle, non‑zero on a interval of
    total length ≈ 2·width."""
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    arg = (i - mid) * np.pi / width
    cosv = np.cos(arg)
    cosv[np.abs(arg) > np.pi / 2] = 0.0
    return np.maximum(cosv, 0.0) ** power


# ----------------------------------------------------------------------
# Fast randomised parametric initialisation
# ----------------------------------------------------------------------
def parametric_initial_candidate(n: int, deadline: float) -> (np.ndarray, float):
    """Search a few thousand random shapes in the time window."""
    rng = np.random.default_rng(123456)
    best_a = np.full(n, 1.0 / n)
    best_val = objective_normed(best_a)

    while time.time() < deadline:
        choice = rng.integers(0, 9)                     # pick a family
        if choice == 0:                                 # Gaussian
            sigma = rng.uniform(0.05 * n, 0.60 * n)
            a = shape_gaussian(n, sigma)
        elif choice == 1:                               # Generalised Gaussian
            sigma = rng.uniform(0.05 * n, 0.60 * n)
            p = rng.uniform(0.5, 3.0)
            a = shape_gen_gaussian(n, sigma, p)
        elif choice == 2:                               # left exponential
            r = rng.uniform(0.9995, 0.999999)
            a = shape_exponential(n, r)
        elif choice == 3:                               # symmetric exponential
            r = rng.uniform(0.999, 0.99999)
            a = shape_sym_exponential(n, r)
        elif choice == 4:                               # power‑law
            p = rng.uniform(0.2, 4.5)
            a = shape_powerlaw(n, p)
        elif choice == 5:                               # symmetric power‑law
            p = rng.uniform(0.2, 4.5)
            a = shape_sym_powerlaw(n, p)
        elif choice == 6:                               # half‑sine
            a = shape_half_sine(n)
        elif choice == 7:                               # cosine bump
            width = rng.uniform(0.10 * n, 0.90 * n)
            power = rng.uniform(1.0, 4.0)
            a = shape_cosine_bump(n, width, power)
        else:                                           # random Dirichlet
            a = rng.dirichlet(np.ones(n))
        # normalise and evaluate
        if a.sum() == 0:
            continue
        a = a / a.sum()
        val = objective_normed(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[init] new best {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted hill‑climbing (exact objective, incremental convolution update)
# ----------------------------------------------------------------------
def hillclimb(a_start: np.ndarray, n: int, deadline: float) -> np.ndarray:
    a = a_start.copy()
    b = full_convolution(a)
    cur_val = 2.0 * n * b.max()
    start_hc = time.time()
    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start_hc) / (deadline - start_hc)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – proportional to contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – prefer positions that hardly affect the maximal entry
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0:
            probs_j = w / w.sum()
            j = np.random.choice(n, p=probs_j)
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > IMPROVE_LIMIT:
                # perturb and restart from the current best
                noise = np.random.normal(scale=0.012, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                b = full_convolution(a)
                cur_val = 2.0 * n * b.max()
                no_improve = 0
        it += 1
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray, n: int, deadline: float) -> np.ndarray:
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = 2.0 * n * full_convolution(a).max()
    eta0 = 0.5 / n                     # initial step size
    eta = eta0
    decay = 0.9995
    start_sg = time.time()
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    it = 0

    while time.time() < deadline:
        b = full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of the maximal entry (see derivation in the analysis)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # gradient step + projection onto the simplex
        v = a - eta * grad
        a = project_simplex(v)

        eta *= decay
        if no_improve > IMPROVE_LIMIT:
            # occasional shake‑up
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
        it += 1
    return best_local_a


# ----------------------------------------------------------------------
# Slight mixing with the uniform distribution (often beneficial)
# ----------------------------------------------------------------------
def mix_with_uniform(a: np.ndarray, n: int) -> (np.ndarray, float):
    best_a, best_val = a, objective_normed(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = objective_normed(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Optimises a step‑function within the given time budget.
    Returns a list of non‑negative floats that minimises `evaluate_sequence`.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # Problem size – 9 000 entries give a good trade‑off between quality and speed.
    n = 9000

    # ------------------------------------------------------------------
    # Phase 1 – rapid parametric sweep
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.25 * (deadline - start_time)
    best_a, best_val = parametric_initial_candidate(n, phase1_deadline)
    print(f"[phase 1] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – targeted hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.75 * (deadline - start_time)
    best_a = hillclimb(best_a, n, phase2_deadline)
    best_val = objective_normed(best_a)
    print(f"[phase 2] after hill‑climb = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (remaining time)
    # ------------------------------------------------------------------
    best_a = subgradient_descent(best_a, n, deadline)
    best_val = objective_normed(best_a)
    print(f"[phase 3] after sub‑gradient = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = mix_with_uniform(best_a, n)
    print(f"[final] polished objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation via the secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
