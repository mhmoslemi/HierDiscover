"""
Optimiser for the step‑function evaluation problem.

The approach adds a few new ingredients to the previous hybrid optimiser:

*   A cheap parametric sweep (exponential, power‑law and Beta‑type shapes)
    that often discovers a much better “seed” than a uniform vector.
*   A new deterministic re‑ordering – *interleaved large* – which spreads
    the biggest entries as far apart as possible (helps to lower the
    peak of the self–convolution).
*   More L‑BFGS restarts with a finer ε–schedule and a slightly larger
    iteration budget.
*   The remaining phases (Adam on the exact max‑convolution, a long
    simulated‑annealing, and a cheap final hill‑climb) are kept from the
    previous state‑of‑the‑art code.

All heavy work (FFT based convolution, gradient calculations) is O(n log n)
and the whole search respects the 1000 s wall‑time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders (required by the L‑BFGS surrogate)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None    # current smoothing parameter
_GLOBAL_N = None      # current problem size


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip numerical negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Exact copy of `evaluate_sequence` for a normalised vector."""
    n = a.shape[0]
    max_b = np.max(_fft_convolve_self(a))
    return float(2.0 * n * max_b)


def _softmax(w: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    w_shift = w - np.max(w)
    ex = np.exp(w_shift)
    return ex / np.sum(ex)


# ----------------------------------------------------------------------
# Smooth surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """Return (L, ∇_w L) for the log‑sum‑exp surrogate of max(conv)."""
    a = _softmax(w)                               # ℓ₁‑normalised non‑negative
    conv = _fft_convolve_self(a)
    max_c = conv.max()

    # smooth approximation of max
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)                  # Σ exp((c−max)/ε)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t a
    weights = np.exp(shifted)
    p = weights / np.sum(weights)                 # soft‑max over the conv entries
    p_rev = p[::-1]

    size = _next_pow2(conv.shape[0] + a.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[:a.shape[0] + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]               # central n entries
    grad_a = 4.0 * n * seg[::-1]                  # ∂L/∂a

    # chain‑rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper for `scipy.optimize.minimize` – uses globals."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Heuristics for re‑ordering
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Deterministic “ends‑large” ordering (big‑small‑big‑small)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left, right = 0, n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


def _interleave_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Spread the largest values as far apart as possible:
    place the biggest entry at position 0, the second biggest at n//2,
    the third at 1, the fourth at n//2+1, ….
    """
    n = a.shape[0]
    vals = np.sort(a)[::-1]                 # descending
    new_a = np.empty_like(a)

    left, right = 0, n // 2
    toggle = True
    for v in vals:
        if toggle:
            if left < n:
                new_a[left] = v
                left += 1
            else:
                new_a[right] = v
                right += 1
        else:
            if right < n:
                new_a[right] = v
                right += 1
            else:
                new_a[left] = v
                left += 1
        toggle = not toggle
    return new_a


# ----------------------------------------------------------------------
# Adam optimisation on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """Projected Adam on the true max‑convolution (sub‑gradient)."""
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)          # parameters for a soft‑max
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_a, best_val = a0.copy(), _eval_fast(a0)

    while time.time() < deadline:
        t += 1
        a = _softmax(w)

        conv = _fft_convolve_self(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active peak
        grad_a = np.zeros_like(a)
        idx = k - np.arange(n)
        ok = (idx >= 0) & (idx < n)
        grad_a[ok] = 2.0 * a[idx[ok]]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m̂ = m / (1 - beta1 ** t)
        v̂ = v / (1 - beta2 ** t)

        lr = 0.5 / (np.sqrt(t) + 1.0)
        w = w - lr * m̂ / (np.sqrt(v̂) + eps)

        a_new = _softmax(w)
        cur = _eval_fast(a_new)
        if cur < best_val:
            best_val, best_a = cur, a_new.copy()
            print(f"[Adam] iter {t:,}  new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (swap + tiny mass moves)
# ----------------------------------------------------------------------
def _simulated_annealing_mass(a: np.ndarray,
                              cur_val: float,
                              deadline: float) -> (np.ndarray, float):
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a, best_val = a.copy(), cur_val

    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99993
    it = 0

    while time.time() < deadline:
        if rng.random() < 0.3:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
        else:                                      # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-13:
                continue
            delta = a[i] * 0.03 * rng.random()
            a[i] -= delta
            a[j] += delta

        new_val = _eval_fast(a)
        d = new_val - cur_val
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[SA] iter {it:,}  new best = {best_val:.6f}")
        else:
            # revert
            if rng.random() < 0.3:     # we performed a swap
                a[i], a[j] = a[j], a[i]
            else:
                a[i] += delta
                a[j] -= delta

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap hill‑climbing
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    n = a.shape[0]
    cur_a, cur_val = a.copy(), cur_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.25:                     # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-13:
                continue
            delta = cur_a[i] * 0.02 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a, cur_val = new_a, val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Parameteric families – quick sweep for good seeds
# ----------------------------------------------------------------------
def _make_exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i   (r∈(0,1))."""
    a = r ** np.arange(n, dtype=float)
    a /= a.sum()
    return a


def _make_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))**p   (p may be negative)."""
    i = np.arange(n, dtype=float) + 1.0
    a = (i * (n - i + 1)) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _make_mixture(base: np.ndarray, shape: np.ndarray, lam: float) -> np.ndarray:
    """λ·base + (1‑λ)·shape   (re‑normalised)."""
    a = lam * base + (1.0 - lam) * shape
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Uses a short parametric sweep for seeding
    followed by the hybrid optimiser (smooth L‑BFGS → Adam → SA → hill‑climb).
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    # tiny safety margin for the final official evaluation
    deadline = start + budget_s - 1.5
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good compromise between quality and speed.
    # ------------------------------------------------------------------
    n = 4000
    global _GLOBAL_N
    _GLOBAL_N = n

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep – generate a variety of sensible seeds.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_val = _eval_fast(best_a)
    print(f"[seed] uniform evaluation = {best_val:.6f}")

    # exponential family
    for r in np.linspace(0.970, 0.9995, 15):
        cand = _make_exponential(n, r)
        val = _eval_fast(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[seed] exp r={r:.5f}  val={val:.6f}")

    # power‑law family (including ends‑large shapes)
    for p in np.linspace(-1.4, 0.4, 20):
        cand = _make_power(n, p)
        val = _eval_fast(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[seed] power p={p:.3f}  val={val:.6f}")

    # mixtures of the above with uniform
    for lam in np.linspace(0.0, 1.0, 6):
        cand = _make_mixture(uniform, best_a, lam)
        val = _eval_fast(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[seed] mix λ={lam:.2f}  val={val:.6f}")

    # a few random perturbations of the current best
    for _ in range(6):
        pert = best_a + 2e-4 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()
        val = _eval_fast(pert)
        if val < best_val:
            best_val, best_a = val, pert.copy()
            print(f"[seed] random pert  val={val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Phase 1 – smooth surrogate minimisation (L‑BFGS) with many restarts.
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-14), num=25)

    restart = 0
    while time.time() < deadline - 300.0:       # keep ~5 min for later phases
        restart += 1
        print(f"[phase1 restart {restart}]")

        # initialise soft‑max parameters from current best (+ noise)
        w = np.log(best_a + 1e-12) + 0.3 * np.random.randn(n)

        for eps in eps_schedule:
            if time.time() >= deadline - 250.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            # L‑BFGS on the smooth surrogate
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            a_candidate = _softmax(w)
            cur_val = _eval_fast(a_candidate)
            if cur_val < best_val:
                best_val, best_a = cur_val, a_candidate.copy()
                print(f"[phase1 eps={eps:.1e}] new best = {best_val:.6f}")

        # after each restart try two deterministic reorderings
        for reorder_fn, name in ((_ends_large_reorder, "ends‑large"),
                                 (_interleave_large_reorder, "interleave‑large")):
            a_ord = reorder_fn(best_a)
            a_ord /= a_ord.sum()
            val_ord = _eval_fast(a_ord)
            if val_ord < best_val:
                best_val, best_a = val_ord, a_ord.copy()
                print(f"[reorder {name}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Phase 2 – Adam projected gradient descent on the exact max‑conv.
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("[phase2] Adam gradient descent")
        best_a, best_val = _adam_opt(best_a, deadline - 140.0)

    # ------------------------------------------------------------------
    # 4️⃣  Phase 3 – long simulated annealing (swap + tiny mass moves).
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[phase3] Simulated annealing")
        best_a, best_val = _simulated_annealing_mass(best_a,
                                                     best_val,
                                                     deadline - 20.0)

    # ------------------------------------------------------------------
    # 5️⃣  Phase 4 – final cheap hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[phase4] Final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a,
                                              best_val,
                                              deadline)

    # ------------------------------------------------------------------
    # Official evaluation (secure version).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    # Return a plain Python list as required.
    return best_a.tolist()
