```
def smooth_gradient_descent(a_start, deadline):
    a = a_start.copy()
    n = len(a)
    best = a.copy()
    best_val = fast_objective(best)
    it = 0
    beta = 10.0
    beta_max = 200.0
    beta_inc = (beta_max - beta) / 20000  # adjust as needed
    alpha0 = 0.5
    momentum = np.zeros_like(a)
    while time.time() < deadline - 10:
        conv = _full_convolution(a)
        max_conv = conv.max()
        # weighted softmax
        # shift conv to avoid large exponents
        diff = conv - max_conv          # <=0
        # compute weights
        # clamp beta * diff to avoid underflow
        w = np.exp(beta * diff)        # max weight = 1
        Z = w.sum()
        # compute weighted grad
        grad = _gradient_weighted(a, w) / Z
        # momentum
        momentum = 0.9 * momentum + 0.1 * grad
        # step size schedule
        alpha = alpha0 / (1 + it * 1e-5)   # slowly decay
        # multiplicative update
        a = a * np.exp(-alpha * momentum)
        # renorm
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        # optionally increase beta
        if it % 500 == 0:
            beta = min(beta_max, beta + beta_inc)
        # evaluate
        if it % 200 == 0:
            cur_val = fast_objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[gd {it}] new best {best_val:.6f}")
            # occasional shuffle
            if it % 2000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = fast_objective(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] new best {best_val:.6f}")
        it += 1
    return best, best_val
```

The objective is  

\[
\text{score}(a)=\frac{2\,n\,\max\bigl(a*a\bigr)}{\bigl(\sum a_i\bigr)^2},
\]

where `a*a` is the linear (full) self‑convolution.  
Because the score is *scale invariant* we can safely keep the vector normalised
(`∑a_i = 1`).  Then  

\[
\text{score}=2\,n\;\max\bigl(a*a\bigr)
\]

and the whole problem reduces to **minimising the largest entry of the
convolution**.

The algorithm consists of three phases

1. **Initialisation** – start from the historic construction (`height_sequence_1`)
   when it is available; otherwise draw a random Dirichlet vector.
2. **Smooth‑max gradient descent** – minimise a smooth approximation of the
   maximum (log‑sum‑exp) using exponential mirror‑descent with momentum.
   The gradient of the smooth‑max is obtained with a *single* FFT, so each
   iteration is cheap even for `n≈8000`.
3. **Simulated‑annealing hill‑climbing** – a lightweight local search that
   accepts occasional worsening moves.  It works on the fast‑computed score
   (`2*n*max(conv)`) and therefore can perform many thousands of proposals
   within the remaining time budget.

All steps keep the vector non‑negative and renormalised, respect the 1000 s
budget and return the best sequence discovered.

```python
"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm
---------
1. Initialise from the historic construction if present, otherwise from a
   random Dirichlet vector (exponential samples normalised to sum 1).

2. Smooth‑max gradient descent:
   * Use a soft‑max (log‑sum‑exp) surrogate of the convolution maximum.
   * Its gradient is a weighted self‑convolution and can be computed with a
     single FFT (O(n log n)).
   * Perform exponential mirror‑descent with momentum, periodically
     increasing the soft‑max temperature β so that the surrogate gradually
     approaches the true max.
   * Keep the vector on the probability simplex (multiplicative update + renorm).

3. Simulated‑annealing hill climbing:
   * Propose random swaps or mass transfers (up to 20 % of the donor mass).
   * Accept a move if it improves the fast objective, otherwise with
     probability exp(‑Δ/ T) where T slowly cools.
   * The objective is evaluated cheaply as `2*n*max(conv)` (no call to the
     heavyweight `evaluate_sequence`).

4. Finally call `evaluate_sequence` once on the best vector and return it.

The whole procedure respects the 1000 s budget and prints intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may create tiny negatives – clip to zero
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _gradient_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Weighted gradient of the smooth‑max term.
    Returns 2· Σ_k w_k a_{k‑i} for i = 0 … n‑1.
    Computed via a single FFT.
    """
    n = a.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    size = _next_pow2(n + L - 1)       # length of a * reversed(w)

    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w[::-1], size)    # reverse w
    conv = np.fft.irfft(fa * fw, size)[: n + L - 1]

    mid = L - 1                        # index corresponding to shift i = 0
    idx = mid - np.arange(n)           # indices for i = 0 … n‑1
    g = 2.0 * conv[idx]                # factor 2 from the two symmetric terms
    return g


def _fast_objective(a: np.ndarray) -> float:
    """Fast evaluation of the score (identical to evaluate_sequence for sum=1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max gradient descent
# ----------------------------------------------------------------------
def _smooth_max_descent(a_start: np.ndarray, deadline: float):
    """
    Exponential mirror‑descent on a smooth approximation of the max.
    Returns the best vector and its fast objective.
    """
    a = a_start.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = _fast_objective(best)

    print(f"[gd start] fast score = {best_val:.6f}")

    # hyper‑parameters
    it = 0
    beta = 10.0                # initial soft‑max temperature
    beta_max = 200.0
    beta_step = (beta_max - beta) / 8000.0   # slow linear increase
    alpha0 = 0.5               # initial step size
    momentum = np.zeros_like(a)

    while time.time() < deadline - 5.0:
        # 1) full self‑convolution
        conv = _full_convolution(a)
        max_conv = conv.max()

        # 2) soft‑max weights (no overflow – diff ≤ 0)
        diff = conv - max_conv               # ≤ 0
        w = np.exp(beta * diff)              # max weight = 1
        Z = w.sum() + 1e-30                  # avoid division by zero

        # 3) weighted gradient (smooth‑max)
        grad = _gradient_weighted(a, w) / Z

        # 4) momentum (exponential moving average)
        momentum = 0.9 * momentum + 0.1 * grad

        # 5) step‑size schedule (very slow decay)
        alpha = alpha0 / (1.0 + it * 1e-5)

        # 6) multiplicative update on the simplex (mirror‑descent)
        delta = -alpha * momentum
        # clip exponent to avoid overflow / underflow
        delta = np.clip(delta, -20.0, 20.0)
        a = a * np.exp(delta)

        # 7) re‑project onto probability simplex
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # 8) bookkeeping
        if it % 200 == 0:
            cur_val = _fast_objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[gd iter {it}] new best = {best_val:.6f}")

            # occasional anti‑symmetric re‑ordering
            if it % 2000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_objective(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] improved to {best_val:.6f}")

        # 9) increase soft‑max temperature
        if it % 100 == 0 and beta < beta_max:
            beta = min(beta_max, beta + beta_step)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Phase 2 – simulated‑annealing hill climbing
# ----------------------------------------------------------------------
def _simulated_hill_climb(a_start: np.ndarray,
                          start_val: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Random local search that accepts worsening moves with a temperature.
    Works on the fast objective; only one call to evaluate_sequence is needed
    after the search finishes.
    """
    cur = a_start.copy()
    cur_val = start_val
    best = cur.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    n = cur.shape[0]

    # temperature schedule
    T = 0.02                # initial temperature
    T_decay = 0.99995

    while time.time() < deadline:
        # ----- propose a move -------------------------------------------------
        if rng.random() < 0.5:               # swap two positions
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:                                 # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] <= 0.0:
                continue
            # up to 20 % of donor mass
            frac = rng.random() * 0.20
            delta = cur[i] * frac
            new = cur.copy()
            new[i] -= delta
            new[j] += delta

        # ----- keep on simplex ------------------------------------------------
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s < 1e-12:
            continue
        new /= s

        # ----- evaluate (fast) ------------------------------------------------
        new_val = _fast_objective(new)
        d = new_val - cur_val

        # ----- Metropolis acceptance -----------------------------------------
        if d < 0.0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
                print(f"[sa] new best = {best_val:.6f}")

        # ----- temperature decay ---------------------------------------------
        T *= T_decay

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 coefficients give a good trade‑off
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from a known construction if possible
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # interpolate to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        # random Dirichlet (exponential) start
        a0 = rng.exponential(scale=1.0, size=n)

    # normalise to a probability vector (sum = 1)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s < 1e-12:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 /= s

    # ------------------------------------------------------------------
    # Initial evaluation (fast)
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_val = _fast_objective(best_a)
    print(f"[init] n={n}, fast score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – few random restarts to avoid a bad start
    # ------------------------------------------------------------------
    for tr in range(3):
        if time.time() >= deadline - 250:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = _fast_objective(a_rand)
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent (≈70 % of the budget)
    # ------------------------------------------------------------------
    gd_deadline = start_time + 0.68 * (deadline - start_time)
    if time.time() < gd_deadline:
        a_gd, val_gd = _smooth_max_descent(best_a, gd_deadline)
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[smooth‑gd] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing hill climbing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _simulated_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final evaluation with the official function
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:       # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = float('inf')

    return best_a.tolist()
```
