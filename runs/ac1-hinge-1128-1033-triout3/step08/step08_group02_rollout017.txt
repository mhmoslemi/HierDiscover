    cur_val = val_start
    best_local_a = cur.copy()
    best_local_val = cur_val

    # precompute uniform for mixing later maybe
    n = cur.shape[0]

    # set schedule parameters
    max_iter = 1000000 (but break if time)
    while time.time() < deadline:
        # compute convolution and index of maximum
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        # identify index k of max
        k = int(np.argmax(b))

        # compute rev contributions for this k: rev[i] = cur[k-i] (if fits)
        # Let's compute contributions for i where 0 <= k-i < n
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # rev indices array correspond to i from lo..hi maps to cur[k-i]
        rev = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        # source index = argmax rev (largest contribution)
        src = int(np.argmax(rev))
        # target index = argmin rev (but not src)
        rev_without_src = rev.copy()
        rev_without_src[src] = np.inf
        tgt = int(np.argmin(rev_without_src))

        if cur[src] <= 0:
            # random jitter if necessary
            # skip
            continue

        # propose move
        # fraction decays linearly with progress
        progress = (time.time() - start) / (deadline - start)
        max_frac = 0.5 * (1 - progress) + 0.001 * progress
        delta = cur[src] * max_frac * rng.random()
        if delta < 1e-15:
            continue

        # test move
        new_a = cur.copy()
        new_a[src] -= delta
        new_a[tgt] += delta

        # compute new objective quickly? Use full conv
        b_new = _full_convolution(new_a)
        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val * (1 - 1e-12): # accept improvement
            cur = new_a
            cur_val = new_val
            if new_val < best_local_val:
                best_local_val = new_val
                best_local_a = new_a.copy()
                print(f"[local] new best = {best_local_val:.6f}")
        else:
            # accept with probability based on temperature
            temperature = 0.001 * (1 - progress) + 1e-6 * progress
            if rng.random() < np.exp(-(new_val - cur_val)/temperature):
                cur = new_a
                cur_val = new_val
                # not update best

        # occasional random jitter to escape plateau
        if rng.random() < 0.001:

            noise = rng.normal(scale=0.01, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()

    return best_local_a, best_local_val

def _full_convolution(a):
    n = a.shape[0]
    out_len = 2*n - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

"""
Step‑function optimizer for `evaluate_sequence`.

The objective is
        f(a) = 2·n·max(conv(a,a)) / (∑a)² .
Because the function is scale‑invariant we keep ∑a = 1 and minimise
        2·n·max(conv(a,a)).

The algorithm consists of three phases (all respecting the 1000 s budget):

* **Phase‑0 – extensive parametric sweep** – a large collection of
  families (power‑law + offset, exponentials, symmetric/asymmetric
  beta‑densities, random Dirichlet draws) is sampled, each candidate is
  normalised and evaluated, and the best one is retained.

* **Phase‑A – simulated‑annealing hill‑climb** – starting from the best
  vector, repeatedly move a small fraction of the mass from the index
  that contributes most to the current maximal convolution entry to the
  index that contributes least.  The transferred fraction decays
  linearly, and occasional Gaussian jitter is added when progress stalls.
  Improvements are always kept; occasional uphill moves are allowed
  according to a simple temperature schedule.

* **Phase‑B – final uniform mixing** – try tiny convex combinations with
  the uniform vector; this often removes a residual peak.

All helper functions are defined at module level (no closures) and the
search is completely self‑contained.  The returned list is a plain list
of Python floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()            # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes ∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return the exact value returned by `evaluate_sequence`
    for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Normalisation / clipping helper
# ----------------------------------------------------------------------
def _prepare_vector(a: np.ndarray) -> np.ndarray:
    """
    Clip to the 0‑1000 range, enforce non‑negativity and normalise to sum‑1.
    The scaling is irrelevant for the objective because it is scale‑invariant.
    """
    a = np.maximum(a, 0.0)
    mx = a.max()
    if mx > 999.0 and mx > 0.0:
        a = a * (999.0 / mx)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Phase‑0 – parametric sweep
# ----------------------------------------------------------------------
def _param_search(best_a: np.ndarray,
                  best_val: float,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Exhaustively search a wide range of analytically‑defined families.
    Returns the best vector found and its objective value.
    """
    rng = np.random.default_rng(987654321)
    uniform = np.full(n, 1.0 / n)
    best_vec = best_a.copy()
    best_score = best_val

    def _try(a_raw: np.ndarray):
        nonlocal best_vec, best_score
        a = _prepare_vector(a_raw)
        val = _objective(a, n)
        if val < best_score:
            best_score, best_vec = val, a
            print(f"[param] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Power‑law with optional offset
    # ------------------------------------------------------------------
    for p in np.linspace(0.5, 4.0, 36):          # exponent
        for off in (0.0, 0.5, 1.0, 2.0, 5.0):   # offset to avoid singularity
            i = np.arange(n, dtype=np.float64)
            a = (i + 1.0 + off) ** (-p) + (n - i + off) ** (-p)
            _try(a)
            for lam in (0.001, 0.005, 0.01, 0.02):
                _try((1 - lam) * a + lam * uniform)
            if time.time() >= deadline:
                return best_vec, best_score

    # ------------------------------------------------------------------
    # 2) Symmetric exponentials r**i + r**(n‑1‑i)
    # ------------------------------------------------------------------
    for r in np.linspace(0.98, 0.9999, 40):
        i = np.arange(n, dtype=np.float64)
        a = r ** i + r ** (n - 1 - i)
        _try(a)
        for lam in (0.001, 0.005, 0.01, 0.02):
            _try((1 - lam) * a + lam * uniform)
        if time.time() >= deadline:
            return best_vec, best_score

    # ------------------------------------------------------------------
    # 3) Symmetric beta densities (U‑shapes)
    # ------------------------------------------------------------------
    for alpha in np.linspace(0.05, 0.95, 19):
        x = (np.arange(n, dtype=np.float64) + 0.5) / n      # mid‑points in (0,1)
        log_raw = (alpha - 1.0) * np.log(x) + (alpha - 1.0) * np.log(1.0 - x)
        log_raw -= log_raw.max()                           # numerical stabilisation
        a = np.exp(log_raw)
        _try(a)
        for lam in (0.001, 0.005, 0.01, 0.02):
            _try((1 - lam) * a + lam * uniform)
        if time.time() >= deadline:
            return best_vec, best_score

    # ------------------------------------------------------------------
    # 4) Asymmetric beta densities (α,β) sampled randomly
    # ------------------------------------------------------------------
    for _ in range(30):
        alpha = 0.05 + rng.random() * 1.5
        beta = 0.05 + rng.random() * 1.5
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        log_raw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
        log_raw -= log_raw.max()
        a = np.exp(log_raw)
        _try(a)
        for lam in (0.001, 0.005, 0.01):
            _try((1 - lam) * a + lam * uniform)
        if time.time() >= deadline:
            return best_vec, best_score

    # ------------------------------------------------------------------
    # 5) Random Dirichlet draws (different concentration parameters)
    # ------------------------------------------------------------------
    for conc in (0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            a = rng.dirichlet(np.full(n, conc))
            _try(a)
            if time.time() >= deadline:
                return best_vec, best_score

    return best_vec, best_score


# ----------------------------------------------------------------------
# Phase‑A – simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_val: float,
                n: int,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer hill‑climb.
    Moves a random fraction of the weight from the index that contributes
    most to the current peak to the index that contributes least.
    The fraction decays from 0.5 → 0.001; a tiny temperature allows
    occasional uphill steps.
    """
    cur = start_a.copy()
    cur_val = start_val
    best_a = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng(123456789)

    # counters for stagnation handling
    no_improve = 0
    IMPATIENT = 250_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # full convolution and current peak
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # the index where the peak occurs
        k = int(np.argmax(b))

        # contributions of each coefficient to that peak
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        rev = np.zeros(n, dtype=np.float64)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]                     # a_{k‑i}

        src = int(np.argmax(rev))                   # biggest contributor
        rev_tmp = rev.copy()
        rev_tmp[src] = np.inf
        tgt = int(np.argmin(rev_tmp))               # smallest contributor (≠ src)

        if cur[src] <= 1e-15:                       # source already depleted
            # occasional jitter if we get stuck
            if no_improve > IMPATIENT:
                noise = rng.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[hill] jitter after stagnation (depleted src)")
            continue

        # linear schedule for the transferred fraction
        progress = (now - start_time) / (deadline - start_time)
        max_frac = 0.5 * (1.0 - progress) + 0.001 * progress

        delta = cur[src] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        new_vec = cur.copy()
        new_vec[src] -= delta
        new_vec[tgt] += delta

        # evaluate new vector
        b_new = _full_convolution(new_vec)
        new_val = 2.0 * n * b_new.max()

        # simple annealing: accept improvements, otherwise accept
        # with a tiny probability that decays with the temperature.
        if new_val < cur_val - 1e-15:
            cur, cur_val = new_vec, new_val
        else:
            # temperature schedule (very low, just to escape flat plateaus)
            temperature = 0.001 * (1.0 - progress) + 1e-6 * progress
            if rng.random() < np.exp(-(new_val - cur_val) / temperature):
                cur, cur_val = new_vec, new_val

        # stagnation handling – global jitter
        if no_improve > IMPATIENT:
            noise = rng.normal(scale=0.006, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[hill] jitter after {IMPATIENT} non‑improving steps")

    return best_a, best_val

# Phase‑B – tiny uniform mixing
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combinations with the uniform vector."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):

        val = _objective(cand, n)
        if val < best_val:

    return best_a, best_val
