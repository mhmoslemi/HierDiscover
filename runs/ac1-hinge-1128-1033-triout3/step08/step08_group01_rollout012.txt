"""
propose_candidate
=================

Search for a non‑negative step‑function (vector) that minimises the
evaluation function

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Fast parametric sweep** – a small collection of hand‑crafted families
   (uniform, exponential, power‑law, sinusoidal, triangular, etc.) is
   instantiated for the chosen length ``n`` (default 2000).  The best
   candidate from this sweep is taken as the start point.

2. **Simulated‑annealing hill‑climbing** – the current vector ``a`` is
   repeatedly modified by moving a tiny fraction of mass from an index
   that contributes most to the current maximal convolution entry to an
   index that contributes little.  Moves that improve the objective are
   always accepted; worsening moves are accepted with probability
   ``exp(-(Δobj)/T)`` where the temperature ``T`` decays exponentially.
   The convolution is updated incrementally in O(n) time, allowing
   hundreds of thousands of iterations within the 1000 s budget.

3. **Uniform mixing** – a final sweep over tiny convex combinations with
   the uniform distribution often reduces the peak further.

The code is fully self‑contained, uses only NumPy and the supplied
``evaluate_sequence`` function, respects all security checks, and
returns a Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (sum of coefficients is assumed to be 1)
# ----------------------------------------------------------------------
def _objective_from_conv(b: np.ndarray, n: int) -> float:
    """Return 2·n·max(b) for a convolution array b (sum of a is 1)."""
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generate a diverse set of parametric candidates (all are normalised)
# ----------------------------------------------------------------------
def _initial_candidates(n: int):
    """Yield a sequence of (a, value) pairs for several simple families."""
    rng = np.random.default_rng(12345)

    # uniform
    a = np.full(n, 1.0 / n)
    yield a, _objective_from_conv(_full_convolution(a), n)

    # exponential decay from the left
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        yield a, _objective_from_conv(_full_convolution(a), n)

    # symmetric exponential (U‑shaped)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        a /= a.sum()
        yield a, _objective_from_conv(_full_convolution(a), n)

    # power‑law symmetric
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p) + (n - i + 1) ** (-p)
        a /= a.sum()
        yield a, _objective_from_conv(_full_convolution(a), n)

    # half‑wave sine (zero at the ends, peak in the middle)
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a /= a.sum()
    yield a, _objective_from_conv(_full_convolution(a), n)

    # triangular with zeros at the ends
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)          # 0 … n/2 … 0
    a = a.astype(float)
    a /= a.sum()
    yield a, _objective_from_conv(_full_convolution(a), n)

    # a few random Dirichlet draws
    for _ in range(8):
        a = rng.dirichlet(np.ones(n))
        yield a, _objective_from_conv(_full_convolution(a), n)


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing on the *exact* objective
# ----------------------------------------------------------------------
def _annealing_hillclimb(a_start: np.ndarray,
                         n: int,
                         start: float,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float) -> np.ndarray:
    """
    Move a small amount of probability mass from an index that contributes
    heavily to the current maximal convolution entry to an index that
    contributes little.  Accept improvements always, accept worsening
    moves with a temperature schedule.
    """
    a = a_start.copy()
    b = _full_convolution(a)                     # convolution array
    cur_val = _objective_from_conv(b, n)

    # annealing schedule parameters
    T0 = cur_val * 0.02          # initial temperature
    Tf = cur_val * 1e-4          # final temperature

    # move size schedule (fraction of source mass)
    max_frac_start, max_frac_end = 0.4, 0.001

    it = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) locate the lag that attains the current maximum
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))

        # --------------------------------------------------------------
        # 2) compute contributions of individual indices to that lag
        # --------------------------------------------------------------
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        source = int(np.argmax(contrib))          # index with largest contribution

        # --------------------------------------------------------------
        # 3) choose a destination index where the added mass would
        #    affect the maximal lag the least
        # --------------------------------------------------------------
        # weight = 1 / (a[k_max - j] + eps); large weight → small product
        rev_idx = k_max - np.arange(n)
        rev_valid = (rev_idx >= 0) & (rev_idx < n)
        weight = np.ones(n, dtype=float)
        # for valid j, look at the partner that would appear in the max lag
        weight[rev_valid] = 1.0 / (a[rev_idx[rev_valid]] + 1e-12)
        weight[source] = 0.0                         # never choose the source again
        if weight.sum() > 0:
            probs = weight / weight.sum()
            dest = int(np.random.choice(n, p=probs))
        else:
            dest = (source + np.random.randint(1, n)) % n

        # --------------------------------------------------------------
        # 4) decide how much mass to move
        # --------------------------------------------------------------
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        delta = a[source] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------
        # 5) incremental update of the convolution (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # remove contribution of the source mass
        b_new[source:source + n] -= 2.0 * delta * a
        # add contribution of the destination mass
        b_new[dest:dest + n] += 2.0 * delta * a
        # correct the diagonal terms and the cross term (source,dest)
        b_new[2 * source] += delta * delta
        b_new[2 * dest]   += delta * delta
        b_new[source + dest] -= 2.0 * delta * delta

        new_val = _objective_from_conv(b_new, n)

        # --------------------------------------------------------------
        # 6) Acceptance rule (simulated annealing)
        # --------------------------------------------------------------
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # temperature decays exponentially
            T = T0 * (Tf / T0) ** progress
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            # apply the move
            a[source] -= delta
            a[dest]   += delta
            b = b_new
            cur_val = new_val

            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")

        # occasional random perturbation to escape plateaus
        if it % 50000 == 0 and it > 0:
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective_from_conv(b, n)

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Small post‑processing: mixing a tiny amount of uniform mass
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective_from_conv(_full_convolution(a), n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_from_conv(_full_convolution(cand), n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,

    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Initialise a – try to down‑sample any previously found solution,
    # otherwise fall back to the parametric sweep.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        # use the best of the families defined above
        best_a = None
        best_val = np.inf
        for a, val in _initial_candidates(n):
            if val < best_val:
                best_val, best_a = val, a.copy()
        a0 = best_a

    best_a = a0.copy()
    best_val = _objective_from_conv(_full_convolution(best_a), n)
