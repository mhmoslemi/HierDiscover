"""
Optimise a non‑negative step function to minimise
    2·n·max(conv(a,a)) / (sum(a)**2)

We view the vector as a probability distribution (the objective is scale‑invariant)
and try to make the self‑convolution as flat as possible.
The search consists of

1.  generating a large, diverse set of analytical candidates (Beta,
    Gaussian, exponential, power‑law, random Dirichlet, etc.);
2.  selecting the best few by a *fast* FFT based evaluator;
3.  refining each selected candidate with a smooth surrogate optimisation
    (log‑sum‑exp smoothing of the max) via L‑BFGS‑B;
4.  a few rounds of exact mirror‑descent on the true max‑convolution;
5.  a simulated‑annealing / hill‑climbing phase with block moves;
6.  a final safety check with the official ``evaluate_sequence`` routine.

The whole routine respects the 1000 s budget and returns a plain Python
list of heights.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smoothed surrogate (they are set before
# each optimisation call).
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may produce tiny negative values
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed objective  L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    where a_i = softmax(w_i).  Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max parametrisation -----
    w_shift = w - np.max(w)          # for stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- convolution a*a -----
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max (log‑sum‑exp) -----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a (chain‑rule through soft‑max) -----
    weights = np.exp(shifted)
    p = weights / weights.sum()      # soft‑max over lag indices

    # convolution of a with reversed p (needed for ∂max/∂a)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # extract the part that aligns with the original a‑vector
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]     # match ordering of a_i

    # ----- back‑prop through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return float(L), grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper for ``scipy.minimize``."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast surrogate for ``evaluate_sequence`` for vectors that already satisfy
    a_i ≥ 0 and Σ a_i = 1.  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that puts large values on the left, small values on the right.
    This often reduces the height of the central peak of the convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass‑transfer moves (80 %).  The vector stays on
    the simplex after each move.
    """
    cur_a = a.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:            # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                             # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[hill‑climb] improved to {cur_val:.6f}")

    return cur_a, cur_val

                                    'disp': False})
            w = res.x

            # Convert back to a probability vector and compute the cheap metric.
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # try the anti‑symmetric re‑order heuristic after every restart
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact mirror‑descent on the true max‑convolution.
    # ------------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.35
    decay = 0.9999
    min_alpha = 0.002
    it = 0
    while time.time() < deadline - 20.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]               # each ordered pair contributes a[jdx]

        # exponential (mirror) update
        alpha = max(alpha0 * (decay ** (it // 500)), min_alpha)
        a = a * np.exp(-alpha * g)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        if it % 250 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[mirror {it}] new best = {best_val:.6f}")

            # occasional anti‑symmetric shuffle
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[mirror reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 5) Simulated‑annealing style random block moves.
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.03                         # initial temperature
    cooling = 0.99994
    rng = np.random.default_rng(seed + 9999)

    while time.time() < deadline - 5.0:
        if rng.random() < 0.6:
            # single‑index mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.12 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # swap two short blocks (length ≤ 40)
            L = rng.integers(1, min(40, n // 8))
            i = rng.integers(0, n - L)
            j = rng.integers(0, n - L)
            new_a = cur_a.copy()
            tmp = new_a[i:i + L].copy()
            new_a[i:i + L] = new_a[j:j + L]
            new_a[j:j + L] = tmp

        # re‑normalise onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)

        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # 6) Final cheap hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # 7) Final safety check with the official evaluator and return.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
