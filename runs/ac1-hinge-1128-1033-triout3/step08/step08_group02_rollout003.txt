"""
Propose a step‑function (non‑negative vector) that minimises

        f(a) = 2·n·max(conv(a,a)) / (sum a)^2

under the scale‑invariant constraint Σa = 1.
The core optimizer is a *soft‑max gradient descent*:

* at each iteration we compute the full self‑convolution c = a∗a,
* we replace the non‑smooth max by a smooth log‑sum‑exp (soft‑max)
  with a temperature β that slowly grows,
* the gradient of the soft‑max w.r.t. a is 2·(p∗a_rev) where
  p are the soft‑max probabilities,
* a multiplicative (exponentiated‑gradient) update preserves non‑negativity,
* after each update we project onto the probability simplex (simple renormalisation).

A cheap family‑sweep provides a good start, then the iterative
soft‑max descent runs for as long as the time budget permits.
Occasional jitteres help escape plateaus.  Finally a tiny uniform
mixing is tried; the best vector found is returned.

The method is completely self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` for the final check, and respects the
1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, b: np.ndarray | None = None) -> np.ndarray:
    """
    Full linear convolution of `a` with `b` (or with itself if `b` is None).
    Returns an array of length len(a)+len(b)-1.
    """
    if b is None:
        b = a
    n = a.shape[0]
    m = b.shape[0]
    out_len = n + m - 1
    # Next power‑of‑two size for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Simple parametric families – quick initial sweep
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)                # uniform
    best_val = _objective(best_a, n)

    def try_update(arr):
        nonlocal best_a, best_val
        a = np.asarray(arr, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0.0:
            return
        a = a / s
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(12345)

    idx = np.arange(n)

    # deterministic families
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # V‑shape linear
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # quadratic decreasing
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law V‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.995, 0.99, 0.985, 0.98, 0.97, 0.96):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # random Dirichlet draws (quick)
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex {x≥0, Σx=1}."""
    # Fast O(n log n) projection (see Wang & Carreira‑Perpiñán 2013)
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max based gradient descent (the heart of the optimizer)
# ----------------------------------------------------------------------
def _softmax_descent(
    a0: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
    target_val: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Perform a soft‑max gradient descent on the exact objective.
    The temperature β grows from β_start to β_end while the step size η
    decays from η_start to η_end.
    """
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 250_000  # after this many non‑improving steps we add jitter

    # schedule parameters
    beta_start, beta_end = 1.0, 120.0
    eta_start, eta_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        # --------------------------------------------------------------
        # evaluate current objective
        # --------------------------------------------------------------
        conv = _full_convolution(cur)               # length 2n‑1
        maxb = conv.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[Softmax {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # temperature and step‑size (linear interpolation w.r.t. time)
        # --------------------------------------------------------------
        progress = (now - start_time) / (deadline - start_time)
        beta = beta_start + (beta_end - beta_start) * progress
        eta = eta_start + (eta_end - eta_start) * progress

        # --------------------------------------------------------------
        # soft‑max probabilities (log‑sum‑exp trick for stability)
        # --------------------------------------------------------------
        maxc = conv.max()                 # already computed
        # subtract maxc to avoid overflow
        exp_terms = np.exp(beta * (conv - maxc))
        p = exp_terms / exp_terms.sum()   # shape (2n‑1,)

        # --------------------------------------------------------------
        # gradient of the soft‑max (∇S = 2·(p∗a_rev))
        # --------------------------------------------------------------
        a_rev = cur[::-1]
        # full convolution of p (len 2n‑1) with a_rev (len n) → length 3n‑2
        grad_full = _full_convolution(p, a_rev)   # length 3n‑2
        # extract entries g_i = 2 * grad_full[n‑1‑i] for i=0..n‑1
        g = 2.0 * grad_full[(n - 1) - np.arange(n)]

        # --------------------------------------------------------------
        # multiplicative update + simplex projection
        # --------------------------------------------------------------
        # Protect against overly large updates
        g = np.clip(g, -1e6, 1e6)

        cur = cur * np.exp(-eta * g)
        # Numerical safety: keep a small floor before projection
        cur = np.maximum(cur, 1e-12)
        cur /= cur.sum()

        # --------------------------------------------------------------
        # occasional jitter if stagnating
        # --------------------------------------------------------------
        if no_improve > IMPATIENT:
            rng = np.random.default_rng()
            noise = rng.normal(scale=0.006, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[Softmax {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Final tiny uniform mixing (often beneficial)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a′ = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

    best_a, best_val = _greedy_hillclimb(
        best_a,
        n,
        start=softmax_deadline,
        deadline=hill_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
    )

    # --------------------------------------------------------------
    # Phase‑C – final tiny uniform mixing
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # --------------------------------------------------------------
    # Final verification via the provided secure wrapper
    # --------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
