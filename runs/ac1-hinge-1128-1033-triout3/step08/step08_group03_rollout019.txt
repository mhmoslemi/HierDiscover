"""
Propose a non‑negative step function that (empirically) beats the previous
best bound (≈ 1.51035).  The algorithm consists of three phases:

1. **Warm‑start** – a Dirichlet‑sampled vector (or the historic
   `height_sequence_1`) is normalised to sum = 1.

2. **Soft‑max exponential‑gradient descent** – instead of using only the
   peaks that are within a tiny tolerance of the current maximum, a smooth
   approximation of the max is employed:
       w_k ∝ exp(β·(c_k – max(c)))      (c = a*a)
   The gradient of the weighted sum  Σ w_k·c_k  can be written in closed form
   and computed with two FFT‑based convolutions (O(n log n)).  The update
   is multiplicative (mirrored gradient descent) and the point stays on the
   simplex.  β is slowly increased so the optimisation starts with a
   global view and ends focusing on the true maximum.

3. **Fine‑grained hill‑climbing** – after the gradient phase a lightweight
   local search (random swaps + tiny mass transfers) is run until the time
   budget expires.

All heavy computation (FFT‑based gradient) is performed without calling
`evaluate_sequence`; the true objective is queried only periodically for
logging and to keep the best solution found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for efficient FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len‑1) using an FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may create tiny negatives – clamp them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Descending order (largest ↔ smallest) – the classic anti‑symmetric
    arrangement that minimises Σ a_i·a_{n‑1‑i}.
    """
    return np.sort(a)[::-1]          # simply descending


# ----------------------------------------------------------------------
# Gradient of a soft‑max weighted sum of convolution entries
# ----------------------------------------------------------------------
def _gradient_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k·c_k   where  c = a * a  (full convolution) and
    w is a non‑negative weight vector of length 2·n‑1.
    The formula is
        g_i = Σ_j a_j·w_{i‑j}           + Σ_j a_j·w_{i+j}
    (indices outside the range give zero).  Both terms are obtained by a
    single FFT for each of the two convolutions a·w and a·w_rev.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    size = _next_pow2(3 * n - 2)       # length needed for a * w (full)

    fa = np.fft.rfft(a, size)

    # first term: Σ_j a_j·w_{i‑j}   (standard convolution)
    fw = np.fft.rfft(w, size)
    conv1 = np.fft.irfft(fa * fw, size)[: 3 * n - 2]   # length 3n‑2

    # second term: Σ_j a_j·w_{i+j}  (convolution with reversed w)
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, size)
    conv2 = np.fft.irfft(fa * fw_rev, size)[: 3 * n - 2]

    L = m - 1                         # = 2·n‑2
    # g_i = conv1[i] + conv2[L‑i]
    idx = L - np.arange(n, dtype=int)
    g = conv1[:n] + conv2[idx]
    return g


# ----------------------------------------------------------------------
# Soft‑max exponential‑gradient descent
# ----------------------------------------------------------------------
def _softmax_exp_gradient(a_start: np.ndarray, deadline: float):
    """
    Mirror‑descent on the smoothed max of the self‑convolution.
    The temperature β grows slowly, turning the smoothed objective into the
    true max.  The step size α is decayed gently.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[sgd start] value = {best_val:.6f}")

    it = 0
    alpha = 0.5               # initial multiplicative step size
    beta = 0.5                # initial soft‑max sharpness
    # schedules
    alpha_decay = 0.92        # after each 2000 iterations
    beta_growth = 1.015       # after each 500 iterations

    while time.time() < deadline - 5.0:
        # full self‑convolution
        conv = _full_convolution(a)
        max_conv = conv.max()

        # soft‑max weights (stabilised around the current maximum)
        # w_k ∝ exp(β·(conv_k – max_conv)); they sum to 1.
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:                # numerical safety – fall back to uniform
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient of Σ w_k·conv_k
        g = _gradient_weighted(a, w)

        # multiplicative mirror‑descent update
        delta = -alpha * g
        delta = np.clip(delta, -20.0, 20.0)   # keep exp(·) numerically safe
        a = a * np.exp(delta)

        # project back onto the probability simplex (positivity + normalisation)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # periodic logging and schedule updates
        if it % 250 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max conv {max_conv:.8e})")
            # increase sharpness of the soft‑max and decay the step size
            beta *= beta_growth
            if it % 2000 == 0 and it > 0:
                alpha *= alpha_decay

        # occasional re‑ordering to keep the pairing term small
        if it % 1200 == 0:
            a = _anti_symmetric_reorder(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (re‑used from the previous solution)
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """Random swaps + tiny mass transfers, keeping only strict improvements."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.25:                     # pure swap (25 % probability)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer (75 %)
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.07 * rng.random()   # slightly larger than before
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:          # accept only strict improvements
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
