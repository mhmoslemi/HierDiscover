def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using FFT of power‑of‑2 size.
    Result length = 2·len(a)−1 and all tiny negatives are clipped to 0.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)   # eliminate round‑off negatives
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)  (a ∈ simplex)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k−max)/ε)

    n = _GLOBAL_N

    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)

    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient of smoothed max w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()      # distribution over lag indices

    # derivative of each conv_k w.r.t. a is the a‑reversed convolution with p
