def _hillclimb_sa(a_start, n, start, deadline, best_a, best_val, target_val=1.503):
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_val_local = best_val
    best_a_local = best_a.copy()
    # schedule parameters
    max_frac_start, max_frac_end = 0.5, 0.001
    temp_start, temp_end = 1e-3, 1e-6
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # compute source and target
        k_max = int(np.argmax(b))
        # compute contributions
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[idx[mask]]
        # find source i
        i = int(np.argmax(contrib))
        # find target j with minimal contribution (non-i)
        # we can also aim for low a[k_max - j]
        # compute contributions for target similarly
        # we could use j = int(np.argmin(contrib + 1e-12))
        # but ensure j != i
        j = int(np.argmin(contrib + (np.arange(n)==i)*np.max(contrib)))
        # delta
        delta = a[i] * max_frac * rng.random()
        if delta <= 0:
            continue

        # incremental update
        b_new = b.copy()
        # update intervals safely (ensuring index within bounds)
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        idx2 = 2*j
        b_new[idx2] += delta * delta
        idx3 = 2*i
        b_new[idx3] += delta * delta
        idx4 = i + j
        b_new[idx4] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept?
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probability exp(-Δ/T)
            if rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val_local:
                best_val_local = cur_val
                best_a_local = a.copy()
                print(f"[SA {it:6d}] new best = {best_val_local:.6f}")
                if best_val_local <= target_val:

            b = _full_convolution(a)

            no_improve = 0
            print(f"[SA {it:6d}] perturb after stagnation")

"""
Propose a non‑negative step‑function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works inside the given time budget (default 1000 s)
and consists of four stages:

1. **Fast parametric sweep** – generate many classic window shapes
   (exponential, power‑law, Gaussian, beta, Hann/Hamming/Blackman,
   sine, triangular, etc.) together with a few random Dirichlet draws.
   The best candidate (lowest objective) is kept.

2. **Adam optimisation** – optimise the smooth log‑sum‑exp surrogate
   of the max‑convolution while keeping the vector on the probability
   simplex via a soft‑max parametrisation.

3. **Simulated‑annealing hill‑climbing** – repeatedly move a small
   amount of mass from the index that contributes most to the current
   maximal convolution entry to the index that contributes the least.
   Down‑hill moves are always accepted; uphill moves are accepted with
   a temperature‑controlled probability.  Periodic perturbations help
   escape plateaus.

4. **Uniform mixing** – a tiny convex combination with the uniform
   distribution often flattens the convolution peak further.

The routine returns a list of non‑negative floats (≈1 in ℓ₁‑norm)
that can be fed directly into ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes nothing about the sum of a)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/(sum(a)**2)."""
    n = a.shape[0]
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    b = _full_convolution(a)
    return 2.0 * n * b.max() / (s * s)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {w ≥ 0, Σ w = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Phase 0 – fast sweep over parametric families
# ----------------------------------------------------------------------
def _parametric_sweep(n, rng, deadline):
    """Return a promising vector (Σ≈1) and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)

    def _evaluate(candidate):
        nonlocal best_a, best_val
        a = np.asarray(candidate, dtype=float)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            return
        a = a / s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[ sweep ] new best = {best_val:.6f}")

    # Uniform (already covered)
    _evaluate(best_a)

    # Exponential (single‑sided)
    for r in np.linspace(0.9995, 0.999999, 8):
        if time.time() > deadline:
            break
        _evaluate(_shape_exponential(n, r))

    # Symmetric exponential (U‑shaped)
    for r in np.linspace(0.9995, 0.999999, 6):
        if time.time() > deadline:
            break
        _evaluate(_shape_sym_exponential(n, r))

    # Power‑law (single‑sided)
    for p in np.linspace(0.3, 4.0, 8):
        if time.time() > deadline:
            break
        _evaluate(_shape_powerlaw(n, p))

    # Symmetric power‑law
    for p in np.linspace(0.3, 4.0, 6):
        if time.time() > deadline:
            break
        _evaluate(_shape_sym_powerlaw(n, p))

    # Gaussian windows (different sigmas)
    for factor in np.linspace(0.05, 0.5, 8):
        if time.time() > deadline:
            break
        sigma = factor * n
        _evaluate(_shape_gaussian(n, sigma))

    # Symmetric beta distributions (α=β)
    for a in np.linspace(0.2, 5.0, 7):
        if time.time() > deadline:
            break
        _evaluate(_shape_beta(n, a, a))

    # Classical signal‑processing windows
    for fn in (_shape_hann, _shape_hamming, _shape_blackman):
        if time.time() > deadline:
            break
        _evaluate(fn(n))

    # Half‑wave sine
    _evaluate(_shape_sine_half(n))

    # Triangular
    _evaluate(_shape_triangular(n))

    # Random Dirichlet draws (quick exploration)
    for _ in range(30):
        if time.time() > deadline:
            break
        a = rng.dirichlet(np.ones(n))
        _evaluate(a)

    return best_a, best_val

# ----------------------------------------------------------------------
def _adam_phase(a0, n, start, deadline, best_a, best_val, target_val=1.503):
    """Adam optimiser on a smoothed max‑convolution objective."""
    w = np.log(a0 + 1e-12)  # logits → simplex via soft‑max

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.004
    eps_start, eps_end = 0.1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 100_000

    it = 0
    while time.time() < deadline:
        # map logits → simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedules
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()                # soft‑max over positions

        # gradient w.r.t. a (see derivation in the description)
        p_rev = p[::-1]
        conv_ab = np.fft.irfft(np.fft.rfft(a, 1 << (2 * n - 2).bit_length()) *
                               np.fft.rfft(p_rev, 1 << (2 * n - 2).bit_length()))
        # extract the segment that corresponds to the convolution a * p_rev
        start_idx = n - 1
        segment = conv_ab[start_idx : start_idx + n][::-1]
        grad_a = 4.0 * n * segment

        # chain rule through the soft‑max parametrisation
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape flat regions
        if rng.random() < 0.005:
            w += rng.normal(scale=0.006, size=n)

        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart (no improvement)")
            w = rng.normal(scale=0.2, size=n)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_sa(a_start, n, start, deadline,
                  best_a, best_val, target_val=1.503):
    """Move mass from high‑contributing to low‑contributing indices."""
    rng = np.random.default_rng()
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    temp_start, temp_end = 1e-3, 1e-7
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate the currently maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[idx[mask]]

        # source (largest contribution)
        i = int(np.argmax(contrib))
        # target (smallest contribution, excluding the source)
        tmp = contrib.copy()
        tmp[i] = np.inf
        j = int(np.argmin(tmp))

        if a[i] <= 0.0:
            # nothing to move – perturb and continue
            a = best_a.copy()
            a += rng.normal(scale=0.01, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            continue

        delta = a[i] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j : j + n] += 2.0 * delta * a
        b_new[i : i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[SA   {it:6d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturbation to jump out of a plateau
            a = best_a.copy()
            a += rng.normal(scale=0.02, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA   {it:6d}] perturb after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – tiny uniform mixing (grid search over α)
# ----------------------------------------------------------------------
def _mix_with_uniform(a, n, cur_val):
    """Try convex combinations (1‑α)·a + α·uniform for many α."""
    best_a, best_val = a, cur_val
    uniform = np.full(n, 1.0 / n)

    alphas = np.concatenate(
        (
            np.linspace(0.001, 0.02, 10),
            np.linspace(0.03, 0.2, 10),
        )
    )
    for alpha in alphas:
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a step‑function within the allotted time budget.
    Returns a list of non‑negative floats (≈1 in sum) that minimises
    ``evaluate_sequence``.
    """
    global height_sequence_1  # might be defined by the harness
    rng = np.random.default_rng(seed)
    start = time.time()
    # small safety margin for the final secure call
    deadline = start + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a few thousand entries provide a good trade‑off.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Phase 0 – parametric sweep (≈5 % of the budget)
    # --------------------------------------------------------------
    phase0_deadline = start + 0.05 * (deadline - start)
    a, val = _parametric_sweep(n, rng, phase0_deadline)

    # --------------------------------------------------------------
    # Phase A – Adam optimiser (≈25 % of the remaining time)
    # --------------------------------------------------------------
    phase_a_deadline = phase0_deadline + 0.25 * (deadline - phase0_deadline)
    a, val = _adam_phase(
        a,
        n,
        start=phase0_deadline,
        deadline=phase_a_deadline,
        best_a=a,
        best_val=val,
        target_val=1.503,
    )

    # --------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (≈70 % of the rest)
    # --------------------------------------------------------------
    phase_b_deadline = phase_a_deadline + 0.70 * (deadline - phase_a_deadline)
    a, val = _hillclimb_sa(
        a,
        n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=a,
        best_val=val,
        target_val=1.503,
    )

    # --------------------------------------------------------------
    # Phase C – final tiny uniform mixing
    # --------------------------------------------------------------
    a, val = _mix_with_uniform(a, n, val)

    # --------------------------------------------------------------
    # Final secure evaluation (uses the provided wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return a.tolist()
