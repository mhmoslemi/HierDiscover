"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Algorithm (≈1000 s budget)

1. **Initialisation** – collect a few diverse candidates
   (uniform, previous best, Dirichlet draws, sinusoid) and keep the best.

2. **Continuation L‑BFGS** – minimise a smooth surrogate of the
   max‑convolution (log‑sum‑exp).  The smoothing parameter ε is decreased
   from 1e‑1 to 1e‑16 over a log‑spaced schedule, with several random
   restarts.  After each schedule we apply an anti‑symmetric re‑ordering
   (large values on the left, small on the right) that often flattens
   the convolution peak.

3. **Exact mirror‑descent** – iterate exponential updates with the true
   sub‑gradient of the max‑convolution, projecting onto the probability
   simplex each step.

4. **Weighted pairwise annealing** – repeatedly move a small fraction of
   mass away from the pair (i,j) that contributes most to the current
   maximal convolution entry, donating it to a low‑mass index.  Moves are
   accepted with a decreasing temperature schedule, and occasional random
   circular shifts are tried.

5. **Final cheap hill‑climb** – stochastic swaps and tiny mass transfers
   that accept only strict improvements.

The best sequence found is finally checked with the official
``evaluate_sequence`` and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None  # smoothing parameter ε
_GLOBAL_N = None    # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Clip tiny negatives caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max to obtain a -----
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- convolution -----
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max -----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    weights = np.exp(shifted)
    p = weights / weights.sum()      # probability over lag indices

    # derivative of conv_k w.r.t a is 2·a_{k‑i} (symmetric)
    # we compute convolution of a with reversed p and extract the
    # relevant segment.
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]      # length n, matches a_i indices
    grad_a = 4.0 * n * seg[::-1]        # factor 4·n from the objective

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    (Wang & Carreira‑Perpiñán, 2013)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the full formula
    reduces to this expression).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places large values on the left side, small values on the
    right side – often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climber that accepts only strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector stays on the
    simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # move a small fraction
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x

            # back‑transform to a probability vector
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[L‑BFGS eps={eps:.1e}] new best = {best_val:.6f}")

        # try anti‑symmetric re‑ordering after each restart
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val - 1e-9:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after restart] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – refined mirror‑descent on the exact max‑convolution.
    # ------------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.7          # initial step size (exponential mirror descent)
    decay = 0.998
    min_alpha = 0.01
    it = 0
    while time.time() < deadline - 150.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idxs = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idxs
        g[idxs] = a[jdx]

        # exponential mirror‑descent update
        alpha = max(alpha0 * (decay ** (it // 500)), min_alpha)
        a = a * np.exp(-alpha * g)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 250 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-10:
                best_val = cur_val
                best_a = a.copy()
                print(f"[MD it {it}] new best = {best_val:.6f}")
            # occasional anti‑symmetric reorder
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val - 1e-10:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[MD reorder] improved to {best_val:.6f}")
        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – weighted pairwise mass‑transfer annealing with occasional shifts.
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.02                # initial temperature
    cooling = 0.9999
    while time.time() < deadline - 30.0:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        # indices that form the peak
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idxs = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idxs
        contrib = cur_a[idxs] * cur_a[jdx]

        # donor index chosen proportionally to its contribution to the peak
        if contrib.sum() == 0:
            donor = rng.integers(0, n)
        else:
            probs = contrib / contrib.sum()
            donor = rng.choice(idxs, p=probs)

        # pick a low‑mass recipient (bottom 5 % of entries)
        low_thresh = np.percentile(cur_a, 5)
        low_inds = np.where(cur_a <= low_thresh)[0]
        if low_inds.size == 0:
            recipient = rng.integers(0, n)
        else:
            recipient = rng.choice(low_inds)

        # move a modest fraction of mass
        delta = cur_a[donor] * 0.05 * rng.random()
        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[recipient] += delta

        # projection onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[pairwise] improved to {best_val:.6f}")

        # occasional random circular shift (helps because of zero padding)
        if rng.random() < 0.01:
            shift = rng.integers(-n // 2, n // 2 + 1)
            shifted = np.roll(cur_a, shift)
            val_shift = _fast_evaluate(shifted)
            if val_shift < cur_val:
                cur_a = shifted
                cur_val = val_shift
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = cur_a.copy()
                    print(f"[shift] improved to {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # Phase 4 – final cheap hill‑climbing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
