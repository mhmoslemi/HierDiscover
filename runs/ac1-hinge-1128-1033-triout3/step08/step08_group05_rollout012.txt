def _max_pair_info(a):
    conv = _full_convolution(a)
    k = conv.argmax()
    i_start = max(0, k - (len(a) - 1))
    i_end = min(len(a) - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    jdx = k - idx
    prod = a[idx] * a[jdx]
    # if i == j (when k even and i=j), product counted once; that's okay.
    pair_idx = np.argmax(prod)
    return k, idx[pair_idx], jdx[pair_idx], prod[pair_idx]

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Initialisation** –  Load the previously discovered sequence
   ``height_sequence_1`` if it exists, otherwise start from a random
   Dirichlet vector.  Normalise the vector onto the probability simplex
   (sum = 1) – the evaluation is homogeneous, therefore fixing the sum
   does not affect the optimal value.

2. **Pre‑processing** –  Apply a cheap anti‑symmetric re‑ordering
   (large values on the left, tiny values on the right).  This heuristic
   is known to flatten the self‑convolution peak.

3. **Simulated‑Annealing (SA) loop** –  The remaining time budget is spent
   in a SA that explores the space of feasible step‑functions.
   Four move types are used:

   * *Swap* – exchange two heights.
   * *Mass transfer* – move a small percentage of the mass of a donor
     index to a recipient index.
   * *Targeted transfer* – identify the pair (i,j) that contributes most
     to the current maximal convolution entry and move a fraction of the
     larger partner to a low‑mass index.
   * *Neighbour smoothing* – replace two adjacent entries by their
     average (a tiny local blur).

   Moves are accepted if they improve the fast surrogate
   ``_fast_evaluate``; otherwise they are accepted with a Metropolis
   probability ``exp(-(Δ)/T)`` where the temperature ``T`` follows a
   slow exponential cooling schedule.

4. **Occasional re‑ordering** – every few thousand iterations the
   anti‑symmetric re‑ordering is tried again; if it yields a better
   value we switch to it.

5. **Final safety check** – the best vector found is handed to the
   official ``evaluate_sequence`` once (the code prints the exact score)
   and then returned as a plain Python list.

The whole routine respects the 1000 s time budget, prints progress
messages, and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – required by the problem statement)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of power‑of‑2 size.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may produce tiny negative values – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of the official ``evaluate_sequence`` for non‑negative
    vectors that sum to 1.  It returns 2·n·max(conv) where conv = a*a.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic re‑ordering that places the largest values on the left side
    and the smallest on the right side – often reduces the maximal
    self‑convolution entry.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _max_pair_info(a: np.ndarray) -> tuple[int, int, int, float]:
    """
    Identify the index k where the self‑convolution attains its maximum
    and return the pair (i,j) with i+j=k that contributes the most
    (i.e. maximises a_i·a_j).  Also return the product value.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k = int(conv.argmax())
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    jdx = k - idx
    prod = a[idx] * a[jdx]
    pair_idx = int(np.argmax(prod))
    return k, idx[pair_idx], jdx[pair_idx], float(prod[pair_idx])
