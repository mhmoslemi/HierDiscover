"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initial seeding** – try a few simple families (uniform,
   exponentially decreasing, a copy of the previous best if available).
   Keep the best according to a fast evaluator.

2. **Linear‑programming refinement** – for the current vector ``f`` we
   compute ``M = max(f*f)``.  The convolution ``f * g`` is linear in ``g``,
   therefore we can maximise the total mass ``sum(g)`` under the linear
   constraints ``f * g ≤ M`` and ``0 ≤ g ≤ 1000``.
   The LP solution ``g`` supplies additional mass that does not raise the
   current worst convolution bucket.  We move a fraction ``t`` towards
   ``g`` (convex combination) and repeat a few times.  This step is the
   main source of improvement over the previous baseline.

3. **Light‑weight local search** – after the LP phase a cheap hill‑climb
   (random swaps and tiny mass‑transfer moves) is run until the time
   budget expires.  The vector is always re‑projected onto the feasible
   box ``[0,1000]``.

All steps respect the 1000 s budget; the best sequence found is finally
checked with ``evaluate_sequence`` and returned as a plain Python list.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors.
    Returns   2·n·max(conv) / (sum(a)^2) .
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:                     # protect against division by zero
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that A @ g = f * g  (full convolution).
    Each diagonal offset d (0 ≤ d < n) contains the constant value f[d].
    """
    n = f.shape[0]
    # each diagonal is a constant vector of length n
    diags = [np.full(n, f[d], dtype=float) for d in range(n)]
    offsets = np.arange(n)      # 0,1,…,n‑1
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _alternating_extremes_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values at the two ends alternating inward.
    This usually reduces the size of the centre bucket of the convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)[::-1]            # descending
    new_a = np.empty_like(a)
    left = 0
    right = n - 1
    toggle = True
    for v in sorted_vals:
        if toggle:
            new_a[left] = v
            left += 1
        else:
            new_a[right] = v
            right -= 1
        toggle = not toggle
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector is kept
    inside the feasible box and renormalised after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-12:
                continue
            delta = cur_a[i] * 0.07 * rng.random()   # up to 7 % of source mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # enforce box constraints and renormalise
        new_a = np.clip(new_a, 0.0, 1000.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[hill climb] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the allotted time budget and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # keep a safety margin

    # ------------------------------------------------------------------
    # Choose problem size ------------------------------------------------
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = min(2000, old.shape[0])       # we keep the problem size modest
        has_old = True
    except Exception:
        n = 2000
        has_old = False

    # ------------------------------------------------------------------
    # 1) Generate a few cheap seeds
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform (scaled to unit sum, later normalised again inside the loop)
    init_candidates.append(np.full(n, 1.0))

    # exponential families (different decay rates)
    for lam in np.logspace(-4, 0, 9):
        a = np.exp(-lam * np.arange(n))
        init_candidates.append(a)

    # copy of previous best (if any), truncated / stretched to length n
    if has_old:
        if old.shape[0] != n:
            # simple truncation or repetition to reach length n
            if old.shape[0] > n:
                a = old[:n].copy()
            else:
                repeats = int(np.ceil(n / old.shape[0]))
                a = np.tile(old, repeats)[:n]
        else:
            a = old.copy()
        init_candidates.append(a)

    # random Dirichlet draws (ensure positivity)
    for _ in range(2):
        a = rng.gamma(shape=1.0, scale=1.0, size=n)
        init_candidates.append(a)

    # normalise everything onto the simplex (sum = 1) – this is convenient
    # for the fast evaluator but not strictly required.
    init_candidates = [c / c.sum() for c in init_candidates]

    # ------------------------------------------------------------------
    # 2) Pick the best cheap candidate.
    # ------------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] fast evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best start value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Linear‑programming refinement phase.
    # ------------------------------------------------------------------
    max_lp_iters = 15          # number of outer LP refinements
    t_mix = 0.25               # convex‑combination weight towards the LP solution
    iteration = 0
    while time.time() < deadline - 30.0 and iteration < max_lp_iters:
        iteration += 1
        # current worst bucket
        conv_f = _full_convolution(best_a)
        M = conv_f.max()
        print(f"[LP {iteration}] current max convolution = {M:.6f}")

        # build the (2n‑1)×n constraint matrix for the current f
        A = _build_conv_matrix(best_a)

        # linear program: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)

        # right‑hand side: the current worst bucket repeated for every constraint
        b_ub = np.full(2 * n - 1, M, dtype=float)

        # variable bounds: 0 ≤ g_i ≤ 1000 (same bound as evaluate_sequence)
        bounds = [(0.0, 1000.0) for _ in range(n)]

        # solve the LP with HiGHS (fast for medium sized problems)
        res = opt.linprog(c,
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True})

        if not res.success:
            print(f"[LP {iteration}] LP failed: {res.message}")
            break

        g = res.x
        g_sum = g.sum()
        print(f"[LP {iteration}] LP optimum sum(g) = {g_sum:.6f}")

        # convex combination with the current vector
        new_a = (1.0 - t_mix) * best_a + t_mix * g

        # enforce the global box constraints again
        new_a = np.clip(new_a, 0.0, 1000.0)

        # renormalise to keep the scale sensible (the objective is scale‑invariant)
        s = new_a.sum()
        if s > 0:
            new_a /= s

        # optional re‑ordering heuristic (alternating extremes)
        new_a = _alternating_extremes_reorder(new_a)

        cur_val = _fast_evaluate(new_a)
        print(f"[LP {iteration}] post‑mix evaluation = {cur_val:.6f}")

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = new_a.copy()
            print(f"[LP {iteration}] NEW BEST = {best_val:.6f}")

        # slowly decay the mixing coefficient so later steps become more conservative
        t_mix = max(0.05, t_mix * 0.85)

    # ------------------------------------------------------------------
    # 4) Final cheap hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # 5) Final safety check with the official evaluator and return.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
