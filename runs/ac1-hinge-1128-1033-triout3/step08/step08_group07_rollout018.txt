"""
Hybrid optimizer for the step‑function minimisation problem.

The search combines several ideas that differ from the reference
implementation:

1. **Larger problem size** – we use n = 6000 (still cheap with an FFT‑based
   convolution).  More degrees of freedom generally allow a smaller
   max‑convolution peak.

2. **Rich set of initialisations** – uniform, exponentially decaying,
   “ends‑large” alternation and a few random restarts.  The best among them
   seeds the optimisation.

3. **Smooth surrogate (log‑sum‑exp) minimisation** – the same L‑BFGS
   routine as before, but the ε‑schedule is extended down to 1e‑15 and
   we perform many short restarts (re‑ordering after each restart).

4. **Projected Adam with a multi‑peak sub‑gradient** – instead of using
   only the single active peak we sum the gradient contributions of all
   convolution entries that are within a tiny tolerance of the current
   maximum.  This provides a more stable descent direction.

5. **Long simulated‑annealing** – swaps of positions plus small mass
   transfers are accepted with a Metropolis rule.  The temperature schedule
   is slower than before, allowing a deeper exploration.

6. **Final cheap hill‑climbing** – deterministic improvement moves until the
   time budget expires.

All heavy work (FFT convolution, gradient computation) is performed with
pre‑allocated buffers to keep the runtime low.  The function respects the
1000 s wall‑time budget and always returns a valid non‑negative sequence.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the L‑BFGS phase
_GLOBAL_N = None     # length of the current sequence
_FFT_SIZE = None     # cached FFT size (next power of two)


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _prepare_fft(size_needed: int):
    """Cache a suitable FFT size for the given required length."""
    global _FFT_SIZE
    _FFT_SIZE = _next_pow2(size_needed)


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    if _FFT_SIZE is None or _FFT_SIZE < out_len:
        _prepare_fft(out_len)
    size = _FFT_SIZE
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)  # clip tiny negatives from rounding
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Exact copy of ``evaluate_sequence`` (used for speed)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max yielding a probability vector."""
    w_shift = w - np.max(w)
    ex = np.exp(w_shift)
    return ex / np.sum(ex)


# ----------------------------------------------------------------------
# Smooth surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    with conv = a∗a and a = softmax(w).  The gradient is derived analytically.
    """
    a = _softmax(w)                     # ℓ₁‑normalised non‑negative vector

    conv = _fft_convolve_self(a)        # length 2·n‑1
    max_c = conv.max()

    # smooth approximation of the max
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)        # ∑exp((c−max)/ε)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / np.sum(weights)       # softmax over the convolution entries

    # convolution a * reverse(p) gives the needed weighted sum
    p_rev = p[::-1]
    size = _next_pow2(conv.shape[0] + a.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[:a.shape[0] + p_rev.shape[0] - 1]

    # central n entries correspond to indices [n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]      # length n
    grad_a = 4.0 * n * seg[::-1]        # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Simple deterministic re‑ordering – large values at the extremes
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values alternately at the two ends."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left, right = 0, n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


# ----------------------------------------------------------------------
# Multi‑peak sub‑gradient (used by Adam)
# ----------------------------------------------------------------------
def _multi_peak_grad(a: np.ndarray, tol: float = 1e-12):
    """
    Compute a sub‑gradient of ``max(conv)`` where all convolution entries
    with value ≥ max(conv)−tol are treated as active.
    Returns ``grad_a`` of shape (n,).
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = conv.max()
    # mask of indices that are (almost) maximal
    mask = conv >= max_b - tol
    # p is the indicator of active peaks (as a vector of length 2n‑1)
    p = mask.astype(float)

    # gradient of Σ_k p_k * conv_k w.r.t. a is 2 * (a * reverse(p))
    p_rev = p[::-1]
    size = _next_pow2(p_rev.shape[0] + a.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    grad = 2.0 * np.fft.irfft(fa * fp, size)[:n]   # length n
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser using the multi‑peak gradient
# ----------------------------------------------------------------------
def _adam_multi_peak(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Adam optimiser on the exact max‑convolution using a sub‑gradient that
    aggregates all (near‑)active peaks.  The iterate is kept on the simplex.
    """
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)          # initialise soft‑max parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a0.copy()
    best_val = _eval_fast(best_a)

    while time.time() < deadline:
        t += 1
        a = _softmax(w)

        # multi‑peak gradient w.r.t. a
        grad_a = _multi_peak_grad(a, tol=1e-12)

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = 0.6 / (np.sqrt(t) + 1.0)   # slightly larger initial step
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _softmax(w)
        cur_val = _eval_fast(a_new)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[Adam] iter {t} new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (swap + tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """Classic SA with a slower temperature decay."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.05, 1e-6)   # start a bit hotter than before
    decay = 0.99996                # slower cooling

    it = 0
    while time.time() < deadline:
        if rng.random() < 0.25:                     # pure swap move
            i, j = rng.integers(0, n, size=2)
            if i != j:
                a[i], a[j] = a[j], a[i]
        else:                                       # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * 0.04 * rng.random()   # up to 4 %
                a[i] -= delta
                a[j] += delta

        new_val = _eval_fast(a)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (it {it})")
        else:
            # revert
            if rng.random() < 0.25:
                a[i], a[j] = a[j], a[i]
            else:
                a[i] += delta
                a[j] -= delta

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Cheap hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                cur_val: float,
                deadline: float) -> (np.ndarray, float):
    """Accept any strictly improving swap or mass‑transfer move."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.3:                     # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-14:
                continue
            delta = cur_a[i] * 0.015 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[hill] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Candidate generators – a few simple parametric families
# ----------------------------------------------------------------------
def _uniform_vector(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _exponential_vector(n: int, rate: float) -> np.ndarray:
    """
    a_i ∝ exp(-rate * i).  The vector is normalised to sum = 1.
    """
    x = np.arange(n, dtype=float)
    a = np.exp(-rate * x)
    a /= a.sum()
    return a


def _random_vector(n: int) -> np.ndarray:
    a = np.random.rand(n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a
