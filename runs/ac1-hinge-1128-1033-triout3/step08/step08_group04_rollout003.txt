"""
Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Summary of the search strategy
--------------------------------
1. **Rich set of parametric seeds** – uniform, previous solution,
   random Dirichlet draws, simple exponentials (fine λ‑grid), power‑laws,
   and a two‑level “high‑block / low‑block’’ family.
   The best seed (according to the cheap exact objective
   ``_fast_evaluate``) is kept.

2. **Smooth surrogate optimisation (L‑BFGS‑B)** – minimise a
   log‑sum‑exp approximation of the max‑convolution.  A long ε‑schedule
   (≈ 25 values from 5·10⁻³ down to 10⁻¹⁴) and many restarts (≈ 25)
   are performed.  After each restart we also try several deterministic
   reorderings (sorted‑decreasing, alternating large/small) to exploit
   the known benefit of placing large masses on one side.

3. **Aggressive mirror‑descent with line‑search** – using the exact
   sub‑gradient of the true max‑convolution.  For each iteration a set
   of candidate step‑sizes (λ) is tested, the one giving the largest
   improvement is applied, then the vector is projected back onto the
   probability simplex.  The step‑size schedule is adaptive and a
   small random perturbation is injected every few hundred iterations.

4. **Short simulated‑annealing / hill‑climb phase** – random block
   swaps and mass‑transfer moves with a geometric cooling schedule.

5. **Final refinement** – cheap hill‑climbing that only accepts strict
   improvements.

All phases respect the global 1000 s budget.  The best vector found is
checked with the official ``evaluate_sequence`` and returned as a plain
Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate.
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise can produce tiny negative values – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already satisfy
    a_i ≥ 0 and Σ a_i = 1.  Returns 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values left, smallest right (alternating)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _desc_reorder(a: np.ndarray) -> np.ndarray:
    """Sort the vector in non‑increasing order (largest left)."""
    return np.sort(a)[::-1]


def _random_permutation(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a random permutation of a (used only for exploration)."""
    perm = rng.permutation(a.shape[0])
    return a[perm]


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution
# ----------------------------------------------------------------------
def _subgrad_maxconv(a: np.ndarray):
    """
    Compute the sub‑gradient g of the functional
        f(a) = max_k (a * a)[k]
    at the current a, together with the index k attaining the maximum.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k = int(conv.argmax())
    # g_i = a_{k-i} for all i such that both indices are inside [0,n)
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = a[k - idx]
    return g, k, conv


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp of the max) used by L‑BFGS‑B
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv - max)/ε)
    where a_i is obtained from a soft‑max parametrisation of w.
    Returns the scalar value and the gradient w.r.t. w.
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max parametrisation -----
    w_shift = w - np.max(w)           # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- convolution -----
    conv = _full_convolution(a)       # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max (log‑sum‑exp) -----
    shifted = (conv - max_c) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()       # soft‑max over lag indices

    # ----- gradient w.r.t. a -----
    # Convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # keep only the segment that aligns with original a‑vector
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    L = 2.0 * n * max_c + 2.0 * n * eps * logsumexp(shifted)
    return float(L), grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses global ε and n)."""
    return _smooth_obj_and_grad(w)


# ----------------------------------------------------------------------
# Helper: simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    O(n) algorithm (Wang & Carreira‑Perpiñán, 2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _candidate_exponential(lam: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum 1."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def _candidate_power_law(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^p, normalised to sum 1."""
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def _candidate_two_level(m_frac: float, ratio: float, n: int) -> np.ndarray:
    """
    Two‑level block distribution:
        first m = floor(m_frac·n) entries have height h_high,
        the remaining entries have height h_low,
        with h_high/h_low = ratio.
    """
    if m_frac <= 0 or m_frac >= 1:
        raise ValueError
    m = int(np.round(m_frac * n))
    if m == 0 or m == n:
        # degenerate – fall back to uniform
        return np.full(n, 1.0 / n)
    # Solve:
    #   m·h_high + (n-m)·h_low = 1,
    #   h_high = ratio·h_low
    denom = m * ratio + (n - m)
    h_low = 1.0 / denom
    h_high = ratio * h_low
    a = np.full(n, h_low, dtype=float)
    a[:m] = h_high
    return a


# ----------------------------------------------------------------------
# Main routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Keep a small safety margin for the final official evaluation.
    deadline = start_time + budget_s - 4.0

    # --------------------------------------------------------------
    # Determine problem size n (use the previous solution if available)
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # --------------------------------------------------------------
    # 1) Build a rich set of cheap initial candidates.
    # --------------------------------------------------------------
    init_candidates = []

    # Uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # Previous solution (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # Random Dirichlet‑like draws (shape parameters 0.5, 2.0, 5.0)
    for shape in (0.5, 2.0, 5.0):
        cand = rng.gamma(shape=shape, scale=1.0, size=n)
        cand /= cand.sum()
        init_candidates.append(cand)

    # Simple sinusoid (positive shift)
    x = np.linspace(0, np.pi, n)
    init_candidates.append((np.sin(x) + 0.1))
    init_candidates[-1] /= init_candidates[-1].sum()

    # --------------------------------------------------------------
    # Parametric families (dense grids)
    # --------------------------------------------------------------
    # Exponential decay (left‑leaning)
    for lam in np.logspace(-4, -0.2, 30):          # finer grid around the sweet spot
        init_candidates.append(_candidate_exponential(lam, n))

    # Power‑law (p negative ⇒ heavier tail)
    for p in np.linspace(-2.5, -0.1, 20):
        init_candidates.append(_candidate_power_law(p, n))

    # Two‑level block families (m_frac from 0.02 to 0.4, ratio from 1.5 to 30)
    for m_frac in np.linspace(0.02, 0.4, 12):
        for ratio in np.logspace(np.log10(1.5), np.log10(30), 12):
            init_candidates.append(_candidate_two_level(m_frac, ratio, n))

    # --------------------------------------------------------------
    # 2) Pick the best cheap candidate.
    # --------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
        if idx % 10 == 0:
            print(f"[init {idx:03d}] value = {val:.6f}")
    print(f"[init] best initial value = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – L‑BFGS on the smooth surrogate of max‑convolution.
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-14), num=25)
    max_restarts = 25
    w = np.log(best_a + 1e-12)          # soft‑max parameters of the best seed
    restart = 0
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        # Occasional full random re‑initialisation of w
        if restart % 6 == 0:
            w = np.random.randn(n) * 1.5

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n

            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 1200,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x

            # Convert back to probability vector and evaluate
            ws = w - np.max(w)
            a_candidate = np.exp(ws)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # --- deterministic re‑orderings after each restart ---
        for reorder_fun, name in [( _alternating_reorder, "alternating"),
                                  ( _desc_reorder, "desc" )]:
            a_ord = reorder_fun(best_a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[reorder {name}] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 2 – Aggressive mirror‑descent with line‑search.
    # --------------------------------------------------------------
    a = best_a.copy()
    it = 0
    # Step‑size candidates (log‑spaced)
    lambda_candidates = np.logspace(-4, 0, 12)   # from 1e‑4 to 1
    # Frequency of random perturbations
    perturb_every = 400

    while time.time() < deadline - 20.0:
        # Compute sub‑gradient at the current max‑convolution index
        g, k_max, conv = _subgrad_maxconv(a)

        # Try each λ and keep the best improvement
        cur_val = _fast_evaluate(a)
        best_local_a = a
        best_local_val = cur_val

        for lam in lambda_candidates:
            # multiplicative update respecting positivity
            upd = a * np.exp(-lam * g)
            upd = np.maximum(upd, 0.0)
            s = upd.sum()
            if s <= 1e-12:
                continue
            upd /= s
            val = _fast_evaluate(upd)
            if val < best_local_val - 1e-12:
                best_local_val = val
                best_local_a = upd

        if best_local_val < cur_val - 1e-12:
            a = best_local_a
            cur_val = best_local_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[mirror iter {it:05d}] new best = {best_val:.6f}")

        # Occasionally inject a tiny random jitter to escape plateaus
        if it % perturb_every == 0 and it > 0:
            jitter = rng.normal(scale=0.02, size=n)  # 2 % noise
            a_pert = a + jitter * a   # relative noise
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= a_pert.sum()
            a = a_pert
            # No immediate evaluation – the next iteration will handle it.

        # Every few hundred iterations also try a cheap reorder
        if it % 500 == 0:
            a_alt = _alternating_reorder(a)
            val_alt = _fast_evaluate(a_alt)
            if val_alt < best_val:
                best_val = val_alt
                best_a = a_alt.copy()
                a = a_alt.copy()
                print(f"[mirror reorder] improved to {best_val:.6f}")

        it += 1

    # --------------------------------------------------------------
    # 5) Phase 3 – Simulated‑annealing style random block moves.
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.025                      # initial temperature
    cooling = 0.99990
    rng_sa = np.random.default_rng(seed + 7777)

    while time.time() < deadline - 5.0:
        # Choose move type
        if rng_sa.random() < 0.6:
            # single‑index mass transfer
            i, j = rng_sa.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.12 * rng_sa.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # block swap (length up to 30)
            L = rng_sa.integers(1, min(30, n // 6))
            i = rng_sa.integers(0, n - L)
            j = rng_sa.integers(0, n - L)
            new_a = cur_a.copy()
            tmp = new_a[i:i + L].copy()
            new_a[i:i + L] = new_a[j:j + L]
            new_a[j:j + L] = tmp

        # Re‑normalise onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng_sa.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA iter {it}] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-6)
        it += 1

    # --------------------------------------------------------------
    # 6) Final cheap hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # 7) Final safety check with the official evaluator and return.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()


# ----------------------------------------------------------------------
# Very cheap hill‑climbing used at the very end of the pipeline.
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Local search that only accepts strict improvements.
    Random swaps (20 %) and mass‑transfer moves (80 %).
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
