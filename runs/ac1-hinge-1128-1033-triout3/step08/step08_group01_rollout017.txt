"""
search for a step function that minimises
    2·n·max(conv(a,a))/sum(a)^2

The implementation uses three stages:
  1.  Generate a diverse set of starting vectors.
  2.  Refine each start with a few L‑BFGS runs on a smooth
      log‑sum‑exp approximation of the max‑convolution (the
      approximation becomes tighter as the schedule proceeds).
  3.  A final exact hill‑climbing phase on the true objective and
      a tiny uniform mixing sweep.

All calculations are performed on the simplex (Σa = 1), because the
objective is scale‑invariant.  The code respects the 1000‑second time
budget and prints progress messages.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of real vectors x and y using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a vector on the simplex)."""
    w_shift = w - np.max(w)
    e = np.exp(w_shift)
    return e / e.sum()


def _objective_true(a: np.ndarray) -> float:
    """Exact objective for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Smooth max‑convolution and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (f, g) where
        f = 2·n·smooth_max(conv(a,a))
        g = ∂f/∂w   (w are log‑softmax logits)

    a = softmax(w) is on the simplex.
    eps is the temperature of the log‑sum‑exp approximation.
    """
    a = _softmax(w)                           # a on simplex
    b = _full_convolution(a)                  # length 2n‑1
    b_max = b.max()

    # smooth max = max + eps * log Σ exp((b-b_max)/eps)
    shifted = (b - b_max) / eps
    exp_shift = np.exp(shifted)
    sum_exp = exp_shift.sum()
    smooth_max = b_max + eps * np.log(sum_exp)

    # soft‑max over convolution positions (weights for the gradient)
    p = exp_shift / sum_exp                    # length 2n‑1
    p_rev = p[::-1]

    # gradient w.r.t. a : 4·n·conv(a , p_rev)   (extract the middle n entries)
    conv_full = _conv_fft(a, p_rev)            # length 3n‑2
    segment = conv_full[n - 1: 2 * n - 1]      # length n
    grad_a = 4.0 * n * segment[::-1]           # shape (n,)

    # chain rule through the soft‑max (log‑softmax gradient)
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    f = 2.0 * n * smooth_max
    return f, grad_w


# ----------------------------------------------------------------------
# Simple hill‑climbing on the exact (non‑smoothed) objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start_time: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.5) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing that moves a tiny amount of mass from heavy to light spots."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # choose a source index proportionally to its contribution
        if total > 0:
            probs = contrib / total
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # pick a target index (prefer a place that does not influence the max entry)
        if np.random.rand() < 0.35:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and restart from the best seen so far
            a = best_a.copy()
            noise = np.random.normal(scale=0.010, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (often helps a little)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective_true(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_true(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Return a list of non‑negative floats (Σ = 1) that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5            # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # Initialise a – use an old construction if it exists, otherwise a rich set.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0 = np.full(n, 1.0 / n)                 # uniform as a fallback

    best_a = a0.copy()
    best_val = _objective_true(best_a)

    # --------------------------------------------------------------
    # Stage 1 – generate a handful of diverse starting points
    # --------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # exponential left‑hand side (different decay rates)
    for r in (0.99999, 0.99997, 0.99995, 0.9999):
        a = r ** np.arange(n, dtype=float)
        a = a / a.sum()
        candidates.append(a)

    # power‑law (symmetric)
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p)
        a = a / a.sum()
        candidates.append(a)

    # a few random Dirichlet draws
    rng = np.random.default_rng(seed + 999)
    for _ in range(12):
        a = rng.dirichlet(np.ones(n))
        candidates.append(a)

    # --------------------------------------------------------------
    # Stage 2 – L‑BFGS refinement on a smooth approximation of max(conv)
    # --------------------------------------------------------------
    eps_schedule = [0.02, 0.005, 0.001, 1e-4]          # increasingly tight
    max_iter_per_stage = 180

    for a_start in candidates:
        if time.time() >= deadline:
            break

        # initialise logits (log‑softmax parametrisation)
        w = np.log(a_start + 1e-12)

        for eps in eps_schedule:
            if time.time() >= deadline:
                break

            # run a short L‑BFGS optimisation for the current temperature
            res = scipy.optimize.minimize(
                lambda w_: _smooth_obj_and_grad(w_, eps, n),
                w,
                jac=True,
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_stage, 'disp': False}
            )
            w = res.x

        # evaluate the true objective for the final w of this run
        a_opt = _softmax(w)
        val = _objective_true(a_opt)
        if val < best_val:
            best_val, best_a = val, a_opt.copy()
            print(f"[L‑BFGS] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – exact hill‑climbing on the true objective
    # --------------------------------------------------------------
    best_a, best_val = _hillclimb_targeted(best_a, n,
                                           start_time=start,
                                           deadline=deadline,
                                           best_a=best_a,
                                           best_val=best_val,
                                           target_val=1.503)

    # --------------------------------------------------------------
    # Stage 4 – tiny uniform mixing (often yields a final dip)
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # --------------------------------------------------------------
    # Final evaluation (secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
