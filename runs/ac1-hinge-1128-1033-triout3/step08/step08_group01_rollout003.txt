"""
A fast, time‑budget aware optimiser for the “max‑self‑convolution” objective.

The objective to minimise is

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

where `a` is a non‑negative vector (the step‑function heights),
`n = len(a)` and `conv` is the linear convolution of `a` with itself.
The evaluation routine `evaluate_sequence` (provided by the harness)
applies the same formula after clipping and normalising, therefore the
optimiser works on the mathematically equivalent problem with the
additional constraint `sum(a)=1`.

The optimiser proceeds in three stages:

1. **Rich initialisation** – a sweep over several parametric families
   (uniform, exponentials, power‑laws, sines, triangles) plus a few
   Dirichlet draws.  The best candidate (lowest objective) is kept.

2. **Adam on a smooth log‑sum‑exp approximation** – the non‑smooth
   `max` is replaced by a temperature‑controlled soft‑max.  Adam updates
   the logits of a soft‑max parametrisation of `a`.  This stage receives
   a modest fraction of the budget (≈10 %) but quickly moves the
   solution into a good basin.

3. **Targeted stochastic hill‑climbing / simulated annealing** – most of
   the remaining time is spent here.  In each iteration a small amount
   of mass is transferred from an index that contributes heavily to the
   current maximal convolution entry to an index that contributes little.
   The move is evaluated by an **incremental O(n) update** of the full
   convolution (no FFT).  The fraction of moved mass (`max_frac`) is
   annealed from 0.6 down to 5·10⁻⁴ and a classic SA temperature
   (acceptance of worsening moves) is also annealed, allowing the
   algorithm to escape shallow local minima.  After long stagnation the
   current solution is perturbed with a small Gaussian noise and the
   search continues.

Finally a tiny convex combination with the uniform distribution is tried,
as this often smoothes the last remaining peak.

All heavy numerics use NumPy and FFT only for the Adam phase; the hill‑climb
operates on plain arrays and therefore stays fast even for `n≈9000`.
The function `propose_candidate` respects the global time budget (default
1000 s) and returns the best sequence found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return the exact evaluation value for a simplex‑normalised vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1} (Euclidean)."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(v.shape[0]) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Rich initialisation – a diverse set of parametric families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector (Σ=1) and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # Uniform
    _try(np.ones(n))

    # Exponential (left‑hand side)
    for r in (0.99999, 0.99997, 0.99995, 0.9999, 0.9998, 0.9995, 0.999):
        a = r ** np.arange(n, dtype=float)
        _try(a)

    # Symmetric exponential (U‑shaped)
    for r in (0.99999, 0.99997, 0.99995, 0.9999, 0.9998, 0.9995):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # Power‑law symmetric
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p) + (n - i + 1) ** (-p)
        _try(a)

    # Half‑wave sine (zero at the ends, peak in the middle)
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _try(a)

    # Triangular (peak in the centre, zero at the ends)
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    _try(a)

    # Quadratic ramp
    a = (np.arange(n) + 1) * (n - np.arange(n))
    _try(a)

    # Random Dirichlet draws – a few quick samples (increase to 30)
    rng = np.random.default_rng(12345)
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam with a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp approximation of the max.
    The optimisation is performed in the log‑soft‑max (logits) space,
    guaranteeing feasibility (non‑negative, Σ=1) at every iteration.
    """
    # initialise logits from a0 (avoid log(0))
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.10, 0.001
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 40000

    it = 0
    while time.time() < deadline:
        # Soft‑max to obtain a feasible a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedules for learning rate and softmax temperature
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max via log‑sum‑exp -----
        b_full = _full_convolution(a)                     # length 2n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps                  # safe numbers
        weights = np.exp(shifted)
        p = weights / weights.sum()                       # soft‑max over positions

        # gradient of the smooth objective w.r.t. a:
        #   d/da 2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                     # length 3n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.005, size=n)

        # restart if stuck for too long
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.25
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Incremental update of the full self‑convolution when moving mass δ
# from index i to index j.
# ----------------------------------------------------------------------
def _incremental_conv_update(b: np.ndarray, a: np.ndarray,
                             i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a fresh copy of the convolution array after the move.
    The update follows from expanding (a‑δ·e_i + δ·e_j) * (a‑δ·e_i + δ·e_j).
    """
    n = a.shape[0]
    # start from a copy to keep the original untouched
    b_new = b.copy()

    # contributions with the unchanged part of a (linear term)
    # careful with slice bounds – they are always inside the array
    b_new[j:j + n] += 2.0 * delta * a
    b_new[i:i + n] -= 2.0 * delta * a

    # quadratic terms (i,i), (j,j) and the mixed (i,j) entry
    b_new[2 * j] += delta * delta
    b_new[2 * i] += delta * delta
    b_new[i + j] -= delta * delta

    return b_new


# ----------------------------------------------------------------------
# Phase B – Targeted stochastic hill‑climbing / simulated annealing
# ----------------------------------------------------------------------
def _hillclimb_sa(a_start: np.ndarray, n: int,
                  start: float, deadline: float,
                  best_a: np.ndarray, best_val: float,
                  target_val: float = 1.503) -> np.ndarray:
    """
    Move a small amount of probability mass from indices that heavily
    contribute to the current maximal convolution entry towards indices
    that hardly affect it.  The acceptance rule follows a classic
    Simulated‑Annealing schedule (temperature decreasing linearly).
    """
    a = a_start.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()

    # annealing parameters
    max_frac_start, max_frac_end = 0.60, 0.0005
    temperature_start, temperature_end = 0.03, 1e-6

    # stagnation handling
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        temperature = temperature_start * (1.0 - progress) + temperature_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to entry k_max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – weighted by its contribution
        if total_contrib > 0 and np.random.rand() < 0.95:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(contrib))

        if a[i] <= 1e-15:
            # nothing to move – pick a random index instead
            i = int(np.random.choice(n, p=np.ones(n) / n))

        # target index – prefers positions that barely affect the max entry
        if np.random.rand() < 0.45:
            # heuristic: low contribution to the *same* entry
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0:
                probs_j = w / w.sum()
                j = int(np.random.choice(n, p=probs_j))
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        # amount to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = _incremental_conv_update(b, a, i, j, delta)
        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # acceptance test – Metropolis criterion
        if new_val < cur_val or np.random.rand() < np.exp((cur_val - new_val) / temperature):
            # accept move
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # restart / perturb after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.010, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbation after stagnation")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Small post‑processing: mixing with the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few convex combinations with the uniform vector."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a list of floats (Σ ≈ 1) that minimises `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Slight safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries is a sweet spot for speed
    # and expressivity.
    # ------------------------------------------------------------------
    n = 9000

    # ------------------------------------------------------------------
    # Initialise a – use a known construction if present,
    # otherwise fall back to a diverse parametric sweep.
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` may be defined in the outer scope of the harness.
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        a0, _ = _initial_candidate(n)
        print("[init] generated fresh parametric start")

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.503                # objective target (corresponds to product = 0.7515)

    # ------------------------------------------------------------------
    # Phase A – Adam optimiser on a smooth approximation of max(conv)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.10 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        a0, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )
    print(f"[phase A] best objective after Adam = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Stochastic hill‑climbing / simulated annealing
    # ------------------------------------------------------------------
    phase_b_deadline = deadline                     # allocate the rest of the budget
    best_a = _hillclimb_sa(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )
    best_val = _objective_value(best_a)
    print(f"[phase B] best objective after hill‑climb = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[final mix] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
