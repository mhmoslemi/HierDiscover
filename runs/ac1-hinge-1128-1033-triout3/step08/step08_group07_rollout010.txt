def _alternating_extremes_reorder(a):
    n = a.shape[0]
    sorted_vals = np.sort(a)[::-1]  # descending
    new_a = np.empty_like(a)
    left = 0
    right = n - 1
    idx = 0
    while idx < n:
        new_a[left] = sorted_vals[idx]
        left += 1
        idx += 1
        if idx >= n:
            break
        new_a[right] = sorted_vals[idx]
        right -= 1
        idx += 1
    return new_a
```

"""
Hybrid optimizer for the step‑function minimisation problem (improved).

The search consists of four phases (all respecting the 1000 s wall‑time
budget):

1. **Smoothed surrogate optimisation** – minimise a log‑sum‑exp
   approximation of the max‑convolution with L‑BFGS.  Several restarts,
   a finer ε‑schedule and a larger iteration budget are used.  After each
   restart the sequence is reordered with two deterministic heuristics
   (large values at the ends and “alternating‑extremes”).  The best
   candidate is kept.

2. **Projected Adam descent on the exact max‑convolution** – a sub‑gradient
   of the true max entry is taken (gradient of the active peak) and the
   soft‑max parameters are updated with Adam.  The optimisation runs for a
   long time with a decreasing learning‑rate schedule.

3. **Targeted simulated annealing** – random swaps, random mass
   transfers, as well as *targeted* swaps (large ↔ small values) and
   *targeted* mass moves based on the current maximal convolution
   contribution are performed.  Temperature decays slowly and the best
   sequence is recorded.

4. **Greedy refinement** – repeatedly moves mass from the index that
   contributes most to the current maximum convolution towards the
   smallest entry, accepting only strictly improving moves.

All heavy work (convolutions, gradients) is performed with FFT‑based
convolution (`_fft_convolve_self`).  The final candidate is evaluated
with the provided `evaluate_sequence` function and returned as a plain
list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for surrogate optimisation (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter for the L‑BFGS phase
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Clip tiny negative values caused by rounding errors.
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast exact copy of `evaluate_sequence` for a normalised vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    w_shift = w - np.max(w)
    ex = np.exp(w_shift)
    return ex / np.sum(ex)


# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and conv = a∗a with a = softmax(w).  The gradient is derived
    analytically and returned w.r.t. the soft‑max parameters.
    """
    a = _softmax(w)                     # ℓ₁‑normalised non‑negative vector

    conv = _fft_convolve_self(a)        # length 2·n‑1
    max_c = conv.max()

    # smooth approximation of the max
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)        # ∑exp((c−max)/ε)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / np.sum(weights)       # softmax over the convolution entries

    # convolution a * reverse(p) gives the needed weighted sum
    p_rev = p[::-1]
    size = _next_pow2(conv.shape[0] + a.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[:a.shape[0] + p_rev.shape[0] - 1]

    # central n entries correspond to indices [n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]      # length n
    grad_a = 4.0 * n * seg[::-1]        # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by `scipy.optimize.minimize`."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Heuristic re‑ordering functions
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place large values alternating with small ones starting at the left."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


def _alternating_extremes_reorder(a: np.ndarray) -> np.ndarray:
    """Largest ↔ smallest extrema alternating from the ends."""
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]   # descending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    idx = 0
    while idx < n:
        new_a[left] = sorted_desc[idx]
        left += 1
        idx += 1
        if idx >= n:
            break
        new_a[right] = sorted_desc[idx]
        right -= 1
        idx += 1
    return new_a


# ----------------------------------------------------------------------
# Projected Adam optimiser (exact max‑convolution)
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Adam optimiser on the true max‑convolution (sub‑gradient of the max entry)
    while keeping the vector on the probability simplex.
    """
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)          # initialise soft‑max parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_a = a0.copy()
    best_val = _eval_fast(best_a)

    while time.time() < deadline:
        t += 1
        a = _softmax(w)

        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # gradient of the active max entry w.r.t. a
        grad_a = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_a[mask] = 2.0 * a[idx[mask]]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update on w
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning‑rate schedule (slow decay)
        lr = 0.5 / (np.sqrt(t) + 1.0)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _softmax(w)
        cur_val = _eval_fast(a_new)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[Adam] iter {t} new best = {best_val:.6f}")

    return best_a, best_val

# Targeted simulated annealing (swap + mass‑transfer moves)

def _anneal_targeted(a: np.ndarray,
                     cur_val: float,
                     deadline: float) -> (np.ndarray, float):
    """
    Simulated‑annealing that mixes random and *targeted* moves:
      * swap a large element with a small one,
      * random swaps,
      * random tiny mass transfers,
      * mass transfer from the index that contributes most to the current
        maximum convolution to the smallest element.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # initial temperature – a small fraction of the objective
    T = max(cur_val * 0.015, 1e-8)
    decay = 0.99996
    it = 0

    while time.time() < deadline:
        a_old = a.copy()
        move = rng.random()

        if move < 0.3:  # targeted swap (large ↔ small)
            k = max(1, int(0.01 * n))
            large_idx = np.argpartition(a, -k)[-k:]   # k largest indices
            small_idx = np.argpartition(a, k)[:k]    # k smallest indices
            i = rng.choice(large_idx)
            j = rng.choice(small_idx)
            a[i], a[j] = a[j], a[i]

        elif move < 0.55:   # plain random swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                a[i], a[j] = a[j], a[i]

        elif move < 0.8:    # random tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = a[i] * 0.02 * rng.random()
                a[i] -= delta
                a[j] += delta

        else:               # targeted mass transfer based on max‑conv contribution
            conv = _fft_convolve_self(a)
            K = int(np.argmax(conv))

            # contributions a[i] * a[K‑i] (where defined)
            idx = np.arange(n)
            jdx = K - idx
            mask = (jdx >= 0) & (jdx < n)
            cont = np.zeros_like(a)
            cont[mask] = a[mask] * a[jdx[mask]]

            i_max = int(np.argmax(cont))
            j_min = int(np.argmin(a))

            if i_max != j_min and a[i_max] > 0:
                delta = a[i_max] * 0.02
                a[i_max] -= delta
                a[j_min] += delta

        new_val = _eval_fast(a)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA‑targeted] new best = {best_val:.6f} (iter {it})")
        else:
            a[:] = a_old   # revert

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.015, 1e-12)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Greedy refinement (mass moves from biggest contributor)
# ----------------------------------------------------------------------
def _greedy_refine(a: np.ndarray,
                   cur_val: float,
                   deadline: float) -> (np.ndarray, float):
    """
    Deterministic greedy improvement: repeatedly move a fraction of the
    element that contributes most to the current max‑convolution towards
    the smallest element, accepting only strictly decreasing objective.
    """
    n = a.shape[0]
    while time.time() < deadline:
        conv = _fft_convolve_self(a)
        K = int(np.argmax(conv))

        idx = np.arange(n)
        jdx = K - idx
        mask = (jdx >= 0) & (jdx < n)
        cont = np.zeros_like(a)
        cont[mask] = a[mask] * a[jdx[mask]]

        i_max = int(np.argmax(cont))
        j_min = int(np.argmin(a))

        if i_max == j_min or a[i_max] < 1e-12:
            break

        # try decreasing fractions until an improvement is found
        improved = False
        for frac in (0.10, 0.05, 0.02, 0.01):
            delta = a[i_max] * frac
            a[i_max] -= delta
            a[j_min] += delta
            new_val = _eval_fast(a)
            if new_val + 1e-14 < cur_val:
                cur_val = new_val
                improved = True
                print(f"[greedy] new best = {cur_val:.6f}")
                break
            # revert and try smaller fraction
            a[i_max] += delta
            a[j_min] -= delta

        if not improved:
            break   # cannot improve further

    return a, cur_val
