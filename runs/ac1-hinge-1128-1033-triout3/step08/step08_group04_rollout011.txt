"""
Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
1.  Initialise a diverse set of candidates (uniform, previous solution,
    random Dirichlet draws, exponentials, inverse‑sqrt, sinusoid, etc.).
2.  Pick the best cheap candidate (fast evaluation = 2·n·max(conv) with
    Σa = 1).
3.  Refine with a smoothed surrogate of the max‑convolution and L‑BFGS.
    The smoothing parameter ε is decreased logarithmically (≈15 stages)
    and several random restarts are performed.
4.  Exact mirror‑descent on the true max‑convolution (exponentiated‑gradient
    on the simplex) for many cheap iterations.
5.  Targeted mass‑transfer descent: repeatedly move a small amount of mass
    from the index that contributes most to the current peak towards a
    low‑mass index; accept only improving moves.  This directly attacks the
    maximal convolution entry.
6.  A final cheap hill‑climbing (random swaps / mass transfers) to polish.
7.  Return the best vector after a check with the official ``evaluate_sequence``.

The whole routine respects a 1000 s wall‑clock budget, keeping a 5 s safety
margin for the final call to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smooth surrogate (set before each call to
# the objective / gradient).
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may create tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate:
        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k−max)/ε)

    with the soft‑max parametrisation a_i = exp(w_i)/Σ_j exp(w_j).

    Returns the scalar value and the gradient w.r.t. w.
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # soft‑max parametrisation (w → a, Σa = 1)
    w_shift = w - np.max(w)       # improve numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution a*a
    conv = _full_convolution(a)   # length 2·n‑1
    max_c = conv.max()

    # soft‑max over the lag indices
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()   # probability distribution over k

    # convolution of a with reversed p   (needed for ∂max/∂a)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # pick the segment that aligns with the original a‑vector
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]   # see derivation in the original notebook

    # chain rule through the soft‑max (w → a)
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return float(L), grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the O(n) algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already satisfy
    Σa = 1 and a_i ≥ 0.  Returns 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places the largest values on the left side and the
    smallest on the right side.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large → left
        new_a[n - 1 - i] = sorted_vals[i]      # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass‑transfer moves (80 %).  The vector is kept
    on the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                      # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[hill‑climb] new best = {cur_val:.6f}")

    return cur_a, cur_val


def _targeted_descent(a: np.ndarray,
                      best_val: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer that directly attacks the dominant convolution entry.
    At each iteration we locate the index i that contributes most to the
    current maximum and move a small fraction of its mass to a low‑mass index.
    Only improving moves are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        # product a_i * a_{max_k-i} for all i (zero where partner is out of range)
        i_idx = np.arange(n)
        j_idx = max_k - i_idx
        valid = (j_idx >= 0) & (j_idx < n)
        prod = np.zeros_like(cur_a)
        prod[valid] = cur_a[i_idx[valid]] * cur_a[j_idx[valid]]

        i_star = int(np.argmax(prod))
        if cur_a[i_star] <= 1e-15:
            break

        # pick a low‑mass target (among the 15 smallest entries)
        small_idx = np.argpartition(cur_a, 15)[:15]
        j_target = int(rng.choice(small_idx))

        if i_star == j_target:
            continue

        # move a modest fraction of the source mass
        delta = cur_a[i_star] * 0.05 * rng.random()
        new_a = cur_a.copy()
        new_a[i_star] -= delta
        new_a[j_target] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[targeted] new best = {best_val:.6f}")

    return cur_a, cur_val

    # Problem size -------------------------------------------------
    # --------------------------------------------------------------

        has_old = True

    # --------------------------------------------------------------

    # exponential families (left‑leaning and symmetric)
    for lam in np.logspace(-4, 0, 9):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        init_candidates.append(a)

        a_sym = np.exp(-lam * np.abs(np.arange(n) - (n - 1) / 2))
        a_sym /= a_sym.sum()
        init_candidates.append(a_sym)

    # power‑law (negative exponents give heavier tails)
    for p in np.linspace(-2.0, 0.0, 5):
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        init_candidates.append(a)

    a_mix /= a_mix.sum()
    init_candidates.append(a_mix)

    # normalise everything onto the simplex (some generators may have zeros)
    init_candidates = [c / c.sum() for c in init_candidates]

    # --------------------------------------------------------------
    # 2) Pick the best cheap candidate.
    # --------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – L‑BFGS on a smooth surrogate of the max‑convolution.
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-13), num=15)
    w = np.log(best_a + 1e-12)               # start from soft‑max parameters
    max_restarts = 20
    restart = 0

    while time.time() < deadline - 40.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        if restart % 5 == 0:
            w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 40.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n

            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # back‑transform to a probability vector
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric reorder after each restart
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 2 – exact mirror‑descent on the true max‑convolution.
    # --------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.35
    decay = 0.9999
    min_alpha = 0.002
    it = 0
    while time.time() < deadline - 30.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a (average over all maximisers)
        # Here we use the simple version with a single maximiser.
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g = np.zeros_like(a)
        g[idx] = a[jdx]

        # exponential (mirror) update
        alpha = max(alpha0 * (decay ** (it // 500)), min_alpha)
        a = a * np.exp(-alpha * g)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        if it % 200 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[mirror {it}] new best = {best_val:.6f}")

            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[mirror reorder] improved to {best_val:.6f}")

        it += 1

    # --------------------------------------------------------------
    # 5) Phase 3 – targeted greedy descent on the dominant convolution entry.
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        best_a, best_val = _targeted_descent(best_a, best_val, deadline - 5.0)

    # --------------------------------------------------------------
    # 6) Final cheap hill‑climbing refinement.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # 7) Final safety check with the official evaluator and return.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
