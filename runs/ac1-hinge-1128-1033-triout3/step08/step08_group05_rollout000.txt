"""
Propose a step‑function (list of non‑negative heights) that minimizes
``evaluate_sequence``.

Approach
--------
* Start from the best known construction (`height_sequence_1` if it exists).
* Normalise it to a probability simplex (sum = 1).  On this simplex the
  objective reduces to `2·n·max(conv)` where `conv = a * a`.
* Perform a focused local‑search that repeatedly:
    – Computes the convolution and identifies the index `kmax`
      where the max occurs.
    – Looks at the contributions `a_i·a_{kmax-i}` and selects a few
      “heavy” indices (large contributions) and a few “light” indices
      (small values).
    – Tries all swaps between a heavy and a light index and accepts the
      best improving swap.
    – If no improving swap exists, attempts a small mass transfer from
      a heavy index to a light one.
* The search is purely greedy (accept only strict improvements) and
  runs until the time budget (≈1000 s) is exhausted, keeping the best
  vector found.
* Finally the candidate is validated with the official ``evaluate_sequence``
  function and returned as a plain Python list.

The algorithm is deliberately simple and cheap to evaluate – the most
expensive operation is a FFT‑based convolution, costing only a few
milliseconds for n≈4000.  By concentrating the search on the active
indices of the current peak we can explore many promising moves per
second and shave off the tiny improvement needed to reach the target
bound (~1.503).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas, all top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) using an FFT of power‑of‑2 size.
    Result is clipped to non‑negative values to avoid tiny negatives from rounding.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip numerical noise
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already sum to 1.
    Returns 2·n·max(conv) where conv = a*a.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Fine‑grained local search
# ----------------------------------------------------------------------


def _swap_refine(a: np.ndarray,
                 best_val: float,
                 deadline: float,
                 top_k: int = 8,
                 low_k: int = 12) -> tuple[np.ndarray, float]:
    """
    Greedy swap optimisation.
    In each iteration the convolution peak `kmax` is located, the
    `top_k` indices with largest contributions a_i*a_{kmax-i} are
    identified, as well as the `low_k` indices with smallest values.
    All swaps between a heavy and a light index are examined and the
    best improving swap (if any) is applied.  The loop stops when the
    time budget expires.
    """
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_b = conv.max()
        kmax = int(conv.argmax())

        # indices that actually contribute to the peak
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = kmax - idx
        contrib = a[idx] * a[jdx]

        # choose heavy indices (largest contributions)
        cur_top_k = min(top_k, len(contrib))
        if cur_top_k == 0:
            break
        heavy_pos = np.argpartition(-contrib, cur_top_k - 1)[:cur_top_k]
        heavy_idx = idx[heavy_pos]

        # choose light indices (smallest a values globally)
        low_idx = np.argsort(a)[:low_k]

        # examine all candidate swaps
        best_swap_val = best_val
        best_pair = None
        for i in heavy_idx:
            for j in low_idx:
                if i == j:
                    continue
                # swap i and j (sum stays 1, non‑negativity preserved)
                new_a = a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]
                val = _fast_evaluate(new_a)
                if val < best_swap_val - 1e-12:
                    best_swap_val = val
                    best_pair = (i, j)

        if best_pair is not None:
            i, j = best_pair
            a[i], a[j] = a[j], a[i]   # apply the best improving swap
            best_val = best_swap_val
            print(f"[swap] improved to {best_val:.6f} (swap {i}<->{j})")
            # after a successful swap we recompute everything in the next loop
            continue

        # ------------------------------------------------------------------
        # No improving swap found – try a tiny mass transfer from a heavy
        # index to a light one.  The transfer size is a small fraction of the
        # donor mass; it is deliberately tiny to act as a fine‑grained
        # gradient step.
        # ------------------------------------------------------------------
        donor = heavy_idx[np.argmax(a[heavy_idx] * a[kmax - heavy_idx])]
        # pick a recipient among the light indices (different from donor)
        rec_candidates = low_idx[low_idx != donor]
        if rec_candidates.size == 0:
            # fallback: random recipient
            rec_candidates = np.arange(n)
            rec_candidates = rec_candidates[rec_candidates != donor]
        recipient = int(rng.choice(rec_candidates))

        delta = a[donor] * 0.015 * rng.random()   # move up to 1.5 % of donor
        if delta <= 1e-12:
            continue

        new_a = a.copy()
        new_a[donor] -= delta
        new_a[recipient] += delta
        # mass is conserved, no need to re‑normalise
        val = _fast_evaluate(new_a)
        if val < best_val - 1e-12:
            a = new_a
            best_val = val
            print(f"[transfer] improved to {best_val:.6f} "
                  f"(donor {donor}->{recipient}, Δ={delta:.2e})")
            # continue loop with the updated vector
            continue

        # if neither swaps nor transfers improve, perform a cheap random
        # perturbation to try to escape a tiny plateau.
        i_rand, j_rand = rng.integers(0, n, size=2)
        if i_rand != j_rand:
            new_a = a.copy()
            new_a[i_rand], new_a[j_rand] = new_a[j_rand], new_a[i_rand]
            val = _fast_evaluate(new_a)
            if val < best_val - 1e-12:
                a = new_a
                best_val = val
                print(f"[rand swap] improved to {best_val:.6f}")

    return a, best_val

    """
    # Initialise RNG
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Keep a small safety margin for the final call to evaluate_sequence

    # ------------------------------------------------------------------

    try:

    else:
