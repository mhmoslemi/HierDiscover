"""
propose_candidate – search for a non‑negative step function that minimises
    evaluate_sequence(sequence)

Algorithm (≈1000 s time budget)

Stage 0 – initialise a diverse pool (uniform, exponential, power‑law,
         alternating‑extreme ordering).  Keep the best candidate.

Stage 1 – projected Adam‑style sub‑gradient descent on the true
         L∞‑objective.  The active set of convolution indices whose value
         is within a relative tolerance of the current maximum is used to
         form a sub‑gradient.  An Adam‑like adaptive step is applied and
         the iterate is projected onto the probability simplex.

Stage 2 – smooth surrogate optimisation (log‑sum‑exp) with L‑BFGS‑B.
         Several restarts from the current best point are performed, each
         with a decreasing smoothing parameter ε.  After each L‑BFGS run
         the “anti‑symmetric’’ and a new “alternating‑extreme’’ reorder are
         tried.

Stage 3 – simulated‑annealing hill‑climbing.  Three move types are used:
         (i) swap two positions, (ii) transfer a fraction of mass from
         one index to another, (iii) rotate a random contiguous block.
         An exponential cooling schedule accepts occasional uphill moves.

Finally the best vector found is returned as a plain Python list.
All updates keep the vector non‑negative and normalised to sum = 1,
which makes the objective equal to 2·n·max(a*a).
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for L‑BFGS (smooth surrogate)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative floating‑point artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    n = v.shape[0]
    if np.sum(v) == 1 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values on the left and the smallest on the right,
    i.e. pair a large coefficient with a small one when the sequence is reversed.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # large → left side
        res[n - 1 - i] = s[i]     # small → right side
    if n % 2:
        res[half] = s[half]
    return res


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """
    Arrange values by alternating extremes:
    largest → leftmost, second largest → rightmost, third → next left, …
    """
    n = a.shape[0]
    s = np.sort(a)[::-1]          # descending
    res = np.empty_like(a)
    left, right = 0, n - 1
    idx = 0
    while left <= right:
        res[left] = s[idx]
        idx += 1
        left += 1
        if left <= right:
            res[right] = s[idx]
            idx += 1
            right -= 1
    return res


def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the current active set.
    g_i = Σ_{k∈active} a_{k‑i}.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n-2) - i] for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)     (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # Soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # Self‑convolution
    conv = _full_convolution(a)
    max_conv = conv.max()

    # Smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # Gradient w.r.t a
    weights = np.exp(shifted)
    p = weights / weights.sum()                       # distribution over conv indices

    # Convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries give derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # ∂L/∂a_i  (factor 4 accounts for symmetry)

    # Chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.minimize (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule: from 5·10⁻³ down to 10⁻¹² (geometric)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=11)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 5.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 1200,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to a (probability simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # try both re‑orderings
        for reorder_fn, tag in ((_anti_symmetric_reorder, "anti"),
                                 (_alternating_reorder, "alt")):
            a_ord = reorder_fn(a)
            val_ord = evaluate_sequence(a_ord.tolist())
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[{tag} reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val

# Stage 1 – projected Adam‑style sub‑gradient descent on true L∞‑objective
# ----------------------------------------------------------------------
def _adam_subgradient_descent(a_start: np.ndarray, deadline: float):
    """
    Adam‑like mirror descent on max‑convolution.
    Returns best (a, value) found before the deadline.
    """
    a = a_start.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.15                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[adam init] value = {best_val:.6f}")

    it = 0
    rel_tol = 0.05                 # start relaxed active‑set tolerance

    while time.time() < deadline - 5.0:
        it += 1

        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten tolerance gradually
        if it % 20000 == 0:
            rel_tol = max(rel_tol * 0.6, 1e-6)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of the max via active set
        g = _active_set_gradient(a, active)

        # Adam update (additive)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr0 / np.sqrt(it)          # simple decay
        a = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)

        # project back onto the simplex
        a = _project_onto_simplex(a)

        # occasional logging & re‑ordering
        if it % 250 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[adam iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")

        if it % 5000 == 0:
            for reorder_fn, tag in ((_anti_symmetric_reorder, "anti"),
                                     (_alternating_reorder, "alt")):
                a_ord = reorder_fn(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()   # continue from improved ordering
                    print(f"[adam reorder {tag} at {it}] improved to {best_val:.6f}")

    return best_a, best_val

# Stage 3 – simulated‑annealing hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    """
    Random local moves (swap, mass transfer, block rotation) with a
    temperature schedule.  Only strictly better solutions are kept for
    the reported best, but occasional uphill moves are accepted.
    """
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.04                      # initial temperature
    it = 0

    while time.time() < deadline:
        it += 1
        # ---- propose a move -------------------------------------------------
        move_type = rng.random()
        if move_type < 0.2:                         # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif move_type < 0.7:                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            # transfer up to 40 % of source mass
            delta = cur_a[i] * 0.4 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        else:                                        # block rotation
            # choose a random block length (2 … n/8)
            max_len = max(2, n // 8)
            block_len = rng.integers(2, max_len + 1)
            start = rng.integers(0, n - block_len + 1)
            shift = rng.integers(1, block_len)      # rotate by 1 … block_len‑1
            new_a = cur_a.copy()
            block = new_a[start:start + block_len]
            block = np.roll(block, shift)
            new_a[start:start + block_len] = block

        # ---- re‑project onto simplex -----------------------------------------
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        # ---- evaluate ---------------------------------------------------------
        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            accept = True
        else:
            # annealing acceptance probability (geometric cooling)
            T = T0 * (0.9997 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal iter {it}] new best = {best_val_local:.6f}")

        # occasional aggressive re‑ordering
        if it % 6000 == 0:
            for reorder_fn, tag in ((_anti_symmetric_reorder, "anti"),
                                     (_alternating_reorder, "alt")):
                a_ord = reorder_fn(cur_a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val_local:
                    best_val_local = val_ord
                    best_a = a_ord.copy()
                    cur_a = a_ord.copy()
                    cur_val = val_ord
                    print(f"[anneal reorder {tag} at {it}] improved to {best_val_local:.6f}")

    return best_a, best_val_local

def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (dist_to_end+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget (≈1000 s).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep a fairly large n to give enough freedom.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialisation – try to reuse a known construction if present.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)                 # uniform fallback

    a = _project_onto_simplex(a)

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – cheap diverse initial candidates
    # ------------------------------------------------------------------
    # (i) exponential random vectors
    for tr in range(5):
        if time.time() >= deadline - 300:
            break
        cand = rng.exponential(scale=1.0, size=n)
        cand = _project_onto_simplex(cand)
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # (ii) symmetric power‑law shapes
    for p in (0.3, 0.5, 0.7, 1.0, 1.5):
        if time.time() >= deadline - 300:

        cand = _powerlaw_candidate(p, n)
        val = evaluate_sequence(cand.tolist())

            print(f"[power‑law p={p:.1f}] new best = {best_val:.6f}")

    a_alt = _alternating_reorder(best_a)
    val_alt = evaluate_sequence(a_alt.tolist())
    if val_alt < best_val:
        best_val, best_a = val_alt, a_alt.copy()
        print(f"[alternating reorder] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam‑style projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 400:
        a_sgd, val_sgd = _adam_subgradient_descent(best_a, deadline)
        if val_sgd < best_val:
            best_val, best_a = val_sgd, a_sgd.copy()
            print(f"[adam descent] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth surrogate optimisation (L‑BFGS) with restarts
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 200 and restarts < 10:
        restarts += 1
        # perturb current best for a fresh start (log‑space noise)
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.3, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[restart {restarts}] L‑BFGS improved to {best_val:.6f}")

        # try both re‑orderings again
        for reorder_fn, tag in ((_anti_symmetric_reorder, "anti"),
                                 (_alternating_reorder, "alt")):
            a_ord = reorder_fn(best_a)
            val_ord = evaluate_sequence(a_ord.tolist())
            if val_ord < best_val:
                best_val, best_a = val_ord, a_ord.copy()
                print(f"[restart {restarts}] {tag} reorder improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final refinement via simulated‑annealing‑style hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_simulated_anneal(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final re‑ordering sweep (alternating + anti‑symmetric) before return
    # ------------------------------------------------------------------
    for reorder_fn, tag in ((_alternating_reorder, "final alt"),
                             (_anti_symmetric_reorder, "final anti")):
        cand = reorder_fn(best_a)
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[{tag} final] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
