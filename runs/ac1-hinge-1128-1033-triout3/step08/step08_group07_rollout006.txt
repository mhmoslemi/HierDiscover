            coeff = f[k - np.arange(j_min, j_max + 1)]
            rows.extend([r] * length)
            cols.extend(np.arange(j_min, j_max + 1))
            data.extend(coeff.tolist())
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(K, n))

        # linear program

        b_ub = np.full(K, M)
        bounds = [(0, None)] * n
        # solve
        lp_res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not lp_res.success:
            print("[LP] failed, break")
            break
        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 0:
            print("[LP] zero sum")
            break

        # Scale to sum S
        scale = S / sum_b
        g = b * scale

        # mixing
        f_new = (1 - t) * f + t * g
        # compute evaluation for new candidate
        cur_val = evaluate_sequence(f_new.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f_new.copy()
            print(f"[iter {iteration}] new best = {best_val:.6f}")
        # update f for next iteration
        f = f_new

        # optional adapt t: if no improvement for several loops, decrease t
        # Not required.

    # final output
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] eval={final_score:.6f}")

"""
Hybrid convex‑LP / mixing optimizer.

The algorithm follows the classic alternating scheme described in the
challenge statement:

1. Keep a vector `f` (the current step‑function) with ℓ₁‑norm
   S = sqrt(2·n).  Its objective value is simply
   max(f∗f) because evaluate_sequence(f) = 2·n·max/(S²) = max.

2. From `f` we compute the current bound M = max(f∗f).  We then solve a
   linear programme

        maximise   Σ b_j
        subject to b_j ≥ 0
                    (f ∗ b)_k ≤ M   for a selected set of “active’’ shifts k

   The LP tries to pack as much mass as possible into a new candidate
   `b` while never exceeding the current worst convolution value `M`.

3. The LP solution is usually over‑normalised ( Σb > S ).  We rescale
   it to the required norm S, obtaining a vector `g`.  By construction
   f∗g ≤ (S/Σb)·M < M.

4. Replace the current iterate by a convex combination
        f ← (1‑t)·f + t·g    (t≈0.2–0.5)
   which strictly reduces the maximum convolution value.  The process
   is repeated until the time budget expires.

5. As a final polishing step a cheap hill‑climber (random swaps and
   tiny mass transfers) is run for the remaining seconds.

The LP is built only for a small “active’’ set of shifts – the K
largest entries of the current convolution – which keeps each solve
tiny (K·n ≈ 2·10⁶ non‑zero coefficients for n=4000, K≈200).  The
overall routine respects the 1000 s wall‑time limit and returns the
best sequence found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Helper: FFT based self‑convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simple hill‑climber (swap + tiny mass transfer)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       deadline: float,
                       max_no_improve: int = 10000) -> np.ndarray:
    """Very fast local search that accepts any strictly improving move."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = evaluate_sequence(cur_a.tolist())
    rng = np.random.default_rng()
    it = 0
    stagnated = 0

    while time.time() < deadline and stagnated < max_no_improve:
        it += 1
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-12:
                continue
            delta = cur_a[i] * 0.02 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_val = evaluate_sequence(new_a.tolist())
        if new_val + 1e-12 < cur_val:
            cur_a = new_a
            cur_val = new_val
            stagnated = 0
            print(f"[refine] iter {it} new best = {cur_val:.6f}")
        else:
            stagnated += 1

    return cur_a

    except Exception as exc:      # pragma: no cover
