def _simulated_annealing(a, best_val, deadline):
    cur_a = a.copy()
    cur_val = best_val
    T = 0.02
    cooling = 0.99997
    rng = np.random.default_rng()
    while time.time() < deadline:
        # random block length up to 30 maybe
        if rng.random() < 0.7:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] < 1e-12:
                continue
            delta = cur_a[i] * 0.2 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            L = rng.integers(1, min(30, n // 20))
            i = rng.integers(0, n - L)
            j = rng.integers(0, n - L)
            new_a = cur_a.copy()

"""
Propose a non‑negative step function that (approximately) minimises
``evaluate_sequence``.  The algorithm is a multi‑phase optimiser:

1. **Problem size** – if a previous construction is available it is
   up‑sampled to a larger size (≈ 6000) to give the optimiser more
   degrees of freedom.  Otherwise a fresh problem of size 6000 is used.

2. **Diverse seeding** – uniform, a few Dirichlet draws, exponential,
   symmetric exponential, power‑law families and a pure sinusoid are
   generated and normalised to sum to one.  The cheapest evaluator
   ``_fast_evaluate`` (exact max‑convolution for a probability vector)
   picks the best seed.

3. **Smooth surrogate optimisation** – a soft‑max parametrisation
   ``a_i = exp(w_i) / Σ exp(w)`` is optimised with L‑BFGS on the
   smoothed objective

        L_ε(a) = 2 n max(conv) + 2 n ε log Σ_k exp((conv_k−max)/ε),

   for a schedule of decreasing ε (1e‑2 → 1e‑15).  Several restarts
   (randomising ``w``) increase robustness.

4. **Exact sub‑gradient descent** – using the true sub‑gradient of the
   non‑smoothed max‑convolution.  At each iteration the index of the
   maximal convolution entry is found; the gradient is
   ``g_i = 2 a_{k−i}`` (with the centre‑term halved).  A projected
   gradient step on the simplex with a slowly decaying step size is
   performed for many cheap iterations.

5. **Simulated‑annealing refinement** – random single‑index mass moves
   and short block swaps are accepted according to a geometric cooling
   schedule, constantly updating the current best solution.

6. **Final hill‑climbing** – a very cheap local search that only keeps
   strict improvements (random swaps / mass transfers).

All phases respect the total time budget (default 1000 s).  The final
candidate is verified with the official ``evaluate_sequence`` call and
returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (set before each
# optimisation call).
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using FFT of power‑of‑2 size.
    Result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate of the original objective.
    a_i = exp(w_i) / Σ exp(w)  (soft‑max on w)
    Returns (value, gradient w.r.t. w).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # soft‑max parametrisation
    w_shift = w - np.max(w)          # numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution a*a
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # smooth max (log‑sum‑exp)
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()      # soft‑max over lag indices

    # convolution of a with reversed p (required for ∂L/∂a)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # align with indices of a
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]     # sign according to derivation

    # back‑propagation through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return float(L), grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses global ε and n)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    O(n) algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for probability vectors
    (Σ a = 1, a_i ≥ 0).  Returns 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places the largest values on the left side,
    the smallest on the right side – often reduces the peak of the
    self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass‑transfer moves (80 %).
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                      # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine respects the allotted time budget and returns the best
    sequence found as a plain Python list.
    """
    # ------------------------------------------------------------------
    # General initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final official check
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine the problem size.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n_old = old.shape[0]
        has_old = True
    except Exception:
        old = None
        n_old = 0
        has_old = False

    # aim for a larger problem (≈ 6000 entries) – more freedom → lower bound
    target_n = max(6000, n_old)
    n = target_n

    if has_old and n_old != n:
        # linear interpolation to up‑sample the old solution
        x_old = np.linspace(0.0, 1.0, n_old)
        x_new = np.linspace(0.0, 1.0, n)
        old = np.interp(x_new, x_old, old)
    elif has_old:
        old = old.copy()

    # --------------------------------------------------------------
    # 1) Build a diverse set of cheap initial candidates.
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous solution (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # random Dirichlet‑like draws
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # pure sinusoid (non‑negative on [0,π])
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x))

    # exponential families
    for lam in np.logspace(-4, 0, 9):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        init_candidates.append(a)

        a_sym = np.exp(-lam * np.abs(np.arange(n) - (n - 1) / 2))
        a_sym /= a_sym.sum()
        init_candidates.append(a_sym)

    # power‑law (negative exponents = heavier tails)
    for p in np.linspace(-2.0, 0.0, 5):
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        init_candidates.append(a)

    # normalise everything onto the simplex (some generators may have zeros)
    init_candidates = [c / c.sum() for c in init_candidates]

    # --------------------------------------------------------------
    # 2) Pick the best cheap candidate.
    # --------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – L‑BFGS on a smooth surrogate of the max‑convolution.
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-2), np.log10(1e-15), num=20)
    w = np.log(best_a + 1e-12)          # soft‑max parameters

    max_restarts = 10
    restart = 0
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        # occasional random re‑initialisation of w
        if restart % 4 == 0:
            w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n

            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate the candidate obtained from the current w
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric reorder after each restart (helps escape plateaus)
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 2 – exact sub‑gradient descent on the true max‑convolution.
    # --------------------------------------------------------------
    a = best_a.copy()
    it = 0
    # step size schedule: start relatively large and decay slowly
    step0 = 0.8
    decay = 0.99995
    min_step = 0.001

    while time.time() < deadline - 20.0:
        conv = _full_convolution(a)
        k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        # derivative is 2*a_j, except when i==j (centre term) where it is a_i
        g[i] = 2.0 * a[j]
        centre = (i == j)
        if np.any(centre):
            g[i[centre]] = a[j[centre]]

        # projected (Euclidean) gradient step on the simplex
        step = max(step0 * (decay ** (it // 200)), min_step)
        a_new = a - step * g
        a_new = _project_simplex(a_new)

        a = a_new
        it += 1

        if it % 250 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[subgrad {it}] new best = {best_val:.6f}")

            # occasional anti‑symmetric reorder to shake the solution
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[subgrad reorder] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 5) Phase 3 – simulated‑annealing style random mass transfers.
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.03                     # initial temperature
    cooling = 0.99996
    rng = np.random.default_rng(seed + 9999)

    while time.time() < deadline - 5.0:
        if rng.random() < 0.65:          # single‑index mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.15 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:                           # short block swap
            L = rng.integers(1, min(40, n // 10))
            i = rng.integers(0, n - L)
            j = rng.integers(0, n - L)
            new_a = cur_a.copy()
            tmp = new_a[i:i + L].copy()
            new_a[i:i + L] = new_a[j:j + L]
            new_a[j:j + L] = tmp

        # re‑normalise
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # --------------------------------------------------------------
    # 6) Final cheap hill‑climbing refinement.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # 7) Final safety check with the official evaluator and return.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
