"""
step_optimize.py
================
A self‑contained optimiser for the “max‑convolution” objective

    2·n·max(conv(a,a)) / (sum(a)^2)

with the additional restriction a_i ≥ 0.  The optimiser proceeds in three
phases:

1.  A large, diverse parametric sweep (uniform, exponential, power‑law,
    sine‑powers, Beta‑densities, …).  Every candidate is normalised to a
    probability vector and its exact objective (via an FFT convolution) is
    recorded.  The best among them seeds the later optimisation.

2.  Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
    The surrogate is controlled by a temperature ε that decays
    exponentially, giving increasingly sharp gradients near the true
    maximum.  Restarts are performed if progress stalls.

3.  Exact hill‑climbing on the true objective followed by a projected
    sub‑gradient descent with an adaptive step‑size.  Both stages move a
    small amount of mass from indices that most contribute to the current
    peak to indices that have little influence, always re‑projecting onto
    the simplex.

A final tiny convex combination with the uniform distribution is tried
to “smooth away” any residual spikes.

The whole routine respects a user supplied time budget (default 1000 s)
and returns a list of non‑negative floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.special import beta as beta_func

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector that already sums to 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w ≥ 0, Σ w = 1}."""
    if np.isclose(np.sum(v), 1.0) and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of parametric candidates
# ----------------------------------------------------------------------
def _candidate_from_array(arr: np.ndarray) -> np.ndarray:
    """Normalise a raw array to a probability vector (sum=1)."""
    a = np.asarray(arr, dtype=np.float64)
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        raise ValueError("Zero sum candidate")
    return a / s


def _initial_candidate(n: int):
    """Search a large family of parametric shapes and keep the best."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(raw):
        nonlocal best_a, best_val
        try:
            a = _candidate_from_array(raw)
        except Exception:
            return
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Uniform
    # --------------------------------------------------------------
    _try(np.ones(n))

    # --------------------------------------------------------------
    # 2. Exponential families (left‑hand & symmetric)
    # --------------------------------------------------------------
    for r in np.linspace(0.5, 0.99, 12):
        _try(r ** np.arange(n))                     # left‑hand side
        _try(r ** np.arange(n) + r ** (n - 1 - np.arange(n)))  # symmetric

    # --------------------------------------------------------------
    # 3. Power‑law (sum) – controls the weight of the ends
    # --------------------------------------------------------------
    for p in np.linspace(0.1, 4.0, 30):
        i = np.arange(1, n + 1, dtype=np.float64)
        _try(i ** (-p) + (n - i + 1) ** (-p))

    # --------------------------------------------------------------
    # 4. Power‑law (product) – U‑shaped “arcsine‑like”
    # --------------------------------------------------------------
    for p in np.linspace(0.1, 4.0, 30):
        i = np.arange(1, n + 1, dtype=np.float64)
        _try((i * (n - i + 1)) ** (-p))

    # --------------------------------------------------------------
    # 5. Sine‑powers (zero at the ends, centre emphasis)
    # --------------------------------------------------------------
    theta = np.pi * (np.arange(1, n + 1) / (n + 1))
    base = np.sin(theta)
    for p in np.linspace(0.2, 4.0, 20):
        _try(base ** p)

    # --------------------------------------------------------------
    # 6. Symmetric Beta densities
    # --------------------------------------------------------------
    xs = (np.arange(1, n + 1) - 0.5) / n         # points in (0,1)
    alk = np.concatenate([np.linspace(0.2, 1.0, 12),
                          np.linspace(1.0, 5.0, 12)])
    for alpha in alk:
        pdf = xs ** (alpha - 1) * (1 - xs) ** (alpha - 1) / beta_func(alpha, alpha)
        _try(pdf)

    # --------------------------------------------------------------
    # 7. Random Dirichlet samples (quick exploration)
    # --------------------------------------------------------------
    rng = np.random.default_rng(12345)
    for _ in range(25):
        _try(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on the log‑sum‑exp surrogate of the max‑convolution."""
    # logits → simplex via soft‑max
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.45, 0.003
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 80_000          # restart if stuck

    it = 0
    while time.time() < deadline:
        # ---- map logits to the simplex ----
        w_shift = w - np.max(w)               # prevent overflow
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---- schedules (linear in log‑time) ----
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ---- smooth max via log‑sum‑exp ----
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()                     # soft‑max over positions

        # gradient of 2·n·∑ p_k (a*a)_k w.r.t. a is 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                   # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]             # the part that corresponds to a*a
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape flat plateaux
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.008, size=n)

        # restart if too many idle steps
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart (no improve >{IMPROVE_LIMIT})")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val

# Phase B – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Exact hill‑climbing that moves a small fraction of mass."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.7, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – pick proportionally to its contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # target index – favour indices that add little to the peak
        if np.random.rand() < 0.42:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            # Small weight for indices that already contribute heavily
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental linear‑time update of the full convolution
        b_new = b.copy()
        # add contributions a_j * a + a_i * a (both sides)
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # adjust the pure self‑terms (they appear twice in the above)
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturb after stagnation")

        it += 1

    return best_a

# Phase C – Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with an adaptive step‑size."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    # initial step proportional to a typical entry size
    eta = 0.6 / n
    decay = 0.9994
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 180_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of max entry (two symmetric contributions)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # Simple step‑size adaptation: if we improve, keep size,
        # otherwise shrink a little
        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # shake and reset step size
            noise = np.random.normal(scale=0.010, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = 0.6 / n
            no_improve = 0
            print(f"[SGD {it:7d}] perturb after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small post‑processing: mixing a tiny amount of uniform mass
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few convex combinations with the uniform distribution."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020, 0.040):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

    # --------------------------------------------------------------
    # Initialise: use a rich parametric sweep; fall back to uniform
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)       # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.503

    # --------------------------------------------------------------
    # Phase A – Adam on the smooth surrogate
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.28 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        a0, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # --------------------------------------------------------------
    # Phase B – Targeted hill‑climbing on the exact objective
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.72 * (deadline - start_time)
    best_a = _hillclimb_targeted(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )
    best_val = _objective_value(best_a)

    # --------------------------------------------------------------
    # Phase C – Exact sub‑gradient descent (projected onto simplex)
    # --------------------------------------------------------------
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # --------------------------------------------------------------
    # Final polishing – tiny uniform mixing
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # --------------------------------------------------------------
    # Final report (uses the secure wrapper provided by the platform)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
