Implementation details of projection onto simplex: code:

```
def _project_simplex(v):
    # ensure nonnegative projection onto simplex sum=1
    # v is np.ndarray
    # algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

The evaluation function is  

\[
F(a)=\frac{2\,n\; \max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 .
\]

Because the objective is homogeneous we can *fix* the sum of the
coefficients to 1; then  

\[
F(a)=2\,n\;\max (a*a) .
\]

Thus the task reduces to **flattening the self‑convolution** of a
non‑negative vector on the simplex.  
The program below implements a fast continuous optimizer:

1. **Parameterisation** – a vector `a` lives on the probability simplex
   (non‑negative, sum = 1).  
   All updates are performed on the simplex, so the objective is
   scale‑invariant.

2. **Soft‑max smoothing of the ∞‑norm** – instead of using only the
   maximal convolution entry we assign a weight  

   \[
   w_k=\exp\!\bigl(\beta\,(c_k-\max c)\bigr)\in[0,1],
   \qquad c=a*a .
   \]

   For small `β` many entries contribute, for large `β` the weight
   concentrates on the true maximum.  The gradient of
   \(\sum_k w_k\,c_k\) w.r.t. `a` is

   \[
   g_i = 2\sum_k w_k\,a_{k-i},
   \]

   which can be obtained with a single FFT‑based convolution.

3. **Mirror‑descent on the simplex** – the update  

   \[
   a_i \leftarrow a_i\;\exp(-\eta\,g_i)
   \]

   keeps all entries non‑negative.  Afterwards we project onto the
   simplex with the O(n log n) algorithm of Duchi *et al.*.

4. **Annealing schedule** – `β` is increased slowly (soft → hard max)
   while the step size `η` decays as `1/√t`.  This gives a rapid early
   flattening and a fine‑grained final polishing.

5. **Anti‑symmetric re‑ordering** – after every improvement we
   rearrange the vector so that the largest values are placed opposite
   the smallest ones (pairwise).  By the rearrangement inequality this
   reduces the central autocorrelation peak.

6. **Local hill‑climbing** – when the time budget is almost exhausted a
   cheap random‑swap / mass‑transfer routine finishes the optimisation.

The whole loop runs for at most the supplied time budget
(`budget_s` ≈ 1000 s).  All heavy work (gradient & FFT) is performed
without calling `evaluate_sequence`; the true objective is queried only
periodically, ensuring many optimisation steps fit into the budget.

```python
"""
Efficient continuous optimisation for the step‑function objective.

The algorithm works on the simplex (∑a = 1, a_i ≥ 0) and minimises the
maximum entry of the self‑convolution a*a.

Key ideas
---------
* Soft‑max smoothing of the ∞‑norm → a differentiable surrogate.
* Mirror‑descent with multiplicative updates, always staying on the
  simplex thanks to a fast projection.
* Annealing of the smoothing parameter β and of the step‑size η.
* Periodic anti‑symmetric re‑ordering (largest opposite smallest) to
  further lower the autocorrelation peak.
* Final cheap hill‑climbing (random swaps / tiny mass transfers).

The function ``propose_candidate`` follows the required signature and
returns the best list of heights found within the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # eliminate tiny negatives
    return conv


def _gradient_weighted_peaks(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Return g_i = Σ_k w_k * a_{k-i}  (indices out of range give 0).

    This implements the gradient of Σ_k w_k (a*a)[k] w.r.t. a.
    Computed in O(n log n) with a single FFT.
    """
    n = a.shape[0]
    # w has length 2*n-1 (full convolution length)
    p_rev = w[::-1]                          # reverse for convolution
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # middle n entries correspond to Σ a_{k-i}
    seg = conv_ap[n - 1 : 2 * n - 1]
    return seg[::-1]                         # undo the reversal


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1                  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest entries with the smallest ones
    (largest on the left side, smallest on the right side).
    This reordering never harms the objective and often improves it.
    """
    n = a.shape[0]
    s = np.sort(a)                           # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[-1 - i]                   # biggest → left
        out[-1 - i] = s[i]                   # smallest → right
    if n % 2:
        out[half] = s[half]
    return out


def _hill_climb_refine(a: np.ndarray, cur_val: float,
                       deadline: float, n: int):
    """
    Very cheap local search:
    * random swaps of two positions,
    * tiny mass transfers from a larger entry to a smaller one.
    Only strict improvements are kept.
    """
    cur_a = a.copy()
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto simplex
        new_a = _project_onto_simplex(new_a)

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:
            cur_a, cur_val = new_a, val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

                best_val, best_a = cur_val, a.copy()
                print(f"[gd iter {it}] improved = {best_val:.6f} "
                      f"(max conv {M:.8e}, β={beta:.2f})")
                # anti‑symmetric re‑ordering (often helps)
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val, best_a = val_ord, a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder] improved to {best_val:.6f}")

        # ----- anneal β slowly -----
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

        it += 1

    # --------------------------------------------------------------
    # Phase 2 – final hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hill_climb_refine(best_a, best_val,
                                              deadline, n)

    # --------------------------------------------------------------
    # Final evaluation & return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
```
