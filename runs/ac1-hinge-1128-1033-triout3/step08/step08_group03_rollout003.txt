Algorithm (soft‑max mirror descent)

   ``height_sequence_1`` if present, otherwise from a Dirichlet draw.
   The vector is normalised to sum 1 (the objective is scale‑invariant).

2. **Smooth max minimisation** – we minimise a smooth approximation of the
   maximum convolution value via a soft‑max (log‑sum‑exp) surrogate.
   For a current distribution ``a`` we compute
       w_k ∝ exp(β·(c_k − max c)),   c = a * a,

       g = 2·(a_rev * w)   (restricted to the first ``n`` entries).
   A multiplicative (exponential‑gradient) update
       a ← a·exp(−η·g)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negatives that appear due to numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    Returns a vector of length ``len(a)``.
    """
    n = a.shape[0]
    max_c = conv.max()

    # soft‑max weights (stable: subtract max_c)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback – uniform weights (should never happen)
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution a_rev * w  (full length = n + len(w) - 1)
    a_rev = a[::-1]
    conv_len = a_rev.shape[0] + w.shape[0] - 1
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv_aw = np.fft.irfft(fa * fw, size)[:conv_len]

    # extract the middle n entries (indices n‑1 … 2n‑2)
    start = n - 1
    g = 2.0 * conv_aw[start:start + n]   # factor 2 comes from ∂c_k/∂a_i = 2 a_{k‑i}
    return g

def _hill_climb_refine(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """Random swaps and tiny mass transfers, keeping only strict improvements."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement required
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

                best_a = a.copy()
                print(f"[reorder] improved to {best_val:.6f}")

        # occasional random jitter to escape shallow plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.02, size=n) * a
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()

    # --------------------------------------------------------------
    # Phase 2 – hill‑climbing refinement (final polishing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline, n)

    # --------------------------------------------------------------
    # Final evaluation & return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
