def subgrad_descent(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy(); best_val = _objective(a)
    iteration = 0
    while time.time() < deadline:
        # compute conv and max index
        b = _full_convolution(a)
        max_b = b.max()
        val = 2.0 * n * max_b
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Subgrad {iteration}] new best = {best_val}")

        # gradient:
        k_max = int(np.argmax(b))
        # compute start and end indices
        start_idx = max(0, k_max - (n-1))
        end_idx = min(n-1, k_max)
        grad = np.zeros_like(a)
        # reversed slice using a segment:
        # i in [start_idx, end_idx]; a_rev_index = k_max - i
        grad[start_idx:end_idx+1] = 4.0 * n * a[k_max - np.arange(start_idx, end_idx+1)]

        # step size schedule: lr = initial_lr / sqrt(iteration+1)
        # initial_lr can be set based on n; e.g., lr0 = 0.2
        lr = 0.2 / math.sqrt(iteration+1)   # decays
        a = a - lr * grad
        # projection onto simplex
        a = _project_simplex(a)
        # ensure nonnegative small tolerance
        iteration += 1

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1️⃣  Initialise a candidate:
    – if a previous solution ``height_sequence_1`` exists, interpolate it;
    – otherwise try a parametric Beta family, a sine‑power family and a few
       random Dirichlet draws.
2️⃣  **Adam optimisation** on a smooth approximation of the true objective
    (2·n·max(conv(a,a))) with an annealed variance penalty.  The parameters
    are kept on the probability simplex via a soft‑max (log‑its) parametrisation.
3️⃣  **Exact sub‑gradient descent**:  use the true gradient of the max‑convolution
    (a shifted copy of the current vector) with a diminishing step‑size and
    Euclidean projection onto the simplex.
4️⃣  **Hill‑climbing (simulated annealing)** – the same fast O(n) local‑search
    routine as before, moving a small amount of mass from indices that
    heavily contribute to the current maximal convolution entry to indices that
    contribute little.
5️⃣  Return the best sequence found (as a plain Python list).  The final
    ``evaluate_sequence`` call is performed within the reserved time budget.

The search runs for at most ``budget_s`` seconds (default 1000) and respects the
required safety margin for the final evaluation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length = 2·len(x)−1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):          # fallback – uniform
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """Smooth approximation of max(conv(a,a)) and its gradient."""
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Variance of the convolution and its gradient
# ----------------------------------------------------------------------
def _var_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Variance of a*a and its gradient (no extra scaling)."""
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # gradient: (4/m) * cross‑correlation of a with v
    conv_ab = _conv_fft(a, v[::-1])              # length n+m-1 = 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad = (4.0 / m) * segment[::-1]
    return var, grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth max + variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimiser for a smooth surrogate of the true objective."""
    # logits for the soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.001
    eps_start, eps_end = 0.8, 1e-8

    # variance‑penalty schedule
    lam_start = 5.0 * n
    lam_end = 0.0

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 20_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # simplex ↔ logits
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedules
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        lam = lam_start * (1.0 - progress) + lam_end * progress

        # smooth‑max
        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)

        # variance penalty
        var_val, grad_var = _var_and_grad(a)

        # total gradient of the loss (to be minimised)
        grad_a = grad_smooth + lam * grad_var

        # chain rule through soft‑max (gradient w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.018, size=n)

        # restart on stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the true max‑convolution gradient."""
    a = a0.copy()
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[Subgrad {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        # index of the maximal convolution entry
        k_max = int(np.argmax(b))

        # compute the gradient (exact sub‑gradient of max(b))
        start_idx = max(0, k_max - (n - 1))
        end_idx = min(n - 1, k_max)
        idx_range = np.arange(start_idx, end_idx + 1)
        grad = np.zeros_like(a)
        # g_i = 4 * n * a_{k_max - i} for the active slice
        grad_slice = 4.0 * n * a[k_max - idx_range]
        grad[start_idx:end_idx + 1] = grad_slice

        # diminishing step size (≈1/√t) scaled by 1/(2n)
        lr = (1.0 / (2.0 * n)) / math.sqrt(iteration + 1)

        a = a - lr * grad
        a = _project_simplex(a)

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (identical to the previous version
# but with a slightly wider transfer‑fraction schedule)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Fast O(n) local search moving small mass from heavy to light contributors."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule for maximal transfer fraction and temperature
    max_frac_start, max_frac_end = 0.55, 0.001
    temp_start, temp_end = 0.015, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 100_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to this entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (probability ∝ contribution)
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index (biased towards low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback uniform among all except i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.028, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _sin_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a one‑parameter sine‑power family a_i ∝ sin(π·i/(n+1))^p."""
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)
    base = np.sin(np.pi * i / (n + 1))
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 10),
                             np.geomspace(1.0, 5.0, 5)[1:]])

    for p in p_vals:
        a = base ** p
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline_total = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 12 000 entries give a good trade‑off between expressivity
    #                and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previous solution if available.
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweeps + a few random Dirichlet draws
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf

    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    # Beta family sweep
    try:
        shape_a, shape_val = _beta_family_best(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] beta‑family sweep improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # Sine‑power family sweep
    try:
        sin_a, sin_val = _sin_family_best(n)
        if sin_val < best_val:
            best_a, best_val = sin_a, sin_val
            print(f"[init] sine‑family sweep improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] sine‑family sweep failed:", exc)

    # random Dirichlet seeds for diversity
    for _ in range(3):
        rand_a = np.random.dirichlet(np.ones(n))
        val = _objective(rand_a)
        if val < best_val:
            best_a, best_val = rand_a, val
            print(f"[init] random Dirichlet seed improved value={best_val:.6f}")

    # fallback random start if nothing else succeeded
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] random start, value=", best_val)

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈55 % of total budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.55 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient descent (≈15 % of total budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.70 * budget_s
    best_a, best_val = _subgrad_descent(best_a, n,
                                        deadline=phase_b_deadline,
                                        start_time=phase_a_deadline,
                                        best_a=best_a,
                                        best_val=best_val,
                                        target_val=1.503)

    # ------------------------------------------------------------------
    # Phase C – Simulated‑annealing hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline_total,
                                         start_time=phase_b_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
