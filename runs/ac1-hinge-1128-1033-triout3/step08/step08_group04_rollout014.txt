"""
Optimisation of a non‑negative step function for the objective

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (∑a)² .

The algorithm mixes several complementary heuristics:

1. **Diverse seeding** – uniform, previous solution, Dirichlet draws,
   exponential / power‑law families, triangular, sinusoidal shapes.
2. **Smooth surrogate optimisation** – L‑BFGS on a log‑sum‑exp
   smoothed version of the max‑convolution (the usual “soft‑max”).
3. **Monotone mirror descent** – sub‑gradient descent on the true
   max‑convolution, always re‑ordering the vector into a decreasing
   sequence (the shape that empirically minimises the max‑correlation).
4. **Simulated‑annealing block moves** – random mass transfers,
   swaps of short blocks and occasional full reshuffles.
5. **Cheap hill‑climbing** – final local search that only accepts strict
   improvements.

All phases respect the allotted time (default 1000 s).  The best
candidate is finally checked with the official ``evaluate_sequence``
and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate – they are set before
# each minimise() call.
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (useful for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using FFT padding to a power of two.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # tiny negatives can appear due to round‑off – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε(a) where a = softmax(w).
    Returns (objective value, gradient w.r.t. w).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ------ soft‑max parametrisation ------
    w_shift = w - np.max(w)                     # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()                                # a lies on the simplex

    # ------ convolution (full linear) ------
    conv = _full_convolution(a)                 # length 2·n‑1
    max_c = conv.max()

    # ------ smooth max (log‑sum‑exp) ------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ------ gradient w.r.t. a ------
    # soft‑max over lag indices
    weights = np.exp(shifted)
    p = weights / weights.sum()                 # ∑_k p_k = 1

    # convolution of a with reversed p (required by chain rule)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment aligning with a (same derivation as in original code)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                # ∂L/∂a

    # ------ chain rule through soft‑max ------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)                 # ∂L/∂w
    return float(L), grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper for ``scipy.minimize`` (uses the globals above)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    O(n) algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast evaluation for vectors that already satisfy a_i ≥ 0 and sum(a)=1.
    Returns 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic: place largest values on the left, smallest on the right.
    Often flattens the convolution peak.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[-1 - i]              # large → left side
        out[n - 1 - i] = s[i]           # small → right side
    if n % 2:
        out[half] = s[half]
    return out


def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing that only accepts strict improvements.
    Random swaps (20 %) and mass‑transfer moves (80 %).
    The vector stays on the simplex after each move.
    """
    rng = np.random.default_rng()
    n = a.shape[0]
    cur_a = a.copy()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.07 * rng.random()   # up to 7 % transfer
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[hill‑climb] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the allotted time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a few seconds margin for the final safety check
    deadline = start_time + budget_s - 5.0
    target_score = 1.503  # early‑exit if we beat this value

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    # ------------------------------------------------------------------
    try:
        # The previous best (if any) is available as a global variable
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        print(f"[info] using previous length n={n}")
        has_old = True
    except Exception:
        n = 5000
        print(f"[info] falling back to n={n}")
        has_old = False

    # ------------------------------------------------------------------
    # 1) Build a diverse initial population (probability vectors)
    # ------------------------------------------------------------------
    init_pop = []

    # uniform distribution
    init_pop.append(np.full(n, 1.0 / n))

    # previous solution (if any)
    if has_old:
        init_pop.append(old / old.sum())

    # Dirichlet / Gamma draws (different concentration parameters)
    for shape in (0.3, 0.7, 1.5, 3.0):
        init_pop.append(rng.gamma(shape=shape, scale=1.0, size=n))

    # simple sinusoid (shifted to stay non‑negative)
    x = np.linspace(0, np.pi, n)
    init_pop.append(np.sin(x) + 0.05)

    # exponential families
    for lam in np.logspace(-5, -0.5, 7):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        init_pop.append(a)

        a_rev = np.exp(-lam * np.arange(n)[::-1])   # increasing exponential
        a_rev /= a_rev.sum()
        init_pop.append(a_rev)

    # power‑law families
    for p in np.linspace(-4.0, -0.2, 7):
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        init_pop.append(a)

    # triangular shape (peak in the middle)
    tri = np.maximum(0.0, 1.0 - np.abs(np.arange(n) - (n - 1) / 2) * 2 / n)
    tri /= tri.sum()
    init_pop.append(tri)

    # normalise everything (some generators may have tiny negative values)
    init_pop = [np.maximum(c, 0.0) for c in init_pop]
    init_pop = [c / c.sum() for c in init_pop]

    # ------------------------------------------------------------------
    # 2) Keep the best few candidates (cheap evaluation)
    # ------------------------------------------------------------------
    pop_vals = [_fast_evaluate(c) for c in init_pop]
    pop = list(zip(init_pop, pop_vals))
    pop.sort(key=lambda t: t[1])
    top_k = 12
    population = [vec for vec, _ in pop[:top_k]]
    best_a = population[0].copy()
    best_val = _fast_evaluate(best_a)
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 1 – L‑BFGS on a smooth surrogate of the max‑convolution
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-14), num=12)

    for cid, cand in enumerate(population):
        if time.time() >= deadline - 200.0:
            break
        w = np.log(cand + 1e-12)      # start from soft‑max parameters
        for restart in range(4):
            if time.time() >= deadline - 200.0:
                break
            # occasional random re‑initialisation of w
            if restart == 3:
                w = np.random.randn(n) * 2.0

            for eps in eps_schedule:
                if time.time() >= deadline - 200.0:
                    break
                global _GLOBAL_EPS, _GLOBAL_N
                _GLOBAL_EPS = eps
                _GLOBAL_N = n

                res = minimize(_smooth_obj_and_grad_wrapper,
                               w,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': 400,
                                        'ftol': 1e-12,
                                        'gtol': 1e-12,
                                        'disp': False})
                w = res.x

                # back‑transform to a probability vector
                w_shift = w - np.max(w)
                a_new = np.exp(w_shift)
                a_new /= a_new.sum()
                cur = _fast_evaluate(a_new)
                if cur < best_val:
                    best_val, best_a = cur, a_new.copy()
                    print(f"[L‑BFGS cid{cid} eps{eps:.1e}] new best = {best_val:.6f}")
                    if best_val < target_score:
                        print("[early exit] target reached during L‑BFGS")
                        final_score = evaluate_sequence(best_a.tolist())
                        return best_a.tolist()
        # occasional anti‑symmetric reorder for this candidate
        a_ord = _anti_symmetric_reorder(cand)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[reorder L‑BFGS cid{cid}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 2 – monotone mirror‑descent on the true max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    alpha = 0.38          # initial step size (exponential mirror)
    decay = 0.9998
    min_alpha = 0.001
    it = 0

    while time.time() < deadline - 120.0:
        conv = _full_convolution(a)
        k_star = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a (derived from convolution definition)
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k_star - idx
        # each pair appears twice in the linear convolution, factor 2:
        g[idx] = 2.0 * a[jdx]

        # exponential mirror update
        a = a * np.exp(-alpha * g)
        a = np.maximum(a, 0.0)

        # enforce a monotone decreasing shape (largest values leftmost)
        a = np.sort(a)[::-1]

        # re‑project onto the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        # periodic reporting and best‑value update
        if it % 150 == 0:
            cur = _fast_evaluate(a)
            if cur < best_val:
                best_val, best_a = cur, a.copy()
                print(f"[mirror {it}] new best = {best_val:.6f}")
                if best_val < target_score:
                    print("[early exit] target reached during mirror descent")
                    final_score = evaluate_sequence(best_a.tolist())
                    return best_a.tolist()
            # occasional anti‑symmetric reorder (helps escape plateaus)
            if it % 900 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val, best_a = val_ord, a_ord.copy()
                    a = a_ord.copy()
                    print(f"[mirror reorder {it}] improved to {best_val:.6f}")

        # step‑size schedule
        alpha = max(min_alpha, alpha * decay)
        it += 1

    # ------------------------------------------------------------------
    # 5) Phase 3 – simulated‑annealing style random moves
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.03               # starting temperature
    cooling = 0.99993

    while time.time() < deadline - 30.0:
        # decide the move type
        if rng.random() < 0.6:
            # single‑index mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.15 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # swap two short contiguous blocks (length up to 50)
            L = rng.integers(1, min(50, n // 6))
            i = rng.integers(0, n - L)
            j = rng.integers(0, n - L)
            new_a = cur_a.copy()
            tmp = new_a[i:i + L].copy()
            new_a[i:i + L] = new_a[j:j + L]
            new_a[j:j + L] = tmp

        # projection onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_a, cur_val = new_a, new_val
            if new_val < best_val:
                best_val, best_a = new_val, new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")
                if best_val < target_score:
                    print("[early exit] target reached during SA")
                    final_score = evaluate_sequence(best_a.tolist())
                    return best_a.tolist()
        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # 6) Final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # 7) Final official evaluation and return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
