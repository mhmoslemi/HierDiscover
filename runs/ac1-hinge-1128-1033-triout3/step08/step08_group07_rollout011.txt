"""
Hybrid optimizer for the step function minimisation using
linear‑programming updates (the “LP‑mix” algorithm) and final hill‑climbing.

The core idea:
    * From a current non‑negative vector a (ℓ₁‑normalised) compute M = max(a∗a).
    * Solve a linear program maximizing Σ g_j while keeping the convolution
      a∗g pointwise ≤ M.  The feasible set contains a, and the optimiser pushes
      mass towards positions where a is small.
    * Mix a←(1−t)a+t g, renormalise, and repeat.  This monotonically reduces
      the value 2·n·max(a∗a).
    * After a few LP steps a cheap stochastic hill‑climb (swaps and tiny mass
      transfers) refines the solution.
"""

def _build_conv_matrix(a):
    n = a.shape[0]
    L = 2 * n - 1
    A = lil_matrix((L, n), dtype=np.float64)
    # positive shifts (including zero)
    for k in range(n):
        row = k + n - 1
        # columns 0 .. n-k-1
        if n - k > 0:
            A[row, 0:n - k] = a[k:]
    # negative shifts
    for k in range(1, n):

"""
Hybrid step‑function optimizer based on a linear‑programming “mix” step.

The algorithm works with a fixed length *n* (default 4000) and
ℓ₁‑normalises the vector a (so that the evaluation reduces to
    2 n · max(a∗a) ).
The optimisation proceeds in three stages:

1. **Initialisation** – load a previously discovered construction if
   `height_sequence_1` exists, otherwise start from a nearly uniform
   random vector.  A deterministic “ends‑large” reordering is applied
   because large values at the extremes tend to reduce the peak of the
   autocorrelation.

2. **LP‑mix updates** – for the current vector a compute M = max(a∗a).
   Solve the linear programme

        maximise   Σ g_j
        subject to  a∗g ≤ M   (point‑wise)
                    g_j ≥ 0

   The feasible set contains a, therefore the optimiser pushes mass to
   positions where a is small.  The new vector is mixed with the old one
   (`a ← (1‑t)a + t·g`) and ℓ₁‑normalised again.  Repeating this step
   monotonically reduces the objective.  Only a handful of LP solves
   (each ≤ 30 s) are performed, keeping the total runtime comfortably
   below the 1000 s budget.

3. **Final cheap hill‑climbing** – random swaps and tiny mass‑transfer
   moves (Metropolis‑type acceptance) are applied until the deadline.
   This extracts any remaining improvement after the LP phase.

The function returns a plain Python list that can be fed to the provided
`evaluate_sequence` routine.

from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

# Fast convolution (exact, using FFT) – replicates evaluate_sequence
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a (full length 2·n‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)      # clip tiny negatives caused by rounding
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for a vector that is already ℓ₁‑normalised.
    Returns 2·n·max(a∗a).
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    # sum_a is 1 because we keep vectors normalised
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple deterministic reordering: large ‑ small alternating from the ends
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values at the extremes, alternating with the smallest."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


# ----------------------------------------------------------------------
# Build the convolution matrix A such that A @ g == a ∗ g (full convolution)
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """
    Returns a sparse matrix A of shape (2·n‑1, n) with the property
        (A @ g)[k] = (a ∗ g)[k-n+1] .
    All entries are non‑negative.
    """
    n = a.shape[0]
    L = 2 * n - 1
    A = lil_matrix((L, n), dtype=np.float64)

    # non‑negative shifts k = 0,…,n‑1 (including the central entry)
    for k in range(n):
        row = k + n - 1
        length = n - k                     # how many columns are involved
        if length > 0:
            A[row, 0:length] = a[k:]       # values a[k], a[k+1], …

    # negative shifts k = -1,…,-(n‑1)
    for k in range(1, n):
        row = n - 1 - k
        length = n - k
        if length > 0:
            A[row, k:n] = a[:length]        # values a[0], a[1], …

    return A.tocsr()
