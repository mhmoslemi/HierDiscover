"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that yields a
low value of ``evaluate_sequence``.  The algorithm is a heavyweight
meta‑heuristic consisting of

1. **Diverse seeding** – a wide family of parametric candidates
   (uniform, exponential, power‑law, mixed exponential/power‑law,
   Dirichlet draws, sinusoid).  The cheapest candidate is kept.

2. **Low‑dimensional grid optimisation** – the best seed is refined by
   many L‑BFGS runs on a smooth log‑sum‑exp surrogate of the maximum
   convolution.  The smoothing parameter ε is driven from 5·10⁻³ down to
   10⁻²⁰, and we allow up to 30 restarts with 20 ε‑stages each.

3. **Mirror‑descent on the exact max‑convolution** – an exponential‑map
   update using the sub‑gradient of the (currently) active lag.  The step
   size is decayed slowly and the iteration runs until time runs out.

4. **Simulated‑annealing with block moves** – random mass transfers,
   large‑block swaps and a *quantisation* move (rounding values to a
   coarser grid).  Acceptance follows a geometric cooling schedule.

5. **Final hill‑climb** – very cheap random swaps / mass transfers that
   only keep strict improvements.

All stages respect the overall 1000 s budget; the best sequence found is
checked with the official ``evaluate_sequence`` before being returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smooth surrogate.
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size (length of the sequence)


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a performed via FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # tiny negative artefacts are removed – they never help the max
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate of the true objective:
        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k – max)/ε)

    The decision variables are the soft‑max parameters w (a_i = exp(w_i)/Σexp).

    Returns (value, gradient w.r.t. w).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ---- soft‑max parameterisation (a lives on the simplex) ----
    w_shift = w - np.max(w)               # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---- convolution and max ----
    conv = _full_convolution(a)            # length 2·n‑1
    max_c = conv.max()

    # ---- smooth max via log‑sum‑exp ----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)           # log Σ exp(...)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (see derivation in the previous code) ----
    weights = np.exp(shifted)
    p = weights / weights.sum()            # soft‑max over the lag indices

    # convolution of a with reversed p  (needed for ∂L/∂a)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the segment that lines up with a
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]           # chain rule will flip again

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return float(L), grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Thin wrapper for scipy.optimize."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (Σ v_i = 1, v_i ≥ 0).
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Cheap version of ``evaluate_sequence`` for vectors with Σ a = 1.
    Returns 2·n·max(conv) (the denominator is 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that puts the largest values on the left side and the smallest
    on the right side (a monotone‑decreasing arrangement).
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing that only accepts strict improvements.
    The move set is 20 % pure swaps, 80 % tiny mass transfers.
    """
    cur_a = a.copy()
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:        # a pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                         # a mass‑transfer move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.07 * rng.random()   # ≤ 7 %
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # keep on the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[hill‑climb] improved to {cur_val:.6f}")

    return cur_a, cur_val

    Main entry point used by the judge.  Returns a Python ``list`` containing
    the best step‑function found before the time budget expires.

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

        has_old = False

    # ------------------------------------------------------------------

    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoid (positive offset)
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.05)

    # ------------------------------------------------------------------
    # Parametric families (exponential, power‑law, mixed)
    # ------------------------------------------------------------------
    idx_range = np.arange(n, dtype=float)

    # exponential decay
    for lam in np.logspace(-4, 1.0, 12):
        a = np.exp(-lam * idx_range)
        a /= a.sum()
        init_candidates.append(a)

        # symmetric exponential about the centre
        a_sym = np.exp(-lam * np.abs(idx_range - (n - 1) / 2.0))
        a_sym /= a_sym.sum()
        init_candidates.append(a_sym)

    # power‑law (negative exponents)
    for p in np.linspace(-2.5, -0.1, 6):
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        init_candidates.append(a)

    # mixed exponential / power‑law (three random draws)
    for _ in range(8):
        lam = 10 ** rng.uniform(-4, 1)
        p = rng.uniform(-2.5, -0.1)
        a = (np.exp(-lam * idx_range) *
             (np.arange(1, n + 1, dtype=float) ** p))
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            continue
        a /= a_sum
        init_candidates.append(a)

    # Normalise everything onto the simplex (guards against accidental zeros)
    init_candidates = [c / c.sum() for c in init_candidates]

    # ------------------------------------------------------------------
    # 2) Pick the best cheap candidate (according to the fast evaluator).
    # ------------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for i, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[seed {i:02d}] fast value = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[seed] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 1 – L‑BFGS on a smooth surrogate.
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-20), num=20)
    w = np.log(best_a + 1e-14)          # start point for the soft‑max parameters

    max_restarts = 30
    restart = 0
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}/{max_restarts}]")
        # every few restarts we inject a fresh random start
        if restart % 7 == 0:
            w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 1500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            # back‑project to the simplex
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best (L‑BFGS) = {best_val:.6f}")

        # try the simple anti‑symmetric reordering after each restart

        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 2 – exact mirror‑descent on the true max‑convolution.
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    alpha0 = 0.42
    decay = 0.99995
    min_alpha = 0.0015
    while time.time() < deadline - 20.0:
        conv = _full_convolution(a)
        # zero‑lag is always the active index – get the gradient analytically
        # grad of max_b = sum_i a_i^2  ⇒  ∂ / ∂a_i = 2 a_i
        # the factor 2·n from the objective multiplies this to 4·n·a_i.

        s = a.sum()

        else:

        if it % 250 == 0:

            if cur_val < best_val - 1e-12:

                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[mirror reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 5) Phase 3 – Simulated annealing with block moves + quantisation.
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.04
    cooling = 0.99990
    rng = np.random.default_rng(seed + 999999)

    while time.time() < deadline - 8.0:
        # decide move type
        r = rng.random()
        if r < 0.55:                     # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.15 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        elif r < 0.85:                   # block swap
            L = rng.integers(2, min(120, n // 6))
            i = rng.integers(0, n - L)
            j = rng.integers(0, n - L)
            if i == j:
                continue
            new_a = cur_a.copy()
            tmp = new_a[i:i + L].copy()
            new_a[i:i + L] = new_a[j:j + L]
            new_a[j:j + L] = tmp
        else:                           # quantisation move – round to a coarser grid
            gran = 10 ** rng.integers(2, 5)   # 100 .. 10 000
            new_a = np.round(cur_a * gran) / gran
            # keep the probability simplex
            new_a = np.maximum(new_a, 0.0)
            s = new_a.sum()
            if s < 1e-12:
                continue
            new_a /= s

        # project onto the simplex (tiny negative numbers may appear after rounding)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # 6) Final cheap hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # 7) Final check with the official evaluator and return.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
