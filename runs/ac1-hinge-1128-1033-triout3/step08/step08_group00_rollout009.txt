"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------

1️⃣ **Low‑dimensional linear programme** (n≈512)  
   Starting from a random Dirichlet vector `f`, we repeatedly
   * compute M = max(conv(f,f))  
   * solve the LP   

        maximise   Σ g_j                     (total mass)
        subject to conv(f,g) ≤ M   (every entry of the convolution ≤ M)
                   g_j ≥ 0

   The solution `g` is scaled to the same ℓ₁‑norm as `f` and mixed:
   `f ← (1‑τ)·f + τ·g` with a slowly decreasing step‑size `τ`.
   Each LP reduces the maximal autocorrelation and we keep the best
   vector found.

2️⃣ **Upscaling** – the best low‑dimensional shape is interpolated to a
   larger size (n≈8000) and re‑normalised to a simplex.

3️⃣ **Hill‑climbing (simulated annealing)** – a fast O(n) local search
   moves a small amount of mass from an index that contributes heavily
   to the current maximal convolution entry to an index that contributes
   little.  Temperature and maximal transfer fraction are annealed to
   zero.  This phase refines the large‑scale solution.

The whole routine respects a 1000 s budget (0.5 s safety margin for the
final ``evaluate_sequence`` call).  The returned list is a plain Python
list whose sum is 1; by construction all entries are non‑negative.
"""

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
# Convolution helpers (full linear convolution using FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (linear convolution)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # fall back to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Build a sparse convolution matrix A such that A @ g = conv(f,g)
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return a (2n‑1)×n sparse matrix representing convolution with f."""
    n = f.shape[0]
    rows = 2 * n - 1
    data = []
    row_ind = []
    col_ind = []
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for each i in [i_start, i_end] we have term f[i] * g[k-i]
        for i in range(i_start, i_end + 1):
            row_ind.append(k)
            col_ind.append(k - i)            # index of g
            data.append(f[i])
    A = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    return A


# ----------------------------------------------------------------------
# Linear‑programming step: given f and its max convolution M,
# find a g that respects conv(f,g) ≤ M and has maximal ℓ₁‑mass.
# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float, warm_start: np.ndarray = None) -> np.ndarray | None:
    """Solve the LP described in the docstring. Returns the optimal g or None."""
    n = f.shape[0]
    A = _conv_matrix(f)
    g = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        g.value = warm_start
    # element‑wise bound A@g ≤ M
    constraints = [A @ g <= M * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver='ECOS', warm_start=True, max_iters=2000, verbose=False)
    except cp.error.SolverError:
        return None
    if g.value is None:
        return None
    g_opt = np.maximum(g.value, 0.0)
    return g_opt


# ----------------------------------------------------------------------
# Hill‑climbing local search (identical in spirit to the previous version)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                best_val: float) -> tuple[np.ndarray, float]:
    """Fast O(n) simulated‑annealing hill‑climber."""
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules
    max_frac_start, max_frac_end = 0.55, 0.001
    temp_start, temp_end = 0.015, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 100_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to this entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # pick source i (prob. proportional to contribution)
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target j (biased towards low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        elif T > 0:
            prob = np.exp(-(new_val - cur_val) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb to escape stagnation
            a = _project_simplex(a + np.random.normal(scale=0.028, size=n))
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Phase 0 – low‑dimensional LP optimisation (≈20 % of time budget)
    # ------------------------------------------------------------------
    n_small = 512                       # small enough for a dense LP
    # initialise with a random Dirichlet vector
    f = np.random.dirichlet(np.ones(n_small))
    best_f = f.copy()
    best_val_small = _objective(best_f)
    print(f"[Init] small problem n={n_small}, value={best_val_small:.6f}")

    phase0_deadline = start_time + 0.20 * budget_s
    prev_g = None
    it_small = 0
    while True:
        now = time.time()
        if now >= phase0_deadline:
            break
        # current maximal convolution entry M
        conv_f = _full_convolution(f)
        M = conv_f.max()
        # solve LP to obtain g
        g_opt = _solve_lp(f, M, warm_start=prev_g)
        if g_opt is None or g_opt.sum() == 0:
            # fallback: random perturbation
            f = np.random.dirichlet(np.ones(n_small))
            prev_g = None
            continue
        prev_g = g_opt.copy()

        # scale g to unit ℓ₁‑norm (same as f) and mix
        g_scaled = g_opt / g_opt.sum()
        # decreasing mixing weight τ
        progress = (now - start_time) / (deadline - start_time)
        tau = 0.25 * (1.0 - progress) + 0.02 * progress
        f = (1.0 - tau) * f + tau * g_scaled
        f = _project_simplex(f)            # enforce exact sum=1

        # evaluate
        val = _objective(f)
        if val < best_val_small:
            best_val_small = val
            best_f = f.copy()
            print(f"[LP {it_small:5d}] new best small‑dim value = {best_val_small:.6f}")
        it_small += 1

    print(f"[LP] finished, best small‑dim value = {best_val_small:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – up‑sample to a larger problem (≈5 % of budget)
    # ------------------------------------------------------------------
    n_large = 8000
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    a = np.interp(x_large, x_small, best_f)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val
    print(f"[Upsample] n={n_large}, starting value = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – hill‑climbing refinement (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _hill_climb(best_a,
                                   deadline=deadline,
                                   start_time=time.time(),
                                   best_val=best_val)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
