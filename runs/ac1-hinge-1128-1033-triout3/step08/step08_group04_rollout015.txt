"""
Propose a non‑negative step function that (hopefully) beats the previous
best value ≈ 1.5106.  

The optimiser works in four complementary phases, all respecting a 1000 s
budget:

1. **Diverse seeding** – many cheap candidates (uniform, previous best,
   Dirichlet draws with several concentration parameters, exponential,
   power‑law families and a sparse “Sidon‑like” support).  The best by the
   cheap fast evaluator is kept.

2. **Smoothed max‑convolution via L‑BFGS** – we minimise a smooth,
   log‑sum‑exp surrogate of the max‑convolution.  A long ε‑schedule
   (15 → 30 values) and many restarts (≈ 30) are performed; after each
   ε‑stage the candidate is projected onto the simplex and evaluated.

3. **Exact mirror‑descent on the true max‑convolution** – a sub‑gradient
   step (exponential update) is applied many thousands of times, with an
   adaptive step‑size schedule and occasional anti‑symmetric re‑ordering.

4. **Aggressive simulated annealing + final hill‑climbing** – random
   mass‑transfer moves (single index or block swaps) are accepted with a
   geometric cooling schedule; the best solution is polished by a cheap
   strict‑improvement hill‑climber.

All phases update a global best vector `best_a`.  At the end the official
`evaluate_sequence` is called for a safety check and the optimal sequence
is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smoothed surrogate (updated before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate of 2·n·max(conv):
        Lε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    a_i = exp(w_i) / Σ_j exp(w_j) (soft‑max parametrisation).

    Returns (objective value, gradient w.r.t. w).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max mapping -----
    w_shift = w - np.max(w)          # for stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- convolution -----
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max (log‑sum‑exp) -----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    # p is the soft‑max over convolution entries
    weights = np.exp(shifted)
    p = weights / weights.sum()      # shape (2·n‑1,)

    # convolution of a with reversed p (necessary for chain rule)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # pick the segment that aligns with the original a‑vector
    seg = conv_ap[n - 1: 2 * n - 1]
    # factor 4·n comes from differentiating 2·n·max and the smoothing term
    grad_a = 4.0 * n * seg[::-1]

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return float(L), grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by scipy.optimize.minimize."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors already
    normalised (a_i ≥ 0, Σ a_i = 1).  Returns 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that interleaves the largest values on the left side and the
    smallest on the right side; often reduces the peak of the convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector is kept on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                      # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
