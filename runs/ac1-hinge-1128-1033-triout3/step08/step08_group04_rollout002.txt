"""
Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
1. Initialise a diverse set of probability vectors (uniform,
   the previous best if available, a few random Dirichlet draws and
   simple exponential families). Keep the best according to the cheap
   ``_fast_evaluate`` (which is exactly ``2·n·max(conv)`` for a
   normalised vector).

2. Refine the best candidate with a long‑running Adam optimiser on a
   smooth surrogate of the true objective:
      L_ε(a)=2·n·max(conv)+2·n·ε·log∑_k exp((conv_k−max)/ε)
   The surrogate approximates the max‑convolution; we start with a
   relatively large ε (≈1e‑2) and halve it every 5000 iterations,
   ending near 1e‑13.  Adam’s per‑coordinate step size and momentum help
   to escape shallow plateaus.

3. Occasionally inject a small random perturbation of the logits to
   avoid getting stuck in a local optimum.

4. After the Adam phase, apply a cheap hill‑climbing routine that
   performs random swaps and tiny mass‑transfer moves, always keeping
   the vector on the simplex and accepting only strict improvements.

5. Finally verify the best vector with the official ``evaluate_sequence``
   and return it as a plain Python list.

All phases respect the 1000 s budget.  The code is deliberately
self‑contained and uses only NumPy and the standard library.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smoothed surrogate (set before each call).
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Result length is 2·len(a)−1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Remove tiny negatives caused by floating‑point noise.
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate of the evaluation function.
    Parameters
    ----------
    w : np.ndarray
        Log‑its (un‑normalised) of the probability vector.
    Returns
    -------
    L, grad_w : float, np.ndarray
        Smoothed loss and its gradient w.r.t. the logits w.
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # --- soft‑max to obtain a probability vector ---
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # --- convolution a*a ---
    conv = _full_convolution(a)      # length 2·n−1
    max_c = conv.max()

    # --- smooth max via log‑sum‑exp ---
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # --- gradient w.r.t. a (see derivation in the original code) ---
    weights = np.exp(shifted)        # ∝ exp((conv−max)/ε)
    p = weights / weights.sum()      # soft‑max over lag indices

    # convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]   # aligns with a‑vector
    grad_a = 4.0 * n * seg[::-1]     # reverse to match ordering of a_i

    # --- chain rule through the soft‑max (log‑its) ---
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return float(L), grad_w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already
    satisfy a_i ≥ 0 and Σ a_i = 1.
    Returns 2·n·max(conv) (the denominator of the official function is 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place large values on the left side, small values on the right."""
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector is kept
    on the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                      # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            # Uncomment the line below for verbose debugging.
            # print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (plain Python list) found before the time
    budget expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final official evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine problem size (reuse previous solution if present).
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # --------------------------------------------------------------
    # 1) Build a cheap set of initial candidates.
    # --------------------------------------------------------------
    init_candidates = []

    # uniform candidate
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction, if any
    if has_old:
        init_candidates.append(old / old.sum())

    # two Dirichlet‑like random draws
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoid (shifted upwards to avoid zeros)
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # exponential families (left‑leaning and symmetric)
    for lam in np.logspace(-4, 0, 7):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        init_candidates.append(a)

        a_sym = np.exp(-lam * np.abs(np.arange(n) - (n - 1) / 2))
        a_sym /= a_sym.sum()
        init_candidates.append(a_sym)

    # power‑law family
    for p in np.linspace(-2.0, 0.0, 5):
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        init_candidates.append(a)

    # Normalise (some generators may have exact zeros)
    init_candidates = [c / c.sum() for c in init_candidates]

    # --------------------------------------------------------------
    # 2) Choose the best cheap candidate.
    # --------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        # print(f"[init {idx}] value = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    # print(f"[init] best value = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Adam optimisation on a smooth surrogate of the max‑convolution.
    # --------------------------------------------------------------
    # Convert the best probability vector into logits.
    w = np.log(best_a + 1e-16)

    # Adam hyper‑parameters
    lr = 0.12
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    it = 0
    while time.time() < deadline - 20.0:
        # schedule for the smoothing parameter ε:
        # start at 1e‑2, halve every 5000 iterations, never go below 1e‑13.
        eps = max(1e-13, 1e-2 * (0.5 ** (it // 5000)))
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        loss, grad = _smooth_obj_and_grad(w)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # Parameter update
        w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Occasionally add a tiny random jitter to avoid stagnation.
        if it % 20000 == 0 and it > 0:
            w += rng.normal(scale=0.02, size=n)

        # Evaluate the corresponding probability vector every few steps.
        if it % 500 == 0:
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _fast_evaluate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                # print(f"[Adam it {it}] new best = {best_val:.6f}")

        it += 1

    # --------------------------------------------------------------
    # 4) Optional anti‑symmetric reorder after the Adam phase.
    # --------------------------------------------------------------
    a_ord = _anti_symmetric_reorder(best_a)
    val_ord = _fast_evaluate(a_ord)
    if val_ord < best_val:
        best_val = val_ord
        best_a = a_ord.copy()
        # print(f"[reorder] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 5) Final cheap hill‑climbing refinement.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # 6) Final safety check with the official evaluator and return.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        # print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        # print("[final] evaluate_sequence raised an exception:", exc)
        pass

    return best_a.tolist()
