"""
Optimization of step function sequence to minimize evaluate_sequence.
Algorithm:
1. Initialise with the known best sequence (height_sequence_1) or uniform.
2. Frank‑Wolfe style iterations:
   - compute full self‑convolution, find active set,
   - sub‑gradient (g) via FFT,
   - locate index with smallest sub‑gradient,
   - line‑search over a small set of step‑sizes γ using the analytic
     expression of the convolution after moving mass to that index,
   - update the sequence and repeat.
3. Permutation hill‑climbing:
   - keep the multiset of coefficients unchanged,
   - propose random swaps,
   - evaluate the new max‑convolution by applying the exact delta
     formula (O(n) update),
   - accept improvements or occasional uphill moves with an annealing
     schedule.
4. Return the best vector found.
All time is bounded by the supplied budget (default 1000 s).
"""

import time
import numpy as np

# -------------------- basic helpers --------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negative artefacts due to round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at the current active set.
    Returns g_i = Σ_{k∈active} a_{k‑i}.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Compute the value returned by evaluate_sequence when sum(a)=1."""
    return 2.0 * n * conv.max()


def _objective(a: np.ndarray) -> float:
    """Convenient wrapper – assumes a is already normalised to sum = 1."""
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.shape[0])


# -------------------- Frank‑Wolfe step --------------------
def _frank_wolfe_step(a: np.ndarray, conv: np.ndarray, n: int,
                     rel_tol: float = 0.01,
                     gamma_grid=None):
    """
    Perform one Frank‑Wolfe inspired move.
    Returns (new_a, new_conv, new_obj) – or (None, None, None) if no improving move.
    """
    max_conv = conv.max()
    # active set: indices whose value is within rel_tol of the maximum
    active = np.where(conv >= max_conv * (1.0 - rel_tol))[0]

    # sub‑gradient restricted to the active set
    g = _active_set_gradient(a, active)

    # choose the index with the smallest sub‑gradient (most promising direction)
    i_star = int(np.argmin(g))

    # build the shifted copy of a that appears in the quadratic formula
    # shift_i[k] = a_{k‑i} (zero outside the original range)
    shift_i = np.concatenate([np.zeros(i_star), a, np.zeros(2 * n - 1 - (i_star + n))])

    # simple grid of step sizes – we keep it short to stay within the time budget
    if gamma_grid is None:
        gamma_grid = np.array([0.0, 0.01, 0.05, 0.1, 0.2, 0.3, 0.5])

    best_gamma = 0.0
    best_obj = 2.0 * n * max_conv   # current value
    best_conv = conv
    for gamma in gamma_grid:
        if gamma == 0.0:
            continue
        # analytic expression for the new convolution after the move:
        #   a' = (1‑γ)·a + γ·e_i
        #   conv' = (1‑γ)²·conv + 2γ(1‑γ)·shift_i + γ²·spike_i
        term1 = (1.0 - gamma) ** 2 * conv
        term2 = 2.0 * gamma * (1.0 - gamma) * shift_i
        term3 = np.zeros_like(conv)
        term3[2 * i_star] = gamma ** 2
        new_conv = term1 + term2 + term3
        obj = 2.0 * n * new_conv.max()
        if obj < best_obj - 1e-12:   # a strict improvement
            best_obj = obj
            best_gamma = gamma
            best_conv = new_conv

    if best_gamma == 0.0:
        # no improving step found
        return None, None, None

    # construct the new vector
    new_a = a * (1.0 - best_gamma)
    new_a[i_star] += best_gamma
    # renormalise to avoid tiny numerical drift
    new_a /= new_a.sum()

    return new_a, best_conv, best_obj


# -------------------- permutation hill‑climbing --------------------
def _swap_delta_conv(conv: np.ndarray, a: np.ndarray, i: int, j: int):
    """
    Return the convolution after swapping a[i] and a[j].
    The update is performed analytically, O(n), without a full FFT.
    """
    if i == j:
        return conv.copy()
    delta = a[j] - a[i]               # amount that moves from i to j (negative if opposite)
    if np.abs(delta) < 1e-18:
        return conv.copy()

    n = a.shape[0]
    new_conv = conv.copy()
    # contribution of the two shifted copies of a
    new_conv[i:i + n] += 2.0 * delta * a
    new_conv[j:j + n] -= 2.0 * delta * a
    # spikes generated by the two unit vectors
    new_conv[2 * i] += delta ** 2
    new_conv[2 * j] += delta ** 2
    new_conv[i + j] -= 2.0 * delta ** 2
    # eliminate possible tiny negative rounding errors
    np.maximum(new_conv, 0.0, out=new_conv)
    return new_conv


def _mass_transfer_delta_conv(conv: np.ndarray, a: np.ndarray, src: int, dst: int, amount: float):
    """
    Transfer ``amount`` from a[src] to a[dst] (src!=dst, amount≤a[src]).
    Return the new convolution array.
    """
    if amount <= 0 or src == dst:
        return conv.copy()
    n = a.shape[0]
    new_conv = conv.copy()
    # linear part: 2·amount·(shift_dst – shift_src)
    new_conv[src:src + n] -= 2.0 * amount * a
    new_conv[dst:dst + n] += 2.0 * amount * a
    # quadratic part: amount²·(spike_src + spike_dst – 2·spike_{src+dst})
    new_conv[2 * src] += amount ** 2
    new_conv[2 * dst] += amount ** 2
    new_conv[src + dst] -= 2.0 * amount ** 2
    np.maximum(new_conv, 0.0, out=new_conv)
    return new_conv


def _permutation_hill_climb(a_init: np.ndarray, deadline: float):
    """
    Random walk that only permutes the multiset of coefficients.
    Uses the fast delta‑convolution update; an annealing schedule
    accepts occasional uphill moves.

    best_a = a.copy()
    best_obj = cur_obj

        else:

            accept = np.exp(-(new_obj - cur_obj) / max(T, 1e-12)) > rng.random()

        if accept:
            # commit the swap
            a[i], a[j] = a[j], a[i]
            conv = new_conv
            cur_obj = new_obj
            if new_obj < best_obj - 1e-12:
                best_obj = new_obj
                best_a = a.copy()
                print(f"[permute] iter {it} new best = {best_obj:.6f}")

        # occasionally try a small mass‑transfer move (helps to break plateaus)
        if it % 500 == 0:
            src = rng.integers(0, n)
            if a[src] > 1e-15:
                dst = rng.integers(0, n)
                if src != dst:

                    new_conv = _mass_transfer_delta_conv(conv, a, src, dst, amount)

                            best_a = a.copy()
                            print(f"[permute] iter {it} (mass) new best = {best_obj:.6f}")

"""
Optimization of a step‑function sequence to minimise
``evaluate_sequence``.  The algorithm works in two phases:

1. **Frank‑Wolfe style moves** – at each iteration we
   * compute the self‑convolution,
   * build a sub‑gradient from the active (near‑maximum) indices,
   * pick the index with the smallest sub‑gradient,
   * perform a line‑search over a small set of step sizes γ using the
     closed‑form expression for the convolution after moving mass to that
     index, and update the probability vector.
   This rapidly reduces the L∞‑norm of the convolution.

2. **Permutation‑only hill‑climbing** – keeping the multiset of
   coefficients unchanged we explore better permutations.
   Swaps are evaluated by an O(n) exact update of the convolution
   (no FFT is needed).  An annealing schedule occasionally accepts
   uphill moves, and small mass‑transfer moves are also tried.

The whole search stops after the allotted time (default 1000 s).  The
final vector is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # kill tiny negatives
    return conv


def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient g_i = Σ_{k∈active} a_{k‑i}.
    ``active_idx`` contains convolution indices (0 … 2n‑2) that are
    currently active (near the maximum).
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective when the vector is already normalised (sum = 1)."""
    return 2.0 * n * conv.max()

    """Wrapper – normalises ``a`` and returns the value of ``evaluate_sequence``."""
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.shape[0])

# Frank‑Wolfe style move
# ----------------------------------------------------------------------
def _frank_wolfe_step(a: np.ndarray, conv: np.ndarray, n: int,
                     rel_tol: float = 0.02,
                     gamma_grid: np.ndarray = None):
    """
    Perform a single Frank‑Wolfe inspired update.
    Returns (new_a, new_conv, new_obj) or (None, None, None) if no
    improving move was found.
    """
    max_conv = conv.max()
    active = np.where(conv >= max_conv * (1.0 - rel_tol))[0]

    # sub‑gradient restricted to the active set
    g = _active_set_gradient(a, active)

    # index with smallest sub‑gradient (most promising direction)
    i_star = int(np.argmin(g))

    # shifted copy of ``a`` that appears in the quadratic formula
    shift_i = np.concatenate([np.zeros(i_star), a,
                              np.zeros(2 * n - 1 - (i_star + n))])

    if gamma_grid is None:
        gamma_grid = np.array([0.0, 0.01, 0.05, 0.1, 0.2, 0.3, 0.5])

    best_gamma = 0.0
    best_obj = 2.0 * n * max_conv
    best_conv = conv

    for gamma in gamma_grid:
        if gamma == 0.0:
            continue
        term1 = (1.0 - gamma) ** 2 * conv
        term2 = 2.0 * gamma * (1.0 - gamma) * shift_i
        term3 = np.zeros_like(conv)
        term3[2 * i_star] = gamma ** 2
        new_conv = term1 + term2 + term3
        obj = 2.0 * n * new_conv.max()
        if obj < best_obj - 1e-13:          # strict improvement
            best_obj = obj
            best_gamma = gamma
            best_conv = new_conv

    if best_gamma == 0.0:
        return None, None, None

    # construct the new probability vector
    new_a = a * (1.0 - best_gamma)
    new_a[i_star] += best_gamma
    new_a /= new_a.sum()                     # guard against rounding drift
    return new_a, best_conv, best_obj
