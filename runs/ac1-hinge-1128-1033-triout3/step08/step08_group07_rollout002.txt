"""
Hybrid refinement for the step‑function max‑convolution minimisation.

The optimiser works in four stages (all respecting the total time budget):

1️⃣  Random warm‑starts + L‑BFGS on a smooth log‑sum‑exp surrogate of the
    max‑convolution.  Several decreasing ε‑values are used; after each
    ε‑stage the current best sequence is optionally reordered with two
    heuristics that push large values towards the ends.

2️⃣  Projected Adam optimiser that follows the true (non‑smooth) max entry.
    The parameters are kept on the probability simplex via a soft‑max.

3️⃣  Long simulated‑annealing search that mixes three move types:
    – swapping two positions,
    – transferring a (small or medium) fraction of mass,
    – reversing a short contiguous block.
    Acceptance follows a standard Metropolis criterion with a slowly
    decreasing temperature.

4️⃣  Final cheap hill‑climbing that only accepts strict improvements.

All convolutions are performed with an FFT‑based routine (`_fft_convolve_self`),
making the inner loop O(n log n).  The algorithm keeps track of the best
sequence found and returns it as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders (used by the smooth surrogate optimiser)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter for the L‑BFGS phase
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Clip tiny negatives caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` for a normalised vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    w_shift = w - np.max(w)
    ex = np.exp(w_shift)
    return ex / np.sum(ex)


# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and conv = a∗a with a = softmax(w).  The gradient is derived
    analytically and returned w.r.t. the soft‑max parameters.
    """
    a = _softmax(w)                     # ℓ₁‑normalised non‑negative vector

    conv = _fft_convolve_self(a)        # length 2·n‑1
    max_c = conv.max()

    # smooth approximation of the max
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)        # ∑exp((c−max)/ε)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (chain rule through the log‑sum‑exp)
    weights = np.exp(shifted)
    p = weights / np.sum(weights)       # soft‑max over the convolution entries

    # convolution a * reverse(p) gives the needed weighted sum
    p_rev = p[::-1]
    size = _next_pow2(conv.shape[0] + a.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[:a.shape[0] + p_rev.shape[0] - 1]

    # central n entries correspond to indices [n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]      # length n
    grad_a = 4.0 * n * seg[::-1]        # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by `scipy.optimize.minimize`."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Heuristics for reordering the vector (large values at the ends)
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place a large value, then a small one, alternating (original heuristic)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


def _ends_descending_reorder(a: np.ndarray) -> np.ndarray:
    """Monotone decrease from the ends towards the centre."""
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]   # descending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    idx = 0
    while left <= right:
        new_a[left] = sorted_desc[idx]
        idx += 1
        left += 1
        if left > right:
            break
        new_a[right] = sorted_desc[idx]
        idx += 1
        right -= 1
    return new_a


# ----------------------------------------------------------------------
# Projected Adam optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Adam optimiser on the true max‑convolution (sub‑gradient of the max entry)
    while keeping the vector on the probability simplex via a soft‑max.
    """
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)          # initialise soft‑max parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a0.copy()
    best_val = _eval_fast(best_a)

    while time.time() < deadline:
        t += 1
        a = _softmax(w)

        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # gradient of the active max entry w.r.t. a
        grad_a = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_a[mask] = 2.0 * a[idx[mask]]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update on w
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # simple decay with √t
        lr = 0.5 / (np.sqrt(t) + 1.0)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate
        a_new = _softmax(w)
        cur_val = _eval_fast(a_new)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[Adam] iter {t} new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (swap, mass‑transfer, block‑reverse)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Long SA that uses three move types:
      * swap two positions,
      * move a fraction of mass from i to j,
      * reverse a short contiguous block.
    Temperature decays slowly.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.05, 1e-6)   # start a bit warmer than before
    decay = 0.99995                # slower cooling

    it = 0
    while time.time() < deadline:
        move_type = rng.random()
        if move_type < 0.35:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]

        elif move_type < 0.75:                  # mass transfer (small or medium)
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-14:
                continue
            # 30 % of the time allow a larger transfer (up to 15 %)
            max_frac = 0.15 if rng.random() < 0.3 else 0.04
            delta = a[i] * max_frac * rng.random()
            a[i] -= delta
            a[j] += delta

        else:                                   # block reversal
            block_len = rng.integers(2, 12)    # reverse a tiny block
            start = rng.integers(0, n - block_len + 1)
            a[start:start + block_len] = a[start:start + block_len][::-1]

        new_val = _eval_fast(a)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] iter {it} new best = {best_val:.6f}")
        else:
            # revert move (needs to know which move was applied)
            # we simply restore from best_seq when the move is rejected:
            a = best_seq.copy()
            cur_val = best_val

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Very fast local search that accepts any strictly improving move."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-14:
                continue
            delta = cur_a[i] * 0.02 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Helper: construct a full‑length vector by piecewise linear interpolation
# ----------------------------------------------------------------------
def _interpolate_to_full(ctrl: np.ndarray, n: int) -> np.ndarray:
    """
    Given a low‑dimensional control vector `ctrl` (length k),
    interpolate it linearly on the grid {0,…,n‑1} and normalise to sum to 1.
    """
    k = len(ctrl)
    x_ctrl = np.linspace(0, 1, k)
    x_full = np.linspace(0, 1, n)
    a = np.interp(x_full, x_ctrl, ctrl, left=ctrl[0], right=ctrl[-1])
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final official evaluation
    deadline = start_time + budget_s - 1.5
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # ------------------------------------------------------------------
    # Problem size – 4000 works well with the FFT code and matches the
    # previously discovered solution size.
    # ------------------------------------------------------------------
    n = 4000
    global _GLOBAL_N
    _GLOBAL_N = n

    # ------------------------------------------------------------------
    # Initialise – reuse a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to requested size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        # fresh start: a modest random Dirichlet‑like vector
        a0 = np.random.rand(n)
        a0 /= np.sum(a0)
        print("[init] using fresh random start")

    a0 = np.maximum(a0, 0.0)
    a0 /= np.sum(a0)          # ℓ₁‑normalise to 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate L‑BFGS with several restarts
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-13), num=18)

    restart = 0
    while time.time() < deadline - 300.0:   # keep ~5 min for later phases
        restart += 1
        print(f"[phase1 restart {restart}]")
        # initialise soft‑max parameters from either the best point or a new random one
        if restart % 3 == 0:
            a_start = np.random.rand(n)
            a_start /= np.sum(a_start)
        else:
            a_start = best_a.copy() + 2e-3 * np.random.randn(n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= np.sum(a_start)

        w = np.log(a_start + 1e-12)

        for eps in eps_schedule:
            if time.time() >= deadline - 250.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            # L‑BFGS on the smooth surrogate
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x

            a_candidate = _softmax(w)
            cur_val = _eval_fast(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[phase1 eps={eps:.1e}] new best = {best_val:.6f}")

        # ------------------------------------------------------------------
        # Apply both re‑ordering heuristics
        # ------------------------------------------------------------------
        for reorder_fn in (_ends_large_reorder, _ends_descending_reorder):
            a_ord = reorder_fn(best_a)
            a_ord /= np.sum(a_ord)
            val_ord = _eval_fast(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[reorder {reorder_fn.__name__}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam projected gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("[phase2] Adam gradient descent")
        best_a, best_val = _adam_opt(best_a, deadline - 140.0)

    # ------------------------------------------------------------------
    # Phase 3 – long simulated annealing (swap + mass‑transfer + block‑reverse)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[phase3] Simulated annealing")
        best_a, best_val = _simulated_annealing(best_a,
                                                best_val,
                                                deadline - 20.0)

    # ------------------------------------------------------------------
    # Phase 4 – final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[phase4] Final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a,
                                              best_val,
                                              deadline)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided security‑checked function)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    # Return a plain Python list as required by the framework
    return best_a.tolist()
