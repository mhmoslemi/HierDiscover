"""
Improved search for a low‑peak self‑convolution step function.

Algorithm
---------
1. Initialise several diverse candidates (uniform, previous best, random
   Dirichlet, anti‑symmetric reorder) and keep the best according to a
   fast evaluation of 2·n·max(conv) (the denominator Σa_i = 1).

2. *Soft mirror‑descent* – minimise a smooth approximation of the
   objective.  At each iteration we compute a soft‑max distribution
   over the convolution entries (temperature ε) and the corresponding
   gradient of J(a)=2·n·max(conv).  The update is multiplicative
   (exponential mirror‑descent) on the simplex:
       a ← a·exp(−η·∇J)   ,   a ← a / Σa .
   ε and the learning rate η are geometrically decayed, allowing a
   coarse‑to‑fine optimisation.

3. *Simulated‑annealing mass‑transfer* – once the smooth phase stalls we
   perform a standard annealing walk: move a small fraction of mass from
   a high‑mass index to a low‑mass index, accepting uphill moves with a
   temperature schedule.

4. *Final cheap hill‑climb* – only strict improvements via random swaps
   or tiny mass transfers (the routine from the previous state‑of‑the‑art).

5. Return the best sequence (as a plain list).  A final call to the
   official ``evaluate_sequence`` is performed for safety.

The whole search respects the 1000‑second budget and prints intermediate
bounds for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)  # eliminate tiny negatives from round‑off
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already sum to 1.
    The denominator Σa_i is 1, therefore the score reduces to 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    # guard against division by zero – should never happen for a proper simplex point
    if max_b < 0:
        return float('inf')
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic reordering that places large values on the left side and small
    values on the right side – often reduces the central autocorrelation peak.
    """
    n = a.shape[0]
    s = np.sort(a)                      # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]           # large → left
        res[n - 1 - i] = s[i]           # small → right
    if n % 2:
        res[half] = s[half]
    return res


def _soft_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        J_ε(a) = 2·n·softmax_ε(conv(a))
    where softmax_ε is the log‑sum‑exp smoothing of the max.
    The returned gradient is ∇_a J_ε (size n), suitable for a mirror‑descent
    update a ← a·exp(−η·∇J_ε).
    """
    n = a.shape[0]
    conv = _full_convolution(a)               # length 2·n‑1
    max_c = conv.max()
    # soft‑max weights over convolution entries
    shifted = (conv - max_c) / eps
    # numerical stability (max of shifted is 0)
    exp_shift = np.exp(shifted - np.max(shifted))
    p = exp_shift / exp_shift.sum()            # distribution over lag indices
    # convolution of a with the reversed weight vector
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # segment that aligns with a_i
    seg = conv_ap[n - 1: 2 * n - 1]            # length n
    # Gradient of J = 4·n·seg[::-1] (see derivation in the analysis)
    grad = 4.0 * n * seg[::-1]
    # guard against tiny numerical noise
    np.maximum(grad, 0, out=grad)
    return grad


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %); after each move the
    vector is projected back onto the probability simplex.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # simplex projection (non‑negative + normalisation)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``. Returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final check
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine the problem size n (reuse the old construction if present)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # Build a collection of diverse initial candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # Uniform distribution
    init_candidates.append(np.full(n, 1.0 / n))

    # Previous construction (if available)
    if has_old:
        init_candidates.append(old / old.sum())

    # Random Dirichlet (α = 1) and (α = 5)
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # Simple sinusoid (slightly shifted to stay positive)
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # Normalise onto the simplex
    init_candidates = [c / c.sum() for c in init_candidates]

    # ------------------------------------------------------------------
    # Pick the best among the initial set (using the fast evaluator)
    # ------------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Soft mirror‑descent on the smoothed objective
    # ------------------------------------------------------------------
    a = best_a.copy()
    lr = 0.5               # initial learning rate for the exponential update
    eps = 0.5              # initial smoothing parameter ε
    lr_decay = 0.9995
    eps_decay = 0.995
    min_eps = 1e-12
    iter_counter = 0
    # Reserve ~200 s for the later stages
    while time.time() < deadline - 200.0:
        grad = _soft_gradient(a, eps)          # ∇J_ε(a)
        # Exponential mirror‑descent update (stay on the simplex)
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:                           # numerical safeguard
            a = np.full(n, 1.0 / n)
            s = 1.0
        a /= s

        # Occasionally evaluate (about 1 % of the iterations)
        if iter_counter % 100 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SMD it {iter_counter}] new best = {best_val:.6f}")

            # Try the anti‑symmetric reorder from time to time
            if iter_counter % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val - 1e-12:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[SMD reorder] improved to {best_val:.6f}")

        # Decay schedules
        lr *= lr_decay
        eps = max(min_eps, eps * eps_decay)

        iter_counter += 1

    # ------------------------------------------------------------------
    # Stage 2 – Simulated‑annealing style mass‑transfer hill climbing
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    temperature = 0.05
    cooling = 0.9999

    while time.time() < deadline - 60.0:
        # Choose a donor (biased towards large entries) and a recipient (among the smallest 5 %)
        donor = rng.choice(n, p=cur_a / cur_a.sum())
        low_idx = np.where(cur_a < np.percentile(cur_a, 5))[0]
        if low_idx.size == 0:
            recipient = rng.integers(0, n)
        else:
            recipient = rng.choice(low_idx)

        if donor == recipient or cur_a[donor] <= 1e-15:
            continue

        # Transfer up to 5 % of the donor's mass
        delta = cur_a[donor] * 0.05 * rng.random()
        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[recipient] += delta

        # Project back onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(temperature, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        temperature = max(temperature * cooling, 1e-6)

    # ------------------------------------------------------------------
    # Stage 3 – Final cheap hill‑climb (strict improvements only)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety evaluation using the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
