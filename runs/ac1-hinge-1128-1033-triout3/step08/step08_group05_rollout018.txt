def _alternating_reorder(a):
    n = a.shape[0]
    sorted_vals = np.sort(a)[::-1]  # descending
    new_a = np.empty_like(a)
    left = 0
    right = n - 1
    for idx, val in enumerate(sorted_vals):
        if idx % 2 == 0:
            new_a[left] = val
            left += 1
        else:
            new_a[right] = val
            right -= 1
    return new_a
```

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The search consists of five stages:

1. **Diverse initialisation** – uniform, previous best, Dirichlet draws,
   simple exponential and sinusoidal shapes.  All candidates are
   normalised to the probability simplex (∑aᵢ = 1).

2. **Aggressive L‑BFGS on a log‑sum‑exp surrogate** – we run a
   decreasing‐ε schedule (1e‑1 … 1e‑16) with many random restarts.
   The surrogate objective is
   ``L_ε = 2·n·max(conv) + 2·n·ε·log∑exp((conv‑max)/ε)``.
   After each ε step we evaluate the exact objective and keep the best
   vector.

3. **Mirror‑descent on the exact peak** – a projected exponential update
   (aᵢ ← aᵢ·exp(‑α·gᵢ)) where gᵢ = a_{K‑i} for the index K attaining the
   maximum of the self‑convolution.  The step size α decays slowly.
   Periodic “anti‑symmetric’’ and “alternating’’ re‑orderings are applied
   to escape plateaux.

4. **Targeted simulated annealing** – repeatedly identify the pair
   (i,j) that contributes most to the current peak and move a small
   fraction of mass from the larger of the two to a low‑mass index.
   Moves are accepted with a Metropolis criterion; the temperature and
   the maximal transfer fraction are cooled down over time.

5. **Very cheap final polish** – a lightweight stochastic hill‑climber
   that tries random swaps and tiny mass transfers.

All phases respect a global 1000 s budget (a 5 s safety margin is kept)
and the code prints intermediate bounds.  The final sequence is handed
to the official ``evaluate_sequence`` for a safety check before being
returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Clip tiny negative values caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ---------- soft‑max ----------
    w_shift = w - np.max(w)          # numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---------- convolution ----------
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ---------- smooth max ----------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    weights = np.exp(shifted)
    p = weights / weights.sum()      # probability distribution over lags

    # derivative of smooth max wrt each conv entry = p
    # grad_a = 4·n·(p * derivative of conv wrt a)
    # derivative of conv_k = Σ_i a_i·δ_{i + j = k} + Σ_j a_j·δ_{i + j = k}
    # which yields a convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(a.shape[0] + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: a.shape[0] + p_rev.shape[0] - 1]

    # segment corresponding to ∂conv/∂a
    seg = conv_ap[a.shape[0] - 1: 2 * a.shape[0] - 1]
    grad_a = 4.0 * n * seg[::-1]    # reverse to align with a_i order

    # ---------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors that sum to 1.
    Returns 2·n·max(conv) because the denominator (∑aᵢ)² = 1.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places large values on the left side and small values
    on the right side – often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """
    Interleave the largest and smallest values (L, S, 2nd‑largest,
    2nd‑smallest, …).  This often reduces the coincidence of large‑large,
    thereby decreasing the peak of the convolution.
    """
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]          # descending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            new_a[left] = val
            left += 1
        else:
            new_a[right] = val
            right -= 1
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector is kept on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def _make_exponential_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a decreasing exponential (geometric) distribution, normalised."""
    # Choose a decay factor r in (0.99, 0.9999)
    r = rng.uniform(0.99, 0.9999)
    a = (1 - r) * (r ** np.arange(n, dtype=float))
    a /= a.sum()
    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine problem size n
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # Initialise a diverse set of candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # 1. uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # 2. previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # 3. random Dirichlet (α = 0.5) – sparser
    init_candidates.append(rng.gamma(shape=0.5, scale=1.0, size=n))

    # 4. random Dirichlet (α = 2.0) – denser
    init_candidates.append(rng.gamma(shape=2.0, scale=1.0, size=n))

    # 5. exponential decay candidate
    init_candidates.append(_make_exponential_candidate(n, rng))

    # 6. sinusoidal shape (always non‑negative)
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.2)

    # Normalise everything onto the simplex
    init_candidates = [c / c.sum() for c in init_candidates]

    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on the smoothed surrogate
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-1), np.log10(1e-16), num=30)
    max_restarts = 12
    w = np.log(best_a + 1e-12)  # initialise soft‑max parameters

    for restart in range(max_restarts):
        if time.time() >= deadline - 150.0:
            break
        print(f"[L‑BFGS restart {restart + 1}]")
        if restart % 2 == 0:
            # every other restart we randomise completely
            w = rng.normal(size=n) * 2.0
        else:
            # otherwise initialise from the current best
            w = np.log(best_a + 1e-12)

        for eps in eps_schedule:
            if time.time() >= deadline - 150.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 3000,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x
            # transform back to a probability vector
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[L‑BFGS eps={eps:.1e}] new best = {best_val:.6f}")

        # occasional re‑orderings after each restart
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val - 1e-9:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder anti‑sym after L‑BFGS] improved to {best_val:.6f}")

        a_alt = _alternating_reorder(best_a)
        val_alt = _fast_evaluate(a_alt)
        if val_alt < best_val - 1e-9:
            best_val = val_alt
            best_a = a_alt.copy()
            print(f"[reorder alternating after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – refined mirror‑descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.8          # initial step size (larger than before)
    decay = 0.9995        # geometric decay
    min_alpha = 0.005
    it = 0
    while time.time() < deadline - 120.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]     # derivative ∂max/∂a_i = a_{K-i}

        # exponential mirror‑descent update
        alpha = max(alpha0 * (decay ** (it // 250)), min_alpha)
        a = a * np.exp(-alpha * g)

        # re‑project onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 250 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = a.copy()
                print(f"[MD it {it}] new best = {best_val:.6f}")

            # occasional re‑orderings
            if it % 1000 == 0:
                a_as = _anti_symmetric_reorder(a)
                val_as = _fast_evaluate(a_as)
                if val_as < best_val - 1e-9:
                    best_val = val_as
                    best_a = a_as.copy()
                    a = a_as.copy()
                    print(f"[MD anti‑sym] improved to {best_val:.6f}")

                a_alt = _alternating_reorder(a)
                val_alt = _fast_evaluate(a_alt)
                if val_alt < best_val - 1e-9:
                    best_val = val_alt
                    best_a = a_alt.copy()
                    a = a_alt.copy()
                    print(f"[MD alternating] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – targeted simulated annealing / pairwise hill‑climbing
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.08                # initial temperature
    cooling = 0.99994
    max_move_frac = 0.08    # initial maximal transfer fraction
    frac_decay = 0.9999

    while time.time() < deadline - 30.0:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        # indices that contribute to the maximal convolution entry
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        contrib = cur_a[idx] * cur_a[jdx]

        # pick the pair (i,j) with the largest product
        pair_idx = np.argmax(contrib)
        i = idx[pair_idx]
        j = jdx[pair_idx]

        # donor = the larger of the two
        donor = i if cur_a[i] >= cur_a[j] else j
        # recipient candidates: among the lightest 8 % entries
        low_thresh = np.percentile(cur_a, 8)
        low_candidates = np.where(cur_a <= low_thresh)[0]
        if low_candidates.size == 0:
            recipient = rng.integers(0, n)
        else:
            recipient = rng.choice(low_candidates)

        # move a small portion of mass from donor to recipient
        delta = cur_a[donor] * max_move_frac * rng.random()
        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[recipient] += delta

        # keep on the simplex (sum is theoretically unchanged)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s > 0:
            new_a /= s
        else:
            continue

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[anneal] improved to {best_val:.6f}")

        # Cool down temperature and the maximal move fraction
        T = max(T * cooling, 1e-6)
        max_move_frac = max(max_move_frac * frac_decay, 0.001)

        # Occasionally try a random swap (helps escape flat regions)
        if rng.random() < 0.07:
            i_swap, j_swap = rng.integers(0, n, size=2)
            if i_swap != j_swap:
                swapped = cur_a.copy()
                swapped[i_swap], swapped[j_swap] = swapped[j_swap], swapped[i_swap]
                swapped_val = _fast_evaluate(swapped)
                if swapped_val < cur_val:
                    cur_a = swapped
                    cur_val = swapped_val
                    if cur_val < best_val - 1e-12:
                        best_val = cur_val
                        best_a = cur_a.copy()
                        print(f"[swap] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final cheap hill‑climbing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
