def _alternating_reorder(a):
    # a is array of values being sorted descending
    v = np.sort(a)[::-1]  # descending
    n = len(v)
    res = np.empty_like(v)
    left = 0
    right = n-1
    toggle = True
    idx = 0
    while idx < n:
        if toggle:
            res[left] = v[idx]; left += 1
        else:
            res[right] = v[idx]; right -= 1
        toggle = not toggle
        idx += 1
    return res

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s budget)

0. Initialise a diverse set of candidates (uniform, exponential, power‑law,
   exponential decay).  Keep the best.

1. Sub‑gradient mirror descent on the true L∞‑objective.  The active set
   (indices whose convolution value is near the current maximum) is used
   to compute a sub‑gradient via a single FFT; an exponentiated update with
   a decaying step size keeps the vector on the simplex.

2. Smooth‐max optimisation: minimise a log‑sum‑exp surrogate with L‑BFGS‑B.
   Several random restarts are performed, gradually decreasing the smoothing
   parameter ε.  After each run the anti‑symmetric and a new “alternating”
   ordering are tried.

3. Permutation annealing – the multiset of values is kept fixed while the
   ordering is optimised.  Random swaps are accepted if they improve the
   objective, otherwise with a temperature schedule (simulated‑annealing).

4. Final hill‑climbing (mass‑transfer + swaps) to polish the solution.

All stages respect the global time‑budget.  The final list of floats is
returned; ``evaluate_sequence`` will be called once more for the official
score.
"""

_GLOBAL_N = None     # vector length for the current optimisation

# Utility functions

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length) via FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative artefacts caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Large values on the left, small values on the right.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]       # large → left side
        res[n - 1 - i] = s[i]       # small → right side
    if n % 2:
        res[half] = s[half]
    return res


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """
    Spread the largest values across the whole interval:
    place the largest at the left end, the second largest at the right end,
    the third largest next to the left end, etc.
    """
    v = np.sort(a)[::-1]          # descending
    n = v.shape[0]
    res = np.empty_like(v)
    left, right = 0, n - 1
    toggle = True
    for val in v:
        if toggle:
            res[left] = val
            left += 1
        else:
            res[right] = val
            right -= 1
        toggle = not toggle
    return res


def _objective_val(a: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(a∗a) (sum(a) is assumed to be 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)
    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t a
    weights = np.exp(shifted)
    p = weights / weights.sum()               # softmax over convolution indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries give derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]              # ∂L/∂a_i

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` – uses the globals."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

# ----------------------------------------------------------------------
# Sub‑gradient (mirror‑descent) on the true L∞‑objective
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the L∞‑norm for the active convolution indices.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n-2) - i] for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent_multi(a_start: np.ndarray, deadline: float):
    """
    Exponentiated mirror‑descent that reduces the active convolution peaks.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.8                       # initial step‑size factor
    rel_tol = 0.05                     # initial active‑set tolerance

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten the tolerance gradually

            rel_tol = max(rel_tol * 0.5, 1e-5)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient via FFT
        g = _active_set_gradient(a, active)

        # decaying step‑size (mirror descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative update
        a = a * np.exp(-alpha * g)

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")
            if it % 5000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[sgd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val

# Simulated‑annealing based hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    """
    Random local moves (swap or mass transfer) with an annealing schedule.
    Only strictly better solutions are kept for the reported best,
    but occasional uphill moves are accepted to escape plateaus.
    """
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.02                     # initial temperature

    it = 0
    while time.time() < deadline:
        it += 1
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            accept = True
        else:
            T = T0 * (0.9995 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val_local:.6f}")

    return best_a, best_val_local

def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (dist_to_end+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a
