def project_simplex(v):
    # ensures non-negative and sum 1
    if np.sum(v) == 0:
        return np.full_like(v, 1.0/v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid optimizer for the step‑function minimisation problem (improved).

The search consists of the following stages (all respect the 1000 s wall‑time
budget and are deliberately independent so that they can be skipped if time
runs out):

1. **Random exponential‑mixture initialisation** – generate many candidates of
   the form (1‑w)·uniform + w·(rⁱ+rⁿ⁻¹⁻ⁱ) and keep the best.  This shape is known
   to concentrate mass at the two ends while keeping a fairly flat convolution.

2. **Smooth surrogate minimisation (log‑sum‑exp)** – L‑BFGS on the same
   ε‑log‑sum‑exp surrogate used in the previous solution, but with a much
   finer ε‑schedule, several random restarts and an additional “ends‑large”
   re‑ordering after each restart.

3. **Variance‑flattening descent** – minimise ∑ₖ(cₖ)², i.e. the L²‑norm of the
   convolution c = a∗a.  Reducing this variance flattens the whole convolution
   and therefore lowers its maximum.  A simple projected gradient step with an
   adaptive step size is used.

4. **Projected Adam on the exact max‑convolution** – as before, but run on the
   best vector obtained so far.

5. **Simulated annealing (swap + mass‑transfer)** – a long SA run with a
   higher initial temperature and both small and large mass moves.

6. **Very cheap hill‑climbing refinement** – final local search.

All heavy work (convolutions, gradients) uses FFT‑based O(n log n) kernels.
The whole routine stops well before the deadline, evaluates the final
candidate with the official ``evaluate_sequence`` function and returns it as a
plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders (required by the smooth‑surrogate optimiser)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter for the L‑BFGS phase
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # Clip tiny negative values caused by floating‑point noise.
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Exact copy of ``evaluate_sequence`` for a normalised vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    w_shift = w - np.max(w)
    ex = np.exp(w_shift)
    return ex / np.sum(ex)


# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and conv = a∗a with a = softmax(w).  The gradient is analytic.
    """
    a = _softmax(w)                     # ℓ₁‑normalised non‑negative vector
    conv = _fft_convolve_self(a)        # length 2·n‑1
    max_c = conv.max()

    # smooth approximation of the max
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)        # log ∑exp((c−max)/ε)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / np.sum(weights)       # soft‑max over the convolution entries

    # convolution a * reverse(p) -> weighted sum needed for ∂L/∂a
    p_rev = p[::-1]
    size = _next_pow2(conv.shape[0] + a.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[:a.shape[0] + p_rev.shape[0] - 1]

    # central n entries correspond to indices [n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]      # length n
    grad_a = 4.0 * n * seg[::-1]        # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Simple deterministic re‑ordering that puts large values at the extremes
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest entries at the two ends and then alternate."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


# ----------------------------------------------------------------------
# Project a vector onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    # ensure finite values
    v = np.clip(v, -1e20, 1e20)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Random exponential‑mixture initialiser
# ----------------------------------------------------------------------
def _random_exp_mixture(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Generate a symmetric distribution consisting of a uniform part and an
    exponential “ends” part:
        a = (1‑w)·uniform + w·(exp(‑r·i) + exp(‑r·(n‑1‑i))) / Z .
    The parameters w∈[0,1] and r>0 are sampled uniformly.
    """
    r = rng.uniform(0.0, 5.0)           # shape of the exponential tail
    w = rng.uniform(0.0, 1.0)           # weight of the exponential component

    # exponential component, symmetric
    exp_vec = np.exp(-r * np.arange(n))
    exp_vec = exp_vec + exp_vec[::-1]
    exp_vec /= exp_vec.sum()           # normalise to 1

    # uniform component
    uniform = np.full(n, 1.0 / n)

    a = (1.0 - w) * uniform + w * exp_vec
    # tiny random jitter to avoid perfectly symmetric stagnation
    a += rng.normal(scale=1e-5, size=n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient descent that minimises the L²‑norm of the convolution
# ----------------------------------------------------------------------
def _flatten_descent(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Projected gradient descent on J(a)=∑ₖ (a∗a)ₖ².
    Large reductions of the variance of the convolution often lower its
    maximum as well.  The step size is adapted automatically.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _eval_fast(best_a)

    # initial step size – a modest number; it will be adapted.
    step = 0.01
    eps = 1e-12

    while time.time() < deadline:
        # convolution and its L²‑norm
        conv = _fft_convolve_self(a)               # length 2n‑1
        # gradient of ∑ conv² wrt a
        a_rev = a[::-1]
        size = _next_pow2(conv.shape[0] + a_rev.shape[0] - 1)
        C_fft = np.fft.rfft(conv, size)
        A_rev_fft = np.fft.rfft(a_rev, size)
        full = np.fft.irfft(C_fft * A_rev_fft, size)[:conv.shape[0] + a_rev.shape[0] - 1]
        grad_a = 4.0 * full[len(a) - 1: len(a) - 1 + len(a)]

        # projected gradient step
        a_new = a - step * grad_a
        a_new = _project_simplex(a_new)

        cur_val = _eval_fast(a_new)

        # simple step‑size adaptation
        if cur_val < best_val:
            best_val, best_a = cur_val, a_new.copy()
            step *= 1.03                     # a little more aggressive after success
            print(f"[flatten] new best = {best_val:.6f}")
        else:
            step *= 0.6                      # back‑off if no improvement

        a = a_new

        # avoid degenerate step size
        if step < 1e-8:
            step = 1e-8
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected Adam optimiser on the exact maximum (sub‑gradient of max)
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """Adam optimiser on the true max‑convolution while staying on the simplex."""
    n = a0.shape[0]
    # initialise soft‑max parameters from a0
    w = np.log(a0 + 1e-12)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a0.copy()
    best_val = _eval_fast(best_a)

    while time.time() < deadline:
        t += 1
        a = _softmax(w)

        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active peak
        grad_a = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_a[mask] = 2.0 * a[idx[mask]]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = 0.5 / (np.sqrt(t) + 1.0)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _softmax(w)
        cur_val = _eval_fast(a_new)

        if cur_val < best_val:
            best_val, best_a = cur_val, a_new.copy()
            print(f"[Adam] iter {t} new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing with both tiny and large mass transfers
# ----------------------------------------------------------------------
def _simulated_annealing_mass(a: np.ndarray,
                              cur_val: float,
                              deadline: float) -> (np.ndarray, float):
    """Long SA with swaps and mass‑transfer moves of varying magnitude."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # start with a relatively high temperature to allow uphill moves
    T = max(cur_val * 0.2, 1e-4)
    decay = 0.99995

    it = 0
    while time.time() < deadline:
        move_type = rng.random()
        if move_type < 0.3:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]            # propose swap
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-14:
                continue
            # choose a magnitude: 30 % tiny (≤3 %), else up to 30 %
            if rng.random() < 0.3:
                delta = a[i] * 0.03 * rng.random()
            else:
                delta = a[i] * 0.3 * rng.random()
            a[i] -= delta
            a[j] += delta

        new_val = _eval_fast(a)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            # revert the move
            if move_type < 0.3:                # swap
                a[i], a[j] = a[j], a[i]
            else:
                a[i] += delta
                a[j] -= delta

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Accept any strictly improving swap or mass‑transfer move."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-14:
                continue
            delta = cur_a[i] * 0.015 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a, cur_val = new_a, val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point required by the challenge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the official evaluation
    deadline = start_time + budget_s - 1.5
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # ------------------------------------------------------------------
    # Problem size – we keep the previously successful size of 4000.
    # ------------------------------------------------------------------
    n = 4000
    global _GLOBAL_N
    _GLOBAL_N = n

    # ------------------------------------------------------------------
    # Initialise – try to load a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using plain uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= np.sum(a0)          # ℓ₁‑normalise to 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Stage 0 – random exponential‑mixture initialisation
    # ------------------------------------------------------------------
    while time.time() < deadline - 300.0:      # reserve ~5 min for later phases
        a_candidate = _random_exp_mixture(rng, n)
        cur_val = _eval_fast(a_candidate)
        if cur_val < best_val:
            best_val, best_a = cur_val, a_candidate.copy()
            print(f"[stage0] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – smooth surrogate L‑BFGS with many restarts & finer eps
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-14), num=30)

    restart = 0
    while time.time() < deadline - 250.0:        # keep enough time for later phases
        restart += 1
        print(f"[stage1 restart {restart}]")
        # initialise soft‑max parameters: occasionally restart from random point
        if restart % 3 == 0:
            w = np.log(best_a + 1e-12) + rng.normal(scale=0.8, size=n)
        else:
            w = np.log(best_a + 1e-12) + 0.5 * rng.standard_normal(n)

        for eps in eps_schedule:
            if time.time() >= deadline - 200.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            # L‑BFGS on the smooth surrogate
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x

            # evaluate corresponding probability vector
            a_candidate = _softmax(w)
            cur_val = _eval_fast(a_candidate)
            if cur_val < best_val:
                best_val, best_a = cur_val, a_candidate.copy()
                print(f"[stage1 eps={eps:.1e}] new best = {best_val:.6f}")

        # reorder “ends‑large” after each restart
        a_ord = _ends_large_reorder(best_a)
        a_ord /= np.sum(a_ord)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – variance‑flattening descent (minimise ∑c²)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("[stage2] variance flattening descent")
        best_a, best_val = _flatten_descent(best_a, deadline - 140.0)

    # ------------------------------------------------------------------
    # Stage 3 – Adam projected gradient descent on the exact max
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print("[stage3] Adam optimiser on max")
        best_a, best_val = _adam_opt(best_a, deadline - 70.0)

    # ------------------------------------------------------------------
    # Stage 4 – long simulated annealing (swap + mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[stage4] Simulated annealing")
        best_a, best_val = _simulated_annealing_mass(best_a,
                                                     best_val,
                                                     deadline - 20.0)

    # ------------------------------------------------------------------
    # Stage 5 – final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[stage5] Final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a,
                                              best_val,
                                              deadline)

    # ------------------------------------------------------------------
    # Official evaluation (uses the provided security‑checked function)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    # Return a plain Python list as required by the framework
    return best_a.tolist()
