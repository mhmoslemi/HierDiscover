def mirror_descent_line_search(a, deadline):
    n = a.shape[0]
    best_val = _fast_evaluate(a)
    cur_a = a.copy()
    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_idx = int(conv.argmax())
        # compute subgradient g
        g = np.zeros_like(cur_a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end+1)
        j = max_idx - idx
        g[idx] = cur_a[j]
        # line search over alphas
        alphas = [0.5, 0.3, 0.1, 0.05, 0.02, 0.01]
        best_local_val = best_val
        best_local_a = cur_a
        for alpha in alphas:
            a_new = cur_a * np.exp(-alpha * g)
            # projection
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s <= 0:
                continue
            a_new /= s
            val = _fast_evaluate(a_new)
            if val < best_local_val - 1e-12:
                best_local_val = val
                best_local_a = a_new
        # if improvement, accept
        if best_local_val < best_val - 1e-12:
            cur_a = best_local_a
            best_val = best_local_val
            # maybe print progress
        else:
            # no improvement, reduce alpha schedule further
            # maybe use smaller alpha
            # try some small alphas gradually down
            for alpha in [0.005, 0.002, 0.001]:
                a_new = cur_a * np.exp(-alpha * g)
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s <= 0: continue
                a_new /= s
                val = _fast_evaluate(a_new)
                if val < best_val - 1e-12:
                    cur_a = a_new
                    best_val = val
                    break
        # maybe occasional reorder based on anti-symmetric or random
        if time.time() > deadline - 30: break
    return cur_a, best_val
```

"""
Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Problem size** – reuse the length of the previously found solution if
   available, otherwise fall back to 4000.

2. **Broad parametric sweep** – generate candidates from several
   low‑dimensional families (uniform, power‑laws, exponentials,
   symmetric exponentials, double exponentials, logistic, beta‑shaped,
   sinusoid, random Dirichlet draws).  The families are explored on a
   fine grid of parameters; each candidate is normalised to the simplex
   and evaluated with a fast convolution routine.  The best candidate is
   kept.

3. **Fine‑grained mirror descent** – starting from the best candidate,
   repeatedly apply the exact sub‑gradient update for the max‑convolution
   (the “worst” index).  At each iteration a small line‑search over a
   handful of step‑sizes is performed; the update that yields the
   greatest improvement is kept.  The loop runs until the time budget is
   almost exhausted.

4. **Optional simulated‑annealing polishing** – a few block‑swap and
   mass‑transfer moves are tried with a geometric cooling schedule to
   escape shallow local minima.

5. **Final hill‑climbing** – cheap strict‑improvement swaps / mass moves.

6. **Safety check** – the returned sequence is evaluated with the
   official ``evaluate_sequence`` before returning.

All steps respect the allotted time (default 1000 s).  The code uses only
NumPy, SciPy (for the optional L‑BFGS refinement) and standard Python
libraries.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Result length = 2*len(a)-1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Clip tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the O(n) algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already satisfy
    a_i ≥ 0 and Σ a_i = 1.  Returns 2·n·max(conv) (the denominator is 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed surrogate for L‑BFGS (optional, kept from the previous code)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None
_GLOBAL_N = None


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate of the max‑convolution:
    L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv - max)/ε)
    with the soft‑max re‑parameterisation a_i = exp(w_i)/Σ exp(w_i).
    Returns (objective, gradient w.r.t. w).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    w_shift = w - np.max(w)          # for stability
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)
    max_c = conv.max()

    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()      # soft‑max over lag indices

    # convolution of a with reversed p (needed for gradient)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return float(L), grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w)
