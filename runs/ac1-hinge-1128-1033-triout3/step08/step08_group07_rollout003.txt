"""
Hybrid optimizer for the step‑function minimisation problem.

The search consists of the following stages (all respecting the 1000 s
wall‑time budget):

1. **Smooth surrogate minimisation** – minimise a log‑sum‑exp
   approximation of the max‑convolution with L‑BFGS.  The surrogate is
   2·n·max(c) + 2·n·ε·logsumexp((c−max)/ε) where c = a∗a.
   Several restarts with a decreasing schedule of ε are performed; after each
   restart the candidate is examined with a few deterministic reorderings
   (largest‑at‑the‑beginning, “ends‑large”, etc.).

2. **Projected Adam on the exact max‑convolution** – a sub‑gradient of the
   active peak is taken and the soft‑max parameters are updated with the
   Adam optimiser; after each run a new random perturbation of the best
   point is tried, while keeping the vector on the probability simplex.

3. **Long simulated‑annealing on permutations + tiny mass transfers** – random
   swaps of positions and small transfers of mass are accepted with a classic
   Metropolis criterion; the temperature decays slowly so that the search
   can escape shallow local minima.

4. **Very cheap hill‑climbing refinement** – additional random swaps and
   mass‑transfer moves are performed until the deadline.

All heavy work (convolutions, gradients) is performed with FFT‑based
convolution (`_fft_convolve_self`) which runs in O(n log n).  The code never
stores or returns intermediate closures, complies with the required function
signature, and prints progress only when a new best value is found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders (used by the L‑BFGS wrapper)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter for the L‑BFGS phase
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)  # clip tiny negatives from rounding
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast (exact) copy of the provided ``evaluate_sequence`` for a
    L1‑normalised vector.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    w_shift = w - np.max(w)
    ex = np.exp(w_shift)
    return ex / np.sum(ex)


# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and conv = a∗a with a = softmax(w).  The gradient is derived
    analytically and returned w.r.t. the soft‑max parameters.
    """
    a = _softmax(w)                     # ℓ₁‑normalised non‑negative vector
    conv = _fft_convolve_self(a)        # length 2·n‑1
    max_c = conv.max()

    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)        # ∑exp((c−max)/ε)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (derived from the smooth max)
    weights = np.exp(shifted)
    p = weights / np.sum(weights)       # soft‑max over the convolution entries

    # convolution a * reverse(p) gives the needed weighted sum
    p_rev = p[::-1]
    size = _next_pow2(conv.shape[0] + a.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[:a.shape[0] + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]      # central n entries
    grad_a = 4.0 * n * seg[::-1]        # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Simple deterministic reorderings
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place large values at the extremes in an alternating fashion:
    largest, smallest, second‑largest, second‑smallest, …
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


def _sorted_desc_reorder(a: np.ndarray) -> np.ndarray:
    """Sort the vector in decreasing order (largest first)."""
    return np.sort(a)[::-1]


def _sorted_asc_reorder(a: np.ndarray) -> np.ndarray:
    """Sort the vector in increasing order (smallest first)."""
    return np.sort(a)


# ----------------------------------------------------------------------
# Simulated annealing on permutations (plus tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing_mass(a: np.ndarray,
                              cur_val: float,
                              deadline: float) -> (np.ndarray, float):
    """
    Random swaps and tiny mass transfers accepted with a classical
    Metropolis rule.  Works directly with the fast evaluator for speed.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # start temperature as a small fraction of the current value
    T = max(cur_val * 0.02, 1e-7)
    decay = 0.99994

    it = 0
    while time.time() < deadline:
        if rng.random() < 0.35:                     # pure swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]                # propose swap
            move_type = 'swap'
        else:                                      # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-12:
                continue
            delta = a[i] * (0.015 + 0.015 * rng.random())  # 1.5 % – 3 %
            a[i] -= delta
            a[j] += delta
            move_type = 'mass'

        new_val = _eval_fast(a)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            # revert the move
            if move_type == 'swap':
                a[i], a[j] = a[j], a[i]
            else:
                a[i] += delta
                a[j] -= delta

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected Adam on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Adam optimiser on the true max‑convolution (sub‑gradient of the active
    peak) while keeping the vector on the probability simplex.
    """
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)          # initialise soft‑max parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a0.copy()
    best_val = _eval_fast(best_a)

    while time.time() < deadline:
        t += 1
        a = _softmax(w)

        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active max entry w.r.t. a
        grad_a = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_a[mask] = 2.0 * a[idx[mask]]

        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update on w
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = 0.5 / (np.sqrt(t) + 1.0)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _softmax(w)
        cur_val = _eval_fast(a_new)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[Adam] iter {t} new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Very fast local search that accepts any strictly improving move."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-12:
                continue
            delta = cur_a[i] * 0.02 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point required by the challenge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin for the final official evaluation
    deadline = start_time + budget_s - 1.5
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # --------------------------------------------------------------
    # Problem size – 4500 provides a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4500
    global _GLOBAL_N
    _GLOBAL_N = n

    # --------------------------------------------------------------
    # Initialise – use a previously saved construction if it exists,
    # otherwise start from a uniform vector with a small perturbation.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        a0 += 5e-4 * np.random.rand(n)
        print("[init] using fresh uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= np.sum(a0)          # L1‑normalise to 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth surrogate L‑BFGS with several restarts
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-14), num=20)

    restart = 0
    # keep ~5 min for the later phases
    phase1_deadline = deadline - 300.0
    while time.time() < phase1_deadline:
        restart += 1
        print(f"[phase1 restart {restart}]")
        # initialise soft‑max parameters from the current best point
        w = np.log(best_a + 1e-12) + 0.6 * np.random.randn(n)

        for eps in eps_schedule:
            if time.time() >= phase1_deadline:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            # L‑BFGS on the smooth surrogate
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 400,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            a_candidate = _softmax(w)
            a_candidate /= np.sum(a_candidate)     # re‑normalise numerically
            cur_val = _eval_fast(a_candidate)

            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[phase1 eps={eps:.1e}] new best = {best_val:.6f}")

        # ------------------------------------------------------------------
        # Deterministic reorderings: try several heuristics
        # ------------------------------------------------------------------
        for reorder_fn in (_ends_large_reorder,
                           _sorted_desc_reorder,
                           _sorted_asc_reorder):
            a_ord = reorder_fn(best_a)
            a_ord /= np.sum(a_ord)
            val_ord = _eval_fast(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[reorder {reorder_fn.__name__}] new best = {best_val:.6f}")

        # small random perturbation before the next restart (helps to escape)
        noise = 0.02 * np.random.randn(n)
        best_a = np.maximum(best_a + noise, 0.0)
        best_a /= np.sum(best_a)

    # --------------------------------------------------------------
    # Phase 2 – Adam projected gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline - 200.0:
        print("[phase2] Adam gradient descent (multiple runs)")
        # allocate the remaining time in 45‑second blocks
        while time.time() < deadline - 120.0:
            # perturb the current best a bit and run Adam from there
            a_start = best_a + 0.01 * np.random.rand(n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= np.sum(a_start)
            sub_dead = time.time() + 40.0
            a_adam, val_adam = _adam_opt(a_start, sub_dead)
            if val_adam < best_val:
                best_val = val_adam
                best_a = a_adam.copy()
                print(f"[phase2] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – long simulated annealing (swap + mass‑transfer moves)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[phase3] Simulated annealing")
        best_a, best_val = _simulated_annealing_mass(best_a,
                                                     best_val,
                                                     deadline - 20.0)

    # --------------------------------------------------------------
    # Phase 4 – final cheap hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[phase4] Final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a,
                                              best_val,
                                              deadline)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the provided security‑checked function)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    # Return a plain Python list as required by the framework
    return best_a.tolist()
