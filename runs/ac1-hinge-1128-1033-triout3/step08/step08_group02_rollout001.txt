"""
Step‑function optimizer for `evaluate_sequence`.

The value returned by `evaluate_sequence` (for a non‑zero sum) is

        f(a) = 2·n·max(conv(a,a)) / (∑a)² .

Because the expression is scale‑invariant we keep the vector normalised
(∑a = 1) and minimise  2·n·max(conv(a,a)).

**What the code does**

1. *Phase‑0* – a very fast sweep over several parametric families
   (uniform, U‑shapes, exponentials, Dirichlet draws …) and keeps the
   best one.

2. *Phase‑1* – *smoothed* exponentiated‑gradient descent.
   Instead of using only the maximal convolution entry we employ a
   soft‑max (log‑sum‑exp) surrogate

        Lλ(a) = (1/λ)·log Σk exp(λ·b_k)   with   b = conv(a,a).

   Its gradient is the correlation of the current vector `a`
   with a probability distribution `w` proportional to
   `exp(λ·b)`.  We update multiplicatively
   `a ← a·exp(‑η·∇Lλ)` and renormalise to the simplex.
   λ is ramped from a modest value (≈ 20) to a large one (≈ 2·10⁴)
   so that the optimisation first moves smoothly and then behaves
   like the ordinary max‑gradient descent.  The learning rate η
   decays from 2·10⁻¹ to 2·10⁻³.

3. *Phase‑2* – deterministic greedy hill‑climbing (the same idea as
   in the previous solution).  Small fractions of mass are moved
   from the index that contributes most to the current peak to the
   index that contributes least; only improving moves are kept.

4. *Phase‑3* – a tiny mixing with the uniform distribution
   (convex combinations with α = 0.001 … 0.02).  This occasionally
   reduces the peak a last time.

All phases respect the global deadline (≈1000 s).  The best vector
found is returned as a plain list of Python floats.  Intermediate
progress (objective value, iteration counters, timestamps) is printed
to aid debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Phase‑0 – cheap families for a good start
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)                # uniform
    best_val = _objective(best_a, n)

    def try_update(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(12345)

    idx = np.arange(n)
    # deterministic families
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # linear U‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # quadratic decreasing
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # random Dirichlet draws (quick)
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑1 – smoothed (log‑sum‑exp) exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _smooth_expgrad_phase(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Weighted (soft‑max) exponentiated‑gradient descent.
    λ ramps from 20 → 20000, η decays from 0.2 → 0.002.
    """
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    # λ schedule parameters
    lam_start = 20.0
    lam_target = 2.0e4

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective (exact)
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SmoothExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule λ (geometric progression)
        progress = (now - start) / (deadline - start)
        lam = lam_start * (lam_target / lam_start) ** progress

        # soft‑max weights w_k ∝ exp(λ·(b_k‑maxb))
        # clipping avoids under‑/overflow
        diff = b - maxb                  # ≤ 0
        diff = np.clip(diff, -60.0, 0.0)  # exp(-60) ≈ 8e‑27, effectively zero
        w = np.exp(lam * diff)
        w_sum = w.sum()
        if w_sum == 0.0:                 # under‑flow – fall back to hard max
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

        # gradient: g_i = 2· Σ_j a_j·w_{i+j}
        # this is a correlation w ⋆ a.  We obtain it via a convolution with a reversed.
        conv_full = np.convolve(w, cur[::-1])                 # length 3·n‑2
        # slice corresponding to i = 0 … n‑1
        g = 2.0 * conv_full[n - 1:n - 1 + n]

        # learning‑rate schedule (linear decay)
        eta = (1.0 - progress) * 0.20 + progress * 0.002

        # exponentiated update – guard against overflow in the exponent
        g_clip = np.clip(g, -50.0, 50.0)
        cur = cur * np.exp(-eta * g_clip)

        # renormalise to the simplex
        total = cur.sum()
        if total == 0.0 or not np.isfinite(total):
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # occasional jitter if the process stalls
        if no_improve > IMPATIENT:
            cur += np.random.normal(scale=0.006, size=n)
            cur = np.maximum(cur, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[SmoothExpGrad {it}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑2 – deterministic greedy hill‑climbing (as before)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Move a fraction of mass from the “worst” index to the “best” one."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # anneal transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the maximal convolution entry
        k = int(np.argmax(b))

        # slice of a that contributes to this peak (a_{k‑i})
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        # source = index with largest contribution
        i = int(np.argmax(rev))

        # target = index with smallest contribution (different from source)
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                cur += np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it}] jitter after stagnation")
            continue

        # amount to move (random fraction of the allowed max)
        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            it += 1
            continue

        # try the move
        cur[i] -= delta
        cur[j] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # accept – improvement already recorded at start of next loop
            pass
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑3 – tiny uniform mixing
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a′ = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under the given time budget.
    Returns the best list of floats found (∑a = 1) which minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – 8000 provides a good trade‑off between quality and speed
    n = 8000

    # ------------------------------------------------------------------
    # Initial guess (reuse previous solution if available)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase‑1 – smoothed exponentiated gradient descent
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.45 * (deadline - start_time)
    best_a, best_val = _smooth_expgrad_phase(
        best_a, n,

        target_val=TARGET
    )

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase‑3 – final tiny mixing with the uniform vector
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification through the provided secure wrapper
    # ------------------------------------------------------------------
    try:

        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
