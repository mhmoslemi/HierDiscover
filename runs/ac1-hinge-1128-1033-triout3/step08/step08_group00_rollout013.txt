def _adam_optimize_exact(a0, n, deadline, start_time, best_a, best_val, target_val=1.503):

    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.9, 0.999
    lr_start, lr_end = 0.15, 0.001

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0

    while time.time() < deadline:
        # softmax to get a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        # compute max conv and gradient
        grad, max_b = _grad_max_conv(a)

        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            idle = 0
            print(f"[AdamExact {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress

        # chain rule
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturb
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.02, size=n)

        if idle > 20000: (similar restart)
            ...

        iteration += 1

    Return best.

def _adam_opt_exact(a0, n, deadline, start_time,
                    best_a, best_val,
                    target_val=1.503):
    import time, numpy as np
    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.9, 0.999
    lr_start, lr_end = 0.2, 0.001  # may adjust
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    max_idle = 15000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute a via softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()
        # compute gradient and current objective
        grad, max_b = _grad_max_conv(a)
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            idle = 0
            print(f"[AdamExact {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1
        # schedule
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        # chain rule to logits
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w * grad_w)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # occasional random perturbation
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.02, size=n)
        if idle > max_idle:
            print(f"[AdamExact {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0
        iteration += 1
    return best_a, best_val

def _hillclimb_simple(a0, n, deadline, start_time, best_a, best_val,
                      target_val=1.503):
    import time, numpy as np
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    iteration = 0
    no_improve = 0
    max_no_improve = 200000
    eps = 1e-12
    while True:
        now = time.time()
        if now >= deadline:
            break
        # recompute max
        k_max = int(np.argmax(b))
        # contributions to max entry
        i_low = max(0, k_max - (n-1))
        i_high = min(n-1, k_max)
        idx = np.arange(i_low, i_high+1)
        contrib = a[idx] * a[k_max - idx]
        # source index = idx with max contrib
        src_idx = idx[np.argmax(contrib)]
        # target: choose index where a small and not contributing strongly
        # compute weight = 1/(contrib+epsilon) (or use a inverse)
        inv_weight = np.full_like(a, eps)
        inv_weight[:] = 1.0 / (a + eps)  # prioritize small a
        inv_weight[src_idx] = 0.0
        probs = inv_weight / inv_weight.sum()
        tgt_idx = np.random.choice(n, p=probs)
        # amount to move
        delta = a[src_idx] * np.random.rand() * 0.5  # max 50% of source
        if delta < eps:
            continue
        # incremental update of convolution
        # recall b_new = b + 2*delta*(e_j - e_i) * a (convolution with a) + delta^2 contributions
        # we approximate with O(n) update similar to previous code
        # Compute update:
        b_new = b.copy()
        # j component added
        b_new[tgt_idx:tgt_idx+n] += 2.0 * delta * a
        # i component removed
        b_new[src_idx:src_idx+n] -= 2.0 * delta * a
        # diagonal terms:
        b_new[2*tgt_idx] += delta * delta
        b_new[2*src_idx] += delta * delta
        b_new[src_idx + tgt_idx] -= delta * delta
        new_max = b_new.max()
        new_val = 2.0 * n * new_max
        if new_val < cur_val - 1e-15:
            # accept
            a[src_idx] -= delta
            a[tgt_idx] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1
        if no_improve > max_no_improve:
            # perturbation
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] perturbation")
        iteration += 1
    return best_a, best_val

Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm outline
-----------------
1️⃣  **Initialisation** – try to reuse a previous solution (if any),
    sweep a cheap two‑parameter Beta family and generate a few
    random Dirichlet draws (all on a moderate size ``n_small``).

2️⃣  **Exact‑gradient Adam** – minimise the true objective
    ``F(a)=2·n·max(conv(a,a))`` using Adam on the log‑softmax
    parametrisation.  The gradient of the *max* is computed analytically
    (no smooth approximation).  The optimizer runs both at the small
    size and after up‑sampling to the final size.

3️⃣  **Greedy hill‑climbing** – repeatedly move a small amount of mass
    from an index that contributes most to the current maximal
    convolution entry to an index with little contribution.
    Moves are accepted only if the objective improves; occasional
    perturbations escape plateaus.

The whole search respects a 1000 s budget (leaving 0.5 s for the final

plain Python list whose sum is 1 (scale‑invariant for the wrapper).

import time

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``2·n·max(conv(a,a))`` for a probability vector."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv(a,a)) (scaled for the full objective)
# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray):
    """
    Returns
        grad : ∇_a ( 2·n·max(conv(a,a)) )
        M    : current maximal convolution value
    """
    n = a.shape[0]
    b = _full_convolution(a)
    k = int(np.argmax(b))                 # index of the maximal entry
    # gradient of the *unsclaed* max is: g_i = 2·a_{k-i} (if k-i in range)
    # scaled by the outer factor 2·n → 4·n·a_{k-i}
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 4.0 * n * a[k - idx]
    return grad, b[k]


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Adam optimiser that uses the *exact* sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _adam_opt_exact(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that directly minimises

        F(a) = 2·n·max(conv(a,a))

    using the exact sub‑gradient (no smoothing).  The parameters are
    kept on the simplex via a log‑softmax (entropy) parametrisation.
    """
    # initialise logits
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001      # exponential decay
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 20000
    iteration = 0

    while True:

        if now >= deadline:
            break

        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        grad, max_b = _grad_max_conv(a)
        cur_val = 2.0 * n * max_b

            idle = 0

                break

        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress

        # gradient w.r.t. logits (soft‑max chain rule)
        dot = np.dot(grad, a)

        # Adam update on logits
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to break symmetry
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.02, size=n)

        # restart if we have been idle for a while
        if idle > MAX_IDLE:
            print(f"[AdamExact {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climbing (deterministic acceptance, occasional shake‑up)
# ----------------------------------------------------------------------
def _hillclimb_greedy(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simple O(n) local search: move a fraction of mass from the index that
    contributes most to the current maximal convolution entry to an index
    with little contribution.  Moves are accepted only when they improve
    the objective; after a long stagnation a small random perturbation is
    applied.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12
    iteration = 0
    no_improve = 0
    MAX_NO_IMPROVE = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to that entry
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k_max - idx]

        # source – the biggest contributor
        src = idx[np.argmax(contrib)]

        if a[src] <= eps:
            # nothing to move – pick a random source instead
            src = int(np.argmax(a))

        # target – bias towards indices with tiny values
        inv_weight = 1.0 / (a + eps)
        inv_weight[src] = 0.0
        probs = inv_weight / inv_weight.sum()
        tgt = np.random.choice(n, p=probs)

        # amount to transfer (up to 50 % of the source mass)
        delta = a[src] * np.random.rand() * 0.5
        if delta < eps:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        # add the contribution of `tgt`
        b_new[tgt:tgt + n] += 2.0 * delta * a
        # remove the contribution of `src`
        b_new[src:src + n] -= 2.0 * delta * a
        # diagonal corrections
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # accept the move
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # shake‑up after a long plateau
        if no_improve > MAX_NO_IMPROVE:
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] perturbation to escape stagnation")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final secure call
    final_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Phase 0 – cheap work on a moderate size (fast exploration)
    # ------------------------------------------------------------------
    n_small = 2000
    init_deadline = start_time + 0.10 * budget_s   # 10 % of the budget
    a_small, val_small = _initial_best(n_small, start_time, init_deadline)

    # Adam with exact gradient on the small problem
    phase_a_deadline = start_time + 0.30 * budget_s   # ~20 % more
    a_small, val_small = _adam_opt_exact(a_small, n_small,
                                          phase_a_deadline,
                                          start_time,
                                          a_small, val_small)

    # Optional quick hill‑climb on the small instance
    if time.time() < phase_a_deadline:
        a_small, val_small = _hillclimb_greedy(a_small, n_small,
                                               phase_a_deadline,
                                               start_time,
                                               a_small, val_small)

    # ------------------------------------------------------------------
    # Upsample to the final size (more expressivity)
    # ------------------------------------------------------------------
    n_big = 8000
    # linear interpolation of the best small solution
    xs_big = np.linspace(0.0, 1.0, n_big)
    xs_small = np.linspace(0.0, 1.0, n_small)
    a_big = np.interp(xs_big, xs_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    val_big = _objective(a_big)
    best_a, best_val = a_big, val_big
    print("[scale‑up] upsampled solution, value =", best_val)

    # ------------------------------------------------------------------
    # Phase 1 – Adam with exact gradient on the full problem
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.80 * budget_s   # leave ~20 % for hill‑climb
    best_a, best_val = _adam_opt_exact(best_a, n_big,
                                        phase1_deadline,
                                        start_time,
                                        best_a, best_val)

    # ------------------------------------------------------------------
    # Phase 2 – Greedy hill‑climbing until the very end
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_greedy(best_a, n_big,
                                         final_deadline,
                                         start_time,
                                         best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (the secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = float("inf")

    return best_a.tolist()
