def _gradient_weighted_peaks(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    assert w.shape[0] == conv_len
    # indicator vector w (weights)
    p_rev = w[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    seg = conv_ap[n-1:2*n-1]
    g = seg[::-1]
    return g
```

"""
Propose an improved step function.

Algorithm
---------
1. **Initialisation** – start from the historic construction if it is
   available, otherwise draw a random Dirichlet (exponential) vector and
   normalise it to sum = 1.

2. **Soft‑max weighted exponential mirror descent** – at each iteration we
   compute the full self‑convolution `c = a*a`.  Let `M = max(c)`.
   A weight vector `w_k = exp(β·(c_k/M − 1))` (β ≥ 0) concentrates on the
   largest convolution entries while keeping the gradient smooth.
   The gradient of the weighted sum `∑_k w_k·c_k` w.r.t. `a` is

        g = 2·reverse( (a  *  w[::-1])[(n‑1): (2n‑1)] ).

   The update is multiplicative (exponentiated‑gradient) and the vector is
   projected back onto the simplex:

        a ← a·exp(−α·g);  a ← max(a,ε);  a ← a / sum(a)

   with a step size `α` that decays slowly and a β that is increased
   gradually (soft‑max becomes sharper).

3. **Anti‑symmetric re‑ordering** – every few hundred iterations the vector
   is reordered by pairing the largest values with the smallest ones.
   This rearrangement (a consequence of the rearrangement inequality) tends
   to lower the peak of the autocorrelation.

4. **Hill‑climbing refinement** – a lightweight local search (random swaps
   and tiny mass transfers) is applied to the best vector found so far.

All expensive operations are performed with FFTs (O(n log n)).  The routine
runs for at most the supplied time budget (default 1000 s) and returns the
best sequence it discovered.  The official objective is finally obtained by
calling the provided `evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding in FFT)."""
    return 1 << (x - 1).bit_length()


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values opposite the smallest ones (pairwise).
    This tends to flatten the autocorrelation.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2*n‑1) using a pre‑allocated FFT size.
    The returned array length is exactly 2*n‑1.
    """
    fa = np.fft.rfft(a, size_fft)
    conv_full = np.fft.irfft(fa * fa, size_fft)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]      # keep only the linear part


def _gradient_weighted_peaks(a: np.ndarray, w: np.ndarray,
                            size_fft: int) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k·a_{k‑i} for all i (indices out of range give 0).
    The computation is performed with a single FFT:
        g = reverse( (a * w[::-1])[(n‑1):(2n‑1)] )
    """
    n = a.shape[0]
    # reverse the weight vector
    w_rev = w[::-1]
    # FFT of a (already computed in the main loop may be reused, but we
    # recompute here for clarity)
    fa = np.fft.rfft(a, size_fft)
    fw = np.fft.rfft(w_rev, size_fft)
    conv = np.fft.irfft(fa * fw, size_fft)
    seg = conv[n - 1 : 2 * n - 1]          # length n
    return seg[::-1]                     # reverse to obtain gradient


# ----------------------------------------------------------------------
# Hill‑climbing refinement (same as the previous version)
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """Random swaps and small mass transfers, keeping only strict improvements."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement required
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a list of floats (the heights of the step function).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for finalisation

    # ------------------------------------------------------------------
    # Problem size – a few ten‑thousands coefficients give a good trade‑off
    # ------------------------------------------------------------------
    n = 12000                     # can be increased if time permits

    # ------------------------------------------------------------------
    # Initialise from a historic construction if available
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the new size
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, old)
        else:
            a = old.copy()
    except Exception:
        a = rng.exponential(scale=1.0, size=n)

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum                     # enforce sum = 1 (scale‑invariant)

    # ------------------------------------------------------------------
    # Pre‑compute constants for FFTs
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    size_fft = _next_pow2(3 * n - 2)   # large enough for a * w_rev convolution

    # ------------------------------------------------------------------
    # Initial evaluation (exact, via the provided evaluator)
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    alpha0 = 0.4                # initial step size for the exponentiated update
    decay = 0.995               # decay of α (slow decay)
    beta = 0.5                  # starting soft‑max sharpness
    beta_max = 30.0
    beta_inc = 0.01             # multiplicative increase per iteration
    min_iter_between_logs = 200
    iteration = 0

    # ------------------------------------------------------------------
    # First convolution (for the initial state)
    # ------------------------------------------------------------------
    fa = np.fft.rfft(a, size_fft)
    conv_full = np.fft.irfft(fa * fa, size_fft)
    conv = conv_full[:conv_len]
    cur_max = conv.max()

    # ------------------------------------------------------------------
    # Main optimisation loop (soft‑max weighted exponential mirror descent)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        M = cur_max
        # Relative weights – 1 for the maximal entry, <1 otherwise
        rel = conv / M - 1.0                 # ≤ 0, zero for the peak(s)
        w = np.exp(beta * rel)               # soft‑max like weighting

        # Gradient of the weighted sum Σ w_k c_k (c_k = (a*a)_k)
        # = 2 * Σ_k w_k a_{k-i}
        g = _gradient_weighted_peaks(a, w, size_fft) * 2.0

        # Step‑size schedule
        alpha = alpha0 * (decay ** (iteration // 2000))

        # Exponentiated (multiplicative) update
        delta = -alpha * g
        delta = np.clip(delta, -30.0, 30.0)        # avoid overflow/underflow
        a = a * np.exp(delta)

        # Project back onto the probability simplex
        a = np.maximum(a, 1e-15)                  # keep positivity
        a /= a.sum()

        # --------------------------------------------------------------
        # Re‑evaluate the new vector (full self‑convolution)
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, size_fft)
        conv_full = np.fft.irfft(fa * fa, size_fft)
        conv = conv_full[:conv_len]
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max               # because sum(a)=1

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[iter {iteration}] new best ≈ {best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic anti‑symmetric re‑ordering (may lower the peak)
        # --------------------------------------------------------------
        if iteration % 500 == 0 and iteration > 0:
            a_ord = _anti_symmetric_reorder(a)
            fa_ord = np.fft.rfft(a_ord, size_fft)
            conv_full_ord = np.fft.irfft(fa_ord * fa_ord, size_fft)
            conv_ord = conv_full_ord[:conv_len]
            val_ord = 2.0 * n * conv_ord.max()
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                a = a_ord.copy()
                conv = conv_ord
                cur_max = conv_ord.max()
                print(f"[reorder {iteration}] improved best to {best_val:.6f}")

        # --------------------------------------------------------------
        # Increase soft‑max sharpness slowly
        # --------------------------------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta * (1.0 + beta_inc))

        iteration += 1

    # ------------------------------------------------------------------
    # Hill‑climbing refinement on the best vector obtained so far
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final official evaluation (required by the harness)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] official evaluation = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    return best_a.tolist()
