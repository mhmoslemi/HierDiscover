"""
Propose a step‑function (non‑negative vector) that minimises
    f(a) = 2·n·max(conv(a, a)) / (sum a)^2 .

The algorithm works in three stages inside the 1000 s budget:

1. **Initial sweep** – generate a few parametric families (uniform,
   power‑law “U‑shapes”, exponentials, random Dirichlet draws) and keep
   the best normalised vector.

2. **Smoothed‑gradient descent** – optimise the exact objective using a
   softmax re‑parameterisation a_i = e^{x_i}/∑e^{x_j}.  The gradient of a
   *smoothed* version of the max‑convolution is obtained by weighting the
   convolution entries with w_k = exp(λ·(b_k/max_b – 1)).  An Adam optimiser
   updates the logits x; λ is slowly increased from 5 to ≈200 as time
   progresses, so the optimisation first moves quickly in a smooth landscape
   and later focuses on the true peak.

3. **Final greedy fine‑tuning** – a cheap hill‑climb that moves a tiny
   fraction of mass from the index that contributes most to the current
   peak to the index that contributes least.  Small random jitters are
   added when progress stalls.

All updates respect the simplex (non‑negative, Σa = 1); the best vector
found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, b: np.ndarray = None) -> np.ndarray:
    """Full linear convolution of `a` with `b` (or with itself if `b` is None)."""
    if b is None:
        b = a
    n, m = a.shape[0], b.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Softmax – logits → probability simplex
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Initial families – cheap sweep; updates global best
# ----------------------------------------------------------------------
def _initial_best(n: int, deadline: float):
    """Return a decent start vector and its exact objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = 2.0 * n * np.max(_full_convolution(best_a))

    def try_update(arr):
        nonlocal best_a, best_val
        a = np.asarray(arr, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = 2.0 * n * np.max(_full_convolution(a))
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(12345)

    # 1) simple deterministic families
    idx = np.arange(n, dtype=float)

    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # linear U‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # quadratic decreasing
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine “U”

    # 2) power‑law U‑shapes (edges heavier)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # 3) symmetric exponentials (V‑shapes)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.996, 0.99, 0.98, 0.96, 0.94, 0.92):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # 4) random Dirichlet draws with a few concentration parameters
    for alpha in (0.3, 0.6, 1.0, 2.0, 5.0):
        for _ in range(6):
            try_update(rng.dirichlet(np.full(n, alpha)))

    # respect the time budget – return early if we are about to overrun
    if time.time() >= deadline:
        return best_a, best_val

    return best_a, best_val


# ----------------------------------------------------------------------
# Smoothed‑gradient descent with Adam on the softmax logits
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   start_time: float, deadline: float,
                   best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on logits; uses a smoothed max‑convolution gradient."""
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)

    # initialise logits from the current best vector
    x = np.log(a0 + 1e-12)

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- construct probability vector -----
        a = _softmax(x)

        # ----- exact objective (for bookkeeping) -----
        b = _full_convolution(a)               # a * a
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, a.copy()
            print(f"[Adam {it:07d}] new best = {best_val:.6f}")

        # ----- smoothed weighting -----
        # scale differences relative to the current maximum
        ratio = b / maxb                       # in [0, 1]
        # λ grows from 5 → 200 as we progress through the allotted time
        progress = (now - start_time) / (deadline - start_time)
        lam = 5.0 + 195.0 * (progress ** 2)     # 5 → 200 (quadratic schedule)

        w = np.exp(lam * (ratio - 1.0))        # w_k = exp(λ·(b_k/maxb – 1))
        Z = w.sum()
        # ----- weighted correlation a ⋆ w (i.e. Σ_j a_j w_{j+i}) -----
        w_rev = w[::-1]                        # reversed for cross‑correlation
        corr_full = _full_convolution(a, w_rev)    # length 3n‑2
        start = len(w) - 1                     # = 2n‑2
        weighted_corr = corr_full[start:start + n] / Z

        # ----- gradient wrt a (exact for the smoothed objective) -----
        grad_a = 4.0 * n * weighted_corr       # d(2n·M)/da_i

        # ----- back‑prop through the softmax -----
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)            # dF/dx = J^T·grad_a

        # ----- Adam update on logits -----
        it += 1
        lr = 0.12 / (1.0 + it * 1e-4)          # slowly decreasing learning rate

        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climb (tiny moves)
# ----------------------------------------------------------------------
def _final_hillclimb(a_start: np.ndarray, n: int,
                     deadline: float,
                     best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    while time.time() < deadline:
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[Hill {it:07d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # locate the peak index
        k = int(np.argmax(b))

        # contributions of each a_i to that peak (a_{k-i})
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        src = int(np.argmax(rev))
        # pick a different index as target
        tmp = rev.copy()
        tmp[src] = np.inf
        dst = int(np.argmin(tmp))

        if cur[src] < 1e-14:
            # source exhausted – jitter a little if we are stuck
            if no_improve > 5000:
                delta_vec = np.random.normal(scale=0.004, size=n)
                cur = np.maximum(cur + delta_vec, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[Hill {it:07d}] jitter after stagnation")
            it += 1
            continue

        # move a small random fraction of source weight
        frac = 0.003 + 0.002 * np.random.rand()   # ≈0.3 %–0.5 %
        delta = cur[src] * frac
        cur[src] -= delta
        cur[dst] += delta

        # accept only if the objective improves
        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-15:
            # keep the move
            pass
        else:
            # revert
            cur[src] += delta
            cur[dst] -= delta

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function within the given time budget.
    Returns the best list of floats found (Σa = 1) which minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size: a few thousand entries gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if the harness
    # provides it under the name `height_sequence_1`.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # linearly interpolate to the current length
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_best(n, deadline)

    best_a = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))

    # ------------------------------------------------------------------
    # Phase‑1 – Adam optimiser on a smoothed max‑convolution
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _adam_optimize(best_a, n,
                                      start_time, phase1_deadline,
                                      best_a, best_val)

    # ------------------------------------------------------------------
    # Phase‑2 – deterministic greedy hill‑climb for fine adjustments
    # ------------------------------------------------------------------
    best_a, best_val = _final_hillclimb(best_a, n,
                                        deadline,
                                        best_a, best_val)

    # ------------------------------------------------------------------
    # Final verification through the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
