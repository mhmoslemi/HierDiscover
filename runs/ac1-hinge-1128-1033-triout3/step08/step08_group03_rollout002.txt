def _gradient_selected_peaks(a, idxs):
    n = a.shape[0]

    p = np.zeros(conv_len)

    seg = conv_ap[n - 1: 2 * n - 1]

    return g

assistantfinal**Solution Overview**

The objective  
\[
F(a)=\frac{2\,n\;\max(a*a)}{(\sum_i a_i)^2}
\]
is *scale invariant*: multiplying the whole vector by a constant does not change the score.  
Therefore we keep the vector on the probability simplex (`∑a_i = 1`).  
The score is then simply `2·n·max(conv)`, where `conv = a*a` is the full linear convolution.

The algorithm consists of three stages:

1. **Fast exponential‑mirror descent** –  
   Repeatedly minimise the maximum of the convolution.  
   In each iteration we
   * compute the full convolution by FFT,
   * select all peaks whose value is within a small relative tolerance of the global maximum,
   * obtain the sub‑gradient of the sum of those peaks (which is a sub‑gradient of the max),
   * apply a *multiplicative* (mirror‑descent) update with momentum,
   * project back onto the simplex,
   * and finally apply the classic *anti‑symmetric* reorder (largest ↔ smallest) which is known to lower central autocorrelations.

   The step size decays slowly and the tolerance is gradually tightened, giving a steady reduction of the maximal peak.

2. **Sparsification & refinement** –  
   After the first descent we zero‑out a large fraction of the smallest coefficients, re‑normalise, and run a short second descent.  
   Sparse vectors tend to have fewer pairwise overlaps, which often lowers the peak further.

3. **Local hill‑climbing** –  
   A cheap random‑swap / mass‑transfer neighbourhood search is performed until the time budget expires.  
   The improvement is evaluated with the same fast FFT‑based objective, guaranteeing many thousands of moves.

The whole procedure respects the 1000 s budget, uses only a single CPU, and relies exclusively on NumPy (no external solvers).  
All helper functions are defined at top level, and the required entry point `propose_candidate` returns the best list of heights found.

```python
"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm
---------
1. Initialise from the historic construction `height_sequence_1` if present,
   otherwise from a random Dirichlet‑like vector (exponential entries)
   and normalise to a probability simplex.

2. Fast exponential‑mirror descent:
   * full self‑convolution via FFT,
   * select all peaks within a relative tolerance of the current maximum,
   * compute the sub‑gradient of the sum of those peaks (FFT based),
   * multiplicative update with momentum,
   * project onto the simplex,
   * anti‑symmetric reorder every few iterations.
   The step size decays and the tolerance tightens over time.

3. Sparsify the best vector (keep only a fraction of the largest entries),
   re‑normalise and run a short second descent.

4. Local hill‑climbing (random swaps / small mass transfers) until the time
   budget is exhausted.

The objective `2·n·max(conv)` is computed with FFT, which is fast enough
to evaluate thousands of candidates inside the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a real vector with itself (length 2·len‑1) using FFT.
    Tiny negative values caused by numerical noise are clipped to zero.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_selected_peaks(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the sum of the convolution peaks indexed by `idxs`.

    For each i,
        g_i = Σ_{k∈idxs} a_{k-i}    (out‑of‑range indices contribute 0)

    The computation is performed with a single FFT.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1

    # indicator of the selected peaks
    p = np.zeros(conv_len, dtype=float)
    p[idxs] = 1.0

    # convolution a * reverse(p)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the middle n entries give Σ a_{k-i}
    seg = conv_ap[n - 1 : 2 * n - 1]
    g = seg[::-1]                      # reverse because of the reversal of p
    return g


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest values with the smallest ones (largest ↔ left side,
    smallest ↔ right side).  This ordering minimises the central autocorrelation
    by the rearrangement inequality.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _objective_fast(a: np.ndarray) -> float:
    """
    Fast evaluation of the score using FFT.
    The vector is assumed to be non‑negative and its sum > 0.01.
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Exponential‑mirror descent with momentum
# ----------------------------------------------------------------------
def _exp_gradient_descent(a_start: np.ndarray,
                          deadline: float,
                          rel_tol_start: float = 5e-4,
                          rel_tol_end: float = 5e-6,
                          alpha0: float = 0.4,
                          decay: float = 0.99,
                          momentum: float = 0.9,
                          reorder_interval: int = 250,
                          min_peaks: int = 10) -> tuple[np.ndarray, float]:
    """
    Multiplicative mirror‑descent on the max‑convolution objective.

    Parameters
    ----------
    a_start : np.ndarray
        Starting probability vector (∑a = 1).
    deadline : float
        Unix time at which the routine must stop.
    rel_tol_start, rel_tol_end : float
        Relative tolerance for peak selection, linearly annealed.
    alpha0 : float
        Initial step‑size for the exponential update.
    decay : float
        Multiplicative decay of the step size every 1000 iterations.
    momentum : float
        Exponential moving average coefficient for the gradient.
    reorder_interval : int
        How often (in iterations) the anti‑symmetric reorder is applied.
    min_peaks : int
        Minimum number of indices used when the max is isolated.

    Returns
    -------
    best_a : np.ndarray
        The best vector found.
    best_val : float
        Corresponding objective value.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective_fast(best_a)

    g_avg = np.zeros_like(a)

    it = 0
    while time.time() < deadline - 2.0:
        # Full self‑convolution
        conv = _full_convolution_fft(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv          # because sum(a) == 1

        # Update the best record
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[gd iter {it}] new best = {best_val:.6f}")

        # Adaptive relative tolerance (tightening)
        rel_tol = rel_tol_start + (rel_tol_end - rel_tol_start) * min(1.0, it / 2000)

        # Indices of peaks close to the maximum
        thresh = max_conv * (1.0 - rel_tol)
        idxs = np.where(conv >= thresh)[0]

        # Guarantee a minimum number of indices
        if idxs.size < min_peaks:
            k = min(min_peaks, conv.size)
            idxs = np.argpartition(-conv, k - 1)[:k]

        # Gradient of the sum of selected peaks
        g = _gradient_selected_peaks(a, idxs)

        # Momentum
        g_avg = momentum * g_avg + (1.0 - momentum) * g

        # Step‑size schedule (decay every 1000 iterations)
        alpha = alpha0 * (decay ** (it // 1000))

        # Multiplicative update, clipped for numerical safety
        delta = -alpha * g_avg
        delta = np.clip(delta, -20.0, 20.0)          # exp(±20) ≈ 5·10⁸
        a = a * np.exp(delta)

        # Projection onto the simplex (keep positivity)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # Periodic anti‑symmetric re‑ordering
        if (it + 1) % reorder_interval == 0:
            a = _anti_symmetric_reorder(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sparsification
# ----------------------------------------------------------------------
def _sparsify(a: np.ndarray, keep_frac: float) -> np.ndarray:
    """
    Keep only the largest `keep_frac` fraction of coefficients,
    zero out the rest and re‑normalise.
    """
    n = a.shape[0]
    k = max(1, int(keep_frac * n))
    # indices of the k largest entries
    idx = np.argpartition(-a, k - 1)[:k]
    mask = np.zeros(n, dtype=bool)
    mask[idx] = True
    a_new = np.where(mask, a, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        return a                     # fallback – keep the original
    a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb_refine(a_start: np.ndarray,
                       cur_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Random local search (swaps and small mass transfers) that accepts only
    strict improvements.  The objective is evaluated with the fast FFT version.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline - 0.5:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-15:
                continue
            delta = a[i] * 0.05 * rng.random()
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _objective_fast(new_a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a.copy()
            a = new_a
            print(f"[hill climb] new best = {best_val:.6f}")

    return best_a, best_val

    # Phase 2 – sparsify and run a short second descent
    # ------------------------------------------------------------------
    for keep_frac in (0.15, 0.10):
        if time.time() >= deadline - 100:
            break
        a_sparse = _sparsify(best_a, keep_frac)
        a_sparse = _anti_symmetric_reorder(a_sparse)
        a_gd, val_gd = _exp_gradient_descent(a_sparse, deadline)
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[sparsify keep={keep_frac:.2f}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final evaluation with the official function (for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] score reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required by the harness
    return best_a.tolist()
```
