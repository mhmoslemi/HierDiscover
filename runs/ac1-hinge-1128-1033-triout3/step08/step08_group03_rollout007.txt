def weighted_gradient(a, w):
    # a length n, w length 2n-1
    # compute full convolution using FFT or np.convolve (but np.convolve O(n^2) would be too slow)
    # use same FFT approach as before: next_pow2.
    n = a.shape[0]
    total_len = n + (2*n-1) - 1 = 3*n - 2
    size = _next_pow2(total_len)
    Fa = np.fft.rfft(a, size)
    Fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(Fa * Fw, size)[:total_len]
    # gradient indices corresponding to i=0..n-1: conv[n-1:2*n-1] (size n)
    grad = 2.0 * conv[n-1:2*n-1]
    return grad

def proj_simplex(v):
   u = np.sort(v)[::-1]; cssv = np.cumsum(u) - 1; rho = np.where(u - cssv/(np.arange(len(v))+1) > 0)[0][-1]; theta = cssv[rho]/(rho+1); w = np.maximum(v - theta, 0); return w

def _proj_simplex(v):
    # projecting onto simplex {x >=0, sum=1}
    # Implementation from https://github.com/pybrain/pybrain... but implement directly
    # ... 
    n = v.shape[0]

    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:

    w = np.maximum(v - theta, 0)
    return w

   * Use the historic construction `height_sequence_1` if it exists,
     otherwise draw a random Dirichlet (exponential) vector.
   * Normalise the vector to the required sum `sqrt(2·n)` (the evaluation
     function returns the maximal self‑convolution value under this
     normalisation).

2. **Soft‑max weighted mirror‑descent**  
   * At each iteration the full self‑convolution `c = a * a` is computed
     with an FFT.  
   * A weight vector `w = exp( β·(c – max(c)) )` emphasises the
     convolution entries that are close to the current maximum
     (β grows slowly, turning the soft‑max into a hard max).  
   * The gradient of the weighted sum `∑ w_k·c_k` w.r.t. `a` can be written
     as a single convolution, also evaluated via FFT.  
   * A **multiplicative** update `a ← a·exp( –α·grad/Σa )` keeps the
     coefficients non‑negative.  After the update the vector is projected
     onto the simplex of sum `sqrt(2·n)`.  
   * Step‑size `α` decays and `β` increases throughout the run.

3. **Anti‑symmetric re‑ordering**  
   Every few thousand iterations the current vector is reordered by pairing
   the largest values with the smallest ones (the rearrangement inequality
   lowers the central autocorrelation).  The reordered vector is kept if it
   improves the objective.

4. **Hill‑climbing / simulated‑annealing refinement**  
   A lightweight local search (random swaps and tiny mass transfers)
   explores the neighbourhood of the best vector found so far.  Worse
   moves are occasionally accepted with a temperature‑controlled
   probability to escape shallow local minima.

The whole procedure respects the 1000 s budget, calls `evaluate_sequence`
only for logging, and returns the best list found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a real vector with itself.
    Length = 2·len(a) – 1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # clip tiny negatives
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g = 2·∑_k w_k·a_{k‑i}  for all i (0 ≤ i < n).

    The computation is performed with a single FFT:
        g = 2· (a * w)[n‑1 : 2·n‑1] .
    """
    n = a.shape[0]
    size = _next_pow2(3 * n - 2)            # enough for a (n) + w (2n‑1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)[: 3 * n - 2]
    grad = 2.0 * conv[n - 1: 2 * n - 1]     # extract the n relevant entries
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, ∑ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimizes
    `evaluate_sequence` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand coefficients give a good trade‑off
    # --------------------------------------------------------------
    n = 8000
    target_sum = math.sqrt(2.0 * n)          # sum needed for the objective = max(conv)

    # --------------------------------------------------------------
    # Initialise from a known construction if possible
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = rng.exponential(scale=1.0, size=n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
        s = 1.0
    a = a / s * target_sum                      # normalise to required sum

    # --------------------------------------------------------------
    # Keep best solution seen so far
    # --------------------------------------------------------------
    best_a = a.copy()
    best_val = _full_convolution(best_a).max()
    print(f"[init] n={n}, initial max‑conv = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – a few random restarts to avoid a poor start
    # --------------------------------------------------------------
    for tr in range(4):
        if time.time() >= deadline - 200:
            break
        cand = rng.exponential(scale=1.0, size=n)
        cand = cand / cand.sum() * target_sum
        val = _full_convolution(cand).max()
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand.copy()
            print(f"[rand restart {tr}] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – soft‑max weighted mirror descent
    # --------------------------------------------------------------
    if time.time() < deadline - 150:
        # hyper‑parameters
        alpha0 = 0.6                # initial step‑size
        alpha_decay = 0.95          # decay factor per 5000 iterations
        beta = 2.0                  # initial soft‑max sharpness
        beta_mult = 1.0005          # increase per iteration
        iter_cnt = 0

        # pre‑compute FFT size (enough for a (n) + w (2n‑1))
        fft_size = _next_pow2(3 * n - 2)

        while time.time() < deadline - 5:
            # ---- FFT of current vector -------------------------------------------------
            Fa = np.fft.rfft(a, fft_size)

            # ---- self‑convolution a*a (full length 2n‑1) ---------------------------------
            conv_full = np.fft.irfft(Fa * Fa, fft_size)[: 2 * n - 1]
            max_conv = conv_full.max()

            # ---- weight vector (soft‑max on the convolution) ---------------------------
            w = np.exp(beta * (conv_full - max_conv))     # values in (0,1]

            # ---- gradient of the weighted sum -----------------------------------------
            Fw = np.fft.rfft(w, fft_size)
            grad_full = np.fft.irfft(Fa * Fw, fft_size)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1: 2 * n - 1]      # length n

            # ---- multiplicative mirror‑descent update ---------------------------------
            # make the gradient dimensionless (divide by target_sum)
            grad_norm = grad / target_sum
            alpha = alpha0 * (alpha_decay ** (iter_cnt // 5000))
            delta = -alpha * grad_norm
            delta = np.clip(delta, -20.0, 20.0)          # avoid overflow in exp
            a = a * np.exp(delta)

            # ---- projection onto the simplex (sum = target_sum) -----------------------
            a = _project_simplex(a) * target_sum

            # ---- occasional logging ----------------------------------------------------
            if iter_cnt % 20 == 0:
                if max_conv < best_val - 1e-12:
                    best_val = max_conv
                    best_a = a.copy()
                    print(f"[gd iter {iter_cnt}] new best = {best_val:.6f}")

            # ---- anti‑symmetric re‑ordering every 2000 iterations --------------------
            if iter_cnt % 2000 == 0 and iter_cnt > 0:
                a_ord = _anti_symmetric_reorder(a)
                a_ord = _project_simplex(a_ord) * target_sum
                conv_ord = np.fft.irfft(
                    np.fft.rfft(a_ord, fft_size) *
                    np.fft.rfft(a_ord, fft_size),
                    fft_size)[: 2 * n - 1]
                max_ord = conv_ord.max()
                if max_ord < best_val:
                    best_val = max_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[gd reorder {iter_cnt}] improved to {best_val:.6f}")

            # ---- update soft‑max temperature -----------------------------------------
            beta *= beta_mult

            iter_cnt += 1

    # --------------------------------------------------------------
    # Phase 2 – hill‑climbing / simulated annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        cur_a = best_a.copy()
        cur_val = best_val
        rng = np.random.default_rng(seed + 123456)
        start_hc = time.time()
        T0 = 1e-2                         # initial temperature for SA

        while time.time() < deadline:
            # propose a local move (swap or mass transfer)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

            if rng.random() < 0.5:
                # pure swap
                new_a = cur_a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]
            else:
                # small mass transfer i → j
                if cur_a[i] <= 1e-15:
                    continue
                delta = cur_a[i] * 0.1 * rng.random()   # up to 10 % of a[i]
                new_a = cur_a.copy()
                new_a[i] -= delta
                new_a[j] += delta

            # re‑project onto the simplex (sum = target_sum)
            new_a = np.maximum(new_a, 0.0)
            s = new_a.sum()
            if s <= 0:
                continue
            new_a = new_a * (target_sum / s)

            # evaluate max‑convolution of the candidate
            Fa = np.fft.rfft(new_a, fft_size)
            conv_full = np.fft.irfft(Fa * Fa, fft_size)[: 2 * n - 1]
            new_val = conv_full.max()

            # accept if strictly better, otherwise possibly accept (SA)
            if new_val < cur_val - 1e-12:
                cur_a, cur_val = new_a, new_val
                if new_val < best_val:
                    best_val = new_val
                    best_a = new_a.copy()
                    print(f"[refine] new best = {best_val:.6f}")
            else:
                # simulated‑annealing acceptance
                elapsed = time.time() - start_hc
                T = max(T0 * (1.0 - elapsed / (deadline - start_hc)), 1e-9)
                if rng.random() < math.exp(-(new_val - cur_val) / T):
                    cur_a, cur_val = new_a, new_val

    # --------------------------------------------------------------
    # Final evaluation & return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
