"""
Propose a non‑negative step‑function that (approximately) minimizes the
``evaluate_sequence`` objective.

Algorithm (different from the reference implementation)
-------------------------------------------------------
1. **Initial seeding** – a small set of diverse candidates (uniform,
   previous best, a few Dirichlet draws).  The cheapest evaluator
   ``_fast_evaluate`` (2·n·max(conv) because the sum is 1) is used to pick the
   best initial vector.

2. **Variance‑flattening gradient descent** – we minimise the variance of
   the convolution ``a*a``:
        V(a) = Σ_k ( (a*a)_k – 1/(2n‑1) )² .
   The gradient of ``V`` can be written as a correlation of ``a`` with the
   residual ``r = a*a – mean`` and is computed by a single FFT‑based
   convolution.  Several short restarts with a little random noise are
   performed; after each step the vector is projected back onto the
   probability simplex.

3. **Exact mirror‑descent on the true max‑convolution** – using the true
   sub‑gradient of the infinity‑norm of the convolution (the “worst’’ index)
   we perform an exponential update on the simplex.  Tens of thousands of
   cheap O(n) iterations are executed; the step size decays slowly.

4. **Final cheap hill‑climbing** – random swaps / mass transfers that are
   accepted only when they strictly improve the objective.

All steps respect the 1000 s budget; the best vector found is finally
checked with the official ``evaluate_sequence`` function and returned as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures) as required.
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of a power‑of‑2 size.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # tiny negative values can appear because of round‑off
    np.maximum(conv, 0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already satisfy
    a_i ≥ 0 and Σ a_i = 1.  Returns 2·n·max(conv) (the denominator is 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the O(n) algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _variance_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the variance of the convolution a*a.
    V(a) = Σ_k ( (a*a)_k – μ )²,   μ = 1/(2n‑1) because Σ (a*a)_k = 1.
    The gradient is
        g = 4 * ( r ★ a_rev )
    where r = conv – μ and ★ denotes a correlation (implemented via a
    convolution with the reversed vector).
    """
    n = a.shape[0]
    mu = 1.0 / (2 * n - 1)
    r = conv - mu                     # residual, length 2n‑1
    a_rev = a[::-1]                  # reversed a, length n

    size = _next_pow2(len(r) + len(a_rev) - 1)
    fr = np.fft.rfft(r, size)
    fa_rev = np.fft.rfft(a_rev, size)
    grad_full = np.fft.irfft(fr * fa_rev, size)[: len(r) + len(a_rev) - 1]

    # the required segment is grad_full[n‑1 : n‑1 + n]
    grad = 4.0 * grad_full[n - 1: n - 1 + n]
    return grad


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector is kept on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:               # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
