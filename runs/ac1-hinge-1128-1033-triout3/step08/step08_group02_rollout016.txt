Algorithm:
- Choose length n (≈8000).
- Initialise with previous solution if available, else uniform.
- Run several restarts of projected Adam on log‑softmax parametrisation,
  using a smoothed max‑convolution (soft‑max) with temperature τ that
  anneals from 0.1 to 0.001.
  The gradient of the smooth objective is computed via FFT‑based
  convolutions (O(n log n)).
- Keep the best vector (minimum 2·n·max(conv)).
- Finish with a short deterministic greedy hill‑climb that moves a
  small fraction of mass from the index contributing most to the current

- Return the best vector found.

import time

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of length `size`."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv

def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector that sums to 1."""
    mx = np.max(x)
    e = np.exp(x - mx)
    s = e.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.shape[0])
    return e / s

def _objective_from_a(a: np.ndarray, n: int) -> float:
    """Exact value returned by evaluate_sequence for a vector already normalized to sum 1."""
    b = _full_conv_fft(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * b.max())

def _gradient_step(x: np.ndarray, n: int, tau: float, size1: int, size2: int,
                   m: np.ndarray, v: np.ndarray, beta1: float, beta2: float,
                   eps: float, t: int, lr: float):
    """Perform one Adam step on the soft‑max approximation of max‑convolution."""
    # a = softmax(x)
    a = _softmax(x)

    # 1) b = a * a (full conv)
    b = _full_conv_fft(a, size1)          # length 2n‑1
    b_max = b.max()

    # 2) soft‑max probabilities over the convolution entries
    #    w_k = exp((b_k - b_max) / tau)  (numerically stable)
    w = np.exp((b - b_max) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        p = np.full_like(w, 1.0 / w.shape[0])
    else:
        p = w / w_sum                     # shape (2n‑1,)

    # 3) gradient w.r.t a: g_a = 2 * (p ⋆ a_rev)[valid], where ⋆ is convolution
    a_rev = a[::-1]
    # full convolution of p with a_rev (size size2)
    fa = np.fft.rfft(p, size2)
    fb = np.fft.rfft(a_rev, size2)
    conv_full = np.fft.irfft(fa * fb, size2)[: (2 * n - 1) + n - 1]   # length 3n‑2
    # valid slice   indices n‑1 … n‑1+n‑1
    conv_valid = conv_full[n - 1 : n - 1 + n]
    g_a = 2.0 * conv_valid               # shape (n,)

    # 4) convert gradient to x‑space (softmax Jacobian)
    dot = np.dot(a, g_a)
    g_x = a * (g_a - dot)                # shape (n,)

    # 5) Adam update on x
    m = beta1 * m + (1.0 - beta1) * g_x
    v = beta2 * v + (1.0 - beta2) * (g_x * g_x)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    x_new = x - step
    return x_new, m, v, a, b_max

def _greedy_refine(a: np.ndarray, n: int, deadline: float, start: float):
    """A short deterministic greedy hill‑climb that moves a small fraction
    of mass from the most harmful index to the least harmful one."""
    cur = a.copy()
    best = cur.copy()
    best_val = _objective_from_a(best, n)
    no_improve = 0
    IMPATIENT = 20000
    max_frac_start, max_frac_end = 0.3, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        # recompute convolution
        b = _full_conv_fft(cur, _next_pow2(2 * n - 1))
        cur_max = b.max()
        cur_val = float(2.0 * n * cur_max)

        if cur_val < best_val - 1e-15:
            best_val, best = cur_val, cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # schedule for transferred fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # index of the peak
        k = int(np.argmax(b))

        # reverse slice contributing to the peak
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        rev = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        i = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                # add a tiny jitter to escape stagnation
                cur += np.random.normal(scale=1e-3, size=n)
                cur = np.maximum(cur, 0.0)
                cur /= cur.sum()
                no_improve = 0
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:

        cur[i] -= delta
        cur[j] += delta

        else:

            cur[j] -= delta

Algorithm
---------

2. Initialise a probability vector `a` (sum = 1).  Parameterise it by logits

    """Full linear convolution a * a using FFT of length `size`."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    s = e.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.shape[0])
    return e / s


def _objective_from_a(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalized vector."""
    b = _full_conv_fft(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# One Adam step on smoothed max‑convolution
# ----------------------------------------------------------------------


def _adam_step(x: np.ndarray, n: int, tau: float,
               size1: int, size2: int,
               m: np.ndarray, v: np.ndarray,
               beta1: float, beta2: float,
               eps: float, t: int, lr: float):
    """Perform a single Adam update on the soft‑max approximation.

    Returns
    -------
    x_new : np.ndarray
        Updated logits.
    m, v : np.ndarray
        Updated Adam moments.
    a    : np.ndarray
        Probability vector corresponding to *old* logits (used for evaluation).
    cur_max : float
        Maximum entry of the convolution of the old `a`.
    """
    # probability vector for the current logits
    a = _softmax(x)

    # 1) full self‑convolution a*a
    b = _full_conv_fft(a, size1)          # length 2n‑1
    cur_max = b.max()

    # 2) smooth max via soft‑max (temperature τ)
    w = np.exp((b - cur_max) / tau)       # all ≤ 1, numerically safe
    w_sum = w.sum()
    if w_sum == 0.0:
        p = np.full_like(w, 1.0 / w.shape[0])
    else:
        p = w / w_sum                       # ∑p = 1

    # 3) gradient w.r.t. a: g_a = 2·(p ⋆ a_rev)[valid]
    a_rev = a[::-1]
    fa = np.fft.rfft(p, size2)
    fb = np.fft.rfft(a_rev, size2)
    conv_full = np.fft.irfft(fa * fb, size2)[: (2 * n - 1) + n - 1]   # length 3n‑2
    # valid slice corresponds to indices n‑1 … n‑1+n‑1
    conv_valid = conv_full[n - 1: n - 1 + n]
    g_a = 2.0 * conv_valid                 # shape (n,)

    # 4) map gradient to logits (soft‑max Jacobian)
    dot = np.dot(a, g_a)
    g_x = a * (g_a - dot)                  # shape (n,)

    # 5) Adam update on logits
    m = beta1 * m + (1.0 - beta1) * g_x
    v = beta2 * v + (1.0 - beta2) * (g_x * g_x)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    x_new = x - step

    return x_new, m, v, a, cur_max


# ----------------------------------------------------------------------
# Small deterministic hill‑climb used as a final polish
# ----------------------------------------------------------------------


def _greedy_refine(a_start: np.ndarray, n: int,
                   deadline: float, start: float):
    """Move a tiny fraction of mass from the worst to the best index."""
    cur = a_start.copy()
    best = cur.copy()
    best_val = _objective_from_a(best, n)

    no_improve = 0
    IMPATIENT = 20000
    max_frac_start, max_frac_end = 0.3, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_conv_fft(cur, _next_pow2(2 * n - 1))
        cur_max = b.max()
        cur_val = float(2.0 * n * cur_max)

        if cur_val < best_val - 1e-15:
            best_val, best = cur_val, cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # anneal transferred fraction
        prog = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # index producing the peak
        k = int(np.argmax(b))

        # contribution to the peak (rev_i = a_{k-i})
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        rev = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        i = int(np.argmax(rev))               # source (largest contribution)
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))           # target (smallest contribution)

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                cur += np.random.normal(scale=1e-3, size=n)
                cur = np.maximum(cur, 0.0)
                cur /= cur.sum()
                no_improve = 0
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        # accept only improvements
        b_new = _full_conv_fft(cur, _next_pow2(2 * n - 1))
        new_val = float(2.0 * n * b_new.max())
        if new_val < cur_val - 1e-15:
            # keep move
            pass
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Return a list of non‑negative heights that (hopefully) beats the previous best.

    The routine respects the 1000 s budget, prints intermediate progress,
    and finally returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Determine problem size – reuse previous solution if available
    # --------------------------------------------------------------
    old_seq = globals().get('height_sequence_1')
    if isinstance(old_seq, (list, tuple, np.ndarray)):
        try:
            old = np.asarray(old_seq, dtype=float)
            n = max(5000, len(old))
            if len(old) != n:
                old_x = np.linspace(0.0, 1.0, len(old))
                new_x = np.linspace(0.0, 1.0, n)
                a0 = np.interp(new_x, old_x, old)
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
            else:
                a0 = np.maximum(old, 0.0)
                if a0.sum() == 0.0:
                    a0 = np.full(n, 1.0 / n)
                else:
                    a0 /= a0.sum()
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        n = 8000
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_val = _objective_from_a(best_a, n)
    print(f"[init] n = {n}, initial value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Pre‑compute FFT lengths
    # --------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)          # for a*a
    size2 = _next_pow2(3 * n - 2)          # for p * a_rev

    # --------------------------------------------------------------
    # Adam optimisation on the smoothed objective
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.6
    tau_start, tau_end = 0.12, 0.001

    restart = 0
    while True:
        now = time.time()
        if now >= deadline - 2.0:
            break

        restart += 1
        # initialise logits – a little perturbation of the current best
        x = np.log(best_a + 1e-12) + 0.1 * np.random.randn(n)
        m = np.zeros(n)
        v = np.zeros(n)
        t_iter = 1

        # give each restart a modest time slice (≈ 0.8 s) or less
        restart_deadline = min(deadline, now + 0.8)

        while True:
            now = time.time()
            if now >= restart_deadline:
                break

            # linear annealing of temperature and learning rate
            prog = (now - start_time) / (deadline - start_time)
            tau = tau_start * (1.0 - prog) + tau_end * prog
            lr = lr0 * (1.0 - prog) + 0.01 * prog

            x, m, v, a, cur_max = _adam_step(
                x, n, tau, size1, size2,
                m, v, beta1, beta2, eps, t_iter, lr)

            cur_val = float(2.0 * n * cur_max)

            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[restart {restart:02d} it {t_iter:06d}] new best = {best_val:.6f}")

            t_iter += 1

    # --------------------------------------------------------------
    # Final deterministic hill‑climb for fine tuning
    # --------------------------------------------------------------
    hill_start = time.time()
    best_a, best_val = _greedy_refine(best_a, n, deadline, hill_start)

    # --------------------------------------------------------------
    # Final verification using the provided secure wrapper
    # --------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
