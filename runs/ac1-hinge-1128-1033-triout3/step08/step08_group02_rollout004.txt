"""
Optimise a non‑negative step function to minimise the objective
returned by `evaluate_sequence`.

The algorithm works inside the allotted budget (≈1000 s) and consists of
multiple *random restarts*.  Each restart performs:

1. **Random initialisation** – a Dirichlet draw with a randomly chosen
   concentration parameter (this yields a wide variety of shapes,
   from very spiky to almost uniform).

2. **Exponentiated‑gradient descent** – a multiplicative update using
   the *sub‑gradient of the current maximal convolution entry*.  To
   avoid being trapped by a single peak we aggregate the sub‑gradients
   of the *top‑K* convolution values (K≈3).  A linear step‑size schedule
   drives the updates from aggressive (≈0.15) to very fine (≈5e‑4).

3. **Deterministic greedy hill‑climb** – repeatedly move a fraction of
   the mass from the index that contributes most to the active peak
   to the index that contributes the least.  The transfer fraction
   anneals from 30 % to 0.1 % and the move is kept only if it strictly
   improves the objective.

4. **Tiny uniform mixing** – finally test a few convex combinations
   with the uniform vector; occasionally this reduces the peak a bit.

All phases respect the overall time limit, keep the vector on the
probability simplex (non‑negative, sum = 1) and maintain the best vector
found across all restarts.  If the target value (1.503) is reached the
search terminates early.

The code uses only `numpy` and the provided `evaluate_sequence`
function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)   # sum(a)=1 ⇒ denominator = 1


# ----------------------------------------------------------------------
# Simple parametric families – a quick fallback if nothing else works
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector (uniform or a simple parametric family)."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(candidate):
        nonlocal best_a, best_val
        cand = np.asarray(candidate, dtype=np.float64)
        if cand.ndim != 1 or cand.shape[0] != n:
            return
        if np.any(cand < 0):
            return
        s = cand.sum()
        if s <= 0:
            return
        cand = cand / s
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()

    rng = np.random.default_rng(0)

    idx = np.arange(n, dtype=float)
    # a couple of deterministic shapes
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # V‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # quadratic decay
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # a few random Dirichlet draws
    for _ in range(8):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (top‑K sub‑gradient)
# ----------------------------------------------------------------------
def _expgrad_phase(a0: np.ndarray,
                   n: int,
                   start_time: float,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   top_k: int = 3,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent using the top‑K active peaks."""
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 250_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        # record best
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---- aggregated sub‑gradient from the K largest convolution entries ----
        # find indices of the K largest values (unsorted)
        if top_k == 1:
            ks = [int(np.argmax(b))]
        else:
            ks = np.argpartition(b, -top_k)[-top_k:]
        g = np.zeros_like(cur)

        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            # contribution a_{k-i}
            g[idx] += 2.0 * cur[k - idx]

        # step‑size schedule (linear decay)
        progress = (now - start_time) / (deadline - start_time)
        eta = (1.0 - progress) * 0.12 + progress * 0.0005   # 0.12 → 5e‑4

        # exponentiated update (preserves non‑negativity)
        cur = cur * np.exp(-eta * g)

        # renormalise to Σa = 1
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # occasional jitter if stuck
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.009, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climb – deterministic mass transfer
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start_time: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     start_frac: float = 0.30,
                     end_frac: float = 0.001,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Move mass from the worst contributor to the best; accept only improvements."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # annealing of the transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        # locate the peak index
        k = int(np.argmax(b))

        # slice a_{k-i}
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev = np.zeros_like(cur)
        rev[idx] = cur[k - idx]

        # source = index with largest contribution to the peak
        src = int(np.argmax(rev))

        # target = index with smallest contribution (excluding source)
        rev_tmp = rev.copy()
        rev_tmp[src] = np.inf
        tgt = int(np.argmin(rev_tmp))

        if cur[src] <= 1e-15:
            # source depleted – occasional random shake
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        # trial move
        cur[src] -= delta
        cur[tgt] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # accept – nothing else to do
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combinations (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

            target_val=TARGET
        )

        # ---- brief hill‑climb (few thousand moves) ----
        # reuse the same slice (if time permits)
        a, _ = _hillclimb_phase(
            a, n,
            restart_start,
            restart_deadline,
            best_a,
            best_val,
            start_frac=0.25,
            end_frac=0.001,
            target_val=TARGET
        )

        # ---- final tiny mixing ----
        a, val = _mix_with_uniform(a, n)

        # ---- update global best ----
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Restart {restart_counter:3d}] new global best = {best_val:.6f}")

        # early exit if we already meet the target
        if best_val <= TARGET:
            print("[Target reached] exiting early.")
            break

        restart_counter += 1

    # ------------------------------------------------------------------
    # Final verification via the supplied secure wrapper (optional)
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return a plain Python list (sum will be ≈1)
    return best_a.tolist()
