"""
Propose a step‑function (non‑negative vector) that minimises
    evaluate_sequence(sequence)

The algorithm is a multi‑stage optimiser:

0.  Initialise a diverse pool of candidates (uniform, exponential,
    several power‑law shapes and an anti‑symmetric ordering).
1.  Mirror‑descent on the true L∞‑objective (gradient from the active
    set of convolution peaks).  A momentum term and a slowly tightening
    tolerance accelerate convergence.
2.  Smooth‑max surrogate (log‑sum‑exp) minimisation with L‑BFGS‑B.
    Several random restarts are performed; after each run the
    anti‑symmetric re‑ordering heuristic is applied.
3.  Targeted local search: the lag that attains the current maximum
    convolution is identified, the most responsible pair of indices is
    selected, and a small amount of mass is transferred to a low‑impact
    index.  Better solutions are accepted immediately; uphill moves are
    occasionally accepted following a simple annealing schedule.

All updates keep the vector on the probability simplex (non‑negative,
unit sum).  The best vector found before the time budget expires is
returned as a plain Python list.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # remove tiny negatives
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values on the left and the smallest on the right,
    i.e. pair a large coefficient with a small one when the sequence is
    read in reverse order.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # large → left side
        res[n - 1 - i] = s[i]     # small → right side
    if n % 2:
        res[half] = s[half]
    return res


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)     (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # self‑convolution
    conv = _full_convolution(a)
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t a (see derivation in the original code)
    weights = np.exp(shifted)
    p = weights / weights.sum()                       # distribution over lag indices

    # convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries give ∂L/∂a_i (the convolution is symmetric)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                      # ∂L/∂a_i

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses module globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

    """

    w = w_init.copy()

    # ε‑schedule – geometric from 5·10⁻³ down to 10⁻¹² (inclusive)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=11)

    best_a = None

            break

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,

                       jac=True,
                       options={'maxiter': 1500,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # back to simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())

# Sub‑gradient descent on the true L∞‑objective (with momentum)
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} a_{k‑i} using a single FFT.
    ``active_idx`` are convolution indices (0 … 2n‑2) belonging to the
    active set.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n-2) - i] for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent_improved(a_start: np.ndarray, deadline: float):

    Exponentiated mirror descent with momentum on the L∞‑objective.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    momentum = np.zeros_like(a)
    beta = 0.9                       # momentum decay
    alpha0 = 0.8                     # initial step‑size factor
    rel_tol = 0.05                   # start with a generous active‑set tolerance

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten tolerance every few thousand iterations

        a = a * np.exp(-alpha * momentum)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:

            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

            if it % 2000 == 0:

                    best_a = a_ord.copy()

        max_conv = conv[max_idx]

        # indices that actually participate in this lag
        start = max(0, max_idx - (n - 1))
        end = min(n - 1, max_idx)
        i_idx = np.arange(start, end + 1)
        j_idx = max_idx - i_idx
        # contribution of each pair (i, j)
        pairs_contrib = cur_a[i_idx] * cur_a[j_idx]

        # choose a source index weighted by its contribution
        if pairs_contrib.sum() == 0.0:
            break
        probs = pairs_contrib / pairs_contrib.sum()
        src = rng.choice(i_idx, p=probs)

        # compute per‑index product for the *same* lag (used for target selection)
        prod = np.zeros_like(cur_a)
        ind = np.arange(n)
        partner = max_idx - ind
        mask = (partner >= 0) & (partner < n)
        prod[mask] = cur_a[mask] * cur_a[partner[mask]]

        # prefer a target with a small product (bottom 20 %)
        thresh = np.percentile(prod, 20)
        low_candidates = np.where(prod <= thresh)[0]
        # avoid picking the source again
        low_candidates = low_candidates[low_candidates != src]
        if low_candidates.size == 0:
            # fall back to any other index
            low_candidates = np.where(np.arange(n) != src)[0]
        tgt = rng.choice(low_candidates)

        # decide how much mass to move (5–15 % of source mass)
        delta = cur_a[src] * rng.uniform(0.05, 0.15)
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[src] -= delta
        new_a[tgt] += delta

        # re‑project onto simplex (numerical safety)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = evaluate_sequence(new_a.tolist())
        if new_val < cur_val_local:
            accept = True
        else:
            # annealing acceptance
            T = T0 * (0.9995 ** it)
            accept = np.exp(-(new_val - cur_val_local) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val_local = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[local search] new best = {best_val:.6f}")

        # occasional re‑ordering (helps escape plateaus)
        if it % 300 == 0:
            a_ord = _anti_symmetric_reorder(cur_a)
            val_ord = evaluate_sequence(a_ord.tolist())
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                cur_a = a_ord.copy()
                cur_val_local = val_ord
                print(f"[local reorder] improved to {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to generate power‑law shaped candidates (stage 0)
# ----------------------------------------------------------------------
def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (dist_to_end+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between expressivity
    # and compute time.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from a previously known construction if available,
    # otherwise fall back to a uniform vector.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)

    # ensure simplex
    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 1e-12 else np.full(n, 1.0 / n)

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – cheap diverse candidates
    # ------------------------------------------------------------------
    # (i) exponential random vectors
    for tr in range(5):
        if time.time() >= deadline - 250:
            break
        cand = rng.exponential(scale=1.0, size=n)
        cand /= cand.sum()
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # (ii) symmetric power‑law shapes
    for p in (0.25, 0.5, 0.75, 1.0, 1.5):
        if time.time() >= deadline - 250:
            break
        cand = _powerlaw_candidate(p, n)
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[power‑law p={p:.2f}] new best = {best_val:.6f}")

    # (iii) anti‑symmetric ordering of the uniform vector
    cand = _anti_symmetric_reorder(np.full(n, 1.0 / n))
    val = evaluate_sequence(cand.tolist())
    if val < best_val:
        best_val, best_a = val, cand.copy()
        print(f"[anti‑sym uniform] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – mirror‑descent on the true L∞‑objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:
        a_sgd, val_sgd = _subgradient_descent_improved(best_a, deadline)
        if val_sgd < best_val:
            best_val, best_a = val_sgd, a_sgd.copy()
            print(f"[sub‑gradient] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max surrogate optimisation (L‑BFGS) with restarts
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 120 and restarts < 12:
        restarts += 1
        # initialise soft‑max parameters from the current best (+ jitter)
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.30, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[restart {restarts}] L‑BFGS improved to {best_val:.6f}")

        # one extra anti‑symmetric sweep after each restart
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[restart {restarts}] reorder improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – targeted local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _targeted_local_search(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
