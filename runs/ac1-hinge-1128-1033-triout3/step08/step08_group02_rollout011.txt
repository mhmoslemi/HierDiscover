def _compute_gradient(a, c, tau):
    maxc = c.max()
    if maxc == 0:
        return np.zeros_like(a)
    shift = (c - maxc) / tau
    # clip shift to avoid overflow
    # maybe shift min -700 ensures exp not underflow to zero
    np.maximum(shift, -700, out=shift)
    w = np.exp(shift)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        w_sum = 1.0
    w /= w_sum

    a_rev = a[::-1]
    L = len(w)
    size = 1 << ((L + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, size)
    A_rev = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(W * A_rev, size)
    start = len(a_rev) - 1  # n-1
    grad = 2.0 * conv[start:start+len(a)]
    return grad

"""
Step‑function optimizer for `evaluate_sequence`.

The goal is to minimise

        f(a) = 2·n·max(conv(a,a)) / (sum a)²

Because the objective is scale‑invariant we work with Σa = 1 and
minimise 2·n·max(conv(a,a)).  The optimiser consists of four stages:

* **Stage‑0 – wide initial sweep** – evaluate a handful of deterministic
  families and many random Dirichlet draws (different concentration
  parameters).  The best vector is kept.

* **Stage‑A – smoothed exponentiated‑gradient descent** – treat the
  max‑convolution entry as a (sub‑)gradient of a *soft‑max* of all
  convolution entries.  A temperature τ is slowly reduced, turning the
  weighted gradient from a global (almost uniform) direction into a
  genuine sub‑gradient of the active peak.  An exponentiated‑gradient
  update (multiplicative) keeps the variables non‑negative; after each
  step we renormalise to Σa = 1.

* **Stage‑B – deterministic greedy hill‑climb** – repeatedly move a
  small fraction of the mass from the index that contributes most to the
  current peak to the index that contributes least.  Only improving
  moves are kept.  The transfer fraction decays from 30 % to 0.1 %.

* **Stage‑C – final uniform mixing** – test a few convex combinations
  with the uniform vector; this often removes a residual spike.

All stages respect the 1000 s wall‑time budget, keep the best vector
found and finally return it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Initial candidate generation (wide sweep)
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)                # uniform
    best_val = _objective(best_a, n)

    def try_update(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # deterministic families
    idx = np.arange(n)
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # U‑shape linear
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996,
              0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # random Dirichlet draws with a spread of concentration parameters
    rng = np.random.default_rng(12345)
    for _ in range(80):
        conc = rng.uniform(0.05, 5.0)                 # low -> spiky, high -> flat
        a = rng.dirichlet(np.full(n, conc))
        try_update(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Soft‑max (temperature) weight vector for the convolution entries
# ----------------------------------------------------------------------
def _soft_weights(c: np.ndarray, tau: float) -> np.ndarray:
    """Return a probability vector proportional to exp((c‑maxc)/τ)."""
    maxc = c.max()
    if tau <= 0:
        # degenerate case – put all mass on the maximal entry
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
        return p
    shift = (c - maxc) / tau                 # ≤ 0
    # clip to avoid overflow/underflow in exp
    np.maximum(shift, -700, out=shift)
    p = np.exp(shift)
    s = p.sum()
    if s == 0.0:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
        s = 1.0
    return p / s


# ----------------------------------------------------------------------
# Gradient of the soft‑max objective (vectorised via FFT)
# ----------------------------------------------------------------------
def _grad_smoothed(a: np.ndarray, c: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute the gradient of  Σ_k p_k   c_k   w.r.t. a,
    where p_k = soft‑max weight (temperature τ).
    The formula reduces to 2·corr(p, a) and is evaluated by FFT.
    """
    p = _soft_weights(c, tau)          # length 2·n‑1
    n = a.shape[0]

    # correlation p ⋆ a (valid mode) = (p * a_rev)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    size = 1 << ((len(p) + n - 1).bit_length())   # next power of two
    P = np.fft.rfft(p, size)
    A_rev = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(P * A_rev, size)          # length = size
    start = n - 1
    grad = 2.0 * conv[start:start + n]            # vector length n
    return grad


# ----------------------------------------------------------------------
# Stage‑A – smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, n: int,
                            start: float, deadline: float,
                            best_a: np.ndarray, best_val: float,
                            target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Multiplicative descent using a soft‑max of the convolution entries."""
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 100_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current convolution and objective
        c = _full_convolution(cur)
        maxc = c.max()
        cur_val = 2.0 * n * maxc

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SmoothGrad {it:8d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # progress fraction (0 … 1)
        prog = (now - start) / (deadline - start)

        # step‑size schedule – start aggressive, end tiny
        eta = (1.0 - prog) * 0.35 + prog * 0.001   # 0.35 → 0.001

        # temperature schedule – start large (≈maxc), end very small
        tau = maxc * ((1.0 - prog) * 1.0 + prog * 0.001)   # maxc → 0.001·maxc

        # gradient w.r.t. a (soft‑max weighted)
        grad = _grad_smoothed(cur, c, tau)

        # exponentiated‑gradient update (keeps positivity)
        cur = cur * np.exp(-eta * grad)

        # renormalise to Σa = 1
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # occasional jitter when stuck
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.002, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[SmoothGrad {it:8d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage‑B – deterministic greedy hill‑climbing (mass transfer)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the index that contributes most to the
    current maximum convolution entry to the index that contributes least.
    The transferred fraction decays from 30 % to 0.1 %.
    """
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # evaluate current state
        c = _full_convolution(cur)
        maxc = c.max()
        cur_val = 2.0 * n * maxc

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:8d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # anneal transfer fraction
        prog = (now - start) / (deadline - start)
        max_frac = 0.30 * (1.0 - prog) + 0.001 * prog

        # locate the active peak
        k = int(np.argmax(c))

        # contributions a_{k‑i} that appear in c_k
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        # source = index with largest contribution
        i = int(np.argmax(rev))
        # target = index with smallest contribution (ignore the source)
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            # source empty – try a random jitter if we are stalled
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.003, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:8d}] jitter after stagnation")
            it += 1
            continue

        # amount to move (random fraction of the allowed max)
        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            it += 1
            continue

        # apply move tentatively
        cur[i] -= delta
        cur[j] += delta

        # evaluate new objective
        c_new = _full_convolution(cur)
        new_val = 2.0 * n * c_new.max()

        if new_val < cur_val - 1e-15:
            # keep the improvement
            cur_val = new_val
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage‑C – final tiny uniform mixing
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations a' = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

def propose_candidate(seed: int = 42,

    """
