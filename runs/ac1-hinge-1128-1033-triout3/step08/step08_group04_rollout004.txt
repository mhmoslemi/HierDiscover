def propose_candidate(seed=42, budget_s=1000, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # Determine problem size n
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 8000
        has_old = False

    rng = np.random.default_rng(seed)

    # Build candidate list
    init_candidates = []
    # Uniform
    init_candidates.append(np.full(n, 1.0/n))
    # previous if any
    if has_old:
        init_candidates.append(old / old.sum())
        init_candidates.append(np.flip(old) / old.sum())
    # random Dirichlet draws
    for shape in [0.5, 1.0, 2.0, 5.0]:
        rand = rng.gamma(shape=shape, scale=1.0, size=n)
        init_candidates.append(rand)

    # exponential decays
    for lam in np.logspace(-5, 0, 12):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        init_candidates.append(a)
        a_sym = np.exp(-lam * np.abs(np.arange(n) - (n-1)/2))
        a_sym /= a_sym.sum()
        init_candidates.append(a_sym)

    # power law
    for p in np.linspace(-3, 0, 7):
        idx = np.arange(1, n+1, dtype=float)
        a = idx**p
        a /= a.sum()
        init_candidates.append(a)

    # sinusoid
    x = np.linspace(0, np.pi, n)
    init_candidates.append((np.sin(x) + 0.1))
    # Normalize each
    init_candidates = [c / c.sum() for c in init_candidates]

    # evaluate cheap and pick best
    best_a = None
    best_val = float('inf')
    for i, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {i}] val = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best {best_val:.6f}")

    # Phase1: L-BFGS on smoothed objective
    # schedule eps
    eps_schedule = np.logspace(np.log10(1e-2), np.log10(1e-13), num=18)
    # Set w initial as log of best_a
    w = np.log(best_a + 1e-12)
    # initial step size for Adam? we skip.

    max_restarts = 20
    restart = 0
    while time.time() < deadline - 30 and restart < max_restarts:
        restart += 1
        print(f"[L-BFGS restart {restart}]")
        # Every few restarts randomize w
        if restart % 5 == 0:
            w = np.random.randn(n) * 3.0

        for eps in eps_schedule:
            if time.time() >= deadline - 30:
                break
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            # L-BFGS
            try:
                res = minimize(_smooth_obj_and_grad_wrapper,
                               w,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': 1200,
                                        'ftol': 1e-12,
                                        'gtol': 1e-12,
                                        'disp': False})
                w = res.x
            except Exception as e:
                print("[L-BFGS] error:", e)
                continue

            # evaluate candidate
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[L-BFGS eps {eps:.1e}] new best {best_val:.6f}")

        # try anti-symmetric reorder
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after restart] improved to {best_val:.6f}")

    # Phase2: Mirror descent subgradient
    a = best_a.copy()
    it = 0
    alpha0 = 0.5  # initial step
    decay = 0.9995
    min_alpha = 0.001
    while time.time() < deadline - 20:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())
        # gradient w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        # contributions: each pair counted twice except when i=j
        # but from earlier note derivative is 2*a_jdx

        # So use factor 2

        a = a * np.exp(-alpha * g)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()

                if val_ord < best_val:

                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[mirror reorder] improved to {best_val:.6f}")

        it += 1

    # Phase3: Simulated annealing
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.03
    cooling = 0.9997
    rng = np.random.default_rng(seed + 9999)

    while time.time() < deadline - 5:
        if rng.random() < 0.6:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0:
                continue
            delta = cur_a[i] * 0.15 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # swap blocks
            L = rng.integers(1, min(50, n // 10))
            i = rng.integers(0, n - L)
            j = rng.integers(0, n - L)
            new_a = cur_a.copy()
            tmp = new_a[i:i+L].copy()
            new_a[i:i+L] = new_a[j:j+L]
            new_a[j:j+L] = tmp

        # renorm
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_val - cur_val) / (T+1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA] new best {best_val:.6f}")

        T = max(T*cooling, 1e-6)

    # Final hill climbing
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # Final check
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] evaluate_sequence:", final_score)
    except Exception as e:
        print("[final] evaluate_sequence error:", e)

    return best_a.tolist()

"""
Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm (≈1000 s budget)
---------------------------
1. **Initial seeding** – a broad set of candidates (uniform,
   previous best, random Dirichlet draws, exponential / symmetric‑exponential
   families, power‑laws, a sinusoid).  The cheapest evaluator
   ``_fast_evaluate`` (conv max only, sum = 1) selects the best start.

2. **Smooth surrogate optimisation** – minimise a
   log‑sum‑exp smoothed max of the convolution.
   The variables are the logits ``w`` for a soft‑max
   probability vector.  A schedule of 15‑20 ε‑values (5·10⁻³ → 1·10⁻¹³)
   is traversed, each time with a bounded L‑BFGS‑B run.
   Multiple restarts (≈20) with random re‑initialisation diversify the search.
   After every restart an “anti‑symmetric” reorder is tried.

3. **Exact mirror‑descent** – sub‑gradient of the true max‑convolution
   (the worst index) drives an exponential / mirror update on the simplex.
   Step‑sizes decay slowly; occasional re‑ordering further shakes the iterate.
   The best vector seen is kept.

4. **Simulated annealing** – random mass‑transfer moves (single index or
   short block swaps) are accepted according to a geometric cooling schedule.
   This helps escape shallow local minima.

5. **Final hill‑climbing** – cheap local swaps / transfers that only accept
   strict improvements, ending just before the deadline.

All stages respect the 1000 s time limit; the final vector is checked with the
official ``evaluate_sequence`` and returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (set before each
# optimisation call).
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Returns the length‑2·len(a)‑1 array.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip numerical negatives
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate Lₑ(a) = 2·n·max(conv) + 2·n·ε·log ∑exp((conv−max)/ε)
    with the soft‑max parametrisation a_i = exp(w_i) / Σ exp(w_i).

    Returns (scalar value, gradient w.r.t. w).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max -----
    w_shift = w - np.max(w)          # numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- convolution -----
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max (log‑sum‑exp) -----
    shifted = (conv - max_c) / eps
    log_sum = np.logaddexp.reduce(shifted)  # log‑sum‑exp
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    weights = np.exp(shifted - log_sum)    # soft‑max over lag indices, sums to 1
    p = weights

    # convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment that aligns with the original a‑vector
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]   # derivative of 2·n·max + 2·n·ε·log‑sum‑exp

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return float(L), grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the O(n) algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for probability vectors
    (Σ a_i = 1).  Returns 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic re‑ordering: biggest values on the left side,
    smallest on the right side – often flattens the convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %); projection onto simplex
    after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                      # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
