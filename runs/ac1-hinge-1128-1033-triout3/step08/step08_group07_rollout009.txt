def solve_lp_max_sum(f, M):
    n = len(f)
    # Build matrix A using diags
    diags_data = []
    offsets = []
    # We'll use list of np arrays of length n - offset
    for offset in range(n):
        diags_data.append(f[:n - offset])
        offsets.append(offset)
    A = sp.diags(diags_data, offsets, shape=(2*n - 1, n), format='csr')
    # Upper bound: A @ b <= M
    # For linprog, constraints must be A_ub @ x <= b_ub
    # Use: A_ub = A, b_ub = np.full(2*n-1, M)
    # bounds for x: (0, None)
    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n - 1, M),
                                bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    else:
        return None

def _refine_hill_climb(a, cur_val, deadline):
    n = len(a)
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        # choose move type: 0 swap two indices, 1 move mass, 2 reverse block, 3 shift mass
        move_type = rng.integers(0, 4)
        if move_type == 0:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        elif move_type == 1:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-12:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        elif move_type == 2:
            # reverse a random subarray
            i, j = sorted(rng.integers(0, n, size=2))
            new_a = cur_a.copy()
            new_a[i:j+1] = new_a[i:j+1][::-1]
        else:
            # shift a block
            i, j = sorted(rng.integers(0, n, size=2))
            block = cur_a[i:j+1]
            dst = rng.integers(0, n - (j-i+1) + 1)
            new_a = cur_a.copy()
            new_a = np.delete(new_a, slice(i, j+1))
            new_a = np.insert(new_a, dst, block)
            # after insertion length still same but np.insert returns new array; but easier to use list? Might be slower.

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

"""
Hybrid LP‑refinement optimizer for the step‑function minimisation problem.

The search consists of the following stages (always respecting the
global 1000 s wall‑time budget):

1. **Initialisation** – try to reuse a previously discovered construction
   (`height_sequence_1`).  If unavailable, start from a tiny random
   perturbation of the uniform distribution, normalised to ℓ₁‑norm = 1.

2. **LP‑driven expansion** – given the current shape `f` compute
   `M = max(f * f)`.  Solve a linear programme that maximises
   `∑ b_i` under the linear constraints `f * b ≤ M` (convolution
   entry‑wise) and `b_i ≥ 0`.  The solution `b` can be normalised to a
   probability vector `g`.  If `g` yields a strictly smaller evaluation
   value than the best seen so far it is adopted, otherwise we move a
   fraction `t` towards `g`.  Multiple LP‑updates are performed while
   time permits.

3. **Local permutation + mass‑transfer hill‑climbing** – a cheap
   random‑swap / tiny‑mass‑transfer search (Metropolis‑free) that
   accepts any improving move.  This stage consumes the remaining budget.

All heavy operations (convolutions, LP matrix construction) use
FFT‑based convolution and sparse matrix arithmetic, allowing 4000‑element
vectors to be processed comfortably within the time limit.  The final
candidate is returned as a plain Python list; the provided
`evaluate_sequence` function is called once for the official score.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities – FFT based convolution, fast evaluation, re‑ordering
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)            # eliminate tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Exact copy of `evaluate_sequence` for a normalised vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Deterministic reordering that puts large values on the extremes."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


# ----------------------------------------------------------------------
# Construction of the convolution matrix for a linear programme
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2·n‑1) × n sparse matrix A such that
        (A @ b)[k] = (f * b)[k]   for all k,
    i.e. each row corresponds to a shifted copy of ``f``.
    """
    n = f.shape[0]
    # Each diagonal offset i (0 ≤ i < n) contains the constant value f[i]
    # repeated n times – this exactly encodes the convolution.
    diags_data = [np.full(n, fi, dtype=np.float64) for fi in f]
    offsets = list(range(n))
    A = sp.diags(diags_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_expand(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP
        maximise   1ᵀ b
        subject to  f * b ≤ M   (entry‑wise)
                    b ≥ 0 .
    Returns the optimal vector ``b`` (or ``None`` on failure).
    """
    n = f.shape[0]
    # Build sparse convolution matrix
    A = _conv_matrix(f)

    # Slight tolerance on the bound to avoid numerical infeasibility
    ub = np.full(2 * n - 1, M * (1.0 + 1e-9), dtype=np.float64)

    # linprog minimises; maximise 1ᵀb  →  minimise -1ᵀb
    c = -np.ones(n, dtype=np.float64)

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 30.0},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Very cheap hill‑climbing refinement (swap + tiny mass moves)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Accept any strictly improving swap or small mass‑transfer move."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        # 30 % swaps, 70 % mass transfers
        if rng.random() < 0.3:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-14:
                continue
            delta = cur_a[i] * 0.02 * rng.random()   # ≤ 2 % of source mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final official evaluation
    deadline = start_time + budget_s - 1.5
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # --------------------------------------------------------------
    # Problem size – 4000 works well with the FFT/L‑P machinery.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------

    try:

            new_x = np.linspace(0, 1, n)

        print("[init] loaded previous construction")
    except Exception:

        a0 = np.full(n, 1.0 / n)
        a0 += 5e-4 * np.random.rand(n)
        print("[init] using fresh uniform start")

    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP driven expansion / mixing
    # ------------------------------------------------------------------
    t_mix = 0.25               # convex‐mixing factor (1‑t towards new vector)
    lp_iter = 0
    while time.time() < deadline - 25.0:   # reserve ~25 s for final hill‑climb
        lp_iter += 1
        M = np.max(_fft_convolve_self(best_a))

        # Solve the LP (may take up to a few seconds)
        b = _lp_expand(best_a, M)
        if b is None:
            print(f"[LP] iteration {lp_iter}: LP failed, aborting LP phase")
            break

        sum_b = np.sum(b)
        if sum_b <= 1e-12:
            print(f"[LP] iteration {lp_iter}: trivial solution, stop")
            break

        # Normalise to a probability vector
        g = b / sum_b

        # Evaluate the pure candidate
        val_g = _eval_fast(g)
        if val_g + 1e-12 < best_val:
            best_a, best_val = g, val_g
            print(f"[LP] iteration {lp_iter}: pure g improved to {best_val:.6f}")
            # after a successful pure move we keep t_mix moderate
            continue

        # Otherwise perform a convex combination (mixing)
        cand = (1.0 - t_mix) * best_a + t_mix * g
        cand = np.maximum(cand, 0.0)
        cand /= np.sum(cand)

        cand_val = _eval_fast(cand)
        if cand_val + 1e-12 < best_val:
            best_a, best_val = cand, cand_val
            print(f"[LP] iteration {lp_iter}: mixed candidate improved to {best_val:.6f}")

        # Periodically apply the “ends‑large” heuristic to break local symmetry
        if lp_iter % 4 == 0:
            ordered = _ends_large_reorder(best_a)
            ordered /= np.sum(ordered)
            ordered_val = _eval_fast(ordered)
            if ordered_val + 1e-12 < best_val:
                best_a, best_val = ordered, ordered_val
                print(f"[LP] iteration {lp_iter}: ends‑large reorder improved to {best_val:.6f}")

        # slowly decay the mixing factor – we want finer steps as we approach a plateau
        t_mix = max(0.05, t_mix * 0.97)

    # ------------------------------------------------------------------
    # Phase 2 – final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] hill‑climbing refinement")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Official evaluation (security‑checked wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    # Return a plain Python list as required by the framework
    return best_a.tolist()
