def _project_simplex(v):
    # Projects onto simplex {x>=0, sum x = 1}
    # using sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _fft_convolve_self(a):
    n = a.shape[0]
    out_len = 2*n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    conv[conv < 0] = 0.0
    return conv

def _project_simplex(v):
    # returns projection onto simplex
    v = np.asarray(v)
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid search for a low‑value step‑function.

The algorithm works on three levels:

1. **Many random restarts of a simple projected sub‑gradient descent**.
   The objective (max of the self‑convolution) is convex in the
   probability simplex, so a short gradient descent from a random Dirichlet
   start quickly lowers the bound.  After each restart we apply the
   “ends‑large” re‑ordering heuristic.

2. **Long simulated‑annealing on the exact objective** (swap + mass‑transfer
   moves).  The temperature is tuned to the current value and the moves are
   accepted with the Metropolis rule.

3. **Very cheap hill‑climbing refinement** to polish the final answer.

All heavy work (convolutions, projections, evaluations) uses only NumPy
and runs well within the 1000 s budget.  The best sequence found is
returned as a plain Python list and finally evaluated by the provided
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (FFT convolution, fast evaluation, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using an FFT of size a power of two.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # eliminate tiny negatives from rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Exact copy of `evaluate_sequence` without the safety wrapper."""
    n = a.shape[0]
    max_b = _fft_convolve_self(a).max()
    s = a.sum()
    # sum is kept > 0 by construction; the denominator never vanishes.
    return 2.0 * n * max_b / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # the projection of the zero vector is the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Heuristics that reorder a vector (large values at the extremes)
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Deterministic reordering: large – small – large – small …"""
    n = a.shape[0]
    sorted_vals = np.sort(a)                     # ascending
    new_a = np.empty_like(a)

    left, right = 0, n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


# ----------------------------------------------------------------------
# Simulated annealing (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing_mass(a: np.ndarray,
                              cur_val: float,
                              deadline: float) -> (np.ndarray, float):
    """
    Classic Metropolis search that alternates swaps and small
    mass‑transfer moves.  The move size grows up to 10 % of the source
    mass, which helps escaping shallow basins.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # initial temperature = few percent of the current value
    T = max(cur_val * 0.04, 1e-6)
    decay = 0.99994

    while time.time() < deadline:
        old_a = a.copy()

        r = rng.random()
        if r < 0.25:                                      # pure swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                a[i], a[j] = a[j], a[i]

        elif r < 0.55:                                    # medium mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and old_a[i] > 1e-12:
                delta = old_a[i] * rng.random() * 0.10   # ≤10 % of source
                a[i] = old_a[i] - delta
                a[j] = old_a[j] + delta

        elif r < 0.80:                                    # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and old_a[i] > 1e-12:
                delta = old_a[i] * rng.random() * 0.03
                a[i] = old_a[i] - delta
                a[j] = old_a[j] + delta

        else:                                             # jitter & re‑project
            i = rng.integers(0, n)
            delta = rng.normal(scale=0.01)
            a[i] = max(old_a[i] + delta, 0.0)
            a = _project_simplex(a)

        new_val = _eval_fast(a)
        diff = new_val - cur_val

        # Metropolis acceptance
        if diff < 0.0 or rng.random() < np.exp(-diff / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f}")
        else:                     # reject → restore old state
            a = old_a

        T *= decay
        # keep temperature from vanishing completely
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap hill‑climbing (swap + tiny transfer, only improvements)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Accepts any strictly improving move (swap or tiny transfer)."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        old_a = cur_a.copy()
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
        else:                                       # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i != j and cur_a[i] > 1e-12:
                delta = cur_a[i] * rng.random() * 0.02
                cur_a[i] -= delta
                cur_a[j] += delta

        val = _eval_fast(cur_a)
        if val + 1e-12 < cur_val:               # improvement
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
        else:
            cur_a = old_a                       # revert

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent from a random start
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         max_iter: int,
                         deadline: float) -> (np.ndarray, float):
    """
    Gradient descent on the exact max‑convolution with projection onto the
    probability simplex.  Uses the sub‑gradient of the active peak.
    """
    a = a0.copy()
    n = a.shape[0]
    step = 0.6                     # initial step size
    best_a = a.copy()
    best_val = _eval_fast(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # compute convolution and locate the active peak
        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the max entry w.r.t. a
        grad = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2.0 * a[idx[mask]]          # 2·a_{k‑i}

        # simple gradient step + projection onto the simplex
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # decay step size slowly
        step *= 0.9995

        if it % 30 == 0:
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[GD] iter {it} new best = {best_val:.6f}")

    # final check
    cur_val = _eval_fast(a)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()
        print(f"[GD] final best = {best_val:.6f}")

    return best_a, best_val
