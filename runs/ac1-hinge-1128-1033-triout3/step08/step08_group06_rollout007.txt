def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Remove tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _objective_fast(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for internal use.
    a must be a 1‑D NumPy array of non‑negative numbers.
    """
    # Apply the same clamping as the official evaluator
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)

    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')

    conv = _full_convolution(a)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (sum_a ** 2))


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values on the left, the smallest on the right."""
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # large → left side
        res[n - 1 - i] = s[i]     # small → right side
    if n % 2:
        res[half] = s[half]
    return res


def _interleaved_reorder(a: np.ndarray) -> np.ndarray:
    """Interleave large and small values (large, small, large, …)."""
    s = np.sort(a)[::-1]          # descending
    n = a.shape[0]
    res = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for v in s:
        if toggle:
            res[left] = v
            left += 1
        else:
            res[right] = v
            right -= 1
        toggle = not toggle
    return res


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # ---------- soft‑max ----------
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # ---------- convolution ----------
    conv = _full_convolution(a)
    max_conv = conv.max()

    # ---------- smooth max ----------
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    weights = np.exp(shifted)
    p = weights / weights.sum()            # distribution over convolution indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries give ∂(Σ p·conv)/∂a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # factor 4n = 2·n·2 (the two symmetric terms)

    # ---------- chain‑rule through soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by `scipy.minimize` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run L‑BFGS with a geometric ε‑schedule.
    Returns the best (a, value) pair discovered before `deadline`.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule (from 5·10⁻³ down to 10⁻¹², 13 points)
    eps_schedule = np.geomspace(5e-3, 1e-12, num=13)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 30.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 1500,
                                'ftol': 1e-13,
                                'gtol': 1e-13,
                                'disp': False})
        w = res.x

        # map back to probability simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_fast(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # try the two re‑orderings
        for reorder, name in ((_anti_symmetric_reorder, "anti‑sym"),
                              (_interleaved_reorder, "interleaved")):
            a_ord = reorder(a)
            val_ord = _objective_fast(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[L‑BFGS {name} reorder eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = _objective_fast(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞‑objective (mirror‑descent)
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute sub‑gradient g_i = Σ_{k∈active} a_{k‑i}
    using a single FFT.  `active_idx` contains convolution indices (0 … 2n‑2).
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n-2) - i] for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent_multi(a_start: np.ndarray, deadline: float):
    """
    Exponentiated mirror‑descent that reduces the *active* convolution peaks.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _objective_fast(a)
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.7          # initial step‑size factor (slightly smaller for stability)
    rel_tol = 0.07        # start with a generous active‑set tolerance

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten tolerance periodically
        if it > 0 and it % 15000 == 0:
            rel_tol = max(rel_tol * 0.5, 5e-6)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # gradient via FFT (sub‑gradient of the L∞‑norm)
        g = _active_set_gradient(a, active)

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative update
        a = a * np.exp(-alpha * g)

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # logging & occasional re‑ordering
        if it % 250 == 0:
            cur_val = _objective_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")

            if it % 6000 == 0:
                for reorder, name in ((_anti_symmetric_reorder, "anti‑sym"),
                                      (_interleaved_reorder, "interleaved")):
                    a_ord = reorder(a)
                    val_ord = _objective_fast(a_ord)
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        print(f"[sgd {name} reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing based hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    """
    Random local moves (swap or mass transfer) with an annealing schedule.
    Only strictly better solutions are kept for the reported best,
    but uphill moves are occasionally accepted.
    """
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.025                     # initial “temperature”
    it = 0

    while time.time() < deadline:
        it += 1
        # propose a move
        if rng.random() < 0.25:                     # pure swap (25 %)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                        # mass transfer (75 %)
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * (0.02 + 0.08 * rng.random())   # up to 10 % of mass at i
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto simplex (mass is already conserved)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _objective_fast(new_a)
        accept = False
        if val < cur_val:
            accept = True
        else:
            # annealing acceptance
            T = T0 * (0.9997 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal iter {it}] new best = {best_val_local:.6f}")

            # occasional global reshuffle when stuck
            if it % 50000 == 0:
                perm = rng.permutation(n)
                cur_a = cur_a[perm]
                cur_a = np.maximum(cur_a, 0.0)
                cur_a /= cur_a.sum()

    return best_a, best_val_local


# ----------------------------------------------------------------------
# Candidate generators (Stage 0)
# ----------------------------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _exponential_candidate(scale: float, n: int, rng: np.random.Generator) -> np.ndarray:
    cand = rng.exponential(scale=scale, size=n)
    cand /= cand.sum()
    return cand


def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (dist_to_end+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


def _exponential_decay_candidate(lam: float, n: int) -> np.ndarray:
    """Geometric decay a_i ∝ exp(-lam * i)."""
    idx = np.arange(n)
    a = np.exp(-lam * idx)
    a /= a.sum()
    return a


def _logistic_candidate(sigma: float, n: int) -> np.ndarray:
    """Logistic (sigmoid) centred at n/2."""
    idx = np.arange(n)
    a = 1.0 / (1.0 + np.exp((idx - n / 2) / sigma))
    a /= a.sum()
    return a


def _alternating_candidate(alpha: float, n: int) -> np.ndarray:
    """Two‑level alternating pattern (even indices get α)."""
    evens = (n + 1) // 2
    odds = n // 2
    total = alpha * evens + (1.0 - alpha) * odds
    a = np.empty(n)
    a[::2] = alpha / evens
    a[1::2] = (1.0 - alpha) / odds
    a /= a.sum()                     # safeguard against rounding error
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the previous successful length.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Start from a known construction if available, otherwise uniform.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = _uniform_candidate(n)

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    a = a / a_sum if a_sum > 1e-12 else _uniform_candidate(n)

    best_a = a.copy()
    best_val = _objective_fast(best_a)
    print(f"[init] n={n}, objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – cheap diversified candidates + random permutations
    # ------------------------------------------------------------------
    # Structured families
    families = []

    families.append(_uniform_candidate(n))

    # exponential random vectors (different scales)
    for scale in (0.5, 1.0, 2.0):
        families.append(_exponential_candidate(scale, n, rng))

    # power‑law exponents
    for p in np.arange(0.2, 2.2, 0.2):
        families.append(_powerlaw_candidate(p, n))

    # exponential decay (geometric) parameters
    for lam in np.logspace(-4, -1, 5):
        families.append(_exponential_decay_candidate(lam, n))

    # logistic sigmoids
    for sigma in np.linspace(n / 20, n / 4, 5):
        families.append(_logistic_candidate(sigma, n))

    # alternating two‑level patterns
    for alpha in np.linspace(0.05, 0.45, 5):
        families.append(_alternating_candidate(alpha, n))

    # ------------------------------------------------------------------
    # Evaluate each base shape and a few random permutations.
    # ------------------------------------------------------------------
    for base in families:
        if time.time() >= deadline - 300:
            break
        # base shape
        val = _objective_fast(base)
        if val < best_val:
            best_val, best_a = val, base.copy()
            print(f"[base] new best = {best_val:.6f}")

        # try both deterministic reorderings
        for reorder, name in ((_anti_symmetric_reorder, "anti‑sym"),
                              (_interleaved_reorder, "interleaved")):
            a_ord = reorder(base)
            val_ord = _objective_fast(a_ord)
            if val_ord < best_val:
                best_val, best_a = val_ord, a_ord.copy()
                print(f"[{name} reorder] improved to {best_val:.6f}")

        # a few random permutations (cheap because we use FFT)
        perm_its = 7
        for _ in range(perm_its):
            perm = rng.permutation(n)
            a_perm = base[perm]
            val_perm = _objective_fast(a_perm)
            if val_perm < best_val:
                best_val, best_a = val_perm, a_perm.copy()
                print(f"[perm] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – smooth‑surrogate optimisation (L‑BFGS) with many restarts
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 200 and restarts < 12:
        restarts += 1
        # perturb current best in the log‑space
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.3, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[restart {restarts}] L‑BFGS improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – sub‑gradient descent (fine‑grained) on the true L∞‑objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        a_sgd, val_sgd = _subgradient_descent_multi(best_a, deadline)
        if val_sgd < best_val:
            best_val, best_a = val_sgd, a_sgd.copy()
            print(f"[sub‑gradient fine] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final refinement via simulated‑annealing hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_simulated_anneal(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer (only once)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
