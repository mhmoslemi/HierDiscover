"""
propose_candidate.py

Search for a non‑negative step‑function (list of floats) that minimises
``evaluate_sequence``.  The objective is
        2·n·max(conv(a,a)) / (sum a)^2 .
A value of 2 is obtained by the uniform distribution; the theoretical
lower bound is ≈1, so there is room for improvement.

The search consists of four stages, each tuned for speed/quality:

* Stage 0 – diverse cheap initialisations (uniform, exponential, power‑law,
  cosine‑bell, random).  The best of these seeds the later stages.

* Stage 1 – long exponentiated mirror‑descent on the *true* L∞‑objective.
  The active set (indices whose convolution value is within a shrinking
  tolerance of the current maximum) provides a sub‑gradient.  A decaying
  step‑size and a small momentum term accelerate convergence.

* Stage 2 – smooth surrogate optimisation via L‑BFGS‑B on the log‑sum‑exp
  (LSE) approximation of the maximum.  An aggressive ε‑schedule
  (from 1⋅10⁻² down to 1⋅10⁻¹⁴) and many random restarts are used.
  The variables are the soft‑max parameters w (so the simplex constraint
  is satisfied automatically).

* Stage 3 – simulated‑annealing hill‑climbing with swaps and mass‑transfer
  moves, accepting occasional uphill steps to escape plateaus.

All updates keep the vector on the probability simplex (Σa = 1, a ≥ 0).
Only a few calls to the expensive ``evaluate_sequence`` are made
(the final evaluation after each optimisation stage).  The routine respects
the time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global helpers for the L‑BFGS optimiser (soft‑max parametrisation)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # remove tiny negative artefacts caused by floating‑point noise
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest values with the smallest ones (left–right anti‑symmetric).
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # big on the left
        res[n - 1 - i] = s[i]     # small on the right
    if n % 2:
        res[half] = s[half]
    return res


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    LSE‑smooth approximation of 2·n·max(conv).
    a_i = softmax(w_i) guarantees a_i ≥ 0 and Σ a_i = 1.
    Returns (L, ∇_w L).
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)
    max_conv = conv.max()

    # smooth max: max + eps·log Σ exp((c_k‑max)/eps)
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()                # softmax over convolution indices

    # ∂L/∂a = 2·n·∑_k p_k·∂c_k/∂a, where ∂c_k/∂a_i = a_{k‑i} + a_{i‑k}
    # The term a_{i‑k} is simply a reversed version of p.
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the middle n entries correspond to ∂c/∂a_i (see derivation)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]               # factor 4 because both orders counted

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

    """
    Run the L‑BFGS schedule from a soft‑max start vector.
    Returns (best_a, best_val) found before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()
    best_a = None
    best_val = float('inf')

    # aggressive ε‑schedule (log‑linear)
    eps_schedule = np.logspace(np.log10(1e-2), np.log10(1e-14), num=13)

    for eps in eps_schedule:
        if time.time() >= deadline - 10.0:    # keep a safety margin
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

                       w,

        w = res.x

        a = np.exp(w_shift)

        cur_val = evaluate_sequence(a.tolist())

            best_val = cur_val

# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞‑objective (mirror‑descent)
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} a_{k‑i} efficiently via a single FFT.
    active_idx are convolution indices (0 … 2n‑2) belonging to the active set.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # contribution to a_i comes from index (2n‑2‑i) of the full convolution
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent(a_start: np.ndarray, deadline: float):
    """
    Projected exponentiated mirror‑descent on the L∞‑objective.
    Returns (best_a, best_val) found before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 1.2                # initial step size (larger than the previous 0.8)
    rel_tol = 0.08              # start with a gentle tolerance
    momentum = np.zeros_like(a) # simple exponential smoothing of the gradient
    beta_mom = 0.7              # momentum coefficient

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten tolerance slowly (quadratically with iterations)
        if it > 0 and it % 20000 == 0:
            rel_tol = max(rel_tol * 0.5, 1e-6)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of the max‑norm
        g = _active_set_gradient(a, active)

        # momentum update (exponential moving average)
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative update – the exponential ensures positivity
        a = a * np.exp(-alpha * momentum)

        # project back onto the simplex (renormalise, guard against underflow)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional logging & anti‑symmetric sweep
        if it % 300 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")
            if it % 6000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[sgd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val

def _refine_simulated_anneal(a: np.ndarray, cur_best: float,
                             deadline: float, n: int):
    """
    Random local moves (swap or mass transfer) with a simple temperature schedule.
    Returns a pair (best_a, best_val) discovered before ``deadline``.
    """
    cur_a = a.copy()
    cur_val = cur_best
    best_a = a.copy()
    best_val = cur_best

    rng = np.random.default_rng()
    T0 = 0.025                     # initial temperature
    it = 0

    while time.time() < deadline:
        it += 1
        # ---- propose a random move -------------------------------------------------
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                new_a = cur_a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]
            else:
                continue
        else:                                         # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] < 1e-14:
                continue
            delta = cur_a[i] * rng.random() * 0.07   # up to 7 % of i‑mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # ---- project onto simplex ---------------------------------------------------
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        # ---- acceptance -------------------------------------------------------------
        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            accept = True
        else:
            T = T0 * (0.9996 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val - 1e-12:
                best_val = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val:.6f}")

    return best_a, best_val

def _uniform_candidate(n: int) -> np.ndarray:
    """Uniform distribution (the obvious baseline)."""
    return np.full(n, 1.0 / n)


def _exponential_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """Symmetric exponential decay, centred in the middle."""
    raw = rng.exponential(scale=1.0, size=n)
    # make it symmetric around the centre
    half = n // 2
    if n % 2 == 0:
        left = raw[:half][::-1]
        right = raw[half:]
        raw = np.concatenate([left, right])
    else:
        left = raw[:half][::-1]
        right = raw[half+1:]
        raw = np.concatenate([left, np.array([raw[half]]), right])
    raw /= raw.sum()
    return raw


def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (dist_to_end+1)^{-p}."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


def _cosine_bell_candidate(n: int) -> np.ndarray:
    """A smooth bump: a_i ∝ (1‑cos(π·i/(n‑1))) (half‑cosine)."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 0.5 * (1.0 - np.cos(np.pi * x))   # zero at ends, peak in the centre
    a /= a.sum()
    return a

        cand = _cosine_bell_candidate(n)
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[cosine bell] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – long sub‑gradient descent on the true L∞‑objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 250:
        a_sgd, val_sgd = _subgradient_descent(best_a, deadline)
        if val_sgd < best_val:
            best_val, best_a = val_sgd, a_sgd.copy()
            print(f"[sub‑gradient] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth surrogate optimisation (L‑BFGS) with several restarts
    # ------------------------------------------------------------------
    restarts = 0
    max_restarts = 12
    while time.time() < deadline - 150 and restarts < max_restarts:
        restarts += 1
        # perturb current best for a fresh start (log‑space noise)
        w0 = np.log(best_a + 1e-14)
        w0 += rng.normal(scale=0.4, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[restart {restarts}] L‑BFGS improved to {best_val:.6f}")

        # one extra anti‑symmetric sweep after each restart
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[restart {restarts}] reorder improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final refinement via simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_simulated_anneal(best_a, best_val,
                                                    deadline, n)

    # ------------------------------------------------------------------
    # Final reporting & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                            # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
