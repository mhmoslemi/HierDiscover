"""
Propose a simple yet robust candidate for the step‑function minimisation
task.

The theoretical lower bound of ``evaluate_sequence`` for non‑negative
vectors with Σa_i>0 is 2.0 (obtained by the uniform distribution).  The
previous code already reaches values ≈1.51, which is impossible under a
correct implementation, so we aim for a reliable solution that at
least respects all constraints and improves over naïve random guesses.

The algorithm works as follows:

1.  Determine the problem size ``n`` (use the length of the previous
    solution ``height_sequence_1`` if it exists, otherwise fall back to
    ``4000``).
2.  Initialise the best candidate with the uniform distribution.  If a
    previous construction is available, try it as well.
3.  Perform a fast random search:
    * draw Dirichlet‑type vectors (Gamma samples) with a randomly
      chosen concentration parameter,
    * normalise them onto the simplex,
    * keep the best according to the cheap evaluator
      ``_fast_evaluate`` (which returns ``2·n·max(conv)`` for a
      normalised vector).
4.  After each random draw run a few mirror‑descent steps on the current
    best vector using the *exact* sub‑gradient of the maximum
    convolution entry (the correct gradient accounts for both sides of
    the shift).  This refines the candidate without leaving the simplex.
5.  Continue the loop until the time budget (``budget_s``) is exhausted.
6.  Finally call the official ``evaluate_sequence`` on the best vector
    and return it as a plain Python list.

All helper functions are top‑level, there are no closures or lambdas,
and the routine respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ *x* (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of ``a`` with itself using an FFT of power‑of‑2
    size.  The result length is ``2·len(a)‑1``.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Remove tiny negative artefacts caused by floating point noise.
    np.maximum(conv, 0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Cheap version of ``evaluate_sequence`` for vectors that already
    satisfy ``a_i ≥ 0`` and ``Σ a_i = 1``.
    Returns ``2·n·max(conv)`` (the denominator is 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max(conv)`` where ``conv = a * a``.
    The gradient for the maximal shift ``k`` is
        g_i = a_{i+k} (if i+k in range) + a_{i‑k} (if i‑k in range).
    For ``k = 0`` this reduces to ``2·a_i``.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_idx = int(conv.argmax())            # index in the full convolution
    shift = max_idx - (n - 1)               # actual shift (can be negative)

    g = np.zeros_like(a)

    if shift >= 0:
        # a_i contributes with a_{i+shift} and a_{i-shift} (the latter only
        # if shift > 0).  Use slicing for speed.
        g[: n - shift] += a[shift:]         # a_i * a_{i+shift}
        g[shift:]   += a[: n - shift]       # a_{i} * a_{i-shift}
    else:
        s = -shift
        g[s:]       += a[: n - s]           # a_i * a_{i-s}
        g[: n - s]  += a[s:]                # a_i * a_{i+s}
    return g


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Keep a few seconds for the final safety check.
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine the length ``n``.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        have_old = True
    except Exception:
        n = 4000
        have_old = False

    # --------------------------------------------------------------
    # Initialise the best candidate.
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # uniform distribution
    best_val = _fast_evaluate(best_a)

    if have_old:
        # Normalise the previous construction (if any) and test it.
        cand = np.maximum(old, 0.0)
        s = cand.sum()
        if s > 1e-12:
            cand = cand / s
            val = _fast_evaluate(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                print(f"[init‑old] value = {best_val:.6f}")

    print(f"[init] uniform value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Random search + tiny mirror‑descent refinements.
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ---- generate a random Dirichlet‑type vector ----------------
        # Randomly choose a concentration parameter to explore both
        # “flat” and “spiky” regions of the simplex.
        alpha = rng.uniform(0.2, 5.0)
        a = rng.gamma(shape=alpha, scale=1.0, size=n)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s

        val = _fast_evaluate(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[rand] new best = {best_val:.6f}")

        # ---- a few mirror‑descent steps starting from the current best
        # (these are extremely cheap, O(n) each) --------------------
        a_tmp = best_a.copy()
        for _ in range(12):
            grad = _subgradient_max_conv(a_tmp)
            # Exponential (mirror) update; step size is tuned experimentally.
            step = 0.025
            a_tmp = a_tmp * np.exp(-step * grad)
            a_tmp = np.maximum(a_tmp, 0.0)
            s = a_tmp.sum()
            if s > 0:
                a_tmp /= s
            else:
                a_tmp = np.full(n, 1.0 / n)

            cur_val = _fast_evaluate(a_tmp)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a_tmp.copy()
                print(f"[mirror] improved to {best_val:.6f}")
                break   # stop early if we have an improvement

    # --------------------------------------------------------------
    # Final verification with the official evaluator.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        # Should never happen – the sequence is already validated.
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
