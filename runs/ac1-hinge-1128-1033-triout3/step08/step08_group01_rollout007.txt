def _adam_phase(a0, n, start, deadline, best_a, best_val, target_val=1.503):
    w = np.log(a0 + 1e-12)
    beta1,beta2 = 0.9,0.999
    lr_start, lr_end = 0.2, 0.001
    eps_start, eps_end = 1e-1, 1e-12
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    it = 0
    no_improve=0
    IMPROVE_LIMIT = 30000
    while time.time() < deadline:
        # compute a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a_sum = a.sum()
        a = a / a_sum
        # symmetrize
        a = _symmetrize(a)
        a = a / a.sum()
        # evaluate objective
        cur_val = _objective_value(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val; best_a = a.copy()
            no_improve=0
            print(f"[Adam {it:6d}] new best {best_val:.6f}")
            if cur_val <= target_val:
                break
        else:
            no_improve+=1
        # schedule
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end/eps_start)**progress
        lr = lr_start * (lr_end/lr_start)**progress
        # compute smooth max gradient
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted - np.max(shifted))  # avoid overflow
        p = weights / weights.sum()
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n-1:2*n-1]
        grad_a = 4.0 * n * segment[::-1]
        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)
        # Adam update
        t += 1
        m = beta1*m + (1-beta1)*grad_w
        v = beta2*v + (1-beta2)*(grad_w**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # occasional jitter
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.005, size=n)
        # restart if idle
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it}] restart after {no_improve}")
            w = np.random.randn(n) * 0.2
            m.fill(0); v.fill(0); t=0; no_improve=0
        it+=1
    return best_a, best_val
```

"""
Propose a non‑negative step function that minimises
   2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1.  Choose a fairly large dimension (n≈20000).  The objective for a
    normalised vector (sum = 1) is 2·n·max(a*a).  As n grows the
    discretisation error shrinks and the attainable bound improves.

2.  Generate a diverse set of *parametric* candidates (exponential,
    power‑law, sinusoidal, triangular, random Dirichlet).  Keep the
    best one as a starting point.

3.  Refine the starter with three complementary optimisation stages,
    each respecting the remaining time budget:

    • Adam on a smooth log‑sum‑exp approximation of the max‑convolution.
      The parametrisation a = softmax(w) guarantees non‑negativity and a
      unit sum.  After each update the vector is explicitly symmetrised
      (a ← (a+rev(a))/2) – the optimal solution is known to be symmetric.

    • Targeted hill‑climbing on the *exact* objective.  In each step a
      small amount of mass is moved from an index that contributes most
      to the current maximal convolution entry to an index that contributes
      little.  The convolution is updated incrementally (O(n) per step).

    • Projected sub‑gradient descent on the true (non‑smooth) objective.
      The step size decays as η₀/√t and the iterate is projected onto the
      probability simplex (Euclidean projection).  Symmetrisation is
      applied after each step.

4.  A final short sweep mixing a tiny amount of uniform mass often
    reduces the peak further.

All phases are time‑controlled; the best vector found at any point is
returned.  The code uses only NumPy and the provided ``evaluate_sequence``
function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised (sum=1) vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w ≥ 0, Σ w = 1}."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Symmetrisation utilities
# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n-1-i} (average with the reversed copy)."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Parametric families – generate a candidate and return (a, value)
# ----------------------------------------------------------------------
def _candidate_exponential(lam: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-lam·i) + exp(-lam·(n-1-i))."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_powerlaw(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p) + (n - i + 1) ** (-p)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_sine_pow(p: float, n: int) -> np.ndarray:
    """Half‑wave sine raised to a power."""
    x = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = np.power(x, p, out=np.empty_like(x))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_triangular_pow(p: float, n: int) -> np.ndarray:
    """Triangular shape |i - (n‑1)/2|^{p} (mirrored)."""
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.ones(n))
    a /= a.sum()
    return a


def _initial_best_candidate(n: int, budget_deadline: float) -> tuple[np.ndarray, float]:
    """
    Sweep a few parametric families (and a few Dirichlet draws)
    and keep the best candidate seen before the given deadline.
    """
    rng = np.random.default_rng(12345)
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _update(a: np.ndarray):
        nonlocal best_a, best_val
        a = a.astype(np.float64)
        if np.any(a < 0):
            return
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[init] new best = {best_val:.6f}")

    # exponential in a reasonable λ‑range
    for lam in np.linspace(0.001, 0.025, 30):
        if time.time() > budget_deadline:
            break
        _update(_candidate_exponential(lam, n))

    # power‑law
    for p in np.linspace(0.2, 4.0, 30):
        if time.time() > budget_deadline:
            break
        _update(_candidate_powerlaw(p, n))

    # sine‑power
    for p in np.linspace(0.5, 3.0, 20):
        if time.time() > budget_deadline:
            break
        _update(_candidate_sine_pow(p, n))

    # triangular‑power
    for p in np.linspace(0.5, 3.0, 15):
        if time.time() > budget_deadline:
            break
        _update(_candidate_triangular_pow(p, n))

    # a few random draws
    for _ in range(15):
        if time.time() > budget_deadline:
            break
        _update(_random_dirichlet(n, rng))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam optimisation on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                global_best_a: np.ndarray, global_best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Optimise using Adam on the log‑sum‑exp smooth of max(conv).
    The iterate is kept on the probability simplex via a soft‑max.
    """
    # initialise logits
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.18, 0.001
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 25000

    while time.time() < deadline:
        # ---- map logits to simplex (and enforce symmetry) ----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()
        a = _symmetrize(a)           # keep the solution symmetric
        a = a / a.sum()              # re‑normalise after symmetrisation

        cur_val = _objective_value(a)

        # ---- keep global best ------------------------------------------------
        if cur_val < global_best_val - 1e-12:
            global_best_val, global_best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:6d}] new global best = {global_best_val:.6f}")
            if global_best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---- schedules -------------------------------------------------------
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ---- smooth max via log‑sum‑exp --------------------------------------
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        # subtract max again to avoid overflow (not strictly needed)
        shifted -= np.max(shifted)
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over convolution entries

        # ---- gradient of smooth objective w.r.t. a ---------------------------
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length 2·n‑1
        segment = conv_ab[n - 1: 2 * n - 1]       # central n entries
        grad_a = 4.0 * n * segment[::-1]         # shape (n,)

        # ---- chain rule through the soft‑max (log‑softmax) ------------------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---- Adam update -----------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ---- occasional jitter ------------------------------------------------
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.004, size=n)

        # ---- restart if stuck ------------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return global_best_a, global_best_val


# ----------------------------------------------------------------------
# Phase B – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        global_best_a: np.ndarray, global_best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from indices that heavily contribute
    to the maximal convolution entry towards indices that contribute little.
    The convolution array is updated incrementally (O(n) per move).
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # select source index proportional to contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # choose target index that hurts the max entry the least
        if np.random.rand() < 0.35:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of convolution (O(n)) ----
        # note: the update formulas are derived from the exact expression
        # for (a+Δ)*(a+Δ) where only a[i] and a[j] change.
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept if we strictly improve the true objective
        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < global_best_val - 1e-12:
                global_best_val, global_best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new global best = {global_best_val:.6f}")
                if global_best_val <= target_val:
                    break
        else:
            no_improve += 1

        # restart if stuck for a long time
        if no_improve > IMPROVE_LIMIT:
            a = global_best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return global_best_a, global_best_val


# ----------------------------------------------------------------------
# Phase C – Exact sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       global_best_a: np.ndarray, global_best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Step size decays like η₀ / sqrt(t).  The iterate is kept symmetric.
    """
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = global_best_val

    eta0 = 0.5 / n          # initial magnitude comparable to average entry
    t = 0
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new local best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # adaptive step size
        t += 1
        eta = eta0 / np.sqrt(t)

        # step & projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # enforce symmetry (keeps the feasible set)
        a = _symmetrize(a)

        # restart if stagnant
        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a = a / a.sum()
            a = _symmetrize(a)
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    # update the global best if needed
    if best_local_val < global_best_val:
        global_best_val, global_best_a = best_local_val, best_local_a.copy()
        print(f"[SGD] final improvement to global best = {global_best_val:.6f}")

    return global_best_a, global_best_val


# ----------------------------------------------------------------------
# Small post‑processing: mixing with uniform
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few convex mixes (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = cand / cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

    )

        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
