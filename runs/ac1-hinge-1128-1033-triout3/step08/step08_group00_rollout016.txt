def _gradient_hillclimb(a0, n, deadline, start_time, best_a, best_val, target_val=1.503):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule params
    max_frac_start, max_frac_end = 0.6, 0.001
    temp_start, temp_end = 0.02, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 100_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # recompute max and K
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[HillGrad {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # identify active indices K (close to max)
        # tolerance relative epsilon maybe 1e-12 absolute
        K = np.where(b >= max_b - 1e-12)[0]

        # compute gradient w.r.t max entry(s)
        grad = np.zeros_like(a)
        for k in K:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # reverse slice of a to match indices: a[k - i] = a[i_start:i_end+1][::-1]
            rev_slice = a[i_start:i_end+1][::-1]
            grad[i_start:i_end+1] += 2.0 * rev_slice

        # source index: choose index with largest grad (most impact)
        src = np.argmax(grad)
        if a[src] <= eps:
            # if src amount near zero, pick another based on a maybe
            src = int(np.argmax(a))
            if a[src] <= eps:
                # cannot move
                break

        # target selection: bias towards low grad
        tgt_weights = 1.0 / (grad + eps)
        tgt_weights[src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            tgt = int(np.random.choice(n, p=tgt_weights / wsum))
        else:
            # fallback random
            tgt = np.random.randint(n-1)
            if tgt >= src:
                tgt += 1

        # amount to move
        delta = a[src] * np.random.rand() * max_frac
        if delta <= 0:
            continue

        # Apply incremental convolution update
        b_new = b.copy()
        b_new[tgt:tgt+n] += 2.0 * delta * a
        b_new[src:src+n] -= 2.0 * delta * a
        b_new[2*tgt] += delta * delta
        b_new[2*src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept if improvement or metropolis with temperature T
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        elif T > 0:
            prob = np.exp(-(new_val - cur_val) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # reinitialize with noise
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillGrad {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

"""
propose_candidate
-----------------

Find a non‑negative step function (list of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

under the 1000 s budget.

Algorithm
~~~~~~~~~
1. **Initialisation** – try to reuse a previous solution, otherwise
   explore a Beta‑family and a few Dirichlet draws.
2. **Adam optimisation** – minimise a smooth approximation of the
   max‑convolution (log‑sum‑exp) with a variance‑penalty.
3. **Gradient‑driven hill‑climbing** – repeatedly move a small amount of
   mass from the index that most reduces the current maximal convolution
   entry (obtained from the exact sub‑gradient) to an index that hurts
   it the least.  A simulated‑annealing acceptance rule allows uphill
   moves early on.
4. **Projected‑gradient polishing** – a few projected‑gradient steps on
   the true max‑convolution objective to squeeze out the last bits.
5. **Final evaluation** – return the best sequence found.

All helper functions are top‑level, no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser (smooth max + variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam on 2·n·smooth_max + λ·Var(conv(a,a))."""
    w = np.log(a0 + 1e-12)               # logits → simplex via soft‑max
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.001
    eps_start, eps_end = 0.8, 1e-8
    lam_start = 5.0 * n
    lam_end = 0.0

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 20_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---------- simplex ↔ logits ----------
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # ---------- schedules ----------
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        lam = lam_start * (1.0 - progress) + lam_end * progress

        # ---------- smooth‑max ----------
        max_b = np.max(_full_convolution(a))
        shifted = (max_b - max_b) / eps  # placeholder for consistency
        # compute soft‑max weights directly (log‑sum‑exp trick)
        b = _full_convolution(a)
        maxb = b.max()
        shifted = (b - maxb) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()            # shape (2n‑1,)

        # gradient of smooth max (2·n factor already included later)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)         # length 3n‑2
        segment = conv_ab[n - 1:2 * n - 1]     # n entries centred
        grad_smooth = 4.0 * n * segment[::-1]

        # ---------- variance ----------
        m_len = 2 * n - 1
        avg = 1.0 / m_len
        v_arr = b - avg
        var = np.mean(v_arr * v_arr)
        conv_var = _conv_fft(a, v_arr[::-1])
        segment_var = conv_var[n - 1:2 * n - 1]
        grad_var = (4.0 / m_len) * segment_var[::-1]

        grad_a = grad_smooth + lam * grad_var

        # ---------- chain rule through soft‑max ----------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---------- Adam update ----------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.018, size=n)

        # restart on stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient‑driven hill‑climbing
# ----------------------------------------------------------------------
def _hillclimb_gradient(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that moves mass from the index with the largest
    sub‑gradient (i.e. the one that most lowers the current maximal
    convolution entry) to an index that hurts it the least.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules
    max_frac_start, max_frac_end = 0.60, 0.001
    temp_start, temp_end = 0.020, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 80_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[HillG {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # indices attaining the current maximum (within tiny tolerance)
        K = np.where(b >= max_b - 1e-12)[0]

        # exact sub‑gradient of the (hard) max entry
        grad = np.zeros_like(a)
        for k in K:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # a_{k-i} = a[i_start:i_end+1][::-1]
            grad[i_start:i_end + 1] += 2.0 * a[i_start:i_end + 1][::-1]

        src = int(np.argmax(grad))
        if a[src] <= eps:
            # fallback to the largest coefficient
            src = int(np.argmax(a))
            if a[src] <= eps:
                break

        # target biased towards indices with small gradient impact
        tgt_weights = 1.0 / (grad + eps)
        tgt_weights[src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            tgt = int(np.random.choice(n, p=tgt_weights / wsum))
        else:
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1

        delta = a[src] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[HillG {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – perturb around the current best
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillG {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Final projected‑gradient polishing (sub‑gradient descent on the true max)
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray,
                                n: int,
                                deadline: float,
                                start_time: float,
                                best_a: np.ndarray,
                                best_val: float,
                                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """A few projected‑gradient steps on the exact max‑convolution objective."""
    a = a0.copy()
    cur_val = 2.0 * n * _full_convolution(a).max()
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[PGD {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        # active set
        K = np.where(b >= max_b - 1e-12)[0]

        # sub‑gradient of the max entry
        grad = np.zeros_like(a)
        for k in K:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            grad[i_start:i_end + 1] += 2.0 * a[i_start:i_end + 1][::-1]

        # simple back‑tracking line search
        progress = (now - start_time) / (deadline - start_time)
        step_start, step_end = 0.02, 1e-7
        step = step_start * (1.0 - progress) + step_end * progress
        improved = False
        for _ in range(8):
            a_candidate = _project_simplex(a - step * grad)
            new_val = 2.0 * n * _full_convolution(a_candidate).max()
            if new_val < cur_val:
                a = a_candidate
                cur_val = new_val
                improved = True
                break
            step *= 0.5
        if not improved:
            # cannot find improvement – stop early
            break
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# A cheap sweep of a Beta‑shaped family (used for initialisation)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a plain Python list (sum of entries is 1, the scaling does
    not affect the final score because evaluate_sequence is homogeneous).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin for final call

    # Problem size – a balance between expressiveness and speed.
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – use the previous solution if present.
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep + a few random Dirichlet seeds
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    try:
        shape_a, shape_val = _beta_family_best(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] beta‑family sweep improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # a few random Dirichlet draws
    for _ in range(3):
        rand_a = np.random.dirichlet(np.ones(n))
        val = _objective(rand_a)
        if val < best_val:
            best_a, best_val = rand_a, val
            print(f"[init] random Dirichlet seed improved value={best_val:.6f}")

    # fallback random start
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] random start, value=", best_val)

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈30 % of total budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # Phase B – Gradient‑driven hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_gradient(best_a, n,
                                           deadline=deadline,
                                           start_time=phase_a_deadline,
                                           best_a=best_a,
                                           best_val=best_val,
                                           target_val=1.503)

    # ------------------------------------------------------------------
    # Phase C – Very short projected‑gradient polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_a, best_val = _projected_gradient_descent(best_a, n,
                                                       deadline=deadline,
                                                       start_time=start_time,
                                                       best_a=best_a,
                                                       best_val=best_val,
                                                       target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
