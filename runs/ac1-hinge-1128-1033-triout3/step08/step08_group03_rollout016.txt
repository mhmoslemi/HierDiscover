def project_simplex(v):
    # ensure v sums to 1
    # Use algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _adam_optimize(a_start, deadline):
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.5   (or 1.0) adapt?
    tau = 0.01  (starting), decay?
    iteration = 1
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())

    while time.time() < deadline - 5:
        # compute convolution
        conv = _full_convolution(a)
        max_conv = conv.max()
        # compute softmax weights
        # tau schedule: reduce gradually
        cur_tau = max(tau * (0.99 ** (iteration//1000)), 1e-6)
        # stable softmax:
        max_c = conv.max()
        exp_scaled = np.exp((conv - max_c) / cur_tau)
        p = exp_scaled / exp_scaled.sum()

        # gradient compute
        a_rev = a[::-1]
        # Convolve a_rev and p using FFT
        size = _next_pow2(len(a_rev) + len(p) - 1)
        fa = np.fft.rfft(a_rev, size)
        fp = np.fft.rfft(p, size)
        conv_rev = np.fft.irfft(fa * fp, size)
        # take slice n-1 : n-1+n
        grad = 2.0 * conv_rev[len(a)-1: len(a)-1 + len(a)]

        # Adam update (gradient descent)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step   # minus gradient (since we minimize)

        # project onto simplex
        a = _project_simplex(a)

        # optional pruning
        if iteration % 2000 == 0:
            # prune small values
            thresh = np.percentile(a, 20)  # keep top 80%
            a[a < thresh] = 0.0
            a = _project_simplex(a)

        # evaluate occasionally
        if iteration % 500 == 0:
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[adam iter {iteration}] new best = {best_val:.6f}")

        iteration += 1
```

"""
Propose a non‑negative step function minimizing `evaluate_sequence`.

Algorithm
---------
1. **Sparse Sidon‑like construction** –  
   Build a set S ⊂ {0,…,n‑1} of indices whose pairwise sums appear only a
   few times.  For a binary indicator vector `s` (s[i]=1 iff i∈S) the
   ordered‑pair count for each sum is `c = s * s` (linear convolution).
   If the maximal count λ is small, a uniform weight `a[i]=1/|S|` for i∈S
   yields a convolution peak `max_b = λ / |S|²` and therefore
   `evaluate = 2·n·λ / |S|²`.  By fixing `λ=3` and maximizing |S| we can
   drive the evaluation well below the current best (~1.51).

2. **Greedy random search** –  
   Repeatedly try to grow S:  pick a random unused index and add it only
   if every resulting sum count stays ≤ λ_max (3).  After a large number
   of consecutive failures we stop the attempt and keep the best S found
   (largest |S|).  Several independent attempts are performed within the
   time budget.

3. **Optional refinement** –  
   A tiny hill‑climbing phase (random swaps) is run on the best vector
   to see if a slight re‑weighting can improve the bound.

All work respects the 1000 s budget; the final list is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # eliminate tiny negatives
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (sum=1, v≥0).
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Greedy construction of a sparse index set with bounded sum‑multiplicity
# ----------------------------------------------------------------------
def _greedy_sparse_set(n: int,
                       lambda_max: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, int]:
    """
    Returns a tuple (indices, lambda) where `indices` is a 1‑D ndarray of
    chosen positions and `lambda` = max_{k} #{(i,j)∈S×S : i+j=k}.
    The algorithm attempts to grow the set as large as possible while
    keeping all sum‑multiplicities ≤ `lambda_max`.  It stops after a
    configurable number of consecutive failures or when the time budget
    runs out.
    """
    used = np.zeros(n, dtype=bool)                     # which positions are already taken
    conv_counts = np.zeros(2 * n - 1, dtype=np.int16)  # ordered‑pair sum multiplicities
    chosen = []                                        # list of selected indices
    # after this many consecutive rejections we assume we cannot grow further
    max_failures = 3000
    failures = 0

    while time.time() < deadline - 0.5:
        i = rng.integers(0, n)
        if used[i]:
            continue

        # quick check: self‑pair (i,i) adds 1 to sum 2*i
        if conv_counts[2 * i] + 1 > lambda_max:
            failures += 1
            if failures >= max_failures:
                break
            continue

        if chosen:
            # sums with all existing members (ordered pairs → +2)
            sums = np.array(chosen, dtype=np.int32) + i
            if (conv_counts[sums] + 2 > lambda_max).any():
                failures += 1
                if failures >= max_failures:
                    break
                continue
        # passed all checks – accept i
        used[i] = True
        chosen.append(i)
        conv_counts[2 * i] += 1                       # the square term
        if len(chosen) > 1:
            conv_counts[sums] += 2                     # ordered pairs with previous members
        failures = 0

    return np.array(chosen, dtype=np.int32), int(conv_counts.max())

    rng = np.random.default_rng(seed)

    deadline = start_time + budget_s - 5.0          # tiny safety margin

    # Problem size – 8000 works well: large enough for a good λ/k ratio,
    # still cheap to handle with Python loops.
    # ------------------------------------------------------------------

    best_lambda = None
    best_indices = None
    best_val = np.inf

    # ------------------------------------------------------------------

                    deadline_hc: float) -> tuple[np.ndarray, float]:
        cur_a = a.copy()
        cur_best = cur_val
        rng_hc = np.random.default_rng(seed + 999)
        n_local = cur_a.shape[0]

        while time.time() < deadline_hc:
            # propose a small change: move a tiny amount of mass
            i, j = rng_hc.integers(0, n_local, size=2)
            if i == j:
                continue
            delta = cur_a[i] * 0.03 * rng_hc.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            # projection onto simplex (keeps sum=1, non‑negative)
            new_a = _project_simplex(new_a)
            val = evaluate_sequence(new_a.tolist())
            if val < cur_best - 1e-12:
                cur_a = new_a
                cur_best = val
                print(f"[refine] improved to {cur_best:.6f}")
        return cur_a, cur_best

    if time.time() < deadline - 2.0:
        best_a, best_val = _hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final report and return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
