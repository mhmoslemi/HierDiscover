"""
Hybrid optimisation for the min‑max auto‑convolution problem.

1️⃣ Initialise the distribution from a beta‑family sweep, a few
   random mixtures of betas and (optionally) an interpolation of the
   previously best solution.
2️⃣ Perform a long Adam run on the *smooth* objective
   L(a)=2·n·smooth_max(a*a)+λ·Var(a*a) with carefully tuned schedules.
   The simplex constraint Σa=1 is enforced via a soft‑max (log‑its) parametrisation.
3️⃣ Finish with a simulated‑annealing hill‑climber that repeatedly
   transfers a small amount of mass from indices contributing strongly
   to the current maximal convolution entry to low‑contributing indices.
   Aggressive transfer‑fraction and temperature schedules are used.

The whole search respects the 1000 s budget (0.5 s safety margin).
The best vector is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights (log‑sum‑exp trick)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Variance of the convolution and its gradient
# ----------------------------------------------------------------------
def _var_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns   var(b) = mean((b‑avg)²)   and   ∂var/∂a  (no extra scaling).
    Here b = a*a and avg = 1/(2n‑1) because Σb = (Σa)² = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # gradient: (4/m) * cross‑correlation of a with v
    conv_ab = _conv_fft(a, v[::-1])              # length n+m-1 = 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad = (4.0 / m) * segment[::-1]
    return var, grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Beta‑family sweep (quick, cheap)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search the two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth max + variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize_mod(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       best_a: np.ndarray,
                       best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises
        L(a) = 2·n·smooth_max(a) + λ·Var(conv(a,a)),
    with schedules tuned for a larger problem size.
    """
    # initialise logits (soft‑max parametrisation)
    w = np.log(a0 + 1e-12) + np.random.randn(n) * 0.01

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.22, 0.001
    eps_start, eps_end = 0.6, 1e-9

    # variance‑penalty schedule (stronger at start)
    lam_start = 8.0 * n
    lam_end = 0.0

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 15000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ------- simplex ↔ logits -------
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # ------- schedules -------
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        lam = lam_start * (1.0 - progress) + lam_end * progress

        # ------- smooth‑max -------
        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)

        # ------- variance -------
        var_val, grad_var = _var_and_grad(a)

        # total gradient of the loss L (to be minimised)
        grad_a = grad_smooth + lam * grad_var

        # chain rule through the soft‑max (gradient w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to avoid lock‑in
        if np.random.rand() < 0.007:
            w += np.random.normal(scale=0.015, size=n)

        # restart on prolonged stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.12
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1
        if iteration % 20000 == 0:
            print(f"[Adam] iteration {iteration}, best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (aggressive version)
# ----------------------------------------------------------------------
def _hillclimb_anneal_mod(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          best_a: np.ndarray,
                          best_val: float,
                          target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that moves mass from indices heavily contributing to the
    current maximal auto‑convolution entry to low‑contributing indices.
    Aggressive transfer‑fraction and temperature schedules are used.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules (more aggressive than the original version)
    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.05, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # index where convolution attains its maximum
        k_max = int(np.argmax(b))

        # contribution of each a[i] to this entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – sample proportionally to contribution
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – bias towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of probability mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        # additive part 2·δ·a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic part δ²
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # escape stagnation by adding small noise and re‑normalising
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1
        if iteration % 50000 == 0:
            print(f"[Hill] iteration {iteration}, best={best_val:.6f}")

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 20000 offers finer resolution without a big
    #                increase in FFT cost (next power‑of‑two stays 2¹⁶).
    # --------------------------------------------------------------
    n = 20000

    # --------------------------------------------------------------
    # Initialise a – try to reuse a previously found solution.
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        best_a, best_val = a0.copy(), _objective(a0)
        print("[init] interpolated previous solution, value={:.6f}".format(best_val))
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweeps (Beta family) + random mixtures.
    # --------------------------------------------------------------
    try:
        shape_a, shape_val = _beta_family_best(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print("[init] beta‑family sweep improved, value={:.6f}".format(best_val))
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # a few random mixtures of two beta distributions
    for _ in range(5):
        p1, q1 = np.random.uniform(0.1, 6.0), np.random.uniform(0.1, 6.0)
        p2, q2 = np.random.uniform(0.1, 6.0), np.random.uniform(0.1, 6.0)
        w = np.random.rand()
        i = np.arange(1, n + 1, dtype=np.float64)
        a_mix = (w * (i ** (p1 - 1.0)) * ((n - i + 1) ** (q1 - 1.0)) +
                 (1 - w) * (i ** (p2 - 1.0)) * ((n - i + 1) ** (q2 - 1.0)))
        s = a_mix.sum()
        if s > 0:
            a_mix /= s
            val = _objective(a_mix)
            if val < best_val:
                best_a, best_val = a_mix, val
                print("[init] beta‑mixture improved, value={:.6f}".format(best_val))

    # a few random Dirichlet draws for diversity
    for _ in range(3):
        rand_a = np.random.dirichlet(np.ones(n))
        val = _objective(rand_a)
        if val < best_val:
            best_a, best_val = rand_a, val
            print("[init] random Dirichlet improved, value={:.6f}".format(best_val))

    # fallback random start
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] random start, value={:.6f}".format(best_val))

    # --------------------------------------------------------------
    # Phase A – Adam optimisation (≈55 % of total budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.55 * budget_s
    best_a, best_val = _adam_optimize_mod(best_a, n,
                                          deadline=phase_a_deadline,
                                          start_time=start_time,
                                          best_a=best_a,
                                          best_val=best_val,
                                          target_val=1.503)

    # --------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (remaining budget)
    # --------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal_mod(best_a, n,
                                             deadline=deadline,
                                             start_time=phase_a_deadline,
                                             best_a=best_a,
                                             best_val=best_val,
                                             target_val=1.503)

    # --------------------------------------------------------------
    # Final evaluation using the provided secure wrapper
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score = {:.6f}".format(final_score))
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
