"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The algorithm consists of four phases:

1. **Diverse initialisation** – several deterministic and random
   probability vectors (uniform, previous best, Dirichlet, U‑shaped
   beta‑distribution) are normalised and the best is kept.

2. **Aggressive L‑BFGS on a smooth surrogate** – the max‑convolution
   is replaced by a log‑sum‑exp approximation with a schedule of
   decreasing smoothing parameters ε.  The optimisation is performed
   in the soft‑max (log‑probability) parametrisation and a handful of
   random restarts are used.  After each run the result is projected
   back onto the simplex and an anti‑symmetric re‑ordering heuristic
   is tried.

3. **Projected sub‑gradient descent on the exact max‑convolution** –
   the exact sub‑gradient of the max‑norm is obtained from all
   indices attaining (or almost attaining) the maximum.  A simple
   Euclidean projected gradient step with a slowly decaying step size
   is applied; every improvement updates the global best.

4. **Simulated‑annealing hill‑climber + cheap polish** – random mass
   transfers and swaps are performed on the simplex with a
   temperature schedule; strict improvements are also fed into a final
   cheap mirror‑descent polishing routine.

All phases respect the 1000 s time budget.  The final candidate is
checked with the official ``evaluate_sequence`` before being returned.
"""
import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None        # smoothing parameter ε
_GLOBAL_N = None          # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negative values caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ---------- soft‑max ----------
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---------- convolution ----------
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ---------- smooth max ----------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    weights = np.exp(shifted)
    p = weights / weights.sum()      # probability distribution over lag indices

    # derivative of smooth max w.r.t. each conv entry = p
    # grad_a = 2·n * d(max)/da = 2·n * Σ_k p_k * d conv_k / da
    # d conv_k / da = a reversed * indicator → convolution with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # slice that corresponds to ∂conv/∂a (indices n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]    # reverse to align with a_i order

    # ---------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the full formula
    simplifies to this expression).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places large values on one side and small values on the
    opposite side – often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %). The vector is kept on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def _subgradient_max_conv(a: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Compute the full self‑convolution, its maximum, and a sub‑gradient of
    max(conv) w.r.t. a (using all indices that attain the maximum up to a
    tiny tolerance).
    Returns (conv, max_val, g) where g_i = Σ_{k∈K} Σ_{j: i+j=k} a_j.
    """
    conv = _full_convolution(a)
    max_val = conv.max()
    # select all entries that are (almost) maximal
    tol = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - tol
    ks = np.nonzero(mask)[0]

    n = a.shape[0]
    g = np.zeros_like(a)

    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx
        g[idx] += a[jdx]

    return conv, max_val, g
