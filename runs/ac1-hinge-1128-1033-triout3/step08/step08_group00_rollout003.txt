"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm Overview
------------------
1. **Initialisation**
   * Try to interpolate a previously found solution (``height_sequence_1``).
   * Perform an exhaustive sweep of the two‑parameter Beta family
     a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}.
   * Augment it with many random Beta draws (log‑uniform sampling of p,q)
     and a few Dirichlet samples.
   * Keep the best among all these cheap candidates.

2. **Adam optimisation**
   * Optimise a smooth surrogate of the true objective:
       L(a) = 2·n·smooth_max(conv(a,a)) + λ·Var(conv(a,a))
   * The simplex constraint Σa = 1 is enforced by a log‑softmax
     (log‑its) parametrisation.
   * λ (variance‑penalty weight) starts very large (≈25·n) and
     anneals to 0, forcing the convolution to flatten.
   * The optimiser restarts if it plateaus for a long time.

3. **Hill‑climbing (simulated annealing)**
   * A fast O(n) local search that repeatedly moves a chunk of mass
     from an index that contributes heavily to the current maximal
     convolution entry to an index that contributes little.
   * The maximal transfer fraction and the temperature start large
     (0.8 and 0.02) and are annealed to near zero; this allows both
     aggressive exploration early and fine polishing later.
   * Stagnation triggers a small random perturbation.

4. **Deterministic sub‑gradient polishing**
   * After the stochastic phase we run a few thousand projected
     sub‑gradient steps on the exact max‑convolution.  The sub‑gradient
     of max(b) (b = a∗a) is readily available:
         ∂max(b)/∂a_j = 2·a_{k_max‑j}   (for all k_max attaining the max).
   * The step size decays linearly; after each step the vector is
     projected back onto the probability simplex.
   * Any improvement updates the current best solution.

The whole procedure constantly checks the remaining time and stops
shortly before the 1000 s budget expires, leaving a final call to
``evaluate_sequence`` for the official score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights (log‑sum‑exp trick)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Variance of the convolution and its gradient
# ----------------------------------------------------------------------
def _var_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns   var(b) = mean((b‑avg)²)   and   ∂var/∂a  (no extra scaling).
    Here b = a*a and avg = 1/(2n‑1) because Σb = (Σa)² = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # gradient: (4/m) * cross‑correlation of a with v
    conv_ab = _conv_fft(a, v[::-1])              # length n+m-1 = 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad = (4.0 / m) * segment[::-1]
    return var, grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth max + variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises
        L(a) = 2·n·smooth_max(a) + λ·Var(conv(a,a)),
    where λ is linearly annealed from λ_start to 0.
    The simplex constraint Σa = 1 is enforced via a log‑softmax
    parametrisation.
    """
    # initialise logits (soft‑max parametrisation)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.001
    eps_start, eps_end = 0.8, 1e-8

    # variance‑penalty schedule (much larger than before)
    lam_start = 25.0 * n          # a stronger flattening force
    lam_end = 0.0

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 15_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- decode logits to probabilities (simplex) -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # ----- schedule parameters -----
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        lam = lam_start * (1.0 - progress) + lam_end * progress

        # ----- smooth‑max and its gradient -----
        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)

        # ----- variance penalty -----
        var_val, grad_var = _var_and_grad(a)

        # total gradient of the loss (we minimise)
        grad_a = grad_smooth + lam * grad_var

        # chain rule through the soft‑max (gradient w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape shallow plateaus
        if np.random.rand() < 0.009:
            w += np.random.normal(scale=0.02, size=n)

        # restart if no progress for a while
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (enhanced schedule)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast local search that moves a small amount of mass from an index that
    heavily contributes to the current maximal convolution entry towards an
    index that contributes little.  The schedule for the maximal transfer
    fraction starts larger (0.8) to allow aggressive jumps early on.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule parameters
    max_frac_start, max_frac_end = 0.8, 0.001
    temp_start, temp_end = 0.02, 1e-8

    no_improve = 0
    IMPROVE_LIMIT = 80_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to this entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – probability ∝ contribution
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – bias towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback (excluding i)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move (up to max_frac of source)
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – small perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503,
                        max_iters: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        f(a) = 2·n·max(conv(a,a))
    using the analytically known sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # sub‑gradient of max(conv) w.r.t. a
        b = _full_convolution(a)
        max_val = b.max()
        k_maxes = np.where(b == max_val)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step size schedule (starts moderately aggressive)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.3 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Subgrad {it:6d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        # occasional jitter to shake off flat regions
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric sweep of a Beta‑shaped family (quick, cheap)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _beta_random_search(n: int, trials: int = 250) -> tuple[np.ndarray, float]:
    """Randomly sample p,q on a log‑scale and keep the best Beta shape."""
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)
    rng = np.random.default_rng()
    for _ in range(trials):
        # log‑uniform sampling of p,q in [1e-3, 10]
        logp = rng.uniform(-6.9, 2.3)   # ln(0.001) .. ln(10)
        logq = rng.uniform(-6.9, 2.3)
        p = np.exp(logp)
        q = np.exp(logq)
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # problem size – a compromise between expressivity and speed
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise a – possibly from a previously best solution
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweeps (Beta grid + random Beta + Dirichlet)
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf

    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    try:
        shape_a, shape_val = _beta_family_best(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] beta‑family grid improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family grid failed:", exc)

    try:
        rand_a, rand_val = _beta_random_search(n, trials=300)
        if rand_val < best_val:
            best_a, best_val = rand_a, rand_val
            print(f"[init] beta‑family random improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta random search failed:", exc)

    # a few Dirichlet draws for extra diversity
    for _ in range(4):
        cand = np.random.dirichlet(np.ones(n))
        cand_val = _objective(cand)
        if cand_val < best_val:
            best_a, best_val = cand, cand_val
            print(f"[init] Dirichlet seed improved value={best_val:.6f}")

    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] fell back to uniform random start, value=", best_val)

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈40 % of the total budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.40 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )

    # ------------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (≈50 % of the budget)
    # ------------------------------------------------------------------
    hill_deadline = start_time + 0.90 * budget_s
    best_a, best_val = _hillclimb_anneal(
        best_a, n,
        deadline=hill_deadline,
        start_time=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )

    # ------------------------------------------------------------------
    # Phase C – Deterministic sub‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=deadline,
        start_time=hill_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
        max_iters=40_000
    )

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
