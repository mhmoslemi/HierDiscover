def _subgradient_polish(a, n, deadline, start_time, best_a, best_val, target_val=1.502):
    cur_a = a.copy()
    cur_val = _objective(cur_a)
    # maybe set step size schedule
    step0 = 0.1
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute conv
        b = _full_convolution(cur_a)
        max_b = b.max()
        # find indices close to max
        # tolerance small relative
        tol = max_b * 1e-12 + 1e-15
        ks = np.where(np.abs(b - max_b) <= tol)[0]
        # compute contributions for each i
        contrib = np.zeros_like(cur_a)
        for k in ks:
            i_min = max(0, k - (n-1))
            i_max = min(n-1, k)
            # indices i from i_min to i_max inclusive
            # j = k - i
            # add a[j] to contrib[i]
            # vectorize:
            i_range = np.arange(i_min, i_max+1)
            j_range = k - i_range
            contrib[i_range] += cur_a[j_range]
        # gradient
        grad = 2.0 * contrib
        # step size decreasing
        progress = (now - start_time) / (deadline - start_time)

"""
Proposes a non‑negative step function that minimizes ``evaluate_sequence``.

Main ideas
---------
* Start from a good analytic shape (Beta family) and, if available,
  interpolate the previous best solution.
* A short Adam phase on a smooth approximation of the true objective
  (max‑convolution + variance penalty) quickly flattens the convolution.
* A long simulated‑annealing hill‑climbing phase moves tiny amounts of mass
  from indices that contribute heavily to the current maximal convolution
  entry to indices that contribute little.  The transfer fraction and the
  temperature are annealed to zero, giving fine‑grained polishing.
* Finally a few sub‑gradient descent steps on the *exact* max are performed
  (using the full FFT convolution) to squeeze out the last bits of improvement.

The whole run respects the 1000 s budget.  The function returns a list of
non‑negative heights whose sum is 1 (any scaling is harmless because the
objective is scale‑invariant).  The code is deliberately self‑contained and
uses only NumPy and SciPy (for a tiny linear‑programming fallback if desired).
"""

import time
import numpy as np
from scipy import optimize  # only for a tiny LP fallback (optional)

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                      # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick (weights sum to 1)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                   # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n

    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Variance of the convolution and its gradient
# ----------------------------------------------------------------------
def _var_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns   var(b) = mean((b‑avg)²)   and   ∂var/∂a  (no extra scaling).
    Here b = a*a and avg = 1/(2n‑1) because Σb = (Σa)² = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # gradient: (4/m) * cross‑correlation of a with v
    conv_ab = _conv_fft(a, v[::-1])              # length 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad = (4.0 / m) * segment[::-1]
    return var, grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max + variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for
        L(a) = 2·n·smooth_max(a) + λ·Var(conv(a,a)).
    The variance‑penalty λ is linearly annealed to zero.
    """
    # logits for soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.003
    eps_start, eps_end = 0.20, 1e-8

    lam_start = 10.0 * n          # stronger flattening at the start
    lam_end = 0.0

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # --- simplex ↔ logits ---
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # --- schedules (based on progress within this phase) ---
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        lam = lam_start * (1.0 - progress) + lam_end * progress

        # --- smooth‑max ---
        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)

        # --- variance ---
        var_val, grad_var = _var_and_grad(a)

        # total gradient of the loss L (we minimise L)
        grad_a = grad_smooth + lam * grad_var

        # chain rule through the soft‑max (gradient w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation (helps escape flat regions)
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.018, size=n)

        # restart if stuck for a long time
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (fast O(n) local search)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass from an index that
    contributes heavily to the current maximal convolution entry to an index
    that contributes little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules
    max_frac_start, max_frac_end = 0.70, 0.0005
    temp_start, temp_end = 0.020, 1e-8

    no_improve = 0
    IMPROVE_LIMIT = 200_000
    full_recalc_interval = 20_000   # recompute b from scratch to avoid drift

    iteration = 0
    last_full = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # full recompute every once in a while (prevents numerical drift)
        if iteration - last_full >= full_recalc_interval:
            b = _full_convolution(a)
            last_full = iteration

        k_max = int(np.argmax(b))

        # contribution of each index to this entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (probability ∝ contribution)
        if total_contrib > 0.0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index (biased towards low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback to uniform among the rest
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        if i == j:
            continue

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        # add contribution of the transferred mass
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # correct the quadratic part
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.028, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exact max, very small steps)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       best_a: np.ndarray,
                       best_val: float,
                       target_val: float = 1.502) -> tuple[np.ndarray, float]:
    """
    A few sub‑gradient steps on the exact max(conv) objective.
    Each step recomputes the full convolution (expensive) but only a few
    hundred iterations are performed, enough for the last bit of polishing.
    """
    a = a0.copy()
    cur_val = _objective(a)
    step0 = 0.04   # initial step size

    while True:
        now = time.time()
        if now >= deadline:
            break

        # full convolution (exact)
        b = _full_convolution(a)
        max_b = b.max()

        # indices that are (virtually) maximal
        tol = max_b * 1e-12 + 1e-15
        max_idx = np.where(np.abs(b - max_b) <= tol)[0]

        # contribution of each a_i to the maximal entries
        contrib = np.zeros_like(a)
        for k in max_idx:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i_range = np.arange(i_min, i_max + 1, dtype=int)
            j_range = k - i_range
            contrib[i_range] += a[j_range]

        # sub‑gradient (any sub‑gradient works; we use 2*contrib)
        grad = 2.0 * contrib

        # step size annealing
        progress = (now - start_time) / (deadline - start_time)
        eta = step0 * (0.1 ** progress)   # decays quickly

        a_next = a - eta * grad
        a_next = _project_simplex(a_next)

        new_val = _objective(a_next)
        if new_val < cur_val:
            a = a_next
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Subgrad] new best = {best_val:.6f}")
            if cur_val <= target_val:
                break
        # otherwise keep the old a (no line‑search for simplicity)

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric sweep of a Beta‑shaped family (quick, cheap)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8000 entries give a good balance between expressivity
    #                and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previously found solution if available.
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Cheap parametric sweep (Beta family) – gives a decent starting shape
    # ------------------------------------------------------------------
    try:
        beta_a, beta_val = _beta_family_best(n)
        if a0 is None or beta_val < np.inf:
            a0 = beta_a
            print(f"[init] beta‑family start, value = {beta_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # ------------------------------------------------------------------
    # A couple of random Dirichlet draws for diversity (optional)
    # ------------------------------------------------------------------
    if a0 is None or np.random.rand() < 0.15:
        a0 = np.random.dirichlet(np.ones(n))
        print("[init] random Dirichlet start")

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    best_a = a0.copy()
    best_val = _objective(best_a)
    print(f"[init] initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – short Adam optimisation (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.15 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.505)

    # ------------------------------------------------------------------
    # Phase B – long hill‑climbing (remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=phase_a_deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # Very tiny sub‑gradient polishing (a few seconds left)
    # ------------------------------------------------------------------
    sub_deadline = deadline - 0.2
    if time.time() < sub_deadline:
        best_a, best_val = _subgradient_polish(best_a, n,
                                               deadline=sub_deadline,
                                               start_time=phase_a_deadline,
                                               best_a=best_a,
                                               best_val=best_val,
                                               target_val=1.502)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
