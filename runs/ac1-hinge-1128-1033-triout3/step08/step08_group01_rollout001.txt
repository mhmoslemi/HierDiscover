"""
Simulated‑annealing + sub‑gradient polishing for the
max‑convolution objective

Goal
----
Minimise                2·n·max(conv(a,a)) / (sum(a))²
with a_i ≥ 0 .  The objective is scale‑invariant, therefore we
keep the sequence normalised (Σ a = 1) and minimise
    2·n·max(conv(a,a))

Algorithm
---------
1. Initialise a with a cheap parametric sweep (uniform, exponentials,
   Dirichlet draws …).  If a previous best (`height_sequence_1`) is
   available it is interpolated to the current length.
2. Run a simulated‑annealing (SA) loop.
   * At each step move a small amount δ from an index that contributes
     heavily to the current maximal convolution entry to an index that
     contributes little.
   * The move is accepted if it improves the objective, otherwise
     with a Metropolis probability that cools down with time.
   * The SA step size (max δ) is also annealed – it starts large
     (≈50 % of the source mass) and shrinks to ≈0.1 %.
   * After long periods without improvement we restart from the best
     solution found so far.
3. A short projected sub‑gradient descent (PGD) stage refines the
   final SA solution.  The sub‑gradient of max(conv) is simply
   2·a_{k*‑i} where k* is the maximal index.
4. Finally try a few convex mixtures with the uniform distribution –
   this often reduces the peak a little more.
5. Return the best sequence (list of floats).  All numbers are non‑negative
   and sum to 1, therefore the security wrapper `evaluate_sequence`
   will not reject the answer.

The whole search respects the 1000 s budget; a small safety margin
is kept for the final call to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two for FFT speed
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise can produce tiny negative values – clamp to zero
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small library of hand‑crafted initial candidates
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising a (Σa=1) and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = 2.0 * n * _full_convolution(best_a).max()

    def _consider(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = 2.0 * n * _full_convolution(a).max()
        if val < best_val:
            best_val, best_a = val, a.copy()

    # uniform
    _consider(np.ones(n))

    # exponential decay (mass concentrated at the left)
    for r in (0.9995, 0.9997, 0.9999):
        a = r ** np.arange(n, dtype=float)
        _consider(a)

    # symmetric exponential (U‑shaped)
    for r in (0.9995, 0.9997, 0.9999):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _consider(a)

    # power‑law symmetric
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p) + (n - i + 1) ** (-p)
        _consider(a)

    # half‑wave sine (zero at the ends)
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _consider(a)

    # triangular (peak in the centre, zero at the ends)
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    _consider(a)

    # a few Dirichlet samples
    rng = np.random.default_rng(12345)
    for _ in range(12):
        a = rng.dirichlet(np.ones(n))
        _consider(a)

    return best_a, best_val

    cur_val = 2.0 * n * cur_max

    best_a = best_sofar_a.copy()
    best_val = best_sofar_val

    # annealing schedules
    max_frac_start, max_frac_end = 0.5, 0.001   # fraction of source mass to move
    temp_start, temp_end = 0.02, 1e-6          # Metropolis temperature
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        # linear temperature decay and max‑fraction decay
        temp = temp_start * (temp_end / temp_start) ** progress
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # --------- locate the currently maximal convolution entry ----------
        k_max = int(np.argmax(b))

        # contributions of each a_i to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a[mask] * a[idx[mask]]
        tot_contrib = contrib.sum()

        # choose source index (high contribution) -------------------------------------------------
        if tot_contrib > 0 and np.random.rand() < 0.9:
            prob_i = contrib / tot_contrib
            i = np.random.choice(n, p=prob_i)
        else:
            # fall back to a random heavy index
            i = int(np.argmax(contrib))

        # choose target index (low contribution) ---------------------------------------------------
        # discourage picking the same index
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        sum_inv = inv.sum()
        if sum_inv > 0 and np.random.rand() < 0.9:
            prob_j = inv / sum_inv
            j = np.random.choice(n, p=prob_j)
        else:
            # random different index
            j = (i + np.random.randint(1, n)) % n

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental update of the convolution (exact) -----
        # we keep a copy only while testing the move
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ----- Metropolis acceptance -----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probability decays with the excess over current value
            if temp > 0:
                prob = np.exp(-(new_val - cur_val) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # commit move
            a[i] -= delta
            a[j] += delta
            # numerical safety
            if a[i] < 0:
                a[i] = 0.0
            # renormalise to avoid any drift (the move conserves mass,
            # but rounding errors may accumulate)
            a_sum = a.sum()
            if abs(a_sum - 1.0) > 1e-12:
                a /= a_sum
            b = b_new
            cur_max = new_max
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[SA {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # occasional full recomputation to kill accumulated rounding error
        if (it + 1) % 5000 == 0:
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max

        # restart if stuck for a long time
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            no_improve = 0
            print(f"[SA {it:7d}] restart after stagnation")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (tiny polishing stage)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         start: float,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """PGD on the true max‑convolution objective."""
    n = a0.shape[0]
    a = a0.copy()
    best_local_a = best_a.copy()
    best_local_val = best_val

    # step‑size schedule (starting roughly at the average entry)
    eta0 = 0.5 / n
    eta = eta0
    decay = 0.9995
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[PGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        # index of the current maximum
        k_max = int(np.argmax(b))

        # sub‑gradient of that entry (2·a_{k_max‑i})
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # projected gradient step onto the simplex
        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # shake and restart from the best seen so far
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[PGD {it:7d}] perturbing after stagnation")
        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small post‑processing: convex mixture with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few convex combinations with the uniform distribution."""
    best_a, best_val = a, 2.0 * n * _full_convolution(a).max()
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = 2.0 * n * _full_convolution(cand).max()
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation objective.
    The implementation mixes a parametric warm‑start, simulated annealing,
    a short projected sub‑gradient polishing phase and a final uniform
    mixing sweep.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem size – a few thousand entries give a good trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # Warm‑start: either interpolate a previously known good sequence or
    #               fall back to a quick parametric sweep.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    # initial evaluation
    cur_val = 2.0 * n * _full_convolution(a0).max()
    best_a, best_val = a0.copy(), cur_val
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Simulated annealing (≈60 % of the time budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.60 * (deadline - start_time)
    best_a, best_val = _simulated_annealing(
        best_a,
        start=start_time,
        deadline=phase_a_deadline,
        best_sofar_a=best_a,
        best_sofar_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase B – Short projected sub‑gradient descent (≈30 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.90 * (deadline - start_time)
    best_a, best_val = _subgradient_descent(
        best_a,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase C – Final uniform mixing sweep
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final secure evaluation (the wrapper clips negatives, NaNs, etc.)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
