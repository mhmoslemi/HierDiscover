"""
propose_candidate.py

Searches for a non‑negative step function (list of floats) that minimises
    evaluate_sequence(sequence)

Algorithm outline
-----------------
0. **Initialisation** – start from the previous best (if available) or a
   uniform vector.  Normalise to the probability simplex (sum = 1).

1. **Broad candidate sweep** – generate many diverse shapes (random
   exponentials, symmetric power‑laws, exponential decays, low‑concentration
   Dirichlet draws) and keep the best according to the true objective.
   A cheap anti‑symmetric re‑ordering is applied to each candidate.

2. **Sub‑gradient descent** – exponentiated mirror‑descent on the exact
   L∞‑objective (the maximal self‑convolution).  An active set of near‑max
   indices is identified each iteration; the gradient is obtained with a
   single FFT.  The step‑size decays as 1/√t and the active‑set tolerance is
   tightened over time.

3. **Smooth‑surrogate optimisation** – minimise a log‑sum‑exp smoothed
   version of the max‑convolution using L‑BFGS‑B.  Several restarts are
   performed; after each run the anti‑symmetric re‑ordering is tried.

4. **Targeted refinement** – fast local moves that explicitly attack the
   pair of coefficients responsible for the current maximal convolution
   peak.  A small fraction of mass is transferred from the “offending”
   index to a low‑mass index (typically near the centre).  Moves are kept
   if they improve the objective, otherwise they are accepted with a
   simple annealing probability.  Periodic re‑ordering keeps the shape
   favourable.

All stages respect the time budget (default 1000 s) and always keep the
vector on the simplex.  The final best vector is returned as a plain list
of floats.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length) via FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)        # eliminate tiny negatives
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values on the left and the smallest on the right,
    i.e. pair a large coefficient with a small one when the sequence is reversed.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # large → left side
        res[n - 1 - i] = s[i]     # small → right side
    if n % 2:
        res[half] = s[half]
    return res


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)                (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # self‑convolution
    conv = _full_convolution(a)
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t a
    weights = np.exp(shifted)
    p = weights / weights.sum()            # p_k over convolution indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries give derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # ∂L/∂a_i

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule: geometric from 5·10⁻³ down to 10⁻¹²
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=12)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,

                       jac=True,

                                'gtol': 1e-12,

        w = res.x

        # map back to a (probability simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())

# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} a_{k‑i} using a single FFT.
    ``active_idx`` contains convolution indices (0 … 2n‑2) belonging
    to the active set.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n-2) - i] for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent_multi(a_start: np.ndarray, deadline: float):
    """
    Exponentiated mirror‑descent that reduces the *active* convolution peaks.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 1.0                     # a slightly larger initial step‑size
    rel_tol = 0.05                   # start with a generous active‑set tolerance

        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten tolerance gradually

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative update
        a = a * np.exp(-alpha * g)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # logging & occasional re‑ordering

            if cur_val < best_val:

            if it % 5000 == 0:

                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[sgd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val

def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (dist_to_end+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


def _exponential_candidate(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential decay: a_i ∝ exp(-α·dist_to_end)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-alpha * d)
    a /= a.sum()
    return a

def _targeted_refinement(a: np.ndarray, cur_best_val: float, deadline: float):
    """
    Focused local search that repeatedly moves a small fraction of the
    mass from the coefficient pair responsible for the current maximal
    convolution peak to a low‑mass index.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_best_val
    best_a = a.copy()
    best_val = cur_best_val

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1

        # fast evaluation (no call to evaluate_sequence)
        conv = _full_convolution(cur_a)
        sum_a = cur_a.sum()
        max_b = conv.max()
        cur_val_est = 2.0 * n * max_b / (sum_a ** 2)

        # index of the maximal convolution entry
        kmax = int(np.argmax(conv))

        # indices i,j that contribute to this kmax
        i_start = max(0, kmax - (n - 1))
        i_end = min(kmax, n - 1)
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = kmax - i_idx
        contrib = cur_a[i_idx] * cur_a[j_idx]

        # pick the most offending pair
        pair = int(np.argmax(contrib))
        i_sel = i_idx[pair]
        j_sel = j_idx[pair]
        src = i_sel if cur_a[i_sel] >= cur_a[j_sel] else j_sel

        # destination = the currently smallest coefficient (usually near centre)
        dst = int(np.argmin(cur_a))
        if src == dst:
            continue

        # move a modest fraction of mass
        delta = cur_a[src] * 0.02 * rng.random()   # up to 2 % of src
        if delta <= 0.0:
            continue

        new_a = cur_a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        # ensure non‑negativity and re‑normalise
        if new_a[src] < -1e-12:
            continue
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        # fast evaluation of the proposal
        conv_new = _full_convolution(new_a)
        max_b_new = conv_new.max()
        val_new = 2.0 * n * max_b_new / (s ** 2)

        # acceptance rule
        if val_new < cur_val_est:
            cur_a = new_a
            cur_val_est = val_new
            if val_new < best_val:
                best_val = val_new
                best_a = new_a.copy()
                if it % 500 == 0:
                    official = evaluate_sequence(best_a.tolist())
                    best_val = official
                    print(f"[targeted iter {it}] new best ≈ {best_val:.6f}")
        else:
            # occasional uphill moves (simple annealing)
            T = 0.01 * (0.9995 ** it)
            if np.exp(-(val_new - cur_val_est) / max(T, 1e-12)) > rng.random():
                cur_a = new_a
                cur_val_est = val_new

        # occasional re‑ordering to keep the anti‑symmetric pattern
        if it % 3000 == 0:
            cur_a = _anti_symmetric_reorder(cur_a)
            conv = _full_convolution(cur_a)
            max_b = conv.max()
            s = cur_a.sum()
            cur_val_est = 2.0 * n * max_b / (s ** 2)
            if cur_val_est < best_val:
                best_val = cur_val_est
                best_a = cur_a.copy()
                print(f"[targeted reorder {it}] improved ≈ {best_val:.6f}")

    # final official evaluation of the best vector discovered in this stage
    final_val = evaluate_sequence(best_a.tolist())
    return best_a, final_val
