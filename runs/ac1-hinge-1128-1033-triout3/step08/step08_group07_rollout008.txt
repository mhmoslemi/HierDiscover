"""
Hybrid optimizer for the step‑function minimisation problem.

The algorithm proceeds in four phases:

1. **Smooth surrogate optimisation** – minimise a log‑sum‑exp
   approximation of the max‑convolution using L‑BFGS.  The surrogate is
   2·n·max(c) + 2·n·ε·log‑sum‑exp((c−max)/ε) where c = a∗a.
   Multiple restarts with a decreasing ε‑schedule are performed.
   The current best probability vector is kept in the simplex
   (soft‑max parametrisation).

2. **Adam on the exact max‑convolution** – sub‑gradient descent on the
   true objective (the active peak of the convolution) while staying
   on the probability simplex.

3. **Permutation optimisation** – the multiset of values obtained so far
   is reordered to minimise the peak of the convolution.
   First a greedy “best‑rotation” among all cyclic shifts of the
   descendingly sorted vector is tried, then a short simulated‑annealing
   on swaps (and tiny mass‑moves) refines the ordering.

4. **Cheap hill–climbing** – final local search that accepts any
   strictly improving swap/mass‑move.

All convolution work uses an FFT‑based O(n log n) routine.
The code respects the 1000 s wall‑time budget and returns the best
sequence found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter for the L‑BFGS phase
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # small negative rounding errors are clipped
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast copy of `evaluate_sequence` for vectors whose entries sum to 1.
    Returns 2·n·max(conv) because sum(a)=1.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    # sum_a is 1 by construction – the denominator disappears.
    return float(2.0 * n * max_b)


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    w_shift = w - np.max(w)
    ex = np.exp(w_shift)
    return ex / np.sum(ex)


# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and conv = a∗a with a = softmax(w).  The gradient is derived
    analytically and returned w.r.t. the soft‑max parameters.
    """
    a = _softmax(w)                     # ℓ₁‑normalised non‑negative vector

    conv = _fft_convolve_self(a)        # length 2·n‑1
    max_c = conv.max()

    # smooth approximation of the max
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)        # ∑exp((c−max)/ε)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (see derivation in the original description)
    weights = np.exp(shifted)
    p = weights / np.sum(weights)       # softmax over the convolution entries

    # convolution a * reverse(p) gives the needed weighted sum
    p_rev = p[::-1]
    size = _next_pow2(conv.shape[0] + a.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[:a.shape[0] + p_rev.shape[0] - 1]

    # central n entries correspond to indices [n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]      # length n
    grad_a = 4.0 * n * seg[::-1]        # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by `scipy.optimize.minimize`."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Permutation optimisation helpers
# ----------------------------------------------------------------------
def _best_cyclic_rotation(a_desc: np.ndarray, cur_val: float,
                          deadline: float) -> (np.ndarray, float):
    """
    Scan all n cyclic rotations of the descendingly sorted vector and
    keep the one with the smallest objective value.
    """
    n = a_desc.shape[0]
    best_seq = a_desc.copy()
    best_val = cur_val
    # we reuse the same array to avoid excessive allocations
    for shift in range(n):
        if time.time() >= deadline:
            break
        cand = np.roll(a_desc, shift)
        val = _eval_fast(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    return best_seq, best_val


def _simulated_annealing_permutation(a: np.ndarray,
                                    cur_val: float,
                                    deadline: float) -> (np.ndarray, float):
    """
    Short SA that swaps positions (and occasionally moves a tiny amount of
    mass).  It starts from the current permutation and tries to improve the
    maximal convolution peak.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    cur_seq = a.copy()
    cur_val = cur_val

    # temperature is a small fraction of the current value
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99992

    while time.time() < deadline:
        # select a move type
        if rng.random() < 0.7:                    # swap move (dominant)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        else:                                      # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_seq[i] <= 1e-14:
                continue
            delta = cur_seq[i] * 0.03 * rng.random()
            cur_seq[i] -= delta
            cur_seq[j] += delta

        new_val = _eval_fast(cur_seq)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[perm‑SA] new best = {best_val:.6f}")
        else:
            # revert move
            if rng.random() < 0.7:   # we performed a swap
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            else:
                cur_seq[i] += delta
                cur_seq[j] -= delta

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected Adam on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Adam optimiser on the true max‑convolution (sub‑gradient of the
    active peak) while keeping the vector on the probability simplex.
    """
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)          # initialise soft‑max parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a0.copy()
    best_val = _eval_fast(best_a)

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        a = _softmax(w)

        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # gradient of the active max entry w.r.t. a
        grad_a = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_a[mask] = 2.0 * a[idx[mask]]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = 0.5 / (np.sqrt(t) + 1.0)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _softmax(w)
        cur_val = _eval_fast(a_new)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[Adam] iter {t} new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Local search that accepts any strictly improving swap or mass move."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-14:
                continue
            delta = cur_a[i] * 0.02 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
