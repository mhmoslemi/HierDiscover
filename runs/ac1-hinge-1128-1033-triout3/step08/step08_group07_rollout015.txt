    n = len(f)
    best_f = f.copy()
    best_val = _eval_fast(best_f)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute conv and M
        conv = _fft_convolve_self(best_f)
        M = conv.max()
        # build constraint matrix
        A = _build_conv_matrix(best_f)   # shape (2n-1, n)
        # RHS vector
        b_ub = np.full(2*n-1, M * (1 + 1e-12))  # small slack
        # linear program
        c = -np.ones(n)  # maximize sum(b)
        # bounds
        bounds = [(0, None)] * n
        # Solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP] iteration {it} linprog failed: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b * (best_f.sum() / sum_b)   # scaling to same L1
        # Mix
        f_new = (1 - t) * best_f + t * g
        f_new = np.maximum(f_new, 0)

        if cur_val < best_val:
            best_val = cur_val

            print(f"[LP] iter {it} improved to {best_val:.6f}")
        else:
            # If not improved, perhaps reduce mixing factor t
            t *= 0.5   # reduce step size
            # If t too small break
            if t < 1e-4:
                break
    return best_f, best_val

"""
Hybrid optimizer for the self‑convolution minimisation problem.

The optimiser proceeds in four stages:

1. **Smooth surrogate L‑BFGS** – minimise a log‑sum‑exp surrogate of the
   max‑convolution.  Soft‑max parametrisation guarantees non‑negativity
   and ℓ₁‑normalisation.  Several restarts with a decreasing ε schedule
   are performed; after each restart the vector is re‑ordered with the
   “ends‑large” heuristic.

2. **Projected Adam** – sub‑gradient descent on the true max‑convolution
   (the active peak) while staying on the probability simplex.

3. **Alternating linear‑programming refinement** – for the current f,
   solve a LP that maximises the total mass of a vector g under the
   constraints (f ∗ g)≤M, where M is the current max‑convolution value.
   The feasible g is rescaled to the same ℓ₁‑norm as f and mixed with f.
   Repeating this step reduces the maximum convolution further.

4. **Final hill‑climbing** – cheap random swaps and tiny mass‑transfer
   moves that accept any strictly improving change.

All heavy work (FFT‑based convolutions, LP construction) is performed
with NumPy / SciPy and stays well within the 1000 s wall‑time budget.
The returned list is a feasible non‑negative step function whose
evaluation by ``evaluate_sequence`` is (hopefully) below the target
1.503.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter for the L‑BFGS phase
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)            # clip tiny negatives from rounding
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Exact copy of ``evaluate_sequence`` for a normalised vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    w_shift = w - np.max(w)
    ex = np.exp(w_shift)
    return ex / np.sum(ex)


# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and conv = a∗a with a = softmax(w).  The gradient w.r.t. the
    soft‑max parameters is returned analytically.
    """
    a = _softmax(w)                     # ℓ₁‑normalised non‑negative vector

    conv = _fft_convolve_self(a)        # length 2·n‑1
    max_c = conv.max()

    # smooth approximation of the max
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)        # ∑exp((c−max)/ε)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / np.sum(weights)       # softmax over the convolution entries

    # convolution a * reverse(p) gives the needed weighted sum
    p_rev = p[::-1]
    size = _next_pow2(conv.shape[0] + a.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[:a.shape[0] + p_rev.shape[0] - 1]

    # central n entries correspond to indices [n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]      # length n
    grad_a = 4.0 * n * seg[::-1]        # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Simple heuristic: put the largest values at the two ends, then alternate
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Deterministic reordering that places large values at the extremes."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


# ----------------------------------------------------------------------
# Simulated annealing on permutations (plus tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing_mass(a: np.ndarray,
                              cur_val: float,
                              deadline: float) -> (np.ndarray, float):
    """
    Fast SA that alternates between swapping two positions and moving a
    tiny amount of mass from one index to another.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99993
    it = 0
    while time.time() < deadline:
        if rng.random() < 0.3:                     # pure swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]                # propose swap
        else:                                      # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-12:
                continue
            delta = a[i] * 0.03 * rng.random()     # move ≤3 % of source mass
            a[i] -= delta
            a[j] += delta

        new_val = _eval_fast(a)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            # revert move
            if rng.random() < 0.3:  # the move we performed was a swap
                a[i], a[j] = a[j], a[i]
            else:
                a[i] += delta
                a[j] -= delta

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected Adam on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Adam optimiser on the true max‑convolution while staying on the
    probability simplex.
    """
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)          # initialise soft‑max parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a0.copy()
    best_val = _eval_fast(best_a)

    while time.time() < deadline:
        t += 1
        a = _softmax(w)

        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active max entry w.r.t. a
        grad_a = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_a[mask] = 2.0 * a[idx[mask]]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update on w
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = 0.5 / (np.sqrt(t) + 1.0)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _softmax(w)
        cur_val = _eval_fast(a_new)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[Adam] iter {t} new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Very fast local search that accepts any strictly improving move."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-12:
                continue
            delta = cur_a[i] * 0.02 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Linear‑programming based alternating refinement
# ----------------------------------------------------------------------
def _build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A that represents the linear constraints
    (f * b)_k ≤ M, i.e. A @ b = f * b.
    A has shape (2·n‑1, n) with entries A[k, j] = f[k‑j] when the index
    is within bounds, otherwise 0.
    """
    n = f.shape[0]
    # Each diagonal offset d (0 ≤ d < n) carries the constant f[d].
    # For offset d the diagonal length is exactly n.
    data = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _lp_refine(f: np.ndarray,
               deadline: float,
               max_iter: int = 12,
               mix_factor: float = 0.4) -> (np.ndarray, float):
    """
    Alternating refinement using a linear programme.
    For the current vector f, maximise ∑b_j under f * b ≤ M (M = max(f*f)).
    The obtained b is rescaled to the same ℓ₁‑norm as f and mixed back.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_val = _eval_fast(best_f)

    t = mix_factor
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        conv = _fft_convolve_self(best_f)
        M = conv.max()
        # Build constraint matrix A (size (2n‑1) × n)
        A = _build_convolution_matrix(best_f)

        # RHS vector – a tiny slack avoids numerical infeasibility
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12))

        c = -np.ones(n)                     # maximise ∑b  → minimise -∑b
        bounds = [(0, None)] * n

        # solve the LP
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP] iteration {it} linprog failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # Rescale to keep the ℓ₁‑norm unchanged
        g = b * (best_f.sum() / sum_b)

        # Mix and renormalise
        new_f = (1.0 - t) * best_f + t * g
        new_f = np.maximum(new_f, 0.0)
        new_f /= new_f.sum()

        cur_val = _eval_fast(new_f)
        if cur_val < best_val:
            best_val = cur_val
            best_f = new_f
            print(f"[LP] iter {it} improved to {best_val:.6f}")
        else:
            # no improvement – halve the mixing factor and try again
            t *= 0.5
            if t < 1e-4:
                break

    return best_f, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the final official evaluation
    deadline = start_time + budget_s - 1.5
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # ------------------------------------------------------------------
    # Problem size – 4000 works well with the FFT code and gives a
    # reasonable trade‑off between solution quality and runtime.
    # ------------------------------------------------------------------
    n = 4000
    global _GLOBAL_N
    _GLOBAL_N = n

    # ------------------------------------------------------------------
    # Initialise – reuse a known construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to requested size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        a0 += 5e-4 * np.random.rand(n)
        print("[init] using fresh uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= np.sum(a0)          # ℓ₁‑normalise
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate L‑BFGS with several restarts
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-13), num=30)

    restart = 0
    # reserve ~400 s for later phases
    while time.time() < deadline - 400.0:
        restart += 1
        print(f"[phase1 restart {restart}]")
        w = np.log(best_a + 1e-12) + 0.5 * np.random.randn(n)

        for eps in eps_schedule:
            if time.time() >= deadline - 380.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            # L‑BFGS on the smooth surrogate
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            a_candidate = _softmax(w)
            cur_val = _eval_fast(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[phase1 eps={eps:.1e}] new best = {best_val:.6f}")

        # ends‑large re‑ordering after each restart
        a_ord = _ends_large_reorder(best_a)
        a_ord /= np.sum(a_ord)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected Adam descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 250.0:
        print("[phase2] Adam gradient descent")
        best_a, best_val = _adam_opt(best_a, deadline - 250.0)

    # ------------------------------------------------------------------
    # Phase 3 – alternating linear‑programming refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        print("[phase3] LP‑based alternating refinement")
        best_a, best_val = _lp_refine(best_a,
                                      deadline - 100.0,
                                      max_iter=12,
                                      mix_factor=0.4)

    # ------------------------------------------------------------------
    # Phase 4 – final cheap hill‑climbing (swap + mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[phase4] Final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a,
                                              best_val,
                                              deadline)

    # ------------------------------------------------------------------
    # Official evaluation (uses the provided security‑checked function)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    return best_a.tolist()
