def _alternating_ends_reorder(a):
    vals = np.sort(a)[::-1] (descending)
    n = len(a)
    new_a = np.empty_like(a)
    left = 0
    right = n-1
    i = 0
    while i < n:
        new_a[left] = vals[i]
        left += 1
        i += 1
        if i >= n: break
        new_a[right] = vals[i]
        right -= 1
        i += 1
    return new_a

"""
Hybrid optimizer with additional pattern search and LP‑mixing.
The algorithm consists of:

1.  A cheap grid‑search over a two‑level “high‑ends / low‑middle” family
    (O(n log n) per candidate).  This often discovers a very good
    starting shape.

2.  A series of L‑BFGS runs on a smooth log‑sum‑exp approximation of the
    max‑convolution, with several restarts and a decreasing smoothing
    schedule.  After each restart two deterministic re‑orderings
    (large‑values at the ends and alternating ends) are tried.

3.  A short linear‑programming mixing step.  For the current sequence *f*
    we solve a tiny LP that maximises the ℓ₁‑norm of a new vector *g*
    while keeping the convolution (f*g) below the current peak.
    A convex combination (1‑t) f + t g (renormalised) is accepted when it
    lowers the objective.

4.  Projected Adam optimisation on the exact max‑convolution (sub‑gradient).

5.  Long simulated‑annealing on swaps and tiny mass transfers.

6.  Very cheap hill‑climbing refinement.

All heavy work (convolutions) uses FFT‑based O(n log n) code.
The routine respects a 1000 s wall‑time budget; when time runs out the
best sequence found so far is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
from scipy import sparse

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for L‑BFGS phase
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)            # clip tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast copy of evaluate_sequence (assumes a is already ℓ₁‑normalised)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:                     # should never happen
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    w_shift = w - np.max(w)
    ex = np.exp(w_shift)
    return ex / np.sum(ex)


# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """Return smooth objective L (log‑sum‑exp surrogate) and its gradient
    w.r.t. the soft‑max parameters w."""
    a = _softmax(w)                     # ℓ₁‑normalised non‑negative

    conv = _fft_convolve_self(a)        # length 2·n‑1
    max_c = conv.max()

    # smooth approximation of the max
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)        # ∑exp((c−max)/ε)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / np.sum(weights)       # soft‑max over the convolution entries

    # convolution a * reverse(p) gives the needed weighted sum
    p_rev = p[::-1]
    size = _next_pow2(conv.shape[0] + a.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[:a.shape[0] + p_rev.shape[0] - 1]

    # central n entries correspond to indices [n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]      # length n
    grad_a = 4.0 * n * seg[::-1]        # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Heuristics for re‑ordering a vector of heights
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values at the two ends, then alternate small‑large."""
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    i = 0
    while i < n:
        new_a[i] = sorted_vals[right]
        right -= 1
        i += 1
        if i >= n:
            break
        new_a[i] = sorted_vals[left]
        left += 1
        i += 1
    return new_a


def _alternating_ends_reorder(a: np.ndarray) -> np.ndarray:
    """Largest values are placed alternately at the leftmost and rightmost
    positions, moving inward."""
    n = a.shape[0]
    vals = np.sort(a)[::-1]            # descending
    new_a = np.empty_like(a)

    left = 0
    right = n - 1
    idx = 0
    while idx < n:
        new_a[left] = vals[idx]
        left += 1
        idx += 1
        if idx >= n:
            break
        new_a[right] = vals[idx]
        right -= 1
        idx += 1
    return new_a


# ----------------------------------------------------------------------
# Simple two‑level grid search (high at the ends, low in the middle)
# ----------------------------------------------------------------------
def _grid_search_two_level(n: int, time_deadline: float):
    """Search over blocks of “high” values at the two ends and a uniform
    low value in the middle.  Returns the best sequence encountered and its
    evaluation value."""
    best_seq = None
    best_val = float('inf')

    # uniform baseline (L = 0)
    a_uniform = np.full(n, 1.0 / n)
    val = _eval_fast(a_uniform)
    best_seq = a_uniform
    best_val = val

    max_L = n // 2
    step_L = max(5, max_L // 200)          # about 200 different L‑values
    for L in range(step_L, max_L + 1, step_L):
        if time.time() >= time_deadline:
            break

        # admissible range for the high value t
        t_high_max = 1.0 / (2 * L)          # puts all mass in the end blocks
        t_high_min = 1.0 / n                # cannot go below uniform height
        # sample a few points on this interval (log‑linear spacing)
        t_samples = np.geomspace(t_high_max, t_high_min, num=7)
        for t in t_samples:
            if time.time() >= time_deadline:
                break
            # low (middle) value derived from the normalisation constraint
            s = (1.0 - 2 * L * t) / (n - 2 * L)
            if s < 0:          # infeasible region, skip
                continue
            a = np.empty(n, dtype=float)
            a[:L] = t
            a[-L:] = t
            if n - 2 * L > 0:
                a[L:n - L] = s
            # enforce non‑negativity and normalise (tiny numerical drift)
            a = np.maximum(a, 0.0)
            a /= np.sum(a)

            val = _eval_fast(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[grid] L={L} t={t:.3e} -> new best {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming mixing step
# ----------------------------------------------------------------------
def _lp_mix_step(a: np.ndarray, cur_val: float,
                 deadline: float, mix_coeff: float = 0.2) -> (np.ndarray, float):
    """
    Solve a tiny LP that maximises the ℓ₁‑norm of a new vector g
    under the constraints (a * g)_i ≤ M for a few active convolution
    indices (where M = max(a*a)).  The resulting g is convex‑combined
    with a (with weight `mix_coeff`) and renormalised.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    M = np.max(conv)

    # pick a handful of indices where the max is attained (or close)
    K = min(12, len(conv))
    top_idx = np.argsort(-conv)[:K]

    # Build sparse matrix A (shape K × n)
    rows = []
    cols = []
    data = []
    for row, sh in enumerate(top_idx):
        # for each variable j we need coefficient a[sh - j] when 0 ≤ sh-j < n
        idx = sh - np.arange(n)          # vector of i indices
        mask = (idx >= 0) & (idx < n)
        cols_row = np.where(mask)[0]
        rows.extend([row] * len(cols_row))
        cols.extend(cols_row.tolist())
        data.extend(a[idx[mask]].tolist())
    A = sparse.csr_matrix((data, (rows, cols)), shape=(K, n))

    # Linear program: maximise sum(g)  ⇔ minimise -sum(g)
    c = -np.ones(n)                      # minimise -∑g
    bounds = [(0.0, 1.0) for _ in range(n)]

    # Solve with HiGHS (fast for sparse problems)
    res = linprog(c, A_ub=A, b_ub=np.full(K, M),
                  bounds=bounds, method='highs', options={'presolve': True})

    if not res.success:
        return a, cur_val   # fall back to the original sequence

    g = res.x
    # Convex combination and renormalisation
    new_a = (1.0 - mix_coeff) * a + mix_coeff * g
    new_a = np.maximum(new_a, 0.0)
    new_a /= np.sum(new_a)

    new_val = _eval_fast(new_a)
    if new_val < cur_val:
        print(f"[LP] mix improved {cur_val:.6f} → {new_val:.6f}")
        return new_a, new_val
    return a, cur_val


# ----------------------------------------------------------------------
# Projected Adam on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """Adam optimiser on the true max‑convolution (sub‑gradient of the peak)
    while keeping the vector on the probability simplex."""
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)          # initialise soft‑max parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0
    best_a = a0.copy()
    best_val = _eval_fast(best_a)

    while time.time() < deadline:
        t_iter += 1
        a = _softmax(w)

        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active max entry w.r.t. a
        grad_a = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_a[mask] = 2.0 * a[idx[mask]]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update on w
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)

        lr = 0.5 / (np.sqrt(t_iter) + 1.0)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        cur_val = _eval_fast(_softmax(w))
        if cur_val < best_val:
            best_val = cur_val
            best_a = _softmax(w).copy()
            print(f"[Adam] iter {t_iter} new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing on permutations + tiny mass transfers
# ----------------------------------------------------------------------
def _simulated_annealing_mass(a: np.ndarray,
                              cur_val: float,
                              deadline: float) -> (np.ndarray, float):
    """Fast SA that alternates between swapping two positions and moving a
    tiny amount of mass from one index to another."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99993
    it = 0
    while time.time() < deadline:
        if rng.random() < 0.3:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
        else:                                      # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-12:
                continue
            delta = a[i] * 0.03 * rng.random()
            a[i] -= delta
            a[j] += delta

        new_val = _eval_fast(a)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            # revert move
            if rng.random() < 0.3:                 # swap was performed
                a[i], a[j] = a[j], a[i]
            else:
                a[i] += delta
                a[j] -= delta

        T = max(T * decay, 1e-12)
        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Local search that accepts any strictly improving move."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-12:
                continue
            delta = cur_a[i] * 0.02 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
