        else:
            f = f * (target_sum / s)
    else:
        f = np.full(n, target_sum / n)

    best_f = f.copy()
    best_score = 2.0 * n * _full_convolution(best_f).max() / (target_sum ** 2)
    print(f"[init] start score = {best_score:.6f}")

    # ---- 2) Main LP‑driven refinement loop ----------------------------
    it = 0
    while time.time() < deadline - 30.0:          # keep time for final hill‑climb
        conv = _full_convolution(best_f)
        M = conv.max()
        cur_score = 2.0 * n * M / (target_sum ** 2)
        print(f"[iter {it}]  M = {M:.6f},  score = {cur_score:.6f}")

        # Solve the LP that tries to increase the total mass while respecting M
        try:
            g = _lp_step(best_f, M)
        except RuntimeError as e:
            print("[LP] failed, exiting loop:", e)
            break

        sum_g = g.sum()
        if sum_g <= 0:
            print("[LP] obtained zero‑mass solution, aborting")
            break

        # Scale back to the required ℓ₁‑norm
        g = g * (target_sum / sum_g)

        # Mixing step – decreasing convex combination
        t = 0.4 * (0.9 ** (it // 10))
        t = max(min(t, 0.5), 0.02)

        new_f = (1.0 - t) * best_f + t * g
        # Re‑project onto the simplex (just in case of tiny rounding errors)
        new_f = np.maximum(new_f, 0.0)
        new_f = new_f / new_f.sum() * target_sum

        new_score = 2.0 * n * _full_convolution(new_f).max() / (target_sum ** 2)
        print(f"[iter {it}]  new score = {new_score:.6f}")

        # Keep the better of the two candidates
        if new_score < best_score - 1e-12:
            best_f = new_f
            best_score = new_score
            print(f"[iter {it}]  *** improvement! best = {best_score:.6f}")

        # Occasionally check whether the raw LP solution (without mixing) is already better
        lp_score = 2.0 * n * _full_convolution(g).max() / (target_sum ** 2)
        if lp_score < best_score - 1e-12:
            best_f = g
            best_score = lp_score
            print(f"[iter {it}]  *** LP alone better! best = {best_score:.6f}")

        it += 1

    # ---- 3) Final cheap hill‑climbing refinement ----------------------
    if time.time() < deadline:
        best_f, best_score = _refine_hill_climb(best_f, best_score, deadline)

    # ---- 4) Final safety check with the official evaluator -------------
    final_score = evaluate_sequence(best_f.tolist())
    print(f"[final] evaluate_sequence reports {final_score:.6f}")

    return best_f.tolist()
