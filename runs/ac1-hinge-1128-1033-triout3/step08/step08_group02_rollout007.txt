"""
Step‑function optimizer for `evaluate_sequence`.

Goal
----
Minimise               f(a) = 2·n·max(conv(a,a)) / (∑a)²
under the non‑negativity and sum‑to‑1 constraints.
The objective is scale‑invariant, therefore we keep Σa = 1
throughout the optimisation.

Algorithm (≈ 1000 s budget)
---------------------------
1. **Initial families** – a quick sweep over several parametric families
   (uniform, power‑law, exponentials, sine‑shapes and random Dirichlet draws)
   to obtain a solid starting point.

2. **Smooth mirror‑descent** – minimise a smooth approximation of the
   “max‑convolution”:
        ϕβ(a) = (1/β)·log(∑ₖ e^{β·bₖ}) ,   b = a∗a .
   The gradient of ϕβ is
        g = 2·conv_rev(a_rev , w)   with   wₖ = softmax(β·bₖ) .
   We perform an exponentiated‑gradient (mirror‑descent) update
        a ← a·exp(‑η·g) ,   a ← a/∑a .
   β is increased from a small to a large value (gradual “hardening” of the
   max) while η decays, yielding a stable descent that avoids the
   plateaus observed with a pure sub‑gradient method.

3. **Deterministic greedy hill‑climb** – repeatedly move a fraction of the
   mass from the index that contributes most to the current maximal
   convolution entry to the index that contributes least, accepting only
   improving moves.  The transfer fraction anneals from 50 % → 0.1 %.

4. **Random fine‑tuning** – occasional stochastic mass transfers
   (simulated‑annealing style) are attempted; only moves that lower the
   exact objective are kept.  This helps to escape tiny local minima.

5. **Uniform mixing** – finally test tiny convex combinations with the
   uniform vector; mixing often reduces the peak a little further.

All phases respect the time budget, keep the vector non‑negative,
renormalise after every modification, and retain the best vector found.
The final candidate is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    lx = x.shape[0]
    ly = y.shape[0]
    out_len = lx + ly - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Simple parametric families – fast initial sweep
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent start vector and its objective."""
    best_a = np.full(n, 1.0 / n)                # uniform
    best_val = _objective(best_a, n)

    def try_update(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(123456)

    idx = np.arange(n, dtype=float)

    # deterministic families
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing linear
    try_update(np.maximum(idx + 1, n - idx))                        # V‑shape (large at ends)
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # quadratic decreasing
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law V‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(1.0, n + 1.0)
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # random Dirichlet draws (quick, a few dozen)
    for _ in range(20):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – smooth mirror‑descent (soft‑max of the convolution)
# ----------------------------------------------------------------------
def _smooth_expgrad_phase(a0: np.ndarray, n: int,
                          start: float, deadline: float,
                          best_a: np.ndarray, best_val: float,
                          target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a smooth approximation of max(conv(a,a)).
    β grows from β_start → β_end; η decays from η_start → η_end.
    """
    cur = a0.copy()
    no_improve = 0
    IMPATIENT = 300_000

    beta_start, beta_end = 0.1, 500.0          # schedule for soft‑max hardness
    eta_start, eta_end   = 0.10, 0.001        # schedule for step size

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective (exact, for bookkeeping)
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SmoothExpGrad] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # progress (0 … 1)
        progress = (now - start) / (deadline - start)

        # schedules
        beta = beta_start * (beta_end / beta_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        # ---- soft‑max weights (stable) ----
        shift = b.max()               # subtract max to avoid overflow
        w_unnorm = np.exp(beta * (b - shift))
        w = w_unnorm / w_unnorm.sum()   # soft‑max vector, length 2·n‑1

        # ---- gradient of the smooth max ----
        a_rev = cur[::-1]
        conv_rev = _fft_convolve(a_rev, w)            # length 3·n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]        # slice to length n

        # ---- exponentiated‑gradient update (mirror‑descent) ----
        # guard against extreme exponent arguments
        upd = np.exp(-eta * np.clip(grad, -50.0, 50.0))
        cur = cur * upd

        # renormalise to Σa = 1
        s = cur.sum()
        if s == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= s

        # occasional jitter if we are stuck
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.005, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print("[SmoothExpGrad] jitter after stagnation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – deterministic greedy hill‑climbing (mass transfer)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Transfer a fraction of the mass from the index that contributes most
    to the current peak to the index that contributes least.  The transfer
    fraction anneals from 0.5 → 0.001.
    """
    cur = a_start.copy()
    no_improve = 0
    IMPATIENT = 400_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # annealed transfer fraction
        progress = (now - start) / (deadline - start)
        frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the maximal convolution entry
        k = int(np.argmax(b))

        # reverse‑relevant slice of a (a_{k‑i})
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        rev = np.zeros(n, dtype=np.float64)
        idx = np.arange(lo, hi + 1, dtype=int)
        rev[idx] = cur[k - idx]

        # source = index with largest contribution to the peak
        i = int(np.argmax(rev))

        # target = index with smallest contribution (skip source)
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            # source depleted – occasional jitter
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.004, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print("[HillClimb] jitter after stagnation")
            continue

        # amount to move (random fraction of the allowed maximum)
        delta = cur[i] * frac * np.random.rand()
        if delta <= 0.0:
            continue

        # trial move
        cur[i] -= delta
        cur[j] += delta

        # evaluate the trial
        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # keep the improvement
            pass
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – random fine‑tuning (stochastic hill‑climb)
# ----------------------------------------------------------------------
def _random_fine_tune(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Randomly pick two indices and move a tiny fraction of mass.
    Accept only improving moves; the transferred fraction decays.
    """
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000
    while True:
        now = time.time()
        if now >= deadline:
            break

        # exact evaluation of current vector
        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[RandomFine] it {it:07d} new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # decreasing transfer fraction
        progress = (now - start) / (deadline - start)
        frac = 0.2 * (1.0 - progress) + 0.001 * progress   # 0.2 → 0.001

        # pick distinct source / target
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        if cur[i] <= 1e-15:
            continue

        delta = cur[i] * frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # keep improvement
            pass
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        it += 1

        if no_improve > IMPATIENT:
            # occasional shake to escape tiny plateaus
            noise = np.random.normal(scale=0.002, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print("[RandomFine] jitter after stagnation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 4 – tiny uniform mixing
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test convex combinations a′ = (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.002, 0.005, 0.010, 0.015, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function within the given time budget.
    Returns the best list of floats found (normalised to sum = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # Problem size – a few‑thousand entries give a good trade‑off.
    # 8000 was used previously; 12000 gives a bit more flexibility
    # while staying comfortably fast.
    n = 12000

    # ------------------------------------------------------------------
    # Initialise (reuse previous construction if provided)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] objective = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase‑1 – smooth mirror‑descent (soft‑max max‑convolution)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * (deadline - start_time)
    best_a, best_val = _smooth_expgrad_phase(
        best_a, n,
        start=start_time,
        deadline=phase1_end,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )
    print(f"[Phase‑1] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 – deterministic greedy hill‑climb
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        start=phase1_end,
        deadline=phase2_end,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )
    print(f"[Phase‑2] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3 – random fine‑tuning (stochastic local search)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        best_a, best_val = _random_fine_tune(
            best_a, n,
            start=phase2_end,
            deadline=deadline,
            best_a=best_a,
            best_val=best_val,
            target_val=TARGET
        )
        print(f"[Phase‑3] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase‑4 – tiny uniform mixing (often yields a final drop)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[FinalMix] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final verification through the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
