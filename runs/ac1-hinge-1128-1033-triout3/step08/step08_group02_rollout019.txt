def generate_candidate(n, p, alpha):
    # Compute logits for a_i: log a_i = p * log(i+1) - alpha * i / n
    i = np.arange(n, dtype=np.float64)
    log_ai = p * np.log(i + 1.0) - alpha * i / n
    # subtract max for numeric stability
    log_ai -= np.max(log_ai)
    a = np.exp(log_ai)
    a /= a.sum()
    return a
```

def _gen_exp_power(n, p, alpha):
    i = np.arange(n, dtype=np.float64)
    log_ai = p * np.log(i+1.0) - alpha * i / n
    # subtract max for stability
    log_ai -= np.max(log_ai)
    a = np.exp(log_ai)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= total
    return a

# ----------------------------------------------------------------------
# Fast convolution (FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by ``evaluate_sequence`` for a normalized vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Simple deterministic families – cheap initial sweep
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent start vector and its objective."""
    best_a = np.full(n, 1.0 / n)                # uniform
    best_val = _objective(best_a, n)

    def try_update(vec):
        nonlocal best_a, best_val
        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.shape[0] != n:
            return
        if np.any(vec < 0):
            return
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        val = _objective(vec, n)
        if val < best_val:
            best_val, best_a = val, vec.copy()

    rng = np.random.default_rng(12345)

    # deterministic families (same as previous code)
    idx = np.arange(n)
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # U‑shape linear
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials (already explored before)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # random Dirichlet draws (quick)
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric families used in the random‑search phase
# ----------------------------------------------------------------------
def _gen_exp(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i  (r in (0,1))."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_rev_exp(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**(n‑1‑i)  (decay from the right)."""
    a = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_mix_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    """Mixture w·r1**i + (1‑w)·r2**(n‑1‑i)."""
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, n - 1 - i)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)**p."""
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power_exp(n: int, p: float, r: float) -> np.ndarray:
    """a_i ∝ (i+1)**p · r**i."""
    i = np.arange(n, dtype=np.float64)
    log_a = p * np.log(i + 1.0) - (np.log(r) * i)   # safe logarithmic form
    # stabilise the exponentials
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Phase A – exponentiated‑gradient descent (same as before)
# ----------------------------------------------------------------------
def _expgrad_phase(a0: np.ndarray, n: int,
                   start: float, deadline: float,
                   best_a: np.ndarray, best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        k = int(np.argmax(b))
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]               # a_{k‑i}
        g = 2.0 * rev                           # sub‑gradient

        progress = (now - start) / (deadline - start)
        eta = (1.0 - progress) * 0.12 + progress * 0.0005

        cur = cur * np.exp(-eta * g)
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.008, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – deterministic greedy hill‑climbing (same as before)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k = int(np.argmax(b))
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        i = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            cur_val = new_val
        else:
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – final uniform mixing
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

    best_a = a0.copy()

    print(f"[Init] objective = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase‑0 – random exploration of rich parametric families
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 999)
    random_deadline = start_time + 0.25 * (deadline - start_time)

    while time.time() < random_deadline:
        choice = rng.integers(0, 5)   # five families
        if choice == 0:                     # pure exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_exp(n, r)
        elif choice == 1:                   # reversed exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_rev_exp(n, r)
        elif choice == 2:                   # mixture of two exponentials
            r1 = rng.uniform(0.90, 0.99999)
            r2 = rng.uniform(0.90, 0.99999)
            w = rng.random()
            cand = _gen_mix_exp(n, r1, r2, w)
        elif choice == 3:                   # pure power‑law
            p = rng.uniform(-5.0, 5.0)
            cand = _gen_power(n, p)
        else:                               # power‑law × exponential
            p = rng.uniform(-5.0, 5.0)
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_power_exp(n, p, r)

        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Random] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase‑A – exponentiated‑gradient descent (exact max‑conv)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.55 * (deadline - start_time)
    best_a, best_val = _expgrad_phase(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase‑B – deterministic greedy hill‑climbing
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.90 * (deadline - start_time)
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase‑C – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification with the provided security wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:                                   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
