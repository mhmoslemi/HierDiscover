def grad_descent(a, n, deadline):

    size1 = ... FFT for conv_a
    size2 = next pow2(3*n-2)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5   # maybe high
    tau0 = 0.5
    tau_max = 100.0
    it = 0
    best_val = objective(a)
    best_a = a.copy()
    while time < deadline:
        ft_a = np.fft.rfft(a, size1)
        b = np.fft.irfft(ft_a * ft_a, size1)[:2*n-1]
        maxb = b.max()
        cur_val = 2.0 * n * maxb
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
        # update tau schedule
        progress = (time.time() - start) / (deadline - start)
        tau = tau0 + (tau_max - tau0) * progress
        # compute probabilities
        maxb = b.max()
        # shift for numerical stability
        shifted = b - maxb
        exps = np.exp(tau * shifted)
        sumexps = exps.sum()
        p = exps / sumexps
        # compute gradient
        rev_a = a[::-1]
        ft_p = np.fft.rfft(p, size2)
        ft_rev = np.fft.rfft(rev_a, size2)
        cross = np.fft.irfft(ft_p * ft_rev, size2)[:3*n-2]
        g = 2.0 * cross[n-1:n-1+n]
        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 * (1 - progress)  # decaying
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # clip negative to zero
        a = np.maximum(a, 0.0)
        # renormalize
        total = a.sum()
        if total == 0:
            a = np.full(n, 1.0/n)
        else:
            a /= total
    return best_a, best_val

import necessary modules

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, size):
    ft = np.fft.rfft(a, size)
    conv = np.fft.irfft(ft * ft, size)[:2*a.shape[0]-1]
    return conv

def _softmax_weights(b, tau):
    b_max = b.max()
    shifted = b - b_max
    exp_vals = np.exp(tau * shifted)
    sumexp = exp_vals.sum()
    if sumexp == 0.0:  # tau huge and all underflow? fallback
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
        return p
    return exp_vals / sumexp

def _gradient(p, a, size):
    rev = a[::-1]
    ft_p = np.fft.rfft(p, size)
    ft_rev = np.fft.rfft(rev, size)
    conv = np.fft.irfft(ft_p * ft_rev, size)[:len(p) + len(rev) - 1]
    n = a.shape[0]
    g = 2.0 * conv[n-1:n-1+n]
    return g

def _objective_given_a(a):
    # assumes sum a = 1
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, size)
    maxb = conv.max()
    return 2.0 * n * maxb

def _parametric_sweep(n):
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_given_a(best_a)
    # define inner update function similar to previous but using objective function.
    def try_candidate(arr):
        nonlocal best_a, best_val
        arr = np.asarray(arr, dtype=np.float64)
        if arr.ndim != 1 or arr.shape[0] != n:
            return
        if np.any(arr < 0):
            return
        s = arr.sum()
        if s <= 0:
            return
        arr = arr / s
        val = _objective_given_a(arr)
        if val < best_val:
            best_val, best_a = val, arr.copy()
    # families...
    idx = np.arange(n, dtype=float)
    # Uniform already set
    try_candidate(np.arange(n, 0, -1, dtype=float))
    try_candidate(np.maximum(idx + 1, n - idx))
    # Power law and exponentials
    for p in np.linspace(0.2, 4.0, 20):
        i = np.arange(1, n + 1, dtype=float)
        a = i**(-p) + (n - i + 1)**(-p)
        try_candidate(a)
    for r in np.linspace(0.95, 0.9999, 30):
        i = np.arange(n, dtype=float)
        a = r**i + r**(n-1-i)
        try_candidate(a)
    # arcsine-like shape
    i = np.arange(0, n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(i * (n - i + 0.5))
    try_candidate(a)
    # random Dirichlet draws
    rng = np.random.default_rng(123456)
    for _ in range(20):
        try_candidate(rng.dirichlet(np.ones(n)))
    return best_a, best_val

def _smooth_gradient_descent(a0, n, start_time, deadline, best_a, best_val, target=1.503):
    cur = a0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    size1 = _next_pow2(2 * n - 1)
    size2 = _next_pow2(3 * n - 2)
    no_improve = 0
    IMPATIENT = 50000
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute convolution and objective
        ft_cur = np.fft.rfft(cur, size1)
        b = np.fft.irfft(ft_cur * ft_cur, size1)[:2*n-1]
        maxb = b.max()
        cur_val = 2.0 * n * maxb
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            # print
            # maybe break if sufficiently low
            if cur_val <= target:
                break
        else:
            no_improve += 1
        # compute softmax probabilities
        progress = (now - start_time) / (deadline - start_time)
        tau = 0.5 + (30.0 - 0.5) * progress  # linear increase
        # shift for numeric stability
        shifted = b - maxb
        exp_vals = np.exp(tau * shifted)
        sumexp = exp_vals.sum()
        if sumexp == 0.0:
            p = np.zeros_like(b)
            p[np.argmax(b)] = 1.0
        else:
            p = exp_vals / sumexp
        # gradient
        grad = _gradient(p, cur, size2)
        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = 0.02 * (1.0 - progress) + 0.001 * progress  # decaying learning rate
        cur = cur - lr * m_hat / (np.sqrt(v_hat) + eps)
        # enforce nonnegativity
        cur = np.maximum(cur, 0.0)
        # renorm
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total
        if no_improve > IMPATIENT:
            # jitter
            noise = np.random.normal(scale=0.005, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
    return best_a, best_val

"""
Optimise a non‑negative step‑function for the provided
`evaluate_sequence` objective.

The algorithm consists of three stages:

1. **Parametric sweep** – a quick search over several simple families
   (uniform, power‑laws, symmetric exponentials, arcsine‑like, random
   Dirichlet draws).  The best vector from this sweep seeds the optimiser.

2. **Smooth gradient descent** – minimise
      f(a) = 2·n·max(conv(a,a))
   while keeping Σa = 1 and a ≥ 0.
   The non‑smooth max is approximated by a soft‑max:
        max(b) ≈ (1/τ)·log Σ exp(τ·b)
   which yields a differentiable surrogate.  The gradient of the
   surrogate can be written in closed form as a cross‑correlation and
   is evaluated with FFTs.  An Adam‑style update (additive, followed by
   projection onto the simplex) is performed.  The temperature τ is
   increased linearly from 0.5 to 30 so that the optimisation first
   follows a smooth landscape and later concentrates on the true peak.
   If the optimiser stalls, a tiny Gaussian jitter is injected.

3. **Final uniform mixing** – test a few convex combinations with a
   uniform vector; this often reduces the peak a little further.

The whole procedure respects the 1000 s budget, keeps all entries
non‑negative, and normalises after every modification.  The best vector
found is returned as a plain list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution a * a using FFT of given size."""
    ft = np.fft.rfft(a, size)
    conv = np.fft.irfft(ft * ft, size)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(p: np.ndarray, a: np.ndarray, size: int) -> np.ndarray:
    """
    Compute 2 * (p ⋆ a) where ⋆ is the cross‑correlation.
    The result has length len(a) and equals
        2 * Σ_k a_k p_{i+k}
    for i = 0,…,len(a)-1.
    """
    rev = a[::-1]
    ft_p = np.fft.rfft(p, size)
    ft_rev = np.fft.rfft(rev, size)
    conv = np.fft.irfft(ft_p * ft_rev, size)[: len(p) + len(rev) - 1]
    n = a.shape[0]
    return 2.0 * conv[n - 1 : n - 1 + n]


def _objective(a: np.ndarray) -> float:
    """
    Objective value 2·n·max(conv(a,a)) assuming Σa = 1.
    (The outer wrapper `evaluate_sequence` applies the same formula after
    clamping, but here we work with the exact numbers.)
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, size)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase‑0 – cheap parametric families
# ----------------------------------------------------------------------
def _parametric_sweep(n: int):
    """Return a decent vector and its value obtained by scanning simple families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)

    def try_update(arr):
        nonlocal best_a, best_val
        arr = np.asarray(arr, dtype=np.float64)
        if arr.shape != (n,) or np.any(arr < 0):
            return
        s = arr.sum()
        if s == 0:
            return
        arr = arr / s
        val = _objective(arr)
        if val < best_val:
            best_val, best_a = val, arr.copy()

    rng = np.random.default_rng(424242)

    idx = np.arange(n, dtype=float)

    # simple deterministic patterns
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # triangular/U‑shape
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine bump

    # power‑law families
    for p in np.linspace(0.2, 4.0, 20):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in np.linspace(0.95, 0.9999, 30):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # arcsine‑like shape (high at the boundaries)
    i = np.arange(0, n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(i * (n - i + 0.5))
    try_update(a)

    # random Dirichlet draws
    for _ in range(20):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑A – smooth (soft‑max) gradient descent with Adam
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a0: np.ndarray,
    n: int,
    global_start: float,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
    target_val: float = 1.503,
):
    """
    Adam‑style additive updates on the soft‑max surrogate of max(conv).
    The loop stops when the deadline is reached or the target
    value is attained.
    """
    cur = a0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    size1 = _next_pow2(2 * n - 1)
    size2 = _next_pow2(3 * n - 2)
    no_improve = 0
    IMPATIENT = 60_000  # after many iterations without progress, inject jitter

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- current objective & best‑track -----
        conv = _full_convolution(cur, size1)
        maxb = conv.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[GradDesc {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- soft‑max weights (smooth approximation of the arg‑max) -----
        progress = (now - global_start) / (deadline - global_start)
        tau = 0.5 + (30.0 - 0.5) * progress  # linear schedule, 0.5 → 30

        # stability shift
        shifted = conv - maxb
        exp_vals = np.exp(tau * shifted)
        sumexp = exp_vals.sum()
        if sumexp == 0.0:
            # extreme case – fall back to a one‑hot at the maximal index
            p = np.zeros_like(conv)
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sumexp

        # ----- gradient of the surrogate -----
        grad = _cross_correlation(p, cur, size2)

        # ----- Adam update (additive) -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (decays a bit as we approach the deadline)
        lr = 0.02 * (1.0 - progress) + 0.001 * progress

        cur = cur - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex (non‑negative + sum 1) -----
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        if s == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= s

        # ----- occasional jitter to escape plateaus -----
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.004, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[GradDesc {it:7d}] jitter after stagnation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑C – tiny uniform mixing (convex combination)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int):
    """Try a few convex blends with the uniform vector and keep the best."""
    best_a, best_val = a.copy(), _objective(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.01, 0.02, 0.05):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
