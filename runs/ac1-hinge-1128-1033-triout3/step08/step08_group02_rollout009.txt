"""
Step‑function optimizer for `evaluate_sequence`.

The goal is to minimise
        f(a) = 2·n·max(conv(a,a)) / (∑a)² .
Because the ratio is scale‑invariant we force ∑a = 1 and minimise
        F(a) = 2·n·max(conv(a,a)) .

The optimiser works in four stages (time‑controlled):
  0️⃣  Fast sweep over a few handcrafted families (including a logistic
      shape) and a handful of random Dirichlet draws.
  1️⃣  Exponentiated‑gradient (mirror descent w.r.t. KL) on the exact
      sub‑gradient of the active convolution entry.
  2️⃣  Adam‑style projected gradient descent (sub‑gradient of the same
      entry, with bias‑corrected moments) together with a Euclidean
      projection onto the probability simplex.
  3️⃣  Deterministic greedy hill‑climb that moves a fraction of mass
      from the “worst” index (largest contribution to the current peak)
      to the “best” one (smallest contribution).  Small random shakes
      are added when progress stalls.
  4️⃣  A tiny uniform mixing is finally tried – it occasionally reduces
      the peak a little more.

All phases respect the global 1000 s budget, keep the vector non‑negative,
renorm to Σa = 1 after every modification, and retain the best vector
found.  The returned list of floats can be fed directly to the provided
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two >= out_len
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the O(n log n) algorithm from
    "Efficient projections onto the l1‑ball for learning in high dimensions".
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families – cheap initial sweep
# ----------------------------------------------------------------------
def _logistic_family(n: int, slope: float, centre: float) -> np.ndarray:
    """a_i ∝ 1 / (1 + exp(slope·(i‑centre))) ."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(slope * (i - centre)))
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s


def _initial_candidate(n: int, rng: np.random.Generator):
    """Search a handful of families and keep the best one."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    idx = np.arange(n)

    # deterministic families
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # V‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # quadratic decreasing
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law V‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # logistic families – slope varies, centre placed at quartiles
    for slope in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for centre_frac in (0.25, 0.5, 0.75):
            centre = centre_frac * (n - 1)
            a = _logistic_family(n, slope, centre)
            try_update(a)

    # random Dirichlet draws
    for _ in range(30):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val

    """
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000

    while True:

        if now >= deadline:
            break

        # current objective & convolution
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # sub‑gradient for the active peak
        k = int(np.argmax(b))
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]                     # a_{k‑i}
        g = 2.0 * rev                               # sub‑gradient (ignoring ½)

        # schedule – linear decay from 0.12 → 5e‑4
        progress = (now - start) / (deadline - start)
        eta = (1.0 - progress) * 0.12 + progress * 0.0005

        # exponentiated update (keeps positivity)
        cur = cur * np.exp(-eta * g)

        # renormalise to Σa = 1
        tot = cur.sum()
        if tot <= 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= tot

        # occasional jitter when stuck
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.008, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Adam‑style projected gradient descent
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int,
                start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam updates on the sub‑gradient of the active convolution entry,
    followed by Euclidean projection onto the simplex.
    """
    cur = a0.copy()
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    no_improve = 0
    IMPATIENT = 250_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # evaluate current state
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # sub‑gradient of the active peak
        k = int(np.argmax(b))
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]
        g = 2.0 * rev

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        it += 1
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (linear decay 0.1 → 0.001)
        progress = (now - start) / (deadline - start)
        eta0 = 0.1
        eta = eta0 * (1.0 - progress) + 0.001 * progress

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # move opposite to the gradient
        cur = cur - step

        # project back onto the probability simplex
        cur = _project_simplex(cur)

        # jitter when completely stuck
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.006, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[Adam {it:7d}] jitter after stagnation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – deterministic greedy hill‑climb (as in the reference)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer between the most‑ and least‑contributing indices.
    """
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # transfer fraction schedule
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate active peak
        k = int(np.argmax(b))

        # reverse‑relevant slice (a_{k‑i})
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        # source = index with largest contribution to the peak
        i = int(np.argmax(rev))
        # target = index with smallest contribution (ignore source)
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            # stagnation – add a tiny shake
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # trial move
        cur[i] -= delta
        cur[j] += delta

        # evaluate the trial
        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # accept – nothing else to do
            pass
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 4 – final tiny uniform mixing
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few convex combinations with the uniform vector."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under the given time budget.
    Returns the best list of floats found (normalised to Σa = 1).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin to guarantee we return before the hard timeout
    total_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries works well for the FFT based
    # computations while still offering enough degrees of freedom.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Warm start – reuse previous construction if the harness supplied it.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n, rng)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Time slicing for the four phases
    # ------------------------------------------------------------------
    total_time = total_deadline - start_time
    t0 = start_time + 0.05 * total_time      # after Phase‑0 (family sweep)
    t1 = start_time + 0.30 * total_time      # after Phase‑1 (exp‑grad)
    t2 = start_time + 0.60 * total_time      # after Phase‑2 (Adam)
    t3 = total_deadline                       # final deadline (incl. hill‑climb & mix)

    # ------------------------------------------------------------------
    # Phase 0 – if we started from a global old vector we already have a
    #           reasonable candidate; otherwise the `_initial_candidate`
    #           call performed a short family sweep.
    # ------------------------------------------------------------------
    # (nothing more to do here)

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    best_a, best_val = _expgrad_phase(best_a, n,
                                      start=start_time,
                                      deadline=t1,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑style projected gradient descent
    # ------------------------------------------------------------------
    best_a, best_val = _adam_phase(best_a, n,
                                   start=t1,
                                   deadline=t2,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=1.503)

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy hill‑climbing
    # ------------------------------------------------------------------
    best_a, best_val = _greedy_hillclimb(best_a, n,
                                         start=t2,
                                         deadline=t3,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # Phase 4 – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification through the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
