import time, numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a):
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    np.maximum(conv, 0, out=conv)
    return conv

def _anti_symmetric_reorder(a):
    n = a.shape[0]
    s = np.sort(a)
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]
        b[-1 - i] = s[i]
    if n % 2:
        b[half] = s[half]
    return b

def _gradient_selected_peaks(a, idxs):
    n = a.shape[0]
    conv_len = 2 * n - 1
    p = np.zeros(conv_len, dtype=float)
    p[idxs] = 1.0
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    seg = conv_ap[n - 1: 2 * n - 1]
    g = seg[::-1]
    return g

def _adaptive_exp_gradient(a_start, deadline):
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[gd start] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.5
    decay_factor = 0.98
    rel_tol = 1e-3
    min_peaks = 5

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()
        thresh = max_conv * (1.0 - rel_tol)
        idxs = np.where(conv >= thresh)[0]
        if idxs.size < min_peaks:
            k = min_peaks
            if k < conv.size:

        delta = np.clip(delta, -20.0, 20.0)
        a = a * np.exp(delta)
        a = np.maximum(a, 1e-15)

                    a = a_ord.copy()
                    print(f"[gd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val

def _grad_lp(a, p):
    n = a.shape[0]
    b = _full_convolution(a)          # length 2n-1
    maxb = b.max()
    if maxb == 0:
        return np.zeros_like(a)
    # weights proportional to (b/maxb)^{p-1}
    w = np.empty_like(b)
    np.power(b / maxb, p - 1, out=w)
    w_rev = w[::-1]
    size = _next_pow2(n + w_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[: n + w_rev.shape[0] - 1]
    idx_start = w_rev.shape[0] - 1    # = 2n-2
    grad = 2.0 * conv[idx_start - np.arange(n)]
    return grad

def _lp_descent(a_start, p_list, deadline, current_best_val):
    a = a_start.copy()
    best_a = a.copy()
    best_val = current_best_val
    iteration = 0
    alpha0 = 0.2
    for p in p_list:
        max_iter = 3000
        for _ in range(max_iter):
            if time.time() >= deadline - 5:
                break
            grad = _grad_lp(a, p)
            # simple step schedule
            alpha = alpha0 * (0.99 ** (iteration // 1000))
            delta = -alpha * grad
            delta = np.clip(delta, -20.0, 20.0)
            a = a * np.exp(delta)
            a = np.maximum(a, 1e-15)
            a /= a.sum()

                if cur_val < best_val:

            iteration += 1
        # reorder after each p-stage
        a = _anti_symmetric_reorder(a)
        # keep best snapshot as start for next p
        a = best_a.copy()
    return best_a, best_val

def _hill_climb_refine(a, best_val, deadline, n):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

    """

    start_time = time.time()

"""
Multi‑stage optimisation for `evaluate_sequence`.

The algorithm works as follows:

1. Initialise a probability vector on the simplex (re‑using a known
   construction if available, otherwise random exponential draw).

2. **Lp‑gradient descent** – minimise the Lₚ‑norm of the self‑convolution
   (with p gradually increased: 2 → 4 → 8 → 16).  The gradient
   ∇ₐ Σₖ ( (a∗a)_k / max )^{p} can be written as a single FFT‑based
   convolution, allowing cheap dense updates:
        a ← a·exp(−α·∇)   (mirror‑descent on the simplex).

   After each p‑stage the vector is re‑ordered by pairing the largest
   entries with the smallest (anti‑symmetric re‑ordering) to spread mass
   and further flatten the convolution peak.

3. **Adaptive exponential‑gradient** – a fine‑grained refinement that
   targets the few largest convolution entries (the approach used in the
   reference solution).

4. **Hill‑climbing** – random swaps and tiny mass‑transfers keep only
   strict improvements, polishing the final result.

All steps respect the 1000 s budget and keep the best sequence found.
"""

import time
import numpy as np

# Utility functions

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    np.maximum(conv, 0, out=conv)               # clamp tiny negatives
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]
        b[-1 - i] = s[i]
    if n % 2:
        b[half] = s[half]
    return b


def _gradient_selected_peaks(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """Gradient of the sum of the convolution entries whose indices are `idxs`."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    p = np.zeros(conv_len, dtype=float)
    p[idxs] = 1.0
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    seg = conv_ap[n - 1: 2 * n - 1]
    return seg[::-1]

            else:

        g = _gradient_selected_peaks(a, idxs)
        alpha = alpha0 * (decay_factor ** (it // 2000))
        delta = -alpha * g
        delta = np.clip(delta, -20.0, 20.0)
        a = a * np.exp(delta)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

            if it % 1000 == 0:

                    a = a_ord.copy()

        it += 1

    return best_a, best_val

# Lp‑gradient descent (smooth approximation of the L∞ norm)
# ----------------------------------------------------------------------
def _grad_lp(a: np.ndarray, p: int) -> np.ndarray:
    """
    Gradient of Σₖ ( (a∗a)_k / max )^{p} w.r.t. the coefficients `a`.
    The computation is performed via two FFTs.
    """
    n = a.shape[0]
    b = _full_convolution(a)               # length 2n‑1
    maxb = b.max()
    if maxb == 0:
        return np.zeros_like(a)

    w = np.empty_like(b)
    np.power(b / maxb, p - 1, out=w)        # weights ∝ (b/max)^{p‑1}
    w_rev = w[::-1]

    size = _next_pow2(n + w_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[: n + w_rev.shape[0] - 1]

    idx_start = w_rev.shape[0] - 1          # = 2n‑2
    grad = 2.0 * conv[idx_start - np.arange(n)]
    return grad


def _lp_descent(a_start: np.ndarray, p_list, deadline, current_best_val):
    """
    Multi‑stage Lp‑descent: for each p∈p_list the vector is updated by
    multiplicative mirror‑descent using the gradient from `_grad_lp`.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = current_best_val
    iteration = 0
    alpha0 = 0.2

    for p in p_list:
        max_iter = 3000
        for _ in range(max_iter):
            if time.time() >= deadline - 5:
                break
            grad = _grad_lp(a, p)
            alpha = alpha0 * (0.99 ** (iteration // 1000))
            delta = -alpha * grad
            delta = np.clip(delta, -20.0, 20.0)
            a = a * np.exp(delta)
            a = np.maximum(a, 1e-15)
            a /= a.sum()

            if iteration % 250 == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[lp p={p} iter={iteration}] new best = {best_val:.6f}")
                if iteration % 1000 == 0:
                    a = _anti_symmetric_reorder(a)

            iteration += 1

        # start next p‑stage from the best snapshot of the current stage
        a = best_a.copy()
        a = _anti_symmetric_reorder(a)
        best_a = a.copy()

        if time.time() >= deadline - 5:
            break

    return best_a, best_val

# Local hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, best_val: float, deadline: float, n: int):
    """Random swaps / tiny mass transfers, keeping only strict improvements."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
