from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp(f, M):
    n = len(f)
    m = 2*n - 1
    # Build CSR matrix: rows of A_ub (size m x n)
    rows = []
    cols = []
    data = []
    for k in range(m):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # for i in [i_start, i_end] inclusive:
        for i in range(i_start, i_end+1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    A_ub = csr_matrix((data, (rows, cols)), shape=(m, n))
    b_ub = np.full(m, M)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # fallback to zeros? maybe random
        return np.zeros(n)
    g = res.x
    return g
```

"""
Non‑negative step‑function optimizer.

The algorithm works in two stages:

1️⃣  **Warm‑start** – try to obtain a good starting distribution.
    • If the global variable ``height_sequence_1`` (the best solution from a
      previous run) exists, interpolate it to the target length.
    • Otherwise fall back to a Beta‑shaped family and keep the best candidate.

2️⃣  **Intensive simulated‑annealing hill‑climbing** – from the warm‑start
    repeatedly moves a small amount of mass from an index that contributes
    heavily to the current maximal convolution entry to an index that contributes
    little.  The maximal transfer fraction and the temperature decay very slowly,
    allowing a long exploration phase (the whole remaining time budget).  The
    current best sequence is kept and occasional “perturb‑restart’’ steps help
    escape stagnation.

All updates respect the simplex constraint (Σa = 1) and the non‑negativity
constraint.  The final sequence is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Beta‑family sweep (cheap parametric search)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (identical to the previous version
# but with a slower schedule).
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from an index that
    contributes heavily to the current maximal convolution entry to an index
    that contributes little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule – deliberately stretched so the search lasts until the deadline
    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress in [0,1]
        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each position to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – draw proportional to its contribution
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index – biased towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback (different from source)
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        # amount of mass to move
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # handling stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to minimise
    ``evaluate_sequence``.  The routine respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave 0.5 s for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries give a good trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from the previously best solution if possible
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Cheap parametric sweep (Beta family) + a couple of Dirichlet seeds
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from previous best, value={best_val:.6f}")

    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_val < best_val:
            best_a, best_val = beta_a, beta_val
            print(f"[init] beta‑family improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # a few random Dirichlet draws for diversity
    for _ in range(3):
        rand_a = np.random.dirichlet(np.ones(n))
        val = _objective(rand_a)
        if val < best_val:
            best_a, best_val = rand_a, val
            print(f"[init] random Dirichlet improved value={best_val:.6f}")

    # fallback if everything else failed
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] random fallback, value=", best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Intensive hill‑climbing (the whole remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=start_time,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
