"""
Propose a non‑negative step function that improves the bound on
``evaluate_sequence``.  The method consists of three fast optimisation
phases that all respect the 1000 s budget:

1. **Mirror‑descent on the true max‑convolution** – using the exact
   sub‑gradient of the peak of the self‑convolution and an adaptive
   multiplicative (exponential) update on the simplex.

2. **Targeted pairwise mass transfers** – repeatedly reduces the
   contribution of the pairs that realise the current maximal convolution
   by moving a small amount of mass from those indices to a location that
   contributes little to the peak.  The step size decays gradually.

3. **Very cheap hill‑climbing** – random swaps / transfers that accept
   only strict improvements; this cleans up any remaining local wiggles.

All phases keep the vector normalised to sum = 1 (the objective is
scale‑invariant) and use a fast FFT‑based convolution.  The best vector
found is finally checked with the official ``evaluate_sequence`` and
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all are top‑level, no closures, no lambdas.
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padded FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using an FFT of size the next
    power of two.  Returns a length‑2·n‑1 array.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise can produce tiny negative values – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors already normalised
    to sum = 1.  Returns 2·n·max(conv) (the denominator is 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic reordering that places the largest entries on the left side
    and the smallest on the right side.  Often flattens the convolution
    peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                 # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[-1 - i]        # large → left
        new_a[n - 1 - i] = sorted_vals[i]    # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts *strict* improvements.
    Random swaps (20 %) and mass‑transfer moves (80 %).  The vector is
    kept on the simplex after each move.
    """
    cur_a = a.copy()
    cur_score = cur_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex (should already sum to 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_score - 1e-12:
            cur_a = new_a
            cur_score = val
            print(f"[refine] new best = {cur_score:.6f}")

    return cur_a, cur_score

# Core optimisation primitives

def _mirror_descent_step(a: np.ndarray, eta: float) -> tuple[np.ndarray, int]:
    """
    One mirror‑descent step on the true max‑convolution.
    Returns the updated probability vector and the index of the current
    maximal convolution entry.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k_max = int(conv.argmax())                     # location of the peak

    # Sub‑gradient of max(conv) w.r.t. a:
    #   g_i = a_{k_max-i}  for all i such that 0 ≤ k_max-i < n, else 0.
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    i_vals = np.arange(i_start, i_end + 1, dtype=int)
    j_vals = k_max - i_vals
    g = np.zeros_like(a)
    g[i_vals] = a[j_vals]

    # Multiplicative (exponential) update on the simplex:
    a_new = a * np.exp(-eta * g)

    # Re‑normalise to the simplex (sum = 1).
    s = a_new.sum()
    if s > 0:
        a_new /= s
    else:                                          # safety fallback
        a_new = np.full_like(a, 1.0 / n)

    return a_new, k_max


def _pairwise_mass_move(a: np.ndarray, k_max: int,
                       rng: np.random.Generator,
                       eps_factor: float) -> tuple[np.ndarray, bool]:
    """
    Transfer a small amount of mass from a pair (i, j) that contributes
    heavily to the current maximal convolution to a location that contributes
    little.  Returns the modified vector and a flag indicating whether a
    move was performed.
    """
    n = a.shape[0]
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    if i_start > i_end:
        return a, False

    i_vals = np.arange(i_start, i_end + 1, dtype=int)
    j_vals = k_max - i_vals
    contrib = a[i_vals] * a[j_vals]               # contribution of each pair

    # Choose a pair among the top contributors.
    K = min(10, len(i_vals))
    top_idx = np.argpartition(-contrib, K - 1)[:K]
    if top_idx.size == 0:
        return a, False
    chosen = rng.choice(top_idx)
    i = i_vals[chosen]
    j = j_vals[chosen]

    # Amount to move: a fraction of the smaller of the two entries.
    max_eps = min(a[i], a[j])
    if max_eps <= 0:
        return a, False
    epsilon = eps_factor * max_eps * rng.random()
    if epsilon < 1e-15:
        return a, False

    # Destination – the index with the smallest mass (excluding i and j).
    t = np.argmin(a)
    if t == i or t == j:
        # pick the second‑smallest if needed
        order = np.argsort(a)
        for cand in order:
            if cand not in (i, j):
                t = cand
                break

    new_a = a.copy()
    new_a[i] -= epsilon
    new_a[j] -= epsilon
    new_a[t] += 2.0 * epsilon

    # Numerical safety – enforce non‑negativity and renormalise.
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s > 0:
        new_a /= s
    else:
        new_a = np.full_like(a, 1.0 / n)

    return new_a, True
