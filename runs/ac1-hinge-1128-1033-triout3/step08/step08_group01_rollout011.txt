def _greedy_mass_transfer(a_start, n, start, deadline, best_a, best_val):

    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a, best_val = a.copy(), cur_val

    it=0
    no_improve=0
    IMPROVE_LIMIT = 200_000

    while time.time()<deadline:
        k = int(np.argmax(b))
        idx = k - np.arange(n)
        mask = (idx>=0) & (idx<n)
        overlap = np.zeros(n)
        overlap[mask] = a[idx[mask]]

        grad = 2.0 * overlap
        # source: max grad
        i = int(np.argmax(grad))
        # target: min overlap (could be zero)
        min_val = np.min(overlap)  # but we may also look at overlapping values, includes zeros if any; if all >0 then pick min.
        # Choose random among indices with that minimally overlapping.
        cand_js = np.where(overlap == min_val)[0]
        # ensure j != i; if i in cand_js remove
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            # pick random different index
            j = (i + np.random.randint(1, n)) % n
        else:
            j = int(np.random.choice(cand_js))

        # delta = a[i] (full move)
        delta = a[i]  # could try full
        if delta <= 1e-15:  # no mass
            # maybe choose second source
            # find second largest grad...
            # for simplicity, random move
            i = int(np.random.choice(n))
            delta = a[i] * 0.5
            if delta <=1e-15:
                continue
            j = int(np.random.choice(n))
        # try series of deltas
        success=False
        for scale in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            d = delta * scale
            if d < 1e-12:
                break
            # compute updated b in O(n)
            b_new = b.copy()
            # Update contributions
            b_new[j:j+n] += 2.0 * d * a
            b_new[i:i+n] -= 2.0 * d * a
            b_new[2*j] += d*d
            b_new[2*i] += d*d
            b_new[i+j] -= 2.0 * d*d
            new_val = 2.0 * n * b_new.max()
            if new_val < cur_val - 1e-12:
                # accept
                a[i] -= d
                a[j] += d
                b = b_new
                cur_val = new_val
                success = True
                no_improve = 0
                if new_val < best_val - 1e-12:
                    best_val, best_a = new_val, a.copy()
                    print(f"[Greedy {it:7d}] new best = {best_val:.6f}")
                break
        if not success:
            no_improve+=1

        if no_improve > IMPROVE_LIMIT:
            # restart: perturb best
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Greedy {it:7d}] perturbing after stagnation")
        it += 1

    return best_a

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

using a three‑stage optimisation:

1.  A rich parametric sweep + a short Adam phase on a smooth
    log‑sum‑exp surrogate of the max‑convolution.
2.  A greedy “mass‑transfer” hill‑climber that repeatedly moves the
    most harmful mass (largest contribution to the current maximal
    convolution entry) to the index that contributes the least.
3.  A projected sub‑gradient descent for fine‑tuning.

All phases respect the global time budget (default 1000 s) and the
sequence is kept on the probability simplex (Σ a = 1, a_i ≥ 0).
The final candidate is optionally mixed a little with the uniform
distribution – a cheap move that often reduces the peak further.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1 – the scale does not matter)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {w ≥ 0, Σ w = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Rich parametric initialisation
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector (Σ a = 1) and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ----- deterministic families -------------------------------------------------
    _try(np.ones(n))                                           # uniform

    # exponential / left‑hand side (dense sweep over r)
    for r in np.linspace(0.99, 0.999999, 120):
        a = r ** np.arange(n, dtype=float)
        _try(a)

    # symmetric exponential (U‑shaped)
    for r in np.linspace(0.99, 0.999999, 80):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # power‑law symmetric (sweep exponent)
    for p in np.linspace(0.2, 4.0, 70):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p) + (n - i + 1) ** (-p)
        _try(a)

    # half‑wave sine
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _try(a)

    # triangular (peak in centre, zero at ends)
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    _try(a)

    # ----- random draws -----------------------------------------------------------
    rng = np.random.default_rng(123456)
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on the log‑sum‑exp smoothed max."""
    # logits → simplex via soft‑max
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.003     # a bit more aggressive than before
    eps_start, eps_end = 1e-1, 1e-13   # keep a very small final epsilon

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000

    while time.time() < deadline:
        # ------- current point on simplex -------
        a = np.exp(w - np.max(w))
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ------- schedules -------
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ------- smooth max (log‑sum‑exp) -------
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted - np.max(shifted))   # stabilise
        p = weights / weights.sum()                  # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑ p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                # length ~3n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to shake out of stagnation
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.008, size=n)

        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Greedy mass‑transfer hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> np.ndarray:
    """Move mass from the most damaging index to the least damaging one."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_val, best_a = cur_val, a.copy()
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 250_000

    while time.time() < deadline:
        k = int(np.argmax(b))                # location of the current peak
        idx = k - np.arange(n)               # counterpart index for each position
        mask = (idx >= 0) & (idx < n)

        # overlap values that enter the peak
        overlap = np.zeros(n)
        overlap[mask] = a[idx[mask]]

        # sub‑gradient (up to a factor 2) – we only need ranking
        grad = overlap

        i = int(np.argmax(grad))             # source: largest contributor
        # target: index with minimal overlap (preferably zero)
        min_val = np.min(overlap)
        cand_js = np.where(overlap == min_val)[0]
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            # fall back to a random different index
            j = (i + np.random.randint(1, n)) % n
        else:
            j = int(np.random.choice(cand_js))

        if a[i] <= 1e-15:
            # nothing to move – pick a random source instead
            i = int(np.random.choice(np.where(a > 1e-15)[0]))
            if a[i] <= 1e-15:
                # all entries extremely tiny, give up
                break

        # try a series of decreasing deltas; the first that improves is kept
        delta_full = a[i]
        success = False
        for scale in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
            d = delta_full * scale
            if d < 1e-12:
                break

            # incremental convolution update (O(n))
            b_new = b.copy()
            b_new[j:j + n] += 2.0 * d * a
            b_new[i:i + n] -= 2.0 * d * a
            b_new[2 * j] += d * d
            b_new[2 * i] += d * d
            b_new[i + j] -= 2.0 * d * d

            new_val = 2.0 * n * b_new.max()
            if new_val < cur_val - 1e-13:
                # accept move
                a[i] -= d
                a[j] += d
                b = b_new
                cur_val = new_val
                success = True
                no_improve = 0
                if new_val < best_val - 1e-12:
                    best_val, best_a = new_val, a.copy()
                    print(f"[Greedy {it:7d}] new best = {best_val:.6f}")
                    if best_val <= target_val:
                        return best_a
                break

        if not success:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb around the best seen so far and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Greedy {it:7d}] perturbing after stagnation")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – Exact sub‑gradient descent (projected onto simplex)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n               # initial step size (≈ half the average entry)
    eta = eta0
    decay = 0.9994
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 180_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k = int(np.argmax(b))
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small post‑processing: mixing with uniform
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combos (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
