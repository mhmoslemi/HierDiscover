Algorithm
---------

   Dirichlet draws, exponentials, power‑laws, sinusoid).  The best cheap

   approximation of the max‑convolution with L‑BFGS.  A schedule of
   decreasing ε is used, and many random restarts (≈30) are performed.
   After each L‑BFGS restart the best vector is stored.

   optimised with the Adam optimiser on the same smooth objective.

4. **Exact sub‑gradient descent** – the true sub‑gradient of
   ``max_k (a*a)[k]`` is ``g_i = 2·a_{k*‑i}`` where ``k*`` is the index of

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # numerical artefacts may produce tiny negatives – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already satisfy
    a_i ≥ 0 and Σ a_i = 1.  Returns 2·n·max(conv) (the denominator is 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate of the objective.
    Parameters are logits w (a_i = softmax(w)).  Returns the scalar value
    and the gradient w.r.t. the logits.
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ---------- soft‑max ----------
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---------- convolution ----------
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ---------- smooth max ----------
    shifted = (conv - max_c) / eps   # (conv‑max)/ε
    # log‑sum‑exp of the shifted values
    # using scipy.special.logsumexp would be ideal but we keep pure numpy
    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    weights = np.exp(shifted - max_shifted)   # unnormalised soft‑max over k
    p = weights / np.sum(weights)            # probabilities over lag indices

    # convolution of a with reversed p (needed for grad_a)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # factor 4 comes from chain rule

    # ---------- chain rule through soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return float(L), grad_w

    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    O(n) algorithm (Wang & Carreira‑Perpiñán, 2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic: largest values on the left side, smallest on the right.
    Often reduces the peak of the autocorrelation.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[-1 - i]       # large → left
        new_a[n - 1 - i] = sorted_vals[i]   # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a

    # 8) Final safety check with the official evaluator and return.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
