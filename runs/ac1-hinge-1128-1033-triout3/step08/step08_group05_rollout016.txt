"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

The algorithm consists of three phases:

1. **Smooth surrogate optimisation** – we minimise a log‑sum‑exp
   approximation of the max‑convolution with L‑BFGS on the soft‑max
   (probability‑simplex) parametrisation.  The smoothing parameter ε is
   decreased on a log‑scale; a few random restarts are performed.

2. **Linear‑programming mixing** – given the current vector f, we solve the
   LP

        maximise   Σ g_i
        subject to 0 ≤ g,   (f * g)_k ≤ M   for all k,
                     where M = max_k (f * f)_k.

   The solution g tries to add as much total mass as possible while keeping
   the convolution peak below the current bound M.  After scaling g to the
   same ℓ₁‑norm as f we form a convex combination
        f ← (1‑t)·f + t·g_scaled
   (t slowly decays).  Repeating this step provably never increases the
   max‑convolution and often reduces it.

3. **Final cheap stochastic hill‑climb** – random swaps / tiny mass transfers
   are used to polish the result until the time budget expires.

All phases respect the 1000 s budget; the best sequence found is finally
checked with the official ``evaluate_sequence`` before being returned.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Global placeholders for the smoothed surrogate (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None          # smoothing parameter ε
_GLOBAL_N = None            # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # remove tiny negatives
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ------- soft‑max ----------
    w_shift = w - np.max(w)          # improve numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ------- convolution ----------
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ------- smooth max ----------
    shifted = (conv - max_c) / eps
    log_sum = np.logaddexp.reduce(shifted)          # log‑sum‑exp
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ------- gradient w.r.t. a ----------
    weights = np.exp(shifted)
    p = weights / weights.sum()      # distribution over lag indices

    # derivative of conv_k w.r.t. a_i = Σ_j a_j·δ_{i+j=k} + Σ_i a_i·δ_{i+j=k}
    # This can be written as the convolution of a with the reversed p.
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    seg = conv_ap[n - 1: 2 * n - 1]                # part needed for ∂conv/∂a
    grad_a = 4.0 * n * seg[::-1]

    # ------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w)


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the full formula
    simplifies to this expression).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        return float('inf')
    return float(2.0 * n * max_b)


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (shape (2n‑1, n)) such that
        A @ g  ==  f * g
    :param f: 1‑D non‑negative array of length n.
    :return: scipy.sparse CSR matrix.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # For offset s (0 ≤ s < n) the diagonal entries are constant = f[s].
    # Every diagonal has length n (the shape guarantees it fits).
    diags_data = [np.full(n, float(f[s]), dtype=float) for s in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(diags_data, offsets, shape=(rows, n), format='csr')
    return A


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise Σ g_i
        subject to 0 ≤ g_i,
                  (f * g)_k ≤ M   for all k.
    Returns the optimal g (or None on failure).
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                # (2n‑1) × n, sparse
    b_ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n)                          # minimise -Σ g_i  (i.e. maximise Σ g_i)
    bounds = [(0.0, None)] * n
    # Use HiGHS solver (fast, accepts sparse matrices)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %); the vector is kept on the
    probability simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                 # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

            best_val = cur_val
            best_a = new_a.copy()
            print(f"[LP mix t={t:.4f}] improved to {best_val:.6f}")
        else:
            # no improvement – shrink the step size
            t *= decay
            if t < min_t:
                t = min_t
        # safety check – if LP does not change the vector much we stop
        if np.linalg.norm(new_a - best_a) < 1e-12:
            break

    # ------------------------------------------------------------------
    # Phase 3 – final cheap stochastic hill‑climbing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
