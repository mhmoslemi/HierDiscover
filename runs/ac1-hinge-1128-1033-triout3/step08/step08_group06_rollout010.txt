"""
Propose a step‑function (non‑negative vector on the probability simplex) that
minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.  The code follows a multi‑stage
pipeline:

1️⃣  Initialise a diverse pool of candidates (uniform, exponential, Gamma,
    power‑law, sinusoidal, and the historic best) and keep the best.
2️⃣  Refine the best candidate with a *weighted* exponentiated mirror‑descent
    that explicitly reduces the excess of the convolution over its uniform
    average.  Momentum (ADAM‑style) and occasional anti‑symmetric re‑ordering
    are used.
3️⃣  Optimise a smooth log‑sum‑exp surrogate of the max‑convolution using
    L‑BFGS‑B with several random restarts and a geometric ε‑schedule.
4️⃣  Final local search with a simulated‑annealing hill‑climber (random swaps
    and mass transfers).
All updates keep the vector non‑negative and normalised (∑a_i = 1), so the
objective depends only on the shape of the vector.  The routine respects the
global time budget (default 1000 s) and prints progress information.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)   # eliminate tiny negatives
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Sort ascending and then place the largest values on the left,
    the smallest on the right (effectively a descending order).
    """
    s = np.sort(a)               # ascending
    n = a.shape[0]
    half = n // 2
    res = np.empty_like(a)
    for i in range(half):
        res[i] = s[n - 1 - i]            # large → left side
        res[n - 1 - i] = s[i]            # small → right side
    if n % 2:
        res[half] = s[half]
    return res


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    w_shift = w - np.max(w)                # stability
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t a
    weights = np.exp(shifted)                # ∝ exp((conv-max)/ε)
    p = weights / weights.sum()              # soft‑max over convolution indices

    # convolution a * reversed(p) (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries give derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]            # ∂L/∂a_i

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule: geometric from 5·10⁻³ down to 10⁻¹² (11 points)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=11)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 18.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 800,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})

        w = res.x

        # map back to a (probability simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric re‑ordering heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val


# ----------------------------------------------------------------------
# Weighted exponentiated mirror‑descent (new Stage 1)
# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k‑i} efficiently via FFT.
    ``w`` has length 2·n‑1 (same as a*a convolution).
    """
    n = a.shape[0]
    # reverse w for cross‑correlation
    w_rev = w[::-1]
    size = _next_pow2(n + w_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv_full = np.fft.irfft(fa * fw, size)[: n + w_rev.shape[0] - 1]
    # indices (2·n‑2)‑i give the required scalar product
    idx = (2 * n - 2) - np.arange(n)
    g = conv_full[idx]
    return g


def _weighted_subgradient_descent(a_start: np.ndarray, deadline: float):
    """
    Exponentiated mirror‑descent that reduces the *excess* of the
    convolution over its average value.  Momentum (Adam‑style) and occasional
    anti‑symmetric re‑ordering are used.
    Returns the best (a, value) pair found before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[weighted‑sgd init] value = {best_val:.6f}")

    # momentum buffers
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    alpha0 = 0.6                     # initial step‑size (in log‑space)
    avg_target = 1.0 / (2 * n - 1)   # uniform average of the convolution

    it = 0
    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # excess over the uniform average (only positive parts)
        w = np.maximum(0.0, conv - avg_target)

        # gradient of Σ_k w_k·conv_k w.r.t a
        g = _weighted_gradient(a, w)

        # Adam‑style momentum update (on the *log*‑space)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        # learning rate decays as 1/√t (mirrored in log‑space)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # exponentiated mirror‑descent step
        a = a * np.exp(-alpha * (m_hat / (np.sqrt(v_hat) + eps_adam)))

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional logging & deterministic re‑ordering
        if it % 250 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[weighted‑sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")
            if it % 2000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[weighted‑sgd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (Stage 3 – unchanged)
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.02                     # initial “temperature”

    it = 0
    while time.time() < deadline:
        it += 1
        if rng.random() < 0.2:                  # pure swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                new_a = cur_a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]
            else:
                continue
        else:                                    # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            accept = True
        else:
            T = T0 * (0.9995 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val_local:.6f}")

    return best_a, best_val_local


# ----------------------------------------------------------------------
# Helper factories for diverse initial candidates (Stage 0)
# ----------------------------------------------------------------------
def _powerlaw_decreasing(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64)) ** (-p)
    a /= a.sum()
    return a


def _sine_candidate(n: int) -> np.ndarray:
    """Smooth sinusoidal shape, strictly positive."""
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 0.5) / n) + 1e-6
    a[a < 0] = 0.0
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 kept for compatibility with historic runs.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from the historic best if available, otherwise uniform.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 1e-12 else np.full(n, 1.0 / n)

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – generate a wide variety of cheap candidates.
    # ------------------------------------------------------------------
    candidates = []

    # (i) historic/ uniform
    candidates.append(best_a.copy())

    # (ii) exponential (Gamma(1))

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
