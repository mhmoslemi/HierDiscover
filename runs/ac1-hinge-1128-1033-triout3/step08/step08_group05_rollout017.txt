"""
Propose a non‑negative step‑function that (almost) flattens its self‑convolution.
The search combines three ideas:

1. **Parameterized families** – simple analytic shapes (power‑law,
   exponential, edge‑heavy mixtures) are scanned with a 1‑D golden‑section
   search.  These give good global structures that already beat the
   previous best ratio.

2. **Aggressive smooth L‑BFGS** – the log‑sum‑exp smooth surrogate of the
   max‑convolution is minimised with many restarts and a very fine ε
   schedule, starting from the best family candidate.

3. **Fine‑grained mirror‑descent + annealed hill‑climbing** – an exact
   sub‑gradient mirror‑descent followed by a temperature‑controlled
   mass‑transfer hill‑climber finishes the optimisation.

All stages respect the 1000 s budget; the final sequence is checked with the
official ``evaluate_sequence`` before being returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smooth surrogate (set inside propose_candidate)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Clip tiny negatives caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the full formula
    simplifies to this expression).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # soft‑max to obtain a
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution
    conv = _full_convolution(a)
    max_c = conv.max()

    # smooth max
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient wrt a
    weights = np.exp(shifted)
    p = weights / weights.sum()          # distribution over lags

    # derivative of conv_k wrt a is the convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]        # part corresponding to ∂conv/∂a
    grad_a = 4.0 * n * seg[::-1]          # (factor 2 per ordered pair absorbed)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climber (strict improvements only).
    Random swaps (20 %) and mass transfers (80 %).  The vector stays on
    the probability simplex.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.03 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # stay on simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Parameterised families (used for the global search)
# ----------------------------------------------------------------------
def _shape_powerlaw(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) + (n-i)^(-p), symmetric, summed to 1."""
    idx = np.arange(n) + 1
    vals = idx ** (-p) + (n - idx + 1) ** (-p)
    vals /= vals.sum()
    return vals


def _shape_exponential(beta: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-β·i) + exp(-β·(n‑1‑i)), symmetric, summed to 1."""
    idx = np.arange(n)
    vals = np.exp(-beta * idx) + np.exp(-beta * (n - 1 - idx))
    vals /= vals.sum()
    return vals


def _shape_edge_mix(lam: float, n: int) -> np.ndarray:
    """Convex mixture of the (edge‑heavy) linear shape and uniform."""
    # linear edge‑heavy: large on both ends, linearly decreasing to centre
    left = np.arange(1, n // 2 + 1)
    right = left[::-1] if n % 2 == 0 else np.append(left, left[-1])
    lin = np.concatenate([left, right])[:n]
    lin = lin.astype(float)
    lin /= lin.sum()
    uniform = np.full(n, 1.0 / n)
    vals = lam * lin + (1.0 - lam) * uniform
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# 1‑D golden‑section search (minimise a univariate function)
# ----------------------------------------------------------------------
def _golden_section(f, a, b, max_iter=30, tol=1e-12):
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    x1 = b - phi * (b - a)
    x2 = a + phi * (b - a)
    f1 = f(x1)
    f2 = f(x2)
    for _ in range(max_iter):
        if f1 > f2:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + phi * (b - a)
            f2 = f(x2)
        else:
            b = x2
            x2 = x1
            f2 = f1
            x1 = b - phi * (b - a)
            f1 = f(x1)
        if abs(b - a) < tol:
            break
    return (x1 + x2) / 2.0

            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on the smooth surrogate of max‑convolution.
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-1, -18, num=30)  # from 1e‑1 down to 1e‑18
    w = np.log(best_a + 1e-12)                  # initialise soft‑max params
    max_restarts = 12

    for restart in range(max_restarts):
        if time.time() >= deadline - 150.0:
            break
        print(f"[L‑BFGS restart {restart + 1}]")
        if restart % 3 == 0:
            # occasional full random restart (drawn from N(0,2))
            w = rng.normal(size=n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 150.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 3000,
                                    'ftol': 1e-14,
                                    'gtol': 1e-14,
                                    'disp': False})
            w = res.x
            # back‑transform to probabilities
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val - 1e-10:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[L‑BFGS eps={eps:.1e}] new best = {best_val:.6f}")

        # occasional simple re‑ordering: descending (the “anti‑symmetric” step)
        a_ord = np.sort(best_a)[::-1]
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val - 1e-10:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – refined mirror‑descent on the exact max‑convolution.
    # ------------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.5
    decay = 0.9997
    min_alpha = 0.003
    it = 0
    while time.time() < deadline - 100.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a (ignoring constant factor 2)
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]                     # see analysis notes

        alpha = max(alpha0 * (decay ** (it // 200)), min_alpha)
        a = a * np.exp(-alpha * g)          # exponential mirror‑descent
        # projection onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 250 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[MD it {it}] new best = {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – targeted pairwise mass‑transfer hill‑climbing with annealing.
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.04                # initial temperature
    cooling = 0.9999
    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        # indices that contribute to the peak
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        contrib = cur_a[idx] * cur_a[jdx]

        pair_idx = np.argmax(contrib)
        i = idx[pair_idx]
        j = jdx[pair_idx]

        # choose the larger of the two as donor
        donor = i if cur_a[i] >= cur_a[j] else j

        # pick a recipient among the smallest 5 % entries
        low_mass = np.where(cur_a < np.percentile(cur_a, 5))[0]
        if low_mass.size == 0:
            recipient = rng.integers(0, n)
        else:
            recipient = rng.choice(low_mass)

        delta = cur_a[donor] * 0.015 * rng.random()   # up to 1.5 % transfer
        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[recipient] += delta

        # simplex projection
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[pairhill] improved to {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # Phase 4 – cheap final polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check using the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
