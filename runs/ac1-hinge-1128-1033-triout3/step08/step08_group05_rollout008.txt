"""

Algorithm (≈1000 s budget)

1. **Initial pool** – several simple seeds (uniform, old solution, Dirichlet,
   sinusoid, anti‑symmetric reorder) are normalised and the best of them
   evaluated.

2. **Aggressive L‑BFGS on a smooth surrogate** – the max‑convolution is
   approximated by a log‑sum‑exp.  We minimise the surrogate with many random
   restarts and a geometric ε‑schedule (1e‑1 → 1e‑14).  The variables are the
   log‑softmax parameters *w* (so the simplex constraint is satisfied implicitly).

3. **Adam fine‑tuning on the same surrogate** – after the L‑BFGS phase we run
   a few thousand Adam steps for each ε on a finer schedule (down to 1e‑14).
   This removes the remaining “jitter” left by the quasi‑Newton method.

4. **Mirror‑descent on the exact max‑convolution** – using the true sub‑gradient
   of the current peak we perform multiplicative updates (exponential mirror‑
   descent) with a slowly decaying step‑size.

5. **Targeted pairwise mass‑transfer hill‑climbing** – repeatedly locate the
   pair (i,j) that contributes most to the maximal convolution entry and move
   a small fraction of mass from the larger of the two to a low‑mass index.
   Simulated‑annealing (tiny temperature) is used to occasionally accept
   uphill moves.

6. **Cheap random polish** – many short local moves (random swaps or tiny mass
   transfers) are accepted only if they improve the true objective.

All intermediate vectors are projected onto the probability simplex
(Σ a_i = 1, a_i ≥ 0).  The best sequence found is finally checked with the

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (power‑of‑2 padding)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may produce tiny negative values – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv

    a_i = exp(w_i) / ∑_j exp(w_j)   (a lives on the simplex)

    L_ε = 2·n·max(conv) + 2·n·ε·log ∑_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max -----
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()                     # a ∈ Δ_n

    # ----- full self‑convolution -----
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ----- smoothed max -----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    weights = np.exp(shifted)        # ∝ exp((conv‑max)/ε)
    p = weights / weights.sum()      # distribution over lag indices

    # derivative of smooth max wrt each conv entry = p
    # grad_a = 4·n·(p * (∂conv/∂a))
    # ∂conv_k/∂a_i = a_{k‑i} + a_{i‑k} (symmetrised).  This can be written
    # as the convolution of a with the reversed p.
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # The segment that aligns with ∂conv/∂a (see derivation in the comment)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)      # ∇_w L = J_softmax^T ∇_a L
    return L, grad_w


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors on the simplex.
    Returns the exact value 2·n·max(conv) (the denominator Σ a_i = 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic: large values on the left, small values on the right.
    Often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = s[n - 1 - i]        # large → left side
        new_a[n - 1 - i] = s[i]        # small → right side
    if n % 2:
        new_a[half] = s[half]
    return new_a


def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climber that only keeps strict improvements.
    Random swaps (20 %) and tiny mass transfers (80 %).  The vector stays on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:  # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:  # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] < 1e-15:
                continue
            delta = cur_a[i] * 0.02 * rng.random()   # up to 2 % of donor
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Returns the best list found before the time budget
    expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 5.0       # keep a safety margin

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined in the surrounding environment
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = int(old.shape[0])
        has_old = True
        print(f"[info] using existing construction of length {n}")
    except Exception:
        n = 4000
        has_old = False
        print(f"[info] falling back to default length {n}")

    # ------------------------------------------------------------------
    # Initial candidate pool
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        candidates.append(old / old.sum())

    # Dirichlet draws (α = 1 and α = 5)
    candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoid plus a tiny offset to avoid zeros
    x = np.linspace(0, np.pi, n)
    candidates.append(np.sin(x) + 0.1)

    # anti‑symmetric version of the uniform vector (helps early on)
    candidates.append(_anti_symmetric_reorder(np.full(n, 1.0 / n)))

    # normalise everything onto the simplex
    candidates = [c / c.sum() for c in candidates]

    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on smoothed surrogate
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-1), np.log10(1e-14), num=20)
    w = np.log(best_a + 1e-12)                 # soft‑max parameters
    max_restarts = 10
    for restart in range(max_restarts):
        if time.time() >= deadline - 150.0:
            break
        print(f"[L‑BFGS restart {restart + 1}/{max_restarts}]")
        if restart % 3 == 0:
            # fresh random start every third iteration
            w = rng.normal(size=n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 150.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 2500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            # back‑transform to simplex
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[L‑BFGS eps={eps:.1e}] new best = {best_val:.6f}")

        # occasional anti‑symmetric re‑ordering
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val - 1e-9:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam fine‑tuning on the smoothed surrogate
    # ------------------------------------------------------------------
    w = np.log(best_a + 1e-12)                 # restart from best L‑BFGS point
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    t = 0
    # finer ε schedule (more points, down to 1e‑14)
    fine_eps = np.logspace(np.log10(5e-2), np.log10(1e-14), num=30)
    for eps in fine_eps:
        if time.time() >= deadline - 80.0:
            break
        _GLOBAL_EPS = eps
        _GLOBAL_N = n
        # run a small batch of Adam steps for each ε
        steps = 400
        for _ in range(steps):
            t += 1
            loss, grad = _smooth_obj_and_grad(w)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            w -= 0.2 * m_hat / (np.sqrt(v_hat) + adam_eps)

        # evaluate current point
        w_shift = w - np.max(w)
        a_candidate = np.exp(w_shift)
        a_candidate /= a_candidate.sum()
        cur_val = _fast_evaluate(a_candidate)
        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_a = a_candidate.copy()
            print(f"[Adam ε={eps:.1e}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – refined mirror‑descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.6
    decay = 0.99995
    min_alpha = 0.004
    it = 0
    while time.time() < deadline - 40.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) (ignoring the constant factor 2)
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]                     # derivative = a_{k‑i}
        # (the factor 2 is absorbed by the step‑size)

        # exponential mirror‑descent update
        alpha = max(alpha0 * (decay ** (it // 500)), min_alpha)
        a = a * np.exp(-alpha * g)

        # project back onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 300 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = a.copy()
                print(f"[MD it {it}] new best = {best_val:.6f}")

            if it % 1500 == 0:
                # occasional anti‑symmetric reordering
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val - 1e-9:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[MD reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 4 – targeted pairwise mass‑transfer hill‑climbing (annealed)
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.03
    cooling = 0.99992

    while time.time() < deadline - 10.0:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        # indices participating in the maximal entry
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        contrib = cur_a[idx] * cur_a[jdx]

        # pair with the largest contribution
        pair_idx = np.argmax(contrib)
        i = idx[pair_idx]
        j = jdx[pair_idx]

        # donor = the larger of the two
        donor = i if cur_a[i] >= cur_a[j] else j

        # choose a low‑mass recipient (bottom 6 % of entries)
        low_mass = np.where(cur_a < np.percentile(cur_a, 6))[0]
        if low_mass.size == 0:
            recipient = rng.integers(0, n)
        else:
            recipient = rng.choice(low_mass)

        # move a modest portion (up to 2.5 % of donor) to the recipient
        delta = cur_a[donor] * 0.025 * rng.random()
        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[recipient] += delta

        # project onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[pairhill] improved to {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # Phase 5 – final cheap stochastic polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
