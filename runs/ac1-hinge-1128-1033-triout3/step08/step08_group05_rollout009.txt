import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Globals used by the smooth surrogate (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than *x* (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)                # eliminate tiny negatives
    return conv


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε)
    together with its gradient w.r.t. the soft‑max parameters w.

    The decision variables are w, with a_i = softmax(w)_i.
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- a = soft‑max(w) -----
    a = _softmax(w)

    # ----- convolution -----
    conv = _full_convolution(a)          # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max -----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    weights = np.exp(shifted)
    p = weights / weights.sum()           # softmax over lag indices

    # derivative of max(conv) w.r.t. each conv entry = p
    # ∂conv/∂a = convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment that corresponds to ∂conv/∂a (see derivation in the comment)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]         # reverse to align with a_i ordering

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already satisfy
    Σ a_i = 1 and are non‑negative.
    Returns 2·n·max(conv).  The denominator Σ a_i² cancels out because it is 1.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place large values on the left side and small values on the right.
    Often flattens the peak of the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _mirror_descent_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One mirror‑descent (exponential) update using the *exact* sub‑gradient of
    the current maximal convolution entry.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k = int(conv.argmax())

    # ---- exact sub‑gradient of max(conv) w.r.t. a ----
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    jdx = k - idx

    # each pair (i,j) contributes a_j to ∂/∂a_i and a_i to ∂/∂a_j
    g[idx] += a[jdx]
    g[jdx] += a[idx]            # diagonal terms receive 2·a_i automatically

    # exponential mirror‑descent
    a_new = a * np.exp(-step * g)

    # project back onto the simplex
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new


def _annealed_pairwise_move(a: np.ndarray, rng: np.random.Generator,
                            temperature: float) -> tuple[np.ndarray, bool]:
    """
    Propose a small transfer of mass from the index participating most in the
    current peak to a low‑mass index.  Returns (new_a, accepted).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k = int(conv.argmax())

    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    jdx = k - idx
    contrib = a[idx] * a[jdx]

    # pair with the largest contribution
    best_pair = np.argmax(contrib)
    i = idx[best_pair]
    j = jdx[best_pair]

    donor = i if a[i] >= a[j] else j

    # pick a low‑mass recipient (among the 5 % smallest entries)
    low_mask = a < np.percentile(a, 5)
    low_ids = np.nonzero(low_mask)[0]
    recipient = rng.choice(low_ids) if low_ids.size else rng.integers(0, n)

    if donor == recipient:
        return a, False

    delta = a[donor] * 0.03 * rng.random()           # up to 3 % of donor's mass
    new_a = a.copy()
    new_a[donor] -= delta
    new_a[recipient] += delta

    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s < 1e-12:
        return a, False
    new_a /= s

    old_val = _fast_evaluate(a)
    new_val = _fast_evaluate(new_a)

    if new_val < old_val:
        accept = True
    else:
        accept = rng.random() < np.exp(-(new_val - old_val) / max(temperature, 1e-12))
    return (new_a, accept) if accept else (a, False)


def _adam_optimize(w0: np.ndarray, eps: float,
                   max_iters: int, deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate L_ε.
    Returns the best vector (as probabilities) and its evaluation value.
    """
    global _GLOBAL_EPS, _GLOBAL_N
    _GLOBAL_EPS = eps
    _GLOBAL_N = w0.shape[0]

    w = w0.copy()
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05                     # learning rate for the w‑space

    best_a = _softmax(w)
    best_val = _fast_evaluate(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        L, grad = _smooth_obj_and_grad(w)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if t % 200 == 0:
            a = _softmax(w)
            val = _fast_evaluate(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a plain Python list) before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin for the final call

    # ------------------------------------------------------------------
    # Determine problem size n (reuse the previous construction if available)
    # ------------------------------------------------------------------

        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # Helper to keep the current best solution
    # ------------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    def update_best(candidate):
        nonlocal best_a, best_val
        val = _fast_evaluate(candidate)
        if val < best_val - 1e-12:
            best_val = val
            best_a = candidate.copy()
            print(f"[update] new best = {best_val:.6f}")
        return val

    # ------------------------------------------------------------------
    # Stage 0 – generate a diverse initial population
    # ------------------------------------------------------------------
    init_candidates = []

    # (0) uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # (1) previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # (2) Dirichlet draws with several concentration parameters
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        cand = rng.gamma(shape=alpha, scale=1.0, size=n)
        init_candidates.append(cand)

    # (3) simple sinusoid (positive)
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # (4) linear decreasing shape
    lin = np.linspace(1.0, 0.0, n) + 0.05
    init_candidates.append(lin)

    # (5) quadratic decreasing shape
    quad = (np.arange(n, 0, -1) ** 2).astype(float) + 0.01
    init_candidates.append(quad)

    # (6) anti‑symmetric re‑ordering of the uniform vector
    init_candidates.append(_anti_symmetric_reorder(np.full(n, 1.0 / n)))

    # normalise all candidates onto the simplex
    init_candidates = [c / c.sum() for c in init_candidates]

    for idx, cand in enumerate(init_candidates):
        val = update_best(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – aggressive L‑BFGS on a smooth surrogate
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-1, -15, num=25)    # 1e‑1 … 1e‑15
    max_restarts = 10
    w = np.log(best_a + 1e-12)                     # initialise soft‑max parameters

    for restart in range(max_restarts):
        if time.time() >= deadline - 200.0:
            break
        print(f"[L‑BFGS restart {restart + 1}]")
        # occasional full random restart of the w‑vector
        if restart % 4 == 0:
            w = rng.normal(size=n) * 0.8

        for eps in eps_schedule:
            if time.time() >= deadline - 200.0:
                break
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 4000,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x
            a_candidate = _softmax(w)
            update_best(a_candidate)

        # try a cheap anti‑symmetric re‑order after each restart
        a_ord = _anti_symmetric_reorder(best_a)
        update_best(a_ord)

    # ------------------------------------------------------------------
    # Stage 2 – refined exact mirror‑descent on the true max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.6
    decay = 0.99995
    min_alpha = 0.004
    it = 0
    while time.time() < deadline - 150.0:
        step = max(alpha0 * (decay ** (it // 500)), min_alpha)
        a = _mirror_descent_step(a, step)
        if it % 300 == 0:
            update_best(a)
            # occasional global re‑order (helps to escape shallow traps)
            if it % 1500 == 0:
                a = _anti_symmetric_reorder(a)
        it += 1

    # ------------------------------------------------------------------
    # Stage 3 – targeted pairwise mass‑transfer hill‑climbing with annealing
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.03
    cooling = 0.99994
    while time.time() < deadline - 70.0:
        cur_a, accepted = _annealed_pairwise_move(cur_a, rng, T)
        if accepted:
            cur_val = _fast_evaluate(cur_a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[pairhill] improved to {best_val:.6f}")
        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # Stage 4 – final Adam optimisation on the smooth surrogate (tiny ε)
    # ------------------------------------------------------------------
    eps_final = 1e-15
    w0 = np.log(best_a + 1e-12)
    a_adam, val_adam = _adam_optimize(w0, eps_final,
                                      max_iters=25000,
                                      deadline=deadline - 15.0,
                                      rng=rng)
    update_best(a_adam)

    # ------------------------------------------------------------------
    # Stage 5 – ultra‑light stochastic polish (strict improvements only)
    # ------------------------------------------------------------------
    def cheap_polish(a_vec, cur_best, until):
        a_cur = a_vec.copy()
        best_local = cur_best
        while time.time() < until:
            if rng.random() < 0.2:      # pure swap
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    new_a = a_cur.copy()
                    new_a[i], new_a[j] = new_a[j], new_a[i]
                else:
                    continue
            else:                       # tiny mass transfer
                i, j = rng.integers(0, n, size=2)
                if i == j or a_cur[i] < 1e-15:
                    continue
                delta = a_cur[i] * 0.02 * rng.random()
                new_a = a_cur.copy()
                new_a[i] -= delta
                new_a[j] += delta
            new_a = np.maximum(new_a, 0.0)
            s = new_a.sum()
            if s < 1e-12:
                continue
            new_a /= s
            val = _fast_evaluate(new_a)
            if val < best_local - 1e-12:
                a_cur = new_a
                best_local = val
                print(f"[polish] new best = {best_local:.6f}")
        return a_cur, best_local

    best_a, best_val = cheap_polish(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
