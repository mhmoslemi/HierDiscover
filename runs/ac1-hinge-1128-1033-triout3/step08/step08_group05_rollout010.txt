import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def _next_pow2(x):
    # return smallest power of two >= x
    return 1 << (x - 1).bit_length()

def _full_convolution(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Remove round‑off negatives
    np.maximum(conv, 0, out=conv)
    return conv

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _fast_evaluate(a):
    """2·n·max(conv) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())

# --- globals for the smooth surrogate -------------------------------------------------
_GLOBAL_EPS = None
_GLOBAL_N = None

def _smooth_obj_and_grad(w):
    """
    Smooth surrogate L_ε = 2 n max(conv) + 2 n ε log‑sum‑exp((conv - max)/ε)
    with soft‑max parametrization w -> a.
    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # soft‑max
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)
    max_c = conv.max()

    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()  # probability over lin. conv indices
    p_rev = p[::-1]

    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # slice needed for ∂conv/∂a
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

def _smooth_obj_and_grad_wrapper(w):
    return _smooth_obj_and_grad(w)

def _anti_symmetric_reorder(a):
    """Heuristic rearrangement: large values left, small values right."""
    n = a.shape[0]
    sorted_vals = np.sort(a)
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]
        new_a[n - 1 - i] = sorted_vals[i]
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a

def _refine_hill_climb(a, best_val, deadline):
    """
    Very cheap stochastic hill‑climber that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %). The vector is kept on the simplex.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

def _gradient_descent_line_search(a_start, cur_best_val, deadline):
    """
    Projected exponential (mirror) descent with back‑tracking line search.
    Guarantees monotone decrease of the objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = cur_best_val
    n = a.shape[0]
    it = 0
    # initial step size
    eta0 = 0.5
    while time.time() < deadline:
        conv = _full_convolution(a)
        max_val = conv.max()
        max_k = int(conv.argmax())

        # compute sub‑gradient of max(conv) with respect to a
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1)
        jdx = max_k - idx
        g[idx] = a[jdx]   # contribution a_{k*-i}

        # step size schedule: slowly decay
        eta = eta0 * (0.99 ** (it // 500))
        min_eta = 1e-6
        while eta >= min_eta:
            cand = a * np.exp(-eta * g)
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s > 0:
                cand /= s
            else:
                cand = np.full_like(a, 1.0 / n)

            cand_val = _fast_evaluate(cand)
            if cand_val < best_val - 1e-12:
                a = cand
                best_val = cand_val
                best_a = cand.copy()
                print(f"[GD it {it}] new best = {best_val:.6f} (eta={eta:.2e})")
                break
            else:
                eta *= 0.5   # backtrack
        # If we fell out of the inner loop without improvement, we still keep the
        # current a (no change) and continue – this usually means we are already
        # close to a stationary point.
        it += 1

        # occasional anti‑symmetric reorder to explore a different region
        if it % 2000 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val - 1e-12:
                a = a_ord
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[GD reorder] new best = {best_val:.6f} at it {it}")

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method combines:
    * a smooth surrogate optimisation (L‑BFGS on a log‑sum‑exp approximation),
    * a monotone projected exponential gradient descent with line‑search,
    * a targeted pairwise mass‑transfer hill‑climber,
    * a cheap final polish.
    The routine respects the given time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    # ------------------------------------------------------------
    # Choose problem size n (use existing solution if present)
    # ------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ----------------------------------------------------------------
    # Build a diverse set of initial candidates on the probability simplex
    # ----------------------------------------------------------------
    candidates = []

    # Uniform
    candidates.append(np.full(n, 1.0 / n))

    # Existing construction (if any)
    if has_old:
        candidates.append(old / old.sum())

    # Dirichlet‑like random draws
    candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # Simple sinusoid, shifted upward to avoid zeros
    x = np.linspace(0, np.pi, n)
    candidates.append(np.sin(x) + 0.1)

    # Triangular (linearly increasing then decreasing)
    tri = np.concatenate([np.linspace(0, 1, n // 2 + 1),
                          np.linspace(1, 0, n - n // 2 - 1)])
    candidates.append(tri + 1e-3)

    # Normalise onto simplex
    candidates = [c / c.sum() for c in candidates]

    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] value = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best value = {best_val:.6f}")

    # ----------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on a smooth surrogate of max‑convolution
    # ----------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-1), np.log10(1e-14), num=20)
    w = np.log(best_a + 1e-12)   # soft‑max parameters
    max_restarts = 6

    for restart in range(max_restarts):
        if time.time() >= deadline - 200.0:
            break
        print(f"[L‑BFGS restart {restart + 1}]")
        if restart % 3 == 0:
            w = rng.normal(size=n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 200.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 1200,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x
            # Transform back to probabilities
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[L‑BFGS eps={eps:.1e}] new best = {best_val:.6f}")

        # occasional anti‑symmetric reordering
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val - 1e-9:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ----------------------------------------------------------------
    # Phase 2 – monotone projected exponential gradient descent (line‑search)
    # ----------------------------------------------------------------
    if time.time() < deadline - 150.0:
        best_a, best_val = _gradient_descent_line_search(best_a, best_val,
                                                          deadline - 150.0)

    # ----------------------------------------------------------------
    # Phase 3 – targeted pairwise mass‑transfer hill‑climbing with annealing
    # ----------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.02
    cooling = 0.99995
    while time.time() < deadline - 70.0:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        contrib = cur_a[idx] * cur_a[jdx]

        pair_idx = np.argmax(contrib)
        i = idx[pair_idx]
        j = jdx[pair_idx]

        donor = i if cur_a[i] >= cur_a[j] else j

        # low‑mass recipients (5 % smallest entries)
        low_mass = np.where(cur_a < np.percentile(cur_a, 5))[0]
        if low_mass.size == 0:
            recipient = rng.integers(0, n)
        else:
            recipient = rng.choice(low_mass)

        # adaptive delta – start with up to 5 % of donor mass, back‑track if needed
        delta = cur_a[donor] * 0.05 * rng.random()
        accepted = False
        while delta > 1e-12:
            new_a = cur_a.copy()
            new_a[donor] -= delta
            new_a[recipient] += delta
            new_a = np.maximum(new_a, 0)
            s = new_a.sum()
            if s < 1e-12:
                break
            new_a /= s
            new_val = _fast_evaluate(new_a)
            if new_val < cur_val:
                accepted = True
                break
            # try a smaller delta
            delta *= 0.5

        if not accepted:
            # possibly accept uphill with small probability (annealing)
            if rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                accepted = True

        if accepted:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[pairhill] improved to {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ----------------------------------------------------------------
    # Phase 4 – final cheap stochastic polish
    # ----------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ----------------------------------------------------------------
    # Final safety check with the official evaluator.
    # ----------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()

"""
Search for a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm works in four stages:

1. **Smooth surrogate optimisation** – L‑BFGS on a log‑sum‑exp
   approximation of the max‑convolution, using a soft‑max
   parametrisation of the probabilities.
2. **Monotone projected exponential gradient descent** – an
   exponential mirror‑descent step with a back‑tracking line search.
   The step size decays slowly and the update is always accepted
   only if the objective improves.
3. **Targeted pairwise mass‑transfer hill‑climbing** – the pair that
   contributes most to the current maximal convolution entry is
   identified and a small fraction of its mass is moved to a low‑mass
   index (with an adaptive step‑size and a tiny annealing component).
4. **Cheap final polish** – a lightweight stochastic hill‑climber that
   tries random swaps and small mass transfers, accepting only strict
   improvements.

All vectors are projected onto the probability simplex (∑aᵢ = 1,
aᵢ ≥ 0).  The routine respects the time budget (≈1000 s) and finally
returns the best sequence it found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Clip tiny negatives caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors
    that sum to 1.  Returns 2·n·max(conv) (the denominator Σ a_i = 1,
    so the full formula simplifies to this expression).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Globals used by the smooth surrogate (set before each L‑BFGS call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None
_GLOBAL_N = None


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ---------- soft‑max ----------
    w_shift = w - np.max(w)          # numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---------- convolution ----------
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ---------- smooth max ----------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    weights = np.exp(shifted)
    p = weights / weights.sum()      # distribution over lag indices
    p_rev = p[::-1]

    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment that corresponds to ∂conv/∂a
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]    # reverse to align with a_i order

    # ---------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places large values on one side and small values on the
    opposite side – often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %). The vector is kept
    on the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def _gradient_descent_line_search(a_start: np.ndarray,
                                  cur_best_val: float,
                                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected exponential (mirror) descent with a simple back‑tracking
    line‑search.  The update is accepted only if the objective improves,
    guaranteeing a monotone descent.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = cur_best_val
    n = a.shape[0]
    it = 0
    eta0 = 0.5                       # initial step size

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a (ignoring the factor 2)
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1)
        jdx = max_k - idx
        g[idx] = a[jdx]

        # slowly decaying step size
        eta = eta0 * (0.99 ** (it // 500))
        min_eta = 1e-6

        while eta >= min_eta:
            cand = a * np.exp(-eta * g)
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s > 0:
                cand /= s
            else:
                cand = np.full_like(a, 1.0 / n)

            cand_val = _fast_evaluate(cand)
            if cand_val < best_val - 1e-12:
                a = cand
                best_val = cand_val
                best_a = cand.copy()
                print(f"[GD it {it}] new best = {best_val:.6f} (eta={eta:.2e})")
                break
            else:
                eta *= 0.5   # back‑track

        it += 1

        # occasional anti‑symmetric re‑ordering
        if it % 2000 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val - 1e-12:
                a = a_ord
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[GD reorder] new best = {best_val:.6f} at it {it}")

    return best_a, best_val
