"""
Improved search for a non‑negative step function minimising
``evaluate_sequence``.

The algorithm consists of four stages:

1️⃣  **Diverse initialisation** – a dense sweep of the two‑parameter
    Beta family, a few classical window shapes (Hann, Hamming,
    Blackman), an arcsine‑type shape, and several random Dirichlet
    draws.  The best of these seeds is kept.

2️⃣  **Adam optimiser on the simplex** – we minimise a smooth
    approximation of the true objective plus a weighted variance penalty.
    The variance weight is considerably larger than before (≈10·n) and
    stays high for a longer fraction of the budget.  Learning rate,
    smooth‑max temperature and the variance weight are all annealed.

3️⃣  **Simulated‑annealing hill‑climbing** – a fast O(n) local search
    that repeatedly moves a small amount of mass from indices that
    contribute heavily to the current maximal convolution to indices that
    contribute little.  Parameters are annealed to zero.

4️⃣  **Fine‑grained sub‑gradient polishing** – after the hill climb we
    perform a few rounds of projected sub‑gradient descent on the exact
    max‑convolution (no smooth approximation) with a tiny step size.
    This often squeezes a little more value out of the current candidate.

All phases respect the 1000 s budget.  The final sequence is returned as a
plain Python list; its sum is 1, therefore the evaluation is fully
scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (handles any sum, but we keep sum≈1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a given probability vector."""
    sum_a = a.sum()
    if sum_a < 0.01:                      # same guard as evaluate_sequence
        return np.inf
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # include 2·n factor
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Variance of the convolution and its gradient
# ----------------------------------------------------------------------
def _var_and_grad(a: np.ndarray):
    """
    Returns   var(b) = mean((b‑avg)²)   and   ∂var/∂a   (no extra scaling).
    Here b = a*a and avg = 1/(2n‑1) because Σb = (Σa)² = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # gradient: (4/m) * cross‑correlation of a with v
    conv_ab = _conv_fft(a, v[::-1])              # length n+m-1 = 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad = (4.0 / m) * segment[::-1]
    return var, grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x >= 0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth max + variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises
        L(a) = 2·n·smooth_max(a) + λ·Var(conv(a,a)),
    where λ is annealed from λ_start to 0.
    Simplex constraint Σa=1 is enforced via a log‑softmax parametrisation.
    """
    # logits that will be soft‑maxed to obtain a probability vector
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.2, 0.001
    eps_start, eps_end = 0.8, 1e-9

    # variance‑penalty schedule (larger than before)
    lam_start = 10.0 * n          # stronger flattening pressure
    lam_end = 0.0

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # --------- obtain probability vector ----------
        w_shift = w - np.max(w)         # for numerical stability
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # --------- schedules ----------
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        lam = lam_start * (1.0 - progress) + lam_end * progress

        # --------- smooth‑max ----------
        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)

        # --------- variance ----------
        var_val, grad_var = _var_and_grad(a)

        # total gradient of the *loss* L (we minimise L)
        grad_a = grad_smooth + lam * grad_var

        # chain rule through the soft‑max (gradient w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter to avoid dead‑ends
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.02, size=n)

        # restart if we are idle for too long
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (same structure, slightly tweaked)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly transfers a small amount of mass from an
    index that contributes heavily to the current maximal convolution entry
    to an index that contributes little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule for maximal transfer fraction and temperature
    max_frac_start, max_frac_end = 0.6, 0.001
    temp_start, temp_end = 0.02, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 80_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each a[i] to this entry (a[i] * a[k_max-i])
        idx = k_max - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[valid] = a[valid] * a[idx[valid]]
        total_contrib = contrib.sum()

        # source index: proportional to contribution
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index: biased towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[valid] = 1.0 / (contrib[valid] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback (any index except i)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # handle stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Fine‑grained projected sub‑gradient polishing on the exact max
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float) -> tuple[np.ndarray, float]:
    """
    A few passes of projected sub‑gradient descent on the exact
    max‑convolution.  Step size is tiny and linearly decayed.
    """
    a = a0.copy()
    cur_val = _objective(a)
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start_time) / (deadline - start_time)
        # step size schedule: start ~1e-5, end ~1e-7 (scaled by 1/(4n))
        step = (1e-5 * (1.0 - progress) + 1e-7 * progress) / (4.0 * n)

        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # gradient of 2*n*max(b) w.r.t a is 4*n * a_rev where a_rev[i] = a[k_max-i]
        grad = np.zeros_like(a)
        idx = k_max - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        grad[valid] = a[idx[valid]]

        # gradient descent step (move opposite to gradient)
        a -= step * grad
        # projection back onto the simplex
        a = _project_simplex(a)

        new_val = _objective(a)
        if new_val < best_val - 1e-12:
            best_val, best_a = new_val, a.copy()
            print(f"[Subgrad {iteration:7d}] new best = {best_val:.6f}")

        cur_val = new_val
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric families – cheap sweeps
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # richer grid
    p_vals = np.concatenate([np.geomspace(0.2, 8.0, 16),
                             np.linspace(0.2, 8.0, 16)])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _window_family(name: str, n: int) -> np.ndarray:
    """Return a standard window shape (unnormalised)."""
    x = np.arange(n, dtype=np.float64)
    if name == "hann":
        w = np.sin(np.pi * (x + 0.5) / n) ** 2
    elif name == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / (n - 1))
    elif name == "blackman":
        w = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
             + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
    else:
        raise ValueError(f"unknown window {name}")
    return w


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape a_i ∝ 1 / sqrt(i·(n‑i+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt(i * (n - i + 1) + eps)
    return a

# Main entry point required by the evaluation framework

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a plain Python list whose sum is (approximately) 1.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8000 entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Stage 0 – initial seeds (cheap parametric sweep + random)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # 0a – try to reuse a previous solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        best_a, best_val = a0, val
        print(f"[init] interpolated previous solution, val={best_val:.6f}")
    except Exception:
        pass

    # 0b – Beta family sweep
    try:
        cand_a, cand_val = _beta_family_best(n)
        if cand_val < best_val:
            best_a, best_val = cand_a, cand_val
            print(f"[init] beta‑family best val={best_val:.6f}")
    except Exception as exc:
        print("[init] beta sweep failed:", exc)

    # 0c – classic windows
    for win in ("hann", "hamming", "blackman"):
        try:
            w = _window_family(win, n)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            val = _objective(w)
            if val < best_val:
                best_a, best_val = w, val
                print(f"[init] {win} window improved val={best_val:.6f}")
        except Exception as exc:
            print("[init] window", win, "failed:", exc)

    # 0d – arcsine‑type shape
    try:
        a_arc = _arcsine_shape(n)
        a_arc = np.maximum(a_arc, 0.0)
        a_arc /= a_arc.sum()
        val = _objective(a_arc)
        if val < best_val:
            best_a, best_val = a_arc, val
            print(f"[init] arcsine shape improved val={best_val:.6f}")
    except Exception as exc:
        print("[init] arcsine failed:", exc)

    # 0e – a few random Dirichlet draws for diversity
    for _ in range(5):
        rand_a = np.random.dirichlet(np.ones(n))
        val = _objective(rand_a)
        if val < best_val:
            best_a, best_val = rand_a, val
            print(f"[init] random Dirichlet improved val={best_val:.6f}")

    # fallback – pure random if everything failed
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] fallback random start, val=", best_val)

    # ------------------------------------------------------------------
    # Stage A – Adam optimisation (large fraction of budget)
    # ------------------------------------------------------------------
    phase_a_end = start_time + 0.70 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_end,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # Stage B – Simulated‑annealing hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a, n,
                                         deadline=deadline,
                                         start_time=phase_a_end,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # Stage C – Fine‑grained sub‑gradient polishing (tiny remaining slice)
    # ------------------------------------------------------------------
    final_deadline = deadline - 0.2  # leave a tiny buffer
    if time.time() < final_deadline:
        best_a, best_val = _subgradient_polish(best_a, n,
                                               deadline=final_deadline,
                                               start_time=time.time(),
                                               best_a=best_a,
                                               best_val=best_val)

    # ------------------------------------------------------------------
    # Final validation via the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
