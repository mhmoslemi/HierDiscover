"""
propose_candidate – analytical “Beta‑shape” step function

The evaluation
    E(a) = 2·n·max(conv(a)) / (∑a)^2
is invariant under scaling, so we may fix ∑a = 1.
For a distribution of the form

    a_i ∝ (p+1)·(1 - i/(n‑1))^p ,  p ≥ 0 ,

the continuous convolution at the centre (where the maximum occurs) can be
computed analytically:

    max(conv) ≈ I(p)/n ,   I(p) = (p+1)^2·B(p+1,p+1),

where B is the beta‑function.
Hence the evaluation is approximately

    E(p) ≈ 2·I(p)   (independent of n)

while the left‑hand endpoint contributes 2·(p+1)^2/n.
The optimum is achieved for a modest integer p (≈5–6 for n≈4000).

The algorithm:

1. Detect the problem size n (use the length of ``height_sequence_1``
   if it exists, otherwise fall back to 4000).
2. Scan a range of exponent values p (both integer and a fine
   fractional grid).  For each p build the vector

        a_i = ((p+1)/n) * (1 - i/(n-1))**p

   and normalise it to sum 1.
3. Evaluate the candidate with a fast FFT‑based convolution
   (``_fast_evaluate``) and keep the best one.
4. Run a short mirror‑descent refinement (exponential updates on the
   simplex) starting from the best candidate – this can capture tiny
   improvements that the analytical family misses.
5. Return the best sequence as a plain Python list.

The whole routine respects a 1000 s budget, prints progress and finally
calls the official ``evaluate_sequence`` once for safety.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – these are deliberately defined at top level
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    # real‑valued FFT (slightly faster than complex FFT)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already sum to 1.
    Returns 2·n·max(conv) (the denominator Σa_i = 1, so the full formula
    simplifies to this expression).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    # Guard against a degenerate (almost‑zero) sum – should never happen.
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _mirror_descent_step(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Perform one exponential‑mirror‑descent update using the exact
    sub‑gradient of max(conv).  The vector is re‑projected onto the simplex.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_k = int(conv.argmax())

    # Sub‑gradient g_i = a_{max_k - i} when indices are valid, else 0.
    g = np.zeros_like(a)
    i_start = max(0, max_k - (n - 1))
    i_end = min(n - 1, max_k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    jdx = max_k - idx
    g[idx] = a[jdx]

    # Exponential update (KL‑mirror descent)
    a_new = a * np.exp(-alpha * g)

    # Projection onto the simplex (ensuring non‑negativity and sum = 1)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        # fallback to uniform if something went wrong
        a_new = np.full(n, 1.0 / n)
        s = 1.0
    a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine uses a parametric Beta‑shape family,
    scans the exponent, and finishes with a short mirror‑descent polish.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Reserve a safety margin for the final call to evaluate_sequence().
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine the problem size n
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    print(f"[info] problem size n = {n}")

    # ------------------------------------------------------------------
    # Scan a family of decreasing power‑law shapes a_i ∝ (p+1)*(1 - i/(n-1))^p
    # ------------------------------------------------------------------
    t = np.arange(n, dtype=float) / (n - 1)           # scaled positions in [0,1]

    best_a = None
    best_val = float('inf')

    # Integer exponents first (quick, give a good baseline)
    integer_ps = list(range(0, 11))                  # 0 … 10
    # Add a finer grid around the expected optimum (≈5–6)
    fine_ps = np.linspace(4.0, 8.0, 81)              # step ≈0.05
    candidate_ps = integer_ps + list(fine_ps)

    for p in candidate_ps:
        if time.time() >= deadline:
            break
        # compute raw (unnormalised) heights
        raw = (p + 1.0) * np.power(1.0 - t, p)
        # Normalise to sum 1 (the analytic factor (p+1)/n is already built‑in,
        # but we normalise numerically to avoid accumulated floating errors)
        a = raw / raw.sum()
        cur_val = _fast_evaluate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[scan] p={p:5.2f} → eval={cur_val:.6f}")

    if best_a is None:
        # Fallback – uniform distribution (should never happen)
        best_a = np.full(n, 1.0 / n)
        best_val = _fast_evaluate(best_a)
        print("[fallback] using uniform distribution")

    print(f"[scan] best after parameter scan: eval={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short mirror‑descent polish (exact max‑convolution gradient)
    # ------------------------------------------------------------------
    a = best_a.copy()
    # step size schedule – start relatively aggressive and decay slowly
    alpha0 = 0.8
    decay = 0.9995
    min_alpha = 0.02
    it = 0
    while time.time() < deadline - 1.0:
        # geometric decay of the step size
        alpha = max(alpha0 * (decay ** it), min_alpha)
        a = _mirror_descent_step(a, alpha)

        # Periodically evaluate
        if it % 50 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[md] it={it:6d} → new best eval={best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator (one call only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
