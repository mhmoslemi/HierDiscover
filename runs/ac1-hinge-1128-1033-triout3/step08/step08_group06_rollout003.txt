"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises ``evaluate_sequence``.

The search consists of several phases within the allocated time budget:

0️⃣  Diverse seed generation – Dirichlet draws with several concentrations,
    power‑law, geometric and two‑level (“ends vs middle”) families.  For
    each seed both the raw vector and its anti‑symmetric re‑ordering are
    evaluated.

1️⃣  Sub‑gradient (mirror) descent on the true L∞‑objective, using an
    active‑set gradient computed with a single FFT.

2️⃣  Smooth‑surrogate optimisation (log‑sum‑exp) via L‑BFGS‑B with a
    geometric ε‑schedule and several random restarts.

3️⃣  Simulated‑annealing hill‑climbing (swap / mass‑transfer moves) for
    final local refinement.

All updates keep the vector on the probability simplex (non‑negative,
unit sum).  The best vector found before the deadline is returned as a
plain Python list of floats.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # eliminate tiny negatives
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest values on the left with the smallest on the right.
    The resulting vector is not symmetric but empirically reduces the
    self‑convolution peak.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # large → left side
        res[n - 1 - i] = s[i]     # small → right side
    if n % 2:
        res[half] = s[half]
    return res


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # self‑convolution
    conv = _full_convolution(a)
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t a
    weights = np.exp(shifted)
    p = weights / weights.sum()          # distribution over convolution indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    seg = conv_ap[n - 1: 2 * n - 1]      # derivative of each a_i
    grad_a = 4.0 * n * seg[::-1]        # ∂L/∂a_i

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

# Sub‑gradient descent on the true L∞‑objective

def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} a_{k‑i} using a single FFT.
    ``active_idx`` contains convolution indices (0 … 2n‑2) belonging
    to the active set.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n-2) - i] for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g

# Simulated‑annealing based hill‑climbing refinement

def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    """
    Random local moves (swap or mass transfer) with an annealing schedule.
    Only strictly better solutions are kept for the reported best,
    but occasional uphill moves are accepted to escape plateaus.
    """
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.02                     # initial “temperature”

    it = 0
    while time.time() < deadline:
        it += 1
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            accept = True
        else:
            T = T0 * (0.9995 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val_local:.6f}")

    return best_a, best_val_local

def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (dist_to_end+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


def _geometric_candidate(r: float, n: int) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _two_level_candidate(L: int, w: float, n: int) -> np.ndarray:
    """
    Two‑level shape:
        - first and last L positions have weight e = w/(2L)
        - the remaining I = n‑2L positions have weight m = (1‑w)/I
    The vector is normalised to sum 1 (and returned on the simplex).
    """
    if L == 0:
        return np.full(n, 1.0 / n)

    I = n - 2 * L
    if I <= 0:
        raise ValueError("L too large for the given n")

    e = w / (2.0 * L)          # weight per “end’’ entry
    m = (1.0 - w) / I          # weight per interior entry

    a = np.full(n, m, dtype=np.float64)
    a[:L] = e
    a[-L:] = e
    # numerical safety
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    return a

            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – diversified cheap candidates (random dirichlet, power‑law,
    # geometric, two‑level).   Stop early if we run out of time.
    # ------------------------------------------------------------------
    def try_update(candidate: np.ndarray, label: str):
        nonlocal best_a, best_val
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 1e-12:
            return
        candidate /= s
        val = evaluate_sequence(candidate.tolist())
        if val < best_val:
            best_val = val
            best_a = candidate.copy()
            print(f"[{label}] new best = {best_val:.6f}")

    # ----- Dirichlet draws with several concentration parameters -----
    dir_alphas = [0.15, 0.25, 0.5, 1.0, 2.0]
    draws_per_alpha = 250
    for alpha in dir_alphas:
        for _ in range(draws_per_alpha):
            if time.time() >= deadline - 250:
                break
            cand = rng.gamma(alpha, 1.0, size=n)
            cand /= cand.sum()
            # raw candidate
            try_update(cand, f"dir(alpha={alpha:.2f})")
            # anti‑symmetric version
            cand_ord = _anti_symmetric_reorder(cand)
            try_update(cand_ord, f"dir(alpha={alpha:.2f})‑reorder")

    # ----- Power‑law shapes -----
    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 2.5, 3.0):
        if time.time() >= deadline - 250:
            break
        cand = _powerlaw_candidate(p, n)
        try_update(cand, f"power‑law(p={p:.1f})")
        cand_ord = _anti_symmetric_reorder(cand)
        try_update(cand_ord, f"power‑law(p={p:.1f})‑reorder")

    # ----- Geometric (exponential) decay -----
    for r in (0.95, 0.96, 0.97, 0.98, 0.99, 0.992, 0.995, 0.998):
        if time.time() >= deadline - 250:
            break
        cand = _geometric_candidate(r, n)
        try_update(cand, f"geo(r={r:.3f})")
        cand_ord = _anti_symmetric_reorder(cand)
        try_update(cand_ord, f"geo(r={r:.3f})‑reorder")

    # ----- Two‑level “ends vs middle’’ families -----
    L_values = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    w_grid = np.linspace(0.0, 1.0, 21)   # total mass allocated to the ends
    for L in L_values:
        if 2 * L >= n:
            continue
        for w in w_grid:
            if time.time() >= deadline - 250:
                break
            cand = _two_level_candidate(L, w, n)
            try_update(cand, f"two‑level(L={L},w={w:.2f})")
            cand_ord = _anti_symmetric_reorder(cand)
            try_update(cand_ord, f"two‑level(L={L},w={w:.2f})‑reorder")

    # ------------------------------------------------------------------
    # Stage 1 – sub‑gradient descent on the true max‑convolution objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:
        a_sgd, val_sgd = _subgradient_descent_multi(best_a, deadline)
        if val_sgd < best_val:
            best_val, best_a = val_sgd, a_sgd.copy()
            print(f"[sub‑gradient] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth surrogate optimisation (L‑BFGS) with several restarts
    # ------------------------------------------------------------------
    restarts = 0
    max_restarts = 12
    while time.time() < deadline - 120 and restarts < max_restarts:
        restarts += 1
        # random perturbation of the soft‑max parameters
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.3, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[restart {restarts}] L‑BFGS improved to {best_val:.6f}")

        # final anti‑symmetric sweep after each restart
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[restart {restarts}] reorder improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final refinement via simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_simulated_anneal(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
