"""
Propose a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`.  The search is split in five short phases that fit
into the 1000 s budget:

1. **Diversified initialization** – several hand‑crafted candidates
   (uniform, Dirichlet draws, sinusoid, possibly a previous solution).

2. **Aggressive L‑BFGS on a log‑sum‑exp surrogate** – the smooth approximation
   of the max‑convolution is minimised many times with decreasing smoothness.
   The variables are the soft‑max parameters, guaranteeing a feasible point
   (non‑negative, sum = 1).

3. **AdaGrad‑style multiplicative mirror descent** – an adaptive
   sub‑gradient method on the exact max‑convolution.  The per‑coordinate step
   shrinks with the accumulated squared sub‑gradient, allowing fine
   adjustments near a local optimum.

4. **Targeted pairwise mass‑transfer hill‑climbing** – repeatedly moves a
   small fraction of mass from the pair that contributes most to the
   maximal convolution entry toward a low‑mass location.  A slow
   simulated‑annealing schedule accepts occasional uphill moves.

5. **Permutation hill‑climbing** – keeps the multiset of heights fixed but
   searches for a better ordering by many random swaps (with a best‑of‑K
   trial).  This directly reduces peaks in the autocorrelation.

A final cheap stochastic polishing (swaps / tiny mass transfers) is applied,
then the official `evaluate_sequence` is called once for safety.  All
intermediate vectors are projected onto the probability simplex to keep the
sum exactly 1.

The code is written with top‑level helper functions only, no closures or
lambdas, and respects the time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Clip tiny negatives caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε = 2·n·max(conv) + 2·n·ε·log∑exp((conv‑max)/ε)
    and its gradient w.r.t. the soft‑max parameters w.

    Decision variables: a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # -- soft‑max ---------------------------------------------------------
    w_shift = w - np.max(w)       # stability
    a = np.exp(w_shift)
    a /= a.sum()

    # -- convolution ------------------------------------------------------
    conv = _full_convolution(a)          # length 2·n‑1
    max_c = conv.max()

    # -- smooth max -------------------------------------------------------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # -- gradient w.r.t. a ------------------------------------------------
    # p is the softmax over (conv‑max)/ε
    weights = np.exp(shifted)
    p = weights / weights.sum()           # shape (2·n‑1,)

    # derivative of max(conv) w.r.t. a is the convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment that corresponds to ∂conv/∂a
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # factor 4·n comes from ∂‖a*a‖∞/∂a

    # -- chain rule through the soft‑max ----------------------------------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the full formula
    simplifies to this expression).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Put large values on the left, small ones on the right – often flattens the peak."""
    n = a.shape[0]
    s = np.sort(a)               # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[-1 - i]       # large → left side
        out[n - 1 - i] = s[i]    # small → right side
    if n % 2:
        out[half] = s[half]
    return out

def _phase_lbfgs(best_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Aggressive L‑BFGS on the smooth surrogate.
    Returns the best (a, value) found within the allocated time.
    """
    n = best_a.shape[0]
    w = np.log(best_a + 1e-12)          # initialise soft‑max parameters
    best_val = _fast_evaluate(best_a)
    best_vec = best_a.copy()

    # schedule from 1e‑1 down to 1e‑14 (20 values) + a final ultra‑tight pass
    eps_schedule = np.logspace(-1, -14, num=22)  # includes 1e‑15
    max_restarts = 10
    rng = np.random.default_rng()

    for restart in range(max_restarts):
        if time.time() >= deadline - 30.0:
            break
        # occasional random restart of soft‑max parameters
        if restart % 3 == 0:
            w = rng.normal(size=n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 1500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # back‑transform to probabilities
            ws = w - np.max(w)
            a_candidate = np.exp(ws)
            a_candidate /= a_candidate.sum()

            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_vec = a_candidate.copy()
                print(f"[L‑BFGS restart {restart+1} eps={eps:.1e}] new best = {best_val:.6f}")

        # try a cheap anti‑symmetric reordering after each restart
        cand = _anti_symmetric_reorder(best_vec)
        val = _fast_evaluate(cand)
        if val < best_val - 1e-12:
            best_val = val
            best_vec = cand.copy()
            print(f"[L‑BFGS reorder after restart {restart+1}] improved to {best_val:.6f}")

    return best_vec, best_val


def _phase_adagrad_md(a_start: np.ndarray, cur_best_val: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    AdaGrad‑style mirror descent on the exact max‑convolution.
    Multiplicative updates a_i ← a_i·exp(‑η·g_i) with per‑coordinate
    scaling based on accumulated squared sub‑gradients.
    """
    a = a_start.copy()
    n = a.shape[0]
    G = np.zeros_like(a)          # sum of squares of past sub‑gradients
    base_eta = 0.35
    best_val = cur_best_val
    best_vec = a.copy()
    it = 0

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a: g_i = 2·a_{max_k‑i} (if defined)
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = 2.0 * a[jdx]

        # AdaGrad accumulation
        G += g * g
        scaled_g = g / (np.sqrt(G) + 1e-12)

        # step size diminishes like 1/√t
        eta = base_eta / np.sqrt(it + 1)

        # multiplicative update on the simplex
        a = a * np.exp(-eta * scaled_g)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-12:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        # periodic monitoring
        if it % 250 == 0:
            val = _fast_evaluate(a)
            if val < best_val - 1e-12:
                best_val = val
                best_vec = a.copy()
                print(f"[AdaGrad it {it}] new best = {best_val:.6f}")

        it += 1

    return best_vec, best_val


def _phase_pair_hill(a_start: np.ndarray, cur_best_val: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Targeted pairwise mass‑transfer hill‑climbing with a slow annealing schedule.
    At each step the pair (i,j) that contributes most to the maximal convolution
    entry is identified; a small fraction of mass is moved from the larger of
    the two to a low‑mass location.
    """
    cur_a = a_start.copy()
    cur_val = cur_best_val
    best_a = cur_a.copy()
    best_val = cur_val
    n = cur_a.shape[0]
    rng = np.random.default_rng()
    T = 0.03                     # start temperature
    cooling = 0.99996

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        # indices involved in the maximal convolution entry
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        contrib = cur_a[idx] * cur_a[jdx]

        # pair with the largest contribution
        pair_idx = np.argmax(contrib)
        i = idx[pair_idx]
        j = jdx[pair_idx]

        # donor = the larger of the two
        donor = i if cur_a[i] >= cur_a[j] else j

        # choose a recipient among the 5 % smallest entries
        low_mass = np.where(cur_a < np.percentile(cur_a, 5))[0]
        if low_mass.size == 0:
            recipient = rng.integers(0, n)
        else:
            recipient = rng.choice(low_mass)

        # move a small fraction (up to 6 %) of the donor's mass
        delta = cur_a[donor] * 0.06 * rng.random()
        if delta <= 0:
            continue
        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[recipient] += delta

        # project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            # annealed uphill acceptance
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[pair‑hill] improved to {best_val:.6f}")

        T = max(T * cooling, 1e-7)

    return best_a, best_val


def _phase_permutation_hill(a_start: np.ndarray, cur_best_val: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Permutation‑only hill‑climbing.
    The multiset of heights is kept fixed; we only shuffle their positions.
    At each iteration several random swaps are tried and the best improving
    one is executed.  A small temperature permits occasional uphill moves.
    """

    cur_val = cur_best_val
    best_a = cur_a.copy()
    best_val = cur_val
    n = cur_a.shape[0]
    rng = np.random.default_rng()
    T = 0.02
    cooling = 0.99990
    K = 5                               # try K random swaps per iteration

    while time.time() < deadline:
        proposals = []
        vals = []
        for _ in range(K):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
            val = _fast_evaluate(cand)
            proposals.append((i, j, cand))
            vals.append(val)

        if not vals:
            continue

        min_idx = int(np.argmin(vals))
        new_val = vals[min_idx]
        i, j, cand = proposals[min_idx]

        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = cand
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[perm‑hill] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-7)

    return best_a, best_val


def _refine_hill_climb(a: np.ndarray, best_val: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %). The vector is kept on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.04 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found (as a plain Python list) before the
    allotted time budget expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final safety check
    final_margin = 5.0
    deadline = start_time + budget_s - final_margin

    # ------------------------------------------------------------------
    # Determine problem size n (use a previous construction if available)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # Initialise a few diverse candidates and keep the best.
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # random Dirichlet (α = 1)
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))

    # random Dirichlet (α = 5)
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoidal shape
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # normalise everything onto the simplex
    init_candidates = [c / c.sum() for c in init_candidates]

    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on smooth surrogate
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 250.0
    if time.time() < phase1_deadline:
        best_a, best_val = _phase_lbfgs(best_a, phase1_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – AdaGrad‑style mirror descent on the exact max‑convolution
    # ------------------------------------------------------------------
    phase2_deadline = phase1_deadline + 250.0
    if time.time() < phase2_deadline:
        best_a, best_val = _phase_adagrad_md(best_a, best_val, phase2_deadline)

    # ------------------------------------------------------------------
    # Phase 3 – targeted pairwise mass‑transfer hill‑climbing
    # ------------------------------------------------------------------
    phase3_deadline = phase2_deadline + 300.0
    if time.time() < phase3_deadline:
        best_a, best_val = _phase_pair_hill(best_a, best_val, phase3_deadline)

    # ------------------------------------------------------------------
    # Phase 4 – permutation‑only hill‑climbing
    # ------------------------------------------------------------------
    phase4_deadline = phase3_deadline + 140.0
    if time.time() < phase4_deadline:
        best_a, best_val = _phase_permutation_hill(best_a, best_val, phase4_deadline)

    # ------------------------------------------------------------------
    # Phase 5 – cheap final polishing (swap / tiny mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
