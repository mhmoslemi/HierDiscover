def _annealed_hillclimb(a_start, n, start, deadline, best_a, best_val):
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = best_a.copy()
    best_local_val = best_val

    # parameters
    max_frac_start = 0.5
    max_frac_end = 0.001
    T_start = 0.01 * cur_val   # maybe
    T_end = 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 200000

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = T_start * (1.0 - progress) + T_end * progress

        # choose source i:
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_i = contrib / total_contrib
            i = np.random.choice(n, p=prob_i)
        else:
            i = np.argmax(a)  # fallback

        # choose target j:
        # choose probability inversely proportional to contribution to max
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0:
            prob_j = w / w.sum()
            j = np.random.choice(n, p=prob_j)
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # update convolution incrementally
        b_new = b.copy()
        # shift of a array for convolution update: for each k from i to i+n-1, add -2*delta*a[k-i] etc.
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i+j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # accept with probability exp(-(new_val - cur_val) / T)
            if np.random.rand() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-12:
                best_local_val = new_val
                best_local_a = a.copy()
                print(f"[Anneal {int(time.time()-start):6d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb around best_local_a
            a = best_local_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {int(time.time()-start):6d}] perturb after stagnation")

    return best_local_a

"""
Propose a non‑negative step‑function (probability vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

using a three‑stage meta‑heuristic:

1. **Warm‑start** – generate a diverse parametric family of candidates
   (uniform, exponential, power‑law, sine, triangular, random Dirichlet)
   and pick the best one.

2. **Smooth Adam optimisation** – minimise a log‑sum‑exp smooth
   approximation of the max‑convolution.  The parameters are logits
   `w` (`a = softmax(w)`).  Learning rate and smoothing parameter are
   exponentially annealed; the whole stage consumes ~45 % of the time
   budget.

3. **Annealed hill‑climbing** – repeatedly move a small fraction of mass
   from indices that heavily contribute to the current maximal convolution
   entry to indices that contribute little.  Moves are accepted with a
   Metropolis criterion (temperature annealed to 0) so the search can
   escape local minima.  This stage gets the bulk of the remaining time.

4. **Fine‑grained projected sub‑gradient descent** (as in the reference
   implementation) for a final polish, followed by a tiny convex mixture
   with uniform (and a sine shape) to shave off any remaining peak.

All stages respect the simplex constraint (`a ≥ 0`, `sum(a)=1`) and run
until the allocated wall‑clock budget expires.  The best vector found is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {w ≥ 0, Σ w = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric initialisation – a diverse set of shapes
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising normalised vector and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # uniform
    _try(np.ones(n))

    # exponential decays (left‑hand side)
    for r in (0.99999, 0.99997, 0.99995, 0.9999, 0.9998, 0.9995, 0.999):
        a = r ** np.arange(n, dtype=float)
        _try(a)

    # symmetric exponential (U‑shaped)
    for r in (0.99999, 0.99997, 0.99995, 0.9999, 0.9998, 0.9995):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # symmetric power‑law
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p) + (n - i + 1) ** (-p)
        _try(a)

    # half‑wave sine (zero at the ends)
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _try(a)

    # triangular (peak in the centre, zeros at ends)
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    _try(a)

    # random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(12):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimisation on a log‑sum‑exp smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam on smooth max‑convolution.  Returns the best vector seen."""
    # work in log‑space to keep the simplex automatically
    w = np.log(a0 + 1e-12)          # logits
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.001
    eps_start, eps_end = 0.5, 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    it = 0
    while time.time() < deadline:
        # ----- simplex point from logits -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam  {it:6d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- schedules -----
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max via log‑sum‑exp -----
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()             # soft‑max over all positions

        # gradient of smooth objective w.r.t. a:
        #   d/da 2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)           # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]        # same length as a

        # chain rule through the soft‑max (log‑softmax) parametrisation
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to avoid stagnation
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.006, size=n)

        # restart if no improvement for a long stretch
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam  {it:6d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Annealed hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _annealed_hillclimb(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Metropolis‑style hill‑climbing that moves mass between indices."""
    a = a_start.copy()
    b = _full_convolution(a)                # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    best_local_a = best_a.copy()
    best_local_val = best_val

    max_frac_start, max_frac_end = 0.6, 0.001
    T_start, T_end = 0.02 * cur_val, 1e-9
    IMPROVE_LIMIT = 200_000
    no_improve = 0

    it = 0
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = T_start * (1.0 - progress) + T_end * progress

        # ---- pick a source index weighted by its contribution to the peak ----
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # ---- pick a target index that currently contributes little ----
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0:
            probs_j = w / w.sum()
            j = np.random.choice(n, p=probs_j)
        else:
            j = (i + np.random.randint(1, n)) % n

        # ---- amount of mass to shift ----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n)) ----
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ---- Metropolis acceptance ----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-12:
                best_local_val = new_val
                best_local_a = a.copy()
                print(f"[Hill  {it:6d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb around the best seen so far to escape a flat region
            a = best_local_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill  {it:6d}] perturb after stagnation")

        it += 1

    return best_local_a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (as in the reference)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray,
                       n: int,
                       start: float,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n
    eta = eta0
    decay = 0.9995
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD   {it:6d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step + projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # shake a little and restart step size
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD   {it:6d}] perturb after stagnation")

        it += 1

    return best_local_a, best_local_val

# Small post‑processing: mixing with simple shapes
# ----------------------------------------------------------------------
def _mix_with_shapes(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations with uniform and a half‑wave sine."""
    best_a, best_val = a, _objective_value(a)

    uniform = np.full(n, 1.0 / n)
    sine = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    sine = np.maximum(sine, 0.0)
    sine /= sine.sum()

    for base, name in [(uniform, "uniform"), (sine, "sine")]:
        for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
            cand = (1.0 - alpha) * a + alpha * base
            cand /= cand.sum()
            val = _objective_value(cand)
            if val < best_val:
                best_val, best_a = val, cand
                print(f"[Mix   ] better with {name} α={alpha:.3f} → {best_val:.6f}")

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given wall‑clock budget.
    Returns a list of non‑negative floats (Σ≈1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few‑ten‑thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise a – use a known construction if present, otherwise fall back
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation on a smooth approximation of max(conv)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        a0, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase B – Annealed hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.80 * (deadline - start_time)
    best_a = _annealed_hillclimb(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )
    best_val = _objective_value(best_a)

    # ------------------------------------------------------------------
    # Phase C – Exact projected sub‑gradient descent (fine polish)
    # ------------------------------------------------------------------
    phase_c_deadline = deadline
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Final polishing – tiny mixture with uniform / sine
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_shapes(best_a, n)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
