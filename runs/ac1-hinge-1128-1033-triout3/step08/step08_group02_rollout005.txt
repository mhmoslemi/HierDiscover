def gradient(a, gamma):
    conv = conv_full(a)  # length 2n-1
    # compute weights
    w = np.exp(gamma * conv)
    w = w / w.sum()
    # correlation
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1) - 1).bit_length()
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a_rev, size)
    corr_full = np.fft.irfft(fw * fa_rev, size)[:len(w) + len(a_rev) - 1]
    # slice for valid correlation
    n = a.shape[0]
    grad = 2.0 * corr_full[n-1 : n-1 + n]
    return grad
```

    maxc = conv.max()

"""
Step‑function optimizer for `evaluate_sequence`.

Approach
--------
1️⃣  Start with a cheap family sweep to obtain a decent vector (≈2000‑entries).
2️⃣  Smooth “log‑sum‑exp” gradient descent:
       * Use a smoothed max:  Lγ(c)= (1/γ)·log Σ exp(γ·c_i).
       * Gradient can be computed analytically → a correlation of the
         weight vector with the current a.
       * Multiplicative (exponentiated) updates keep non‑negativity.
       * γ is increased gradually (continuation) while the step size η
         decays – this helps escape shallow local minima.
3️⃣  Upsample the solution to a larger dimension (8000) and continue the
    smooth descent.
4️⃣  Finish with a deterministic greedy hill‑climb (mass‑transfer) and a
    tiny uniform mixing, exactly as in the reference implementation.
5️⃣  Return the best sequence found before the time budget expires.

The algorithm respects the 1000 s budget, works only with NumPy (no I/O)
and prints progress messages for debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalized vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Simple families – quick initial sweep
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent start vector and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(12345)
    idx = np.arange(n)

    # deterministic families
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # U‑shape linear
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # random Dirichlet draws
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp gradient step (multiplicative update)
# ----------------------------------------------------------------------
def _smooth_update(a: np.ndarray, eta: float, gamma: float) -> np.ndarray:
    """
    One iteration of the smoothed gradient descent.
    - η : step size (positive)
    - γ : smoothing parameter (larger → closer to true max)
    """
    n = a.shape[0]
    L = 2 * n - 1

    # 1. full auto‑convolution a*a
    size = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:L]

    # 2. soft‑max weights w_i ∝ exp(γ·conv_i)
    maxc = conv.max()
    w_raw = np.exp(gamma * (conv - maxc))          # shift to avoid overflow
    w = w_raw / w_raw.sum()

    # 3. gradient = 2 * correlation(w, a)
    a_rev = a[::-1]
    size2 = 1 << ((L + n - 1) - 1).bit_length()
    fw = np.fft.rfft(w, size2)
    fa_rev = np.fft.rfft(a_rev, size2)
    corr_full = np.fft.irfft(fw * fa_rev, size2)[:L + n - 1]
    grad = 2.0 * corr_full[n - 1:n - 1 + n]       # length n

    # 4. multiplicative update and renormalisation
    a = a * np.exp(-eta * grad)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent (continuation on γ)
# ----------------------------------------------------------------------
def _smooth_expgrad_phase(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Runs many smooth‑gradient steps while slowly increasing γ
    (from ≈0.5 to ≈20) and decaying η.
    """
    a = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 150_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # schedule based on overall progress of this phase
        phase_progress = (now - start) / (deadline - start)
        eta = (1.0 - phase_progress) * 0.4 + phase_progress * 0.001   # 0.4 → 0.001
        gamma = 0.5 + phase_progress * 20.0                           # 0.5 → 20.5

        a = _smooth_update(a, eta, gamma)

        cur_val = _objective(a, n)
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SmoothExpGrad {it:07d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        if no_improve > IMPATIENT:
            # tiny random perturbation to escape a flat region
            noise = np.random.normal(scale=0.006, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            no_improve = 0
            print(f"[SmoothExpGrad {it:07d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climbing (as in the reference code)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Move a small fraction of mass from the “worst” to the “best” index."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:07d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule the transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # location of the maximal convolution entry
        k = int(np.argmax(b))

        # reverse‑relevant slice of a (a_{k‑i})
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        i = int(np.argmax(rev))                     # source index
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))                 # target index

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:07d}] jitter after stagnation")
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # keep the move (cur already updated)
            pass
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val

def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test convex combinations a' = (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a non‑negative step function under the given time budget.
    Returns the best list of floats (Σa = 1) for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 1️⃣  Small‑scale initial search (n≈2000)
    # ------------------------------------------------------------------
    n_small = 2000
    a_small, _ = _initial_candidate(n_small)
    best_a = a_small.copy()
    best_val = _objective(best_a, n_small)

    # ------------------------------------------------------------------
    # 2️⃣  Smooth exponentiated‑gradient descent on the small problem
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * total_time
    best_a, best_val = _smooth_expgrad_phase(best_a, n_small,
                                            start=start_time,
                                            deadline=phase1_end,
                                            best_a=best_a,
                                            best_val=best_val,
                                            target_val=1.503)

    # ------------------------------------------------------------------
    # 3️⃣  Upsample to a larger dimension (n≈8000) – more degrees of freedom
    # ------------------------------------------------------------------
    n_large = 8000
    xs_old = np.linspace(0.0, 1.0, n_small)
    xs_new = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(xs_new, xs_old, best_a)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    best_a = a_large
    best_val = _objective(best_a, n_large)
    print(f"[Upsample] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Smooth descent on the large problem
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * total_time
    best_a, best_val = _smooth_expgrad_phase(best_a, n_large,
                                            start=start_time,
                                            deadline=phase2_end,
                                            best_a=best_a,
                                            best_val=best_val,
                                            target_val=1.503)

    # ------------------------------------------------------------------
    # 5️⃣  Deterministic greedy hill‑climbing for fine tuning
    # ------------------------------------------------------------------
    # leave a small amount of time for the final mixing & verification
    hill_end = deadline - 0.20
    best_a, best_val = _greedy_hillclimb(best_a, n_large,
                                         start=phase2_end,
                                         deadline=hill_end,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # 6️⃣  Tiny uniform mixing (often gives a last dip)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n_large)

    # ------------------------------------------------------------------
    # Final verification via the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
