"""
Hybrid optimiser for the step‑function problem
------------------------------------------------
The objective is   2·n·max(conv(a,a)) / (∑a)²      (a ≥ 0).
All our candidates are kept on the probability simplex (∑a = 1),
hence we minimise   2·n·max(conv(a,a)).

The optimiser consists of three complementary stages, each
respecting the global time budget (≈1000 s):

1. **Adam on a smoothed max** – log‑sum‑exp of the convolution,
   with an annealing ε and a decreasing learning rate.
2. **Lp‑gradient descent** – minimise Σ(conv(a,a)^p) for a
   sequence of increasing exponents p (p≈2…32).  For p→∞ the
   objective approaches the true max, but the gradient is cheap
   (two FFT‑convolutions per iteration) and becomes increasingly
   focused on the worst entry.
3. **Fine‑grained hill‑climbing** – move a tiny fraction of mass
   from indices that contribute heavily to the current maximal
   convolution entry to indices that hardly contribute.

A short final sweep mixing a tiny amount of the uniform vector
often reduces the peak a little more.

All operations use only NumPy and run on a single CPU.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex {w ≥ 0, Σw = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective utilities (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalized vector."""
    n = a.shape[0]
    conv = _full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric / random initialisers (Σ = 1)
# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator):
    """Yield a few diverse starting vectors (already normalised)."""
    # uniform
    yield np.full(n, 1.0 / n)

    # half‑wave sine (zero at ends, peak in the middle)
    x = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    yield x / x.sum()

    # triangular (peak in centre, zero at the ends)
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i)
    yield tri / tri.sum()

    # exponential decay (left‑hand side)
    for r in (0.9999, 0.9995, 0.999):
        exp = r ** np.arange(n, dtype=float)
        yield exp / exp.sum()

    # symmetric exponential (U‑shaped)
    for r in (0.9999, 0.9995, 0.999):
        i = np.arange(n, dtype=float)
        u = r ** i + r ** (n - 1 - i)
        yield u / u.sum()

    # random Dirichlet draws (concentration = 0.5 … 5)
    for alpha in (0.5, 1.0, 2.0, 5.0):
        a = rng.dirichlet(np.full(n, alpha))
        yield a


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth log‑sum‑exp approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int,
                start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected Adam optimiser on a soft‑max of the convolution."""
    # work in logits to keep the simplex constraint exact
    w = np.log(a0 + 1e-15)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.18, 0.002      # a bit more aggressive than before
    eps_start, eps_end = 1e-1, 1e-14    # final epsilon a touch smaller

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    it = 0
    while time.time() < deadline:
        # map logits to simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        conv = _full_convolution_fft(a)
        cur_val = float(2.0 * n * conv.max())

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedules
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        max_c = conv.max()
        shifted = (conv - max_c) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # softmax over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂ = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        grad_a = 4.0 * n * _conv_fft(a, p_rev)[n - 1: 2 * n - 1]

        # chain rule through log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.008, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Lp‑gradient descent (smooth approximation of max)
# ----------------------------------------------------------------------
def _lp_gradient_phase(a0: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected gradient descent on Σ(conv^p) with increasing p."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    # schedule parameters
    lr_start, lr_end = 0.12, 0.001
    p_max = 32                       # final exponent
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 300_000

    while time.time() < deadline:
        # progress ∈ [0,1]
        progress = (time.time() - start) / (deadline - start)
        # learning rate annealing
        lr = lr_start * (lr_end / lr_start) ** progress
        # exponent annealing (integer, grows roughly exponentially)
        p = int(2 ** (1 + 4 * progress))   # 2,4,8,16,32
        if p > p_max:
            p = p_max

        conv = _full_convolution_fft(a)                 # length 2n‑1
        cur_val = float(2.0 * n * conv.max())
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[LpGrad {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        # gradient of Σ(conv^p) w.r.t. a : 2·p·conv(a_rev, conv^{p‑1})
        w = np.power(conv, p - 1)               # (2n‑1,)  – non‑negative
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, w)         # length 3n‑2
        grad = grad_full[n - 1:n - 1 + n]        # slice to length n
        grad = 2.0 * p * grad

        # gradient step + projection onto simplex
        cand = a - lr * grad
        a = _project_simplex(cand)

        # occasional perturbation if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[LpGrad {it:7d}] perturbing after stagnation")
            noise = np.random.normal(scale=0.008, size=n)
            a = _project_simplex(best_local_a + noise)
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase C – Fine‑grained hill‑climbing on the exact max‑convolution
# ----------------------------------------------------------------------
def _hillclimb_fine(a_start: np.ndarray, n: int,
                    start: float, deadline: float,
                    best_a: np.ndarray, best_val: float,
                    target_val: float = 1.503) -> np.ndarray:
    """Move small mass‑chunks from high‑contrib indices to low‑contrib ones."""
    a = a_start.copy()
    b = _full_convolution_fft(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.2, 0.00001   # finer than before
    IMPROVE_LIMIT = 400_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # index of current worst convolution entry
        k_max = int(np.argmax(b))

        # contribution of each position to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index: proportional to its contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – preferably far from k_max‑pairing
        if np.random.rand() < 0.45:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n) update)
        b_new = b.copy()
        # add/subtract 2·δ·a to the convolution slices
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # correct the quadratic terms at the exact positions
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # restart‑like perturbation if stagnating for a while
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill {it:7d}] perturbation after stagnation")
            a = best_a.copy()
            noise = np.random.normal(scale=0.006, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution_fft(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Small post‑processing – mixing with a tiny amount of uniform mass
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few convex combinations (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.018):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the given time budget.
    Returns a list of floats (Σ = 1) that (hopefully) beats the target.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 8500

    # ------------------------------------------------------------------
    # Initialise a – use a stored construction if available, else a sweep.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        # pick the best among a handful of parametric families
        best_a = None
        best_val = np.inf
        for cand in _initial_candidates(n, rng):
            val = _objective_value(cand)
            if val < best_val:
                best_val, best_a = val, cand
        a0 = best_a

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth approximation of max(conv)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.20 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        a0, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase B – Lp‑gradient descent (p grows from 2 to 32)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * (deadline - start_time)
    best_a, best_val = _lp_gradient_phase(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase C – Fine‑grained hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    phase_c_deadline = deadline
    best_a = _hillclimb_fine(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    best_val = _objective_value(best_a)

    # ------------------------------------------------------------------
    # Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also evaluate)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
