"""
Propose a non‑negative step function that (hopefully) beats the previous
best bound 1.5085.

The algorithm consists of three stages

1️⃣  **Initialisation** – a quick β‑family sweep and (if available) an
    interpolation of the previous best sequence.

2️⃣  **Adam on a smooth max** – minimise the smooth approximation of
    ``2·n·max(conv(a,a))`` (no variance penalty).  The optimisation works
    on logits so that Σa = 1 is satisfied automatically.  The smoothing
    parameter ε is annealed exponentially, ending at ~1e‑9, which makes
    the optimiser focus on the true ℓ∞‑norm.

3️⃣  **Greedy hill‑climbing** – a deterministic local search that always
    moves a small fraction of mass from an index that contributes most
    to the current maximal convolution entry to an index that contributes
    the least.  The step size is linearly annealed to zero.

All heavy lifting (FFT‑based convolutions, smooth‑max, projection onto the
simplex) is performed with NumPy; the expensive ``np.convolve`` call is
only used once at the very end when the final score is reported.
The routine respects the 1000 s budget (a 0.5 s safety margin is kept
for the final call to ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a (the factor 2·n is already included).
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick (weights sum to 1)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted, dtype=np.float64)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries (the relevant half)
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of the smooth max (no variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize_smooth(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          best_a: np.ndarray,
                          best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises   L(a) = 2·n·smooth_max(a)
    using a soft‑max (log‑its) parametrisation (ensures Σa = 1).
    """
    # initialise logits
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.004           # a bit more aggressive than before
    eps_start, eps_end = 0.5, 1e-9           # final ε is essentially exact max

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- logits → probabilities (a) -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()    # simplex

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
        else:
            idle += 1

        # ----- schedules -----
        prog = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # ----- smooth‑max & its gradient -----
        smooth_val, grad = _smooth_max_and_grad(a, eps)

        # ----- chain rule through the soft‑max (log‑its) -----
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation (helps to escape plateaus)
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.017, size=n)

        # restart if stuck
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.12
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climbing (no temperature)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly moves a small fraction of mass from an
    index that contributes most to the current maximal convolution entry
    to an index that contributes the least.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-14

    # annealing for the maximal transfer fraction
    max_frac_start, max_frac_end = 0.55, 0.001

    stagnation = 0
    STAG_LIMIT = 150_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # locate the current maximal convolution index
        k_max = int(np.argmax(b))

        # contributions of each index to this entry: g_i = 2 * a_{k_max - i}
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]          # product (unordered)
        # source index: the one with largest product
        src = int(np.argmax(contrib))
        if a[src] <= eps:
            # nothing to move – random perturbation and restart
            stagnation += 1
            if stagnation > STAG_LIMIT:
                a = _project_simplex(a + np.random.normal(scale=0.03, size=n))
                b = _full_convolution(a)
                cur_val = 2.0 * n * b.max()
                stagnation = 0
            continue

        # target index: pick the index with the *smallest* product
        # (avoid the source itself)
        targ_mask = np.arange(n) != src
        # among those, prefer positions where counterpart a_{k_max - j} is tiny
        idx_t = k_max - np.arange(n)
        mask_t = (idx_t >= 0) & (idx_t < n)
        prod_t = np.full(n, np.inf)
        prod_t[mask_t] = a[idx_t[mask_t]]
        # low counterpart => low contribution to the peak
        # combine with availability (non‑negativity) via weight
        weights = 1.0 / (prod_t + eps)
        weights[src] = 0.0
        # normalise
        total_w = weights.sum()
        if total_w > 0:
            j = int(np.random.choice(n, p=weights / total_w))
        else:
            # fallback: uniform among all except src
            j = np.random.randint(n - 1)
            if j >= src:
                j += 1

        # amount of mass to move
        delta = a[src] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        # the convolution update follows the same pattern as in the previous code
        b_new[j:j + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        # self‑interaction terms
        b_new[2 * j] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept only if we improve (strictly)
        if new_val < cur_val - 1e-14:
            a[src] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            stagnation = 0

            if new_val < best_val - 1e-13:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
        else:
            stagnation += 1
            if stagnation > STAG_LIMIT:
                # perturb to escape a possible plateau
                a = _project_simplex(a + np.random.normal(scale=0.025, size=n))
                b = _full_convolution(a)
                cur_val = 2.0 * n * b.max()
                stagnation = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple β‑family sweep (cheap) – useful for a quick initial guess
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val
