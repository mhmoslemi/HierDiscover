def _optimal_mix_t(conv_f, conv_g, conv_fg, max_iter=30, tol=1e-4):
    lo, hi = 0.0, 1.0
    phi = (np.sqrt(5)-1)/2
    for _ in range(max_iter):
        t1 = lo + (1-phi)*(hi-lo)
        t2 = lo + phi*(hi-lo)
        val1 = np.max((1-t1)**2 * conv_f + 2*t1*(1-t1) * conv_fg + t1**2 * conv_g)
        val2 = np.max((1-t2)**2 * conv_f + 2*t2*(1-t2) * conv_fg + t2**2 * conv_g)
        if val1 < val2:
            hi = t2
        else:
            lo = t1
        if hi - lo < tol:
            break
    t_opt = (lo+hi)/2
    max_val = np.max((1-t_opt)**2 * conv_f + 2*t_opt*(1-t_opt) * conv_fg + t_opt**2 * conv_g)
    return t_opt, max_val
```

"""
Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm (≈1000 s budget)
-------------------------
1. **Seed generation** – build a modest portfolio of diverse candidates
   (uniform, previous best, Dirichlet draws, simple sinusoid, exponential
   and power‑law families).  Keep the best by a cheap fast evaluator.

2. **Smooth surrogate optimisation** – minimise a log‑sum‑exp smoothed
   version of the max‑convolution with L‑BFGS‑B.  A short restart schedule
   (10 restarts, 10 eps‑levels) is used; after each restart an
   anti‑symmetric reordering is tried.

3. **Exact mirror‑descent** – sub‑gradient descent on the true objective
   (max‑convolution) using the exponential mirror map.  The step size is
   slowly decayed.

4. **Mix‑and‑match refinement** – repeatedly generate a nearby candidate
   (random mass transfer or a single mirror step), compute the optimal
   convex‑mixing parameter *t* (golden‑section search on the exact
   max‑convolution of the mixture) and accept the mixture if it improves
   the bound.

5. **Final cheap hill‑climb** – random swaps / mass‑transfers that only
   accept strict improvements.

All phases respect the global time budget; the best vector is finally
checked with the official ``evaluate_sequence`` and returned as a plain
Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (set before each
# optimisation call).
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of power‑of‑2 size.
    Result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Clip tiny negatives that may appear due to round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


def _convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two (possibly different) vectors."""
    n, m = a.shape[0], b.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate   L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    with the soft‑max parametrisation a_i = exp(w_i)/Σ_j exp(w_j).

    Returns the scalar value and the gradient w.r.t. w.
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max -----
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- convolution -----
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max -----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    weights = np.exp(shifted)
    p = weights / weights.sum()      # soft‑max over lag indices

    # convolution of a with reversed p (needed for gradient)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment that aligns with the original a‑vector
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]     # reverse to match ordering of a_i

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return float(L), grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses global ε and n)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the O(n) algorithm from
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that already
    satisfy a_i ≥ 0 and Σ a_i = 1.
    Returns 2·n·max(conv) (the denominator is 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places the largest values to the left side
    and the smallest to the right side, often flattening the peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector is kept
    on the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                      # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def _gradient_at_max(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a: for k = argmax(conv) the gradient
    is g_i = Σ_{j : i+j = k} a_j (the factor 2 is omitted – it does not
    affect the mirror‑descent direction).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_k = int(conv.argmax())
    g = np.zeros_like(a)
    i_start = max(0, max_k - (n - 1))
    i_end = min(n - 1, max_k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    jdx = max_k - idx
    g[idx] = a[jdx]
    return g


def _optimal_mix_t(conv_f: np.ndarray, conv_g: np.ndarray,
                   conv_fg: np.ndarray, max_iter: int = 30,
                   tol: float = 1e-4) -> tuple[float, float]:
    """
    Given the three convolution vectors for f, g and f*g, find the mixing
    coefficient t∈[0,1] minimising max((1‑t)²·conv_f + 2·t·(1‑t)·conv_fg + t²·conv_g).
    Golden‑section search (the objective is unimodal in practice).
    Returns (t_opt, max_convolution_at_t_opt).
    """
    lo, hi = 0.0, 1.0
    phi = (np.sqrt(5) - 1) / 2   # ≈0.618
    for _ in range(max_iter):
        t1 = lo + (1 - phi) * (hi - lo)
        t2 = lo + phi * (hi - lo)

        mix1 = (1 - t1) ** 2 * conv_f + 2 * t1 * (1 - t1) * conv_fg + t1 ** 2 * conv_g
        mix2 = (1 - t2) ** 2 * conv_f + 2 * t2 * (1 - t2) * conv_fg + t2 ** 2 * conv_g

        val1 = mix1.max()
        val2 = mix2.max()

        if val1 < val2:
            hi = t2
        else:
            lo = t1
        if hi - lo < tol:
            break
    t_opt = (lo + hi) / 2.0
    best_mix = (1 - t_opt) ** 2 * conv_f + 2 * t_opt * (1 - t_opt) * conv_fg + t_opt ** 2 * conv_g
    return t_opt, float(best_mix.max())
