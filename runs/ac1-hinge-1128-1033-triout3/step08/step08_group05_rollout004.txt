"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The search uses four stages:

1. **Initial pool** – a few diverse candidates (uniform, old solution,
   Dirichlet draws, a sine shape).

2. **Aggressive L‑BFGS on a smooth surrogate** – we minimise a
   log‑sum‑exp approximation of the maximal self‑convolution.  Several
   random restarts and a fine ε‑schedule are employed.

3. **Multi‑peak mirror‑descent + simulated annealing** – after the
   surrogate optimisation we repeatedly:
   * compute the full convolution,
   * form a sub‑gradient that aggregates the contributions of the
     *K* largest convolution entries,
   * update the probabilities with an exponential (entropy) mirror step,
   * occasionally reorder the vector (anti‑symmetric or alternating‑ends)
   * and finally run a simple annealed mass‑transfer walk that accepts
     uphill moves with a decaying temperature.

4. **Cheap final polish** – a lightweight stochastic hill‑climber that
   tries random swaps and small mass transfers, accepting only strict
   improvements.

All iterates are projected onto the probability simplex (Σa_i = 1,
a_i ≥ 0).  The routine respects the 1000 s budget (a 5 s safety
margin) and finally returns the best sequence found as a plain Python
list.  The official ``evaluate_sequence`` is called once at the end for
a safety check.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
from typing import List

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (set before each call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None  # smoothing parameter ε
_GLOBAL_N = None    # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)   # clip tiny negatives from rounding
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    Decision variables:
        a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ---------- soft‑max ----------
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---------- convolution ----------
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ---------- smooth max ----------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    weights = np.exp(shifted)
    p = weights / weights.sum()      # distribution over lag indices

    # derivative of smooth max w.r.t. each conv entry = p
    # grad_a = 4·n·(p * derivative of conv wrt a)
    # derivative of conv_k = Σ_i a_i·δ_{i + j = k} + Σ_j a_j·δ_{i + j = k}
    # which yields a convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment that corresponds to ∂conv/∂a
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]    # reverse to align with a_i order

    # ---------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the full formula
    simplifies to this expression).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places large values on the left half and small values on the right.
    Often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _alternating_ends_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest element at the left end, the second largest at the right end,
    the third largest next to the left, etc.  This often reduces the peak of the
    autocorrelation.
    """
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]          # descending
    out = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for val in sorted_desc:
        if toggle:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
        toggle = not toggle
    return out


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climber that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %).  The vector is kept on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found (as a plain Python list) before the
    allocated time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a safety margin for the final check
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
        print(f"[info] Using existing sequence of length {n}")
    except Exception:
        n = 4000
        has_old = False
        print(f"[info] No previous sequence – starting from scratch (n={n})")

    # ------------------------------------------------------------------
    # Create a few diverse initial candidates and keep the best.
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # random Dirichlet (α = 1)
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))

    # random Dirichlet (α = 5)
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoidal shape (half‑wave)
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # normalise everything onto the simplex
    init_candidates = [c / c.sum() for c in init_candidates]

    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on a smooth surrogate.
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-1), np.log10(1e-14), num=25)
    w = np.log(best_a + 1e-12)           # initialise soft‑max parameters
    max_restarts = 12

    for restart in range(max_restarts):
        if time.time() >= deadline - 150.0:
            break
        print(f"[L‑BFGS restart {restart + 1}/{max_restarts}]")
        if restart % 3 == 0:
            w = rng.normal(size=n) * 2.0   # full random restart

        for eps in eps_schedule:
            if time.time() >= deadline - 150.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x
            # back‑transform to probabilities
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[L‑BFGS eps={eps:.1e}] new best = {best_val:.6f}")

        # two cheap reorderings after each restart
        for reorder_fn in (_anti_symmetric_reorder, _alternating_ends_reorder):
            a_ord = reorder_fn(best_a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val - 1e-9:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[reorder {reorder_fn.__name__}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multi‑peak mirror descent (exact max‑convolution).
    # ------------------------------------------------------------------
    a = best_a.copy()
    step = 1.0               # initial mirror‑descent step size
    min_step = 0.001
    decay = 0.9997
    topK = 5                 # number of largest convolution entries to aggregate

    while time.time() < deadline - 200.0:
        conv = _full_convolution(a)
        # indices of the K largest entries
        idxs = np.argpartition(-conv, topK)[:topK]

        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            g[i_range] += a[j_range]

        # exponential mirror‑descent update
        a = a * np.exp(-step * g)
        a = _project_simplex(a)

        cur_val = _fast_evaluate(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[MD] new best = {best_val:.6f}")

        # occasional re‑ordering to shake the structure
        if int(time.time() * 10) % 1000 == 0:  # very cheap periodic check
            for reorder_fn in (_anti_symmetric_reorder, _alternating_ends_reorder):
                a_ord = reorder_fn(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val - 1e-12:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[MD reorder {reorder_fn.__name__}] improved to {best_val:.6f}")

        step = max(step * decay, min_step)

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing mass‑transfer hill climbing.
    # ------------------------------------------------------------------
    T = 0.02                # initial temperature
    cooling = 0.99995
    cur_val = best_val
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        donor, recipient = (i, j) if a[i] >= a[j] else (j, i)
        if a[donor] <= 1e-15:
            continue
        delta = a[donor] * 0.03 * rng.random()   # up to 3 % of donor mass
        new_a = a.copy()
        new_a[donor] -= delta
        new_a[recipient] += delta
        new_a = _project_simplex(new_a)

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[anneal] improved to {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # Phase 4 – final cheap stochastic polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
