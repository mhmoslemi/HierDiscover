"""
Propose a non‑negative step function minimising `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – start from a uniform distribution (or from the
   historic `height_sequence_1` if it exists) and normalise it to sum = 1.
2. **Soft‑max exponential mirror descent** – at each iteration the full
   self‑convolution `c = a*a` is computed.  A soft‑max over the
   convolution entries (temperature β) yields weights `w` that concentrate
   on the largest peaks.  The gradient of the weighted sum has a closed
   form  
       g_i = 2·∑_k w_k·a_{k‑i},
   which is a single correlation (`np.correlate`).  The coefficients are
   updated multiplicatively (`a ← a·exp(−α·g)`) and projected onto the
   probability simplex.  α decays and β grows during the run, so the
   method first smooths the whole distribution and later focuses on the
   remaining peaks.
3. **Occasional anti‑symmetric re‑ordering** – after a successful
   improvement the vector is reordered by pairing large and small values
   (rearrangement inequality) which often reduces the maximal
   convolution.
4. **Hill‑climbing refinement** – a lightweight local search (random
   swaps and tiny mass transfers) finishes the optimisation, keeping only
   strict improvements.
5. The best vector seen so far is returned.  All heavy work is done
   without calling `evaluate_sequence`; the true objective is queried only
   sparingly for logging and final verification.

The whole routine respects the 1000 s time budget and makes use of only
NumPy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding in FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length = 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)   # clip tiny negatives
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    s = np.sort(a)                     # ascending
    n = s.shape[0]
    b = np.empty_like(s)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # big → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _correlation_gradient(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i}.

    This is exactly twice the “valid” correlation of w with a.
    """
    # `np.correlate` returns (w * a)[lag] = Σ_i w[i+lag]·a[i] for lag >=0.
    # For lag = i (0 ≤ i < n) we obtain Σ_j w[i + j]·a[j] = Σ_k w_k·a_{k‑i}.
    return 2.0 * np.correlate(w, a, mode='valid')


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Random swaps / tiny mass transfers, accepting only strict improvements."""
    cur_a = a.copy()
    cur_val = best_val
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:          # safety – evaluate_sequence may reject bad input
            continue
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a list of non‑negative heights that minimises
    `evaluate_sequence` within the supplied time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a few seconds safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    # Use the historic construction if it exists, otherwise fall back to
    # a uniform distribution of length 4000 (FFT‑friendly).
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        a = old.copy()
        print(f"[init] using historic sequence of length {n}")
    except Exception:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=float)
        print(f"[init] starting from uniform of length {n}")

    # make sure we are on the simplex (scale invariance of the objective)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    # --------------------------------------------------------------
    # Keep the best solution seen so far
    # --------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[baseline] evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Soft‑max exponential mirror descent
    # --------------------------------------------------------------
    # Hyper‑parameters (tuned empirically)
    alpha = 2.0          # initial step size (multiplicative update)
    beta = 0.2           # temperature for the soft‑max (low → smooth)
    decay_alpha = 0.9999   # per‑iteration decay of α
    grow_beta = 1.0005     # per‑iteration increase of β (capped below)

    it = 0
    while time.time() < deadline - 5.0:
        # --- convolution and weighting ---------------------------------
        conv = _full_convolution(a)
        max_conv = conv.max()

        # soft‑max weights (stable: subtract the max)
        shifted = conv - max_conv          # max entry becomes 0
        # exponentials stay ≤ 1, so no overflow
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the weighted sum of convolution peaks ---------
        grad = _correlation_gradient(w, a)   # shape (n,)

        # --- multiplicative mirror descent update ----------------------
        delta = -alpha * grad
        delta = np.clip(delta, -20.0, 20.0)   # avoid overflow in exp
        a = a * np.exp(delta)

        # projection onto the probability simplex (keeps positivity)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # --------------------------------------------------------------
        # Logging & occasional re‑ordering
        # --------------------------------------------------------------
        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[gd iter {it}] new best = {best_val:.6f} (max conv {max_conv:.8e})")
            else:
                print(f"[gd iter {it}] val = {cur_val:.6f}  max_conv = {max_conv:.8e}")

            # every 1000 iterations try the anti‑symmetric shuffle
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[reorder] improved to {best_val:.6f}")

        # --------------------------------------------------------------
        # Update schedules
        # --------------------------------------------------------------
        alpha *= decay_alpha
        beta = min(beta * grow_beta, 10.0)   # cap β to keep numbers stable

        it += 1

    # --------------------------------------------------------------
    # Final hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # Final evaluation & return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
