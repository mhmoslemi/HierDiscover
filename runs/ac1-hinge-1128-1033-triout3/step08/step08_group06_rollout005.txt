"""
propose_candidate.py

A self‑contained optimizer for the “max‑self‑convolution” problem.
It combines cheap structured initialisations, a long
exponentiated (mirror) sub‑gradient descent that focuses on the
tightest convolution peaks, a smooth‑surrogate L‑BFGS optimiser,

directly on the probability simplex (∑a_i = 1, a_i ≥ 0) and
periodically applies the anti‑symmetric re‑ordering heuristic
(which tends to push large coefficients far apart and lowers the
maximum of the self‑convolution).  The code is written to stay
well within the 1000 s time budget and strives to push the final
evaluation below the target 1.503.

The public entry point is ``propose_candidate`` – the evaluation harness
will call it and expects a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global state for the smooth surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter ε
_GLOBAL_N = None     # length of the vector currently optimised


# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of a real vector with itself using FFT.
    Returns a vector of length 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # numerical noise may produce tiny negative values – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest numbers on the left and the smallest on the right,
    i.e. pair a large coefficient with a small one when the sequence is
    reversed.  This often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # large → left side
        res[n - 1 - i] = s[i]     # small → right side
    if n % 2:
        res[half] = s[half]
    return res


# ----------------------------------------------------------------------
# Fast objective (identical to evaluate_sequence but without the safety
# checks) – used for every internal iteration.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Compute 2·n·max(conv) / (sum(a)^2) for a non‑negative vector a.
    The function is invariant to scaling, so we keep a on the simplex.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS.
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. the logits w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (soft‑max – guarantees a_i ≥ 0,
                                      Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # ----- soft‑max (numerically stable) -----
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- self‑convolution -----
    conv = _full_convolution(a)
    max_conv = conv.max()

    # ----- smooth max via log‑sum‑exp -----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    weights = np.exp(shifted)            # proportional to exp((c‑max)/ε)
    p = weights / weights.sum()          # softmax over convolution indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries give ∂L/∂a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # ∂L/∂a_i

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)          # ∂L/∂w_i
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Thin wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters
    ``w_init``.  Returns (best_a, best_val) seen before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule: geometric decay from 5·10⁻³ down to 10⁻¹²
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=11)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 25.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 800,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # back‑to‑probability‑simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric re‑ordering heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = _objective(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback – should never happen
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = _objective(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞‑objective.
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} a_{k‑i} using a single FFT.
    ``active_idx`` holds convolution indices (0 … 2n‑2) that belong to
    the active set.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n‑2) – i] for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent(a_start: np.ndarray, deadline: float):
    """
    Exponentiated mirror‑descent that reduces the *active* convolution
    peaks.  The active set is forced to be a singleton (the true maximum)
    after a brief warm‑up, which gives a much sharper sub‑gradient.
    Returns (best_a, best_val) discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _objective(a)
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 1.0          # larger step‑size than the original code
    rel_tol = 0.05        # starts generous, will be forced to zero shortly

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()
        # After a few thousand iterations we lock onto the true maximum
        if it > 2000:
            rel_tol = max(rel_tol * 0.5, 1e-9)   # quickly shrink to ~0
        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of the max‑convolution
        g = _active_set_gradient(a, active)

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative (exponentiated) update
        a = a * np.exp(-alpha * g)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # periodic logging / re‑ordering
        if it % 200 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")
            if it % 4000 == 0:
                a = _anti_symmetric_reorder(a)
                # evaluate after re‑ordering
                cur_val = _objective(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[sgd reorder {it}] improved to {best_val:.6f}")

        # early stop if we have already crossed the target
        if best_val < 1.503:
            print("[sgd] target reached – exiting early")
            break

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, cur_val: float,
                             deadline: float, n: int):
    """
    Classic hill‑climbing with occasional uphill moves.  The schedule
    is deliberately lazy – we only accept a move if it improves the
    objective or if a temperature‑based Metropolis criterion is met.
    """
    cur_a = a.copy()
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    T0 = 0.02                # initial temperature
    it = 0

    while time.time() < deadline:
        it += 1
        # ----- propose a move ------------------------------------------------
        if rng.random() < 0.2:            # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                              # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.07 * rng.random()   # 7 % of source mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # ----- re‑project onto simplex ----------------------------------------
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _objective(new_a)

        if val < cur_val:
            accept = True
        else:
            T = T0 * (0.9995 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val - 1e-12:
                best_val = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Miscellaneous candidate generators (Stage 0)
# ----------------------------------------------------------------------
def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (dist_to_end+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


def _beta_candidate(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on [0,1] (sorted to the anti‑symmetric order).
    """
    x = (np.arange(n) + 0.5) / n
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a = np.nan_to_num(a, nan=0.0)  # safety for α<1 or β<1
    a = _anti_symmetric_reorder(a)  # encourage anti‑symmetry immediately
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the historical size (8000) unless a different
    # existing solution is available.
    # ------------------------------------------------------------------
    default_n = 8000
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        if n != default_n:
            # interpolate to default_n (keeps the shape)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, default_n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        n = default_n
        a = np.full(n, 1.0 / n)

    # Ensure a is a proper probability vector
    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 1e-12 else np.full(n, 1.0 / n)

    best_a = a.copy()
    best_val = _objective(best_a)
    print(f"[init] n={n}, objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – cheap diverse initial candidates
    # ------------------------------------------------------------------
    # (i) random Dirichlet vectors
    for tr in range(5):
        if time.time() >= deadline - 250:
            break
        cand = rng.gamma(shape=1.0, scale=1.0, size=n)
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # (ii) symmetric power‑law shapes
    for p in (0.3, 0.5, 0.7, 0.9, 1.1):
        if time.time() >= deadline - 250:
            break
        cand = _powerlaw_candidate(p, n)
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[power‑law p={p:.1f}] new best = {best_val:.6f}")

    # (iii) Beta‑like candidates
    for α, β in ((0.5, 0.5), (0.8, 0.8), (0.5, 2.0), (2.0, 0.5)):
        if time.time() >= deadline - 250:
            break
        cand = _beta_candidate(α, β, n)
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[beta α={α:.1f} β={β:.1f}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – sub‑gradient descent (long run, focuses on true max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:
        a_sgd, val_sgd = _subgradient_descent(best_a, deadline)
        if val_sgd < best_val:
            best_val, best_a = val_sgd, a_sgd.copy()
            print(f"[sub‑gradient] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth surrogate optimisation (L‑BFGS) with many restarts
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 120 and restarts < 12:
        restarts += 1
        # perturb the current best to create a fresh start
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.3, size=n)          # moderate noise

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[restart {restarts}] L‑BFGS improved to {best_val:.6f}")

        # one extra anti‑symmetric sweep
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _objective(a_ord)
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[restart {restarts}] reorder improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final refinement via simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_simulated_anneal(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final validation using the provided evaluate_sequence (safety wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
