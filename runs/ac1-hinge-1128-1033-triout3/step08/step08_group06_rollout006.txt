def _alternating_reorder(a):
    s = np.sort(a)[::-1]  # descending
    res = np.empty_like(a)
    left = 0
    right = len(a) - 1
    i = 0
    while left <= right:
        res[left] = s[i]
        i += 1
        left += 1
        if left <= right:
            res[right] = s[i]
            i += 1
            right -= 1
    return res

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # sizes schedule
    sizes = [512, 1024, 2048, 4096, 8000] # ensure last <= maybe 12000
    best_a = None
    best_val = float('inf')
    # initial candidate for smallest size
    for n in sizes:
        if time.time() >= deadline:
            break
        # allocate time budget fraction: give more time later
        time_left = deadline - time.time()
        # determine size-specific budget: proportion to n
        # simple heuristic: allocate proportionally to n: time_target = time_left * (n / sum(sizes))
        # but we can allocate based on remaining sizes ...

        # generate initial candidate
        if best_a is None:
            # start with uniform
            a = np.full(n, 1.0 / n)
        else:
            # upscale
            x_old = np.linspace(0.0, 1.0, best_a.shape[0])
            x_new = np.linspace(0.0, 1.0, n)

    mass‑transfer, three‑way transfer) finishes the search.

All updates keep the vector on the probability simplex (non‑negative,
unit sum).  The routine respects the 1000 s time budget and prints
intermediate values so the progress can be inspected.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = rfft(x, size)
    conv = irfft(fx * fx, size)[:out_len]
    # Eliminate tiny negative floating‑point artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Largest values on the left, smallest on the right (pair large‑small)."""
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # large → left side
        res[n - 1 - i] = s[i]     # small → right side
    if n % 2:
        res[half] = s[half]
    return res


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values alternately at the two ends."""
    n = a.shape[0]
    s = np.sort(a)[::-1]          # descending
    res = np.empty_like(a)
    left = 0
    right = n - 1
    i = 0
    while left <= right:
        res[left] = s[i]
        i += 1
        left += 1
        if left <= right:
            res[right] = s[i]
            i += 1
            right -= 1
    return res


# ----------------------------------------------------------------------
# Power‑law candidate (used in the initial pool)
# ----------------------------------------------------------------------
def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (dist_to_end+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)          (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # stable soft‑max
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)
    max_conv = conv.max()

    shifted = (conv - max_conv) / eps
    weights = np.exp(shifted)
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t a
    p = weights / weights.sum()                       # distribution over convolution indices

    # convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = rfft(a, size)
    fp = rfft(p_rev, size)
    conv_ap = irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries give ∂L/∂a_i (see derivation in the original code)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                      # ∂L/∂a_i

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()
    best_a = None
    best_val = float('inf')

    # ε‑schedule: geometric from 5·10⁻³ down to 1e‑15 (11 points)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-15), num=11)

    for eps in eps_schedule:
        if time.time() >= deadline - 15.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 1200,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to a (probability simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # try the two re‑ordering heuristics
        for reorder_fn, name in ((_anti_symmetric_reorder, "anti‑sym"),
                                 (_alternating_reorder, "alternating")):
            a_ord = reorder_fn(a)
            val_ord = evaluate_sequence(a_ord.tolist())
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[L‑BFGS {name} reorder eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback – should never happen
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())
    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} a_{k‑i} using a single FFT.
    ``active_idx`` contains convolution indices (0 … 2n‑2) belonging
    to the active set.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = rfft(a, size)
    fm = rfft(mask_rev, size)
    conv_full = irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n-2) - i] for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent_multi(a_start: np.ndarray, deadline: float):
    """
    Exponentiated mirror‑descent that reduces the *active* convolution peaks.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 1.0               # a slightly larger initial step‑size
    rel_tol = 0.05             # start with a generous active‑set tolerance

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten the tolerance gradually
        if it > 0 and it % 15000 == 0:
            rel_tol = max(rel_tol * 0.5, 1e-5)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]   # indices of near‑max peaks

        # gradient via FFT (sub‑gradient of the L∞‑norm)
        g = _active_set_gradient(a, active)

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative update
        a = a * np.exp(-alpha * g)

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional logging & re‑ordering
        if it % 250 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")
            if it % 5000 == 0:
                for reorder_fn, name in ((_anti_symmetric_reorder, "anti‑sym"),
                                         (_alternating_reorder, "alternating")):
                    a_ord = reorder_fn(a)
                    val_ord = evaluate_sequence(a_ord.tolist())
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        print(f"[sgd {name} reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing based hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    """
    Random local moves (swap, mass transfer, three‑way split) with an
    annealing schedule.  Only strictly better solutions are saved for the
    reported best, but occasional uphill moves are accepted.
    """
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.02                     # initial “temperature’
    it = 0

    while time.time() < deadline:
        it += 1
        # propose a move -------------------------------------------------
        r = rng.random()
        if r < 0.15:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif r < 0.70:                   # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-16:
                continue
            delta = cur_a[i] * 0.07 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        else:                            # three‑way split
            i, j, k = rng.integers(0, n, size=3)
            if i == j or i == k or cur_a[i] <= 1e-16:
                continue
            delta = cur_a[i] * 0.04 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta * 0.5
            new_a[k] += delta * 0.5

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        # evaluate
        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            accept = True
        else:
            # annealing acceptance probability
            T = T0 * (0.9998 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal iter {it}] new best = {best_val_local:.6f}")

    return best_a, best_val_local


# ----------------------------------------------------------------------
# Helper to generate a Dirichlet‑like sparse candidate
# ----------------------------------------------------------------------
def _sparse_dirichlet_candidate(alpha: float, n: int, rng) -> np.ndarray:
    """Draw from Gamma(alpha,1) and normalise – very sparse when α≪1."""
    g = rng.gamma(shape=alpha, scale=1.0, size=n)
    if g.sum() == 0:
        return np.full(n, 1.0 / n)
    return g / g.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Multi‑resolution schedule – the last element is the final length.
    # ------------------------------------------------------------------
    size_schedule = [512, 1024, 2048, 4096, 8000]

    # ------------------------------------------------------------------
    # Try to reuse a previously found construction if it exists.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # keep the shape of the final size – we will later upscale/downscale.
        a_global = old.copy()
    except Exception:
        a_global = None

    best_a = None
    best_val = float('inf')

    for n in size_schedule:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise at the current resolution
        # --------------------------------------------------------------
        if best_a is None:
            if a_global is not None and a_global.shape[0] == n:
                a = a_global.copy()
            else:
                a = np.full(n, 1.0 / n)
        else:
            # upscale previous best via linear interpolation
            x_old = np.linspace(0.0, 1.0, best_a.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, best_a)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 1e-12 else np.full(n, 1.0 / n)

        # --------------------------------------------------------------
        # Stage 0 – cheap diverse candidates
        # --------------------------------------------------------------
        init_pool = [a.copy(),
                     rng.exponential(scale=1.0, size=n),
                     _powerlaw_candidate(0.5, n),
                     _powerlaw_candidate(0.8, n),
                     _sparse_dirichlet_candidate(0.08, n, rng)]

        for cand in init_pool:
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum == 0:
                cand = np.full(n, 1.0 / n)
            else:
                cand = cand / cand_sum
            val = evaluate_sequence(cand.tolist())
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                print(f"[init size {n}] new best = {best_val:.6f}")

        # --------------------------------------------------------------
        # Stage 1 – sub‑gradient descent on the true L∞‑objective
        # --------------------------------------------------------------
        sub_deadline = time.time() + (deadline - time.time()) * 0.35
        if time.time() < sub_deadline - 5.0:
            a_sgd, val_sgd = _subgradient_descent_multi(best_a, sub_deadline)
            if val_sgd < best_val:
                best_val = val_sgd
                best_a = a_sgd.copy()
                print(f"[sub‑gradient size {n}] improved to {best_val:.6f}")

        # --------------------------------------------------------------
        # Stage 2 – smooth surrogate optimisation (L‑BFGS) with restarts
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 30.0:
            lbfgs_deadline = time.time() + remaining * 0.55
            restarts = 0
            while time.time() < lbfgs_deadline and restarts < 10:
                restarts += 1
                w0 = np.log(best_a + 1e-12)
                w0 += rng.normal(scale=0.35, size=n)
                a_opt, val_opt = _l_bfgs_optimize_from_w(w0, lbfgs_deadline)
                if val_opt < best_val:
                    best_val = val_opt
                    best_a = a_opt.copy()
                    print(f"[L‑BFGS restart {restarts} size {n}] best = {best_val:.6f}")

                # one extra re‑ordering sweep
                for reorder_fn, name in ((_anti_symmetric_reorder, "anti‑sym"),
                                         (_alternating_reorder, "alternating")):
                    a_ord = reorder_fn(best_a)
                    val_ord = evaluate_sequence(a_ord.tolist())
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        print(f"[{name} reorder after L‑BFGS size {n}] improved to {best_val:.6f}")

        # --------------------------------------------------------------
        # Stage 3 – final refinement via simulated annealing
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            best_a, best_val = _refine_simulated_anneal(best_a, best_val, deadline, n)
            print(f"[anneal final size {n}] best = {best_val:.6f}")

        # Prepare for the next resolution (if any)
        # The best vector is kept and will be interpolated in the next loop.
        # ------------------------------------------------------------------
        # (optional) apply a quick re‑ordering after each resolution
        # ------------------------------------------------------------------
        for reorder_fn, name in ((_anti_symmetric_reorder, "anti‑sym"),
                                 (_alternating_reorder, "alternating")):
            a_tmp = reorder_fn(best_a)
            val_tmp = evaluate_sequence(a_tmp.tolist())
            if val_tmp < best_val:
                best_val = val_tmp
                best_a = a_tmp.copy()
                print(f"[{name} post‑resolution reorder size {n}] improved to {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Final evaluation & return
    # ----------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    return best_a.tolist()
