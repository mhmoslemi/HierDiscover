"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a,a)) / (sum(a)^2)

where `conv` is the linear convolution of the vector with itself.
The objective is homogeneous; only the *shape* of the vector matters.
The best possible value for a length‑n vector is (2n)/(2n‑1)≈1, which is
approached by making the self‑convolution as flat as possible.

The algorithm consists of four quick stages (≤1000 s total):

  * Stage 0 – generate a small pool of diverse deterministic / random
    candidates (uniform, exponential, power–law, inverse‑sqrt).  Each
    candidate and its anti‑symmetric re‑ordering are evaluated and the
    best kept.

  * Stage 1 – exponentiated mirror‑descent (sub‑gradient) on the true
    L∞–objective.  Active‑set tolerance tightens over time; the step‑size
    decays as 1/√t.  The best point seen is retained.

  * Stage 2 – smooth surrogate optimisation (log‑sum‑exp) with L‑BFGS‑B.
    A schedule of smoothing parameters ε from 5·10⁻³ down to 1·10⁻¹² is used.
    Several random restarts from the current best point are performed.

  * Stage 3 – simulated‑annealing hill‑climbing (random swaps / mass
    transfers).  Occasionally uphill moves are accepted to escape plateaus.

All updates keep the vector on the probability simplex
(non‑negative, unit sum).  The final best vector is returned as a plain
Python list of floats.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative artefacts caused by floating point error
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair large values with small ones:
    largest half → left side, smallest half → right side.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # large → left
        res[n - 1 - i] = s[i]     # small → right
    if n % 2:
        res[half] = s[half]
    return res


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed L∞ loss Lε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)      (ensures a_i ≥ 0, Σ a_i = 1)

    Lε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # self‑convolution
    conv = _full_convolution(a)
    max_conv = conv.max()

    # smoothed max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # p_k over convolution idx

    # convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries give ∂L/∂a_i (see paper for derivation)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                    # ∂L/∂a_i

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper for ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule (geometric)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=11)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 800,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # try the anti‑symmetric re‑ordering heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback – should never happen
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the current active set.
    Returns g_i = Σ_{k∈active} a_{k‑i} (computed via a single FFT).
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n-2) - i]
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent_multi(a_start: np.ndarray, deadline: float):
    """
    Exponentiated mirror descent that reduces the *active* convolution peaks.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.8                     # initial step‑size factor
    rel_tol = 0.05                    # start with generous active‑set tolerance

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten the tolerance gradually
        if it > 0 and it % 20000 == 0:
            rel_tol = max(rel_tol * 0.5, 1e-5)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient via FFT
        g = _active_set_gradient(a, active)

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative update
        a = a * np.exp(-alpha * g)

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional logging & re‑ordering
        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")
            if it % 5000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[sgd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing based hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    """
    Random local moves (swap or mass transfer) with a simple annealing schedule.
    Only strictly better solutions are kept for the reported best,
    but occasional uphill moves are accepted to escape plateaus.
    """
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.02
    it = 0

    while time.time() < deadline:
        it += 1
        # propose a move
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            accept = True
        else:
            # annealing acceptance
            T = T0 * (0.9995 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val_local:.6f}")

    return best_a, best_val_local

def _uniform_candidate(n: int) -> np.ndarray:
    """Uniform distribution (reference)."""

def _exponential_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """Exponential random vector, normalised to sum‑one."""
    cand = rng.exponential(scale=1.0, size=n)

def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    a /= a.sum()
    return a


def _inverse_sqrt_candidate(n: int) -> np.ndarray:
    """
    a_i ∝ 1/√(i+½) – the continuous solution of
    f * f ≈ constant on [0,2] (up to the finite‑support effect).
    """
    idx = np.arange(0, n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Length of the vector – 8000 is a good compromise between flexibility
    # and speed for the 1000 s budget.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Try to start from a previously discovered construction if it exists.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 1e-12 else np.full(n, 1.0 / n)

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – generate a handful of cheap, diverse candidates
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(_uniform_candidate(n))

    # a few exponential random vectors
    for _ in range(3):
        candidates.append(_exponential_candidate(rng, n))

    # power‑law shapes
    for p in (0.3, 0.5, 0.7, 1.0, 1.5):
        candidates.append(_powerlaw_candidate(p, n))

    # inverse‑sqrt (theoretically good)
    candidates.append(_inverse_sqrt_candidate(n))

    # evaluate each candidate and its anti‑symmetric re‑ordering
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline - 250:
            break
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[cand {idx}] new best = {best_val:.6f}")

        # anti‑symmetric version
        cand_ord = _anti_symmetric_reorder(cand)
        val_ord = evaluate_sequence(cand_ord.tolist())
        if val_ord < best_val:
            best_val, best_a = val_ord, cand_ord.copy()
            print(f"[cand {idx} reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – sub‑gradient descent on the true max‑convolution objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:
        a_sgd, val_sgd = _subgradient_descent_multi(best_a, deadline)
        if val_sgd < best_val:
            best_val, best_a = val_sgd, a_sgd.copy()
            print(f"[sub‑gradient] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth surrogate optimisation (L‑BFGS) with several restarts
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 120 and restarts < 12:
        restarts += 1
        # perturb the current best for a fresh start
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.25, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[restart {restarts}] L‑BFGS improved to {best_val:.6f}")

        # one extra anti‑symmetric sweep (cheap)
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[restart {restarts}] reorder improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final refinement via simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_simulated_anneal(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
