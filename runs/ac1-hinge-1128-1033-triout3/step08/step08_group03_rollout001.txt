"""
Gradient‑based search with smooth max‑approximation + simulated‑annealing refinement.

The algorithm works on a probability vector `a` (non‑negative, sum = 1).
For a given `a` we compute its self‑convolution `c = a*a`.
Instead of a hard threshold on the top‑k peaks we approximate the
maximum by a weighted sum using a softmax (log‑sum‑exp) style weight

    w_k = exp( β * (c_k - max(c)) )    (β grows over time)

The gradient of the weighted sum w.r.t. `a` is a (scaled) cross‑correlation
between `a` and `w`.  This gradient is obtained with a single FFT, so many
iterations are cheap.  An adaptive step size `η = 0.5 / max(g)` performs a
multiplicative (mirror‑descent) update `a_i ← a_i·exp(-η·g_i)`.  After every
update the vector is projected back onto the probability simplex.

The descent runs for the bulk of the time budget.  Occasionally the
anti‑symmetric re‑ordering (pair largest with smallest) is applied – it
often reduces the autocorrelation peak.

When the time left is small a lightweight simulated‑annealing hill‑climb
does random swaps / small mass transfers and accepts non‑improving moves
with a slowly decreasing temperature.  The best vector found is returned.

The whole routine respects the 1000 s limit, prints progress, and never
produces negative numbers (the security checks of `evaluate_sequence`
therefore never reject the output).
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a real vector with itself (length 2·n‑1).
    Uses an FFT of length the next power‑of‑two for speed.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)      # clip tiny negatives
    return conv


def _gradient_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of Σ_k w_k (a*a)_k w.r.t. a.
    The exact gradient (ignoring the diagonal ½ factor) is
        g_i = 2 * Σ_k w_k a_{k-i}
    which is a cross‑correlation between a and w.
    Implemented via one FFT.
    """
    n = a.shape[0]
    conv_len = w.shape[0]                # = 2*n‑1
    w_rev = w[::-1]
    size = _next_pow2(n + conv_len - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[: n + conv_len - 1]
    # middle n entries correspond to Σ_k w_k a_{k-i}
    seg = conv[n - 1: 2 * n - 1]
    return 2.0 * seg                     # factor 2 as explained above


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                       # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]                 # large → left side
        b[-1 - i] = s[i]                 # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Core optimisation phases
# ----------------------------------------------------------------------
def _smooth_max_descent(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using a smooth approximation of the max‑convolution.
    Returns the best vector seen and its evaluation.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[descent] start value = {best_val:.6f}")

    it = 0
    beta = 0.5               # softness of the max‑approximation
    beta_max = 20.0
    while time.time() < deadline - 5.0:
        # 1) self‑convolution
        conv = _full_convolution(a)
        max_conv = conv.max()

        # 2) soft‑max weights (max weight = 1)
        w = np.exp(beta * (conv - max_conv))

        # 3) gradient
        g = _gradient_weighted(a, w)

        # 4) adaptive step size: shrink the most‑influential coordinate by ≈exp(-0.5)
        max_g = g.max()
        if max_g <= 0:
            break
        eta = 0.5 / max_g

        # multiplicative mirror update
        delta = -eta * g
        delta = np.clip(delta, -30.0, 30.0)           # avoid overflow/underflow
        a = a * np.exp(delta)

        # project back onto the simplex
        a = _project_simplex(a)

        # occasional logging & re‑ordering
        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[descent iter {it}] new best = {best_val:.6f} (max conv {max_conv:.8e})")
            # every 1000 iterations try the anti‑symmetric permutation
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[descent reorder] improved to {best_val:.6f}")

        # slowly increase β to focus more on the true maximum
        beta = min(beta * 1.008, beta_max)

        it += 1

    return best_a, best_val


def _annealing_refine(a_start: np.ndarray, start_val: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Light simulated‑annealing hill‑climb (swap + tiny mass transfers).
    Accepts worsening moves with a decaying temperature.
    """
    cur_a = a_start.copy()
    cur_val = start_val
    best_a = cur_a.copy()
    best_val = cur_val
    n = cur_a.shape[0]
    rng = np.random.default_rng()
    T = 1e-3                     # initial temperature

    while time.time() < deadline:
        # ----- propose a move ------------------------------------------------
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                        # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.07 * rng.random()   # ≤ 7 % of source mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # ----- projection onto simplex --------------------------------------
        new_a = _project_simplex(new_a)

        # ----- evaluate ------------------------------------------------------
        new_val = evaluate_sequence(new_a.tolist())

        # ----- acceptance rule -----------------------------------------------
        if new_val < cur_val:
            cur_a, cur_val = new_a, new_val
            if new_val < best_val:
                best_val, best_a = new_val, new_a.copy()
                print(f"[anneal] new best = {best_val:.6f}")
        else:
            # Metropolis criterion
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    cur_a, cur_val = new_a, new_val

        # ----- temperature schedule -------------------------------------------
        T *= 0.9997   # slow cooling

    return best_a, best_val
