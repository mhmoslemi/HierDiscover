def _solve_lp(a, ks, M):
    n = a.shape[0]
    A = _build_constraint_rows(a, ks)
    # Append sum constraint row
    A = np.vstack([A, np.ones(n, dtype=float)])
    b = np.concatenate([np.full(len(ks), M), np.array([1.0])])
    c = -np.ones(n)  # maximize sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        g = res.x
        # ensure nonnegativity & normalization (sum may be less than 1)
        g = np.maximum(g, 0.0)
        s = g.sum()
        if s > 0:
            g = g / s
        return g
    else:
        # fallback: uniform distribution
        return np.full(n, 1.0 / n)

def _mix(a, g, t):
    new = (1 - t) * a + t * g

    s = new.sum()
    if s > 0:
        new = new / s
    else:
        new = np.full_like(new, 1.0 / new.shape[0])

"""
Propose a non‑negative step function minimizing `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – start from the historic construction
   (`height_sequence_1`) if present, otherwise from a random Dirichlet
   (exponential) vector.  The vector is normalised to sum = 1.

2. **LP‑based projection** – at each iteration we compute the full
   self‑convolution `c = a*a` and take the `k` largest peaks
   (default 8).  Keeping the current maximal value `M = max(c)`,
   we solve a tiny linear programme

        maximise   Σ_j g_j
        subject to   (a * g)[p] ≤ M   for every selected peak p
                     Σ_j g_j ≤ 1
                     g_j ≥ 0

   with `g` having the same dimension as `a`.  Because only a few
   constraints are used the LP is tiny (≈ 9 constraints for
   8000 variables) and solves in a few milliseconds.

3. **Mixing step** – we replace the current candidate by a convex
   combination `a ← (1‑t)·a + t·g` with a decaying mixing weight `t`
   (initially 0.4, multiplied by 0.999 each iteration).  The sum stays
   equal to 1, so normalisation is not required.

4. **Anti‑symmetric re‑ordering** – every 40 iterations we try to
   improve the candidate by pairing the largest entries with the
   smallest ones (the rearrangement inequality gives a lower central
   autocorrelation).  If the reordered vector improves the score it
   is kept.

5. **Hill‑climbing refinement** – after the main loop a lightweight
   local search (random swaps and tiny mass transfers) is performed
   until the time budget expires.

The whole procedure respects the 1000 s budget and returns the best
vector found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (length 2·n‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may create tiny negatives – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Build constraint matrix for a set of convolution peaks
# ----------------------------------------------------------------------
def _build_constraint_rows(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    For each peak index k in `ks` return a row vector r such that
        (a * g)[k] = r @ g .
    This is just the reversed slice of `a` that aligns with the
    convolution.  The result has shape (len(ks), len(a)).
    """
    n = a.shape[0]
    rows = np.zeros((len(ks), n), dtype=float)

    for row_idx, k in enumerate(ks):
        # i runs over indices of a that actually contribute to (a*g)[k]
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        seg = a[i_start:i_end + 1]          # a[i_start … i_end]
        # Corresponding g‑indices run from j = k-i_start down to j = k-i_end
        j_start = k - i_start
        j_end = k - i_end
        # Assign the reversed segment to the appropriate slice.
        rows[row_idx, j_end:j_start + 1] = seg[::-1]

    return rows


# ----------------------------------------------------------------------
# Solve the tiny linear programme described in the docstring.
# ----------------------------------------------------------------------
def _solve_lp(a: np.ndarray, ks: np.ndarray, M: float) -> np.ndarray:
    """
    Maximise Σ g_j subject to:
        (a * g)[k] ≤ M   for all k in ks,
        Σ g_j ≤ 1,
        g_j ≥ 0.
    Returns a feasible vector `g` (normalised to sum = 1).
    """
    n = a.shape[0]

    # Build the constraint matrix for the selected peaks.
    A_peaks = _build_constraint_rows(a, ks)          # shape (|ks|, n)

    # Append the global-sum constraint.
    A = np.vstack([A_peaks, np.ones(n, dtype=float)])
    b = np.concatenate([np.full(len(ks), M), np.array([1.0])])

    # Linear programme: maximise Σ g_j  ⇔ minimise −Σ g_j
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method="highs", options={"presolve": True})

    if not res.success:
        # Fallback – a uniform distribution is always feasible.
        g = np.full(n, 1.0 / n, dtype=float)
        return g

    g = np.maximum(res.x, 0.0)
    total = g.sum()
    if total > 0.0:
        g /= total
    else:
        g = np.full(n, 1.0 / n, dtype=float)
    return g


# ----------------------------------------------------------------------
# Mixing helper
# ----------------------------------------------------------------------
def _mix(a: np.ndarray, g: np.ndarray, t: float) -> np.ndarray:
    """Return (1‑t)·a + t·g with a simplex projection."""
    new = (1.0 - t) * a + t * g
    # Floating‑point drift may break the simplex; re‑project.
    s = new.sum()
    if s > 0.0:
        new /= s
    else:
        new = np.full_like(new, 1.0 / new.shape[0])
    return new


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (same idea as the previous code)
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, cur_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Random swaps / tiny mass transfers, keeping only strict improvements."""
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            delta = best_a[i] * 0.02 * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val + 1e-12 < best_val:
            best_a = new_a
            best_val = val
            print(f"[refine] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand coefficients give a good trade‑off
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise from the historic construction if possible
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # Interpolate to the desired length.
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        # Dirichlet‑like start (exponential, normalised).
        a = rng.exponential(scale=1.0, size=n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a /= s

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop – LP‑based projection & mixing
    # --------------------------------------------------------------
    iteration = 0
    mix_weight = 0.4                # initial mixing parameter t
    decay_factor = 0.999            # geometric decay of t
    min_peaks = 5
    max_peaks = 20

    while time.time() < deadline - 30.0:
        # Full self‑convolution of the current best candidate.
        conv = _full_convolution(best_a)
        max_conv = conv.max()

        # Select a set of peak indices to form the linear constraints.
        # The number of constraints grows slowly with the iteration count.
        k_peaks = min(max_peaks, max(min_peaks,
                                     min_peaks + iteration // 200))
        if k_peaks >= conv.size:
            ks = np.arange(conv.size)
        else:
            ks = np.argpartition(-conv, k_peaks - 1)[:k_peaks]

        # Solve the tiny LP to obtain a feasible direction g.
        g = _solve_lp(best_a, ks, max_conv)

        # Mix the direction into the current candidate.
        new_a = _mix(best_a, g, mix_weight)

        # Occasionally try the anti‑symmetric reordering.
        if iteration % 40 == 0:
            cand = _anti_symmetric_reorder(new_a)
            val_reorder = evaluate_sequence(cand.tolist())
            if val_reorder < best_val:
                best_a = cand
                best_val = val_reorder
                print(f"[reorder iter {iteration}] improved to {best_val:.6f}")

        # Evaluate the mixed candidate.
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            # Should never happen as we keep the simplex projection.
            val = math.inf

        if val < best_val:
            best_a = new_a
            best_val = val
            print(f"[iter {iteration}] new best = {best_val:.6f}")

        # Decay the mixing weight – we become more conservative over time.
        mix_weight = max(0.01, mix_weight * decay_factor)
        iteration += 1

    # --------------------------------------------------------------
    # Final hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline, rng)

    # --------------------------------------------------------------
    # Final evaluation & return
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
