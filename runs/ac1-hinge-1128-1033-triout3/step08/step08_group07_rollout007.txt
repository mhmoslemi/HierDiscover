def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a∗a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # remove tiny negatives caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for a normalised vector.
    The vector must be non‑negative and sum to 1.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    return float(2.0 * n * max_b)   # sum_a == 1 → denominator drops out


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    w_shift = w - np.max(w)
    ex = np.exp(w_shift)
    return ex / np.sum(ex)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and conv = a∗a with a = softmax(w).  Gradient is computed
    analytically and returned with respect to the soft‑max parameters.
    """
    a = _softmax(w)                     # ℓ₁‑normalised, non‑negative
    conv = _fft_convolve_self(a)        # length 2·n‑1
    max_c = conv.max()

    # smooth approximation of the max
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)        # Σ exp((c−max)/ε)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (derived from the smoothed max)
    weights = np.exp(shifted)           # shape (2n‑1,)
    p = weights / np.sum(weights)       # soft‑max over the convolution entries

    # convolution a * reverse(p) gives ∂L/∂a up to the factor 2·n
    p_rev = p[::-1]
    size = _next_pow2(conv.shape[0] + a.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[:a.shape[0] + p_rev.shape[0] - 1]

    # the relevant segment corresponds to indices [n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]      # length n
    grad_a = 4.0 * n * seg[::-1]        # ∂L/∂a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by `scipy.optimize.minimize`."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Heuristic permutation schemes
# ----------------------------------------------------------------------
def _alternating_ends(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values at the two ends, then the next largest
    at the opposite side, alternating inward.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)[::-1]          # descending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    i = 0
    while i < n:
        new_a[left] = sorted_vals[i]
        left += 1
        i += 1
        if i >= n:
            break
        new_a[right] = sorted_vals[i]
        right -= 1
        i += 1
    return new_a


def _bitrev_permutation(n: int) -> np.ndarray:
    """
    Returns a permutation of ``range(n)`` that corresponds to the
    bit‑reversal ordering (requires n to be a power of two).
    """
    bits = n.bit_length() - 1
    perm = np.arange(n, dtype=int)
    rev = np.zeros_like(perm)
    for i in range(n):
        b = i
        r = 0
        for _ in range(bits):
            r = (r << 1) | (b & 1)
            b >>= 1
        rev[i] = r
    perm[:] = rev
    return perm


def _apply_permutation(a: np.ndarray, perm: np.ndarray) -> np.ndarray:
    """Reorder ``a`` according to ``perm``."""
    return a[perm]


def _random_permutation(a: np.ndarray) -> np.ndarray:
    """Return a random shuffle of ``a``."""
    rng = np.random.default_rng()
    perm = np.arange(a.shape[0])
    rng.shuffle(perm)
    return a[perm]


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------
def _generate_candidates(n: int, max_candidates: int, deadline: float):
    """
    Produce a stream of candidate sequences.  Each candidate is built
    from an exponential weight profile a_i ∝ r^i (0.90 ≤ r ≤ 0.9999)
    which is then permuted by one of several deterministic schemes.
    Only the best few candidates are retained.
    """
    rng = np.random.default_rng()
    best_seqs = []          # list of (value, array)
    max_keep = 8            # keep at most this many for later refinement

    while len(best_seqs) < max_candidates and time.time() < deadline:
        # ----- pick a decay factor -------------------------------------------------
        r = rng.uniform(0.90, 0.9999)                # exponential decay
        base = r ** np.arange(n)                     # decreasing profile
        a0 = base / np.sum(base)                     # ℓ₁‑normalise

        # ----- pick a permutation --------------------------------------------------
        perm_type = rng.integers(0, 4)
        if perm_type == 0:
            a = _alternating_ends(a0)
        elif perm_type == 1:
            # bit‑reversal works only for powers of two; fall back to random otherwise
            if (n & (n - 1)) == 0:
                perm = _bitrev_permutation(n)
                a = _apply_permutation(a0, perm)
            else:
                a = _random_permutation(a0)
        elif perm_type == 2:
            a = _random_permutation(a0)
        else:
            # identity (pure exponential)
            a = a0.copy()

        # ----- fast evaluation ------------------------------------------------------
        val = _eval_fast(a)

        # keep track of the best candidates seen so far
        if len(best_seqs) < max_keep:
            best_seqs.append((val, a.copy()))
            best_seqs.sort(key=lambda x: x[0])
        else:
            if val < best_seqs[-1][0]:
                best_seqs[-1] = (val, a.copy())
                best_seqs.sort(key=lambda x: x[0])

        # small pause to keep time checks reasonably cheap
        if len(best_seqs) % 50 == 0:
            if time.time() >= deadline:
                break

    return best_seqs


# ----------------------------------------------------------------------
# Adam optimiser on the exact max convolution
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float):
    """
    ADAM optimiser on the true max‑convolution (sub‑gradient of the
    active peak) while keeping the vector on the probability simplex.
    """
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)                     # initialise soft‑max parameters
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_a = a0.copy()
    best_val = _eval_fast(best_a)

    while time.time() < deadline:
        t += 1
        a = _softmax(w)

        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        # gradient of the active peak w.r.t. a
        grad_a = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_a[mask] = a[idx[mask]]               # derivative of a_i * a_{k-i}
        # factor 2 because each product appears twice except at borders (insignificant)
        grad_a *= 2.0

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ADAM update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = 0.6 / (np.sqrt(t) + 1.0)          # slowly decaying learning rate
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate new point (fast)
        a_new = _softmax(w)
        cur_val = _eval_fast(a_new)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[Adam] iter {t:5d}  new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing + tiny mass transfers
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float, deadline: float):
    """
    Classic SA that alternates between swapping two positions and moving
    a small percentage (≤ 3 %) of mass from one index to another.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq, best_val = a.copy(), cur_val

    # temperature starts as a few per‑cent of the current value
    T = max(cur_val * 0.02, 1e-7)
    decay = 0.99994

    while time.time() < deadline:
        if rng.random() < 0.35:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
        else:                                        # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-14:
                continue
            delta = a[i] * 0.03 * rng.random()      # ≤ 3 % of source mass
            a[i] -= delta
            a[j] += delta

        new_val = _eval_fast(a)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA]  new best = {best_val:.6f}")
        else:
            # revert move
            if rng.random() < 0.35:
                a[i], a[j] = a[j], a[i]
            else:
                a[i] += delta
                a[j] -= delta

        T = max(T * decay, 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float, deadline: float):
    """Greedy local search – accepts any strictly improving move."""
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_a, cur_v = a.copy(), cur_val

    while time.time() < deadline:
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                        # tiny mass move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-14:
                continue
            delta = cur_a[i] * 0.015 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_v:
            cur_a, cur_v = new_a, val
            print(f"[refine] new best = {cur_v:.6f}")

    return cur_a, cur_v


# ----------------------------------------------------------------------
# Main entry point required by the challenge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give a tiny margin for the final official evaluation
    deadline = start_time + budget_s - 1.0
    print(f"[init] time budget = {budget_s}s, deadline @ {deadline:.2f}")

    # ------------------------------------------------------------------
    # Problem size -------------------------------------------------------
    # ------------------------------------------------------------------
    n = 4000                       # a comfortable size for FFT‑based ops
    global _GLOBAL_N
    _GLOBAL_N = n

    # ------------------------------------------------------------------
    # Warm‑start ---------------------------------------------------------
    # ------------------------------------------------------------------
    # Prefer an already known construction if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        # start from a tiny random perturbation of the uniform vector
        a0 = np.full(n, 1.0 / n)
        a0 += 5e-4 * np.random.rand(n)
        print("[init] using fresh uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= np.sum(a0)                         # ℓ₁‑normalise to 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – generate a diverse pool of cheap candidates --------------
    # ------------------------------------------------------------------
    cand_deadline = deadline - 300.0          # reserve ~5 min for later phases
    if time.time() < cand_deadline:
        print("[phase0] generating cheap candidates")
        cand_pool = _generate_candidates(
            n=n,
            max_candidates=40,               # keep 40 best from the raw pool
            deadline=cand_deadline
        )
        # keep the top few for refinement
        best_pool = cand_pool[:8]
        # update best_a / best_val with the best from the pool
        if best_pool and best_pool[0][0] < best_val:
            best_val, best_a = best_pool[0]
            print(f"[phase0] new best from pool = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate L‑BFGS (multiple restarts) -------------
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        print("[phase1] L‑BFGS on smoothed surrogate")
        eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-16), num=15)

        restart = 0
        while time.time() < deadline - 200.0:
            restart += 1
            print(f"[phase1] restart #{restart}")

            # initialise from the current best point (soft‑max params)
            w = np.log(best_a + 1e-12) + 0.5 * np.random.randn(n)

            for eps in eps_schedule:
                if time.time() >= deadline - 180.0:
                    break
                global _GLOBAL_EPS
                _GLOBAL_EPS = eps

                # L‑BFGS step
                res = minimize(_smooth_obj_and_grad_wrapper, w,
                               method='L‑BFGS‑B',
                               jac=True,
                               options={'maxiter': 500,
                                        'ftol': 1e-12,
                                        'gtol': 1e-12,
                                        'disp': False})
                w = res.x

                # evaluate
                a_candidate = _softmax(w)
                cur_val = _eval_fast(a_candidate)
                if cur_val < best_val:
                    best_val, best_a = cur_val, a_candidate.copy()
                    print(f"[phase1] eps={eps:.1e} → new best = {best_val:.6f}")

            # --- cheap re‑ordering after each restart ------------------------
            # Use a stricter “alternating‑ends” reordering
            a_ord = _alternating_ends(best_a)
            a_ord /= np.sum(a_ord)
            val_ord = _eval_fast(a_ord)
            if val_ord < best_val:
                best_val, best_a = val_ord, a_ord.copy()
                print(f"[phase1] after re‑order → best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – ADAM on the exact max‑convolution ------------------------
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print("[phase2] ADAM optimisation")
        best_a, best_val = _adam_opt(best_a, deadline - 70.0)

    # ------------------------------------------------------------------
    # Phase 3 – long simulated annealing ---------------------------------
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[phase3] simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val, deadline - 20.0)

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb -----------------------------------
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[phase4] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final official evaluation (security‑checked)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    # return plain Python list as required
    return best_a.tolist()
