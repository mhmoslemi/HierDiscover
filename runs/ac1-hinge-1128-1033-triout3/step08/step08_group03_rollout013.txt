Project onto simplex algorithm: Given vector v, we want to find projection onto {x >=0, sum=1}. Implementation:

```
def _project_simplex(v):
    # standard algorithm: sort v descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # clip small negatives
    conv = np.maximum(conv, 0.0)
    return conv

def _gradient_weighted_peaks(a, p):
    # a: (n,), p: (2n-1,)
    n = a.shape[0]
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # segment
    seg = conv[n - 1 : 2 * n - 1]
    grad = seg[::-1]
    return grad

def _project_simplex(v):
    # Project onto the probability simplex (sum=1, v>=0)
    n = v.shape[0]
    # sort v decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _anti_symmetric_reorder(a):
    n = a.shape[0]
    s = np.sort(a)  # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]  # large
        b[-1 - i] = s[i]  # small
    if n % 2:
        b[half] = s[half]
    return b

"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – several restarts with different simple seeds:
   uniform, sinusoidal, exponential random.  All vectors are normalised
   to sum = 1 (the objective is scale‑invariant).

2. **Smoothed max‑convolution descent** – minimise a smooth surrogate of
   the true objective:
        f_ε(a) = (1/ε) * log∑_k exp(ε·b_k)  ,  where b = a*a .
   The gradient of f_ε is written as a single FFT‑based convolution,
   allowing us to treat *all* convolution peaks simultaneously.
   An exponential (multiplicative) mirror‑descent step
        a ← a·exp(−η·∇f_ε) ,   a ← a / sum(a)
   keeps the vector non‑negative and on the probability simplex.
   The smoothing parameter ε is decreased in stages (0.5 → … → 0.01)
   so that the optimisation first smooths the landscape and then
   focuses on the true maximum.

3. **Anti‑symmetric re‑ordering** – after each ε‑stage the vector is
   reordered by pairing the largest and smallest entries; the
   rearrangement inequality guarantees that this never worsens the
   convolution peak.

4. **Hill‑climbing refinement** – a cheap local search (random swaps
   and tiny mass transfers) is performed until the time budget is
   exhausted.

All heavy work (FFT‑based gradient steps) is performed without calling
`evaluate_sequence`; the true objective is queried only occasionally to
track progress.  The routine respects the 1000 s budget and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of ∑_k p_k·b_k w.r.t. a,
    where b_k = (a*a)[k] and p is a (2n‑1) weight vector (∑p=1, p≥0).

    Result: g_i = Σ_k p_k·a_{k‑i}   (out‑of‑range indices are zero).
    Computed with one FFT.
    """
    n = a.shape[0]
    # reverse the weight vector
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # the relevant slice corresponds to indices n‑1 … 2n‑2 (inclusive)
    seg = conv[n - 1 : 2 * n - 1]
    # reverse back to obtain the gradient
    g = seg[::-1]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0 , Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]            # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Smoothed‑max descent (multiplicative mirror‑descent)
# ----------------------------------------------------------------------
def _smoothed_descent(a_start: np.ndarray, deadline: float):
    """
    Minimise a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector found and its true evaluation.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[descent init] objective = {best_val:.6f}")

    # schedule of smoothing parameters (large → small)
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    # total number of inner iterations (adjusted to the remaining budget)
    # we aim for roughly 2000 – 3000 FFT steps overall.
    total_iters = 2500
    it_per_eps = max(1, total_iters // len(eps_schedule))

    for eps in eps_schedule:
        # step size – a modest constant that decays slowly
        eta = 0.7

        for it in range(it_per_eps):
            if time.time() >= deadline - 5.0:
                break

            # ----- compute convolution and smooth weights -----
            conv = _full_convolution(a)          # length 2n‑1
            max_conv = conv.max()

            # stable softmax: shift by the maximum
            shift = max_conv
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()        # Σp = 1, p≥0

            # ----- gradient of the smooth surrogate -----
            grad = 2.0 * _gradient_weighted_peaks(a, p)   # shape (n,)

            # ----- multiplicative mirror‑descent step -----
            delta = -eta * grad
            # avoid overflow/underflow in the exponentials
            delta = np.clip(delta, -20.0, 0.0)    # exp(‑20) ≈ 2·10⁻⁹
            a = a * np.exp(delta)
            a = _project_simplex(a)               # enforce Σa = 1, a≥0

            # ----- occasional true‑objective check -----
            if it % 20 == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[eps {eps:.3f} it {it}] new best = {best_val:.6f} (max conv {max_conv:.6e})")

                # anti‑symmetric re‑ordering every 200 steps
                if it % 200 == 0:
                    a_ord = _anti_symmetric_reorder(a)
                    val_ord = evaluate_sequence(a_ord.tolist())
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        print(f"[reorder eps {eps:.3f} it {it}] improved to {best_val:.6f}")

            # slowly decay the step size (helps stability for small ε)
            eta *= 0.9995

        # a final re‑ordering after each ε‑stage (cannot increase the max)
        a = _anti_symmetric_reorder(a)

    return best_a, best_val

**Explanation of key ideas**

* **Log‑sum‑exp smoothing** turns the non‑differentiable `max` into a smooth convex function, allowing the use of a single gradient that aggregates information from *all* convolution positions (instead of a few‑peak selection).  
* **Multiplicative mirror‑descent** (exponential update followed by projection onto the simplex) keeps the coefficients non‑negative and automatically respects the scale invariance of the problem.  
* **Continuation in ε** – starting with a large smoothing factor and gradually decreasing it lets the optimizer first move quickly on a smooth landscape and then hone in on the true max‑convolution.
