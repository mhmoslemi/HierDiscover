assistantfinal**Solution Overview**

The objective  

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{(\sum_i a_i)^2}
\]

is scale‑invariant, therefore we may restrict the search to the probability simplex  
\(\sum_i a_i = 1,\; a_i\ge 0\).  
The term \(\max_k (a*a)_k\) is the supremum of the *self‑convolution* of the
step‑function.  Minimising it is a highly non‑convex problem, but three
complementary optimisation ideas work well in practice:

1. **Parametric warm‑starts** – power‑law, exponential and a product of both
   generate many diverse candidates cheaply.
2. **Sub‑gradient mirror descent** – an exponential update driven by the
   gradient of the *active* (near‑maximal) convolution entries.
3. **Smooth‑max surrogate & L‑BFGS** – the max is replaced by a
   log‑sum‑exp, yielding a smooth objective whose gradient can be obtained
   by a single FFT‑based convolution.  A schedule of decreasing smoothing
   parameters drives the solution to the true optimum.
4. **Targeted simulated‑annealing** – local mass‑transfer moves between
   high‑mass and low‑mass coordinates (or distant swaps) finish the search.

All operations stay on the simplex, the vector is kept monotone decreasing
(the anti‑symmetric ordering that pairs large and small coefficients) because
this arrangement is provably optimal for a given multiset of magnitudes.

The algorithm respects the 1000‑second budget, prints intermediate progress
and finally returns the best coefficient list it discovered.

```python
"""
propose_candidate – search for a non‑negative step function that minimises
evaluate_sequence(sequence).  The method combines:

* a dense grid of parametric warm‑starts (power‑law, exponential, mixed);
* exponentiated mirror‑descent on the true L∞‑objective;
* a smooth‑max surrogate optimised by L‑BFGS-B with a fine ε‑schedule;
* a targeted simulated‑annealing hill‑climber with mass‑transfer moves.

The vector is always projected onto the probability simplex and finally
re‑ordered in decreasing order (large coefficients on the left, small on the
right), which minimises the central convolution term.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global parameters (filled by the optimiser)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε for the surrogate
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # eliminate tiny negatives
    return conv


def _descending_order(a: np.ndarray) -> np.ndarray:
    """Return a sorted‑descending copy of *a* (large ↔ left, small ↔ right)."""
    return np.sort(a)[::-1]


# ----------------------------------------------------------------------
# Parametric families (stage 0)
# ----------------------------------------------------------------------
def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (monotonically decreasing)."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = idx ** (-p)
    a /= a.sum()
    return a


def _exponential_candidate(r: float, n: int) -> np.ndarray:
    """a_i ∝ r**i   (0 < r < 1)."""
    idx = np.arange(n, dtype=np.float64)
    # use log to avoid underflow for large i
    a = np.exp(idx * np.log(r))
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _mixed_candidate(p: float, r: float, n: int) -> np.ndarray:
    """
    a_i ∝ (i+1)^(-p) * r**i   (product of power‑law and exponential).
    """
    idx = np.arange(n, dtype=np.float64)
    a = (idx + 1.0) ** (-p) * np.exp(idx * np.log(r))
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. log‑softmax parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)          (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # stable soft‑max
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # self‑convolution
    conv = _full_convolution(a)
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # weights for the smooth max
    weights = np.exp(shifted)
    p = weights / weights.sum()               # p_k over convolution indices

    # convolution of a with reversed p  (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the middle n entries correspond to ∂max/∂a_i (see derivation)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]              # ∂L/∂a_i

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.minimize – uses globals."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from log‑softmax parameters ``w_init``.
    Returns the best pair (a, value) discovered before the deadline.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule – from 5·10⁻³ down to 10⁻¹⁴ (geometric)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-14), num=13)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 1200,
                                'ftol': 1e-14,
                                'gtol': 1e-14,
                                'disp': False})
        w = res.x

        # map back to the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # try the descending order (large ↔ small pairing)
        a_ord = _descending_order(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞‑objective (stage 1)
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} a_{k‑i} using a single FFT.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n‑2)‑i]  for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent_multi(a_start: np.ndarray, deadline: float):
    """
    Exponentiated mirror‑descent that reduces the *active* convolution peaks.
    Returns the best (a, value) pair found before the deadline.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.9                     # initial step‑size factor
    rel_tol = 0.08                  # start with a generous active‑set tolerance

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten tolerance every few thousand iterations
        if it and it % 15000 == 0:
            rel_tol = max(rel_tol * 0.6, 5e-6)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of the L∞‑norm
        g = _active_set_gradient(a, active)

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative (exponentiated) update
        a = a * np.exp(-alpha * g)

        # project onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional logging & re‑ordering
        if it % 250 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")
            if it % 6000 == 0:
                a_ord = _descending_order(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[sgd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (stage 3)
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    """
    Targeted local moves with an annealing schedule.
    Swaps are made between distant positions, and mass is transferred from
    the heavy tail to the light tail.  Only strictly better solutions update
    the reported best; uphill moves are accepted with a temperature that
    decays exponentially.
    """
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.02                     # initial temperature
    it = 0

    # pre‑compute index groups (top / bottom 10 %)
    top_k = max(1, n // 10)
    bottom_k = max(1, n // 10)

    while time.time() < deadline:
        it += 1
        if rng.random() < 0.45:                     # distant swap
            i = rng.integers(0, n)
            # ensure the second index is at least n/2 away
            j = (i + rng.integers(n // 2, n)) % n
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                        # mass transfer
            # pick donor from the heavy tail
            idx_desc = np.argsort(cur_a)[::-1]
            donor = idx_desc[rng.integers(0, top_k)]
            # pick recipient from the light tail
            idx_asc = np.argsort(cur_a)
            recipient = idx_asc[rng.integers(0, bottom_k)]
            if donor == recipient or cur_a[donor] <= 1e-15:
                continue
            delta = cur_a[donor] * 0.07 * rng.random()   # up to 7 % of donor
            new_a = cur_a.copy()
            new_a[donor] -= delta
            new_a[recipient] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            accept = True
        else:
            T = T0 * (0.9995 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val_local:.6f}")

    return best_a, best_val_local


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    evaluate_sequence under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – larger n gives more freedom but costs more time.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initial vector (fallback: uniform)
    # ------------------------------------------------------------------
    try:
        # if a previous construction named `height_sequence_1` exists, reuse it
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)   # uniform start

    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 1e-12 else np.full(n, 1.0 / n)

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – a dense grid of parametric warm‑starts
    # ------------------------------------------------------------------
    # (i) power‑law sweep
    for p in np.linspace(0.05, 2.5, 45):
        if time.time() >= deadline - 250:
            break
        cand = _downward = _descending_order(_powerlaw_candidate(p, n))
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[power‑law p={p:.3f}] new best = {best_val:.6f}")

    # (ii) exponential sweep
    for r in np.linspace(0.95, 0.9999, 30):
        if time.time() >= deadline - 250:
            break
        cand = _descending_order(_exponential_candidate(r, n))
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[exp r={r:.5f}] new best = {best_val:.6f}")

    # (iii) mixed power‑law / exponential grid
    p_grid = np.linspace(0.2, 1.6, 14)
    r_grid = np.linspace(0.96, 0.999, 11)
    for p in p_grid:
        for r in r_grid:
            if time.time() >= deadline - 250:
                break
            cand = _descending_order(_mixed_candidate(p, r, n))
            val = evaluate_sequence(cand.tolist())
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[mixed p={p:.3f} r={r:.4f}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – sub‑gradient descent on the true max‑convolution objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:
        a_sgd, val_sgd = _subgradient_descent_multi(best_a, deadline)
        if val_sgd < best_val:
            best_val, best_a = val_sgd, a_sgd.copy()
            print(f"[sub‑gradient] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth surrogate optimisation (L‑BFGS) with several restarts
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 120 and restarts < 12:
        restarts += 1
        # perturb current best for a fresh start in the log‑space
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.2, size=n)

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[restart {restarts}] L‑BFGS improved to {best_val:.6f}")

        # one more descending‑order sweep
        a_ord = _descending_order(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[restart {restarts}] reorder improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final refinement via simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_simulated_anneal(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
```
