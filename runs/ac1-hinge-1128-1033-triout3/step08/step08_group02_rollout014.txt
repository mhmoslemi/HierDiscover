def simplex_projection(v):
    if v.sum() == 1 and np.all(v>=0): return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _smooth_gradient_descent(a0, n, start_time, deadline, best_a, best_val, target=1.503):
    # parameters
    max_iters = 500000  # guess.
    alpha_start = 0.01
    alpha_end = 20.0
    lr_start = 0.4
    lr_end = 0.001
    momentum = 0.9
    velocity = np.zeros_like(a0)
    cur = a0.copy()
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        # compute full conv
        conv = _full_convolution(cur)
        maxc = conv.max()
        cur_val = 2.0 * n * maxc
        # update best
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, cur.copy()
            last_improve = 0
            print(f"[Grad {it:07d}] new best = {best_val:.6f}")
            if best_val <= target:
                break
        else:
            last_improve += 1

        # schedule parameters
        progress = it / max_iters
        alpha = (1 - progress) * alpha_start + progress * alpha_end
        lr = (1 - progress) * lr_start + progress * lr_end

        # weight distribution
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient via correlation
        grad = 2.0 * _valid_correlation(w, cur)   # shape n

        # momentum update
        velocity = momentum * velocity - lr * grad
        cur = cur + velocity

        # project onto simplex
        cur = _simplex_projection(cur)

        # enforce symmetry
        cur = (cur + cur[::-1]) * 0.5

        # occasional jitter if stuck
        if last_improve > 100000:
            noise = np.random.normal(scale=0.005, size=n)
            cur = np.maximum(cur + noise, 0)
            cur = _simplex_projection(cur)
            last_improve = 0
            print(f"[Grad {it:07d}] jitter after stagnation")

"""
propose_candidate – smooth‑gradient optimizer for the self‑convolution bound

The objective
        f(a) = 2·n·max(conv(a,a)) / (∑a)²
is scale‑invariant, therefore we keep ∑a = 1 and minimise
        2·n·max(conv(a,a)).

Algorithm
---------
* **Phase‑0 – cheap families** – a few deterministic/in‑random families
  (uniform, power‑law, exponential, sine, …) are evaluated and the best
  one is used as a starting point.

* **Phase‑A – smoothed gradient descent** – we minimise the smooth
  surrogate
        φ_α(a) = 2·n·log‑sum‑exp(α·conv(a,a)) / α
  which converges to the original max‑convolution as α → ∞.
  The gradient of φ_α is
        ∇φ = 2·∑_k w_k·a_{k‑i},   w_k ∝ exp(α·conv_k) ,
  i.e. a weighted correlation of the current weight vector w with the
  current a.  The gradient step is projected onto the probability simplex
  (non‑negative, sum = 1) and the vector is symmetrised after each step.
  α and the learning rate η decay linearly from a small to a large value,
  allowing a smooth global search followed by a focused refinement.

* **Phase‑B – deterministic hill‑climb (optional)** – a fast greedy
  mass‑transfer that moves a small fraction of the probability mass from
  the index that contributes most to the current peak to the index that
  contributes least.  This removes residual ``spikes'' that the smooth
  phase may leave behind.

* **Phase‑C – tiny uniform mixing** – convex combinations with a uniform
  vector are tried (α∈{0.001,…,0.020}) – a classic “smoothing” trick that
  sometimes lowers the maximum convolution entry further.

All phases respect the overall 1000 s time budget, keep the best
candidate found, and finally verify the result with the supplied
`evaluate_sequence` wrapper.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (real‑valued, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y using FFT (both real)."""
    out_len = x.shape[0] + y.shape[0] - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection on the probability simplex (∑=1, a_i≥0)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0

    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by evaluate_sequence when Σa = 1."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Small family sweep – cheap initial candidates
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starter vector and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(arr):
        nonlocal best_a, best_val
        a = np.asarray(arr, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    rng = np.random.default_rng(12345)
    idx = np.arange(n, dtype=float)

    # deterministic families
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing linear
    try_update(np.maximum(idx + 1, n - idx))                        # linear U‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # random Dirichlet draws
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Smoothed gradient descent on  φ_α(a) = 2·n·log‑sum‑exp(α·conv)/α
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, n: int,
                             start: float, deadline: float,
                             best_a: np.ndarray, best_val: float,
                             target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of max(conv).
    The state is kept on the probability simplex and forced to be symmetric
    after each step.
    """
    cur = a0.copy()
    velocity = np.zeros_like(cur)      # simple momentum
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    # hyper‑parameters (tuned experimentally)
    max_iters = 600_000
    alpha_start, alpha_end = 0.02, 20.0          # smoothing schedule
    lr_start, lr_end = 0.4, 0.001               # learning‑rate schedule
    momentum = 0.9

    while it < max_iters and time.time() < deadline:
        # --- evaluate current objective (exact) --------------------------------
        conv = _full_convolution(cur)          # shape 2n‑1
        maxb = conv.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[Grad {it:07d}] new best = {best_val:.6f}")
            if best_val <= target:
                break
        else:
            no_improve += 1

        # --- schedules ---------------------------------------------------------
        progress = it / max_iters
        alpha = (1.0 - progress) * alpha_start + progress * alpha_end
        lr = (1.0 - progress) * lr_start + progress * lr_end

        # --- softmax weights over the convolution entries -----------------------
        # max‑stable exponentiation
        w = np.exp(alpha * (conv - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:               # should never happen, but be safe
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the smoothed loss ------------------------------------
        # g_i = 2 * Σ_k w_k * a_{k‑i}  (valid for i∈[0,n‑1])
        # Compute via convolution of w with the reversed a.
        a_rev = cur[::-1]
        conv_wa = _conv_general(w, a_rev)                     # length 3n‑2
        start_idx = n - 1
        grad = 2.0 * conv_wa[start_idx:start_idx + n]        # shape n

        # --- momentum update + projection --------------------------------------
        velocity = momentum * velocity - lr * grad
        cur = cur + velocity
        cur = _simplex_projection(cur)

        # enforce perfect symmetry (helps convergence and reduces dimensionality)
        cur = (cur + cur[::-1]) * 0.5

        # --- occasional jitter if we are stuck ---------------------------------
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.006, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur = _simplex_projection(cur)
            cur = (cur + cur[::-1]) * 0.5
            no_improve = 0
            print(f"[Grad {it:07d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climb (fine‑tuning)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Small mass‑transfer moves: take a fraction of the probability mass from the
    index that contributes most to the current peak and give it to the index
    that contributes least.  Only accepting strictly improving moves.
    """
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 150_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while time.time() < deadline:
        # evaluate current state
        conv = _full_convolution(cur)
        maxb = conv.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[Hill {it:07d}] new best = {best_val:.6f}")
            if best_val <= target:
                break
        else:
            no_improve += 1

        # anneal the transferred fraction
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # index of the maximal convolution entry and the corresponding slice
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        rev = np.zeros(n)
        rev[lo:hi + 1] = cur[k - np.arange(lo, hi + 1)]

        # source = index with largest contribution, target = smallest (different)
        src = int(np.argmax(rev))
        tmp = rev.copy()
        tmp[src] = np.inf
        tgt = int(np.argmin(tmp))

        if cur[src] <= 1e-15:                   # source empty – jitter if needed
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.004, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur = _simplex_projection(cur)
                cur = (cur + cur[::-1]) * 0.5
                no_improve = 0
                print(f"[Hill {it:07d}] jitter after stagnation")
            continue

        delta = cur[src] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[src] -= delta
        cur[tgt] += delta

        # check if the move improved the true objective
        new_conv = _full_convolution(cur)
        new_val = 2.0 * n * new_conv.max()
        if new_val < cur_val - 1e-12:
            # keep move (already reflected in cur)
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (final polishing)
# ----------------------------------------------------------------------
def _final_mix(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combos with a tiny amount of the uniform vector."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = cand / cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Return a sequence of non‑negative heights minimising `evaluate_sequence`."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin

    # -------------------------------------------------------------
    # Choose a problem size – 8000 works well for the 1000 s budget
    # -------------------------------------------------------------
    n = 8000

    # -------------------------------------------------------------
    # Try to reuse a previous construction if the harness provides one
    # -------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    TARGET = 1.503

    # -------------------------------------------------------------
    # Phase‑0 – quick sweep over families (already done in _initial_candidate)
    # -------------------------------------------------------------
    # (nothing extra needed – best_a already the best of those families)

    # -------------------------------------------------------------
    # Phase‑A – smoothed gradient descent (major improvement driver)
    # -------------------------------------------------------------
    phase_a_deadline = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _smooth_gradient_descent(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target=TARGET,
    )

    # -------------------------------------------------------------
    # Phase‑B – deterministic greedy hill‑climb (optional fine‑tune)
    # -------------------------------------------------------------
    phase_b_deadline = start_time + 0.90 * (deadline - start_time)
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target=TARGET,
    )

    # -------------------------------------------------------------
    # Phase‑C – final mixing with a uniform vector
    # -------------------------------------------------------------
    best_a, best_val = _final_mix(best_a, n)

    # -------------------------------------------------------------
    # Verify with the provided secure wrapper
    # -------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
