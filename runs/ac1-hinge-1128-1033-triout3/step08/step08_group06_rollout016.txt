"""
search_candidate_v2.py

A multi‑stage search for a non‑negative step‑function that minimises

        evaluate_sequence(a)

The algorithm improves on the previous version by

* adding a richer pool of initial candidates (Gaussian, cosine, triangular,
  several power‑law exponents, random Dirichlet, etc.);
* performing many more L‑BFGS restarts with a finer ε‑schedule;
* complementing L‑BFGS with an Adam‑style optimiser for the smooth (log‑sum‑exp)
  surrogate;
* applying the “anti‑symmetric” re‑ordering after every optimiser run;
* allowing two short sub‑gradient descent runs from different starts;
* keeping all work within the 1000 s time budget.

The code follows the same conventions as the reference implementation:
the optimisation variables are the soft‑max logits `w` (so the
coefficients are always non‑negative and sum to one), the convolution is
computed by FFT, and all updates are projected back onto the probability
simplex.  The final vector (a Python list) is returned by ``propose_candidate``.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser (L‑BFGS / Adam)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector

# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x – used for zero‑padding FFT."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative floating‑point artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values on the left and the smallest on the right,
    i.e. pair a large coefficient with a small one when the sequence is reversed.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # large → left side
        res[n - 1 - i] = s[i]     # small → right side
    if n % 2:
        res[half] = s[half]
    return res


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS / Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max logits w.

    a_i = exp(w_i) / Σ_j exp(w_j)                     (ensures a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # --- soft‑max (stable) ---------------------------------------------
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # --- self‑convolution -----------------------------------------------
    conv = _full_convolution(a)
    max_conv = conv.max()

    # --- smooth max via log‑sum‑exp --------------------------------------
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # --- gradient w.r.t. a (see derivation in the reference code) -----
    weights = np.exp(shifted)
    p = weights / weights.sum()                       # distribution over convolution indices

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the middle n entries give ∂L/∂a_i
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                      # ∂L/∂a_i

    # --- chain rule through the soft‑max ---------------------------------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max logits ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule: from 5·10⁻³ down to 10⁻¹² (geometric)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=11)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 800,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to a (probability simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric re‑ordering heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val


def _adam_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Adam‑style optimisation of the smooth surrogate.
    Works on the logits w, keeps the best point seen.
    """
    n = w_init.shape[0]
    w = w_init.copy()
    best_a = None
    best_val = float('inf')

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.03                # base learning‑rate for the logits

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t_iter = 0

    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=11)

    for eps in eps_schedule:
        if time.time() >= deadline - 10.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

                val = evaluate_sequence(a.tolist())

                    best_a = a.copy()

    if best_a is None:
        a = np.exp(w - np.max(w))
        a /= a.sum()
        best_a = a
        best_val = evaluate_sequence(a.tolist())

    return best_a, best_val

# Sub‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} a_{k‑i} using a single FFT.
    ``active_idx`` contains convolution indices (0 … 2n‑2) belonging
    to the active set.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n-2) - i]  for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent_multi(a_start: np.ndarray, deadline: float):
    """
    Exponentiated mirror‑descent that reduces the *active* convolution peaks.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.8                     # initial step‑size factor
    rel_tol = 0.05                    # start with a generous active‑set tolerance

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten the tolerance gradually
        if it > 0 and it % 20000 == 0:
            rel_tol = max(rel_tol * 0.5, 1e-5)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]   # indices of near‑max peaks

        # gradient via FFT (sub‑gradient of the L∞‑norm)
        g = _active_set_gradient(a, active)

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative update
        a = a * np.exp(-alpha * g)

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional logging & re‑ordering
        if it % 200 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")
            if it % 5000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[sgd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing based hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    """
    Random local moves (swap or mass transfer) with an annealing schedule.
    Only strictly better solutions are kept for the reported best,
    but occasional uphill moves are accepted to escape plateaus.
    """
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.02                     # initial “temperature”

    it = 0
    while time.time() < deadline:
        it += 1
        # propose a move
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        # evaluate
        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            accept = True
        else:
            # annealing acceptance probability
            T = T0 * (0.9995 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val_local:.6f}")

    return best_a, best_val_local

def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (dist_to_end+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


def _gaussian_candidate(sigma: float, n: int) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the vector."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def _cosine_candidate(n: int) -> np.ndarray:
    """Raised‑cosine window (Hann‑type) – non‑negative and smooth."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    a += 1e-12                     # avoid exact zero (helps numerical stability)
    a /= a.sum()
    return a


def _triangular_candidate(n: int) -> np.ndarray:
    """Symmetric triangular shape that peaks in the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    a += 1e-12
    a /= a.sum()
    return a


def _exponential_candidate(tau: float, n: int) -> np.ndarray:
    """Decaying exponential (forward) – normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / tau)
    a /= a.sum()
    return a

        cand = rng.dirichlet(alpha=np.ones(n))
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[dirichlet] new best = {best_val:.6f}")

    # (ii) exponential random vectors
    for tr in range(3):
        if time.time() >= deadline - 200:
            break
        cand = rng.exponential(scale=1.0, size=n)
        cand /= cand.sum()
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand exp {tr}] new best = {best_val:.6f}")

    # (iii) symmetric power‑law shapes
    for p in (0.3, 0.5, 0.7, 1.0, 1.5):
        if time.time() >= deadline - 200:
            break
        cand = _powerlaw_candidate(p, n)
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[power‑law p={p:.1f}] new best = {best_val:.6f}")

    # (iv) Gaussian candidates (several σ)
    for sigma_factor in (0.08, 0.12, 0.18):
        if time.time() >= deadline - 200:
            break
        sigma = sigma_factor * n
        cand = _gaussian_candidate(sigma, n)
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[gauss σ={sigma_factor:.2f} n] new best = {best_val:.6f}")

    # (v) cosine (Hann) window
    if time.time() < deadline - 150:
        cand = _cosine_candidate(n)
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[cosine] new best = {best_val:.6f}")

    # (vi) triangular window
    if time.time() < deadline - 150:
        cand = _triangular_candidate(n)
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[triangular] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – sub‑gradient descent (run twice from different starts)
    # ------------------------------------------------------------------
    if time.time() < deadline - 250:
        # first run: from the current best
        a_sgd, val_sgd = _subgradient_descent_multi(best_a, deadline)
        if val_sgd < best_val:
            best_val, best_a = val_sgd, a_sgd.copy()
            print(f"[sgd run0] new best = {best_val:.6f}")

    if time.time() < deadline - 200:
        # second run: from a fresh random Dirichlet point
        w0 = np.log(rng.dirichlet(alpha=np.ones(n)) + 1e-12)
        a_start, _ = _l_bfgs_optimize_from_w(w0, deadline)   # cheap L‑BFGS just to get a start on the simplex
        a_sgd2, val_sgd2 = _subgradient_descent_multi(a_start, deadline)
        if val_sgd2 < best_val:
            best_val, best_a = val_sgd2, a_sgd2.copy()
            print(f"[sgd run1] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – intensive smooth‑surrogate optimisation (L‑BFGS + Adam)
    # ------------------------------------------------------------------
    # L‑BFGS restarts
    restarts = 0
    while time.time() < deadline - 120 and restarts < 12:
        restarts += 1
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.25, size=n)        # moderate perturbation

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[L‑BFGS restart {restarts}] improved to {best_val:.6f}")

        # single anti‑symmetric sweep
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[L‑BFGS reorder {restarts}] improved to {best_val:.6f}")

    # Adam optimiser restarts (fewer because it is heavier)
    for run in range(4):
        if time.time() >= deadline - 80:
            break
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.3, size=n)

        a_opt, val_opt = _adam_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Adam restart {run+1}] improved to {best_val:.6f}")

        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[Adam reorder {run+1}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final refinement via simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_simulated_anneal(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
