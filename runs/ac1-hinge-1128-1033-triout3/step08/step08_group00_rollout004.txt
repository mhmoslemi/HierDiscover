"""
search for a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – try to reuse the previous best solution (if
   available), sweep a two‑parameter Beta family and draw a few random
   Dirichlet vectors.  All candidates are normalised to sum = 1 and the
   best of them becomes the starting point.

2. **Adam (smooth‑max + variance)** – minimise a smooth approximation of
   the true objective together with a variance penalty (to flatten the
   auto‑convolution).  The parameters of Adam, the smoothing factor and
   the variance weight are annealed.  This phase uses a log‑softmax
   parametrisation so the simplex constraint Σa=1 is satisfied automatically.

3. **Hill‑climbing (local mass transfer)** – repeatedly move a small
   amount of mass from a “source’’ index that contributes heavily to the
   current maximal convolution entry to a “target’’ index that contributes
   little.  The convolution update is performed incrementally (O(n)) with a
   *correct* cross‑term (the previous version subtracted only one
   Δ² instead of two).  Every few thousand steps the full convolution is
   recomputed to cancel accumulated floating‑point error.

4. **Fine‑tune Adam** – a short Adam run that minimises only the smooth
   max (variance penalty switched off) with a very small smoothing factor,
   polishing the solution found by the hill‑climber.

The whole search respects the 1000 s budget (a 0.5 s safety margin is kept
for the final call to ``evaluate_sequence``).  The best sequence found is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp in a numerically stable way
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    weights_sum = weights.sum()
    p = weights / weights_sum                     # soft‑max probabilities

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 3n‑2
    # the part that corresponds to the same indices as b
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 2·n already accounted for
    smooth_val = max_b + eps * np.log(weights_sum)
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Variance of the convolution and its gradient
# ----------------------------------------------------------------------
def _var_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns   var(b) = mean((b‑avg)²)   and   ∂var/∂a.
    Here b = a*a and avg = 1/(2n‑1) because Σb = (Σa)² = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # gradient: (4/m) * cross‑correlation of a with v
    conv_ab = _conv_fft(a, v[::-1])              # length 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad = (4.0 / m) * segment[::-1]
    return var, grad


# ----------------------------------------------------------------------
# Adam optimiser (smooth max + variance)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503,
                   lam_start_factor: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises
        L(a) = 2·n·smooth_max(a) + λ·Var(conv(a,a)),
    where λ is linearly annealed from λ_start to 0.
    The simplex constraint Σa = 1 is enforced via a log‑softmax
    parametrisation.
    """
    # logits for soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.18, 0.0015
    eps_start, eps_end = 0.9, 1e-9

    # variance‑penalty schedule
    lam_start = lam_start_factor * n
    lam_end = 0.0

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 22_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- current point on the simplex ----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_obj = _objective(a)
        if cur_obj < best_val:
            best_val, best_a = cur_obj, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # ---- schedules ----
        prog = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog
        lam = lam_start * (1.0 - prog) + lam_end * prog

        # ---- smooth‑max ----
        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)

        # ---- variance ----
        var_val, grad_var = _var_and_grad(a)

        # total gradient of the loss (we minimise)
        grad_a = grad_smooth + lam * grad_var

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter to escape shallow minima
        if np.random.rand() < 0.009:
            w += np.random.normal(scale=0.02, size=n)

        # restart if we have been idle for too long
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.13
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing with exact incremental convolution update
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float,
               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast O(n) local search that transfers a small amount of mass from an
    index heavily influencing the current maximal convolution entry to an
    index that influences it little.  The convolution is updated
    incrementally; every ``recalc_every`` iterations the full convolution
    is recomputed to avoid drift.
    """
    a = a0.copy()
    b = _full_convolution(a)                # current full auto‑convolution
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules
    max_frac_start, max_frac_end = 0.58, 0.0012
    temp_start, temp_end = 0.018, 1e-7
    IMPROVE_LIMIT = 150_000
    recalc_every = 4000

    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each a_i to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (probability ∝ contribution)
        if total_contrib > 0.0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index (biased towards low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform over the remaining indices
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental convolution update (exact) ----
        b_new = b.copy()
        # contribution of moved mass with the unchanged part of a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # self‑terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        # cross‑term (the only place where the previous version was wrong)
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # Re‑compute the full convolution periodically to cancel numerical drift
        if iteration % recalc_every == 0:
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        # Stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.025, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Fine‑tuning Adam (smooth max only, very low smoothing)
# ----------------------------------------------------------------------
def _final_adam(a0: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    A short Adam run that minimises only the smooth max (λ = 0) with a
    very small smoothing factor, serving as a polish step.
    """
    # same log‑softmax parametrisation
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.0012
    eps_start, eps_end = 5e-2, 1e-12   # start relatively smooth, finish extremely sharp

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_obj = _objective(a)
        if cur_obj < best_val:
            best_val, best_a = cur_obj, a.copy()
            print(f"[FineAdam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        prog = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)

        # gradient of loss (no variance term)
        grad_a = grad_smooth

        # back‑prop through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam step
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # tiny random perturbation to avoid dead‑ends
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.007, size=n)

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple Beta‑family sweep (cheap initialisation)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # use a moderately fine grid; the cost is cheap compared with later phases
    p_vals = np.concatenate([np.linspace(0.1, 1.5, 13),
                             np.geomspace(1.5, 8.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a power of two gives a cheap FFT size.
    # ------------------------------------------------------------------
    n = 16384          # 16 K entries give extra expressivity

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previous solution if available.
    # ------------------------------------------------------------------
    a0 = None
    try:
        # ``height_sequence_1`` may be defined in the surrounding environment.
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep + a few random seeds
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    try:
        shape_a, shape_val = _beta_family_best(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] beta‑family sweep improved value={best_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[init] beta‑family sweep failed:", exc)

    # a few random Dirichlet draws for diversity
    for _ in range(4):
        rand_a = np.random.dirichlet(np.ones(n))
        val = _objective(rand_a)
        if val < best_val:
            best_a, best_val = rand_a, val
            print(f"[init] random Dirichlet improved value={best_val:.6f}")

    # fallback random start
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] random start, value =", best_val)

    # ------------------------------------------------------------------
    # Phase A – Adam (smooth max + variance) – ~35 % of budget
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.35 * budget_s
    best_a, best_val = _adam_optimize(best_a, n,
                                      deadline=phase_a_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503,
                                      lam_start_factor=6.0)

    # ------------------------------------------------------------------
    # Phase B – Hill‑climbing – ~50 % of budget
    # ------------------------------------------------------------------
    hill_deadline = start_time + 0.85 * budget_s
    best_a, best_val = _hillclimb(best_a, n,
                                  deadline=hill_deadline,
                                  start_time=phase_a_deadline,
                                  best_a=best_a,
                                  best_val=best_val,
                                  target_val=1.503)

    # ------------------------------------------------------------------
    # Phase C – final fine‑tune Adam (smooth max only) – remaining time
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_val = _final_adam(best_a, n,
                                   deadline=final_deadline,
                                   start_time=hill_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
