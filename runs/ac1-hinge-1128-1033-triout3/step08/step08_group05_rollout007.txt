def _alternating_reorder(a):
    sorted_vals = np.sort(a)[::-1]  # descending
    new_a = np.empty_like(a)
    left = 0
    right = len(a) - 1
    toggle = True
    for val in sorted_vals:
        if toggle:
            new_a[left] = val
            left += 1
        else:
            new_a[right] = val
            right -= 1
        toggle = not toggle
    return new_a

   Very light stochastic hill‑climbing (random swaps / tiny mass
   transfers) that stops just before the deadline.

All stages respect the global time budget (≈ 1000 s).  The best vector
found is finally checked with the official ``evaluate_sequence`` and
returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders: they are set before each call to the smooth surrogate
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # problem size n


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)               # clip tiny negatives
    return conv


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate L_ε and its gradient wrt the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ----- soft‑max -----
    w_shift = w - np.max(w)                # stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ----- convolution -----
    conv = _full_convolution(a)            # length 2·n‑1
    max_c = conv.max()

    # ----- smooth max -----
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ----- gradient wrt a -----
    weights = np.exp(shifted)
    p = weights / weights.sum()            # soft‑max over convolution entries

    # derivative of conv wrt a is a convolution with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment that corresponds to ∂conv/∂a (central part)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]           # factor 4 matches the derivation

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Thin wrapper required by ``scipy.minimize``."""
    return _smooth_obj_and_grad(w)


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator disappears because Σa = 1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Large values on the left side, small values on the right side.
    Often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[-1 - i]             # biggest → left
        res[n - 1 - i] = s[i]          # smallest → right
    if n % 2:
        res[half] = s[half]
    return res


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """
    Interleaves the largest remaining value on the left,
    the second largest on the right, the third on the next left slot,
    etc.  This spreads large values over many different sums.
    """
    n = a.shape[0]
    vals = np.sort(a)[::-1]            # descending
    res = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for v in vals:
        if toggle:
            res[left] = v
            left += 1
        else:
            res[right] = v
            right -= 1
        toggle = not toggle
    return res


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climber that accepts only strict improvements.
    Moves consist of random swaps (20 %) and small mass transfers (80 %).
    The vector is projected back onto the probability simplex after each move.
    """
    cur = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] <= 1e-15:
                continue
            delta = cur[i] * 0.05 * rng.random()
            new_a = cur.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur, cur_val = new_a, val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur, cur_val


def _perm_hill_climb(a: np.ndarray, best_val: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Permutation‑focused local search.
    Random swaps and deterministic re‑orderings are tried;
    only strictly improving moves are kept.
    """
    cur = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.7:                     # random swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # deterministic re‑order (chosen at random)
            if rng.random() < 0.5:
                new_a = _anti_symmetric_reorder(cur)
            else:
                new_a = _alternating_reorder(cur)

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur, cur_val = new_a, val
            print(f"[perm] new best = {cur_val:.6f}")

    return cur, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (plain Python list) before the budget expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0                     # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # Initialise diverse candidates (all normalised to sum = 1)
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # previous construction if present
    if has_old:
        candidates.append(old / old.sum())

    # Dirichlet draws
    candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoid
    x = np.linspace(0, np.pi, n)
    candidates.append(np.sin(x) + 0.1)

    candidates = [c / c.sum() for c in candidates]

    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] value = {val:.6f}")
        if val < best_val:
            best_val, best_a = val, cand.copy()
    print(f"[init] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on a smooth surrogate
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-1), np.log10(1e-15), num=30)
    max_restarts = 12
    w = np.log(best_a + 1e-12)                     # soft‑max parameters

    for restart in range(max_restarts):
        if time.time() >= deadline - 250.0:
            break
        print(f"[L‑BFGS restart {restart + 1}]")
        # occasional full random restart
        if restart % 4 == 0:
            w = rng.normal(size=n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 250.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 3000,
                                    'ftol': 1e-14,
                                    'gtol': 1e-14,
                                    'disp': False})
            w = res.x
            # back‑transform to probabilities
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a_candidate.copy()
                print(f"[L‑BFGS eps={eps:.1e}] new best = {best_val:.6f}")

        # apply deterministic re‑orderings after each restart
        for reorder_fun, name in [( _anti_symmetric_reorder, "anti‑sym"),
                                  ( _alternating_reorder, "alternating")]:
            a_ord = reorder_fun(best_a)
            val_ord = _fast_evaluate(a_ord)
            if val_ord < best_val - 1e-12:
                best_val, best_a = val_ord, a_ord.copy()
                print(f"[{name} reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – permutation‑based local search
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        best_a, best_val = _perm_hill_climb(best_a, best_val, deadline - 150.0)

    # ------------------------------------------------------------------
    # Phase 3 – refined exponential mirror‑descent on the exact max‑conv.
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    alpha0 = 0.6                # initial step size
    decay = 0.99995
    min_alpha = 0.004

    while time.time() < deadline - 130.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # sub‑gradient of max(conv) wrt a
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]                 # derivative = a_{k-i}

        alpha = max(alpha0 * (decay ** (it // 500)), min_alpha)
        a = a * np.exp(-alpha * g)      # exponential mirror step

        # projection onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        if it % 250 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[MD it {it}] new best = {best_val:.6f}")
            # occasional deterministic reorder to shake things up
            if it % 1250 == 0:
                a_tmp = _alternating_reorder(a)
                val_tmp = _fast_evaluate(a_tmp)
                if val_tmp < best_val - 1e-12:
                    best_val, best_a = val_tmp, a_tmp.copy()
                    a = a_tmp.copy()
                    print(f"[MD reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing pairwise mass transfer
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.025
    cooling = 0.99993

    while time.time() < deadline - 70.0:
        conv = _full_convolution(cur_a)
        max_k = int(conv.argmax())

        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        contrib = cur_a[idx] * cur_a[jdx]

        # pair that most contributes to the peak
        pair_idx = np.argmax(contrib)
        i = idx[pair_idx]
        j = jdx[pair_idx]

        donor = i if cur_a[i] >= cur_a[j] else j

        # pick a low‑mass recipient (among the 5 % smallest entries)
        low = np.where(cur_a < np.percentile(cur_a, 5))[0]
        if low.size == 0:
            recipient = rng.integers(0, n)
        else:
            recipient = rng.choice(low)

        delta = cur_a[donor] * 0.035 * rng.random()      # up to 3.5 %
        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[recipient] += delta

        # renormalise
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a, cur_val = new_a, new_val
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, cur_a.copy()
                print(f"[pairhill] improved to {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # Phase 5 – final cheap polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final official check
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
