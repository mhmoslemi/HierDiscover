def propose_candidate(seed: int=42, budget_s: int=1000, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 5

    # choose n:
    n = 12000  # maybe adapt; if time insufficient later fallback.

    # start with old best if defined, else uniform.
    if 'height_sequence_1' in globals():
        old = np.asarray(height_sequence_1, dtype=float)
        if old.shape[0] != n:
            old_x = np.linspace(0,1, old.shape[0])
            new_x = np.linspace(0,1, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    else:
        a = np.full(n, 1.0 / n)

    # ensure sum to 1:
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-12:
        a = a / s
    else:
        a = np.full(n, 1.0/n)

    best_a = a.copy()
    best_val = _fast_evaluate(best_a)
    print(f"[init] n={n}, eval={best_val:.6f}")

    # Stage 0
    # candidate types
    candidate_generators = []

    # exponential
    candidate_generators.append(lambda: rng.exponential(scale=1.0,size=n))
    # uniform
    candidate_generators.append(lambda: np.full(n,1.0/n))
    # power law
    for p in (0.3,0.5,0.7,1.0,1.5,2.0):
        candidate_generators.append(lambda p=p: _powerlaw_candidate(p,n))
    # sinusoidal (sin^p)
    for p in (1,2,3):

    # random Dirichlet
    candidate_generators.append(lambda: rng.dirichlet(np.ones(n)))
    # maybe mixture of two spikes:
    candidate_generators.append(lambda: _two_spike_candidate(n))

    for gen in candidate_generators:
        if time.time() >= deadline - 200:
            break
        cand = gen()
        # ensure non-negative and sum=1
        cand = np.maximum(cand,0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.full(n,1.0/n)
        else:
            cand = cand / s
        val = _fast_evaluate(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            print(f"[stage0] new best = {best_val:.6f}")

    # Stage 1 subgradient descent
    if time.time() < deadline - 300:
        a_sgd, val_sgd = _subgradient_descent_multi(best_a, deadline)
        if val_sgd < best_val:
            best_val = val_sgd
            best_a = a_sgd.copy()
            print(f"[sgd] new best = {best_val:.6f}")

    # Stage 2 L-BFGS
    # We'll use over restarts
    restarts = 0
    while time.time() < deadline - 120 and restarts < 12:
        restarts += 1
        # perturb best
        w0 = np.log(best_a + 1e-12)
        w0 += rng.normal(scale=0.15, size=n)
        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[l-bfgs restart {restarts}] new best = {best_val:.6f}")

        # maybe anti-symmetric
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after l-bfgs {restarts}] new best = {best_val:.6f}")

    # Stage 3: simulated annealing
    if time.time() < deadline:
        best_a, best_val = _refine_simulated_anneal(best_a, best_val, deadline, n)

    # final evaluation using evaluate_sequence (the original) for reporting.
    final_val = evaluate_sequence(best_a.tolist())
    print(f"[final] evaluate_sequence -> {final_val:.6f}")

    # Return list
    return best_a.tolist()

def _alternating_reorder(a):
    s = np.sort(a)[::-1] # descending
    n = len(a)
    out = np.empty_like(a)
    # fill alternately from start and end
    left = 0
    right = n-1
    toggle = True
    for val in s:
        if toggle:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
        toggle = not toggle
    return out

"""
Propose a low‑value step function for the ``evaluate_sequence`` objective.

The algorithm works in four stages within a fixed time budget (default 1000 s)

1. **Initialisation** – pull a known good solution if available, otherwise
   create a few diverse candidates (uniform, exponential, power‑law,
   sinusoidal, Dirichlet).  Keep the best one.

2. **Exponentiated sub‑gradient descent** on the true L∞‑objective.
   The active set of convolution entries that are within a relative
   tolerance of the current maximum is used to form a sub‑gradient via a
   single FFT.  An exponential mirror‑descent update keeps the vector on
   the probability simplex.  The tolerance is tightened over time.

3. **Smooth‑max L‑BFGS optimisation** – the max‑convolution is replaced by
   a log‑sum‑exp surrogate with a decreasing smoothing parameter ε.
   The optimiser works on soft‑max parameters (guaranteeing non‑negativity
   and unit sum) and is restarted several times from perturbed copies of
   the current best.

4. **Simulated‑annealing hill‑climbing** – random swaps and mass‑transfer
   moves are proposed; uphill moves are accepted with a geometric cooling
   schedule to escape shallow local minima.

All updates keep the sequence non‑negative and normalised (∑a = 1); the
result is returned as a plain Python list of ``float``.  A fast internal
evaluation (identical to ``evaluate_sequence`` for normalised vectors) is
used throughout the search; the official ``evaluate_sequence`` is called
once at the very end for reporting.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth‑surrogate optimiser (L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative artefacts caused by floating‑point noise
    np.maximum(conv, 0.0, out=conv)
    return conv


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for normalised vectors (sum ≈ 1).
    Returns 2·n·max(conv) because the denominator (∑a)² cancels out.
    """
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = _full_convolution(a)
    max_b = conv.max()
    return 2.0 * a.shape[0] * max_b / (s * s)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values on the left, the smallest on the right,
    i.e. pair a large coefficient with a small one when the sequence is reversed.
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = s[n - 1 - i]     # large → left side
        res[n - 1 - i] = s[i]     # small → right side
    if n % 2:
        res[half] = s[half]
    return res


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """
    Interleave the sorted values: largest, smallest, second‑largest,
    second‑smallest, …  This often spreads large entries more evenly.
    """
    s = np.sort(a)[::-1]           # descending
    n = len(a)
    out = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for v in s:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


# ----------------------------------------------------------------------
# Power‑law and sinusoidal candidates (Stage 0)
# ----------------------------------------------------------------------
def _powerlaw_candidate(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (dist_to_end+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


def _sinusoid_candidate(p: float, n: int) -> np.ndarray:
    """Sinusoidal shape: a_i ∝ sin(π·(i+½)/n)^p, normalised."""
    idx = np.arange(n)
    a = np.sin(np.pi * (idx + 0.5) / n) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)                (a lies on the simplex)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = a.sum()
    a = a / a_sum

    # self‑convolution
    conv = _full_convolution(a)
    max_conv = conv.max()

    # smoothed max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t a
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # soft‑max over convolution entries

    # convolution a * reversed(p)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the middle n entries give ∂conv_k/∂a_i summed over active k
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                     # ∂L/∂a_i

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` – uses globals for eps and n."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _l_bfgs_optimize_from_w(w_init: np.ndarray, deadline: float):
    """
    Run a short L‑BFGS schedule starting from soft‑max parameters ``w_init``.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    n = w_init.shape[0]
    w = w_init.copy()

    # ε‑schedule: from 5·10⁻³ down to 10⁻¹² (geometric)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=11)

    best_a = None
    best_val = float('inf')

    for eps in eps_schedule:
        if time.time() >= deadline - 20.0:
            break
        global _GLOBAL_EPS, _GLOBAL_N
        _GLOBAL_EPS = eps
        _GLOBAL_N = n

        res = minimize(_smooth_obj_and_grad_wrapper,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 800,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # map back to simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _fast_evaluate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # anti‑symmetric re‑ordering heuristic
        a_ord = _anti_symmetric_reorder(a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS eps {eps:.1e}] improved to {best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        best_a = a
        best_val = _fast_evaluate(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------
def _active_set_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} a_{k‑i} using a single FFT.
    ``active_idx`` – indices (0 … 2n‑2) belonging to the active set.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return np.zeros_like(a)

    mask = np.zeros(2 * n - 1, dtype=np.float64)
    mask[active_idx] = 1.0
    mask_rev = mask[::-1]

    size = _next_pow2(n + mask_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fm = np.fft.rfft(mask_rev, size)
    conv_full = np.fft.irfft(fa * fm, size)[: n + mask_rev.shape[0] - 1]

    # g_i = conv_full[(2*n-2) - i] for i = 0 … n‑1
    g = conv_full[(2 * n - 2) - np.arange(n)]
    return g


def _subgradient_descent_multi(a_start: np.ndarray, deadline: float):
    """
    Exponentiated mirror‑descent that reduces the *active* convolution peaks.
    Returns the best (a, value) pair discovered before ``deadline``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _fast_evaluate(best_a)
    print(f"[sgd init] value = {best_val:.6f}")

    it = 0
    alpha0 = 0.8                     # initial step‑size factor
    rel_tol = 0.05                    # start with a generous active‑set tolerance

    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        max_conv = conv.max()

        # tighten tolerance gradually (logarithmic)
        if it > 0 and it % 25000 == 0:
            rel_tol = max(rel_tol * 0.5, 1e-6)

        thresh = max_conv * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # gradient via FFT (sub‑gradient of the L∞‑norm)
        g = _active_set_gradient(a, active)

        # decaying step‑size (mirror‑descent)
        alpha = alpha0 / np.sqrt(it + 1.0)

        # multiplicative update
        a = a * np.exp(-alpha * g)

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # logging & occasional re‑ordering
        if it % 200 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sgd iter {it}] new best = {best_val:.6f} (max {max_conv:.8e})")
            if it % 5000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[sgd reorder] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing based hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_simulated_anneal(a: np.ndarray, best_val: float,
                             deadline: float, n: int):
    """
    Random local moves (swap or mass transfer) with an annealing schedule.
    Only strictly better solutions are kept for the reported best,
    but occasional uphill moves are accepted to escape plateaus.
    """
    cur_a = a.copy()
    cur_val = best_val
    best_a = a.copy()
    best_val_local = best_val

    rng = np.random.default_rng()
    T0 = 0.025                     # initial temperature
    it = 0

    while time.time() < deadline:
        it += 1
        # propose a move
        if rng.random() < 0.15:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-18:
                continue
            # transfer up to 10 % of the source mass, scaled with temperature
            max_delta = cur_a[i] * 0.10
            delta = max_delta * rng.random() * (T0 / (T0 + it * 1e-3))
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val:
            accept = True
        else:
            # annealing acceptance probability
            T = T0 * (0.9997 ** it)
            accept = np.exp(-(val - cur_val) / max(T, 1e-12)) > rng.random()

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val_local - 1e-12:
                best_val_local = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val_local:.6f}")

        # occasional aggressive re‑ordering
        if it % 15000 == 0:
            cand = _alternating_reorder(cur_a)
            val_alt = _fast_evaluate(cand)
            if val_alt < best_val_local:
                best_val_local = val_alt
                best_a = cand.copy()
                cur_a = cand.copy()
                cur_val = val_alt
                print(f"[anneal reorder] improved to {best_val_local:.6f}")

    return best_a, best_val_local

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a slightly larger n gives a better asymptotic bound.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise from the known construction if it exists, otherwise uniform.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            old = np.asarray(height_sequence_1, dtype=float)
            if old.shape[0] != n:
                # linear interpolation to the required length
                old_x = np.linspace(0.0, 1.0, old.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                a = np.interp(new_x, old_x, old)
            else:
                a = old.copy()
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 1e-12 else np.full(n, 1.0 / n)

    best_a = a.copy()
    best_val = _fast_evaluate(best_a)
    print(f"[init] n={n}, fast evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – cheap diverse candidates
    # ------------------------------------------------------------------
    # 1) exponential random vectors
    for tr in range(4):
        if time.time() >= deadline - 250:
            break
        cand = rng.exponential(scale=1.0, size=n)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _fast_evaluate(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand exp {tr}] new best = {best_val:.6f}")

    # 2) uniform vector (already covered but keep for completeness)
    if time.time() < deadline - 240:
        cand = np.full(n, 1.0 / n)
        val = _fast_evaluate(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[uniform] new best = {best_val:.6f}")

    # 3) power‑law shapes
    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        if time.time() >= deadline - 230:
            break
        cand = _powerlaw_candidate(p, n)
        val = _fast_evaluate(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[power‑law p={p:.1f}] new best = {best_val:.6f}")

    # 4) sinusoidal shapes
    for p in (1, 2, 3):
        if time.time() >= deadline - 220:
            break
        cand = _sinusoid_candidate(p, n)
        val = _fast_evaluate(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[sinusoid p={p}] new best = {best_val:.6f}")

    # 5) Dirichlet (almost uniform but with fluctuations)
    if time.time() < deadline - 210:
        cand = rng.dirichlet(np.ones(n))
        val = _fast_evaluate(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[dirichlet] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – sub‑gradient descent on the true max‑convolution objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:
        a_sgd, val_sgd = _subgradient_descent_multi(best_a, deadline)
        if val_sgd < best_val:
            best_val, best_a = val_sgd, a_sgd.copy()
            print(f"[sub‑gradient] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑surrogate optimisation (L‑BFGS) with several restarts
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 120 and restarts < 12:

        a_opt, val_opt = _l_bfgs_optimize_from_w(w0, deadline)
        if val_opt < best_val:

            print(f"[restart {restarts}] L‑BFGS improved to {best_val:.6f}")

        # anti‑symmetric sweep
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()

    # ------------------------------------------------------------------
    # Stage 3 – final refinement via simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:

    # ------------------------------------------------------------------
    # Final evaluation & return (use the official function for reporting)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
