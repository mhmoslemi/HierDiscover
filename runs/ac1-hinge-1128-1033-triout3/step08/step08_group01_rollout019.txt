def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    # Choose n based on time budget
    n = 2000  # moderate for speed

    # init a uniform
    a = np.full(n, 1.0 / n)

    # compute convolution b via FFT
    b = _full_convolution(a)

    best_a = a.copy()
    best_val = 2.0 * n * b.max()
    # print initial

    # Phase annealing
    T0 = 0.5  # initial temperature relative to objective magnitude
    # may adapt: T0 = best_val * 0.1

    while time.time() < deadline:
        # schedule temperature decreasing: linear
        elapsed = time.time() - start_time
        t_frac = elapsed / (deadline - start_time)
        T = T0 * (1 - t_frac)  # linearly decrease to zero

        # choose i and j
        # sample i weighted by contribution to b_max (like hill)
        # compute contribution for current max shift

        k_max = int(np.argmax(b))
        # contributions for each index to this max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0 and np.random.rand()<0.7:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = np.random.randint(n)

        # Choose j uniformly distinct from i
        j = (i + np.random.randint(1, n)) % n

        # Determine max transfer fraction
        max_frac = 0.1 * (1 - t_frac) + 0.001 * t_frac  # start larger
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0:
            continue

        # Update a and b incrementally
        # b_new = b.copy()
        # for each shift s:
        #   b_new[s + i] -= 2*delta*a[s] (since we move mass away from i)
        # Actually we can use similar update as hillclimb: 
        # b_new[j:j+n] += 2*delta*a
        # b_new[i:i+n] -= 2*delta*a
        # Plus delta terms self: 
        # b_new[2*j] += delta*delta
        # b_new[2*i] -= delta*delta (since we removed)
        # b_new[i+j] -= delta*delta

        b_new = b.copy()
        # additive updates
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        # adjust terms for delta^2
        b_new[2*j] += delta * delta
        b_new[2*i] -= delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_obj = 2.0 * n * new_max

        # accept?
        if new_obj < best_val:
            accept = True
        else:
            # acceptance probability
            # if T > 0 else False
            if T > 0:
                prob = np.exp(-(new_obj - best_val) / T)
                accept = np.random.rand() < prob
            else:
                accept = False

        if accept:
            # update a,b,obj
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_obj
            if new_obj < best_val:
                best_val = new_obj
                best_a = a.copy()
                print(f"[SA] new best = {best_val:.6f} time={time.time()-start_time:.1f}s")
        # else ignore

        # occasionally renormalize to sum=1 (due to floating errors)
        if np.abs(a.sum() - 1.0) > 1e-12:
            a /= a.sum()
            b = _full_convolution(a)

    # after loop, maybe refine with Adam as earlier for few iterations?

    # final: ensure normalization
    best_a /= best_a.sum()
    return best_a.tolist()

def _full_convolution(a: np.ndarray) -> np.ndarray:

    fa = np.fft.rfft(a, size)

def _initial_uniform(n: int) -> np.ndarray:

        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:                       # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
