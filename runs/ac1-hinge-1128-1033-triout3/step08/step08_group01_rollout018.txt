def propose_candidate(seed=..., budget_s=1000, **kwargs):
    import time, numpy as np
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 9000

    best_a = None
    best_val = float('inf')

    # uniform base
    uniform = np.full(n, 1.0 / n)

    # function to evaluate candidate arr (already sum=1)
    def eval_arr(arr):
        # compute via FFT
        conv = _full_convolution(arr)
        maxb = conv.max()
        return 2.0 * n * maxb

    # Exponential search
    for alpha in np.linspace(0.5, 6.0, 111): # step 0.055
        if time.time() > deadline:
            break
        r = 1.0 - alpha / n
        if r <= 0:
            continue
        a = np.power(r, np.arange(n, dtype=np.float64))
        a_sum = a.sum()
        a /= a_sum  # normalize
        # optionally mix with uniform
        for w in [0.0, 0.001, 0.003, 0.005, 0.01]:
            if time.time()>deadline:
                break
            cand = (1-w) * a + w * uniform
            # cand sum = 1
            val = eval_arr(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                print(f"[grid] alpha={alpha:.3f}, w={w} -> {val:.6f}")
                if val <= 1.503:
                    # maybe early exit
                    pass

    # Power-law search
    for p in np.linspace(0.5, 5.0, 46):
        if time.time()>deadline:
            break
        a = 1.0 / (np.arange(1, n+1, dtype=np.float64) ** p)
        a /= a.sum()
        for w in [0.0, 0.001, 0.003, 0.005, 0.01]:
            if time.time()>deadline:
                break
            cand = (1-w) * a + w * uniform
            val = eval_arr(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                print(f"[grid power] p={p:.3f}, w={w} -> {val:.6f}")

    # Additional random restarts: draw Dirichlet and hillclimb a few steps
    rng = np.random.default_rng(int(time.time()) % 2**32)
    while time.time() < deadline - 5:  # keep some time for hillclimb
        a = rng.dirichlet(np.ones(n))
        # simple local smoothing: apply some gradient descent steps via subgradient
        # let's run limited subgradient phase for a few iterations
        a = _subgradient_phase_limited(a, n, start=time.time(), deadline=deadline - 2, max_iter=2000)[0] # we need a limited version
        val = eval_arr(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[rand] dirichlet improved -> {val:.6f}")

    # final mixing with uniform fine-grained
    best_a, best_val = _mix_with_uniform(best_a, n) # using existing function

    # final evaluate
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] best evaluation reported =", final_score)
    except Exception as e:
        print("[final] error", e)
    return best_a.tolist()

"""
propose_candidate
-----------------
A fast, deterministic optimizer for the max‑convolution objective

1.  Build a diverse set of parametric seed vectors (uniform,
    geometric, power‑law, mixtures with a tiny uniform component,
    a few random Dirichlet draws) – all normalized to Σa=1.
2.  Keep the best seed (lowest exact objective) as a starting point.
3.  Run an aggressive hill‑climbing phase that repeatedly transfers a
    small amount of mass from indices that contribute heavily to the
    current maximal convolution entry to indices that hardly contribute.
    The step size is annealed from 100 % of the source mass down to
    0.1 % over the remaining budget.
4.  Finish with a short uniform‑mix sweep (α∈[0,0.02]) – in practice
    this often reduces the peak a little more.
5.  Return the resulting list; all numbers are non‑negative and sum to 1,
    which makes the internal evaluation `2·n·max(conv(a,a))` identical
    to ``evaluate_sequence`` (which divides by the squared sum).

The code uses only NumPy and no closures/lambdas, fits into the 1000 s
budget and prints progress for debugging.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# --------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# --------------------------------------------------------------
# Exact objective (assumes sum(a) == 1)
# --------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalized to sum 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# --------------------------------------------------------------
# Simple building blocks
# --------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex mixtures (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a.copy(), _objective(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        # already normalized (both summands sum to 1)
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# --------------------------------------------------------------
# Generate a collection of simple parametric seeds
# --------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a list of candidate vectors (already normalized)."""
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n))

    # geometric families: a_i ∝ r^i, with r = 1 - α/n
    for alpha in np.linspace(0.5, 6.0, 111):          # α from 0.5 … 6.0
        r = 1.0 - alpha / n
        if r <= 0.0:
            continue
        a = r ** np.arange(n, dtype=np.float64)
        a /= a.sum()
        seeds.append(a)

    # power‑law families: a_i ∝ (i+1)^{-p}
    for p in np.linspace(0.2, 5.0, 49):
        a = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        a /= a.sum()
        seeds.append(a)

    # mixtures of the geometric families with a tiny uniform component
    uniform = np.full(n, 1.0 / n)
    for alpha in (1.0, 2.0, 3.0, 4.0):
        r = 1.0 - alpha / n
        if r <= 0.0:
            continue
        base = r ** np.arange(n, dtype=np.float64)
        base /= base.sum()
        for w in (0.001, 0.003, 0.005, 0.010):
            seeds.append((1.0 - w) * base + w * uniform)

    # a handful of random Dirichlet draw (low‑dimensional, diverse)
    for _ in range(12):
        a = rng.dirichlet(np.ones(n))
        seeds.append(a)

    return seeds


# --------------------------------------------------------------
# Aggressive hill‑climbing on the exact objective
# --------------------------------------------------------------
def _hillclimb(a_start: np.ndarray,
               n: int,
               start: float,
               deadline: float,
               best_a: np.ndarray,
               best_val: float,
               target_val: float = 1.503) -> np.ndarray:
    """Move mass from high‑contributing indices to low‑contributing ones."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    max_frac_start, max_frac_end = 1.0, 0.001          # aggressive start
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # linear schedule for the maximal transferable fraction
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contributions of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # pick a source index proportional to its contribution
        if total_contrib > 0.0:
            prob_src = contrib / total_contrib
            i = rng.choice(n, p=prob_src)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            # nothing to take from this index, try next iteration
            continue

        # pick a target index that contributes little to the max entry
        # (reverse‑convolution view)
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0.0:
            prob_tgt = w / w.sum()
            j = rng.choice(n, p=prob_tgt)
        else:
            # fallback: any other index
            j = (i + rng.integers(1, n)) % n

        # amount to transfer – a fraction of the source mass
        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n) vectorised work)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # accept improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # occasional large perturbation to escape flat plateaus
        if no_improve > IMPROVE_LIMIT:
            # restart from the best solution so far with a small jitter
            a = best_a.copy()
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 9000

    # ------------------------------------------------------------------
    # 1️⃣  Build a rich set of parametric seeds and keep the best one.
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng)
    best_a = None
    best_val = float("inf")
    for idx, cand in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Seed {idx:03d}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Aggressive hill‑climbing on the exact objective.
    # ------------------------------------------------------------------
    hill_deadline = deadline   # use the rest of the budget
    best_a = _hillclimb(
        best_a, n,
        start=start_time,
        deadline=hill_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )
    best_val = _objective(best_a)

    # ------------------------------------------------------------------
    # 3️⃣  Final tiny uniform mixing (often reduces the peak a bit).
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # 4️⃣  Final evaluation through the provided wrapper.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
