def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Compute gradient g_i = 2 * sum_j a_j * w_{i+j}
    where w_rev is reversed w vector (length 2n-1).
    We can compute conv = np.convolve(a, w_rev) (full length n+2n-2)
    Then g = 2 * conv[2n-2:2n-2 + n] (since w_rev length 2n-1).
    Use FFT for speed.

    start = len(w_rev) - 1 # 2n-2

"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  

The algorithm consists of three phases:

* **Initialisation** – use the historic construction if present,
  otherwise start from a random Dirichlet vector (exponential draws normalized
  to sum 1).

* **Smoothed exponential‑mirror descent** – at each iteration we compute the
  self‑convolution `c = a * a`.  Instead of taking only the peak(s) we build a
  smooth approximation of the maximum by weighting every entry with
  `exp(beta·(c‑max(c)))`.  The resulting weighted sum is differentiable:
  ∂/∂aᵢ  Σₖ wₖ·cₖ = 2·(a * w_rev)[i]  (linear convolution, computed
  with a single FFT).  We update the probability vector multiplicatively
  (`a ← a·exp(‑α·g)`) and renormalise onto the simplex.  The step size `α`
  decays slowly, while the smoothing parameter `β` is increased from 1 to a
  few hundred, thus gradually focusing the descent onto the true maximum.

* **Hill‑climbing refinement** – after the main descent finishes we perform
  a cheap stochastic local search (random swaps and tiny mass transfers)
  that accepts only strict improvements.

All heavy work (FFT‑based gradient updates) is performed without calling
`evaluate_sequence`; the objective is queried only periodically for logging
and to keep the best solution found so far.  The whole procedure respects the
1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)      # eliminate tiny negatives
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of Σₖ wₖ·(a*a)ₖ w.r.t. `a` using convolution.
    The output is gᵢ = 2·Σⱼ aⱼ·wᵢⱼ, where wᵢⱼ = w_{i+j}.
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1          # = 3n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    start = w_rev.shape[0] - 1                 # 2n‑2
    seg = conv[start:start + n]                # length n
    return 2.0 * seg
