def generate_sparse_set(n, max_m, C_max, rng):
    sum_counts = np.zeros(2*n-1, dtype=int)  # for unordered counts
    indices = []
    candidate_pool = list(range(n))
    rng.shuffle(candidate_pool)
    for i in candidate_pool:
        # compute number of pairs new with each existing j
        # check compatibility
        ok = True
        # self pair count: sum_counts[2*i] will be increased by 1
        if sum_counts[2*i] + 1 > C_max:

"""
Sparse‑set construction for the auto‑convolution peak minimisation problem.

The evaluation function is
    score = 2·n·max(conv(a,a)) / (sum(a)**2)

For a probability vector (sum = 1) the denominator disappears and we only have to
minimise the peak of the convolution.  If the support consists of m indices and
every possible sum of two indices occurs at most C times (counting the i+i
self‑pair), then each convolution entry is bounded by
    (2·C−1) / m²                      (ordered pairs ≤ 2·C−1)

so the score is at most 2·n·(2·C−1)/m².  Choosing a moderate bound C=3 and a
support size m≈160 already yields a theoretical score below the target
1.503.  The algorithm below repeatedly builds random sparse supports that obey
the bound C, evaluates the corresponding uniform‑weight vector and keeps the
best one.  A cheap mirror‑descent refinement is applied at the end.

The whole search respects a 1000 s budget; it falls back to the uniform
distribution if no better candidate is found.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper functions (FFT‑based full convolution and utilities)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑2 FFT.
    The result length is 2·len(a)−1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # tiny negative values can appear due to round‑off
    np.maximum(conv, 0, out=conv)
    return conv


# --------------------------------------------------------------
# Sparse support construction
# --------------------------------------------------------------
def _greedy_sparse_set(n: int, target_m: int, C_max: int, rng: np.random.Generator):
    """
    Build a set of indices S ⊆ {0,…,n‑1} with |S|≈target_m such that
    for every possible sum s=i+j (i,j∈S) the number of unordered pairs that
    realise s never exceeds C_max.
    Returns the list of chosen indices (may be shorter than target_m).
    """
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)   # unordered pair counters
    indices = []
    # Random permutation gives diverse candidates
    order = rng.permutation(n)
    for i in order:
        # fast reject if the self‑pair would violate the bound
        if sum_counts[2 * i] + 1 > C_max:
            continue
        ok = True
        for j in indices:
            if sum_counts[i + j] + 1 > C_max:
                ok = False
                break
        if not ok:
            continue
        # accept i: update all newly created sums
        for j in indices:
            sum_counts[i + j] += 1          # unordered (i,j) – (j,i) counted once
        sum_counts[2 * i] += 1              # the diagonal pair (i,i)
        indices.append(i)
        if len(indices) >= target_m:
            break
    return indices, sum_counts


# --------------------------------------------------------------
# Very cheap evaluation for a probability vector (sum == 1)
# --------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv) for a vector that already sums to 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())

# Main entry point required by the competition harness
# --------------------------------------------------------------

    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method builds large, very sparse supports (≤3 collisions per sum), then
    refines the result with a few mirror‑descent steps.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a few seconds margin for the final official check
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine problem size (use the length of a previous solution if present)
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)      # type: ignore[name-defined]
        n = int(old.shape[0])
        print(f"[init] re‑using previous length n={n}")
        # normalise the previous vector to sum = 1 for a fair baseline
        old_norm = old / old.sum()
        best_vec = old_norm.copy()
        best_val = evaluate_sequence(best_vec.tolist())
        print(f"[init] previous best score = {best_val:.6f}")
    except Exception:
        n = 4000
        best_vec = np.full(n, 1.0 / n)   # uniform baseline (score = 2)
        best_val = 2.0
        print("[init] no previous solution – start with uniform (score = 2)")

    # --------------------------------------------------------------
    # Phase 1 – random greedy construction of a low‑collision sparse set
    # --------------------------------------------------------------
    # We try a small set of collision budgets; C=3 is the hardest but gives

    return best_vec.tolist()
