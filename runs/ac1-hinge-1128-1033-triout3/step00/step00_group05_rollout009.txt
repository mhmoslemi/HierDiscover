def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1 # small margin
    # Use height_sequence_1 if exists else random
    best_sequence = list(height_sequence_1) if 'height_sequence_1' in globals() else [np.random.rand() for _ in range(500)]
    best_score = evaluate_sequence(best_sequence)
    curr_sequence = best_sequence.copy()
    
    while time.time() < deadline:
        # Compute current M and conv
        conv_f = np.convolve(curr_sequence, curr_sequence)
        M = conv_f.max()
        # Determine active constraints: top K lags with conv close to M
        # We pick indices where conv_f >= M*0.9 (some fraction)
        thresh = M * 0.95
        active_idx = np.where(conv_f >= thresh)[0]
        # Limit to at most max_constraints
        max_constraints = 200
        if len(active_idx) > max_constraints:
            # Random sample
            active_idx = np.random.choice(active_idx, size=max_constraints, replace=False)
        # Solve LP for g
        g = solve_convolution_lp_subset(curr_sequence, M, active_idx)
        if g is not None:
            # Normalize g sum to match current sum? not needed but we keep g as is.
            # Evaluate mixing with several alphas.
            for alpha in [0.01, 0.05, 0.1, 0.2, 0.5]:
                cand = [(1 - alpha) * x + alpha * y for x, y in zip(curr_sequence, g)]
                # clip to non-negative (linprog ensures >=0)
                cand = [max(0.0, v) for v in cand]
                score = evaluate_sequence(cand)
                if score < best_score - 1e-12:
                    best_score = score
                    best_sequence = cand[:]
                    curr_sequence = cand[:]
                    break
            else:
                # If none improved, do small random perturbation
                idx = np.random.randint(len(curr_sequence))
                cand = curr_sequence[:]
                cand[idx] = max(0, cand[idx] + np.random.randn() * 0.01 * np.mean(cand))
                score = evaluate_sequence(cand)
                if score < best_score:
                    best_score = score
                    best_sequence = cand[:]
                    curr_sequence = cand[:]
        else:
            # Fallback to random perturbation
            idx = np.random.randint(len(curr_sequence))
            delta = np.random.randn() * 0.01
            new_seq = curr_sequence[:]
            new_seq[idx] = max(0, new_seq[idx] + delta)
            score = evaluate_sequence(new_seq)
            if score < best_score:
                best_score = score
                best_sequence = new_seq[:]
                curr_sequence = new_seq[:]
        # Occasionally random restart
        if np.random.rand() < 0.001:
            # random restart using best sequence plus noise
            curr_sequence = [max(0, x + np.random.randn()*0.05) for x in best_sequence]
    # After loop, return best_sequence
    return best_sequence

def solve_convolution_lp_subset(f_seq, M, ks):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum(b)
    # Build sparse matrix A_ub and vector b_ub
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(ks):
        # compute range of j values
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        # For j in [j_min, j_max], i = k - j
        # coefficient = f_seq[i]
        # i in range 0..n-1
        for j in range(j_min, j_max+1):
            i = k - j
            rows.append(row_idx)
            cols.append(j)
            data.append(f_seq[i])
    # Add nonnegativity constraints: -I <= 0
    # Create identity rows after the above rows.
    # We'll append n rows: with -1 on diagonal
    # for each variable
    N_constraints = len(ks) + n
    row_ind = np.array(rows + list(range(len(ks), N_constraints)))
    col_ind = np.array(cols + list(range(n)))
    data_arr = np.array(data + [-1.0]*n)
    A_ub = scipy.sparse.csr_matrix((data_arr, (row_ind, col_ind)), shape=(N_constraints, n))
    b_ub = np.concatenate([np.full(len(ks), M), np.zeros(n)])
    # Call linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'time_limit': 2.0})
    if res.success:
        return np.maximum(0.0, res.x)  # ensure nonnegative
    else:
        return None

Simple iterative improvement using a sparse LP direction.
Starts from height_sequence_1 (if present) or a random uniform vector.
In each iteration we:
- compute current max convolution M
- pick the lag indices where the convolution is close to M
- solve a linear program that maximizes the sum of a new vector g under the
  constraints (f * g)[k] <= M for the selected lags.
- keep only those g that satisfy all lag constraints (checked explicitly)
- form several convex combinations of f and g with different mixing weights,
  evaluate them and keep the best.
- occasionally perform a small random perturbation to escape plateaus.
The algorithm runs for the allotted time budget and returns the best
sequence found.
"""

import time
import numpy as np
from scipy import optimize, sparse

def solve_convolution_lp_subset(f_seq, M, lag_indices, max_time=1.0):
    """
    Solve the linear program:
        maximize sum(b)
        subject to conv(f_seq, b)[k] <= M   for k in lag_indices
                    b >= 0
                    b <= 1000
    Returns the optimal b as a 1‑D numpy array, or None if the solver failed.
    """
    n = len(f_seq)
    # Objective: maximize sum(b)  -> minimise -sum(b)
    c = -np.ones(n, dtype=float)

    # Build sparse constraint matrix A_ub * b <= b_ub
    rows = []
    cols = []
    data = []

    # Build rows for each selected lag
    for row, k in enumerate(lag_indices):
        # j ranges where 0 <= j < n and i = k-j also in [0,n)
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        # For each j, coefficient = f_seq[i] where i = k - j
        for j in range(j_min, j_max + 1):
            i = k - j
            rows.append(row)
            cols.append(j)
            data.append(f_seq[i])

    # Non‑negativity handled by bounds; also add upper bound 1000 using bounds
    # Add the -b <= 0 rows for explicit feasibility check with low overhead.
    # They are not strictly needed when we set bounds but we keep them to
    # give the solver a clear representation of b >= 0.
    base = len(lag_indices)
    for i in range(n):
        rows.append(base + i)
        cols.append(i)
        data.append(-1.0)   # -b_i <= 0

    nr_constraints = len(lag_indices) + n
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(nr_constraints, n))

    b_ub = np.concatenate([np.full(len(lag_indices), M), np.zeros(n)])

    bounds = [(0.0, 1000.0) for _ in range(n)]

    # Use the HiGHS interior‑point method; it supports sparse matrices.
    res = optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': max_time}
    )
    if res.success:
        # Guard against tiny negative values caused by numerical noise.
        return np.maximum(0.0, res.x)
    return None

def convex_mix(f, g, alphas):

Iterative improvement of a non‑negative step function.

The algorithm starts from the provided `height_sequence_1` (or a uniform
fallback) and repeatedly performs the following steps:

1. Compute the current maximal self‑convolution `M`.
2. Select a modest set of lag indices where the convolution is close to `M`.
3. Solve a *sparse* linear program that maximises the sum of a new vector
   `g` while keeping the cross‑convolution `conv(f, g)` below `M` on the
   selected lags.  The LP has `len(active_lags)` constraints and is
   solved with HiGHS (supports sparse matrices).
4. Verify that `g` respects *all* lag constraints (full convolution).
5. Form several convex combinations ` (1‑α)·f + α·g ` for a small list of
   α‑values, evaluate each candidate with the given `evaluate_sequence`,
   and keep the best one.
6. If no improvement is found, apply a tiny random perturbation to escape
   plateaus.  Occasionally a noisy restart around the current best sequence
   is performed.

The loop runs until the allotted time budget expires (default 1000 s) and
returns the best sequence discovered.
"""

import time
import numpy as np
from scipy import optimize, sparse

# ----------------------------------------------------------------------
# Helper: sparse linear program (maximise sum(g) s.t. conv(f,g)[k] <= M)
# ----------------------------------------------------------------------
def solve_convolution_lp_subset(f_seq, M, lag_indices, max_time=0.8):
    """
    Solve  max sum(g)  subject to  (f_seq * g)[k] <= M  for k in lag_indices,
    with g >= 0 and g <= 1000.
    The constraints are built as a sparse matrix and handed to HiGHS.
    Returns a numpy array `g` on success, otherwise None.
    """
    n = len(f_seq)
    # objective: minimise -sum(g)  (equivalent to maximising sum(g))
    c = -np.ones(n, dtype=float)

    # ----- build sparse A_ub for the selected lag constraints -----
    rows = []
    cols = []
    data = []
    for row, k in enumerate(lag_indices):
        # j runs over indices of g that appear in conv(f,g)[k]
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            i = k - j            # the matching index of f_seq
            rows.append(row)
            cols.append(j)
            data.append(f_seq[i])

    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(len(lag_indices), n))
    b_ub = np.full(len(lag_indices), M, dtype=float)

    # bounds for each variable: 0 <= g_i <= 1000  (the same clipping
    # performed later by evaluate_sequence)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # solve with HiGHS; it understands sparse matrices
    res = optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': max_time}
    )
    if res.success:
        # numerical noise may produce tiny negatives – clamp to zero
        return np.maximum(0.0, res.x)
    return None

    """

    `max_lags` entries (largest values kept).

    max_val = np.max(conv_vals)
    # keep lags with value >= (1 - rel_gap) * max_val

        cur = np.array(height_sequence_1, dtype=float)
    else:
        # fallback: a modest uniformly‑valued sequence
        length = np.random.randint(300, 900)
        cur = np.full(length, 1.0, dtype=float)

    # Enforce the evaluator’s hard clipping limits
    cur = np.clip(cur, 0.0, 1000.0)

    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] score={best_score:.6f}, length={len(best_seq)}")

    it = 0
    stagnated = 0          # counts iterations without improvement

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- 1. current max convolution and active lags -----
        conv_f = np.convolve(cur, cur)
        M = float(conv_f.max())
        if M == 0.0:          # degenerate – should never happen
            break

        active = locate_active_lags(conv_f, rel_gap=0.05, max_lags=120)

        # ----- 2. LP direction -------------------------------------------------
        g = solve_convolution_lp_subset(cur, M, active, max_time=0.6)

        if g is None:
            # LP failed – fall back to a tiny random jiggle
            idx = np.random.randint(len(cur))
            delta = np.random.randn() * 0.02 * max(1.0, np.mean(cur))
            cand = cur.copy()
            cand[idx] = max(0.0, cand[idx] + delta)
            cur = cand
            stagnated += 1
            continue

        # ----- 3. sanity‑check full cross‑convolution ----- 
        if np.max(np.convolve(cur, g)) > M * 1.001:
            # the sparse LP ignored a critical lag – discard this g
            g = None

        # ----- 4. try convex mixes of current and g ----- 
        if g is not None:
            improved = False
            # a few α values; smaller α keeps us close to the safe region,
            # larger α exploits any extra sum that g may provide
            for α, candidate in convex_mixtures(cur.tolist(), g.tolist(),
                                                alphas=[0.01, 0.04, 0.07,
                                                        0.12, 0.20, 0.35]):
                score = evaluate_sequence(candidate)
                if score < best_score:
                    best_score = score
                    best_seq = np.array(candidate, dtype=float)
                    cur = best_seq.copy()
                    improved = True
                    stagnated = 0
                    print(f"[it {it:4d}] improved: score={best_score:.6f} (α={α:.2f})")
                    break   # accept the first improving α

            if not improved:
                # no α yielded improvement → small random perturbation
                idx = np.random.randint(len(cur))
                delta = np.random.randn() * 0.02 * max(1.0, np.mean(cur))
                cand = cur.copy()
                cand[idx] = max(0.0, cand[idx] + delta)
                cur = np.clip(cand, 0.0, 1000.0)
                stagnated += 1
        else:
            # g unavailable – random local move
            idx = np.random.randint(len(cur))
            delta = np.random.randn() * 0.03 * max(1.0, np.mean(cur))
            cand = cur.copy()
            cand[idx] = max(0.0, cand[idx] + delta)
            cur = np.clip(cand, 0.0, 1000.0)
            stagnated += 1

        # ----- 5. occasional noisy restart to escape long plateaus -----
        if stagnated >= 30 and it % 150 == 0:
            noise = np.random.randn(*best_seq.shape) * 0.01 * max(1.0, np.mean(best_seq))
            cur = np.clip(best_seq + noise, 0.0, 1000.0)
            stagnated = 0
            print(f"[it {it:4d}] noisy restart")

    # ------------------------------------------------------------------
    print(f"[done] iterations={it}, best_score={best_score:.6f}")
    return best_seq.tolist()
