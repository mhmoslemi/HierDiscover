"""
Simple simulated‑annealing improvement for the “step‑function” challenge.

The algorithm starts from the classic inverse‑square‑root shape

        a_i = 1 / sqrt(i+1)

which already gives a fairly flat self‑convolution (value ≈ π, giving a score
≈ π/2 ≈ 1.57).  We then iteratively move a small amount of mass from an index
that participates in the current maximal convolution entry to a random other
index.  The move keeps the total sum unchanged (the objective is scale‑invariant)
and usually reduces the peak.  A temperature schedule permits occasional uphill
steps to escape local plateaus.

The procedure runs until the allocated time (≈ budget_s – a few seconds) is
exhausted and returns the best sequence seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def convolution_peak(seq: np.ndarray):
    """
    Return the maximal convolution value and the index where it occurs.
    """
    conv = np.convolve(seq, seq)
    idx = int(np.argmax(conv))
    peak = float(conv[idx])
    return peak, idx, conv


def propose_move(seq: np.ndarray, k_max: int, max_move_frac: float = 0.2):
    """
    Propose a new sequence by moving a random fraction (up to max_move_frac)
    of the mass from a randomly chosen index that is part of the peak
    (i + j = k_max) to a random destination index.
    The total sum is preserved.
    """
    n = len(seq)

    # viable i such that 0 <= i <= n-1 and 0 <= j = k_max-i <= n-1
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low > i_high:        # should never happen
        return None

    i = np.random.randint(i_low, i_high + 1)
    j = k_max - i

    # pick a destination index different from i (and preferably far from i)
    p = np.random.randint(0, n)
    while p == i:
        p = np.random.randint(0, n)

    # amount to move – a fraction of the source value
    src_val = seq[i]
    if src_val <= 0.0:
        return None
    max_delta = src_val * max_move_frac
    delta = np.random.rand() * max_delta

    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[p] += delta
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget (seconds). The function returns before this limit.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)

    # --------------------------------------------------------------
    # 1.  Choose a moderate length – large enough for a good bound,
    #     small enough for fast convolutions.
    # --------------------------------------------------------------
    n = 2500                     # you may tune this (2000‑3000 works well)

    # --------------------------------------------------------------
    # 2.  Initialise with the inverse‑sqrt shape (already decent).
    # --------------------------------------------------------------
    seq = np.array([1.0 / math.sqrt(i + 1) for i in range(n)], dtype=np.float64)
    # add a tiny random jitter to break perfect symmetry
    seq += 0.001 * np.random.rand(n)

    # evaluate the initial sequence
    best_score = evaluate_sequence(seq.tolist())
    best_seq = seq.copy()
    cur_score = best_score

    # --------------------------------------------------------------
    # 3.  Simulated‑annealing loop.
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a safety margin
    iteration = 0

    # temperature schedule parameters
    T_start = 0.15
    T_end = 0.001

    while time.time() < deadline:
        # compute the current convolution peak
        peak, k_max, _ = convolution_peak(seq)

        # propose a move that tries to lower the peak
        new_seq = propose_move(seq, k_max, max_move_frac=0.15)
        if new_seq is None:
            # fallback: a tiny random perturbation
            idx = np.random.randint(n)
            sigma = max(1e-6, seq[idx] * 0.03)
            new_seq = seq.copy()
            new_seq[idx] = max(0.0, new_seq[idx] + np.random.randn() * sigma)

        # evaluate the candidate
        new_score = evaluate_sequence(new_seq.tolist())
        if math.isinf(new_score):
            # discard illegal candidates
            continue

        # temperature (linear decay)
        elapsed = time.time() - start_time
        T = T_start + (T_end - T_start) * (elapsed / budget_s)

        # acceptance rule (Metropolis)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability to accept a worse move
            prob = math.exp((cur_score - new_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            seq = new_seq
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[{elapsed:6.2f}s] New best score: {best_score:.6f}")

        # occasional random shake‑up to avoid stagnation
        if iteration % 800 == 0 and iteration > 0:
            shake_idx = np.random.randint(n)
            shake = max(1e-8, seq[shake_idx] * 0.1)
            seq[shake_idx] = max(0.0, seq[shake_idx] + np.random.randn() * shake)

        iteration += 1

    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
