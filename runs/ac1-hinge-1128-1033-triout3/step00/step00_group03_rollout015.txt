"""
Propose a step‑function sequence that minimizes the evaluation function.

The algorithm proceeds in three stages:

1. **Analytic families** – explore a few simple parametric families that are
   known to give low convolution peaks:
   * inverse‑square‑root shape   a_i = 1/√(i+offset+1)
   * power‑law shape            a_i = (i+1)^{-p}
   * geometric decay            a_i = r^i  (optimal r found by ternary search)
   For each family a handful of parameter values and several lengths are
   tried; the best candidate so far is kept.

2. **Local hill‑climbing** – starting from the best sequence obtained above,
   repeatedly pick two positions and transfer a small random amount of weight
   from one to the other (keeping the entries non‑negative).  Accept the move
   whenever the evaluation improves.

3. **LP refinement** – use the LP routine from the reference implementation.
   Given a sequence `f` we solve a linear program that finds a sequence `g`
   with maximal ℓ₁‑norm under the current convolution bound.  Mixing `f` and
   `g` (with a modest step size) strictly reduces the maximum convolution
   value.  Several LP steps are performed while time permits.

The whole search runs for the allotted time (default 1000 s) and finally
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: linear program that maximises the sum of a non‑negative vector g
# while keeping the convolution f * g bounded by rhs.
# ----------------------------------------------------------------------
def solve_convolution_lp(f_sequence, rhs):
    """Solve: maximise sum(g)  s.t.  conv(f,g) <= rhs, g >= 0."""
    n = len(f_sequence)
    c = -np.ones(n)                       # maximise → minimise -sum(g)

    # Build the inequality matrix for the convolution constraints.
    # For each k = 0 … 2n‑2 the row contains the coefficients of g_j
    # in (f * g)_k = sum_i f_i * g_{k-i}.
    a_rows = []
    b_vals = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # i runs over all positions where f_i participates in (f*g)_k
        for i in range(i_low, i_high + 1):
            j = k - i
            row[j] = f_sequence[i]
        a_rows.append(row)
        b_vals.append(rhs)

    # Non‑negativity constraints for g:  -g <= 0
    a_ub = np.vstack([a_rows, -np.eye(n)])
    b_ub = np.hstack([b_vals, np.zeros(n)])

    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


# ----------------------------------------------------------------------
# Helper: one LP step that moves the current sequence towards a better one.
# ----------------------------------------------------------------------
def get_good_direction_to_move_into(sequence):
    """Return a new sequence that is a small convex combination with a LP‑found g."""
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-9:
        return None

    # Normalise to the standard sum sqrt(2n) used in the reference approach.
    normaliser = np.sqrt(2.0 * n) / sum_seq
    f_norm = [x * normaliser for x in sequence]

    # Bounding value = max convolution of the normalised sequence.
    rhs = float(np.max(np.convolve(f_norm, f_norm)))

    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    # Rescale g back to the same normalisation.
    sum_g = np.sum(g)
    if sum_g == 0:
        return None
    g_norm = [x * normaliser for x in g]

    # Convex combination; a modest step size keeps the process stable.
    t = 0.04
    new_seq = [(1.0 - t) * a + t * b for a, b in zip(sequence, g_norm)]
    return new_seq

                cur = new
                print(f"[+] Hill‑climb improvement: {best_score:.6f}")
                import sys
                sys.stdout.flush()

    # ------------------------------------------------------------------
    # 3) LP‑based refinement (optional, runs while time permits)
    # ------------------------------------------------------------------
    # Start from the best sequence found so far, otherwise fall back to a
    # random initialisation.
    curr_seq = best_seq if best_seq is not None else [np.random.rand() for _ in range(500)]

    while time.time() < deadline:
        nxt = get_good_direction_to_move_into(curr_seq)
        if nxt is None:
            # very cheap random perturbation as a fallback
            idx = np.random.randint(len(curr_seq))
            curr_seq[idx] = max(0.0, curr_seq[idx] + np.random.randn() * 0.01)
            continue
        sc = evaluate_sequence(nxt)
        if sc < best_score:
            best_score = sc
            best_seq = nxt.copy()
            print(f"[+] LP step improvement: {best_score:.6f} (len={len(nxt)})")
            import sys
            sys.stdout.flush()
        curr_seq = nxt

    print(f"=== Final best score: {best_score:.6f} (length {len(best_seq)}) ===")
    return best_seq
