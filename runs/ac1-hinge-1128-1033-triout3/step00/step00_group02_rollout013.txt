"""
Stochastic hill‑climbing search for a step‑function with a small
2·n·max(conv(f,f))/sum(f)^2 value.

The algorithm works on a probability vector `p` (sum(p)=1).  The
objective depends only on `p` (scaling cancels), so we keep the sum
exactly 1 after every move.

Procedure
---------

1.  Initialise `p` uniformly (or from a previous construction if
    available).  Compute the convolution `c = p∗p` using a fast FFT
    based routine and the current score `score = 2·n·max(c)`.

2.  While time permits:
        * Locate the index `k` where `c` attains its maximum.
        * Choose an index `i` that participates in this maximal sum
          (`i ∈ [max(0,k‑(n‑1)), min(n‑1,k)]`).
          Its partner is `j = k‑i`.
        * Transfer a small amount `ε` of mass from the pair
          `(i,j)` to two random positions `r1, r2` (different from
          `i` and `j`).  This directly reduces the contribution
          `p[i]·p[j]` to the peak.
        * Build a candidate vector `p′` with the moved mass,
          recompute its convolution with the FFT routine and evaluate
          the new score.
        * Accept the move only if the score improves (simple greedy
          descent).  Keep the best sequence found so far.

   If a long streak without improvement occurs, a small random
   perturbation (Gaussian noise + renormalisation) is applied to
   escape possible plateaus.

3.  Return the best sequence observed.  All entries are non‑negative,
   the sum is 1 (well above the 0.01 safety limit) and the length
   is kept modest (≈ 5 000) so that each FFT is cheap.

The algorithm explores the search space without building large
constraint matrices (as required by the LP approach) and therefore
fits easily into the 1000 s time budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: fast convolution of a real vector with itself (full length)
def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Return np.convolve(a, a) using real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1                     # length of full convolution
    # Next power of two for efficient FFT
    N = 1 << ((m - 1).bit_length())
    # FFT of the vector (real → rfft)
    fa = np.fft.rfft(a, n=N)
    # Convolution in frequency domain (multiply, then inverse)
    conv_full = np.fft.irfft(fa * fa, n=N)
    return conv_full[:m]              # keep only the linear part


# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    the provided `evaluate_sequence` function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function stops a little before the
        limit to leave time for returning the result).

    Returns
    -------
    list[float]
        The best sequence found during the search.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # --------------------------------------------------------
    # Problem size – larger gives a lower theoretical bound.
    n = 5030                     # length used in the reference solution

    # --------------------------------------------------------
    # Initialise the probability vector.
    # If a previous construction `height_sequence_1` exists, use it
    # (it is already normalised by the scoring routine).  Otherwise
    # start from the uniform distribution.
    if "height_sequence_1" in globals():
        # make a copy and renormalise to sum 1
        init = np.array(height_sequence_1, dtype=float)
        if init.ndim != 1 or init.size == 0:
            p = np.full(n, 1.0 / n)
        else:
            p = init.copy()
            if p.sum() == 0:
                p = np.full(n, 1.0 / n)
            else:
                p = p / p.sum()
            if p.shape[0] != n:
                # resize (truncate / pad) to the required length
                if p.shape[0] > n:
                    p = p[:n]
                else:
                    p = np.pad(p, (0, n - p.shape[0]), constant_values=0.0)
    else:
        p = np.full(n, 1.0 / n)

    # Pre‑compute the first convolution and score.
    conv = _self_convolution(p)
    cur_score = 2.0 * n * conv.max()
    best_p = p.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # --------------------------------------------------------
    # Parameters governing the stochastic search
    max_no_improve = 5000          # after this many rejects do a perturbation
    no_improve_cnt = 0
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # 1) locate the current peak of the convolution
        k_max = int(np.argmax(conv))

        # 2) pick a contributing index i (and its partner j=k_max-i)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        i = np.random.randint(i_low, i_high + 1)
        j = k_max - i

        # ensure i <= j to treat the pair only once
        if i > j:
            i, j = j, i

        # 3) decide on the amount of mass to move
        if i == j:   # the central term (square)
            eps_limit = p[i]
        else:
            eps_limit = min(p[i], p[j])

        if eps_limit <= 0:
            # nothing to move – fall back to a tiny random perturbation
            eps = 0.0
        else:
            # random fraction between 0.01 and 0.5 of the available amount
            eps = eps_limit * (0.01 + 0.49 * np.random.rand())

        # 4) choose two destination indices different from i and j
        r1 = np.random.randint(n)
        while r1 == i or r1 == j:
            r1 = np.random.randint(n)
        r2 = np.random.randint(n)
        while r2 == i or r2 == j or r2 == r1:
            r2 = np.random.randint(n)

        # 5) build the candidate vector
        cand = p.copy()
        if eps > 0:
            cand[i] -= eps
            if i != j:
                cand[j] -= eps
            cand[r1] += eps
            cand[r2] += eps

        # Numerical safety – clamp tiny negatives caused by round‑off
        np.maximum(cand, 0.0, out=cand)
        # Re‑normalise to keep sum exactly 1 (scale does not affect the score)
        s = cand.sum()
        if s == 0:
            # extremely unlikely, restart from uniform
            cand = np.full(n, 1.0 / n)
        else:
            cand /= s

        # 6) evaluate the candidate
        conv_cand = _self_convolution(cand)
        cand_score = 2.0 * n * conv_cand.max()

        # 7) greedy acceptance
        if cand_score < cur_score:
            p = cand
            conv = conv_cand
            cur_score = cand_score
            no_improve_cnt = 0

            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                print(f"[{iteration:07d}] new best score = {best_score:.6f}")

        else:
            no_improve_cnt += 1

        # ----------------------------------------------------
        # Occasional random shake‑up if we are stuck for a while
        if no_improve_cnt >= max_no_improve:
            # Add a small Gaussian jitter and renormalise
            jitter = np.random.randn(n) * 0.01
            cand = p + jitter
            np.maximum(cand, 0.0, out=cand)
            total = cand.sum()
            if total > 0:
                cand /= total
                p = cand
                conv = _self_convolution(p)
                cur_score = 2.0 * n * conv.max()
                no_improve_cnt = 0
                print(f"[{iteration:07d}] shake‑up performed, score={cur_score:.6f}")

    # --------------------------------------------------------
    print(f"[finish] best score found = {best_score:.6f}")
    # The scorer expects a list of floats; convert now.
    return best_p.tolist()
