def compute_score(seq):
    arr = np.array(seq, dtype=np.float64)
    arr = np.clip(arr, 0, 1000)
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    n = arr.size
    # use FFT
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_val = conv.max()
    return 2.0 * n * max_val / (sum_a * sum_a)

def compute_score(seq):
    arr = np.clip(np.array(seq, dtype=np.float64),0,1000)
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    n = len(arr)
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(arr, m)
    conv = np.fft.irfft(fa*fa, m)[:2*n-1]
    max_c = conv.max()
    return 2.0 * n * max_c / (sum_a*sum_a)

    best_seq = None
    best_score = np.inf
    # Search over lengths

    for n in lengths:
        # Determine range for delta: from 0 to maybe n*2
        # Step size maybe 20% of n?
        max_delta = n  # can try up to n
        for delta in np.linspace(0, max_delta, num=10):

            for p in [0.45, 0.5, 0.55]:

                score = compute_score(seq_arr)
                if score < best_score:
                    best_score = score
                    best_seq = seq_arr.tolist()
                    print(f"New best score {best_score:.6f} n={n} delta={delta:.2f} p={p}")

    # Random local refinement: start from best_seq
    if best_seq is not None:
        seq = np.array(best_seq)
        n = len(seq)
        # Determine step size for perturbations based on sequence scale.
        while time.time() < deadline:
            # pick random index
            idx = np.random.randint(n)
            # propose perturbation: multiply by factor 1 + delta
            factor = 1 + np.random.randn()*0.05  # small relative change
            new_val = max(0, seq[idx] * factor)

                best_seq = seq.tolist()

                # revert
                seq[idx] = old_val
        # if time left, try random shuffling of a few values

"""
Propose a candidate step‑function that minimises
    2·n·max(a*a)/(sum(a)^2)

The search is divided in two phases:

1. **Parametric scan** –  for a family of monotone sequences
       a_i = (i + offset + 1)^(-p)
   we sweep the length `n`, the exponent `p` and the offset.
   The score is evaluated with an FFT based convolution (O(n log n)).

2. **Local refinement** – starting from the best sequence found,
   random element‑wise perturbations are tried.  A change is kept
   only if it improves the score.

All candidate sequences are clipped to the limits enforced by
`evaluate_sequence` (0 ≤ a_i ≤ 1000) and the final sequence with the
lowest score before the time budget expires is returned.
"""

import time
import math
import numpy as np
from scipy import optimize  # only needed for the optional LP refinement


def _score_fft(seq):
    """
    Fast version of evaluate_sequence (same formula, no security checks).
    Returns the benchmark score; np.inf for illegal input.
    """
    a = np.array(seq, dtype=float)
    # apply the same caps as evaluate_sequence
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = a.size
    # convolution via FFT (real‑to‑real transform)
    m = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    maxc = conv.max()
    return float(2.0 * n * maxc / (s * s))


def _generate_powerlaw(n, p, offset):
    """
    a_i = (i + offset + 1)^(-p)   for i = 0 .. n-1
    """
    i = np.arange(n, dtype=float)
    return np.power(i + offset + 1.0, -p)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point used by the grader.
    Returns a list of non‑negative floats (the step heights).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # keep a few seconds margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – systematic grid search over a simple parametric family
    # ------------------------------------------------------------------
    # length range – a few hundred up to several thousand (still cheap for FFT)
    lengths = list(range(400, 4801, 400))          # 400, 800, …, 4400
    # exponents – we know 0.5 is near‑optimal, explore a neighbourhood
    exponents = [0.40, 0.45, 0.475, 0.50, 0.525, 0.55, 0.60]
    # offsets – integer shift inside the power law; 0 means the classic 1/√i
    offsets = [0, 1, 2, 5, 10, 20]

    for n in lengths:
        if time.time() > deadline:
            break
        for p in exponents:
            for off in offsets:
                seq_arr = _generate_powerlaw(n, p, off)
                sc = _score_fft(seq_arr)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq_arr.tolist()
                    print(
                        f"[grid] new best: score={best_score:.6f}, n={n}, p={p}, offset={off}"
                    )
        # also try a pure exponential family (geometric) – useful for very
        # low‑sum, high‑n regimes
        for r in [0.98, 0.99, 0.995, 0.997]:
            i = np.arange(n, dtype=float)
            seq_arr = np.power(r, i)
            sc = _score_fft(seq_arr)
            if sc < best_score:
                best_score = sc
                best_seq = seq_arr.tolist()
                print(
                    f"[geom] new best: score={best_score:.6f}, n={n}, r={r}"
                )

    # ------------------------------------------------------------------
    # Phase 2 – simple stochastic hill‑climbing starting from the best
    # ------------------------------------------------------------------
    if best_seq is None:          # fallback (should never happen)
        best_seq = [1.0] * 1000
        best_score = _score_fft(best_seq)

    cur_seq = np.array(best_seq, dtype=float)
    cur_score = best_score
    n = cur_seq.size

    # pre‑compute a few useful numbers for the perturbation scale
    sigma = max(cur_seq) * 0.05 + 1e-3

    while time.time() < deadline:
        # pick a random index and propose a multiplicative perturbation
        idx = np.random.randint(n)
        factor = 1.0 + np.random.randn() * 0.03   # ~3 % change
        new_val = max(0.0, cur_seq[idx] * factor)

        old_val = cur_seq[idx]
        cur_seq[idx] = new_val
        new_score = _score_fft(cur_seq)

        if new_score < cur_score:
            cur_score = new_score
            # keep the change; update global best if needed
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur_seq.tolist()
                print(
                    f"[mhc] improved: score={best_score:.6f} (n={n})"
                )
        else:
            # revert – the move was not beneficial
            cur_seq[idx] = old_val

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq
