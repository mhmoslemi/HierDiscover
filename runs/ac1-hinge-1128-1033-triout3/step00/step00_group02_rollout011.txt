def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    # Determine n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 5000
    # Initialize sequence: u[i] = 1.0/np.sqrt(i+1)
    a = np.array([1.0/np.sqrt(i+1) for i in range(n)], dtype=np.float64)
    # Normalize to target sum = sqrt(2*n)
    target_sum = np.sqrt(2.0 * n)
    a *= target_sum / a.sum()
    best_sequence = a.copy()
    best_score = evaluate_sequence(best_sequence.tolist())
    print(f'Initial score {best_score:.6f}')
    
    # Hyperparameters
    lr = 0.5   # initial step size
    min_lr = 1e-5
    patience = 0
    # loop
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute convolution
        conv = np.convolve(a, a)
        max_c = conv.max()
        idx = int(conv.argmax())
        S = a.sum()
        # compute gradient
        term = max_c / S
        # counterpart values
        idx_arr = np.arange(n)
        counterpart = idx - idx_arr
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = a[counterpart[mask]]
        coeff = 4.0 * n / (S * S)
        grad = coeff * (g - term)
        # line search
        improved = False
        step_lr = lr
        for _ in range(10):
            a_candidate = a - step_lr * grad
            # enforce nonnegative
            a_candidate = np.clip(a_candidate, 0.0, None)
            # re-normalize sum to target_sum
            sum_cand = a_candidate.sum()
            if sum_cand == 0:
                # degenerate, break
                break
            a_candidate *= target_sum / sum_cand
            # compute score
            score = evaluate_sequence(a_candidate.tolist())
            if score < best_score:
                best_score = score
                best_sequence = a_candidate.copy()
                a = a_candidate
                improved = True
                # increase lr slightly
                lr = step_lr * 1.2
                break
            else:
                # try smaller step
                step_lr *= 0.5
        if not improved:
            # decrease lr
            lr = max(lr * 0.7, min_lr)
            # Occasionally random perturb to escape
            if iteration % 200 == 0:
                # random small perturbation
                perturb = np.random.randn(n) * 0.01 * a
                a = np.clip(a + perturb, 0.0, None)
                a *= target_sum / a.sum()
        # occasionally log
        if iteration % 100 == 0:
            print(f'Iter {iteration}: best_score={best_score:.6f}, lr={lr:.6f}')
    print('Finished, best score', best_score)
    return best_sequence.tolist()

    """
    n = arr.shape[0]
    conv = np.convolve(arr, arr)
    max_c = conv.max()
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Run a projected gradient descent to minimise the bound.

    Returns
    -------
    list[float]
        The best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0       # a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Determine length and initialise a “inverse‑sqrt’’ sequence.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)               # use the given construction if present
    except NameError:
        n = 5030                                   # fallback length
    # a_i ≈ 1/√(i+1)  (the classical almost‑flat convolution shape)
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    # Normalise the ℓ¹‑norm to √(2n) – this does not change the score
    target_sum = math.sqrt(2.0 * n)
    a *= target_sum / a.sum()

    best_seq = a.copy()
    best_val = _score(best_seq)
    print(f"[init] n={n}, initial score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Gradient‑descent loop.
    # ------------------------------------------------------------------
    lr = 0.5                     # initial learning rate
    min_lr = 1e-6
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # current convolution, max value and its index
        conv = np.convolve(a, a)
        max_c = conv.max()
        k_max = int(conv.argmax())
        S = a.sum()

        # ----- sub‑gradient of the objective ---------------------------------
        term = max_c / S                               # M / S
        idx = np.arange(n, dtype=np.int64)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        # a_{k−i} is zero where the partner index is out of range
        a_partner = np.zeros_like(a)
        a_partner[mask] = a[partner[mask]]
        coeff = 4.0 * n / (S * S)
        grad = coeff * (a_partner - term)

        # ----- simple back‑tracking line search -------------------------------
        step = lr
        improved = False
        for _ in range(10):
            a_candidate = a - step * grad
            # projection onto the feasible set (non‑negative)
            np.maximum(a_candidate, 0.0, out=a_candidate)

            # re‑normalise the ℓ¹‑norm (scale‑invariant objective)
            s_cand = a_candidate.sum()
            if s_cand == 0.0:
                break
            a_candidate *= target_sum / s_cand

            cand_val = _score(a_candidate)

            if cand_val < best_val:               # accept only improving moves
                best_val = cand_val
                best_seq = a_candidate.copy()
                a = a_candidate
                improved = True
                lr = step * 1.1                    # slightly increase step size
                break
            else:
                step *= 0.5                        # try a smaller step

        if not improved:
            # no improvement – reduce learning rate and (occasionally) jitter
            lr = max(lr * 0.7, min_lr)
            if iteration % 200 == 0:
                jitter = np.random.randn(n) * 0.001 * a
                a += jitter
                np.maximum(a, 0.0, out=a)
                a *= target_sum / a.sum()

        # ------------------------------------------------------------------
        # 3. Periodic progress report
        # ------------------------------------------------------------------
        if iteration % 100 == 0:
            elapsed = time.time() - start_time
            print(
                f"[{iteration:5d}] best={best_val:.6f} lr={lr:.6f} "
                f"time={elapsed:.1f}s"
            )

    # ----------------------------------------------------------------------
    # 4. Return the best found sequence (as a plain Python list)
    # ----------------------------------------------------------------------
    print(f"[done] iterations={iteration}, best score={best_val:.6f}")
    return best_seq.tolist()
