def project_to_simplex(v, s):
    # v is 1-d numpy array
    # returns w
    n = len(v)
    if s < 0: raise
    # standard algorithm:
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative?
        rho = 0
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def local_adjust(seq, S, deadline):
    n = len(seq)
    cur = seq.copy()
    cur_score = compute_score(cur)
    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # propose moving epsilon from i to j
        epsilon = cur[i] * 0.1 * np.random.rand()  # random fraction up to 10%
        if cur[i] < epsilon:
            continue
        new = cur.copy()
        new[i] -= epsilon
        new[j] += epsilon
        # no negative
        new_score = compute_score(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
        # early exit if near target
        if cur_score < 1.51:
            break
    return cur, cur_score

"""
Propose a step‑function (a list of non‑negative heights) that minimises
the objective used in ``evaluate_sequence``.

The algorithm is a hybrid search:

1.  Initialise several candidate families (uniform, triangular,
    exponential, power‑law, Gaussian).  All candidates are normalised so
    that   sum(a) = sqrt(2·n).

2.  For each candidate run a *smooth‑max* gradient descent.
    The objective  max_k (a∗a)[k]  is smoothed by a log‑sum‑exp
    (parameter α) which yields a differentiable surrogate.
    The gradient of the surrogate can be computed by one correlation.
    After each gradient step the vector is projected onto the probability
    simplex of total mass ``S = sqrt(2·n)`` (the scaling does not change the
    final score).

3.  Random sparse patterns (a small set of equal‑mass positions) are
    generated and refined by a simple hill‑climbing swap procedure which
    keeps the total mass fixed.  The best sparse pattern is again polished
    by the smooth‑max descent.

4.  Finally a cheap coordinate‑wise random perturbation is applied while
    time permits.

All intermediate evaluations use a fast FFT based convolution, which is
identical to the definition used in ``evaluate_sequence`` but avoids the
sec‑level overhead of the reference implementation.  The best sequence
found before the time budget expires is returned as a plain Python list.
"""

import time
import numpy as np

# Helper utilities

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """ Euclidean projection of v onto {x >= 0, sum(x) = s}. """
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """ Linear convolution a*a using FFT (real valued). """
    n = a.shape[0]
    N = 2 * n                      # zero‑padding length (power of two not needed)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[: 2 * n - 1]


def _score(seq: np.ndarray, target_sum: float) -> float:
    """
    Compute the same metric as ``evaluate_sequence``.
    The sequence is assumed to be non‑negative; if its sum differs from
    ``target_sum`` it is renormalised first.
    """
    sum_a = np.sum(seq)
    if sum_a <= 1e-12:
        return np.inf
    if abs(sum_a - target_sum) > 1e-9:
        seq = seq * (target_sum / sum_a)
    conv = _conv_fft(seq)
    max_b = np.max(conv)
    # formula: 2·n·max_b / (sum_a**2); when sum_a = sqrt(2·n) this equals max_b
    n = seq.shape[0]
    return float(2.0 * n * max_b / (np.sum(seq) ** 2))

def _uniform(n: int, S: float) -> np.ndarray:
    """Uniform distribution with total mass S."""
    return np.full(n, S / n, dtype=float)


def _triangular(n: int, S: float) -> np.ndarray:
    """Symmetric triangular shape (peak in the centre)."""
    i = np.arange(n)
    peak = (n - 1) / 2.0
    a = np.minimum(i, n - 1 - i) + 1.0
    a = a.astype(float)
    a = a * (S / np.sum(a))
    return a


def _exponential(n: int, S: float, tau_factor: float = 0.3) -> np.ndarray:
    """a_i = exp(-i / tau)  (tau scales with the length)."""
    tau = tau_factor * n
    i = np.arange(n, dtype=float)
    a = np.exp(-i / tau)
    a = a * (S / np.sum(a))
    return a


def _power_law(n: int, S: float, p: float = 1.0) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p)
    a = a * (S / np.sum(a))
    return a


def _gaussian(n: int, S: float, sigma_factor: float = 0.4) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    sigma = sigma_factor * n
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-((i - centre) ** 2) / (2.0 * sigma ** 2))
    a = a * (S / np.sum(a))
    return a


def _sparse_random(n: int, m: int, S: float) -> np.ndarray:
    """Uniform mass on a random subset of size m (m >= 2)."""
    idx = np.random.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = S / m
    return a

def _smooth_max_descent(
    init_seq: np.ndarray,
    S: float,
    deadline: float,
    max_iters: int = 500,
    alpha: float = 12.0,
) -> (np.ndarray, float):
    """
    Gradient descent on a smooth approximation of max_k (a*a)[k].
    The approximation is  (1/α)·log Σ exp(α·c_k) where c_k are the
    convolution entries.  After each step the vector is projected onto the
    simplex of total mass S.
    """
    n = init_seq.shape[0]
    a = init_seq.copy()
    best_a = a.copy()
    best_score = _score(a, S)
    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution
        conv = _conv_fft(a)
        # smooth max: weights = softmax(α·conv)
        z = alpha * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)               # stabilise
        w = exp_z / np.sum(exp_z)               # size 2n‑1, sums to 1

        # gradient = 2 * correlate(w, a)[:n]
        grad = 2.0 * np.correlate(w, a, mode="full")[:n]

        # step size – simple diminishing schedule
        eta = 0.25 / (1.0 + 0.005 * it)

        a_new = a - eta * grad
        a = _project_to_simplex(a_new, S)

        cur_score = _score(a, S)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # early exit if we are already below the target bound
            if best_score < 1.51:
                break
        it += 1
    return best_a, best_score


def _hill_climb_sparse(
    start_seq: np.ndarray,
    S: float,
    deadline: float,
    max_swaps: int = 2000,
) -> (np.ndarray, float):
    """
    Simple hill‑climbing on a sparse uniform‑mass pattern.
    Swaps a randomly chosen non‑zero index with a zero index;
    the total number of non‑zeros stays constant, hence the mass per
    non‑zero entry (v = S / m) also stays constant.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    # identify the support
    nonzero = set(np.flatnonzero(a > 1e-12))
    zero = set(range(n)) - nonzero
    m = len(nonzero)
    if m == 0:
        return a, _score(a, S)
    v = S / m
    # enforce exact equality of all non‑zero entries
    a[:] = 0.0
    for idx in nonzero:
        a[idx] = v
    cur_score = _score(a, S)

    swaps = 0
    while time.time() < deadline and swaps < max_swaps:
        i_rem = np.random.choice(tuple(nonzero))
        i_add = np.random.choice(tuple(zero))

        # perform swap
        a[i_rem] = 0.0
        a[i_add] = v

        new_score = _score(a, S)
        if new_score < cur_score:
            # keep swap
            cur_score = new_score
            nonzero.remove(i_rem)
            nonzero.add(i_add)
            zero.remove(i_add)
            zero.add(i_rem)
            if cur_score < 1.51:
                break
        else:
            # revert
            a[i_rem] = v
            a[i_add] = 0.0
        swaps += 1
    return a, cur_score


def _coordinate_adjust(
    seq: np.ndarray,
    S: float,
    deadline: float,
    max_steps: int = 1200,
) -> (np.ndarray, float):
    """
    Very cheap random coordinate moves: transfer a random fraction of mass
    from one entry to another and keep the improvement if any.
    """
    a = seq.copy()
    best_score = _score(a, S)
    n = a.shape[0]
    steps = 0
    while time.time() < deadline and steps < max_steps:
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] < 1e-12:
            steps += 1
            continue
        # transfer a small amount (up to 10 % of a[i])
        delta = a[i] * 0.1 * np.random.rand()
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # projection is unnecessary – the move preserves the total sum
        cur = _score(a_new, S)
        if cur < best_score:
            a = a_new
            best_score = cur
            if best_score < 1.51:
                break
        steps += 1
    return a, best_score

    # ------------------------------------------------------------------
    # 1) start from a few deterministic families
    # ------------------------------------------------------------------
    candidates = [
        _uniform(n, S),
        _triangular(n, S),
        _exponential(n, S, tau_factor=0.25),
        _exponential(n, S, tau_factor=0.55),
        _power_law(n, S, p=0.6),
        _power_law(n, S, p=1.2),
        _gaussian(n, S, sigma_factor=0.45),
        _gaussian(n, S, sigma_factor=0.65),
    ]

    best_seq = None
    best_score = np.inf

    for seq in candidates:
        sc = _score(seq, S)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[init] Best score from deterministic families: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) smooth‑max gradient descent from each deterministic start
    # ------------------------------------------------------------------
    for base in candidates:
        if time.time() >= deadline:
            break
        seq_opt, sc_opt = _smooth_max_descent(base, S, deadline, max_iters=400)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = seq_opt.copy()
            print(f"[gd] Improved to {best_score:.6f} from a deterministic start")
        # also try a short descent from the current best (helps escape)
        seq_opt2, sc_opt2 = _smooth_max_descent(best_seq, S, deadline, max_iters=200)
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = seq_opt2.copy()
            print(f"[gd] Further improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) random sparse patterns + hill‑climbing + polishing
    # ------------------------------------------------------------------
    sparse_factors = [0.05, 0.07, 0.10, 0.13, 0.18]  # proportion of non‑zeros
    for fac in sparse_factors:
        if time.time() >= deadline:
            break
        m = max(2, int(fac * n))
        for trial in range(4):
            if time.time() >= deadline:
                break
            seq0 = _sparse_random(n, m, S)
            # a few hill‑climbing swaps
            seq_hc, sc_hc = _hill_climb_sparse(seq0, S, deadline, max_swaps=1500)
            # polish with smooth‑max descent
            seq_gd, sc_gd = _smooth_max_descent(seq_hc, S, deadline, max_iters=300)
            if sc_gd < best_score:
                best_score = sc_gd
                best_seq = seq_gd.copy()
                print(
                    f"[sparse] m={m} trial={trial} → score {best_score:.6f}"
                )
            # occasionally try a coordinate‑wise tweak from the current best
            seq_co, sc_co = _coordinate_adjust(best_seq, S, deadline, max_steps=800)
            if sc_co < best_score:
                best_score = sc_co
                best_seq = seq_co.copy()
                print(f"[coord] Small tweak → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final cheap random local search (if time remains)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        seq_tmp, sc_tmp = _coordinate_adjust(best_seq, S, deadline, max_steps=600)
        if sc_tmp < best_score:
            best_score = sc_tmp
            best_seq = seq_tmp.copy()
            print(f"[final] Random tweak gave {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[result] Best score after search: {best_score:.6f}")
    # Convert to plain Python list for the external evaluator
    return best_seq.tolist()
