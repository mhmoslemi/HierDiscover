    fft_len = 1 << (2 * n - 1).bit_length()      # constant FFT length for the run

    # -----------------------------------------------------------------
    # main optimisation loop
    # -----------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- compute convolution and locate the current peak -----
        fa = np.fft.rfft(cur_seq, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2 * n - 1]
        max_idx = int(conv.argmax())

        # ----- gradient of the active peak -----
        grad = compute_gradient(cur_seq, max_idx)

        # ----- choose source (largest gradient) and sink (smallest) -----
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # ----- transfer a fraction of the source mass toward the sink -----
        delta = 0.0
        if cur_seq[src] > 0:
            fraction = rng.random() * 0.3               # up to 30 % of the source weight
            delta = cur_seq[src] * fraction

        new_seq = cur_seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta
        np.clip(new_seq, 0.0, 1000.0, out=new_seq)

        new_score = fast_score(new_seq)

        # ----- simulated‑annealing acceptance -----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = new_seq
            cur_score = new_score
            temperature *= 0.9997                     # gentle cooling
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                it_without_improve = 0
                if iteration % 200 == 0:
                    print(f"[{iteration}] New best score: {best_score:.6f}")
            else:
                it_without_improve += 1
        else:
            it_without_improve += 1

        # ----- random jitter to break stagnation -----
        if it_without_improve > 5000:
            cur_seq = best_seq.copy()
            jitter = rng.normal(scale=0.01, size=n)
            cur_seq += jitter
            np.clip(cur_seq, 0.0, 1000.0, out=cur_seq)
            cur_score = fast_score(cur_seq)
            temperature = 0.2
            it_without_improve = 0
            print(f"[{iteration}] Restart after stagnation – best score {best_score:.6f}")

    print(f"Search finished after {iteration} iterations – best score {best_score:.6f}")
    return best_seq.tolist()
