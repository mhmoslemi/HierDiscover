import numpy as np
import time

def linear_convolve(a):
    """Return the linear convolution a*a using FFT (real sequences)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    # rfft works on real input; result length size//2+1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def max_subgradient(a, conv, M):
    """Subgradient of the maximum convolution entry.
    Returns vector g where g_i = sum_{k : conv[k]==M} 2*a_{k-i} (if index exists)."""
    n = a.shape[0]
    # indices where conv is (almost) the maximum
    active = np.where(np.isclose(conv, M, atol=1e-12))[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # segment of a that participates
        seg = a[lo : hi + 1]
        # counterpart indices are reversed
        g[lo : hi + 1] += 2.0 * seg[::-1]
    return g

import numpy as np
import time

# --------------------------------------------------------------
# Helper: fast linear convolution using FFT (real sequences)
# --------------------------------------------------------------
def _linear_convolve(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)                     # rfft works on real data
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------
# Helper: sub‑gradient of the max‑convolution entry
# --------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray, M: float) -> np.ndarray:
    """
    Compute a sub‑gradient of the function  max_k  (a*a)[k]
    with respect to the vector a (non‑negative, sum fixed).
    For every index k where conv[k] equals the maximum M we add
    the contribution 2 * a_{k-i} to g_i.
    """
    n = a.shape[0]
    # indices that realise the maximum (allow a tiny tolerance)
    active = np.where(np.isclose(conv, M, atol=1e-12))[0]
    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        seg = a[lo: hi + 1]                 # a[i] for i in [lo,hi]
        # the counterpart a_{k-i} runs in the reverse order
        g[lo: hi + 1] += 2.0 * seg[::-1]
    return g


# --------------------------------------------------------------
# Main entry point required by the competition framework
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a step‑function (non‑negative vector) that
    minimises the supplied `evaluate_sequence` function.

    The method:
      * works on a fixed length (≈ 5000) – large enough to give a
        substantial improvement over the naïve uniform sequence.
      * starts from a random or simple “U‑shaped’’ distribution,
        every vector is kept symmetric (a[i] == a[-i-1]) and normalised.
      * repeatedly:
          – computes the self‑convolution,
          – builds a sub‑gradient of the maximal convolution entry,
          – projects this direction onto the simplex (preserves the sum,
            stays in the non‑negative orthant),
          – takes a decreasing step, symmetrises and renormalises.
      * keeps the best vector seen and performs occasional random restarts
        if no improvement occurs for a long while.

    The whole routine stops when the time budget expires and returns the
    best sequence found.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a nice trade‑off.
    # ------------------------------------------------------------------
    n = 5000                         # length of the candidate vector

    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Function to produce a fresh feasible starting point.
    # ------------------------------------------------------------------
    def _initial_vector():
        if np.random.rand() < 0.5:
            a0 = np.random.rand(n)               # completely random
        else:
            # a simple “U’’ shape: larger values at the ends, smaller in the centre
            x = np.arange(n, dtype=float)
            mid = (n - 1) / 2.0
            coeff = np.random.rand() * 5.0        # how pronounced the U is
            a0 = 1.0 + coeff * ((x - mid) / mid) ** 2
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        # enforce symmetry – the optimum is symmetric, this halves the search space
        a0 = (a0 + a0[::-1]) * 0.5
        return a0

    # ------------------------------------------------------------------
    # Initialise the first candidate and record its score.
    # ------------------------------------------------------------------
    a = _initial_vector()
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] evaluation = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the sub‑gradient descent
    # ------------------------------------------------------------------
    step0 = 0.5              # initial step magnitude (will be decayed)
    min_step = 1e-5
    max_no_improve = 2000   # after this many iterations without progress we restart

    it_without_improve = 0
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 1. convolution and maximal entry
        conv = _linear_convolve(a)
        M = conv.max()

        # 2. sub‑gradient of the max‑entry
        g = _max_subgradient(a, conv, M)

        # 3. project onto the affine space {sum x = 0}
        avg = g.mean()
        direction = -g + avg                     # sum(direction) == 0

        # 4. step length – linearly decreasing with the remaining budget
        time_left = deadline - time.time()
        frac = max(time_left / (budget_s - 0.0), 0.0)   # fraction of total budget left
        step = max(step0 * frac, min_step)

        # 5. take a step, enforce non‑negativity and normalise
        a_new = a + step * direction
        a_new = np.maximum(a_new, 0.0)
        tot = a_new.sum()
        if tot == 0.0:
            a_new = _initial_vector()
        else:
            a_new /= tot

        # 6. keep the vector symmetric
        a_new = (a_new + a_new[::-1]) * 0.5

        # 7. evaluate the new candidate
        cur_score = float(evaluate_sequence(a_new.tolist()))

        # 8. bookkeeping
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            it_without_improve = 0
            print(f"[{iteration:6d}] new best = {best_score:.6f}")
        else:
            it_without_improve += 1

        # 9. simple acceptance rule (mostly keep the improvement)
        if cur_score <= best_score or np.random.rand() < 0.005:
            a = a_new

        # 10. restart if stuck for a while
        if it_without_improve > max_no_improve:
            a = _initial_vector()
            it_without_improve = 0

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    return best_seq.tolist()
