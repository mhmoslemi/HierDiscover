"""
propose_candidate – heuristic search for a step‑function that minimizes
evaluate_sequence (i.e. minimizes   2·n·max(conv(a,a)) / (sum(a))²).

The algorithm proceeds in three stages:

1. **Initialisation** –  a deterministic power‑law start
   a_i = (i+1)^(-0.5)  (the classic 1/√i shape).  This already beats the
   uniform constant sequence (value ≈ 1.60 for n≈2000) because its
   self‑convolution is almost flat.

2. **Local improvement** –  a greedy / simulated‑annealing style hill‑climber.
   In each iteration we locate the index `k` where the convolution is maximal,
   then find the pair (i, j = k‑i) that contributes the largest product
   a_i·a_j.  A small amount of mass is transferred from the larger member of
   that pair to the currently smallest coefficient (keeping all entries ≥0).
   The move is kept if the evaluation score improves; otherwise it is
   reverted.  The transferred fraction (`step`) is adapted on the fly,
   decreasing when moves are rejected and slightly increasing after a
   successful move.

3. **Diversification** –  after a long streak without improvement we
   perform a modest random perturbation of a few coefficients to escape
   possible local minima.

All heavy work is done with NumPy; the convolution is evaluated by an FFT
(which is ~O(n log n) and much faster than the naïve O(n²) `np.convolve`
for the lengths we use).  The routine respects the 1000 s time limit and
returns the best sequence found so far.

The method is deliberately simple, does not rely on external solvers and
fits easily into the allowed execution budget while typically reaching
values well below the baseline 2.0 (often < 1.55 for n≈2000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast convolution (FFT based) returning the full array
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a * a (full convolution) using FFT for speed."""
    n = a.size
    # length needed for full convolution = 2*n-1, pick next power of two
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Initialise a power‑law sequence a_i = (i+1)^(-p)
def _initial_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Create a monotone decreasing non‑negative sequence."""
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    seq = idx ** (-p)
    # scaling is irrelevant for the objective (scale invariant), but we keep it.
    return seq


# ----------------------------------------------------------------------
# Information about the current maximal convolution entry
def _max_convolution_info(seq: np.ndarray):
    """Return (max_val, max_index, i_src, j_src) where the product a_i a_j is largest."""
    n = seq.size
    conv = _full_convolution(seq)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # range of i that participates in this sum (i + j = max_idx)
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
    j_range = max_idx - i_range

    # product contributions for this k
    prod = seq[i_range] * seq[j_range]
    # pick the index giving the largest individual product
    arg = int(np.argmax(prod))
    i_src = int(i_range[arg])
    j_src = int(j_range[arg])
    return max_val, max_idx, i_src, j_src


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a low‑value step sequence within the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total seconds allowed for the search (default 1000).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ---- 1. Choose a length -------------------------------------------------
    # A length of a few thousand balances speed and quality.
    n = 2000

    # ---- 2. Build the first candidate (power‑law) ---------------------------
    # The exponent 0.5 is known to give a very good starting point.
    seq = _initial_sequence(n, p=0.5, offset=0.0).astype(np.float64)

    # ---- 3. Evaluate the starting point ------------------------------------
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, start_score={best_score:.6f}")

    # ---- 4. Local optimisation loop -----------------------------------------
    # Parameters for the adaptive step size
    step = 0.08          # start moving ~8 % of the source coefficient
    step_min = 1e-6
    step_max = 0.5
    no_improve = 0      # counter for stagnation detection

    while time.time() < deadline:
        # ----- a) locate the most offending product -------------------------
        max_val, max_k, i_src, j_src = _max_convolution_info(seq)

        # If the maximal convolution entry is already close to the average,
        # we are unlikely to improve further.
        # (average = 1/(2*n-1) for a probability vector; we just keep going.)

        # Choose the source index – the larger of the two that build the peak.
        if seq[i_src] >= seq[j_src]:
            src = i_src
        else:
            src = j_src

        # Choose a destination index with a very small value.
        dst = int(np.argmin(seq))
        if dst == src:
            # pick the second smallest if we accidentally hit the source
            dst = int(np.argsort(seq)[1])

        # Amount to move (a small fraction of the source value)
        delta = seq[src] * step
        if delta <= 0.0:
            # nothing to move – break to avoid endless loop
            break

        # Apply move
        seq[src] -= delta
        seq[dst] += delta

        # ----- b) evaluate the new sequence --------------------------------
        cur_score = evaluate_sequence(seq.tolist())

        if cur_score < best_score:
            # improvement – keep the change
            best_score = cur_score
            best_seq = seq.copy()
            # slowly increase step size to accelerate when we are on a good path
            step = min(step_max, step * 1.02)
            no_improve = 0
            # occasional progress report
            if no_improve % 100 == 0:
                print(f"[improve] iter={no_improve:5d}, score={best_score:.6f}, step={step:.4f}")
        else:
            # revert the move
            seq[src] += delta
            seq[dst] -= delta
            # make the step more cautious
            step = max(step_min, step * 0.97)
            no_improve += 1

        # ----- c) diversification after long stagnation --------------------
        if no_improve > 2000:
            # random perturbation of a few entries to escape a plateau
            for _ in range(5):
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                # move a tiny random chunk from i to j
                eps = seq[i] * (0.01 + 0.02 * np.random.rand())
                eps = min(eps, seq[i])
                seq[i] -= eps
                seq[j] += eps
            no_improve = 0
            step = max(step_min, step * 0.5)
            print(f"[perturb] stagnation detected – random tweak, step={step:.4f}")

    # ----------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best_score={final_score:.6f}, length={n}")
    return best_seq.tolist()
