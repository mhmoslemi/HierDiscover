"""
propose_candidate
=================

This implementation builds a two‑stage step function:

* a *core* that follows the optimal 1/√i decay (the shape whose
  self‑convolution is almost constant and gives the theoretical bound
  ≈π/2 for the evaluation function),

* a long *tail* of a constant small height.  The tail increases the total
  ℓ¹‐norm while keeping the maximum of the convolution essentially
  unchanged, thereby lowering the ratio
      L = 2·n·max(conv) / (∑a)² .

The algorithm searches over the core length and the tail height on a
log‑grid, evaluates the exact score using an FFT‑based convolution (fast
enough for the given sizes), and finally performs a short simulated‑annealing
refinement.  The returned list of non‑negative numbers respects all
constraints of ``evaluate_sequence``.
"""

import time
import numpy as np

def _conv_max(seq: np.ndarray) -> float:
    """
    Return the maximum value of the linear convolution of *seq* with itself.
    Uses an FFT of size power‑of‑two for speed.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """Exact evaluation of the objective for a given sequence."""
    n = seq.shape[0]
    total = float(seq.sum())
    if total < 0.01:
        return np.inf
    max_conv = _conv_max(seq)
    return 2.0 * n * max_conv / (total * total)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Build a candidate sequence by combining an inverse‑square‑root core
    with a constant tail.  Perform a small grid search over core length
    and tail height, then improve the best candidate with a few
    simulated‑annealing steps.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Length of the construction: use the known length if available,
    # otherwise fall back to a reasonable default.
    n = len(globals().get("height_sequence_1", [])) or 8000

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1) Grid search over core size and tail height.
    # ------------------------------------------------------------------
    # Core fractions to try (5 % … 50 % of the total length).
    core_fracs = np.linspace(0.05, 0.5, 10)

    # Tail‑height exponents (10⁻⁵ … 10⁻⁰·⁵).  The tail height is the
    # dominant knob that controls how much ℓ¹‑mass we add.
    log_heights = np.linspace(-5, -0.5, 30)

    for frac in core_fracs:
        if time.time() > deadline:
            break
        core_len = max(2, int(frac * n))          # at least two elements
        # inverse‑sqrt core: a_i = 1 / sqrt(i+1)
        core = 1.0 / np.sqrt(np.arange(1, core_len + 1, dtype=np.float64))

        tail_len = n - core_len
        if tail_len <= 0:
            continue

        for logh in log_heights:
            delta = 10 ** logh                       # constant tail height
            tail = np.full(tail_len, delta, dtype=np.float64)
            seq = np.concatenate([core, tail])
            val = _score(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(
                    f"[grid] new best: {best_val:.6f} "
                    f"(core={core_len}, δ={delta:.2e})"
                )
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # 2) Simple simulated‑annealing refinement around the best point.
    # ------------------------------------------------------------------
    if best_seq is None:                 # fallback – uniform sequence
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _score(best_seq)

    cur_seq = best_seq.copy()
    cur_val = best_val
    start = time.time()
    # initial temperature proportional to the current score
    T0 = max(0.01, 0.1 * best_val)

    while time.time() < deadline:
        # pick two distinct indices and move a small amount from i → j
        i, j = np.random.randint(n, size=2)
        if i == j or cur_seq[i] <= 0:
            continue
        # move at most 1 % of the donor entry
        amount = cur_seq[i] * np.random.rand() * 0.01
        new_seq = cur_seq.copy()
        new_seq[i] -= amount
        new_seq[j] += amount

        # quick reject if we produced a negative (should not happen)
        if new_seq[i] < 0:
            continue

        new_val = _score(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T0):
            cur_seq, cur_val = new_seq, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, new_seq.copy()
                print(f"[SA] improved to {best_val:.6f}")

        # linear cooling
        elapsed = time.time() - start
        T0 = max(1e-6, T0 * (1.0 - elapsed / budget_s))

    # Return a plain Python list as required by the evaluation harness.
    print(f"final best value: {best_val:.6f}")
    return best_seq.tolist()
