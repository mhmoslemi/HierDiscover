"""
Simulated‑annealing / hill‑climbing search for a non‑negative step‑function
sequence that minimises the provided `evaluate_sequence` function.

The algorithm works with a fixed length (taken from the global
`height_sequence_1` if it exists).  It keeps the ℓ¹‑norm of the vector
constant (the objective is scale‑invariant) and repeatedly moves a small
amount of mass between two positions.  Moves that improve the objective
are always accepted; worsening moves are accepted with a temperature‑
controlled probability (Metropolis rule).  A cheap FFT‑based routine is
used for fast evaluation of the objective inside the annealing loop.
Occasional Gaussian smoothing steps help escape plateaus.

The best sequence found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation of a sequence (identical to the reference evaluate_sequence)
# ----------------------------------------------------------------------
def _fast_evaluation(arr: np.ndarray) -> float:
    """Return the same value as evaluate_sequence for a numpy array."""
    n = arr.shape[0]
    # Full linear convolution via FFT (real → real)
    fft = np.fft.rfft(arr, n * 2)
    conv = np.fft.irfft(fft * fft, n * 2 - 1)
    max_b = float(conv.max())
    sum_a = float(arr.sum())
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Small Gaussian blur helper (no closure, top‑level as required)
# ----------------------------------------------------------------------
def _blur_sequence(arr: np.ndarray, sigma: float = 1.5) -> np.ndarray:
    """Return a mildly blurred copy of `arr` using a 1‑D Gaussian kernel."""
    radius = int(3 * sigma)
    x = np.arange(-radius, radius + 1, dtype=float)
    kernel = np.exp(-(x ** 2) / (2.0 * sigma ** 2))
    kernel /= kernel.sum()
    blurred = np.convolve(arr, kernel, mode='same')
    # Clip to non‑negative values and renormalise to the original ℓ¹‑norm
    blurred = np.maximum(blurred, 0.0)
    total = blurred.sum()
    if total == 0.0:
        return arr.copy()
    return blurred * (arr.sum() / total)

        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (default 1000).
    kwargs : dict

    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine the problem size (use the global construction if available)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # pylint: disable=undefined-variable
    except Exception:                       # pragma: no cover

    a = np.clip(a, 0.0, None)
    a *= target_sum / a.sum()

    # ------------------------------------------------------------------
    # Keep track of the best solution discovered

    best_seq = a.copy()

    cur_seq = a.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # Annealing schedule (linear cooling)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin
    it = 0

    T_end = best_score * 0.001 + 1e-6

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1
