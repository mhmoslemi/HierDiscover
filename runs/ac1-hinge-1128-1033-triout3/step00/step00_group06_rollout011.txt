"""
Propose a low‑value step‑function by using simple power‑law shapes.

For a sequence a_i = (i+1)^p (with p≥0) and length n the evaluation
value tends to the constant 2·C(p) where

    C(p) = (p+1)^2 * B(p+1, p+1) = (p+1)^2·Γ(p+1)^2 / Γ(2p+2).

Choosing a moderate exponent (p=5 or 6) and a length larger than the
threshold 1/B(p+1,p+1) makes the maximal convolution occur at the
central shift, giving an evaluation ≈2·C(p).  For p=6 the constant is
2·C≈0.016, far below the target 1.503.

The function enumerates a few lengths and exponents, normalises each
candidate (so that all values are ≤1000 and the scale is irrelevant),
evaluates it with the provided `evaluate_sequence`, and returns the
best one found within the time budget.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate and test a few power‑law sequences (a_i ∝ (i+1)^p).
    Returns the sequence with the smallest evaluation score found
    before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()

    best_sequence = None
    best_score = float('inf')

    # candidate lengths and exponents (feel free to adjust)
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    exponents = [3, 4, 5, 6, 7]

    for n in lengths:
        for p in exponents:
            # time‑out check
            if time.time() - start_time > budget_s - 1:
                break

            # raw power‑law values
            raw = np.arange(1, n + 1, dtype=np.float64) ** p

            # normalise so that the sum is 1 (any scale works because the
            # objective is homogeneous).  This also guarantees all entries
            # are far below the 1000‑clipping limit.
            total = raw.sum()
            seq = (raw / total).tolist()

            # evaluate the candidate
            try:
                score = evaluate_sequence(seq)
            except Exception:
                # defensive: treat any error as an infinite score
                score = float('inf')

            # keep the best one
            if score < best_score:
                best_score = score
                best_sequence = seq
                print(
                    f"New best: length={n}, exponent={p}, score={score:.6f}"
                )

        # outer break if out of time
        if time.time() - start_time > budget_s - 1:
            break

    # fallback – should never be needed
    if best_sequence is None:
        best_sequence = [1.0] * 1000
        best_score = evaluate_sequence(best_sequence)

    print(
        f"Returning best sequence (length={len(best_sequence)}) with score {best_score:.6f}"
    )
    return best_sequence
