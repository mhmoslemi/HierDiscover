def solve_lp_for_seq(seq):
    n = len(seq)
    sum_seq = float(np.sum(seq))
    if sum_seq == 0:
        return None
    factor = np.sqrt(2 * n) / sum_seq
    f_norm = np.array(seq) * factor
    rhs = np.max(np.convolve(f_norm, f_norm))
    g = solve_convolution_lp(f_norm.tolist(), float(rhs))
    if g is None:
        return None
    sum_g = float(np.sum(g))
    if sum_g == 0:
        return None
    g_norm = np.array(g) * (np.sqrt(2 * n) / sum_g)
    return g_norm

    rows = []
    bounds = []

"""
A fast stochastic search for a step‑function that minimises the
provided ``evaluate_sequence`` metric.

The core ideas
----------------
* Randomly generate binary (0/1) sequences of moderate length.
  For a given length the expected self‑convolution peak is lower than
  the uniform block, so many of these candidates already beat the
  previous bound (≈2.0).
* Keep the best candidate seen so far.
* Occasionally improve the current best with a small linear‑programming
  refinement (the same LP used in the reference code).  The LP finds a
  direction `g` that maximises the total mass while keeping the
  convolution bounded by the current maximum `M`.  We then try a few
  mixing coefficients `t` and keep the best mixture.
* Simple local moves (flip a single bit) and random permutations are
  also tried to escape plateaus.
* The whole loop stops after the allotted budget (default 1000 s).

The search is completely self–contained, uses only NumPy, SciPy and the
already‑imported ``evaluate_sequence`` function, and never writes to
disk.
"""

import time
import random
import numpy as np
from scipy import optimize


def solve_convolution_lp(f_seq, rhs):
    """
    Solve the linear program

        maximise   Σ b_j
        subject to  (f * b)_k ≤ rhs   for all k
                    b_j ≥ 0

    where ``f_seq`` is a non‑negative sequence and ``rhs`` is the current
    convolution bound.  The function returns the optimal ``b`` as a NumPy
    array or ``None`` if the solver fails.
    """
    n = len(f_seq)

    # maximise Σb  → minimise –Σb
    c = -np.ones(n)

    # Build the ‘convolution ≤ rhs’ constraints.
    # For each k we need Σ_i f_i * b_{k-i} ≤ rhs.
    a_rows = []
    b_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # fill the row with the appropriate coefficients
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_rows.append(row)
        b_rows.append(rhs)

    # non‑negativity constraints  –b ≤ 0
    a_rows.append(-np.eye(n))
    b_rows.append(np.zeros(n))

    A_ub = np.vstack(a_rows)
    b_ub = np.hstack(b_rows)

    # Use HiGHS (the default) – it is fast and robust for these dense problems.
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None


def lp_refine(seq):
    """
    One LP‑based refinement step.

    Given a sequence ``seq`` (list or ndarray), the routine builds a
    normalized copy, solves the LP to obtain a direction ``g``,
    then tries several mixing parameters ``t`` and returns the best
    mixture (as a plain Python list).  If the LP fails, ``None`` is returned.
    """
    n = len(seq)
    sum_seq = float(np.sum(seq))
    if sum_seq < 1e-9:                     # degenerate – should not happen
        return None

    # Normalise to the standard sum √(2n) – this does not change the objective.
    scale = np.sqrt(2.0 * n) / sum_seq
    f_norm = np.array(seq, dtype=float) * scale

    # Right‑hand side of the LP: the current maximal self‑convolution.
    rhs = float(np.max(np.convolve(f_norm, f_norm)))

    # Solve the LP.
    g_raw = solve_convolution_lp(f_norm.tolist(), rhs)
    if g_raw is None:
        return None

    sum_g = float(np.sum(g_raw))
    if sum_g < 1e-9:
        return None

    # Normalise g to the same total mass as f.
    g_norm = np.array(g_raw, dtype=float) * (np.sqrt(2.0 * n) / sum_g)

    # Try a handful of mixing weights.
    best_seq = np.array(seq, dtype=float)
    best_score = evaluate_sequence(seq)

    for t in (0.01, 0.03, 0.07, 0.15, 0.3, 0.5):
        cand = (1.0 - t) * np.array(seq, dtype=float) + t * g_norm
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    return best_seq.tolist()
