def project_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    # algorithm
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - s) / (np.arange(1, n+1)) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - s) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

**Project onto simplex**: Use algorithm by Duchi et al.

```
def project_simplex(v, s=1.0):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError('v must be a 1D array')
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - s) / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - s) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def refine_sequence_pg(a, n_iter=200, step=0.01):
    n = a.shape[0]
    for _ in range(n_iter):
        conv = np.convolve(a, a)
        k = np.argmax(conv)
        # gradient
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_arr = np.arange(i_start, i_end + 1)
        j_arr = k - i_arr
        grad[i_arr] = 2.0 * a[j_arr]  # gradient of max term
        a_new = a - step * grad
        a_new = project_simplex(a_new, 1.0)
        # optionally adapt step if objective increases
        # compute new conv max
        if np.convolve(a_new, a_new).max() < conv.max():
            a = a_new
        else:
            step *= 0.5
            if step < 1e-6:
                break
    return a

"""
propose_candidate

Algorithm
---------
The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

Since the score is homogeneous we may fix sum(a)=1.
For a fixed length n the problem reduces to minimizing the
maximum entry of the auto‑convolution.  A uniform sequence gives
max(conv)=1/n and score=2, but a “U‑shaped’’ distribution (large
weights at the ends, small in the middle) reduces the central peak
while keeping the end peaks moderate.  The optimizer therefore
searches over simple parametric families of symmetric U‑shapes:

* exponential decay from the ends:
      a_i ∝ exp(-λ·min(i, n‑1-i))

* power‑law decay:
      a_i ∝ (min(i, n‑1-i)+1)^{-p}

For each length we scan a grid of λ and p values, keep the best
candidate, and optionally improve it with a few steps of projected
gradient descent on the non‑smooth objective (sub‑gradient w.r.t. the
max‑convolution entry).  Random Dirichlet starts are also tried to
escape local minima.

The whole search runs until the supplied time budget expires (≈
1000 s).  The best sequence found is returned as a plain list of
floats; it is already non‑negative and normalized to sum 1, so the
evaluation routine will accept it directly.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex {x>=0, sum(x)=s}
def project_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if s < 0:
        raise ValueError("s must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # find rho
    rho_arr = np.where(u - (cssv - s) / (np.arange(1, n + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = (cssv[rho] - s) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------
# Shape families
def generate_exponential_sequence(n, lam):
    """Symmetric exponential decay from the ends, normalized to sum 1."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-lam * d)
    a /= a.sum()
    return a

def generate_power_sequence(n, p):
    """Symmetric power‑law decay from the ends, normalized to sum 1."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1) ** (-p)      # +1 avoids division by zero when d=0
    a /= a.sum()
    return a

# --------------------------------------------------------------
# Simple projected‑gradient refinement for the non‑smooth objective.
def refine_sequence_pg(a, max_iter=200, init_step=0.02):
    """
    Perform a few projected‑gradient steps that try to reduce
    max(conv(a,a)).  The gradient of the maximal convolution entry
    with respect to a is 2*a shifted by the index of that entry.
    The sequence is kept on the simplex (non‑negative, sum=1).
    """
    n = a.shape[0]
    step = init_step
    for _ in range(max_iter):
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))          # index of current maximum
        # compute sub‑gradient of the max‑entry
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        grad[i_idx] = 2.0 * a[j_idx]

        prev_score = 2.0 * n * conv.max()
        a_candidate = a - step * grad
        a_candidate = project_simplex(a_candidate, s=1.0)
        conv_cand = np.convolve(a_candidate, a_candidate)
        cand_score = 2.0 * n * conv_cand.max()

        # accept if we improve, otherwise shrink the step size
        if cand_score < prev_score:
            a = a_candidate
            step = min(step * 1.2, 0.1)  # slowly increase step when successful
        else:
            step *= 0.5
            if step < 1e-6:
                break
    return a

# --------------------------------------------------------------
# Main search over lengths and shape parameters
def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0   # safety margin

    best_seq = None
    best_score = float('inf')

    # lengths to try – larger lengths give a better theoretical lower bound
    lengths = [100, 200, 400, 800, 1600]

    for n in lengths:
        if time.time() > deadline:
            break

        # ---- uniform baseline -------------------------------------------------
        a_uni = np.full(n, 1.0 / n)
        score_uni = 2.0 * n * np.max(np.convolve(a_uni, a_uni))
        if score_uni < best_score:
            best_score = score_uni
            best_seq = a_uni.copy()
            print(f"[{n}] uniform score = {score_uni:.6f}")

        # ---- parameter grids ---------------------------------------------------
        max_lambda = np.log(n) + 1.0          # enough to push mass to the ends
        lam_grid = np.linspace(0.0, max_lambda, 30)
        p_grid = np.linspace(0.0, 6.0, 30)

        # exponential family
        for lam in lam_grid:
            if time.time() > deadline:
                break
            a = generate_exponential_sequence(n, lam)
            score = 2.0 * n * np.max(np.convolve(a, a))
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{n}] exp λ={lam:.3f} score = {score:.6f}")

        # power‑law family
        for p in p_grid:
            if time.time() > deadline:
                break
            a = generate_power_sequence(n, p)
            score = 2.0 * n * np.max(np.convolve(a, a))
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{n}] pow p={p:.3f} score = {score:.6f}")

        # ---- local refinement of the current best for this length ----------
        if best_seq is not None and best_seq.shape[0] == n:
            # a few gradient steps (less iterations for large n)
            iters = 200 if n <= 400 else 100
            a_ref = refine_sequence_pg(best_seq.copy(),
                                        max_iter=iters,
                                        init_step=0.02)
            score_ref = 2.0 * n * np.max(np.convolve(a_ref, a_ref))
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{n}] refined score = {score_ref:.6f}")

        # ---- random Dirichlet starts + light refinement ---------------------
        for _ in range(12):
            if time.time() > deadline:
                break
            a = np.random.rand(n)
            a /= a.sum()
            # a light refinement to escape very bad local points
            a = refine_sequence_pg(a, max_iter=50, init_step=0.03)
            score = 2.0 * n * np.max(np.convolve(a, a))
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{n}] random start improved score = {score:.6f}")

        # early exit if we have already reached the target bound
        if best_score < 1.51:
            print(f"Target vicinity reached (score={best_score:.6f}) – stop.")
            break

    # Fallback in the unlikely case nothing was found
    if best_seq is None:
        best_seq = generate_exponential_sequence(20, lam=0.5)

    # Ensure the sequence is a plain Python list of floats (the evaluator
    # expects a list).  The values are already non‑negative and sum≈1.
    return best_seq.tolist()
