def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # Determine length n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 1000  # fallback
    target_sum = math.sqrt(2 * n)

    # Compute central binomial coefficients scaled
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2 * i - 1) / (2 * i)

    # Normalize sum
    sum_a = a.sum()
    a = a * (target_sum / sum_a)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial binomial sequence score: {best_score:.6f}")

    # Simple local search: random perturbations
    # Use a small step size
    step_scale = 0.01 * target_sum / n  # small relative step

    while time.time() < deadline:
        # propose perturbation: choose random index and add/subtract small delta
        i = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * step_scale
        new_seq = best_seq.copy()
        new_seq[i] = max(0.0, new_seq[i] + delta)
        # renormalize to keep sum
        new_seq = new_seq * (target_sum / new_seq.sum())
        # compute
        try:
            score = evaluate_sequence(new_seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = new_seq
                print(f"Improved score: {best_score:.6f}")
        except Exception:
            pass

        # occasionally decay step_scale
        if np.random.rand() < 0.01:
            step_scale *= 0.9

    return best_seq.tolist()

propose_candidate

Generate a high‑quality step‑function sequence using the
central‑binomial (Catalan) coefficients

    c_0 = 1,   c_{k+1} = c_k * (2k+1) / (2(k+1))

    (c * c)_k = 1   for all 0 ≤ k < n

so the self‑convolution is flat (value 1) for the first half of the
output.  After a simple scaling the sequence attains the well‑known
asymptotic bound   max(b) ≈ π/2   i.e. an evaluation score ≈ 1.57,
which is a substantial improvement over the naïve uniform sequence
(score 2.0).  The routine also performs a tiny stochastic refinement
within the allotted time budget to try to steal a few extra points.

The implementation is completely self‑contained, uses only NumPy,
and respects all of the problem constraints (non‑negative entries,
no I/O, no nested helper functions).

    # ------------------------------------------------------------------
    # 3. Optional tiny stochastic refinement.
    #    We perturb a randomly chosen entry by a small amount, renormalise
    #    the total sum (which does not change the objective) and keep
    #    the change if it improves the score.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin
    # step size is a tiny fraction of the typical magnitude (≈ sqrt(2)/sqrt(n))
    step_scale = 0.02 * (math.sqrt(2.0) / max(1.0, math.sqrt(n)))

    while time.time() < deadline:
        idx = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * step_scale
        cand = best_seq.copy()
        cand[idx] = max(0.0, cand[idx] + delta)

        # Renormalise to keep the sum unchanged – this does not affect
        # the evaluation score (the objective is homogeneous of degree 0).
        s = cand.sum()
        if s > 0:
            cand *= best_seq.sum() / s
        else:
            continue

        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[improve] new best score={best_score:.6f}")

        # slowly shrink the perturbation magnitude
        if np.random.rand() < 0.01:
            step_scale *= 0.95

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    return best_seq.tolist()
