"""propose_candidate
A fast heuristic that starts from the analytic 1/√ shape (the known extremal
function for the continuous problem) and then performs many small
gradient‑type steps that reduce the largest entry of the self‑convolution.
The method works entirely with NumPy, uses FFT for the convolution (O(n log n))
and respects all constraints of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: convolution using a power‑of‑two FFT (real valued, fast)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of two real arrays using FFT."""
    n = len(x) + len(y) - 1
    # next power of two for speed
    N = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:n]
    return conv


# ----------------------------------------------------------------------
# Core optimisation step
# ----------------------------------------------------------------------
def _improve_sequence(a: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    Starting from vector ``a`` (non‑negative), iteratively move weight away
    from the indices that contribute to the current maximal convolution entry.
    The sum of the vector is kept constant (scale‑invariance of the objective)
    and all elements stay non‑negative.
    """
    n = len(a)
    sum_a = np.sum(a)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # main loop – runs until the time budget expires
    while time.time() < time_deadline:
        # 1. convolution and position of the current maximum
        conv_full = _fft_convolve(a, a)                # length 2n‑1
        max_val = conv_full.max()
        # numerical tolerance – treat all entries within 1 e‑12 of the max as active
        tol = max_val * 1e-12
        max_positions = np.where(conv_full >= max_val - tol)[0]
        # pick one of the active positions (randomly for a tiny amount of noise)
        k = max_positions[np.random.randint(len(max_positions))]

        # 2. gradient of the (non‑smooth) max operator:
        #    g_i = 2 * a_{k-i} for those i where the index exists, else 0
        grad = np.zeros_like(a)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals
        grad[i_vals] = 2.0 * a[j_vals]

        # 3. choose a step size – we keep the update modest to stay in the
        #    neighbourhood of the current point.
        #    The step is expressed as a fraction of the mean entry.
        mean_a = sum_a / n
        max_grad = grad.max()
        if max_grad <= 0:
            # nothing to improve – stop early
            break
        # try a few decreasing step sizes and keep the best improvement
        candidate = None
        candidate_score = np.inf
        for factor in (0.5, 0.25, 0.125, 0.0625):
            eta = factor * mean_a / max_grad           # scale‑invariant step
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0.0)              # enforce non‑negativity
            # re‑normalise to the original sum (scale invariance of the objective)
            new_sum = a_new.sum()
            if new_sum < 1e-8:          # avoid division by zero
                continue
            a_new *= sum_a / new_sum
            cur_score = evaluate_sequence(a_new.tolist())
            if cur_score < candidate_score:
                candidate_score = cur_score
                candidate = a_new

        # if none of the trial steps improved, perform a tiny random jitter
        if candidate is None or candidate_score >= best_score:
            # occasional random perturbation to escape plateaus
            if np.random.rand() < 0.02:
                idx = np.random.randint(n)
                delta = 0.01 * mean_a
                a[idx] = max(0.0, a[idx] + np.random.randn() * delta)
            continue

        # accept the best trial step
        a = candidate
        if candidate_score < best_score:
            best_score = candidate_score
            best_seq = candidate.copy()
            # simple progress report
            print(f"New best score {best_score:.6f} at t={time.time():.2f}")

    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generates a sequence of non‑negative heights that tries to minimise the

    The procedure:

      2. Optionally start from the previously supplied ``height_sequence_1`.
      3. Run a fast gradient‑type optimiser that repeatedly reduces the
         contribution of the current maximal convolution entry while keeping
         the total sum fixed.
      4. Return the best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5          # a few seconds margin

    # ------------------------------------------------------------------
    # 1. Choose an initial construction.
    # ------------------------------------------------------------------
    try:
        # If the previous best construction exists, use it as a candidate.
        prev = np.array(height_sequence_1, dtype=float)
        # Ensure non‑negativity and a reasonable length.
        if prev.size > 0 and np.any(prev < 0):
            prev = np.maximum(prev, 0.0)
        init_seq = prev
        init_score = evaluate_sequence(init_seq.tolist())
        print(f"Starting from previous sequence, score = {init_score:.6f}")
    except Exception:
        init_seq = None
        init_score = np.inf

    # Analytic 1/√ shape (length can be tuned – we pick 5 000 to match earlier work)
    n = 5030
    analytic = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))

    # Evaluate the analytic seed
    analytic_score = evaluate_sequence(analytic.tolist())
    print(f"Analytic 1/√ seed, length={n}, score={analytic_score:.6f}")

    # Decide which seed to use
    if init_score < analytic_score:
        current = init_seq.copy()
        best_overall = init_seq.copy()
        best_score = init_score
        print("Using previous best as the starting point.")
    else:
        current = analytic.copy()
        best_overall = analytic.copy()
        best_score = analytic_score
        print("Using analytic 1/√ as the starting point.")

    # ------------------------------------------------------------------
    # 2. Run the optimisation for as long as we have time.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Perform a local improvement phase (few hundred iterations)
        # Spend at most a few seconds per phase to keep the loop responsive.
        phase_deadline = min(deadline, time.time() + 2.0)
        improved = _improve_sequence(current, phase_deadline)
        # Evaluate the result of the phase
        phase_score = evaluate_sequence(improved.tolist())
        if phase_score < best_score:
            best_score = phase_score
            best_overall = improved.copy()
            print(f"Global best updated: {best_score:.6f}")
        # Prepare for the next phase – start from the best we have so far
        current = best_overall.copy()

        # Occasionally try a completely fresh random start (helps avoid

            if rand_seq.sum() < 0.01:
                rand_seq += 0.1
            current = rand_seq
            print(f"Random restart with length {rand_len}")

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"Finished search – best score = {best_score:.6f}")
    # Convert to plain Python list as required.
    return best_overall.tolist()
