def hill_climb(sequence, budget):
    best_seq = sequence[:]
    best_score = evaluate_sequence(best_seq)
    n = len(sequence)
    S = np.sqrt(2*n)
    start = time.time()
    while time.time() < budget:
        # compute convolution
        conv = np.convolve(sequence, sequence)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]

        # compute gradient for that shift
        # Create array zeros.
        grad = np.zeros(n)
        # i iterates over valid i range
        i_start = max(0, max_k - (n-1))
        i_end = min(n-1, max_k)  # inclusive
        for i in range(i_start, i_end+1):
            j = max_k - i
            if i != j:
                grad[i] += sequence[j]
                grad[j] += sequence[i]  # symmetrical
            else:
                grad[i] += sequence[i]  # only one

        # step direction: move opposite gradient
        # Use step size proportional to current values
        step = 0.05  # tune
        new_seq = sequence - step * grad
        # ensure non-negative
        new_seq = np.maximum(0, new_seq)
        # renormalize sum to S
        sum_seq = np.sum(new_seq)
        if sum_seq == 0:
            continue
        new_seq = new_seq * (S / sum_seq)

        # evaluate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.tolist()
            sequence = new_seq  # accept improvement
        else:
            # maybe accept with some probability? not necessary
            # revert or keep old
            # maybe random jitter occasionally
            if np.random.rand() < 0.01:
                sequence = new_seq

    return best_seq

"""
Propose a high‑quality step‑function (non‑negative height sequence).

Algorithm
---------
1.  Initialise a monotone decreasing exponential distribution.
    For length *n* we solve for the optimal decay `ε` that minimises the
    peak of the self‑convolution of a pure exponential (the solution of
    `exp(x)=1+2x`).  This gives a very good starting point (≈1.80 on the
    benchmark).

2.  Greedy “mass‑transfer” refinement.
    * Compute the convolution and locate the index `k_max` where it attains
      its maximum.
    * All pairs `(i, j)` with `i+j = k_max` contribute to that peak.
      The indices participating in these pairs form the *active set*.
    * Transfer a small fraction of the total mass in the active set to
      indices outside it (the *inactive set*) – this directly reduces the
      products that form the peak while keeping the L¹‑norm (∑aᵢ) fixed.
    * After each transfer we renormalise the whole sequence to the exact
      target sum `sqrt(2 n)`.  If the new sequence improves the evaluation
      score we keep it, otherwise we keep the old one and shrink the step
      size.

3.  Random perturbations are occasionally injected to escape possible
    plateaus.

The routine runs for at most `budget_s` seconds (default 1000 s) and
returns the best sequence discovered.  All numbers are non‑negative and
well below the 1000‑clamp, so the evaluation function is never forced to
truncate them.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def optimal_exponential_parameter(n):
    """
    Solve for the optimal decay ε = x_opt / n where x_opt satisfies
        exp(x) = 1 + 2*x.
    Returns the decay ε and the base r = 1‑ε.
    """
    # Newton solve for x in exp(x) - 1 - 2*x = 0
    x = 1.2                       # initial guess (close to the root)
    for _ in range(30):
        f = np.exp(x) - 1 - 2 * x
        df = np.exp(x) - 2
        x_new = x - f / df
        if abs(x_new - x) < 1e-12:
            break
        x = x_new
    eps = x / n
    r = 1.0 - eps
    return eps, r


def exponential_initial_sequence(n, eps, r):
    """
    Build a decreasing exponential sequence a_i = a0 * r**i
    with ∑a_i = sqrt(2*n).  The formula for a0 uses the geometric sum.
    """
    S = np.sqrt(2.0 * n)                     # required L1‑norm
    # r**n may under‑/overflow for extreme values; use log1p for safety
    r_pow_n = np.exp(n * np.log1p(-eps))     # (1‑ε)**n
    denom = 1.0 - r_pow_n
    a0 = S * eps / denom                     # a0 = S·ε / (1‑rⁿ)
    # generate the whole vector
    indices = np.arange(n, dtype=np.float64)
    seq = a0 * np.exp(indices * np.log(r))    # a0·r**i
    return seq


def max_convolution_info(seq):
    """
    Compute full convolution, its maximum value and the index where it
    occurs.  Returns (conv, max_val, max_idx).
    """
    conv = np.convolve(seq, seq)
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return conv, max_val, max_idx


def transfer_mass(seq, max_idx, step_frac, target_sum):
    """
    Perform one greedy mass‑transfer step.
    *seq* – current non‑negative array (float64) of length n.
    *max_idx* – index of the current maximal convolution coefficient.
    *step_frac* – fraction of the total active mass to move (0 < step≤1).
    *target_sum* – exact L¹‑norm we must keep (√(2 n)).
    Returns a new sequence (still non‑negative) and a boolean telling
    whether any change was actually performed.
    """
    n = seq.size
    # indices i that appear in a pair (i, j) with i+j = max_idx
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)          # inclusive
    active_mask = np.zeros(n, dtype=bool)

    # build the active mask (both i and its partner j)
    for i in range(i_start, i_end + 1):
        j = max_idx - i
        active_mask[i] = True
        active_mask[j] = True

    if not active_mask.any():
        return seq, False

    inactive_mask = ~active_mask
    n_inactive = np.count_nonzero(inactive_mask)
    if n_inactive == 0:
        # nothing to move mass to – give up this operation
        return seq, False

    active_sum = float(seq[active_mask].sum())
    if active_sum == 0.0:
        return seq, False

    # total mass we intend to relocate this step
    delta = step_frac * active_sum

    # proportional reduction inside the active set
    reduction = (seq[active_mask] / active_sum) * delta
    new_seq = seq.copy()
    new_seq[active_mask] -= reduction

    # uniform distribution of the removed mass over the inactive set
    addition = delta / n_inactive
    new_seq[inactive_mask] += addition

    # numerical safety: clip to non‑negative, renormalise exact L1‑norm
    np.maximum(new_seq, 0, out=new_seq)
    current_sum = new_seq.sum()
    if current_sum <= 0:
        # fallback – keep the old sequence
        return seq, False
    new_seq *= target_sum / current_sum
    return new_seq, True


def random_perturbation(seq, magnitude, target_sum, rng):
    """
    Add a small Gaussian noise to a random subset of coordinates.
    Afterwards renormalise to the required L1‑norm.
    """
    n = seq.size
    pert = rng.normal(scale=magnitude, size=n)
    new_seq = seq + pert
    np.maximum(new_seq, 0, out=new_seq)
    s = new_seq.sum()
    if s == 0:
        return seq                       # discard pathological case
    new_seq *= target_sum / s
    return new_seq

# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Generate a step‑function (list of non‑negative heights) whose
    evaluation score is as low as possible within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Determine problem size (use the provided construction if available)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        # a reasonable default if the global is missing
        n = 1742

    target_sum = np.sqrt(2.0 * n)   # the L1 value that makes evaluate = max_conv

    # --------------------------------------------------------------
    # 1) Initialise with the analytically motivated exponential shape
    # --------------------------------------------------------------
    eps, r = optimal_exponential_parameter(n)
    seq = exponential_initial_sequence(n, eps, r)

    # Occasionally try a few neighbouring ε values (helps with the
    # discrete‑grid effect of the finite length)
    for factor in (0.7, 0.9, 1.1, 1.3):
        alt_eps = eps * factor
        alt_r = 1.0 - alt_eps
        cand = exponential_initial_sequence(n, alt_eps, alt_r)
        if evaluate_sequence(cand.tolist()) < evaluate_sequence(seq.tolist()):
            seq = cand

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # --------------------------------------------------------------
    # 2) Greedy refinement driven by the current maximal convolution
    # --------------------------------------------------------------
    # dynamic step size – start relatively aggressive and decay if no progress
    step_frac = 0.25
    min_step = 1e-4
    max_step = 0.5
    no_improve_counter = 0

    while time.time() < deadline:
        conv, max_val, max_idx = max_convolution_info(seq)

        # mass‑transfer step
        new_seq, changed = transfer_mass(seq, max_idx, step_frac, target_sum)

        # occasional random jitter to avoid stagnation
        if not changed or rng.random() < 0.02:
            new_seq = random_perturbation(seq, magnitude=0.02 * target_sum / n,
                                          target_sum=target_sum, rng=rng)

        # evaluate
        try:
            score = evaluate_sequence(new_seq.tolist())
        except Exception:
            # any numerical mishap – discard candidate
            score = np.inf

        # accept if strictly better
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            seq = new_seq
            # succeed – we can be a bit bolder next time
            step_frac = min(max_step, step_frac * 1.05)
            no_improve_counter = 0
        else:
            # not an improvement – shrink the step size gradually
            step_frac = max(min_step, step_frac * 0.92)
            no_improve_counter += 1
            # after a while try a fresh restart from the exponential base
            if no_improve_counter >= 150:
                eps, r = optimal_exponential_parameter(n)
                seq = exponential_initial_sequence(n, eps, r)
                no_improve_counter = 0

        # periodic progress report (does not affect the returned answer)
        if int(time.time()) % 30 == 0:
            print(f"[{time.time():.0f}] Best score: {best_score:.6f}  step: {step_frac:.5f}")

    # --------------------------------------------------------------
    # Return the best sequence (as plain Python list of floats)
    # --------------------------------------------------------------
    print(f"Finished optimisation – best evaluation = {best_score:.6f}")
    return best_seq.tolist()
