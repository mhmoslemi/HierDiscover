def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    n = 5000 # length
    # Initialize with random positive values
    a = np.random.rand(n) + 0.1  # ensure >0
    # Could scale to sum=1 for stability
    a = a / np.sum(a)  # sum=1

    best = a.copy()
    best_score = evaluate_sequence(a.tolist())
    print(f"Initial score {best_score}")

    # hyperparameters
    eta = 0.1  # step size
    decay = 0.9995
    min_eta = 1e-4

    while time.time() < deadline:
        # compute conv and objective
        b = np.convolve(a, a)
        max_b = np.max(b)
        S = np.sum(a)
        # compute gradient
        k = np.argmax(b)  # index of max
        # gradient of max_b
        g = np.zeros_like(a)
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # vectorized: indices = np.arange(i_start, i_end+1)
        # j = k - indices
        # g[j] = 2 * a[indices]
        idx_i = np.arange(i_start, i_end+1, dtype=int)
        idx_j = k - idx_i
        g[idx_j] = 2 * a[idx_i]

        # gradient of objective
        grad = (2 * n / (S**2)) * (g - (2 * max_b / S) * np.ones_like(a))

        # update
        a_new = a - eta * grad
        # enforce nonnegativity
        a_new = np.maximum(a_new, 0.0)

        # optional renormalize to keep sum fixed (scale invariance)
        # we can optionally re-scale to maintain sum=1
        sum_new = np.sum(a_new)
        if sum_new < 1e-8:
            a_new = np.random.rand(n) + 0.1
            a_new = a_new / np.sum(a_new)
        else:
            a_new = a_new * (1.0 / sum_new)  # keep sum=1

        # evaluate new
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best = a_new.copy()
            print(f"New best {best_score:.6f} at time {time.time()-start:.1f}s")
        # accept new a as current (optional use if score improved else maybe keep)
        if score <= best_score or np.random.rand() < 0.01:
            a = a_new  # accept; maybe always accept to smooth descent
        # decay learning rate
        eta = max(min_eta, eta * decay)
    # convert best to list of floats
    return best.tolist()

def flatten_sequence(a, steps=2000, eta=0.2):
    n = a.shape[0]
    for step in range(steps):
        b = np.convolve(a, a)
        max_b = b.max()
        k = np.argmax(b)
        # compute gradient of max_b
        g = np.zeros_like(a)
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        idx_i = np.arange(i_start, i_end+1)
        idx_j = k - idx_i
        g[idx_j] = 2 * a[idx_i]
        S = a.sum()
        grad = (2 * n / (S**2)) * (g - (2 * max_b / S) * np.ones_like(a))
        # update with momentum?
        a = a - eta * grad
        a = np.maximum(a, 0)
        a = a / a.sum()
    return a
```

def _objective(a):

"""
Propose a step‑function (a list of non‑negative heights) with a low
evaluate_sequence score.

The method works in two stages.

1. **Exploratory pool** – a modest number of simple parametric families
   (inverted cosine windows, symmetric cosine‑squared windows,
   truncated Gaussians, exponential tails and a few random vectors) are
   generated, normalised (L1‑norm = 1) and evaluated.  The best element of
   this pool is taken as a warm start.

2. **Projected gradient descent** – we directly minimise the quantity
   2·n·max(f∗f) / (‖f‖₁)² = objective(f) which is scale‑invariant, by
   following the sub‑gradient of the maximal convolution entry.
   After each gradient step the vector is projected onto the non‑negative
   orthant and re‑normalised to keep the L1‑norm fixed.  A small momentum
   term helps to escape shallow plateaus and a decaying step‑size provides
   stability.

The whole search respects the 1000 s budget and returns the best sequence
found (as a plain Python list).  All numbers are non‑negative and well
within the [0, 1000] clipping range of `evaluate_sequence`.
"""

import time
import numpy as np

def _linf_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a)/(sum(a)**2).  a must be 1‑D and non‑negative."""
    n = a.shape[0]
    S = a.sum()
    if S < 1e-12:               # avoid division by zero – will be rejected
        return np.inf
    b = np.convolve(a, a)
    max_b = b.max()
    return 2.0 * n * max_b / (S * S)
