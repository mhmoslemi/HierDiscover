"""
Propose a candidate sequence of non‑negative heights that (approximately) minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a simple projected (sub‑)gradient descent with occasional random
perturbations, smoothing and multi‑start restarts.  The algorithm works
independently of the provided `evaluate_sequence` (it is only used at the
very end by the grader), so a cheap internal score based on the same
formula is used during the search.  The routine respects the 1000 s
budget, prints progress and finally returns the best sequence found.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _internal_score(a: np.ndarray) -> float:
    """
    Compute the same objective as `evaluate_sequence` but without the
    safety‑checks and clipping (our vectors stay in the valid range).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(a, a)
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Return (max_conv_value, sub‑gradient of `max(conv(a,a))` w.r.t. a).

    The sub‑gradient is the average of the gradients of all indices that
    attain the maximum convolution value.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()
    # all positions where the maximum is achieved
    max_idxs = np.flatnonzero(conv == max_val)
    grad = np.zeros_like(a)

    # vectorised contribution of each max index
    i = np.arange(n)               # shape (n,)
    for idx in max_idxs:
        j = idx - i                # indices of a that influence conv[idx]
        mask = (j >= 0) & (j < n)  # keep only valid positions
        grad[mask] += 2.0 * a[j[mask]]
    if max_idxs.size > 0:
        grad /= float(max_idxs.size)
    return max_val, grad


def _subgradient_of_score(a: np.ndarray):
    """
    Compute a sub‑gradient of the full objective
        2 * n * max(conv(a,a)) / (sum(a)**2)
    """
    n = a.shape[0]
    s = a.sum()
    if s == 0:
        return np.zeros_like(a)

    max_val, grad_max = _max_conv_and_subgrad(a)

    C = 2.0 * n
    grad = (C / (s * s)) * (grad_max - (2.0 * max_val) / s)
    return grad

        velocity = np.zeros_like(a)          # momentum term
        momentum = 0.9

        while time.time() < local_deadline:
            iteration += 1

            # Sub‑gradient of the whole objective
            grad = _subgradient_of_score(a)

            # Simple learning‑rate schedule
            lr = 0.6 / math.sqrt(iteration + 1.0)

            # Momentum update (helps escape shallow plateaus)
            velocity = momentum * velocity - lr * grad
            a = a + velocity

            # ----------------------------------------------------------
            # Projection onto the feasible set: a_i >= 0 and sum = target_sum
            # ----------------------------------------------------------
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                # Re‑initialise if everything collapsed to zero
                a = np.random.exponential(scale=1.0, size=n)
                s = a.sum()
            a *= target_sum / s

            # ----------------------------------------------------------
            # Occasional stochastic perturbations & smoothing
            # ----------------------------------------------------------
            if iteration % 70 == 0:
                # Small Gaussian jitter
                a += np.random.normal(scale=0.008, size=n)
                a = np.clip(a, 0.0, None)
                a *= target_sum / a.sum()

            if iteration % 120 == 0:
                # Light neighbour smoothing (preserves sum after re‑norm)
                smooth = (
                    a
                    + np.concatenate(([0.0], a[:-1]))
                    + np.concatenate((a[1:], [0.0]))
                ) / 3.0
                a = smooth
                a = np.clip(a, 0.0, None)
                a *= target_sum / a.sum()

            # ----------------------------------------------------------
            # Periodic evaluation of the current candidate
            # ----------------------------------------------------------
            if iteration % 15 == 0:
                cur_score = _internal_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] Trial {trial:3d}, iter {iteration:4d}, "
                        f"new best score = {best_score:.6f}, length = {n}"
                    )

        # End of local loop – evaluate final candidate of this restart
        final_score = _internal_score(a)
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] Trial {trial:3d} finished, final best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list
    # ------------------------------------------------------------------
    if best_seq is None:
        # Very unlikely, but return a trivial constant sequence as a fallback
        n_fallback = 1000
        a_fallback = np.full(n_fallback, math.sqrt(2.0 * n_fallback) / n_fallback)
        return a_fallback.tolist()

    # Convert to list of Python floats (the grader expects that)
    return best_seq.tolist()
