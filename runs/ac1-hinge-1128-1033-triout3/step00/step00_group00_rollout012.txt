"""
propose_candidate
-----------------

Searches for a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is based on the analytically optimal “constant‑convolution”
sequence whose coefficients are

    b_k = C(2k, k) / 4**k   (central binomial / 4^k)

For a truncated prefix of length n this sequence satisfies
max(conv(b, b)) = 1, so the objective reduces to

    score(n) = 2 * n / (sum_{k < n} b_k) ** 2

The score can be evaluated instantly for any n.  We scan a range of lengths,
pick the length with the smallest analytic score, and finally perform a very
lightweight local‑search (random perturbations) to see whether small tweaks
improve the true objective evaluated by `evaluate_sequence`.  The whole
procedure respects the 1000 s time budget.

The returned sequence consists of the first `best_n` coefficients
`b_0 … b_{best_n‑1}` (all non‑negative, ≤1, well inside the 0‑1000 clamp).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the constant‑convolution coefficients up to a given size
# ----------------------------------------------------------------------
def generate_binomial_prefix(max_n: int):
    """
    Returns a list `coeffs` where coeffs[k] = binom(2k, k) / 4**k for 0 <= k < max_n.
    Computed by the recurrence  b_{k+1} = b_k * (2k+1) / (2(k+1)).
    """
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0                     # b_0
    coeffs[0] = cur
    for k in range(1, max_n):
        # recurrence derived from central binomial identity
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Produce a sequence of non‑negative heights with a low evaluation score.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – analytical scan over many lengths
    # ------------------------------------------------------------------
    # Upper bound on length – the binomial coefficients decay ~ 1/sqrt(k)
    # Lengths above ~3000 give virtually no extra gain while costing time.
    MAX_LEN = 3000

    # generate once, reuse for all lengths
    coeffs = generate_binomial_prefix(MAX_LEN)

    cumulative = 0.0
    best_len = 1
    best_score = float('inf')
    # iterate and keep the prefix with minimal analytic score
    for n in range(1, MAX_LEN + 1):
        cumulative += coeffs[n - 1]          # sum_{k < n} b_k
        # max of self‑convolution of this prefix is exactly 1
        score = 2.0 * n / (cumulative * cumulative)
        if score < best_score:
            best_score = score
            best_len = n

    # ------------------------------------------------------------------
    # Phase 2 – construct the starting candidate from the best prefix
    # ------------------------------------------------------------------
    best_seq = coeffs[:best_len].tolist()   # plain Python list of floats
    # sanity check with the provided evaluator (should match the analytic score)
    try:
        true_score = evaluate_sequence(best_seq)
        # keep the best observed real score
        best_real_score = true_score
        print(f"[init] length={best_len}, analytic={best_score:.6f}, "
              f"real={true_score:.6f}")
    except Exception:   # protect against any unexpected issue
        best_real_score = best_score

    # ------------------------------------------------------------------
    # Phase 3 – tiny stochastic hill‑climb (optional, limited by time)
    # ------------------------------------------------------------------
    cur_seq = best_seq[:]
    cur_score = best_real_score

    # parameters for the random walk
    sigma_rel = 0.05          # relative std‑dev of perturbations
    max_no_improve = 2000    # stop after many attempts without improvement

    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        # pick a random index and propose a small change
        idx = np.random.randint(len(cur_seq))
        old_val = cur_seq[idx]
        # Gaussian perturbation, keep non‑negative
        delta = np.random.randn() * sigma_rel * max(old_val, 1e-12)
        new_val = max(0.0, old_val + delta)

        # create a candidate copy
        cand = cur_seq[:]
        cand[idx] = new_val

        # evaluate – cheap because the sequence is short (≤ 3000)
        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            # any invalid sequence is simply discarded
            continue

        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
            # record global improvement
            if cand_score < best_real_score:
                best_seq = cand[:]
                best_real_score = cand_score
                print(f"[improve] new best {best_real_score:.6f} "
                      f"len={len(best_seq)} idx={idx}")
        else:
            no_improve += 1

    # final sanity – make sure no negative numbers slipped through
    best_seq = [float(max(0.0, x)) for x in best_seq]

    return best_seq
