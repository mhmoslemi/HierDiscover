def project_to_simplex(v):
    # standard algorithm to project onto simplex of sum 1 and nonnegative
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=12345, budget_s=1000):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    
    # list of lengths
    length_candidates = [30, 50, 80, 100, 150, 200, 300, 500]
    # families
    for n in length_candidates:
        if time.time() >= deadline: break
        # uniform
        a = np.full(n, 1.0 / n)
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score, best_seq = score, a.tolist()
        # linear decreasing
        shape = np.arange(n, 0, -1)  # n, n-1,...1
        a = shape / shape.sum()
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score, best_seq = score, a.tolist()
        # quadratic decreasing
        shape = np.arange(n, 0, -1) ** 2
        a = shape / shape.sum()
        score = evaluate_sequence(a.tolist())
        # store best
        # exponential family
        for p in np.geomspace(0.001, 0.5, num=30):
            if time.time() >= deadline: break
            shape = np.exp(-p * np.arange(n))
            a = shape / shape.sum()
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score, best_seq = score, a.tolist()
        # power law decreasing
        for p in np.linspace(0.1, 3.0, num=30):
            if time.time() >= deadline: break
            shape = (np.arange(1, n+1) ** (-p))
            a = shape / shape.sum()
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score, best_seq = score, a.tolist()
        # ends-favoring shape: a_i = 1/ ((i+1)*(n-i))**p
        for p in np.linspace(0.1, 4.0, num=20):
            if time.time() >= deadline: break
            idx = np.arange(n)
            shape = 1.0 / (( (idx+1) * (n - idx) ) ** p)
            a = shape / shape.sum()
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score, best_seq = score, a.tolist()
    # Now local refinement hill climbing around best_seq
    if best_seq is None:
        # fallback uniform 10
        best_seq = [1.0/10]*10
        best_score = evaluate_sequence(best_seq)
    a = np.array(best_seq, dtype=float)
    # hill climbing parameters
    move_factor = 0.1
    temperature = 1.0
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        # random move: choose i!=j
        i, j = rng.integers(len(a), size=2)
        if i == j:
            continue
        delta = a[i] * move_factor * rng.random()
        # ensure not exceed a[i]
        delta = min(delta, a[i])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # project negativities not needed
        # compute score
        cur_score = evaluate_sequence(a_new.tolist())
        if cur_score < best_score or rng.random() < math.exp((best_score - cur_score) / temperature):
            a = a_new
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
        # anneal temperature
        temperature = max(0.001, temperature * 0.9999)
        # occasionally adjust move_factor
        if iter_count % 1000 == 0:
            move_factor = max(0.001, move_factor * 0.9)
    print(f"Best score: {best_score:.6f}, length: {len(best_seq)}")
    return best_seq

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    # try several lengths
    for n in range(3, 31):  # moderate lengths
        if time.time() >= deadline:
            break
        # random restarts per n
        for restart in range(20):
            # initialize random Dirichlet
            a = rng.dirichlet(alpha=np.ones(n))
            cur_score = evaluate_sequence(a.tolist())
            # local search
            temp = 1.0
            for it in range(5000):
                if time.time() >= deadline:
                    break
                # propose random move: pick two indices
                i, j = rng.integers(n, size=2)
                if i == j:
                    continue
                # propose delta move from i to j
                # Choose delta proportion of a[i]
                delta = a[i] * rng.random() * 0.2  # up to 20% of source
                if delta <= 0:
                    continue
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # ensure non-negative
                if a_new[i] < 0:
                    continue
                # Evaluate
                new_score = evaluate_sequence(a_new.tolist())
                # Accept if better or with probability
                if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / temp):
                    a = a_new
                    cur_score = new_score
                # Cool down
                temp *= 0.9999
                # Record global best
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.tolist()
                    # Print progress
                    print(f"New best: {best_score:.6f} with n={n}")
            # end iters
            # after restart maybe adjust
        # end restarts
    # After loops, try refinement on best_seq using small moves

    # final refinement on best_seq

"""
Propose a step‐function (list of non‑negative heights) that minimizes the
evaluation function `evaluate_sequence`.

The algorithm consists of three stages:

1. **Analytic candidates** –  
   For each length n (3 ≤ n ≤ 30) we use the analytically derived shape
   consisting of one large entry a0 and n‑1 equal entries.  The optimal
   a0 for this family solves   a0² = 2·a0·y + (n‑2)·y²   where
   y = (1‑a0)/(n‑1).  This gives a closed‑form solution that already beats the
   uniform sequence for small n.

2. **Random parametric families** –  
   A few simple parametric families (linear, quadratic, exponential,
   power‑law, and “inverse‑product”) are generated for the same lengths and
   evaluated.

3. **Local stochastic optimisation** –  
   Starting from the best candidate found so far we perform a simulated‑annealing
   hill‐climb: a small amount of mass is transferred from one index to another.
   Moves that improve the score are always accepted; worsening moves are
   accepted with a temperature‑dependent probability.  The temperature is
   slowly cooled and the step size gradually reduced.  The search is bounded
   by the remaining time budget.

The procedure explores many short sequences (where the score can be far below
2.0) while keeping the runtime well within the 1000 s limit.
"""

import time
import math
import numpy as np

def _analytic_one_big(n: int):

    # t = fraction of the total mass that is *not* in a0
    # t = N * (N + 1 - sqrt(N)) / (N**2 + N + 1)
    t = N * (N + 1 - math.sqrt(N)) / (N * N + N + 1)
    a0 = 1.0 - t
    if n == 1:
        return [1.0]
    y = (1.0 - a0) / N
    seq = np.empty(n, dtype=float)
    seq[0] = a0
    seq[1:] = y
    return seq.tolist()


def _family_sequence(n: int, kind: str, param: float):
    """Generate a deterministic sequence of length n belonging to the
    requested family and parameter."""
    if kind == "linear":
        shape = np.arange(n, 0, -1, dtype=float)          # n, n‑1, …, 1
    elif kind == "quadratic":
        shape = np.arange(n, 0, -1, dtype=float) ** 2
    elif kind == "exponential":
        shape = np.exp(-param * np.arange(n, dtype=float))
    elif kind == "power":
        shape = (np.arange(1, n + 1, dtype=float) ** -param)
    elif kind == "inverse_product":
        idx = np.arange(n, dtype=float)
        shape = 1.0 / ((idx + 1) * (n - idx)) ** param
    else:
        raise ValueError("unknown family")
    if shape.sum() == 0:
        shape = np.ones_like(shape)
    shape /= shape.sum()
    return shape.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights (the step function)."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------
    # 1. Analytic 1‑big‑element candidates (n from 3 to 30)
    # ------------------------------------------------------------
    for n in range(3, 31):
        if time.time() >= deadline:
            break
        seq = _analytic_one_big(n)
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score, best_seq = score, seq
            print(f"[analytic] new best {best_score:.6f} (n={n})")

    # ------------------------------------------------------------
    # 2. Parametric families (small n, few parameter values)
    # ------------------------------------------------------------
    families = [
        ("linear", None),
        ("quadratic", None),
        ("exponential", np.geomspace(0.001, 0.5, 12)),
        ("power", np.linspace(0.2, 4.0, 12)),
        ("inverse_product", np.linspace(0.2, 4.0, 12)),
    ]
    for n in range(3, 31):
        if time.time() >= deadline:
            break
        for kind, params in families:
            if params is None:
                seq = _family_sequence(n, kind, 0.0)
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score, best_seq = score, seq
                    print(f"[family] new best {best_score:.6f} (n={n}, {kind})")
            else:
                for p in params:
                    if time.time() >= deadline:
                        break
                    seq = _family_sequence(n, kind, p)
                    score = evaluate_sequence(seq)
                    if score < best_score:
                        best_score, best_seq = score, seq
                        print(
                            f"[family] new best {best_score:.6f} (n={n}, {kind}, p={p:.4f})"
                        )

    # ------------------------------------------------------------
    # 3. Stochastic local optimisation (simulated annealing)
    # ------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform of length 10
        n = 10
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    a = np.array(best_seq, dtype=float)
    n = len(a)

    # temperature schedule
    temp = 1.0
    # step size factor (relative to the source entry)
    step_factor = 0.2

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # pick two distinct indices
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # propose moving a fraction of a[i] to a[j]
        delta = a[i] * step_factor * rng.random()
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        cur_score = evaluate_sequence(a_new.tolist())
        # accept if better, otherwise with Metropolis probability
        if cur_score < best_score or rng.random() < math.exp((best_score - cur_score) / temp):
            a = a_new
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[refine] new best {best_score:.6f} (len={n})")
        # cooling
        temp = max(1e-4, temp * 0.9995)
        # slowly reduce step size
        if iteration % 3000 == 0:
            step_factor = max(0.001, step_factor * 0.9)

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
