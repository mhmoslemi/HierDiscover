    edge_indices = np.r_[np.arange(edge_width), np.arange(n-edge_width, n)]
    # central region maybe same width
    # loop
    while time.time() < deadline:
        conv = compute_conv(a)
        M = conv.max()
        # find indices achieving max (could be multiple)
        ks = np.where(conv == M)[0]
        k = ks[np.random.randint(len(ks))]  # pick one max shift
        # Determine range of i for this shift
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        i_candidates = np.arange(i_min, i_max+1)
        # weights proportional to a[k-i]
        weights = a[k - i_candidates]
        if weights.sum() == 0:
            i = np.random.choice(i_candidates)
        else:
            probs = weights / weights.sum()
            i = np.random.choice(i_candidates, p=probs)
        # choose j from edges (low a) with probability favoring lower a
        # compute probability inversely proportional to a (plus epsilon)
        a_edge = a[edge_indices]
        inv_weights = 1.0 / (a_edge + 1e-12)
        probs_edge = inv_weights / inv_weights.sum()
        j = np.random.choice(edge_indices, p=probs_edge)
        if i == j:
            continue
        # move fraction of a[i] to a[j]
        delta = 0.05 * a[i]  # 5%
        # ensure delta isn't too large relative to a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renorm to target sum (to correct rounding)
        a_new = a_new / a_new.sum() * target_sum
        # compute new max
        conv_new = compute_conv(a_new)
        M_new = conv_new.max()
        # acceptance
        if M_new < M or np.random.rand() < np.exp((M - M_new) / (temperature + 1e-12)):
            a = a_new
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
            # update temperature
        temperature = max(temperature * decay, min_temp)
        # early exit if M is good
        if best_M <= 1.51:
            break
    return best_a, best_M

"""
Simple simulated‑annealing / hill‑climbing optimizer.

* Normalises every candidate so that  sum(a)=sqrt(2·n) .
* For a normalised sequence the objective returned by
  ``evaluate_sequence`` is exactly the maximum of its aperiodic
  autocorrelation (the “max‑conv” value).  Thus we minimise
  ``max(conv(a,a))``.
* Several deterministic “window” seeds (uniform, Hann, Hamming,
  half‑sine, exponential) and the previously found ``height_sequence_1`` are
  initialised.  The best seed is taken as a start.
* The main loop repeatedly:
    – computes the full convolution (FFT based – O(n log n));
    – picks a shift k where the maximum occurs;
    – chooses an index i that contributes heavily to that shift
      (probability proportional to a[k‑i]);
    – chooses a low‑valued edge index j (inverse‑probability weighting);
    – moves a small fraction (≈5 %) of a[i] to a[j];
    – accepts the move if the new maximum is lower, otherwise with a
      decreasing “temperature’’ probability (simulated annealing).
* The procedure runs until the time budget is exhausted (≈ 1000 s) and
  returns the best sequence found.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a*a (full discrete convolution) using FFT for speed."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    # tiny negative round‑off can appear – clip to non‑negative
    return np.maximum(conv, 0.0)


def _normalise(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale the vector so that its L1‑norm equals target_sum."""
    s = a.sum()
    if s == 0:
        # fall back to a uniform tiny vector
        a[:] = target_sum / a.size
        return a
    return a * (target_sum / s)


# ------------------------------------------------------------
# Local optimiser
# ------------------------------------------------------------
def _improve_sequence(a: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      init_temp: float = 1.0,
                      decay: float = 0.9995,
                      min_temp: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing + simulated annealing.
    Returns a (possibly) improved copy of ``a`` and its best max‑conv value.
    """
    n = a.size
    a = _normalise(a, target_sum)

    # parameters for the edge‑biased moves
    edge_width = max(1, n // 20)                     # ~5 % of the sequence
    edge_idx = np.r_[np.arange(edge_width),
                     np.arange(n - edge_width, n)]

    best_a = a.copy()
    best_M = _full_convolution(a).max()

    temp = init_temp

    while time.time() < deadline:
        # current convolution and the worst shift(s)
        conv = _full_convolution(a)
        M = conv.max()
        # pick one of the shifts attaining the maximum (random tie‑break)
        ks = np.flatnonzero(conv == M)
        k = ks[np.random.randint(ks.size)]

        # indices that participate in this shift
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        cand_i = np.arange(i_min, i_max + 1)

        # contribution weight = a[k‑i]; larger ⇒ higher impact on M
        w = a[k - cand_i]
        if w.sum() == 0:
            i = np.random.choice(cand_i)
        else:
            p = w / w.sum()
            i = np.random.choice(cand_i, p=p)

        # choose a destination index on the edges, preferring small values
        a_edge = a[edge_idx]
        inv = 1.0 / (a_edge + 1e-12)
        p_edge = inv / inv.sum()
        j = np.random.choice(edge_idx, p=p_edge)

        if i == j:
            continue   # nothing to move

        # transfer a modest fraction of a[i] to a[j]
        delta = 0.05 * a[i]          # 5 % of the source entry
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # re‑normalise (to correct round‑off)
        a_new = _normalise(a_new, target_sum)

        M_new = _full_convolution(a_new).max()

        # accept if improvement, otherwise with a temperature‑dependent probability
        if M_new < M or np.random.rand() < math.exp((M - M_new) / (temp + 1e-12)):
            a = a_new
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
                print(f'  ** new local best max‑conv = {best_M:.6f}')

        # temperature schedule
        temp = max(temp * decay, min_temp)

        # early stop if we already beat the target bound
        if best_M <= 1.51:
            break

    return best_a, best_M


# ------------------------------------------------------------
# Main entry point required by the evaluation harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) attains a
    max‑conv value below the current upper bound 2.0.
    The algorithm explores several sensible initial shapes and then
    performs a fast simulated‑annealing hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a safety margin

    # -----------------------------------------------------------------
    # Determine sequence length and target L1‑norm (sqrt(2·n))
    # -----------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        n = base.size
    else:
        n = 2000                     # reasonable default when no previous solution
    target_sum = math.sqrt(2.0 * n)

    # -----------------------------------------------------------------
    # Build a collection of plausible seeds
    # -----------------------------------------------------------------
    seeds = []

    # 1) uniform (all entries equal)
    uniform = np.full(n, 1.0, dtype=float)
    uniform = _normalise(uniform, target_sum)
    seeds.append(uniform)

    # 2) random positive vector
    rnd = np.random.rand(n).astype(float)
    rnd = _normalise(rnd, target_sum)
    seeds.append(rnd)

    # 3) Hann window
    i = np.arange(n, dtype=float)
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * (i + 1) / (n + 1)))
    hann = _normalise(hann, target_sum)
    seeds.append(hann)

    # 4) Hamming window
    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * (i + 1) / (n + 1))
    hamming = _normalise(hamming, target_sum)
    seeds.append(hamming)

    # 5) half‑sine (positive on the whole interval)
    half_sine = np.sin(np.pi * (i + 0.5) / n)
    half_sine = _normalise(half_sine, target_sum)
    seeds.append(half_sine)

    # 6) exponential decay (fast drop‑off)
    exp_decay = np.exp(-i / (n / 5.0))
    exp_decay = _normalise(exp_decay, target_sum)
    seeds.append(exp_decay)

    # 7) previous best if it exists (already normalised)
    if 'height_sequence_1' in globals():
        seq1 = np.asarray(height_sequence_1, dtype=float)
        seq1 = _normalise(seq1, target_sum)
        seeds.append(seq1)

    # -----------------------------------------------------------------
    # Pick the best among the seeds (cheap evaluation)
    # -----------------------------------------------------------------
    best_seq = None
    best_M = float('inf')
    for s in seeds:
        M = _full_convolution(s).max()
        if M < best_M:
            best_M = M
            best_seq = s.copy()
    print(f'Initial best max‑conv from seeds = {best_M:.6f}')

    # -----------------------------------------------------------------
    # Main optimisation loop – run local improvement on the current best,
    # occasionally restart from a fresh random seed if time permits.
    # -----------------------------------------------------------------
    while time.time() < deadline:
        # Run a local optimiser starting from the current best
        improved_seq, improved_M = _improve_sequence(
            best_seq,
            target_sum,
            deadline,
        )
        if improved_M < best_M:
            best_M = improved_M
            best_seq = improved_seq.copy()
            print(f'Global best updated: max‑conv = {best_M:.6f}')

        # Early exit if we are already below the target bound
        if best_M <= 1.51:
            break

        # If time still remains, try a fresh random start
        remaining = deadline - time.time()
        if remaining > 1.0:
            rnd_start = np.random.rand(n).astype(float)
            rnd_start = _normalise(rnd_start, target_sum)
            best_seq, best_M = max(
                [(best_seq, best_M), (rnd_start, _full_convolution(rnd_start).max())],
                key=lambda x: -x[1]   # want the lower max‑conv → larger negative key
            )
        else:
            break

    print(f'Finished. Best max‑conv achieved = {best_M:.6f}')
    return best_seq.tolist()
