        if method == 'uniform':

        elif method == 'random':

        elif method == 'triangular':

def anneal(seq, max_iters):
    s = seq.sum()
    seq = seq / s
    cur_score = evaluate_sequence(seq.tolist())
    best_seq, best_score = seq.copy(), cur_score
    T0 = 1.0 * cur_score
    for i in range(max_iters):
        # choose two indices i, j
        i1, i2 = np.random.randint(len(seq), size=2)
        # propose moving small amount delta from i1 to i2
        delta = np.random.exponential(scale=0.01) * seq[i1]  # small fraction
        new_seq = seq.copy()
        new_seq[i1] = max(0, new_seq[i1] - delta)
        new_seq[i2] = new_seq[i2] + delta
        # renorm to sum 1
        new_seq = new_seq / new_seq.sum()
        new_score = evaluate_sequence(new_seq.tolist())
        # accept
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / (T0 * (1 - i / max_iters))):
            seq, cur_score = new_seq, new_score
            if new_score < best_score:
                best_seq, best_score = new_seq.copy(), new_score
        # optionally reheat early steps
    return best_seq, best_score
