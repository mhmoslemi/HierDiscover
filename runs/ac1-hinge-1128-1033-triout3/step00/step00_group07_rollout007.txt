def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # small buffer

    # Determine length:
    n0 = len(height_sequence_1) if 'height_sequence_1' in globals() else 500
    # maybe try different lengths: can try multiple lengths that are divisors of budget
    # We'll just use n0.

    target_sum = np.sqrt(2 * n0)
    # start from height_sequence_1 if available else random.
    if 'height_sequence_1' in globals() and np.random.rand() < 0.7:
        a = np.array(height_sequence_1, dtype=float)
        # Normalize sum
        a = a * (target_sum / np.sum(a))
    else:
        a = np.random.rand(n0)
        a = a * (target_sum / np.sum(a))

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score {best_score:.6f}")

    # Adam parameters
    lr = 0.2
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iter = 0
    beta_smooth = 5.0
    beta_inc = 0.05  # gradually increase
    max_iter = 1000000
    while time.time() < deadline and iter < max_iter:
        # compute convolution
        conv = np.convolve(a, a)
        # smooth max
        max_conv = np.max(conv)
        # compute w (softmax)
        # shift for numeric stability
        shift = conv - max_conv
        expb = np.exp(beta_smooth * shift)
        w = expb / np.sum(expb)

        # compute gradient
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (iter + 1))
        v_hat = v / (1 - beta2 ** (iter + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = np.maximum(a, 0.0)
        # renormalize sum
        s = np.sum(a)
        if s == 0:
            # reinitialize
            a = np.random.rand(n0)
            a = a * (target_sum / np.sum(a))
            m[:] = 0
            v[:] = 0
            continue
        a = a * (target_sum / s)
        # evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iter}, improved score {best_score:.6f}")
        # increase smoothing
        beta_smooth = min(50.0, beta_smooth + beta_inc)
        iter += 1

    # After gradient descent, optionally local random search:
    # do random perturbations for remaining time
    while time.time() < deadline:
        # random small perturb in neighbourhood of best_seq
        a = best_seq.copy()
        # pick random index(s)
        for _ in range(np.random.randint(1, 5)):
            idx = np.random.randint(n0)
            a[idx] += np.random.randn() * 0.1 * a[idx]  # relative perturb
        a = np.maximum(a, 0.0)
        a = a * (target_sum / np.sum(a))
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Random perturb improved to {best_score:.6f}")
        # else maybe accept occasional worse for diversification
    print(f"Final best score {best_score:.6f}")
    return best_seq.tolist()

def run_optim(a, deadline):
    n = len(a)
    target_sum = np.sqrt(2*n)
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # init Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr = 0.3
    beta_smooth = 5.0
    max_beta = 60.0
    beta_inc_per_iter = (max_beta - beta_smooth) / 2000.0
    iter = 0
    while time.time() < deadline - 0.1:
        # compute conv:
        conv = np.convolve(a, a)
        # compute smooth max weighting
        # subtract max for stability
        shift = conv - np.max(conv)
        expb = np.exp(beta_smooth * shift)
        w = expb / np.sum(expb)

        grad = 2.0 * np.correlate(w, a, mode='valid')
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (iter + 1))
        v_hat = v / (1 - beta2 ** (iter + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection and renorm
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s == 0:
            a = np.random.rand(len(a))
            a = a * (target_sum / np.sum(a))
            m.fill(0)
            v.fill(0)
        else:
            a = a * (target_sum / s)

        # evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # optional print
            print(f"Iter {iter}, score {best_score:.6f}")

        # update beta_smooth gradually
        beta_smooth = min(max_beta, beta_smooth + beta_inc_per_iter)

        # optional random jitter every few iterations
        if iter % 200 == 0 and iter > 0:
            # add small noise
            a = a + np.random.randn(len(a)) * 0.001 * a
            a = np.maximum(a, 0.0)
            a = a * (target_sum / np.sum(a))

        iter += 1
        # break condition for iteration count perhaps
        if iter >= 5000:
            break

    return best_seq, best_score
```

"""
Propose a good step‑function sequence by smooth‑max gradient descent.

The objective to minimise is
    2 * n * max(conv(a,a)) / (sum(a)^2)
which, after normalising   sum(a) = sqrt(2n), equals simply max(conv(a,a)).
We therefore normalise the vector to the target L1‑norm after each update
and minimise a smooth approximation of the maximum (log‑sum‑exp) using an
Adam optimiser.  The gradient of the smooth‑max is
    ∇_a  log∑_k exp(β·conv_k) = 2 * correlate(w, a)
where w_i = exp(β·conv_i)/∑_j exp(β·conv_j).  Starting from a random
vector (or the provided SOTA sequence) the algorithm quickly breaks
symmetry and drives the convolution towards a flatter profile, reducing the
maximum value.  Once the time budget is almost exhausted a small random
perturbation phase performs a final local search.

The function respects all security checks of `evaluate_sequence` by keeping
all entries non‑negative and well below the 1000‑value cap.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector so that its L1‑norm equals target_sum."""
    total = np.sum(vec)
    if total == 0.0:
        # avoid division by zero – re‑initialise uniformly
        vec = np.full_like(vec, target_sum / vec.size)
    else:
        vec = vec * (target_sum / total)
    return vec


def _smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_i = exp(beta * (conv_i - max(conv))) / Σ.
    The subtraction stabilises the exponentials.
    """
    shift = conv - np.max(conv)          # ≤ 0, max becomes 0
    exp_beta = np.exp(beta * shift)
    w = exp_beta / np.sum(exp_beta)
    return w


def _adam_step(a: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               grad: np.ndarray,
               it: int,
               lr: float = 0.25,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform a single Adam update and return new (a, m, v).
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    # bias‑corrected estimates
    m_hat = m / (1.0 - beta1 ** (it + 1))
    v_hat = v / (1.0 - beta2 ** (it + 1))

    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v


def _run_adam_opt(initial: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a single start vector with Adam + smooth‑max.

    Returns the best sequence seen during this run and its (approximate)
    max‑convolution value.
    """
    a = _normalize(initial.copy(), target_sum)

    n = a.size
    best_seq = a.copy()
    # because sum == target_sum, the score equals max(conv)
    best_score = np.max(np.convolve(a, a))

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    beta = 0.5          # start with a very smooth approximation
    beta_target = 30.0  # final sharpness
    beta_inc = (beta_target - beta) / 8000.0  # linear schedule (≈8000 iters)

    # main optimisation loop
    while time.time() < deadline - 0.05:
        # convolution and smooth‑max weights
        conv = np.convolve(a, a)                # length 2n‑1
        w = _smooth_max_weights(conv, beta)     # length 2n‑1, sums to 1

        # gradient of the smooth‑max (2 * correlate(w, a) )
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # Adam step
        a, m, v = _adam_step(a, m, v, grad, it)

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _normalize(a, target_sum)

        # current (exact) objective value
        cur_score = np.max(np.convolve(a, a))

        if cur_score < best_score:
            best_score = float(cur_score)
            best_seq = a.copy()
            # log only when an improvement happens
            print(f"[Iter {it:05d}]  new best max‑conv = {best_score:.6f}")

        # schedule update
        beta = min(beta_target, beta + beta_inc)
        it += 1

        # occasional jitter to escape plateaus (every 500 iters)
        if it % 500 == 0:
            jitter = np.random.randn(n) * 0.002 * a
            a += jitter
            a = np.maximum(a, 0.0)
            a = _normalize(a, target_sum)

        # hard stop if we have exhausted a reasonable number of iterations
        if it >= 20000:
            break

    return best_seq, best_score


def _random_perturb(seq: np.ndarray,
                    target_sum: float,
                    magnitude: float = 0.03) -> np.ndarray:
    """
    Apply a few small random adjustments while preserving the sum.
    `magnitude` controls the relative size of the perturbation.
    """
    a = seq.copy()
    n = a.size
    # choose a random number of indices to touch (1‑5)
    for _ in range(np.random.randint(1, 5)):
        idx = np.random.randint(n)
        delta = np.random.randn() * magnitude * (a[idx] + 1e-12)
        a[idx] = max(0.0, a[idx] + delta)
    # renormalise to the target L1‑norm
    a = _normalize(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑value step‑function sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function must return before this limit).

    Returns
    -------
    list[float]
        The best sequence found within the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1. Determine the length of the sequence.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        # keep a copy of the SOTA sequence for a possible start point
        sota_seq = np.array(height_sequence_1, dtype=float)
    else:
        # fallback – pick a moderate size that fits comfortably in the budget
        n = 1200
        sota_seq = None

    target_sum = np.sqrt(2.0 * n)   # normalisation used in the original paper

    # ------------------------------------------------------------------
    # 2. Perform a few independent optimisation runs (random restarts).
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_score = np.inf

    # we will try at most three distinct restarts (more if time permits)
    restarts = 3 if deadline - time.time() > 200 else 1
    for rs in range(restarts):
        if time.time() >= deadline:
            break

        # initialise: 70 % chance of starting from the known SOTA,
        # otherwise from a fresh random vector
        if sota_seq is not None and np.random.rand() < 0.7:
            init_vec = sota_seq.copy()
        else:
            init_vec = np.random.rand(n).astype(float)

        # run the Adam‑based optimiser
        seq, score = _run_adam_opt(init_vec, target_sum, deadline)

        if score < best_overall_score:
            best_overall_score = float(score)
            best_overall_seq = seq.copy()
            print(f"[Restart {rs}]  best score so far = {best_overall_score:.6f}")

        # update the SOTA reference for the next restart (helps transfer
        # knowledge between runs)
        sota_seq = seq.copy()

    # ------------------------------------------------------------------
    # 3. Final refinement phase – cheap random perturbations.
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # Very unlikely, but guard against an empty result.
        best_overall_seq = np.full(n, target_sum / n)

    while time.time() < deadline - 0.1:
        candidate = _random_perturb(best_overall_seq, target_sum,
                                   magnitude=0.02 + 0.03 * np.random.rand())
        cand_score = np.max(np.convolve(candidate, candidate))

        if cand_score < best_overall_score:
            best_overall_score = float(cand_score)
            best_overall_seq = candidate.copy()
            print(f"[Perturb]  improved max‑conv = {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence seen.
    # ------------------------------------------------------------------
    print(f"Final best max‑conv = {best_overall_score:.6f} (length={n})")
    return best_overall_seq.tolist()
