def simplex_projection(v):
   """Project vector v onto simplex {x >=0, sum x = 1}."""
   sorted_v = np.sort(v)[::-1]
   cssv = np.cumsum(sorted_v) - 1
   ind = np.arange(len(v)) + 1
   cond = sorted_v - cssv / ind > 0
   rho = ind[cond][-1]
   theta = cssv[cond][-1] / rho
   w = np.maximum(v - theta, 0)
   return w

def simplex_projection(v):
    """Project a vector onto the probability simplex (non-negative, sum=1)."""
    v = np.asarray(v, dtype=float)
    # Based on algorithm by Wang & Carreira-Perpinán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # All entries <=0, return uniform
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
Simple simulated‑annealing + projected gradient optimiser.

The evaluation function is scale invariant, therefore we keep the
sequence normalised to a probability vector (sum = 1).  The score then
reduces to 2·n·max(conv(a,a)).  The optimiser repeatedly proposes a tiny
mass transfer between two random positions (SA move) and, every few
iterations, takes a projected gradient step that explicitly tries to
decrease the currently maximal convolution entry.  All intermediate
vectors are projected onto the simplex (non‑negative, sum = 1).  The
best vector found before the time budget expires is returned.
"""

import time
import math
import numpy as np

def _simplex_projection(v):
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative -> return uniform distribution
        return np.full(n, 1.0 / n, dtype=float)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a, fft_len):
    """Return max(a * a) using an FFT of length `fft_len` (real RFFT)."""
    # a is a 1‑D array with sum = 1
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]
    return conv.max()


def _score(a, n, fft_len):
    """Score = 2·n·max(conv); sum(a)=1."""
    return 2.0 * n * _max_convolution(a, fft_len)


def _gradient_step(a, k_max, step):
    """
    Perform a single projected gradient step that tries to reduce the
    convolution entry at index `k_max`.  The step size `step` multiplies
    the direction.
    """
    n = a.shape[0]
    # gradient of conv[k_max] w.r.t a_i is 2 * a_{k_max-i} (if in range)
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros(n, dtype=float)
    grad[mask] = 2.0 * a[j[mask]]
    # move opposite to gradient, keep zero sum by removing the mean
    direction = -grad
    direction -= direction.mean()
    cand = a + step * direction
    cand = _simplex_projection(cand)
    return cand
