"""
Geometric‑decay construction for the step‑function problem.

The evaluation metric is
    score = 2 * n * max(conv(a,a)) / (sum(a)^2)

For a sequence a_i = r^i (i = 0 … n‑1) the centre of the convolution is
    (a * a)[n‑1] = n * r^{n‑1}

Choosing r = n^{‑1/(n‑1)} makes this centre exactly 1, which is also the
value at index 0 (a_0^2).  Hence max(conv) = 1, independent of n.

The sum of the geometric series is
    S = (1 - r^n) / (1 - r)  ≈  n / log(n)
so the score becomes roughly 2·(log n)^2 / n, which for n≈10 000 is
≈ 0.017 – far below the target 1.503.

The function below builds such a sequence (length can be tuned) and returns
it.  It performs only a single evaluation, well inside the 1000 s budget.
"""

import math
import time
import numpy as np


def geometric_sequence(length: int) -> list[float]:
    """
    Return a list a_i = r**i, i = 0 … length‑1,
    where r = length^{‑1/(length‑1)} so that the centre of the
    self‑convolution equals a_0^2 = 1.
    """
    # r = n^{-1/(n-1)}
    r = math.pow(length, -1.0 / (length - 1))
    # Build the sequence using the exact power (float) for each index.
    seq = [float(r ** i) for i in range(length)]
    return seq


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Generate a geometric‑decay sequence that yields a very low evaluation
    score.  The routine respects the time budget, but the computation is
    inexpensive – a single call to ``evaluate_sequence`` is made for logging.
    """
    start_time = time.time()

    # Choose a length that balances speed and an extremely low score.
    # n = 10 000 gives a score ≈ 0.017 and the convolution finishes quickly.
    n = 10000

    # Build the candidate sequence.
    candidate = geometric_sequence(n)

    # Optional: report the score for monitoring.
    try:
        score = evaluate_sequence(candidate)
        print(f"[propose_candidate] length={n}, score={score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[propose_candidate] evaluation error:", exc)
        score = None

    # Ensure we stop well before the deadline.
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1.0:
        # Fallback to a trivial short uniform sequence (guaranteed valid).
        print("[propose_candidate] time limit imminent – returning fallback.")
        return [1.0] * 10

    return candidate
