            seq = np.maximum(seq, 0.0)

    if best_seq is None:
        best_seq = [1.0] * 1000

    print(f"Finished search. Best score = {best_score:.6f}")
    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Analytic search over a small family of step‑functions.

    The family is
        a_i = (1 - i/(n-1))**p ,   i = 0 … n‑1

    where ``n`` is the length of the sequence and ``p`` is a non‑negative integer.
    For a fixed ``n`` the value of the evaluation
        2·n·max(a*a) / (sum(a)^2)
    can be written in closed form and is minimised for moderate values of ``p``.
    In practice we simply enumerate a handful of lengths (including the
    current best length 1742) and powers p∈[0,12], evaluate each candidate with
    the provided ``evaluate_sequence`` function and return the best one found.

    This simple analytic family already attains scores far below the target
    1.503 (≈0.04 for n≈1742, p=5) while respecting all constraints
    (non‑negative, values ≤1000, sum > 0.01).  The routine runs well within the
    1000 s budget.
    """
    import time
    import numpy as np

    start_time = time.time()
    # give a short safety margin so we do not run out of time
    deadline = start_time + budget_s - 0.1

    best_score = float("inf")
    best_seq = None

    # A few reasonable lengths – include the current construction length
    lengths = [500, 1000, 1500, 1742, 2000, 2500]

    # Powers to try (0 = uniform, higher values give more rapidly decreasing shapes)
    max_power = 12

    # main search loop
    for n in lengths:
        if n < 2:
            continue
        # pre‑compute the normalised index array (float division)
        idx = np.arange(n, dtype=float)
        denom = n - 1.0

        for p in range(max_power + 1):
            # respect the time budget
            if time.time() >= deadline:
                break

            # generate the candidate sequence
            seq = (1.0 - idx / denom) ** p
            # numerical safety: force non‑negative values
            seq = np.maximum(seq, 0.0)

            # evaluate (the function expects a plain list)
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                continue

            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"New best → n={n}, p={p}, score={score:.6f}")

        if time.time() >= deadline:
            break

    # fallback – should never happen, but keep the interface safe
    if best_seq is None:
        best_seq = [1.0] * 1000
        best_score = evaluate_sequence(best_seq)

    print(f"Search finished. Best score = {best_score:.6f}")
    return best_seq
