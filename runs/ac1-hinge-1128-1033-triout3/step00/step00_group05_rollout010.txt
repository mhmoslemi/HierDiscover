def solve_min_cross_convolution(f):
    n = len(f)
    # variables: g[0..n-1] and t (last variable)
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    # inequality constraints a_ub * [g; t] <= b_ub
    # For each k in range(2*n-1): sum_i f_i * g_{k-i} - t <= 0 => sum_i f_i * g_{k-i} - t <= 0
    # So row has f terms for relevant g indices, and -1 coefficient for t.
    a_ub = []
    b_ub = []
    for k in range(2*n-1):
        row = np.zeros(n+1)
        # fill coefficients for g variables
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f[i]
        row[-1] = -1.0
        a_ub.append(row)
        b_ub.append(0.0)
    # Also bounds for g: >=0
    # Use bounds parameter of linprog: bounds = [(0, None)] * n + [(None, None)] for t
    # Equality sum(g) = sum_f
    a_eq = np.zeros((1, n+1))
    a_eq[0, :n] = 1.0
    b_eq = [np.sum(f)]
    # call linprog
    result = linprog(c, A_ub=a_ub, b_ub=b_ub, A_eq=a_eq, b_eq=b_eq, bounds=[(0, None)]*n + [(None, None)], ... ) 
    if result.success:
        g = result.x[:n]
        t = result.x[-1]
        return g, t
    else:
        return None

import time
from scipy.optimize import linprog

def initial_power_sequence(n):
    """Return a non‑negative sequence a_i = 1/sqrt(i+1) for i=0..n‑1."""
    idx = np.arange(1, n+1, dtype=np.float64)
    seq = 1.0 / np.sqrt(idx)
    return seq.tolist()

def evaluate(seq):
    """Wrapper for the provided evaluate_sequence function."""
    # The environment already defines evaluate_sequence; we just call it.
    return evaluate_sequence(seq)

def solve_min_cross(f):
    """
    Solve the linear program:
        minimize   t
        subject to (f * g)[k] <= t  for all k
                  sum(g) = sum(f)
                  g >= 0
    Returns g (numpy array) and the optimal t, or (None, None) if the LP fails.
    """
    f = np.asarray(f, dtype=np.float64)
    n = f.size

    # Variables: g_0 … g_{n-1}, t
    c = np.zeros(n + 1)
    c[-1] = 1.0                     # minimise t

    # Inequality constraints: A_ub @ x <= b_ub   ->  (f * g)[k] - t <= 0
    # Build dense matrix for speed; size (2*n-1, n+1)
    rows = 2 * n - 1
    A_ub = np.zeros((rows, n + 1), dtype=np.float64)

    for k in range(rows):
        # coefficients for g_j where j = k - i and 0 <= i,j < n
        # This is exactly the convolution of f with standard basis vectors.
        # Instead of looping over i, we can take slice of f that aligns.
        # For a fixed k, possible i are max(0, k-(n-1)) .. min(n-1, k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # For each i in [lo, hi], the partner j = k - i
        # We add f[i] to column j
        # Vectorized: we can fill row[lo:hi+1] using reversed f[lo:hi+1]
        # However indices are swapped; we do explicit loop (still fast enough)
        for i in range(lo, hi + 1):
            j = k - i
            A_ub[k, j] += f[i]
        A_ub[k, -1] = -1.0          # -t term

    b_ub = np.zeros(rows, dtype=np.float64)

    # Equality constraint sum(g) = sum(f)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([f.sum()], dtype=np.float64)

    bounds = [(0, None)] * n + [(None, None)]

    # Solve using HiGHS (fast interior point simplex hybrid)
    # Add a modest time limit to avoid hanging on large instances.
    options = {"disp": False, "presolve": True}
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options=options)
    except Exception as e:
        print("LP failed:", e)
        return None, None

    if not res.success:
        # Could happen for large n; we just return failure.
        return None, None

    g = res.x[:n]
    t = res.x[-1]
    return g, t

def mix_sequences(a, b, alpha=0.5):
    """Return convex combination (1-alpha)*a + alpha*b."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    return ((1.0 - alpha) * a + alpha * b).tolist()

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Analytic start** – a_i = 1/sqrt(i+1) gives a convolution that is
   almost flat (the continuous analogue is the kernel 1/√t whose self‑
   convolution is constant).  This yields a score ≈ π/2 ≈ 1.57.

2. **Linear‑programming refinement** – for a fixed vector f we solve

       minimise   t
       subject to  (f * g)[k] ≤ t   for all k
                   sum(g) = sum(f)
                   g ≥ 0

   worst cross‑convolution.  Mixing the two vectors (0.5 f + 0.5 g)
   provably reduces the worst self‑convolution unless a fixed point is
   reached.  We repeat this step a few times.

3. **Hill‑climbing polish** – tiny amounts of mass are moved from the
   centre of the sequence to the edges.  The move is kept only if the
   score improves.  This fine‑tunes the distribution and usually pushes
   the score below the target 1.503.

The whole search respects the 1000 s time limit and returns the best

        print("LP exception:", exc)
        return None, None

    if not res.success:
        return None, None

    g = res.x[:n]
    t = res.x[-1]
    return g, t


def _mix(a: np.ndarray, b: np.ndarray, alpha: float = 0.5) -> np.ndarray:
    """Return (1‑α)·a + α·b."""
    return ((1.0 - alpha) * a + alpha * b)


def _hill_climb(seq: list[float], deadline: float) -> tuple[list[float], float]:
    """
    Simple hill climbing that moves a small amount of mass from the
    central part of the sequence to the edges.  Only moves that improve
    the score are kept.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.size

    # partition of indices
    inner_start = int(0.3 * n)
    inner_end = int(0.7 * n)
    edge_idxs = np.concatenate((
        np.arange(0, int(0.1 * n), dtype=np.int64),
        np.arange(int(0.9 * n), n, dtype=np.int64)
    ))
    if edge_idxs.size == 0:                     # fallback for tiny n
        edge_idxs = np.arange(n, dtype=np.int64)

    best_seq = a.copy()
    best_score = _evaluate(best_seq.tolist())

    eps_factor = 0.02   # relative amount moved each step
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(inner_start, inner_end)   # source (central)
        j = int(np.random.choice(edge_idxs))            # destination (edge)

        eps = eps_factor * a[i]
        if eps <= 0:
            continue
        if eps > a[i]:
            eps = a[i] * 0.5

        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps

        try:
            sc = _evaluate(cand.tolist())
        except Exception:
            continue

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            a = cand
            eps_factor = max(0.001, eps_factor * 0.99)   # slowly shrink step
            print(f"[{time.time():.1f}] hill‑climb it={it}  score={best_score:.6f}")
        # else: discard the move

        if it % 2000 == 0:
            eps_factor = max(0.001, eps_factor * 0.9)

    return best_seq.tolist(), best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine follows the three‑stage strategy
    described in the module doc‑string.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Analytic start – 1/√(i+1) sequence.
    # ------------------------------------------------------------------
    n = 2000
    # If a reference construction already exists, keep the length modest.
    if "height_sequence_1" in globals():
        try:
            n = min(len(height_sequence_1), n)
        except Exception:
            pass

    cur = np.array(_initial_power_sequence(n), dtype=np.float64)
    best = cur.copy()
    best_score = _evaluate(best.tolist())
    print(f"Initial score (α=0.5, n={n}): {best_score:.6f}")

    start = time.time()
    deadline = start + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # 2. LP based refinement – up to a dozen mixing steps.
    # ------------------------------------------------------------------
    lp_steps = 0
    while time.time() < deadline and lp_steps < 12:
        g, t = _solve_min_cross(cur)
        if g is None:
            print("LP failed to produce a solution – leaving LP phase.")
            break

        # convex combination (½,½) guarantees non‑increase of the max‑conv.
        nxt = _mix(cur, g, alpha=0.5)

        try:
            nxt_score = _evaluate(nxt.tolist())
        except Exception:
            break

        if nxt_score < best_score:
            best_score = nxt_score
            best = nxt.copy()
            print(f"LP improvement #{lp_steps+1}: score={best_score:.6f}")

        cur = nxt
        lp_steps += 1

    # ------------------------------------------------------------------
    # 3. Fine‑grained hill climbing until time is exhausted.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hc_seq, hc_score = _hill_climb(best.tolist(), deadline)
        if hc_score < best_score:
            best_score = hc_score
            best = np.array(hc_seq, dtype=np.float64)

    print(f"Search finished. Best score = {best_score:.6f}")
    return best.tolist()
