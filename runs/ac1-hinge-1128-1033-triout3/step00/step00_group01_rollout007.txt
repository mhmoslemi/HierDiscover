def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)

    start = time.time()
    end = start + budget_s - 0.5

    # Choose lengths to try
    lengths = [500, 1000, 2000]
    best_seq = None
    best_val = float('inf')
    for n in lengths:
        # initialize with edge-heavy shape to try flatten convolution
        idx = np.arange(n)
        a = (np.maximum(idx, n-1-idx) + 1.0).astype(float)  # edge heavy
        a = a / a.sum()  # normalize sum 1
        
        # optionally random perturbation start
        if np.random.rand() < 0.3:
            a = np.random.rand(n)
            a /= a.sum()
        
        # run iterative improvement
        cur = a.copy()
        cur_val = evaluate_sequence(cur.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur.tolist()
            print(f'Init length {n} val {cur_val}')
        # iteration parameters
        t_max = 0.2
        while time.time() < end:
            # compute current max convolution
            conv_cur = np.convolve(cur, cur)
            M = conv_cur.max()
            # solve LP direction
            g = solve_convolution_lp(cur.tolist(), M)
            if g is None:
                # fallback: random small perturb
                i = np.random.randint(n)
                cur[i] = max(0.0, cur[i] + np.random.randn() * 0.01)
                # renormalize
                cur = cur / cur.sum()
                continue
            
            g = np.array(g)
            # renorm g to sum 1
            g = g / g.sum()
            # try a few t values
            best_local_t = 0.0
            best_local_val = cur_val
            for t in np.linspace(0.01, t_max, 5):
                cand = (1 - t) * cur + t * g
                cand = np.clip(cand, 0, None)
                cand = cand / cand.sum()
                val = evaluate_sequence(cand.tolist())
                if val < best_local_val:
                    best_local_val = val
                    best_local_t = t
            if best_local_t > 0:
                cur = (1 - best_local_t) * cur + best_local_t * g
                cur = np.clip(cur, 0, None)
                cur = cur / cur.sum()
                cur_val = best_local_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur.tolist()
                    print(f'Improved length {n} val {best_val} t {best_local_t}')
            else:
                # no improvement, perform random perturb to escape
                idx = np.random.randint(n)
                cur[idx] = max(0.0, cur[idx] + np.random.randn() * 0.01)
                cur = cur / cur.sum()
        # end while for this length
    # after loops, return best_seq
    print(f'Best final score: {best_val}')
    return best_seq

def gradient_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    # indices i from lo to hi inclusive
    i = np.arange(lo, hi + 1)
    j = k - i
    g[i] = 2.0 * a[j]
    return g

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """Generate a step function sequence that minimizes the evaluation function.
    
    The algorithm uses a simple projected gradient descent on the maximum 
    value of the autocorrelation (convolution with itself).  Starting from an 
    edge‑heavy distribution the method repeatedly moves mass from indices 
    that contribute most to the current maximal convolution value to indices 
    that do not contribute, while keeping the sequence non‑negative and 
    normalised (sum = 1).  The step size decays over time and random 
    perturbations are added occasionally to escape plateaus.
    """
    import numpy as np
    import time
    # Ensure reproducibility
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # leave margin

    # Choose sequence length
    n = 2000  # moderate size, balances quality and speed

    # Initial edge‑heavy shape: larger values at the ends, smaller in the centre
    idx = np.arange(n)
    a = np.maximum(idx, n - 1 - idx) + 1.0   # values grow towards the ends
    a = a.astype(float)
    a = a / a.sum()                           # normalise to unit sum

    # Occasionally replace the start with a random Dirichlet to diversify searches
    if np.random.rand() < 0.2:
        a = np.random.rand(n)
        a = a / a.sum()

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score (n={n}): {best_score:.6f}")

    # Hyper‑parameters
    lr = 0.8                 # initial step size (will be scaled)
    lr_decay = 0.99995
    min_lr = 1e-6
    iter_count = 0
    perturb_interval = 150   # iterations between random perturbations

    while time.time() < deadline:
        iter_count += 1

        # Compute convolution and its maximum
        b = np.convolve(a, a)
        max_b = b.max()
        # Find all shifts attaining (or very close to) the maximum
        # Using a tolerance prevents missing ties due to floating point noise
        tol = 1e-12
        ks = np.where(b >= max_b - tol)[0]

        # Build gradient of the (selected) maximal convolution value

"""
Propose a sequence of non‑negative heights that (approximately) minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works on a moderate size (n = 2000) and uses a projected
gradient descent on the maximal entry of the autocorrelation.
Starting from an “edge‑heavy’’ distribution we repeatedly move mass away
from indices that contribute most to the current maximal convolution
value.  After each step the vector is clipped to non‑negative values and
renormalised – the objective is scale‑invariant, so fixing the sum to 1
removes the trivial scaling degree of freedom.  A small amount of random
noise is injected from time to time to escape plateaus.  The best
sequence found before the time budget expires is returned.
"""

import numpy as np
import time
from typing import List

def _grad_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the convolution entry b_k = sum_i a_i a_{k-i}
    with respect to the vector a (using the convention of np.convolve).
    For a given shift k the derivative is
        ∂b_k/∂a_i = 2 * a_{k-i}   if 0 ≤ k-i < n, else 0
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights (the step function) that attempts
    to minimise the provided evaluate_sequence() function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop before this time).

    Returns
    -------
    List[float]
        The best sequence found within the time budget.
    """
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # a little safety margin

    # Sequence length – a trade‑off between theoretical bound (better for large n)
    # and speed of the inner loop.  n = 2000 works comfortably inside the budget.
    n = 2000

    # ------------------------------------------------------------------
    # Initial distribution – larger values near the edges, smaller in the centre.
    # This shape already reduces the peak of the autocorrelation compared to a flat
    # sequence and moves us away from the uniform stationary point.
    # ------------------------------------------------------------------
    idx = np.arange(n)
    a = np.maximum(idx, n - 1 - idx).astype(float) + 1.0   # edge‑heavy
    a = a / a.sum()                                        # normalise to sum = 1

    # Occasionally start from a random Dirichlet to diversify the search.
    if np.random.rand() < 0.2:
        a = np.random.rand(n)
        a = a / a.sum()

    cur = a.copy()
    cur_score = evaluate_sequence(cur.tolist())
    best_seq = cur.copy()
    best_score = cur_score
    print(f"[Init] n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the projected gradient descent
    # ------------------------------------------------------------------
    lr = 0.8                     # initial step size
    lr_decay = 0.99995          # slow multiplicative decay
    min_lr = 1e-7
    max_backtrack = 5           # how many step‑size reductions we try
    it = 0
    perturb_every = 150         # inject random noise periodically

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) Compute the current autocorrelation and locate the maximal shift(s)
        # --------------------------------------------------------------
        b = np.convolve(cur, cur)            # length 2n‑1
        max_b = b.max()
        # Numerical tolerance to catch ties caused by floating‑point noise
        ks = np.where(b >= max_b - 1e-12)[0]

        # --------------------------------------------------------------
        # 2) Assemble the (sub‑)gradient of the maximal entry
        # --------------------------------------------------------------
        grad = np.zeros_like(cur)
        for k in ks:
            grad += _grad_for_shift(cur, k)
        grad /= max(1, ks.size)   # average if several shifts share the max

        # --------------------------------------------------------------
        # 3) Try a forward step (negative gradient) with back‑tracking line search
        # --------------------------------------------------------------
        step_accepted = False
        trial_lr = lr
        for _ in range(max_backtrack):
            cand = cur - trial_lr * grad
            cand = np.clip(cand, 0.0, None)          # keep non‑negative
            s = cand.sum()
            if s == 0.0:                             # avoid division by zero
                cand = np.full_like(cand, 1.0 / n)
            else:
                cand = cand / s                      # renormalise to sum = 1

            cand_score = evaluate_sequence(cand.tolist())
            # Accept if we have not increased the objective
            if cand_score <= cur_score:
                cur = cand
                cur_score = cand_score
                step_accepted = True
                # Record the best ever seen
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur.copy()
                    print(f"[Iter {it:5d}] New best: {best_score:.6f}  lr={trial_lr:.3e}")
                break
            # otherwise shrink step size and retry
            trial_lr *= 0.5

        # If none of the tried step sizes helped we simply decay the learning rate
        if not step_accepted:
            lr = max(min_lr, lr * 0.9)

        # Decay the learning rate smoothly
        lr = max(min_lr, lr * lr_decay)

        # --------------------------------------------------------------
        # 4) Occasional random perturbation to escape possible plateaus
        # --------------------------------------------------------------
        if it % perturb_every == 0:
            noise = np.random.randn(n) * 0.01 * cur
            cur = cur + noise
            cur = np.clip(cur, 0.0, None)
            s = cur.sum()
            cur = cur / s if s > 0 else np.full_like(cur, 1.0 / n)
            cur_score = evaluate_sequence(cur.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"[Iter {it:5d}] Perturb improved best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered before the timeout.
    # ------------------------------------------------------------------
    print(f"[Done] Best score found: {best_score:.6f}")
    return best_seq.tolist()
