"""
Propose a candidate sequence that minimizes the evaluation function.

Algorithm
---------
We generate families of exponentially decaying sequences
    a_i = exp(-λ·i)   (i = 0 … n‑1)

The score of such a sequence can be estimated analytically:
    sum_a = (1‑exp(-λ·n)) / (1‑exp(-λ))
    max_b = n·exp(-λ·(n‑1))
    score  = 2·n·max_b / sum_a²

For large n the product x = λ·n determines the score:
    score(x) = 2·x²·e^{-x} / (1‑e^{-x})²,
and the optimum lies near x ≈ 2.5 (score ≈ 1.2).  By randomising
n and x we quickly find promising candidates.  Each candidate is first
filtered by the cheap analytic estimate; only if it looks better than the
current best we evaluate the true score with ``evaluate_sequence``.
A simple local refinement around the best parameters is performed before
returning the best sequence found within the time budget.

The search runs for the allotted time (default 1000 s) and never exceeds
the limits on value magnitude or negativity, because the generated
sequences consist of positive numbers ≤ 1.

"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def geometric_sequence(n: int, lam: float) -> list[float]:
    """Return a length‑n non‑negative sequence a_i = exp(-lam·i)."""
    if lam <= 0.0:
        # Uniform sequence (lam = 0) – the score is exactly 2.
        return [1.0] * n
    # Use numpy for speed, then convert to plain Python list.
    return np.exp(-lam * np.arange(n)).tolist()


def analytic_score(n: int, lam: float) -> float:
    """
    Cheap analytic approximation of the evaluation score for an exponential
    sequence of length n with decay λ = lam.
    """
    if lam <= 0.0:
        return 2.0
    # Avoid loss of precision when lam is tiny: use series expansion.
    if lam < 1e-12:
        return 2.0
    exp_n = math.exp(-lam * n)
    one_minus_exp = 1.0 - math.exp(-lam)
    # sum_a = (1 - e^{-λn}) / (1 - e^{-λ})
    sum_a = (1.0 - exp_n) / one_minus_exp
    # max_b = n·e^{-λ·(n‑1)}
    max_b = n * math.exp(-lam * (n - 1))
    return 2.0 * n * max_b / (sum_a * sum_a)


def local_refine(n: int, lam: float, best_score: float,
                 budget: float, start: float) -> tuple[list[float], float, float]:
    """
    Small hill‑climbing around (n, lam).  Returns (seq, lam, score)
    if an improvement is found within the remaining time budget.
    """
    # Search a tiny neighbourhood of λ (±20 %) and of n (±10 %).
    # Steps are coarse – we only aim for a quick polish.
    for factor in (0.9, 0.95, 1.0, 1.05, 1.1):
        lam2 = lam * factor
        if lam2 <= 0:
            continue
        # Keep n fixed for the inner loop; try a few nearby lengths.
        for dn in (-2, -1, 0, 1, 2):
            n2 = max(2, n + dn)      # length must be at least 2
            # Quick analytic filter
            est = analytic_score(n2, lam2)
            if est >= best_score:
                continue
            # Respect the remaining time budget.
            if time.time() - start > budget:
                return None, None, None
            seq = geometric_sequence(n2, lam2)
            try:
                scr = evaluate_sequence(seq)
                if scr < best_score:
                    return seq, lam2, scr
            except Exception:
                # Defensive – any problem with the evaluator is ignored.
                continue
    return None, None, None


# ----------------------------------------------------------------------
# Main entry point required by the challenge framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑score step‑function sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # keep a small safety margin

    # ------------------------------------------------------------------
    # Initialise with the provided SOTA sequence if it exists.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    best_n = None
    best_lam = None

    # height_sequence_1 may be present in the global namespace.
    try:
        cand = list(height_sequence_1)          # type: ignore[name-defined]
        sc = evaluate_sequence(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            best_n = len(cand)
            # estimate λ from first two entries (if they differ)
            if len(cand) >= 2 and cand[0] > 0 and cand[1] > 0:
                best_lam = -math.log(cand[1] / cand[0])
    except Exception:
        # No SOTA sequence or it is invalid – ignore.
        pass

    # ------------------------------------------------------------------
    # Main stochastic search: generate exponential families, keep the
    # best one (filtered by cheap analytic estimate).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Random length: between 200 and 8000 (larger lengths give slightly
        # better asymptotic behaviour but are slower to evaluate).
        n = random.randint(200, 8000)

        # Choose a target product x = λ·n.
        # Theory suggests x ∈ [1.5, 4.0] is promising.
        x = random.uniform(1.5, 4.0)
        lam = x / n

        # Quick analytic bound – prune unpromising candidates.
        est = analytic_score(n, lam)
        if est >= best_score:
            # Occasionally still evaluate a few bad candidates to avoid
            # being trapped by a misleading analytic estimate.
            if random.random() > 0.001:
                continue

        seq = geometric_sequence(n, lam)

        # Evaluate the true objective.
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue

        if score < best_score:
            best_score = score
            best_seq = seq
            best_n = n
            best_lam = lam
            print(f"[{time.time() - start_time:.1f}s] New best: {best_score:.6f} "
                  f"n={n} λ={lam:.6g} (x≈{x:.3f})")

        # Occasionally perform a tiny local refinement around the current best.
        if best_seq is not None and random.random() < 0.05:
            rem_budget = deadline - time.time()
            if rem_budget > 0.1:
                refined_seq, refined_lam, refined_score = local_refine(
                    best_n, best_lam, best_score, deadline, start_time)
                if refined_seq is not None and refined_score < best_score:
                    best_seq = refined_seq
                    best_score = refined_score
                    best_n = len(refined_seq)
                    best_lam = refined_lam
                    print(f"[{time.time() - start_time:.1f}s] Refined best: "
                          f"{best_score:.6f} n={best_n} λ={best_lam:.6g}")

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (should not happen) return a simple uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 100
    return best_seq
