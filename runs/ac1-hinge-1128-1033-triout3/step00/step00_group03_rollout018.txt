"""
propose_candidate
-----------------
Searches for a step‑function (non‑negative list of heights) that yields a low
value of ``evaluate_sequence``.  The method works directly on the probability
vector ``p = a / sum(a)`` (the objective is scale invariant).  It performs a
projected gradient descent on a smooth approximation of the
``max(conv(p,p))`` term:

    smooth_max_β(b) = (1/β)·log Σ_k exp(β·b_k)

where ``b = conv(p,p)``.  The gradient of the smooth max can be written as a
cross‑correlation and is obtained efficiently via a single convolution.
After each gradient step the vector is projected onto the probability simplex
(keeps all entries ≥0 and the total sum fixed).  The smoothing parameter β is
gradually increased, turning the smooth objective into the true max.
Random perturbations and occasional restarts help escape plateaus.
The best sequence found before the time budget expires is returned.

The algorithm needs no external LP solver and runs with only NumPy.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the simplex {x >= 0, Σ x = target_sum}."""
    # Standard O(n log n) simplex projection
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / (np.arange(len(u)) + 1) > 0)[0]
    if len(rho) == 0:
        # all entries will be zero
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    current best bound.  The function uses a projected gradient descent on a
    smooth approximation of the max auto‑convolution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.05   # keep a tiny margin

    # ------------------------------------------------------------------
    # 1. Initialise the sequence.
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=float)  # existing construction
    except NameError:

        n = np.random.randint(3000, 6000)
        init_seq = np.random.random(n)

    n = len(init_seq)
    # make sure we start with a legal (non‑negative) vector
    init_seq = np.maximum(0.0, init_seq)

    # The evaluation is scale‑invariant; we fix the L1‑norm to sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    if init_seq.sum() == 0:
        init_seq = np.full(n, 1.0)
    init_seq = init_seq * (target_sum / init_seq.sum())

    seq = init_seq.copy()
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # 2. Main optimisation loop.
    # ------------------------------------------------------------------
    beta = 1.0          # smoothing parameter for the soft‑max
    beta_max = 30.0
    step_factor = 0.5   # controls absolute step size (relative to gradient norm)
    no_improve = 0

    while time.time() < deadline:
        # convolution of the current sequence with itself
        conv = np.convolve(seq, seq)                # length = 2n‑1
        max_conv = conv.max()

        # ----- smooth approximation of the max (soft‑max) -----
        # weights w_k = exp(beta*(conv_k - max_conv)) are in (0,1]
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:        # numerical safety net
            break
        w = w / w_sum           # now Σ w_k = 1

        # ----- gradient of the smooth max w.r.t. seq -----
        # grad_i = 2 * Σ_j seq_j * w_{i+j}   (cross‑correlation)
        w_rev = w[::-1]                         # reverse for cross‑correlation
        full = np.convolve(seq, w_rev)          # length = n + (2n‑1) - 1 = 3n‑2
        # extract the needed slice: indices (2n‑2 - i) for i = 0 … n‑1
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * full[idx]

        # ----- take a step in the descent direction -----
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            # pure random shake when the gradient vanishes
            i, j = np.random.choice(n, 2, replace=False)
            delta = seq[i] * 0.01
            seq[i] -= delta
            seq[j] += delta
            seq = np.maximum(seq, 0.0)
            seq = _project_simplex(seq, target_sum)
            continue

        lr = step_factor / grad_norm                # adaptive step size
        cand = seq - lr * grad
        cand = np.maximum(cand, 0.0)                # enforce non‑negativity
        cand = _project_simplex(cand, target_sum)   # keep Σ = target_sum

        # ----- evaluate candidate -----
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            # any numerical or validation issue – skip this candidate
            continue

        # keep the globally best one
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # decide whether to move to the new candidate (simple greedy)
        if cand_score <= best_score:
            seq = cand
        else:
            # occasional exploration
            if np.random.rand() < 0.01:

        # slowly increase the smoothness parameter (approach true max)
        beta = min(beta * 1.05, beta_max)

        # occasional restart if stuck for a while
        if no_improve > 300:
            seq = np.random.rand(n)
            seq = seq * (target_sum / seq.sum())
            no_improve = 0

    # ------------------------------------------------------------------
    # 3. Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"Best score obtained: {best_score:.6f} (length = {len(best_seq)})")
    return best_seq.tolist()
