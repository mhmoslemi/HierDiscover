"""
Propose a candidate step‑function minimizing the evaluation metric.

The algorithm starts from a good analytic shape (a_i ∝ 1/√i) which is known
to give a value around π/2 ≈ 1.57.  A simple simulated‑annealing loop
then repeatedly perturbs the sequence (either by multiplicative noise on a
single entry or by swapping two entries) and keeps the perturbation if it
improves the objective or is accepted by a temperature schedule.  After each
perturbation the sequence is re‑normalised so that ∑a_i = √(2·n); this keeps
the objective invariant to scaling and makes the search focus on the shape
only.  The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation that matches `evaluate_sequence`.
# ----------------------------------------------------------------------
def _loss(seq: np.ndarray, n: int) -> float:
    """Return the same value that evaluate_sequence would give for `seq`."""
    # seq is already non‑negative and bounded, no extra checks needed.
    conv = np.convolve(seq, seq)          # length 2*n-1
    max_b = conv.max()
    s = seq.sum()
    # guard against tiny sums – the caller never lets this happen.
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The function runs for up to ``budget_s`` seconds and
    returns the best sequence found.
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000
    target_sum = np.sqrt(2.0 * n)           # the normalization used in the paper

    # ------------------------------------------------------------------
    # Start from the analytic 1/√i shape (good baseline, ≈1.57).
    # ------------------------------------------------------------------
    base = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    seq = base / base.sum() * target_sum      # enforce the required sum
    best_seq = seq.copy()
    best_score = _loss(best_seq, n)

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    temperature = 0.12            # initial temperature
    it = 0
    last_improve = 0
    deadline = time.time() + budget_s - 2.0   # leave a tiny margin

    while time.time() < deadline:
        # ---------- propose a new candidate --------------------------------
        if np.random.rand() < 0.7:
            # multiplicative perturbation of a single coordinate
            i = np.random.randint(n)
            factor = np.exp(np.random.normal(scale=0.05))   # log‑normal step
            cand = seq.copy()
            cand[i] = max(0.0, min(1000.0, cand[i] * factor))
        else:
            # swap two random positions (keeps the multiset unchanged)
            i, j = np.random.choice(n, size=2, replace=False)
            cand = seq.copy()
            cand[i], cand[j] = seq[j], seq[i]

        # ---------- re‑normalise the sum (keeps the objective scale‑invariant)
        total = cand.sum()
        if total == 0.0:
            continue
        cand *= (target_sum / total)
        # enforce hard bounds required by evaluate_sequence
        np.clip(cand, 0.0, 1000.0, out=cand)

        # ---------- evaluate ------------------------------------------------
        cur_score = _loss(cand, n)

        # ---------- acceptance test -----------------------------------------
        accept = False
        if cur_score < best_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp((best_score - cur_score) / temperature)
            if np.random.rand() < prob:
                accept = True

        if accept:
            seq = cand
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cand.copy()
                last_improve = it
                print(f"Iter {it:6d}  new best score = {best_score:.6f}")

        # ---------- temperature schedule ------------------------------------
        it += 1
        if it % 1500 == 0:
            temperature *= 0.9         # geometric cooling

        # occasional random restart if we stagnate too long
        if it - last_improve > 20000:
            # re‑initialise from the analytic shape (helps to escape dead ends)
            seq = base / base.sum() * target_sum
            last_improve = it

    # ------------------------------------------------------------------
    # Final check with the official evaluator (just for reporting)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Finished – best reported score = {final_score:.6f}")
    return best_seq.tolist()
