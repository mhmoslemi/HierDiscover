def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Construct a simple linearly increasing step function.
    
    The heights are 1, 2, …, 1000 (floats).  
    For a sequence a_i = i+1 of length n the evaluation metric simplifies to
    
        score = 2 * n * max(a*a) / (sum(a)**2)
              = 4 * (n + 2) / (3 * (n + 1))

    which tends to 4/3 ≈ 1.333 as n grows.  With the maximal allowed length
    n = 1000 we obtain a score of 4008 / 3003 ≈ 1.335, comfortably below the
    target of 1.503.  All values are ≤ 1000, so no clipping occurs and the
    sequence satisfies every security check in ``evaluate_sequence``.
    
    No iterative optimisation is required – the construction is instantaneous,
    well within the 1000 s budget.
    """
    # maximal length allowed without violating the 0‑1000 clipping rule
    n = 1000

    # generate the linear sequence 1, 2, …, n as floats
    seq = [float(i) for i in range(1, n + 1)]

    # optional sanity check (the platform already imports ``evaluate_sequence``)
    try:
        score = evaluate_sequence(seq)
        print(f"[propose_candidate] Linear sequence of length {n}, score = {score:.6f}")
    except Exception as exc:
        print("[propose_candidate] evaluation error:", exc)

    return seq
