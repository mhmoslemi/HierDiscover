Pseudo code for simplex:

def simplex_projection(v):
    """Projects vector v onto probability simplex {x>=0, sum x = 1}."""
    n=v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1,n+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

function simplex_projection(v):

   u = np.sort(v)[::-1]
   cssv = np.cumsum(u)
   rho = np.where(u - (cssv - 1) / (np.arange(len(v)) + 1) > 0)[0]
   rho = rho[-1]
   theta = (cssv[rho] - 1) / (rho + 1)
   w = np.maximum(v - theta, 0)
   return w

def propose_candidate(seed=...budget_s=1000,...):

   np.random.seed(seed)
   start_time = time.time()
   deadline = start_time + budget_s - 2

   # Determine n
   if 'height_sequence_1' in globals():
       n = len(height_sequence_1)
   else:
       n = 2000

   # Random initialization: Dirichlet distribution

   seq = init / init.sum()  # sum 1

   best_seq = seq.copy()
   best_val = 2 * n * np.max(np.convolve(seq, seq)) / (np.sum(seq) ** 2)  # internal

   # Parameters
   p = 6   # exponent for weighting
   epsilon = 0.01 # for noise
   step_factor = 0.5 # step scaling
   min_step = 1e-6

   # main loop
   iter = 0
   step = 0.1
   while time.time() < deadline:
       iter += 1

       # compute convolution and weighting
       b = np.convolve(seq, seq)
       max_b = b.max()

       # compute weight: w = (b / max_b) ** (p - 1)
       # raise to high exponent to emphasize top values
       w = (b / max_b) ** (p - 1)
       # pad for gradient: w_rev
       w_rev = w[::-1]
       convg = np.convolve(seq, w_rev)
       offset = len(w) - 1
       gradient = 2.0 * convg[offset - np.arange(n)]

       # Adaptive step: scale so that max absolute update about step_factor * mean(seq)
       max_abs = np.max(np.abs(gradient))
       if max_abs == 0:
           step_size = min_step
       else:
           step_size = step_factor * seq.mean() / max_abs

       # Proposed update
       new_seq = seq - step_size * gradient

       # Add small random jitter to avoid stagnation
       jitter = epsilon * np.random.randn(n) * seq.mean()
       new_seq += jitter

       # Project onto simplex (nonnegative, sum=1)
       new_seq = simplex_projection(new_seq)

       # Compute new objective
       b_new = np.convolve(new_seq, new_seq)
       obj_new = 2 * n * b_new.max() / (np.sum(new_seq) ** 2)

       if obj_new < best_val:

       else:
           # Occasionally accept worse move to escape local minima
           if np.random.rand() < 0.001:
               seq = new_seq
           # else keep old seq
           # adapt step down if not improving
           step_factor = max(0.01, step_factor * 0.99)

       # Update current sequence for next iteration
       seq = new_seq

       # Occasionally reinitialize if stuck
       if iter % 2000 == 0:

           else:
               # keep as is

           # Print progress
           print(f"Iter {iter}, best obj {best_val:.6f}")

   # Return best_seq
   return best_seq.tolist()

"""
propose_candidate – heuristic optimisation for the step‐function evaluation.

The objective is 2·n·max(conv(a,a)) / (sum(a)^2).  Since the value is
scale–invariant we keep the vector on the probability simplex (sum = 1,
all entries ≥ 0).  The algorithm works with the following ideas:

1.  Random Dirichlet initialisation (or the provided height_sequence_1).
2.  At every iteration we compute the full convolution b = a∗a.
3.  A smooth “soft‑max’’ weight w_k = (b_k / max(b))^p (p≈2) is built.
    The gradient of the weighted sum  Σ w_k·b_k w.r.t. a is
        ∇_i = 2·Σ_j a_j·w_{i+j}
    which can be obtained by one additional convolution.
4.  A gradient step of size proportional to 1/‖∇‖ is taken,
    a small random jitter is added to break symmetry,
    and the result is projected back onto the simplex.
5.  The best vector seen so far (lowest evaluation) is kept.
6.  Simple simulated‑annealing / random‑restart tricks are added so the
    search does not get stuck at the uniform fixed point.

The whole loop runs until the supplied time budget expires (≈1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # Find the largest rho such that u_rho - cssv_rho / rho > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # the projection is the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise again
    w /= w.sum() if w.sum() > 0 else 1.0
    return w

# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Heuristic search for a sequence that minimises evaluate_sequence().
    Returns a list of non‑negative floats (the best sequence found).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # leave a tiny safety margin

    # --------------------------------------------------------------
    # Determine the length of the problem – use the provided construction
    # if it exists, otherwise fall back to a moderate size.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        # use the given construction as a possible start point
        start_from_sota = np.maximum(0.0, np.array(height_sequence_1, dtype=float))
        start_from_sota /= start_from_sota.sum() if start_from_sota.sum() > 0 else 1.0
    else:
        n = 2000
        start_from_sota = None

    # --------------------------------------------------------------
    # Initialise the search vector a (on the simplex)
    # --------------------------------------------------------------
    if start_from_sota is not None and np.random.rand() < 0.4:
        a = start_from_sota.copy()
    else:
        # Dirichlet‑like random start
        a = np.random.exponential(scale=1.0, size=n)
        a /= a.sum()

    best_seq = a.copy()
    # internal objective (identical to evaluate_sequence for sum == 1)
    def _objective(x):
        b = np.convolve(x, x)
        return 2.0 * n * b.max() / (x.sum() ** 2)

    best_val = _objective(best_seq)

    # hyper‑parameters (tuned empirically)
    p = 2.0                # exponent for the soft‑max weighting (p>1)
    jitter_scale = 1e-3    # magnitude of random jitter (relative to mean)
    step_factor = 0.5      # controls absolute step size
    min_step = 1e-8
    iteration = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- convolution and weighting ---------------------------------
        b = np.convolve(a, a)
        max_b = b.max()
        if max_b == 0:
            # degenerate case – re‑initialise
            a = np.random.exponential(scale=1.0, size=n)
            a /= a.sum()
            continue

        # soft‑max weights (values in [0,1]); p≈2 gives a smooth focus on the peaks
        w = (b / max_b) ** (p - 1.0)

        # ----- gradient of Σ w_k b_k w.r.t. a ----------------------------
        # gradient_i = 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]
        conv_grad = np.convolve(a, w_rev)                # length = n + (2n‑1) - 1 = 3n‑2
        offset = len(w) - 1                               # = 2n‑2
        grad = 2.0 * conv_grad[offset - np.arange(n)]

        # ----- step size (scaled to the typical magnitude of a) ---------
        max_abs = np.max(np.abs(grad))
        if max_abs < 1e-12:
            step = min_step
        else:
            step = step_factor * a.mean() / max_abs

        # ----- propose a new point ---------------------------------------
        a_new = a - step * grad

        # add a tiny random jitter to avoid the symmetric uniform fixed point
        jitter = jitter_scale * a.mean() * (np.random.random(n) - 0.5)
        a_new += jitter

        # enforce feasibility (non‑negative, sum = 1)
        a_new = _project_onto_simplex(a_new)

        # ----- evaluate candidate -----------------------------------------
        val_new = _objective(a_new)

        # ----- accept / reject logic ---------------------------------------
        if val_new < best_val:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new
            # increase aggressiveness slightly after a success
            step_factor = min(1.0, step_factor * 1.02)
            print(f"[{iteration:07d}]  NEW BEST = {best_val:.6f}")
        else:
            # occasional uphill moves to escape shallow local minima
            if np.random.rand() < 0.001:
                a = a_new
                # slightly shrink the step factor after a bad move
                step_factor = max(0.01, step_factor * 0.95)
            else:
                # keep the current a unchanged
                pass

        # ----- periodic random restarts / perturbations -------------------
        if iteration % 5000 == 0:
            # if we have not improved for a while, re‑mix with random noise
            if val_new > best_val * 1.01:
                noise = np.random.exponential(scale=1.0, size=n)
                a = 0.9 * best_seq + 0.1 * noise
                a = _project_onto_simplex(a)
                print(f"[{iteration:07d}]  RE‑START triggered, best = {best_val:.6f}")

        # keep the loop responsive
        if iteration % 2000 == 0 and time.time() < deadline:
            elapsed = time.time() - (deadline - budget_s + 2.0)
            print(f"Iter {iteration:07d} – best = {best_val:.6f} – elapsed {elapsed:.1f}s")

    # --------------------------------------------------------------
    # Return the best sequence discovered (as a plain list)
    # --------------------------------------------------------------
    print(f"Finished after {iteration} iterations – best objective = {best_val:.6f}")
    return best_seq.tolist()
