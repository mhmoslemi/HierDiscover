def simplex_projection(v, z=1):
    # ensures nonnegative, sum = z.
    # v may be np.ndarray.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient–projected simplex descent for minimizing
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm keeps the L1‐norm of the sequence fixed to
    S = sqrt(2·n)   (the normalisation used in the reference solution)
and iteratively reduces the peak of the auto‑convolution.
At each step we compute the gradient of the current maximal
convolution entry (sub‑gradient of a max‑function), take a step
in the opposite direction and project back onto the non‑negative
simplex of radius S.  A simple backtracking line‑search guarantees
monotone improvement of the peak value.

Multiple restarts with different lengths (including the length of the
provided `height_sequence_1`) are performed until the time budget is
exhausted.  The best sequence found (according to `evaluate_sequence`)
 is returned.
"""

import time
import numpy as np
from scipy import optimize  # only used for a tiny fallback LP (optional)

# --------------------------------------------------------------
# Helper: projection onto the non‑negative simplex of radius z
def project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project a vector onto {x >= 0, sum(x) = z}."""
    if z < 0:
        raise ValueError("Radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    # find rho = max{i | u_i > cssv_i / i}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros case – just distribute uniformly
        return np.full_like(v, z / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Fast convolution using FFT (real input)
def fft_convolve(a: np.ndarray) -> np.ndarray:
    """Return the autocorrelation a*a (full mode) using FFT."""
    n = a.shape[0]
    # length for FFT – next power of two ≥ 2·n‑1
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------
def gradient_of_max(b: np.ndarray, a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(b) w.r.t. a (a ≥ 0, sum fixed).
    active – indices k where b_k equals the current maximum (within tolerance).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    # each active index contributes 2*a_{k-i} to the i‑th component
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # loop over the overlapping region
        for i in range(i_start, i_end + 1):
            j = k - i
            grad[i] += 2.0 * a[j]
    return grad


# --------------------------------------------------------------
def refine_sequence(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Perform projected gradient descent on a single sequence.
    Returns the best sequence encountered and its evaluation score.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)          # normalisation target for the sum
    a = project_to_simplex(a, S)   # guarantee feasibility

    best_a = a.copy()
    best_maxb = np.max(fft_convolve(a))

    # initial step size (relative to the L1‑radius)
    step_size = 0.1 * S

    # iteration limit – we stop earlier if time runs out
    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution and its maximal entry
        b = fft_convolve(a)
        maxb = b.max()
        # active set (tolerance avoids floating‑point noise)
        eps = 1e-12
        active = np.where(b >= maxb - eps)[0]

        # compute sub‑gradient of the max entry
        grad = gradient_of_max(b, a, active)
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            break

        # ---------- backtracking line search ----------
        step = step_size
        improved = False
        while step > 1e-14:
            a_candidate = a - step * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = project_to_simplex(a_candidate, S)

            b_candidate = fft_convolve(a_candidate)
            maxb_candidate = b_candidate.max()

            if maxb_candidate < maxb - 1e-12:
                # accept
                a = a_candidate
                maxb = maxb_candidate
                improved = True
                step_size = step * 1.05               # slightly enlarge next step
                break
            step *= 0.5

        if not improved:
            step_size *= 0.5          # shrink step if we made no progress

        # keep the best seen so far
        if maxb < best_maxb:
            best_maxb = maxb
            best_a = a.copy()

        # optional early stop if the peak is already very low
        if maxb <= 1.5:
            break

        # occasional random jitter to escape possible stagnation
        if it % 200 == 0 and np.random.rand() < 0.3:
            jitter = np.random.randn(n) * (step_size * 0.1)
            a = np.maximum(best_a + jitter, 0.0)
            a = project_to_simplex(a, S)

    # final score using the provided evaluator
    try:
        score = evaluate_sequence(best_a.tolist())
    except Exception:  # defensive – should never happen
        score = np.inf
    return best_a, float(score)
