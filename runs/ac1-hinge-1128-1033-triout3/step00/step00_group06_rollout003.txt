"""
Simple sub‑gradient optimiser for the step‑function objective.

The evaluation function is

    f(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

If we keep the ℓ1‑norm of the sequence equal to
    S = sqrt(2·n)
the denominator equals the numerator’s constant factor, so
    f(a) = max(conv(a,a))

Thus the problem reduces to:  minimise the largest element of the
self‑convolution while keeping the entries non‑negative and the sum fixed.
The algorithm performs a projected sub‑gradient descent on the L∞‑norm
of the convolution (the max entry).  At each step we:

1. compute the convolution and locate all indices attaining its maximum;
2. build a sub‑gradient (the sum of the contributions a_i·a_{k-i});
3. take a step opposite to the sub‑gradient;
4. project back onto the simplex {x≥0, sum(x)=S}.

Random restarts, occasional Gaussian noise and a small set of
different sequence lengths are used to avoid local minima.
The best sequence found within the time budget is returned.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = z, x_i >= 0)
# ------------------------------------------------------------
def simplex_projection(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector v onto the simplex {x >= 0, sum(x)=z}.
    Implementation follows the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if z < 0:
        raise ValueError("z must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)

    # Find the rho index (last positive entry of u - cssv/ind)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Core: one projected sub‑gradient step
# ------------------------------------------------------------
def subgradient_step(seq: np.ndarray, target_sum: float, step_factor: float,
                     it: int) -> np.ndarray:
    """
    Perform a single projected sub‑gradient step aimed at lowering
    max(conv(seq, seq)).  Returns the new sequence (still non‑negative)
    with the same ℓ1‑norm = target_sum.
    """
    n = seq.shape[0]
    # convolution (full)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    # active convolution indices (could be more than one)
    active = np.where(conv == max_b)[0]

    # build sub‑gradient (vector of same size as seq)
    grad = np.zeros_like(seq)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution 2 * a_{k-i}
        grad[idx] += 2.0 * seq[k - idx]

    # step size: decays like O(1/√t)
    norm = np.linalg.norm(grad) + 1e-16
    step = step_factor * target_sum / norm / math.sqrt(it + 1.0)

    # gradient descent step (move opposite to gradient)
    v = seq - step * grad

    # projection back onto the simplex {x>=0, sum = target_sum}
    new_seq = simplex_projection(v, target_sum)
    return new_seq

# Main entry point required by the competition harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    evaluate_sequence.  Uses random restarts and a projected sub‑gradient
    optimiser described above.  Returns the best sequence discovered before
    the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little safety margin

    best_seq = None
    best_val = float("inf")
    iter_counter = 0

    # -----------------------------------------------------------------
    # Optionally start from the previously known construction if it exists
    # -----------------------------------------------------------------
    try:
        init = list(height_sequence_1)   # global supplied by the environment
        n0 = len(init)
        S0 = math.sqrt(2.0 * n0)
        init_arr = np.array(init, dtype=np.float64)
        if init_arr.sum() > 0:
            init_arr = init_arr * (S0 / init_arr.sum())
        val0 = evaluate_sequence(init_arr.tolist())
        best_seq = init_arr.tolist()
        best_val = val0
        print(f"[init] height_sequence_1 value = {best_val:.6f}")
    except Exception:
        # no supplied sequence – continue with random starts only
        pass

    # ---------------------------------------------------------------
    # Main search loop – keep generating new random starts until time up
    # ---------------------------------------------------------------
    while time.time() < deadline:
        # Choose a length for the current restart (moderate size)
        n = np.random.randint(300, 1300)          # 300 … 1300 gives a good trade‑off
        target_sum = math.sqrt(2.0 * n)

        # Random start: exponential distribution gives a varied shape
        raw = np.random.exponential(scale=1.0, size=n)
        if raw.sum() == 0:
            raw = np.ones(n)
        seq = raw * (target_sum / raw.sum())

        # Number of sub‑gradient iterations for this restart
        max_iter = np.random.randint(300, 800)

        # Sub‑gradient descent
        for it in range(max_iter):
            if time.time() > deadline:
                break

            # Full convolution and objective value (no extra call to evaluate_sequence)
            conv = np.convolve(seq, seq)
            max_b = conv.max()
            # Because sum(seq) == target_sum (up to rounding), the true score equals max_b
            cur_val = float(max_b)

            # Record improvement
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_seq = seq.copy()
                print(f"[new best] val={best_val:.6f}  n={n}  iter={it}  total_iters={iter_counter}")

            # One (projected) sub‑gradient step
            seq = subgradient_step(seq, target_sum, step_factor=0.15, it=it)

            # Occasional random perturbation to escape flat regions
            if np.random.rand() < 0.003:
                noise = np.random.normal(scale=target_sum * 0.0008, size=n)
                seq = np.maximum(seq + noise, 0.0)
                seq = simplex_projection(seq, target_sum)

        iter_counter += 1

    # -----------------------------------------------------------------
    # Fallback in the very unlikely case that no sequence was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 2
        S = math.sqrt(2.0 * n)
        best_seq = [S / n] * n
        best_val = evaluate_sequence(best_seq)

    # Ensure the result is a plain Python list of floats (as required)
    best_seq = [float(x) for x in best_seq]

    # Final sanity check (optional – cheap)
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] best value = {final_score:.6f}  length = {len(best_seq)}")
    except Exception:
        pass

    return best_seq
