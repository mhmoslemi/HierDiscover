def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 2500  # length
    S = np.sqrt(2 * n)

    # evaluate uniform baseline
    best_seq = np.full(n, S / n)
    best_score = evaluate_sequence(best_seq.tolist())

    print(f'Initial uniform score {best_score}')
    
    # Phase 1: random sparsity
    while time.time() < deadline * 0.4: # maybe 40% time
        k = np.random.randint(max(2, int(0.02*n)), int(0.2*n)) # between 2% and 20% n
        seq = np.zeros(n)
        idx = np.random.choice(n, size=k, replace=False)
        seq[idx] = S / k
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_seq = seq.copy()
            best_score = score
            print(f'Random sparse k={k} score {score}')
    
    # Phase 2: local refinement using gradient moves
    seq = best_seq.copy()
    
    # iteration
    while time.time() < deadline:
        # compute convolution, find max index
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        # find all ks with max (maybe multiple)
        max_idxs = np.flatnonzero(conv == max_val)
        k_idx = max_idxs[0]  # choose one
        
        # compute gradient
        grad = np.zeros_like(seq)
        # for each i, if k_idx - i within range
        for i in range(n):
            j = k_idx - i
            if 0 <= j < n:
                if i == j:
                    grad[i] += seq[i]  # derivative of a_i^2
                else:
                    grad[i] += 2 * seq[j]
        # Identify source: index with largest grad and nonzero seq
        source_candidates = np.where(seq > 1e-12)[0]
        if source_candidates.size == 0:
            break
        src = source_candidates[np.argmax(grad[source_candidates])]
        # target: index with smallest grad (including zero seq)
        tgt = np.argmin(grad)
        if src == tgt:
            # choose random
            tgt = np.random.randint(n)
            if tgt == src:
                continue
        
        # delta move
        delta = min(seq[src] * 0.2, S * 0.001)  # small portion
        new_seq = seq.copy()
        new_seq[src] -= delta
        new_seq[tgt] += delta
        # re-scale to keep sum S (avoid drift)
        scale = S / new_seq.sum()
        new_seq *= scale
        
        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < best_score:
            best_seq = new_seq
            best_score = new_score
            seq = new_seq
            print(f'Improved to {best_score} at time {time.time():.2f}')
        else:
            # Accept with small probability?
            if np.random.rand() < 0.01:
                seq = new_seq
        
    print(f'Final best score {best_score}')
    return best_seq.tolist()

"""
propose_candidate
-----------------

A simple but effective local‑search algorithm that starts from a uniform (or the
previous best) step‑function, then repeatedly moves a small amount of mass from
indices that increase the current maximum of the self‑convolution to indices
that have the smallest influence.  Because the evaluation function is
`2*n*max(conv)/sum(a)^2`, we keep the total sum fixed (≈√(2n)) so that the
objective is exactly the maximum convolution value.  The method consists of:

1. Initialise a sequence of length ``n`` (default 2500) with a uniform
   distribution or the historic ``height_sequence_1`` if it is available.
2. Randomly try a number of sparse sequences (few non‑zero entries) and keep
   the best one.
3. Run a gradient‑guided hill‑climbing loop:
   * locate the index where the convolution attains its maximum,
   * compute a sub‑gradient of this maximum w.r.t. each coefficient,
   * move a modest amount of mass from the coefficient with the largest
     gradient (and positive value) to the coefficient with the smallest
     gradient,
   * renormalise the sequence to keep the sum exactly ``sqrt(2*n)``,
   * accept the move if the evaluation improves (occasional random acceptance
     is allowed to escape plateaus).

The loop runs until the allotted time (``budget_s``) is almost exhausted.
The best sequence found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def _max_convolution(seq: np.ndarray) -> np.ndarray:
    """
    Linear convolution of ``seq`` with itself using FFT (fast for the
    moderate lengths we use).  Returns the full convolution array.
    """
    n = seq.size
    # size for linear convolution (next power of two for speed)
    size = int(2 ** np.ceil(np.log2(2 * n - 1)))
    fft_a = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]
    return conv


def _gradient_of_max(conv: np.ndarray, seq: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Compute the (sub‑)gradient of the current maximal convolution entry
    ``conv[max_idx]`` with respect to each coefficient of ``seq``.
    For a self‑convolution the derivative is:
        d/d a_i [ (a*a)_k ] = a_{k-i} + a_{k-i}  (i ≠ k-i)
                          = a_i               (i = k-i)
    The result is a vector of length ``len(seq)``.
    """
    n = seq.size
    grad = np.zeros_like(seq)

    # valid i such that 0 <= k-i < n
    i_min = max(0, max_idx - (n - 1))
    i_max = min(n - 1, max_idx)
    i_vals = np.arange(i_min, i_max + 1, dtype=int)
    j_vals = max_idx - i_vals

    # where i == j (the centre of the convolution)
    centre_mask = i_vals == j_vals
    grad[i_vals[centre_mask]] += seq[i_vals[centre_mask]]
    # where i != j
    grad[i_vals[~centre_mask]] += 2.0 * seq[j_vals[~centre_mask]]

    return grad


def _random_sparse_sequence(n: int, k: int, target_sum: float) -> np.ndarray:
    """
    Build a sequence of length ``n`` with exactly ``k`` equally‑weighted
    non‑zero entries and total sum ``target_sum``.
    """
    seq = np.zeros(n, dtype=float)
    idx = np.random.choice(n, size=k, replace=False)
    seq[idx] = target_sum / float(k)
    return seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # keep a small safety margin

    # ------------------------------------------------------------------
    # Problem size – a compromise between solution quality and speed
    # ------------------------------------------------------------------
    n = 2500                     # length of the sequence (feel free to change)
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Start from a reasonable baseline: uniform or the historic best
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be defined in the environment; use it if it
        # exists and has the right length.
        if isinstance(height_sequence_1, (list, np.ndarray)):
            seq0 = np.asarray(height_sequence_1, dtype=float)
            if seq0.size > 0:
                # re‑scale to the target sum while keeping non‑negativity
                seq0 = seq0 * (target_sum / seq0.sum())
                best_seq = seq0.copy()
                best_score = float("inf")
            else:
                raise NameError
        else:
            raise NameError
    except Exception:
        # fall back to a uniform sequence
        best_seq = np.full(n, target_sum / float(n), dtype=float)
        best_score = float("inf")

    # Evaluate the initial choice (uniform or historic)
    try:
        cur_score = evaluate_sequence(best_seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            print(f"[init] score from initial sequence = {best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random sparse trials (exploration)
    # ------------------------------------------------------------------
    # Allocate roughly 35 % of the time budget to this coarse search.
    phase1_deadline = start_time + 0.35 * budget_s
    while time.time() < min(phase1_deadline, deadline):
        # Choose a random sparsity level (2 % – 15 % of the entries)
        k = np.random.randint(int(0.02 * n), int(0.15 * n) + 1)
        cand = _random_sparse_sequence(n, k, target_sum)
        try:
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(
                    f"[rand] k={k:4d}  score={score:.6f}  time={time.time() - start_time:.1f}s"
                )
        except Exception:
            # Should never happen because we respect the input constraints
            continue

    # ------------------------------------------------------------------
    # Phase 2 – gradient‑guided local optimisation (exploitation)
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    # Parameters controlling the hill‑climbing dynamics
    max_no_improve = 2000               # stop early if nothing improves for a while
    no_improve_counter = 0
    temperature = 0.0                    # not used for now (set to zero)

    while time.time() < deadline:
        # Compute the full self‑convolution and locate the maximal entry
        conv = _max_convolution(seq)
        max_val = conv.max()
        max_positions = np.flatnonzero(conv == max_val)

        # Take the first maximal index (any works – the algorithm quickly
        # adapts if there are several equal peaks)
        k_max = int(max_positions[0])

        # Gradient of the current maximal convolution entry
        grad = _gradient_of_max(conv, seq, k_max)

        # ------------------------------------------------------------------
        # Choose a source (high gradient, positive mass) and a destination
        # (low gradient, possibly zero mass)
        # ------------------------------------------------------------------
        positive_mask = seq > 1e-12
        if not np.any(positive_mask):
            break  # nothing to move

        # source: index with largest *gradient* among those with some mass
        src_candidates = np.where(positive_mask)[0]
        src = src_candidates[np.argmax(grad[src_candidates])]

        # destination: index with the smallest gradient (could be zero)
        dst = int(np.argmin(grad))

        if src == dst:
            # pick a random partner if the gradient map is flat
            dst = np.random.randint(0, n)

        # Amount to move – a modest fraction of the source value
        delta = min(seq[src] * 0.15, target_sum * 0.001)
        if delta <= 0:
            break

        new_seq = seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # Renormalise to keep the total sum exactly ``target_sum``.
        # This multiplication does not change the direction of the move.
        scale = target_sum / new_seq.sum()
        new_seq *= scale

        # ------------------------------------------------------------------
        # Acceptance test
        # ------------------------------------------------------------------
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            # numerical trouble – reject this move
            continue

        # Accept if strictly better, otherwise keep the current sequence.
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            seq = new_seq
            no_improve_counter = 0
            print(
                f"[impr] iter={no_improve_counter:5d}  score={best_score:.6f}  time={time.time() - start_time:.1f}s"
            )
            # If we have already achieved the target, we can stop early.
            if best_score <= 1.503:
                break
        else:
            # Occasionally accept a worse move to escape flat regions
            if temperature > 0 and np.random.rand() < np.exp(-(new_score - best_score) / temperature):
                seq = new_seq
            else:
                # reject – keep the old sequence
                no_improve_counter += 1

        # Simple cooling schedule for the optional temperature (kept at 0)
        # temperature = max(0.0, temperature * 0.9999)

        # Early exit if we have been stagnant for a long while
        if no_improve_counter > max_no_improve:
            # Re‑seed the search from the current best with a small random jitter
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            seq = best_seq + jitter
            seq = np.clip(seq, 0, None)                # enforce non‑negativity
            seq = seq * (target_sum / seq.sum())      # renormalise
            no_improve_counter = 0
            continue

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain list)
    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} after {time.time() - start_time:.1f}s")
    return best_seq.tolist()
