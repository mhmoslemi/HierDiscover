def project_simplex(v, s):
    """Project v onto the simplex {x >= 0, sum x = s}."""
    if s < 0:
        raise ValueError
    n = v.shape[0]
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # Find rho
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all negative
        theta = cssv[-1] / n
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def project_simplex(v, z):
    """Project vector v onto the simplex {x >= 0, sum x = z}."""
    # Duchi et al. (2008)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]  # sorted descending
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0, return uniform
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def autocorr(a):
    """Linear (non‑circular) autocorrelation of a using FFT."""
    n = a.shape[0]
    # length for convolution: next power of two >= 2*n-1
    L = 1 << ((2 * n - 1).bit_length())
    # real FFT
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]

def gradient_of_max(a, conv, max_val, tol=1e-12):
    """Return sub‑gradient of max(conv) w.r.t. a."""
    n = a.shape[0]
    # active indices (where conv is (nearly) maximal)
    K = np.where(conv >= max_val - max(tol * max_val, 1e-15))[0]
    g = np.zeros_like(a)
    for k in K:
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # indices of a that appear in this term
        idx = k - np.arange(j_start, j_end + 1)
        g[j_start : j_end + 1] += 2.0 * a[idx]
    return g

"""
search for a low‑value step function (list of heights)

The evaluation function is
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²)

If we enforce the normalisation  sum(a) = sqrt(2·n)  then the
evaluation equals simply  max(conv(a,a)).  Consequently the task
reduces to:  minimise the peak of the autocorrelation of a
non‑negative vector a with a prescribed ℓ₁‑norm.

Algorithm
---------
* Work on a sequence length n that grows during the run (start at n≈200,
  finish ≈5000).  The current best vector is linearly interpolated to the
  larger size before continuing.
* Initialise a∈ℝ₊ⁿ with a smooth “tent” shape and scale it so that
  sum(a)=√(2n).  (A random start is also possible.)
* Repeatedly
    – compute the linear autocorrelation  b = a∗a  (FFT based, O(n log n));
    – find the current maximum m = max(b);
    – evaluate the sub‑gradient of m w.r.t. a (only the active indices
      where b≈m contribute);
    – take a projected sub‑gradient step with a diminishing step size,
      followed by a projection onto the simplex {x≥0, sum x = √(2n)}.
    – if the step does not improve the peak, halve the step and retry;
      after a few failures a small random perturbation is applied.
* Keep the best vector ever seen (lowest peak).  The loop stops when
  time runs out (the caller supplies a 1000 s budget) or when the peak
  is within 0.5 % of the theoretical lower bound  2n/(2n−1).

The final list of heights is returned; the external ``evaluate_sequence``
will recompute the same value (the peak of the autocorrelation) and
award the score.

import time
import numpy as np


def _project_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, sum x = z} (Duchi et al.)."""
    n = v.size
    if n == 0:
        return v.copy()
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become zero – fallback to uniform distribution
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _autocorr(a: np.ndarray) -> np.ndarray:
    """Linear autocorrelation of a (a∗a) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def _gradient_of_max(a: np.ndarray, conv: np.ndarray, max_val: float,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of  max(conv)  w.r.t. a.
    Only indices k where conv[k] is (almost) maximal are active.
    """
    n = a.shape[0]
    # treat *all* indices that are within a tiny absolute tolerance
    tol = max(rel_tol * max_val, 1e-15)
    K = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in K:
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # indices of a that appear in the term for this k
        idx = k - np.arange(j_start, j_end + 1)
        g[j_start:j_end + 1] += 2.0 * a[idx]
    return g


def _tent_initial(n: int) -> np.ndarray:
    """Triangular (tent) shape centred in the interval [0,n)."""
    idx = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(idx - centre) / (centre + 1e-12)
    return np.maximum(a, 0.0)

            best_seq = np.ones(1, dtype=float) * np.sqrt(2.0)

    return best_seq.tolist()
