def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Returns a sequence of non‑negative heights (list of floats) that
    minimizes ``evaluate_sequence``.
    The function explores a few values of the exponent p in the
    decreasing power‑law shape a_i = (n-i)^p, rescales to the required L1
    norm and keeps the best candidate.
    """
    # Timing – stop early if we are running out of the allocated budget.
    deadline = time.time() + budget_s - 0.1   # leave a tiny safety margin

    # Use the same length that was used in the previous best construction.
    # It is large enough to make the asymptotic formula accurate.
    n = 5808

    # The required L1 norm (the code normalises the sequence to this value).
    target_sum = math.sqrt(2.0 * n)

    best_seq = None
    best_score = float("inf")
    best_p = None

    # Test a small range of integer exponents.  Larger p give dramatically
    # smaller scores, but the gain saturates because the peak of the
    # convolution eventually moves to the very first entry.
    for p in range(1, 11):      # exponents 1 … 10
        # ----- build the raw (unscaled) sequence -------------------------
        # Use float64 to avoid integer overflow for large powers.
        raw = np.arange(n, 0, -1, dtype=np.float64) ** p

        # ----- scale so that the L1 norm equals √(2 n) --------------------
        scale = target_sum / raw.sum()
        seq = raw * scale

        # Convert to a plain Python list – this is what evaluate_sequence expects.
        candidate = seq.tolist()

        # ----- evaluate ----------------------------------------------------
        try:
            score = evaluate_sequence(candidate)
        except Exception as e:
            # Any unexpected error – treat as infinitely bad.
            print(f"Evaluation failed for p={p}: {e}")
            continue

        print(f"p = {p:2d} | eval = {score:.6f}")

        # Keep the best result seen so far.
        if score < best_score:
            best_score = score
            best_seq = candidate
            best_p = p

        # Stop early if we are about to exceed the time budget.
        if time.time() > deadline:
            print("Time budget exhausted, stopping search.")
            break

    # Final report
    print(f"\nBest exponent found: p = {best_p} with evaluation = {best_score:.6f}")
    return best_seq
