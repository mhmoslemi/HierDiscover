"""
Propose a candidate step‑function sequence that improves over the
previous bound (≈2.0).  The algorithm works in two phases:

1.  **Generation of diverse starting shapes** – several simple parametric
    families (uniform, linear, quadratic, exponential, U‑shaped, random
    Dirichlet) are created and the best among them is kept.

2.  **Iterative flattening of the worst convolution entry** – at each
    iteration the index `k_max` where the self‑convolution is maximal is
    located.  The two factors that contribute most to this entry are
    identified (`i_max` and `j_max = k_max‑i_max`).  Their values are
    reduced a little while the removed mass is transferred to the
    currently smallest element.  This operation keeps the total sum
    (approximately) constant but lowers the peak of the convolution.
    If the new sequence yields a lower score it is accepted; otherwise
    the change is discarded.  Random perturbations and occasional full
    restarts keep the search from stagnating.

The whole search runs for the allotted time (default 990 s, leaving a
few seconds for a clean return).  The final best sequence found is
returned as a plain Python list of floats.
"""

import time
import numpy as np

def build_initial_candidates(n, rng):
    """
    Produce a list of candidate sequences using simple parametric families.
    Each candidate is a NumPy array of length ``n`` with non‑negative values.
    """
    candidates = []

    # 1) Uniform
    cand = np.ones(n)
    candidates.append(cand)

    # 2) Linear ramp (increase)
    cand = np.linspace(0.1, 1.0, n)
    candidates.append(cand)

    # 3) Linear ramp (decrease)
    cand = np.linspace(1.0, 0.1, n)
    candidates.append(cand)

    # 4) Quadratic (convex)
    t = np.linspace(0, 1, n)
    cand = (t - 0.5) ** 2
    cand = cand - cand.min() + 0.1        # shift to stay positive
    candidates.append(cand)

    # 5) Exponential decay
    cand = np.exp(-np.linspace(0, 5, n))
    candidates.append(cand)

    # 6) U‑shaped (large at the edges, small in the middle)
    idx = np.arange(n)
    cand = 1.0 / (idx + 1) + 1.0 / (n - idx)
    candidates.append(cand)

    # 7) Random Dirichlet (dense positive vector)
    cand = rng.dirichlet(np.ones(n))
    candidates.append(cand)

    # 8) Sparse random support (like a Sidon‑set pattern)
    m = int(np.sqrt(2 * n))            # maximal size of a Sidon‑like set
    support = np.zeros(n)
    chosen = set()
    # simple greedy Sidon construction using triangular numbers
    i = 0
    while len(chosen) < m and i < n:
        t = i * (i + 1) // 2
        if t < n:
            support[t] = 1.0
            chosen.add(t)
        i += 1
    candidates.append(support + 0.001)  # tiny background to avoid zeros

    return candidates


def flatten_peak(a, eps, rng):
    """
    Perform one local improvement step.
    Identify the convolution peak and move a small amount of mass
    from the two entries that contribute most to that peak towards the
    current smallest entry.
    Returns a new array (may be identical to the input if no improvement).
    """
    n = a.shape[0]
    # full convolution (length 2n‑1)
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))          # index of the worst entry

    # valid indices i such that 0 <= i < n and 0 <= k_max-i < n
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    i_vals = np.arange(i_low, i_high + 1)
    j_vals = k_max - i_vals

    # contributions of each ordered pair (i, j)
    contrib = a[i_vals] * a[j_vals]
    # locate the pair with the largest product
    max_idx = int(np.argmax(contrib))
    i_max = i_vals[max_idx]
    j_max = j_vals[max_idx]

    # Amount to take away – a small fraction of the smaller of the two
    take = min(a[i_max], a[j_max]) * eps
    if take <= 0.0:
        return a.copy()     # nothing to do

    a_new = a.copy()
    a_new[i_max] -= take
    a_new[j_max] -= take

    # Transfer the removed mass to the globally smallest entry
    # (this raises the denominator while hardly affecting the peak)
    idx_min = int(np.argmin(a_new))
    a_new[idx_min] += 2.0 * take

    # Numerical safety – enforce the non‑negative, bounded domain
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new


def random_perturbation(a, rng):
    """
    Small random change to avoid being stuck.
    """
    n = a.shape[0]
    idx = rng.integers(0, n)
    delta = rng.normal() * 0.05 * a[idx]  # 5% Gaussian perturbation
    a_new = a.copy()
    a_new[idx] = max(0.0, a_new[idx] + delta)
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000):
    """
    Return a list of non‑negative heights (floats) that gives a low
    value of ``evaluate_sequence``.  The routine spends up to
    ``budget_s`` seconds searching; it finally returns the best sequence
    discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5   # leave a tiny margin

    # ------------------------------------------------------------------
    # Choose a problem size that balances exploration speed and quality.
    # Empirically, n≈500–800 works well within the given time budget.
    # ------------------------------------------------------------------
    n = 600

    # ------------------------------------------------------------------
    # 1.  Evaluate a set of diverse starting points.
    # ------------------------------------------------------------------
    candidates = build_initial_candidates(n, rng)
    best_seq = None
    best_score = np.inf

    for cand in candidates:
        try:
            score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    if best_seq is None:
        # fallback – pure random dense vector
        best_seq = rng.random(n)
        best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # 2.  Iterative refinement of the current best sequence.
    # ------------------------------------------------------------------
    current = best_seq.copy()
    eps = 0.04                     # fraction of mass moved each step
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # local peak‑flattening step
        new_seq = flatten_peak(current, eps, rng)

        # occasional random jitter to escape plateaus
        if rng.random() < 0.02:
            new_seq = random_perturbation(new_seq, rng)

        # evaluate the candidate
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            # any numerical hiccup – skip this candidate
            continue

        # accept if we made progress
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            print(f"Iter {iteration:6d} – new best score: {best_score:.6f}")

        # simple deterministic acceptance: keep improvement, otherwise keep
        # the current sequence only if we improved the current score.
        # (This prevents wandering away from a good region.)
        cur_score = evaluate_sequence(current.tolist())
        if new_score <= cur_score:
            current = new_seq

        # occasional full restart to diversify search
        if rng.random() < 0.001:
            current = rng.random(n)
            eps = 0.04

    # Final sanity: enforce limits required by ``evaluate_sequence``.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
