def local_search(seq, deadline):
    n = len(seq)
    sum_target = 1.0 # after normalization
    cur_seq = np.array(seq)
    cur_score = evaluate_sequence(cur_seq.tolist())
    best_seq = cur_seq.copy()
    best_score = cur_score

    temp = 1.0
    decay = 0.9999

    while time.time() < deadline:
        # propose mutation: pick random index(s)
        idx = np.random.randint(n)
        # random delta in [-d, d]
        d = 0.05 * temp  # adapt
        delta = np.random.randn() * d
        new_seq = cur_seq.copy()
        new_seq[idx] = max(0.0, new_seq[idx] + delta)
        # Renormalize
        total = new_seq.sum()
        if total <= 1e-12:
            continue
        new_seq = new_seq / total
        # Evaluate
        new_score = evaluate_sequence(new_seq.tolist())
        # Accept if better or with prob
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / max(temp, 1e-6)):
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                print(f"Local best: {best_score:.6f}")
        # maybe occasionally do larger perturb or random restart
        if np.random.rand() < 0.001:
            # Random restart
            cur_seq = np.random.rand(n)
            cur_seq = cur_seq / cur_seq.sum()
            cur_score = evaluate_sequence(cur_seq.tolist())

        temp *= decay
        # also break if temp low
        if temp < 1e-4:
            temp = 1.0 # reset? or break
    return best_seq.tolist(), best_score

"""
Propose candidate sequence using hybrid simulated annealing and LP-guided direction.
We start from several seeds (constant, random, and provided height_sequence_1),
run a simple annealing that keeps the sum constant, and occasionally invoke the
LP from the reference implementation to compute a promising direction.  The
best sequence seen before the deadline is returned.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog

def solve_convolution_lp(f_sequence, rhs):
    """Solve LP: maximize sum(b) s.t. conv(f,b) <= rhs, b>=0."""
    n = len(f_sequence)
    c = -np.ones(n)  # maximize sum => minimize -sum
    # Build convolution constraints
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non-negativity constraints
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)
    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])
    # solve with a modest time limit
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', options={'time_limit': 5})
    if res.success:
        return res.x
    return None

def lp_mix_step(seq):
    """Return a new sequence obtained by mixing seq with an LP solution.
    The function also performs a line search over mixing weights."""
    n = len(seq)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return None, None
    # Normalise to the canonical sum sqrt(2n) used by the reference algorithm
    normalized = seq * np.sqrt(2 * n) / sum_seq
    # Current worst convolution value (the bound that the LP must respect)
    rhs = np.max(np.convolve(normalized, normalized))
    g = solve_convolution_lp(normalized, rhs)
    if g is None:
        return None, None
    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None, None
    # Normalise g in the same way
    g_norm = g * np.sqrt(2 * n) / sum_g

    # Try a few mixing ratios and keep the best
    base_score = evaluate_sequence(seq.tolist())
    best_seq = seq
    best_score = base_score
    for t in (0.001, 0.003, 0.01, 0.03, 0.1):
        new_seq = (1 - t) * seq + t * g_norm
        # enforce non‑negativity (numerical noise may produce tiny negatives)
        new_seq = np.maximum(new_seq, 0.0)
        # No need to re‑scale because the objective is scale invariant
        try:
            score = evaluate_sequence(new_seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = new_seq
        except Exception:
            pass
    return best_seq, best_score

def perturb_sequence(seq, epsilon):
    """Small random redistribution that keeps the total sum unchanged."""
    n = len(seq)
    if n < 2:
        return seq
    i, j = np.random.choice(n, size=2, replace=False)
    delta = np.random.randn() * epsilon
    # Apply delta to i, -delta to j, and clip to keep non‑negativity
    new_seq = np.copy(seq)
    new_seq[i] = max(0.0, new_seq[i] + delta)
    new_seq[j] = max(0.0, new_seq[j] - delta)
    # If clipping destroyed the total sum, renormalise to the original sum
    s = np.sum(new_seq)
    if s < 1e-12:
        # fallback to original
        return seq
    new_seq = new_seq * (np.sum(seq) / s)
    return new_seq

def local_anneal(start_seq, deadline):
    """Simple simulated‑annealing loop that tries perturbations and LP steps."""
    seq = np.array(start_seq, dtype=float)
    # Normalise sum to 1 to have a stable scale (scale does not affect objective)
    total = np.sum(seq)
    if total == 0:
        seq[:] = 1.0
        total = np.sum(seq)
    seq = seq / total
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    # Annealing schedule
    temperature = 1.0
    temp_decay = 0.9995
    while time.time() < deadline:
        # Occasionally use the LP guided move (more expensive)
        if np.random.rand() < 0.05:
            new_seq, new_score = lp_mix_step(seq)
            if new_seq is not None and new_score is not None:
                seq = new_seq
                cur_score = new_score
            else:
                # fall back to a cheap perturbation
                seq = perturb_sequence(seq, epsilon=temperature * 0.05)
                cur_score = evaluate_sequence(seq.tolist())
        else:
            # cheap random perturbation
            seq = perturb_sequence(seq, epsilon=temperature * 0.05)
            cur_score = evaluate_sequence(seq.tolist())

        # Acceptance test (Metropolis)
        improve = cur_score < best_score
        if improve or np.random.rand() < np.exp((best_score - cur_score) / max(temperature, 1e-8)):
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"New global best {best_score:.6f} at time {time.time():.1f}")
        # cool down
        temperature = max(temperature * temp_decay, 1e-4)

        # occasional random restart to avoid stagnation
        if np.random.rand() < 0.001:
            seq = np.random.rand(len(seq))
            seq = seq / np.sum(seq)
            temperature = 1.0
    return best_seq.tolist(), best_score

"""
Hybrid search for a non‑negative step function that minimises
``evaluate_sequence``.

The algorithm combines three ideas:

1. **Diverse seeds** – constant, random, and the provided
   ``height_sequence_1`` (trimmed/padded).

2. **Simulated annealing** – cheap random redistributions that keep the
   total sum fixed. The Metropolis rule occasionally accepts worsening
   moves to escape local minima.

3. **LP‑guided direction** – periodically solve the linear programme used
   in the reference SOTA method, mix the current sequence with the LP
   solution and perform a tiny line‑search over the mixing weight.
   This provides a principled descent step.

The best sequence seen before the time budget expires is returned.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Linear programme used by the reference algorithm
# ----------------------------------------------------------------------
linprog = optimize.linprog


def solve_convolution_lp(f_sequence, rhs):
    """Solve the LP:
       maximise    sum(b)
       subject to  conv(f, b) <= rhs   (for all 2n‑1 positions)
                   b >= 0.
       Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_sequence)
    # maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints: for each k, sum_i f[i] * b[k-i] <= rhs
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints: -b_i <= 0
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)

    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    # Use the modern HiGHS solver; give a modest time limit
    try:
        res = linprog(
            c,
            A_ub=a_ub,
            b_ub=b_ub,
            method="highs",
            options={"time_limit": 5.0},
        )
    except Exception:  # pragma: no cover
        return None

    if res.success:
        return res.x
    return None


def lp_mix_step(seq):
    """
    Construct a candidate by mixing ``seq`` with an LP solution.
    A few tiny mixing coefficients are tried and the best is returned.
    """
    n = len(seq)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return None, None

    # Normalise exactly as the reference code does
    normalized = seq * np.sqrt(2 * n) / sum_seq
    rhs = np.max(np.convolve(normalized, normalized))

    g = solve_convolution_lp(normalized, rhs)
    if g is None:
        return None, None

    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None, None
    g_norm = g * np.sqrt(2 * n) / sum_g

    # Evaluate the base sequence once
    try:
        best_score = evaluate_sequence(seq.tolist())
    except Exception:  # pragma: no cover
        return None, None
    best_seq = seq

    # Tiny line‑search over mixing coefficients
    for t in (0.001, 0.003, 0.01, 0.03, 0.1):
        new_seq = (1.0 - t) * seq + t * g_norm
        new_seq = np.maximum(new_seq, 0.0)          # safety
        try:
            score = evaluate_sequence(new_seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = new_seq
        except Exception:
            pass
    return best_seq, best_score


def perturb_sequence(seq, epsilon):
    """
    Perform a tiny redistribution of mass between two random indices.
    The total sum is preserved (up to numeric noise) and non‑negativity
    is enforced.
    """
