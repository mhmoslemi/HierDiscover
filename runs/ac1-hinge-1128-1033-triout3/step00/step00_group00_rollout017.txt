def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum
    # build A_ub matrix: shape (2n-1, n)
    rows = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # for i in i_min..i_max:
        # j = k - i
        # contribution f_seq[i] * b_j; in matrix row we set row[j] = f_seq[i]
        # Since we build linear inequality sum_i f_i * b_{k-i} <= rhs
        # So each b_j coefficient is sum over i where j= k - i => i = k - j => f_seq[k - j]
        # We can fill as:
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2*n - 1, rhs)
    # add nonnegativity constraints: -I * b <= 0
    A_ub = np.vstack([A_ub, -np.eye(n)])
    b_ub = np.concatenate([b_ub, np.zeros(n)])
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Propose a step‑function sequence that improves the evaluation score.

The construction is based on the optimal “square‑root’’ sequence
a_k = C(2k,k) / 4**k (central binomial coefficients divided by 4^k).
For a truncated version of this infinite sequence the self‑convolution
is almost constant, giving an evaluation ≈ π/2 ≈ 1.57, well below the
previous bound 2.0.

Starting from this sequence we run a few iterations of a linear‑program
based improvement (the same idea as the published SOTA).  In each
iteration we solve

    maximise   Σ b_j
    subject to   (f * b)_k ≤ M   for all k,
                 b_j ≥ 0

where f is the current sequence, M = max(f*f) and * denotes the discrete
convolution.  The solution b provides a direction that can increase the
total mass without raising the worst convolution value.  A convex
combination of f and the normalised b is tried with several step sizes
and the best improvement is kept.

The procedure runs for the allotted time (default 1000 s) and finally
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _solve_convolution_lp(f, rhs):
    """
    Solve the linear program:
        maximise   Σ b_j
        subject to (f * b)_k ≤ rhs   for all k,
                   b_j ≥ 0
    Returns the optimal b (numpy array) or None on failure.
    """
    n = len(f)
    # Objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build the inequality matrix A_ub @ b ≤ b_ub.
    # Each row k corresponds to Σ_i f_i * b_{k-i} ≤ rhs.
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over indices where 0 ≤ i < n and 0 ≤ k-i < n
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # fill the row: for each i, column j = k-i gets coefficient f[i]
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, rhs)

    # Non‑negativity bounds are supplied via the 'bounds' argument.
    bounds = [(0, None) for _ in range(n)]

    # Use the HiGHS solver (fast and reliable).
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights (the step‑function) that
    minimises `evaluate_sequence`.  The routine builds the square‑root
    starting sequence and then performs a few LP‑based improvement steps.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # a little safety margin

    # ------------------------------------------------------------------
    # 1. Build the classic square‑root sequence a_k = C(2k,k)/4^k
    #    (recurrence a_{k+1} = a_k * (2k+1)/(2k+2)).
    # ------------------------------------------------------------------
    n = 1742                     # length of the construction (can be changed)
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)

    # Normalise to the required sum sqrt(2n)  (the evaluation then equals max_b).
    target_sum = np.sqrt(2.0 * n)
    f = a * (target_sum / a.sum())

    # keep the best sequence seen so far
    best_seq = f.copy()
    best_score = np.max(np.convolve(best_seq, best_seq))   # = evaluate_sequence result
    print(f"Initial score (evaluation) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Iteratively improve the sequence using the LP direction.
    # ------------------------------------------------------------------
    # pre‑selected step sizes to test in each iteration
    step_candidates = (0.05, 0.1, 0.2, 0.5)
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # normalise the current vector (numerical drift may appear)
        f_sum = f.sum()
        if f_sum == 0:
            break
        f_norm = f * (target_sum / f_sum)

        # current worst convolution value M
        M = np.max(np.convolve(f_norm, f_norm))

        # --------------------------------------------------------------
        # solve the LP to obtain a direction that can increase total mass
        # while keeping the convolution bound ≤ M
        # --------------------------------------------------------------
        g = _solve_convolution_lp(f_norm, M)
        if g is None:
            # LP failed – break out and return the best sequence
            print("LP solver failed, terminating early.")
            break

        # normalise the LP solution to the same sum
        g_norm = g * (target_sum / g.sum())

        # --------------------------------------------------------------
        # try a few convex combinations (different step sizes) and keep
        # the one that yields the lowest max‑convolution (i.e. lowest score)
        # --------------------------------------------------------------
        improved = False
        best_local = None
        best_local_score = best_score

        for t in step_candidates:
            cand = (1 - t) * f_norm + t * g_norm
            cand_score = np.max(np.convolve(cand, cand))
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local = cand
                improved = True

        if improved:
            f = best_local
            best_seq = f.copy()
            best_score = best_local_score
            print(f"Iter {iteration:3d}: improved score = {best_score:.6f}")
        else:
            # No improvement – perform a tiny random perturbation to
            # escape possible stagnation.
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and f[i] > 1e-12:
                delta = 0.05 * f[i]                     # move 5 % of mass
                f[i] -= delta
                f[j] += delta
                # keep sum exactly target_sum (tiny rounding error ignored)
                # continue with the perturbed vector
                # (do not update best_seq yet – we will accept only if it helps)
        # safety break in case we have converged to machine precision
        if best_score < 1.50:
            # reached an impressively low value, stop early
            print("Achieved score below 1.50 – stopping early.")
            break

    # ------------------------------------------------------------------
    # 3. Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    print(f"Final best score = {best_score:.6f}")
    return best_seq.tolist()
