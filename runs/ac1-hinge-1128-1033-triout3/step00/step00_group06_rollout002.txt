    np.random.seed(seed)

    # Choose length

    # Or we can try the length of height_sequence_1 if known to be 5857; but we can use that as baseline.
    # Let's try n = len(height_sequence_1) if available; but to avoid slowdown we pick smaller.

        # maybe cut to n or pad
        if len(seq) > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n-len(seq)), 'constant')
    else:
        seq = np.random.rand(n)
    # Normalize sum to 1.0
    seq = seq / seq.sum()
    
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score}")
    
    # Precompute FFT size
    conv_len = 2*n - 1
    n_fft = 1 << ((conv_len).bit_length())
    
    # Main loop
    iteration = 0
    eta = 0.1  # initial step size
    while time.time() < deadline:
        # Compute convolution via FFT
        a_fft = np.fft.rfft(seq, n=n_fft)
        b = np.fft.irfft(a_fft * a_fft, n=n_fft)[:conv_len]
        max_b = b.max()
        # Compute average
        avg_b = b.mean()
        # compute error e = b - avg_b
        e = b - avg_b
        # Compute gradient: grad = 2 * correlation(e, seq)
        # correlation via FFT: conv(e, rev(seq))
        rev_seq = seq[::-1]
        e_fft = np.fft.rfft(e, n=n_fft)
        rev_seq_fft = np.fft.rfft(rev_seq, n=n_fft)
        grad_full = np.fft.irfft(e_fft * rev_seq_fft, n=n_fft)[:conv_len]
        # valid part: length n starting at index n-1? For correlation(e, seq)[i] = sum_t e[i+t] * seq[t] for i=0..n-1.
        # This is conv(e, rev(seq)) at indices (n-1) to (n-1 + n-1) = (n-1, 2n-2) inclusive, which is slice [n-1:2n-1].
        grad = 2.0 * grad_full[n-1:2*n-1]   # (n elements)
        
        # Gradient descent step
        seq_new = seq - eta * grad
        seq_new = np.clip(seq_new, 0, None)
        # Renormalize sum to 1
        sum_new = seq_new.sum()
        if sum_new < 1e-12:
            # if all zeros, reinitialize
            seq_new = np.random.rand(n)
            sum_new = seq_new.sum()
        seq_new = seq_new / sum_new
        
        # Possibly add small random jitter occasionally to escape local minima
        if iteration % 200 == 0:
            jitter = np.random.randn(n) * 0.001
            seq_new = np.clip(seq_new + jitter, 0, None)
            seq_new = seq_new / seq_new.sum()
        
        seq = seq_new
        
        # Evaluate score every certain iterations
        if iteration % 10 == 0:
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"Iter {iteration}, new best score: {best_score:.6f}")
                # adapt step size maybe
                eta = min(eta * 1.05, 0.5)  # if improving, increase step
            else:
                # if not improving, gradually reduce step
                eta = max(eta * 0.99, 1e-4)
        
        iteration += 1
    
    print(f"Best score after {iteration} iterations: {best_score}")
    return best_seq.tolist()

'''
Propose a candidate step‑function by minimizing the variance of the
self‑convolution (which drives the maximal convolution value towards its
average).  The algorithm uses projected gradient descent on the objective

    L(a) = ½ ∑_k ( (a*a)_k - μ )² ,  μ = (∑ a)²/(2n-1)

where a is forced to be non‑negative and re‑normalised after each step.
The gradient of L w.r.t. a is   ∇L = 2 * corr(e, a)   with
e = (a*a) – μ (the error of the convolution).  Correlation is computed
via FFTs for speed (≈ O(n log n)).  An adaptive step size and occasional
jitter are used to escape plateaus.  The best sequence seen during the
run is returned.
'''

import numpy as np
import time
import math

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises
    evaluate_sequence(...).  Returns the best sequence found within the
    allotted time budget (default 1000 s).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose an initial length.
    # If a previous good construction exists, reuse its length,
    # otherwise pick a random length in a reasonable range.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(height_sequence_1, dtype=float)
        n = init_seq.size
        # optionally trim or pad to keep the size manageable
        if n > 6000:
            n = 6000
            init_seq = init_seq[:n]
    else:
        n = np.random.randint(2000, 5000)
        init_seq = np.random.rand(n)

    # Normalise to unit sum (the objective is scale‑invariant)
    seq = init_seq.copy()
    seq = np.clip(seq, 0, None)          # ensure non‑negative
    seq_sum = seq.sum()
    if seq_sum < 1e-12:                  # pathological case – re‑initialise
        seq = np.random.rand(n)
        seq_sum = seq.sum()
    seq = seq / seq_sum

    # ------------------------------------------------------------------
    # Pre‑compute FFT size (next power of two >= 3·n‑2) – this length
    # is sufficient for both a*a and the correlation needed for the
    # gradient.
    # ------------------------------------------------------------------
    needed = 3 * n - 2
    n_fft = 1
    while n_fft < needed:
        n_fft <<= 1

    # ------------------------------------------------------------------
    # Helper: evaluate the current sequence (cached value kept for step
    # size adaptation).
    # ------------------------------------------------------------------
    try:
        current_score = evaluate_sequence(seq.tolist())
    except Exception:
        current_score = float('inf')
    best_seq = seq.copy()
    best_score = current_score

    print(f"[INIT] length={n}, initial score={current_score:.6f}")

    # Adaptive step size (starts moderate, will be tuned)
    eta = 0.5
    min_eta = 1e-6
    max_eta = 2.0
    last_score = current_score
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- 1. Compute self‑convolution via FFT --------------------
        a_fft = np.fft.rfft(seq, n=n_fft)
        conv_full = np.fft.irfft(a_fft * a_fft, n=n_fft)
        b = conv_full[:2 * n - 1]                  # (a * a)

        # ---- 2. Form the error w.r.t. the constant average ----------
        # The average of b is analytically equal to 1/(2n‑1) because
        # we have normalised sum(seq) == 1.
        avg_b = b.mean()                           # ≈ 1/(2n‑1)
        e = b - avg_b                               # error vector

        # ---- 3. Gradient of the variance w.r.t. a ------------------
        rev_seq = seq[::-1]                         # reversed a
        rev_fft = np.fft.rfft(rev_seq, n=n_fft)
        e_fft = np.fft.rfft(e, n=n_fft)
        corr_full = np.fft.irfft(e_fft * rev_fft, n=n_fft)

        # valid slice corresponds to indices n‑1 … 2n‑2 (inclusive)
        grad = 2.0 * corr_full[n - 1:2 * n - 1]     # same length as seq

        # ---- 4. Gradient descent step (projected onto non‑negative) ---
        candidate = seq - eta * grad
        candidate = np.clip(candidate, 0.0, None)

        # Re‑normalise to keep the scale invariant (sum == 1)
        cand_sum = candidate.sum()
        if cand_sum < 1e-12:                        # avoid division by zero
            candidate = np.random.rand(n)
            cand_sum = candidate.sum()
        candidate /= cand_sum

        # ---- 5. Occasionally add a tiny random jitter to escape
        #      possible plateaus ---------------------------------------
        if iteration % 250 == 0 and iteration > 0:
            jitter = np.random.randn(n) * 0.0005
            candidate = np.clip(candidate + jitter, 0.0, None)
            candidate /= candidate.sum()

        # ---- 6. Evaluate candidate ------------------------------------
        try:

        except Exception:
            cand_score = float('inf')

        # ---- 7. Keep the best ever seen -------------------------------
