    n = 2000
    # init: arcsine shape
    i = np.arange(n)
    a_raw = 1.0/np.sqrt((i + 0.5)*(n - i - 0.5))
    a = a_raw / a_raw.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    iter_count = 0
    # control parameters
    total_iters = 0
    while time.time() < deadline:
        # compute convolution via FFT
        L = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(a, n=L)
        B = A * A
        b_full = np.fft.irfft(B, n=L)[:2*n-1]
        # find index of max
        k_max = int(np.argmax(b_full))
        # compute gradient
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n-1))
        i_end = min(n-1, k_max)

        idx = np.arange(i_start, i_end+1)
        jdx = k_max - idx
        # accumulate gradient
        g[idx] = a[idx] + a[jdx]
        # step scale schedule
        # step_scale decays over time
        step_scale = max(0.1, 5.0 * (1 - total_iters / 2000))
        avg_g = g.mean() + 1e-12
        eta = step_scale / avg_g
        # multiplicative update
        a_new = a * np.exp(- eta * g)
        # renormalize to sum 1
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            # fallback to random perturb
            a_new = np.random.random(n)
            a_new = a_new / a_new.sum()
        else:
            a_new = a_new / a_new_sum
        # evaluate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            score = np.inf
        # accept if better
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"Iter {total_iters}: improvement, score={score:.6f}, max_b={b_full.max():.6e}")
        # optionally also accept if not worse to allow exploration?

import numpy as np, time

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Proposes a step‑function (list of non‑negative heights) that attempts to
    minimise `evaluate_sequence`.  The optimiser works on the probability simplex:
    it keeps the vector of heights normalised to sum 1 and repeatedly
    reduces the peak of the self‑convolution by a multiplicative‑weights
    update (mirrored‑descent on the sub‑gradient of the max‑convolution).
    The routine starts from a heuristic “edge‑heavy’’ distribution (the
    discrete arcsine shape) or from the global `height_sequence_1` if available,
    then performs many cheap FFT‑based iterations until the time budget is
    exhausted.  The best sequence seen is returned.
    """
    # ------------------------------------------------------------------
    # Helper: project to the probability simplex (used only as a fallback)
    # ------------------------------------------------------------------
    def normalize_to_simplex(v):
        """Return v rescaled to have non‑negative entries and sum exactly 1."""
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0:
            # uniform fallback
            return np.full_like(v, 1.0 / len(v))
        return v / s

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a small safety margin

    # If a previous construction is present, use it as a warm start.
    if 'height_sequence_1' in globals():
        init = np.array(height_sequence_1, dtype=float)
        init = np.clip(init, 0.0, None)          # protect against negative
        if init.sum() > 0:
            a = normalize_to_simplex(init)
            n = len(a)
        else:
            # fallback to a fresh random start
            n = 2000
            a = np.random.random(n)
            a = normalize_to_simplex(a)
    else:
        # default length – a few thousand gives a good trade‑off
        n = 2000
        # start from an “edge‑heavy’’ arcsine‑like shape.
        i = np.arange(n, dtype=float)
        a_raw = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a = normalize_to_simplex(a_raw)

    # keep the best sequence discovered so far
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial score: {best_score:.6f} (n={n})")

    # pre‑compute FFT length (next power of two >= 2*n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    iteration = 0
    # main optimisation loop
    while time.time() < deadline:
        # ----- compute self‑convolution (FFT, O(L log L)) -----
        A = np.fft.rfft(a, n=L)
        B = A * A
        b_full = np.fft.irfft(B, n=L)[:2 * n - 1]

        # ----- locate the worst index (largest convolution entry) -----
        k_max = int(np.argmax(b_full))
        max_val = b_full[k_max]

        # ----- gradient of the max‑convolution w.r.t. a (sub‑gradient) -----
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k_max - idx
        # contribution a_i + a_j (the same term appears twice when i==j)
        g[idx] = a[idx] + a[jdx]

        # scale the step so that the average exponent ≈ -step_scale.
        # step_scale is decreased slowly to improve stability.
        step_scale = max(0.1, 5.0 * (1.0 - iteration / 5000.0))

        eta = step_scale / avg_g

        expo = -eta * g
        expo = np.clip(expo, -50.0, 50.0)
        a_new = a * np.exp(expo)
        a_new = normalize_to_simplex(a_new)

        # ----- evaluate candidate -----
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            score = np.inf

        # ----- keep the best ever seen -----
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"Iter {iteration:04d}: new best {best_score:.6f}, max_b={max_val:.6e}")

        # decide whether to move to a_new – we always move (the method is
        # a descent on a smooth surrogate, but we keep the simple rule).
        a = a_new

            jitter = np.random.randn(n) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a = normalize_to_simplex(a)

    # final sanity check – make sure the vector is a plain python list
    return best_a.tolist()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative step‑function that minimises ``evaluate_sequence``.

    The optimiser works on the probability simplex (the heights are
    normalised to sum to 1).  Starting from a heuristic “edge‑heavy’’
    distribution – a discrete arcsine shape – or from the global
    ``height_sequence_1`` if it exists, it repeatedly reduces the peak of
    the self‑convolution using a multiplicative‑weights (mirror‑descent)
    update on the sub‑gradient of the maximum convolution entry.

    Each iteration:
        * computes the full convolution a*a via an FFT (O(L log L));
        * finds the index k_max where the convolution attains its maximum;
        * builds the sub‑gradient g_i = a_i + a_{k_max-i} (zero elsewhere);
        * updates a_i ← a_i * exp(‑η·g_i) and renormalises to sum = 1;
        * keeps the best sequence seen.

    The step size η adapts to the current average gradient magnitude and a
    slowly decreasing “step_scale’’ is used to anneal the updates.
    Random jitter is injected every 1000 iterations to avoid stagnation.

    The routine runs until the time budget is exhausted and returns the
    best list of heights discovered.
    """

    # Helper: project a vector onto the probability simplex
    # --------------------------------------------------------------
    def normalize_to_simplex(v: np.ndarray) -> np.ndarray:

        s = v.sum()

        return v / s
