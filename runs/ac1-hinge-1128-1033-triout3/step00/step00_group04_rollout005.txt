def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    while time.time() < deadline:
        # random length between 1000 and 5000
        n = np.random.randint(1500, 5000)
        # pick c between maybe 0.5 and 0.9
        c = np.random.uniform(0.4, 0.9)
        # compute r
        eps = c / np.sqrt(n)
        r = 1.0 - eps
        # generate exponential decaying values
        a = np.power(r, np.arange(n, dtype=np.float64))
        # normalize
        total = a.sum()
        if total < 1e-12:
            continue
        a = a / total
        # evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f'New best score {score:.6f} with n={n}, c={c:.4f}, r={r:.6f}')
    if best_seq is None:
        # fallback default uniform to avoid empty
        best_seq = [1.0] * 100
        best_score = evaluate_sequence(best_seq)
        print('Fallback uniform sequence')
    print('Final best score', best_score)
    return best_seq

import numpy as np
import time

def exponential_sequence(n, c):
    """
    Generates a normalized exponential decreasing sequence of length n.
    Parameter c controls the decay: the ratio of (1-r) = c/sqrt(n).
    The resulting evaluation score is approximately 2*c**2.
    """
    eps = c / np.sqrt(n)
    r = 1.0 - eps
    # generate powers r^i
    idx = np.arange(n, dtype=np.float64)
    a = np.power(r, idx)
    total = a.sum()
    if total == 0:
        return None
    a = a / total
    return a

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """

    The decay is chosen as (1 - c / sqrt(n)) with c in a range where

Exponential‐decay construction.

The evaluation function is scale‑invariant:
score = 2·n·max(conv(a,a)) / (∑a)².
If we normalise ∑a = 1 the score simplifies to 2·n·max(conv).

For a decreasing geometric sequence a_i = rⁱ (0<r<1) the central
convolution term is ≈ n·rⁿ⁻¹ and the edge term is a₀².
Choosing the decay so that r = 1 – c/√n with a modest constant c

c>0.2.  Hence the score is approximately 2·c², independent of n.
Taking c≈0.5 gives a score ≈0.5, well below the target 1.503.

import numpy as np
import time

def _make_exponential(n: int, c: float) -> np.ndarray | None:
    """
    Return a length‑n non‑negative sequence whose entries are proportional
    to r**i with r = 1 - c/√n, normalised to unit sum.
    If the sum is zero (numerical failure) returns None.
    """
    eps = c / np.sqrt(n)           # 1 - r
    r = 1.0 - eps

    idx = np.arange(n, dtype=np.float64)
    a = np.power(r, idx)
    total = a.sum()
    if total <= 0.0:
        return None
    return a / total               # normalise to sum = 1
