def local_step(seq):
   n = len(seq)
   conv = np.convolve(seq, seq)
   k = conv.argmax()
   # compute contributions:
   i_inds = np.arange(n)
   j_inds = k - i_inds
   mask = (j_inds >= 0) & (j_inds < n)
   products = np.zeros(n)
   products[mask] = seq[i_inds[mask]] * seq[j_inds[mask]]
   # find i_max with highest product
   i_max = products.argmax()
   j_max = k - i_max
   # choose source = i_max if seq[i_max] >= seq[j_max] else j_max
   src = i_max if seq[i_max] >= seq[j_max] else j_max
   # choose dest = random index not src
   dest = np.random.randint(0, n-1)
   if dest >= src:
       dest += 1  # ensure not src
   # move amount epsilon = min(seq[src] * 0.5, random factor)
   eps = min(seq[src] * 0.5, seq[src] * np.random.rand() * 0.5)
   # update:
   seq_new = seq.copy()
   seq_new[src] -= eps
   seq_new[dest] += eps
   # renorm sum:
   total = seq_new.sum()
   if total > 0:
       seq_new = seq_new / total
   return seq_new

def generate_set(n, m_target, K=2, max_trials=1000000):
    sum_counts = np.zeros(2*n-1, dtype=np.int16)
    in_set = np.zeros(n, dtype=np.bool_)
    S = []
    trials =0
    while len(S) < m_target and trials < max_trials:
        i = np.random.randint(0, n)
        if in_set[i]:
            trials+=1
            continue
        # compute list of sums to check: self i+i, plus i+j for each j in S
        # check if any sum_counts[s] + 1 > K
        ok = True
        if sum_counts[i+i] + 1 > K:
            ok=False
        else:
            for j in S:
                s = i+j
                if sum_counts[s] + 1 > K:
                    ok=False
                    break
        if ok:
            # update
            in_set[i] = True
            S.append(i)
            sum_counts[i+i] += 1
            for j in S[:-1]: # exclude newly added i
                s = i + j
                sum_counts[s] += 1
        trials += 1
    return S

"""
Propose a candidate sequence that minimizes
    2 * n * max(conv(seq, seq)) / (sum(seq) ** 2)

The idea:
* The convolution value at a lag k counts the *ordered* number of pairs
  (i, j) with i + j = k, each weighted by a_i * a_j.
* If the sequence has only a few non‑zero entries (size m) and each
  unordered sum i + j occurs at most K times, then the ordered count
  at any lag is ≤ 2·K (plus at most one diagonal term).
* With equal weights w = 1/m the maximal convolution is
        M ≈ w² · (2·K) = (2·K) / m² .
  Hence the evaluation value is approximately
        F ≈ 2·n·M = 4·n·K / m² .
  If we can make K small while m is close to the theoretical maximum
  satisfying      m·(m+1)/2 ≤ K·(2·n-1)      (all unordered pair sums ≤ K),
  then F can be driven well below 2.  For n ≈ 5400 the bound allows
  m ≈ 200 with K = 2, yielding an expected score ≈ 1.1.

The algorithm repeatedly tries to construct a set S of indices
with the above property using random permutations and a greedy
insertion test.  The best sequence found within the time budget is
returned.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function height sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found (list of non‑negative floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a small safety margin

    # length of the candidate sequence – keep the same magnitude as the
    # previous best construction.
    n = 5396

    best_seq = None
    best_score = float("inf")

    # Helper – quick estimate of the score for a given (n, m, K)
    # (used only for debugging/printing)
    def estimated_score(n_, m_, K_):
        # max ordered pair count ≤ 2*K_  (ignore the diagonal term)
        return (4.0 * n_ * K_) / (m_ * m_)

    # Try two caps for the unordered sum multiplicity.
    for K in (2, 3):
        # Maximum possible size of a set S with unordered multiplicity ≤ K:
        #    m(m+1)/2 ≤ K * (2n-1)
        max_cap = K * (2 * n - 1)
        m_limit = int((np.sqrt(1 + 8 * max_cap) - 1) // 2)

        # We aim a little below the theoretical limit to make construction easier.
        target_m = min(m_limit, 250)   # 250 is a safe ceiling for K=3

        attempts = 0
        while time.time() < deadline and attempts < 2000:
            attempts += 1

            # Random order of indices – a different order gives a different greedy set.
            perm = rng.permutation(n)

            # sum_counts[k] = number of unordered pairs (i, j) with i+j = k (i≤j)
            sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
            S = []                      # list of selected indices

            # Greedy insertion
            for i in perm:
                # self‑pair i+i must stay ≤ K
                if sum_counts[i + i] >= K:
                    continue

                # check all sums with already chosen elements
                ok = True
                for j in S:
                    if sum_counts[i + j] >= K:
                        ok = False
                        break
                if not ok:
                    continue

                # accept i
                S.append(i)
                sum_counts[i + i] += 1                     # self‑pair
                # update all new unordered sums with previous elements
                for j in S[:-1]:
                    sum_counts[i + j] += 1

                if len(S) >= target_m:
                    break

            m = len(S)
            if m < 2:
                continue          # trivial sets are useless

            # Build the sequence (equal weight on each selected index)
            seq_arr = np.zeros(n, dtype=float)
            seq_arr[S] = 1.0 / m
            seq = seq_arr.tolist()

            # Evaluate – this uses the provided security‑checked function.
            try:
                score = evaluate_sequence(seq)
            except Exception:
                continue

            # Keep the best one seen so far
            if score < best_score:
                best_score = score
                best_seq = seq
                print(
                    f"[{time.time() - start_time:5.1f}s] New best: "
                    f"score={score:.6f}, K={K}, m={m}, attempts={attempts}"
                )
                # Stop early if we already beat the target.
                if best_score < 1.503:
                    return best_seq

        # After exhausting attempts for this K we continue with the next K.
        # (If a good solution was already found we would have returned.)

    # -----------------------------------------------------------------
    # Fallback: if nothing was built (very unlikely) return a uniform list.
    # -----------------------------------------------------------------
    if best_seq is None:
        uniform_val = 1.0 / n
        best_seq = [uniform_val] * n
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence, score={best_score:.6f}")

    return best_seq
