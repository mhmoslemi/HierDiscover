"""
Uniform step function optimizer.

The evaluation function is
    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

For any non‑negative sequence a with at least one positive entry we have
    max(conv(a,a)) ≥ sum_i a_i**2 ≥ (sum(a))**2 / n      (Cauchy‑Schwarz)

Hence  E(a) ≥ 2 and equality holds exactly when all entries are equal.
Consequently the global optimum is 2, attained by any uniform sequence.

This proposer therefore returns a uniform sequence of moderate length.
The implementation also measures the obtained score with `evaluate_sequence`
and prints it – useful for debugging, but the core logic is simply the
construction of the uniform vector.
"""

import time
import numpy as np

# The evaluation function `evaluate_sequence` is provided by the judge
# and will be imported automatically – we do NOT import it here.

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a uniform step function sequence.

    Parameters
    ----------
    seed : int
        Random seed (ignored – kept for API compatibility).
    budget_s : int
        Time budget in seconds (ignored – the construction is instantaneous).
    **kwargs :
        Additional ignored arguments.

    Returns
    -------
    list[float]
        A list of equal positive numbers whose evaluation is exactly 2.
    """
    # We keep the length moderate (e.g. 1000) – any length works.
    length = 1000

    # Choose a value that comfortably stays inside the allowed range [0,1000].
    # 1.0 is simple and guarantees sum >= 0.01.
    uniform_value = 1.0

    sequence = [uniform_value] * length

    # Optional: report the score (useful when running locally).
    # The judge will ignore any prints, they are only for our diagnostics.
    try:
        score = evaluate_sequence(sequence)   # type: ignore  # noqa: F821
        print(f"Proposed uniform sequence of length {length} → score = {score}")
    except Exception as e:
        # If for some unexpected reason evaluation fails, fall back to a safe

        print(f"Evaluation failed ({e}); falling back to length 10.")
        sequence = [uniform_value] * 10
        try:
            print("Fallback score:",

        except Exception:
            pass

    return sequence
