import numpy as np
import time
import random

def project_simplex(v):
    """Project a vector v onto the probability simplex (nonnegative sum=1)."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all <=0
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    current_score = best_score

    iter_count = 0

        max_val = conv[max_idx]
        # ----- gradient of the max‑convolution term -----
        grad_max = np.zeros_like(a)
        # contribution from ordered pairs that appear in conv[max_idx]
        # derivative = 2 * a_{max_idx - i}
        idxs = np.arange(n)

"""
Simple gradient–based search for a step‑function with a flat self‑convolution.

The objective to minimise is
    E(a) = 2·n·max_k (a*a)[k] / (sum a)^2 .

Because E is homogeneous of degree 0 we may fix sum(a)=1 and only try to lower
the maximal convolution value.  The algorithm works on the simplex (non‑negative
entries summing to 1) and combines two gradients:

* ∇ max – the sub‑gradient of the current maximal convolution entry.
* ∇ MSE – the gradient of the mean‑square deviation of the whole convolution
  from its average value (which is 1/(2 n‑1) when sum(a)=1).

A projected gradient step followed by a simplex projection keeps the iterate
feasible.  Simulated‑annealing style uphill moves and occasional random jitter
help to escape local plateaus.  The routine stops as soon as a value below the
target (1.503) is found or when the time budget expires.

The returned list can be any length; here we use n≈200 which is cheap to
evaluate but still gives enough degrees of freedom to flatten the convolution.
"""

import numpy as np
import time
import random

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project the vector v onto the probability simplex (v_i ≥ 0, Σv_i = 1)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    # find the right rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    def _score(vec: np.ndarray) -> float:
        return evaluate_sequence(vec.tolist())

    # Initialise on the simplex (uniform‐Dirichlet point)
    a = np.random.dirichlet(np.ones(n))
    best_a = a.copy()
    best_score = _score(a)
    current_score = best_score

    # Hyper‑parameters (feel free to tune)
    eta = 0.12           # base step size (scaled by gradient norm)
    alpha = 0.8          # weight for the max‑convolution sub‑gradient
    beta = 0.2           # weight for the MSE flattening gradient
    temp_start = 0.10    # initial temperature for uphill moves
    temp_end = 1e-4      # final temperature

    # Pre‑compute the target average of the convolution (sum(a)=1)
    avg_conv = 1.0 / (2 * n - 1)

    iter_cnt = 0
    # ------------------------------------------------------------------ #
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) current convolution and the index of its maximum entry
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # --------------------------------------------------------------
        # 2) gradient of the max‑convolution term
        grad_max = np.zeros_like(a)
        idx = np.arange(n)
        rev = k_max - idx
        mask = (rev >= 0) & (rev < n)
        grad_max[mask] = 2.0 * a[rev[mask]]

        # --------------------------------------------------------------
        # 3) gradient of the MSE (flattening) term
        diff = conv - avg_conv                      # shape (2n‑1,)
        # Correlation of diff with a gives ∂/∂a of Σ(diff^2)
        grad_mse = 4.0 * np.correlate(diff, a, mode='valid')
        # --------------------------------------------------------------

        # Combine the two directions
        grad = alpha * grad_max + beta * grad_mse
        norm = np.linalg.norm(grad)
        if norm == 0.0:                     # degenerate case – re‑initialise
            a = np.random.dirichlet(np.ones(n))
            continue

        # Projected gradient step (step size scaled by norm)
        step = eta * grad / norm
        a_new = a - step
        a_new = _project_simplex(a_new)

        # --------------------------------------------------------------
        # 4) evaluate the candidate
        new_score = _score(a_new)

        # --------------------------------------------------------------
        # 5) acceptance rule (down‑hill always, occasional up‑hill)
        if new_score < current_score:
            a = a_new
            current_score = new_score
        else:
            # linear temperature schedule
            elapsed = time.time() - start_time
            t_frac = max(0.0, min(1.0, elapsed / (budget_s - 5.0)))
            temperature = temp_start * (1 - t_frac) + temp_end * t_frac
            prob = np.exp(-(new_score - current_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a = a_new
                current_score = new_score

        # --------------------------------------------------------------
        # 6) keep the global best
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            # early exit if we have already beaten the target
            if best_score <= 1.503:
                print(f"Target reached after {iter_cnt} iterations: score={best_score:.6f}")
                break

        # --------------------------------------------------------------
        # 7) occasional random jitter to avoid stagnation
        iter_cnt += 1
        if iter_cnt % 5000 == 0:
            jitter = np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a + jitter)

        # --------------------------------------------------------------
        # 8) slowly decay the step size
        if iter_cnt % 2000 == 0:
            eta *= 0.92

        # --------------------------------------------------------------
        # 9) status output
        if iter_cnt % 1000 == 0:
            print(f"Iter {iter_cnt:6d} | best={best_score:.6f} | cur={current_score:.6f}")

    # Return the best sequence found as a plain Python list
    return best_a.tolist()
