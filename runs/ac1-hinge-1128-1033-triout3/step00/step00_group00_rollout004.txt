def gradient_descent_step(a, eta):
    # a normalized to sum S
    c = np.convolve(a, a)
    # get max index(s)
    max_idx = np.argmax(c)
    # gradient of max c w.r.t a: g_i = 2 * a[max_idx - i] if in range else 0
    g = np.zeros_like(a)
    k = max_idx
    for i in range(len(a)):
        j = k - i
        if 0 <= j < len(a):
            g[i] = 2 * a[j]
    # take step
    a_new = a - eta * g
    # project onto simplex
    a_new = project_onto_simplex(a_new, np.sqrt(2*len(a)))
    return a_new

import numpy as np

def project_onto_simplex(v, s):
    """Project vector v onto the simplex {x >=0, sum x = s}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    # Determine rho
    rho_arr = u - cssv / ind
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # Numerical correction for sum
    w_sum = w.sum()
    if w_sum > 0:
        w *= s / w_sum
    return w

def gradient_step(a, step):
    """Perform one projected sub‑gradient step that reduces the max convolution."""
    n = a.shape[0]
    # convolution
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    # compute sub‑gradient of max_k (a*a)[k] w.r.t. a
    g = np.zeros_like(a)
    # indices i where 0<=k-i<n
    i_min = max(0, k - n + 1)
    i_max = min(k, n - 1)
    idxs = np.arange(i_min, i_max + 1)

    # take step (negative sub‑gradient direction)
    a_new = a - step * g
    # project back onto simplex with target sum sqrt(2 n)
    target_sum = np.sqrt(2.0 * n)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, conv[k]

"""
Propose a better step‑function sequence by stochastic projected
sub‑gradient descent on the max‑convolution objective.

The evaluation function is
    J(a) = 2·n·max_k (a*a)[k] / (sum a)^2 .
If we keep the sum of the coefficients equal to √(2·n), then
    J(a) = max_k (a*a)[k]               (the peak of the self‑convolution).

The algorithm:
1. Choose a length n (the length of the provided `height_sequence_1`
   if it exists, otherwise 500) and fix the target sum S = √(2·n).
2. Initialise a random point on the simplex {a_i ≥ 0, Σa_i = S},
   optionally starting from the known construction.
3. In each iteration
   * compute the convolution a*a and locate the index k where it attains
     its maximum;
   * the sub‑gradient of that maximal term w.r.t. a is
        g_i = 2·a_{k‑i}  (if 0 ≤ k‑i < n, else 0);
   * move a a little opposite to the sub‑gradient:
        a_new = a – step·g,
     then project back onto the simplex Σa = S.
   * if the new sequence yields a lower evaluation score, accept it and
     increase the step size slightly; otherwise shrink the step.
   * occasional tiny random perturbations and very rare full random
     restarts help escape plateaus.
4. Continue until the time budget (≈ 1000 s) is exhausted and return the
   best sequence discovered.

The projection onto the simplex is performed by the classic O(n log n)
algorithm of sorting and thresholding.
"""

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)

    # find the largest rho where u_i > (cssv_i / i)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)

    # renormalise numerically (optional, but guards against tiny drift)
    w_sum = w.sum()
    if w_sum > 0:
        w *= s / w_sum
    return w


# ----------------------------------------------------------------------
# One projected sub‑gradient step
# ----------------------------------------------------------------------
def gradient_step(a, step):
    """Perform a single sub‑gradient step that reduces the max convolution."""
    n = a.shape[0]

    # Full convolution (length 2n‑1)
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))               # index of the current peak

    # Sub‑gradient of (a*a)[k] w.r.t. a:
    #   ∂/∂a_i (a*a)[k] = 2·a_{k‑i}  if 0 ≤ k‑i < n, else 0
    g = np.zeros_like(a)
    i_min = max(0, k - n + 1)
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    g[idx] = 2.0 * a[k - idx]

    # Take a step opposite to the sub‑gradient and re‑project
    a_new = a - step * g
    target_sum = np.sqrt(2.0 * n)
    a_new = project_onto_simplex(a_new, target_sum)

    return a_new, conv[k]   # also return the current peak value (optional)


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=1000):
    """
    Stochastic projected sub‑gradient descent on the max‑convolution
    objective.  Returns a list of non‑negative heights (floats) that
    improves upon the current best bound.
    """
    np.random.seed(seed)
    # Give a small safety margin before the external timeout
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine problem size (use the supplied construction if available)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # existing best sequence length
    except Exception:
        n = 500                              # fallback size
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate safely (guards against unexpected errors)
    # ------------------------------------------------------------------
    def safe_eval(vec):
        try:
            # evaluate_sequence expects a plain Python list
            return evaluate_sequence(vec.tolist())
        except Exception:
            return float("inf")

    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    # Random start on the simplex
    a = np.random.rand(n)
    a = a / a.sum() * target_sum
    cur_score = safe_eval(a)
    best_a = a.copy()
    best_score = cur_score

    # Try the known construction if it is present
    try:
        baseline = np.array(height_sequence_1, dtype=float)
        baseline_score = safe_eval(baseline)
        if baseline_score < best_score:
            best_a = baseline.copy()
            best_score = baseline_score
        # Begin descent from the baseline because it is already reasonable
        a = baseline.copy()
        cur_score = baseline_score
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    step = 0.5          # initial step size
    min_step = 1e-6
    it = 0
    last_print = time.time()

    while time.time() < deadline:
        # One projected sub‑gradient move
        a_candidate, _ = gradient_step(a, step)
        cand_score = safe_eval(a_candidate)

        if cand_score < best_score - 1e-12:
            # Improvement: accept and slightly enlarge step
            best_a = a_candidate.copy()
            best_score = cand_score
            a = a_candidate
            cur_score = cand_score
            step = min(step * 1.2, 5.0)
        else:
            # No improvement: shrink step
            step = max(step * 0.5, min_step)

            # Tiny random jitter to escape shallow plateaus
            if np.random.rand() < 0.02:
                jitter = np.random.randn(n) * 0.01 * target_sum / n
                a_pert = a + jitter
                a_pert = project_onto_simplex(a_pert, target_sum)
                pert_score = safe_eval(a_pert)
                if pert_score < best_score:
                    best_a = a_pert.copy()
                    best_score = pert_score
                    a = a_pert
                    step = step * 1.1

            # Very occasional full restart
            if np.random.rand() < 0.005:
                a = np.random.rand(n)
                a = a / a.sum() * target_sum
                step = 0.5

        it += 1

        # Periodic progress report (every ~5 s)
        if time.time() - last_print > 5.0:
            print(f"Iter {it:6d} | best score {best_score:.6f} | step {step:.6f}")
            last_print = time.time()

    print("Optimisation finished – best score obtained:", best_score)
    return best_a.tolist()
