        n = len(f_seq)

        scale = sqrt_2n(n) / sum_f
        f_norm = f_seq * scale

        # compute rhs = max conv of f_norm with itself
        conv = np.convolve(f_norm, f_norm)
        rhs = conv.max()

        # LP: maximize sum b; constraints: conv(f_norm, b) <= rhs, b >=0
        # Build constraint matrix A_ub, b_ub
        # For each k in [0, 2n-2], conv_k = sum_i f_norm[i] * b_{k-i}
        # We'll vectorize.

        # Number of variables = n
        A = np.zeros((2*n-1, n), dtype=float)

            # Equivalent to A[k, j] = f_norm[i] where i = k - j
            # We'll loop j

            for i in range(n):

                if 0 <= j < n:

"""
Hybrid linear-programming and gradient search for low‑autocorrelation step functions.

The algorithm keeps a current candidate `seq`.  In each iteration it
* solves a linear program that maximises the total mass of a new
  non‑negative sequence `g` while keeping the convolution `seq * g`
  below the current peak value `M`;
* mixes `seq` and `g` with several step sizes and keeps the mixture
  that gives the largest improvement in the evaluation function;
* occasionally performs a simple sub‑gradient step that reduces the
  peak of the autocorrelation directly;
* adds random perturbations and restarts to avoid stagnation.

The loop stops after the allotted time budget and returns the best
sequence found.
"""
import numpy as np
import time
import math
import random
from scipy.optimize import linprog

def build_convolution_matrix(f_norm):
    """Return matrix A where A @ b = conv(f_norm, b) (length 2*len(f)-1)."""
    n = f_norm.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes f_norm[i] to rows i .. i+n-1
    for i, val in enumerate(f_norm):
        if val != 0.0:
            A[i:i + n, i] = val
    return A

def solve_lp_dir(seq):
    """Given a non‑negative sequence `seq`, return a direction `g`
    that maximises sum(g) while satisfying conv(seq_norm, g) <= M,
    where `seq_norm` is normalised to have sum sqrt(2n) and
    M = max conv(seq_norm, seq_norm).  Returns None on failure."""
    n = seq.shape[0]
    total = seq.sum()
    if total == 0.0:
        return None
    # normalise the current sequence to the canonical sum sqrt(2n)
    scale = math.sqrt(2.0 * n) / total
    f_norm = seq * scale
    # rhs: current peak of the autocorrelation
    rhs = np.max(np.convolve(f_norm, f_norm))
    # Build A such that A @ b = conv(f_norm, b)
    A = build_convolution_matrix(f_norm)
    # Upper‑bound constraints: conv(f_norm,b) <= rhs
    # and b >= 0  (expressed as -I @ b <= 0)
    A_ub = np.vstack([A, -np.eye(n)])
    b_ub = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])
    # Objective: maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    # Solve with HiGHS – fast and deterministic for medium sized problems
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None
    return res.x

def mix_sequences(f, g, ts):
    """Return the best mixture (1‑t)·f + t·g over the list of step sizes `ts`.
    The function also returns the corresponding evaluation value."""
    best_seq = None
    best_val = None
    for t in ts:
        cand = (1.0 - t) * f + t * g
        # keep variables inside allowed range (the evaluator also clips,
        # but we avoid unnecessary overflow)
        cand = np.clip(cand, 0.0, 1000.0)
        val = evaluate_sequence(cand.tolist())
        if (best_val is None) or (val < best_val):
            best_val = val
            best_seq = cand
    return best_seq, best_val

def gradient_step(seq, step=0.02):
    """One sub‑gradient step that tries to lower the current peak of the
    autocorrelation.  The step size `step` controls how much mass is moved."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))
    # gradient of conv_k with respect to seq_i is
    #   2 * seq_{k-i}   if 0 ≤ k-i < n, else 0
    grad = np.zeros_like(seq)
    # indices i that contribute are those where j = k_max - i lies in [0,n)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    # vectorised assignment
    i_range = np.arange(i_low, i_high + 1, dtype=int)
    j_range = k_max - i_range
    grad[i_range] = 2.0 * seq[j_range]
    # take a small step opposite to the gradient
    new_seq = seq - step * grad
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    # renormalise to keep the sum at the canonical value
    total = new_seq.sum()
    if total > 0.0:
        new_seq = new_seq * (math.sqrt(2.0 * n) / total)
    return new_seq

    total0 = seq.sum()

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score {best_score:.6f}")

    # 2. Main optimisation loop.
    # ------------------------------------------------------------------
    iteration = 0
    # step sizes that will be tried when mixing LP direction with the current sequence
    mix_ts = [0.01, 0.04, 0.08, 0.15, 0.25, 0.4, 0.6]

            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.05)

        else:                               # Gradient‑driven move
            new_seq = gradient_step(seq, step=0.015)
            val = evaluate_sequence(new_seq.tolist())
            if val < best_score:
                best_score = val
                best_seq = new_seq.copy()
                seq = new_seq
                print(f"[{iteration}] gradient step improved to {best_score:.6f}")

        # --------------------------------------------------------------
        # Exploration helpers (jitter & restarts)
        # --------------------------------------------------------------
        if iteration % 80 == 0:          # small jitter around the current best
            jitter = best_seq.copy()
            mask = np.random.rand(len(jitter)) < 0.12
            jitter[mask] = np.clip(jitter[mask] + np.random.randn(np.sum(mask)) * 0.12,
                                   0.0, 1000.0)
            total = jitter.sum()
            if total > 0.0:
                jitter = jitter * (math.sqrt(2.0 * len(jitter)) / total)
            val = evaluate_sequence(jitter.tolist())
            if val < best_score:
                best_score = val
                best_seq = jitter.copy()
                seq = jitter
                print(f"[{iteration}] jitter improved to {best_score:.6f}")

        if iteration % 250 == 0:         # full restart with a fresh random length
            new_n = np.random.randint(200, 1000)
            seq = np.random.rand(new_n).astype(np.float64)
            seq = seq * (math.sqrt(2.0 * new_n) / seq.sum())
            print(f"[{iteration}] restart with new length {new_n}")

    # ------------------------------------------------------------------
    # 3. Return the best sequence found within the time budget
    # ------------------------------------------------------------------
    print(f"Search finished after {iteration} iterations. Best score {best_score:.6f}")
    return best_seq.tolist()
