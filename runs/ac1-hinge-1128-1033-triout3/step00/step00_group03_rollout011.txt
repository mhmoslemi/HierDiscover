Pseudo:
```
def project_simplex(v):
    # v is 1-D array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    """Projects vector v onto probability simplex (non-negative, sum=1)."""
    # Following Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv/(np.arange(1, n+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence that beats the trivial uniform bound (evaluation = 2).

The construction is based on the discrete analogue of the generating function
(1‑x)^{-1/2} = Σ_{k≥0} C_k x^k  with  C_k = binom(2k,k)/4^k .
For a finite length n the first n coefficients C_k are positive, decreasing
and satisfy
    (Σ C_k)^2 ≈ (4/π)·n .
If we normalise the coefficients to sum to 1, the self‑convolution of the
infinite series would be the constant sequence 1.  After truncation the
central part of the convolution stays almost constant, giving the asymptotic
evaluation 2·n·max(conv) → π/2 ≈ 1.5708, which already improves the previous
upper bound 2.0.

Starting from this analytically motivated “square‑root” sequence we run a
few iterations of projected gradient descent on the objective

    F(a) = 2·n·max_k (a∗a)_k      (sum a_i = 1 , a_i ≥ 0)

The gradient is taken only w.r.t. the active index/indices where the maximum
is attained.  After a gradient step the vector is projected onto the probability
simplex (non‑negative, sum = 1).  Simple random perturbations are added from
time to time to escape possible plateaus.

The routine uses only NumPy and the supplied `evaluate_sequence` function.
It respects the 1000 s time budget and returns the best sequence found.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: project a vector onto the probability simplex
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex {x >= 0, sum x = 1} (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Initialise the “square‑root” sequence of length n
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """
    Return the normalised first n coefficients of (1‑x)^(-1/2):
        w_0 = 1,
        w_{k+1} = w_k * (2k+1) / (2(k+1)).
    The vector is normalised to sum to 1.
    """
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# --------------------------------------------------------------
# Gradient of the max‑convolution w.r.t. a
# --------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of max_k (a∗a)_k at the indices `max_idx`.
    For each active index k the contribution to the gradient is:
        g_i += 2 * a_{k-i}   for all i with 0 <= i < n and 0 <= k-i < n.
    The result is averaged over all active indices.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]
    if len(max_idx) > 1:
        grad /= len(max_idx)
    return grad

    Generate a sequence of non‑negative heights that minimises the provided

    1. Build the analytic “square‑root” start (coefficients of (1‑x)^(-1/2)).

    4. Keep and finally return the best sequence observed during the allotted

    """

    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    a = _sqrt_series(n)                     # already normalised, sum = 1
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial evaluation={best_score:.6f}")

    # gradient‑descent parameters

    decay = 1e-4            # linear decay of the step size
