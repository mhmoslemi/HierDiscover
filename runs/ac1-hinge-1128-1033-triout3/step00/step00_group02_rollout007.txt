"""
Propose a candidate step‑function height sequence that beats the current bound (≈2.0).

The algorithm is a simple heuristic that repeatedly
* finds the index k where the self‑convolution a*a attains its maximum,
* identifies a pair (i,j) contributing most to this peak,
* moves a small amount of mass from that pair to two locations with
  the smallest current values (while keeping the total sum constant),
* accepts the move only if the bound improves.

The convolution is computed with an FFT (real‑valued) for speed.
Occasional random restarts and a decreasing step size help to escape
local minima.  The procedure runs until the allocated time budget is
exhausted and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility: next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_self(a: np.ndarray, nfft: int) -> np.ndarray:
    """Return the linear convolution a*a using real FFT (length 2*len(a)-1)."""
    # rfft returns N/2+1 complex numbers for real input of length N
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[: 2 * len(a) - 1]
    return conv


def _bound(a: np.ndarray, nfft: int) -> float:
    """
    Compute the evaluation function for a candidate sequence.
    This mirrors evaluate_sequence but avoids the extra checks for speed.
    """
    n = len(a)
    conv = _conv_self(a, nfft)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is guaranteed > 0 by the caller
    return float(2.0 * n * max_b / (sum_a ** 2))


def _improve_step(a: np.ndarray, target_sum: float, nfft: int,
                  step_frac: float) -> tuple[np.ndarray, float]:
    """
    Perform a single improvement step that tries to lower the maximal
    convolution value.

    Parameters
    ----------
    a : np.ndarray
        Current non‑negative sequence (sum == target_sum).
    target_sum : float
        Desired L1 norm (sqrt(2*n)); used for renormalisation.
    nfft : int
        Length of the FFT (power of two >= 2*len(a)-1).
    step_frac : float
        Fraction of the smaller entry in the selected pair that will be
        transferred (controls step size).

    Returns
    -------
    new_a : np.ndarray
        Updated sequence (still non‑negative, sum≈target_sum).
    new_score : float
        The bound after the update.
    """
    n = len(a)
    conv = _conv_self(a, nfft)
    k_max = int(np.argmax(conv))          # index of the current peak
    # Find the pair (i,j) with i+j == k_max giving the largest product a_i*a_j
    # (i!=j, because diagonal contribution alone is never the sole maximum)
    best_pair = None
    best_val = -1.0
    for i in range(n):
        j = k_max - i
        if 0 <= j < n and i != j:
            prod = a[i] * a[j]
            if prod > best_val:
                best_val = prod
                best_pair = (i, j)

    # If no off‑diagonal pair exists (very degenerate case) fall back to a tiny random move.
    if best_pair is None:
        # tiny random perturbation
        i, j = np.random.randint(n, size=2)
        eps = step_frac * a[i] if a[i] > 0 else step_frac * target_sum / n
        eps = min(eps, a[i])
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        # renormalise (should already sum to target, but guard against round‑off)
        s = new_a.sum()
        if s > 0:
            new_a *= (target_sum / s)
        else:
            new_a = np.full_like(a, target_sum / n)
        return new_a, _bound(new_a, nfft)

    i0, j0 = best_pair
    # Transfer a fraction of the smaller of the two entries
    eps = step_frac * min(a[i0], a[j0])
    if eps <= 0.0:  # nothing to move – give up on this iteration
        return a, _bound(a, nfft)

    # Choose two destination indices with the smallest values.
    # We also avoid creating a new peak at k_max (i.e., ensure p+q != k_max).
    low_idx = np.argsort(a)[:4]  # a few candidates in case of conflict
    # Simple deterministic selection of two distinct low indices
    p, q = low_idx[0], low_idx[1]
    # If by chance p+q == k_max, try the next candidates
    for cand in low_idx[2:]:
        if (p + q) != k_max:
            break
        q = cand
    # As a final safeguard pick random indices until condition satisfied
    tries = 0
    while (p + q) == k_max and tries < 10:
        p, q = np.random.randint(n, size=2)
        tries += 1

    new_a = a.copy()
    # Remove mass from the offending pair
    new_a[i0] -= eps
    new_a[j0] -= eps
    # Add the same amount elsewhere
    new_a[p] += eps
    new_a[q] += eps

    # Numerical safety: clip negatives (they should not appear)
    new_a = np.maximum(new_a, 0.0)

    # Renormalise to keep the L1 norm exactly target_sum (scale invariance)
    s = new_a.sum()
    if s > 0:
        new_a *= (target_sum / s)
    else:
        new_a = np.full_like(a, target_sum / n)

    new_score = _bound(new_a, nfft)
    return new_a, new_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine runs for roughly `budget_s` seconds
    (leaving a small safety margin) and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered (list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a few seconds buffer

    # Determine the length to use.  If a previous construction is available,
    # reuse its length; otherwise fall back to a size of ~5000.
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 5000

    target_sum = np.sqrt(2.0 * n)           # the normalisation used in the SOTA code
    nfft = _next_pow2(2 * n - 1)           # FFT length (power of two)

    # --------------------------------------------------------------
    # Initialisation: start from the uniform distribution with a
    # small random perturbation to break symmetry.
    # --------------------------------------------------------------
    uniform = np.full(n, target_sum / n)
    a = uniform + 0.01 * uniform * (np.random.rand(n) - 0.5)
    a = np.maximum(a, 0.0)
    a *= (target_sum / a.sum())           # re‑normalise exactly

    best_seq = a.copy()
    best_score = _bound(best_seq, nfft)
    print(f"[init] length={n}, start bound={best_score:.6f}")

    # Main optimisation loop -------------------------------------------------
    step_frac = 0.30          # initial fraction of mass moved each step
    no_improve_iters = 0
    while time.time() < deadline:
        # Perform one directed improvement step
        a_candidate, cand_score = _improve_step(a, target_sum, nfft, step_frac)

        # Accept only if we improve (this is a simple hill‑climbing scheme)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = a_candidate.copy()
            a = a_candidate
            no_improve_iters = 0
            # gradually shrink the step size after a successful move
            step_frac = max(step_frac * 0.98, 0.01)
            print(f"[improve] new best bound={best_score:.6f}  step={step_frac:.4f}")
        else:
            # No improvement – keep the old sequence but possibly shrink step size
            a = a_candidate if np.random.rand() < 0.001 else a
            no_improve_iters += 1
            if no_improve_iters % 200 == 0:
                step_frac = max(step_frac * 0.95, 0.01)

        # Occasionally restart from a fresh random perturbation of the uniform vector
        if time.time() > deadline - 20.0 or no_improve_iters > 800:
            # final phase: try a few random restarts to escape possible plateaus
            a = uniform + 0.05 * uniform * (np.random.rand(n) - 0.5)
            a = np.maximum(a, 0.0)
            a *= (target_sum / a.sum())
            step_frac = 0.30
            no_improve_iters = 0
            print("[restart] new random start")

    print(f"[final] best bound achieved: {best_score:.6f}")
    # The evaluation function is scale‑invariant, but we keep the exact L1 norm
    # to stay safely inside the evaluator's checks.
    return best_seq.tolist()
