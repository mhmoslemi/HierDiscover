def solve_convolution_lp(f_sequence, rhs, time_limit=1.0):
    n= len(f_sequence)
    c = -np.ones(n)
    # build ub matrix
    a_rows = []
    b_vals = []
    for k in range(2*n-1):
        row = np.zeros(n)
        # i iterates over f indices 0..n-1
        # j = k - i must be in range.
        # We fill row[j] = f[i] for suitable i.
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # for each i in [i_start,i_end], j = k-i
        # set row[j] = f[i]
        # Could vectorize but simple loops okay.
        for i in range(i_start, i_end+1):
            j = k - i
            row[j] = f_sequence[i]
        a_rows.append(row)
        b_vals.append(rhs)

    # nonnegativity constraints: -I * b <= 0
    a_rows.append(-np.eye(n))
    b_vals.append(np.zeros(n))

    A_ub = np.vstack(a_rows)
    b_ub = np.hstack(b_vals)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub,
                    options={'time_limit': time_limit, 'disp': False})
    if result.success:
        return result.x
    else:
        return None

"""
Propose a better step‐function heights sequence.

The algorithm is an accelerated version of the published
iterative LP scheme:

* Keep the current vector `a` (length n) and normalise it to the
  standard sum √(2 n).  Its self‑convolution peak M = max(a*a) is the
  current value of the evaluation function (because
  evaluate_sequence = 2 n M / sum² and sum = √(2 n) ⇒ value = M).

* Solve a linear programme
        maximise   1ᵀ b
        subject to  a ⊛ b ≤ M               (convolution bound)
                    b ≥ 0
  The LP is linear because a is fixed; the feasible region consists of
  all non‑negative vectors whose convolution with a never exceeds the
  current peak M.

* Scale the LP solution to the same normalising sum and mix it with the
  current vector.  Rather than a tiny fixed step (t=0.01) we perform a
  line‑search over several mixing coefficients and keep the best
  improvement (or stay unchanged).

* Occasionally apply small random perturbations to escape possible
  stagnation and keep the best sequence seen so far.

The whole search runs for the allotted time (default 1000 s minus a
small safety margin) and finally returns the best sequence discovered.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def _max_convolution(seq: np.ndarray) -> float:
    """Return max value of the discrete convolution of seq with itself."""
    return np.max(np.convolve(seq, seq))


def _normalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale `seq` so that its ℓ¹‑norm equals `target_sum` (preserving sign)."""
    total = seq.sum()
    if total == 0:
        # avoid division by zero – fall back to uniform
        seq = np.ones_like(seq)
        total = seq.sum()
    return seq * (target_sum / total)


def _solve_convolution_lp(f: np.ndarray, bound: float,
                         time_limit: float = 1.5) -> np.ndarray | None:
    """
    Linear programme:
        maximise   Σ b_j
        subject to  (f ⊛ b)_k ≤ bound   for all k
                    b_j ≥ 0

    Returns the optimal b (as a NumPy array) or None on failure.
    """
    n = f.shape[0]
    # objective: maximise sum(b) → minimise -1ᵀb
    c = -np.ones(n)

    # build the convolution constraints A_ub @ b ≤ bound
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # for each i where 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1
        # set row[k‑i] = f[i]
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)

    # non‑negativity of b  →  -I·b ≤ 0
    rows.append(-np.eye(n))
    A_ub = np.vstack(rows)
    b_ub = np.concatenate([np.full(2 * n - 1, bound), np.zeros(n)])

    # call HiGHS via scipy.optimize.linprog with a time limit
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
            options={"time_limit": time_limit, "disp": False},
        )
    except Exception:          # pragma: no cover – safety net
        return None

    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000, **_: dict) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided `evaluate_sequence` function.  Runs for `budget_s` seconds,
    returning the best sequence found.
    """
    np.random.seed(seed)
    # give ourselves a safety margin before the hard timeout
    deadline = time.time() + budget_s - 10.0

    # -----------------------------------------------------------------
    # Initialisation – use the known construction if available,
    # otherwise fall back to a random vector of moderate length.
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.array(height_sequence_1, dtype=float)
    else:
        init_len = np.random.randint(300, 800)
        init_seq = np.random.rand(init_len)

    # ensure positivity and normalise to the standard sum √(2 n)
    n = init_seq.shape[0]
    target_sum = np.sqrt(2.0 * n)
    init_seq = np.maximum(0.0, init_seq)
    init_seq = _normalise(init_seq, target_sum)

    best_seq = init_seq.copy()
    best_score = float("inf")

    # -----------------------------------------------------------------
    # Main optimisation loop
    # -----------------------------------------------------------------
    while time.time() < deadline:
        n = init_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)

        # normalise current vector to the prescribed ℓ¹‑norm
        cur = _normalise(init_seq, target_sum)

        # current peak M = max(cur ⊛ cur)
        M = _max_convolution(cur)

        # -------------------------------------------------------------
        # 1.  Solve the LP to obtain a direction with larger total mass
        # -------------------------------------------------------------
        g_raw = _solve_convolution_lp(cur, M, time_limit=1.2)

        if g_raw is None:
            # LP failed – make a tiny random perturbation and continue
            idx = np.random.randint(n)
            cur[idx] = max(0.0, cur[idx] + np.random.randn() * 0.01)
            cur = _normalise(cur, target_sum)
            init_seq = cur
            continue

        # scale the LP solution to the same ℓ¹‑norm
        g = _normalise(g_raw, target_sum)

        # -------------------------------------------------------------
        # 2.  Line‑search over mixing coefficients t ∈ [0,1]
        # -------------------------------------------------------------
        # candidate mixing parameters (including the endpoints)
        t_vals = [0.0, 0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.75, 1.0]

        seq_candidate = cur
        score_candidate = evaluate_sequence(cur.tolist())

        for t in t_vals:
            if t == 0.0:
                cand = cur
            elif t == 1.0:
                cand = g
            else:
                cand = (1.0 - t) * cur + t * g
                # numerical drift may alter the total sum slightly
                cand = _normalise(cand, target_sum)

            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < score_candidate:
                score_candidate = cand_score
                seq_candidate = cand

        # -------------------------------------------------------------
        # 3.  Accept the best candidate and possibly record a new global best
        # -------------------------------------------------------------
        init_seq = seq_candidate

        if score_candidate < best_score:
            best_score = score_candidate
            best_seq = seq_candidate.copy()
            print(
                f"New best score {best_score:.6f}  length={n}  time={time.time():.1f}"
            )

        # -------------------------------------------------------------
        # 4.  Occasional random perturbation to avoid stagnation
        # -------------------------------------------------------------
        if np.random.rand() < 0.015:
            # perturb a random slice (5‑10 % of the entries)
            k = max(1, n // 20)
            idxs = np.random.choice(n, size=k, replace=False)
            noise = np.random.randn(k) * 0.03
            cur[idxs] = np.maximum(0.0, cur[idxs] + noise)
            cur = _normalise(cur, target_sum)
            init_seq = cur

        # -------------------------------------------------------------
        # 5.  Optional restart with a fresh random vector if no progress
        #    has been seen for a while (simple heuristic)
        # -------------------------------------------------------------
        # (implemented as a tiny probability each iteration)
        if np.random.rand() < 0.003:
            new_len = np.random.randint(300, 1000)
            new_vec = np.random.rand(new_len)
            new_target = np.sqrt(2.0 * new_len)
            init_seq = _normalise(new_vec, new_target)

    # final return: a plain Python list (the evaluation code expects a list)
    return [float(v) for v in best_seq]
