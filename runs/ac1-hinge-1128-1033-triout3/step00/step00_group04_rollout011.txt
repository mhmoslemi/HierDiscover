    conv = np.fft.ifft(np.fft.fft(seq, n*2) * np.fft.fft(seq, n*2)).real[:2*n-1]

"""
Propose a candidate step‐function sequence.

The algorithm builds a non‑negative sequence that tries to make its
self‑convolution as flat as possible.  The target continuous solution
for a uniform convolution is the density f(x)=1/(2√x) (Beta(½,1)),
which gives a discrete bound of about π/2≈1.571.  By iteratively
adjusting the sequence with a multiplicative “NMF‑style” update we
drive the convolution towards a constant (the average value).  The
updates are cheap (FFT based) and the loop runs until the time budget
expires or a good bound (<1.503) is reached.

The returned list is a viable candidate for ``evaluate_sequence``.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """
    Convolution of two real vectors using FFT.
    Returns the linear convolution (len(a)+len(b)-1 elements).
    """
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[: len(a) + len(b) - 1]


def _self_convolution(x: np.ndarray, size: int) -> np.ndarray:
    """Return x * x (linear convolution) using the cached FFT size."""
    X = np.fft.rfft(x, size)
    return np.fft.irfft(X * X, size)[: 2 * len(x) - 1]


def _score_from_seq(seq: np.ndarray, size: int) -> float:
    """Fast evaluation of the objective (without the security checks)."""
    n = len(seq)
    conv = _self_convolution(seq, size)
    max_b = conv.max()
    sum_a = seq.sum()          # should be 1 after normalisation
    # protect against tiny sums (same safeguard as evaluate_sequence)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))

        a = np.maximum(a, 0.0)                # non‑negative
        s = a.sum()
        if s == 0.0:                          # extremely unlikely, but guard
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional tiny random perturbation ----------
        # helps to escape very flat plateaus.
        if it % 100 == 0:
            noise = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()

        # ----- fast score -------------------------------------------------
        cur_score = _score_from_seq(a, fft_size)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {it:5d} – new best score {best_score:.6f}")

        # ----- early exit if we already beat the target -------------------
        if best_score < 1.503:
            print("Target reached, exiting early.")
            break

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    print(f"Finished after {it} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
