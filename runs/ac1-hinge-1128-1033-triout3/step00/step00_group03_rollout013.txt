def compute_conv_max(a):
    n = len(a)
    L = 2*n - 1
    # compute FFT with length L (maybe next pow2 for speed)
    fft_len = 1 << (L-1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:L]
    # conv may have small negative due to rounding
    return float(conv.max())

* At each iteration we locate the index `k` where the convolution attains
  its maximum.  The sub‑gradient of the maximum‑convolution with respect to
  `a` is known analytically:
        ∂max/∂a_i = 2·a_{k-i}   (when the index k-i is inside the array,
                                otherwise 0).
  This gradient points to the coordinates that contribute most to the
  current peak.
* We take a small step opposite to this sub‑gradient, renormalise the
  vector to keep the sum equal to one and clip negative entries.
  The step size is proportional to the “excess” of the current peak over
  the average convolution value – this drives the method towards a
  flatter convolution.
* If a step does not improve the score we occasionally accept it
  (simulated‑annealing style) to avoid getting stuck.  When many
  iterations pass without improvement we restart from a fresh random
  vector (or from a noisy perturbation of the best solution found so far).
* The whole process runs until the time budget is exhausted (default
  1000 s).  The best sequence discovered during the run is returned.

The search is deliberately lightweight, avoids any file I/O and uses only
NumPy and the standard library.  The implementation works entirely with
NumPy arrays, which makes the convolution (`O(n log n)`) fast enough to
perform thousands of iterations within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures)
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a*a using FFT (real input)."""
    n = a.shape[0]
    L = 2 * n - 1
    # next power‑of‑two for efficiency
    fft_len = 1 << ((L - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:L]
    # tiny negative artefacts due to floating point are harmless for max()
    return conv


def _score_fast(a: np.ndarray) -> float:
    """Fast replica of evaluate_sequence (no clipping, no IO)."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_full_fft(a)
    max_b = conv.max()
    return 2.0 * a.shape[0] * max_b / (s * s)


def _subgradient_at_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the peak of the convolution at index `peak_idx`.

    For the linear convolution `c_k = Σ_i a_i a_{k-i}`, the derivative
    with respect to a_i (when k-i lies inside the array) is
        ∂c_k/∂a_i = 2·a_{k-i}.
    The sub‑gradient for the max‑convolution uses this formula at the
    peak index.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    if peak_idx < n:
        # i = 0 … peak_idx  =>  k-i runs from peak_idx … 0
        g[: peak_idx + 1] = 2.0 * a[peak_idx::-1]
    else:
        # i = peak_idx-n+1 … n-1  =>  k-i runs from n-1 … peak_idx-n+1
        start = peak_idx - n + 1
        length = n - start
        g[start:] = 2.0 * a[:length][::-1]
    return g


def _random_perturb(a: np.ndarray, scale: float = 0.02) -> np.ndarray:
    """Add a small zero‑mean Gaussian noise and re‑normalise."""
    b = a + scale * np.random.randn(a.shape[0])
    b = np.maximum(b, 0.0)
    total = b.sum()
    if total < 1e-12:
        b = np.random.rand(a.shape[0])
        total = b.sum()
    return b / total


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of heights with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function must return before this
        deadline, otherwise the submission is discarded).
    kwargs : dict
        Additional unused keyword arguments (kept for compatibility).

    Returns
    -------
    list[float]
        The best sequence found (list of non‑negative floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0           # small safety margin

    # keep the overall best sequence
    best_seq = None
    best_score = float("inf")

    # a few candidate lengths – longer vectors give a larger search space
    length_pool = [512, 1024, 2048, 4096]

    # main outer loop – each round starts from a fresh random vector
    while time.time() < deadline:
        n = int(np.random.choice(length_pool))
        # initialise a random probability vector (sum == 1)
        a = np.random.rand(n)
        a = a / a.sum()

        # occasionally blend in the current best solution (if dimensions match)
        if best_seq is not None and len(best_seq) == n and np.random.rand() < 0.3:
            a = 0.85 * a + 0.15 * np.array(best_seq, dtype=float)
            a = a / a.sum()

        cur_score = _score_fast(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Init] length={n}  score={best_score:.6f}")

        # ----- inner optimisation loop (sub‑gradient descent) -----
        stagnation = 0
        max_inner = 4000                       # safe upper bound per restart
        for it in range(max_inner):
            if time.time() >= deadline:
                break

            # full convolution – needed for the peak index and the average value
            conv = _conv_full_fft(a)
            max_val = float(conv.max())
            peak_idx = int(np.argmax(conv))

            # sub‑gradient of the peak
            g = _subgradient_at_peak(a, peak_idx)

            # step size – proportional to how much the peak exceeds the average
            n = a.shape[0]
            avg = (a.sum() ** 2) / (2 * n - 1)   # average of the convolution
            excess = max_val - avg
            norm_sq = np.dot(g, g) + 1e-16
            eta = 0.5 * excess / norm_sq          # 0.5 is a safe damping factor
            eta = min(eta, 0.12)                 # avoid overly large steps

            # gradient step + occasional random jitter
            a_new = a - eta * g
            if np.random.rand() < 0.01:          # tiny stochastic exploration
                a_new = _random_perturb(a_new, scale=0.015)

            a_new = np.maximum(a_new, 0.0)
            total = a_new.sum()
            if total < 1e-12:                    # degenerated – re‑initialise
                a_new = np.random.rand(n)
                total = a_new.sum()
            a_new /= total

            new_score = _score_fast(a_new)

            # accept improvement; otherwise keep the old vector (with a tiny
            # chance of accepting a worsening move to escape plateaus)
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                stagnation = 0
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[Iter] len={n}  it={it}  score={best_score:.6f}")
            else:
                stagnation += 1
                if np.random.rand() < 0.001:     # occasional uphill move
                    a = a_new
                    cur_score = new_score

            # if we have made no progress for a while, break and restart
            if stagnation > 1200:
                break

        # end of one restart – loop continues with a new random length/vector

    # ------------------------------------------------------------------
    # Return the best candidate (as a plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # extremely unlikely – fall back to a trivial constant sequence
        best_seq = np.ones(1, dtype=float)
    return best_seq.tolist()
