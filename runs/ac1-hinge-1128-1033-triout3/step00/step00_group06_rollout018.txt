    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000
    target_sum = np.sqrt(2 * n)

    # Prepare best found
    best_seq = None
    best_score = float('inf')

    # function to evaluate and update best
    def try_update(seq):
        nonlocal best_seq, best_score
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"New best score: {best_score:.6f} at time {time.time():.2f}")

    # define descent run
    def run_descent(initial):
        a = initial.copy()
        it = 0
        while time.time() < deadline and it < 20000:
            # compute convolution and gradient
            c = np.convolve(a, a)
            max_val = c.max()
            k = int(c.argmax())
            # gradient of c_k
            i_arr = np.arange(n)
            j_arr = k - i_arr
            valid = (j_arr >= 0) & (j_arr < n)
            i_valid = i_arr[valid]
            j_valid = j_arr[valid]
            grad_c = np.zeros_like(a)
            same = i_valid == j_valid
            grad_c[i_valid[same]] += a[i_valid[same]]
            not_same = ~same
            grad_c[i_valid[not_same]] += 2.0 * a[j_valid[not_same]]
            # compute gradient of objective
            S = a.sum()
            if S == 0:
                # reinitialize
                a = np.random.rand(n)
                a *= target_sum / a.sum()
                continue
            factor = 2.0 * n / (S * S)
            grad_f = factor * (grad_c - (2.0 * max_val / S) * np.ones_like(a))
            # step size schedule
            lr = 0.5 / math.sqrt(it + 1)  # can tune
            a = a - lr * grad_f
            # enforce nonnegativity
            a = np.maximum(a, 0.0)
            # renormalize sum
            Snew = a.sum()
            if Snew > 0:
                a *= target_sum / Snew
            else:
                a = np.random.rand(n)
                a *= target_sum / a.sum()
            # evaluate occasionally
            if it % 100 == 0:
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.tolist()
                # optionally print progress
                #print(f"Iter {it}, score {score:.6f}")
            it += 1

    # Starting candidates
    uniform = np.full(n, target_sum / n)
    run_descent(uniform)

    # random restarts
    while time.time() < deadline:
        rand_seq = np.random.rand(n)
        rand_seq *= target_sum / rand_seq.sum()
        run_descent(rand_seq)

    # ensure something returned
    if best_seq is None:
        best_seq = uniform.tolist()
    return best_seq

    S = np.sum(seq)
    if S < 1e-12:
        return float("inf")
    conv = np.convolve(seq, seq)
    M = np.max(conv)
    return float(2.0 * n * M / (S * S))


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function runs for at most ``budget_s`` seconds and returns the
    best sequence (as a plain list of floats) discovered.
    """
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------------
    # Length of the problem – use the known construction if present
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 5857                     # fallback length
    target_sum = math.sqrt(2.0 * n)   # the normalising sum (scale‑invariant)

    deadline = time.time() + max(0.0, budget_s - 5.0)   # keep a margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------------
    # Initialise from the provided SOTA sequence (if any)
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(height_sequence_1, dtype=float)
        if seq0.sum() > 0:
            seq0 = seq0 * (target_sum / seq0.sum())
        sc0 = _compute_score(seq0)
        best_seq = seq0.copy()
        best_score = sc0
        print(f"[init] start from height_sequence_1, score = {best_score:.6f}")

    # --------------------------------------------------------------------
    # Main optimisation loop – many independent random restarts
    # --------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        # random start with the correct L1 norm
        a = np.random.rand(n)
        a = a * (target_sum / a.sum())

        local_iters = 0
        # local improvement phase (fixed number of iterations per restart)
        while time.time() < deadline and local_iters < 2000:
            conv = np.convolve(a, a)
            M = np.max(conv)
            k = int(conv.argmax())
            cur_score = _compute_score(a)

            # indices that actually contribute to the maximal convolution entry
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            idx = np.arange(low, high + 1)
            other = k - idx
            contrib = a[idx] * a[other]

            i_max = idx[np.argmax(contrib)]
            j_max = k - i_max

            # choose two target positions among the currently smallest entries
            # (they are the least likely to increase the maximum)
            smallest = np.argsort(a)[:max(2, n // 50)]   # ≈ 2 % smallest
            target_pool = np.setdiff1d(smallest, [i_max, j_max])
            if target_pool.size < 2:
                target_pool = np.setdiff1d(np.arange(n), [i_max, j_max])
            t1, t2 = np.random.choice(target_pool, size=2, replace=False)

            # transfer a modest fraction of the mass from the dominant pair
            frac = 0.05
            delta_i = min(frac * a[i_max], a[i_max])
            delta_j = min(frac * a[j_max], a[j_max])

            a_new = a.copy()
            a_new[i_max] -= delta_i
            a_new[j_max] -= delta_j
            a_new[t1] += delta_i
            a_new[t2] += delta_j

            # renormalise to keep the L1 norm exactly target_sum
            S = a_new.sum()
            if S > 0:
                a_new *= (target_sum / S)

            new_score = _compute_score(a_new)

            # accept improvement unconditionally, otherwise with a tiny chance
            if new_score < cur_score or random.random() < 0.001:
                a = a_new
                cur_score = new_score

            # keep the best overall sequence found so far
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[best] new global best = {best_score:.6f}")

            # occasional random jitter to escape plateaus
            if local_iters % 200 == 0:
                jitter = (np.random.rand(n) - 0.5) * 0.02 * a
                a = np.maximum(a + jitter, 0.0)
                a = a * (target_sum / a.sum())

            local_iters += 1

        restart += 1
        if restart % 5 == 0:
            print(f"[restart] completed {restart} restarts, best score = {best_score:.6f}")

    # --------------------------------------------------------------------
    # Fallback: a uniform sequence (guaranteed to be feasible)
    # --------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
