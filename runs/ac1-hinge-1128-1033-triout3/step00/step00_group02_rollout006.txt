def compute_gradient(a, max_indices):
    n = len(a)
    gradient = np.zeros(n)
    for k in max_indices:
        # iterate over i where 0 <= k-i < n => i in [k-n+1, k]
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)  # inclusive
        # For i in [i_start, i_end], a_{k-i} = a[k-i]

        # gradient[idx_i] += 2 * a[idx_j]

    conv = np.convolve(a, a)

"""
Simple projected sub‑gradient optimizer for the step‑function problem.
The algorithm builds a probability distribution (non‑negative, sum=1) that
makes the auto‑convolution as flat as possible.  It starts from an
U‑shaped (arcsine) initialization (or from the previously found
`height_sequence_1` if available) and repeatedly:
    * computes the convolution,
    * builds a sub‑gradient using the most extreme convolution entries,
    * takes a small projected step,
    * normalises to keep the sum equal to one,
    * occasionally performs a random mass‑transfer move to escape plateaux.
The best sequence seen during the run is returned.
"""

import time
import numpy as np

def _score_of(a: np.ndarray) -> float:
    """Return the evaluation function value for a non‑negative array a."""
    # sum of a is guaranteed >0
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = np.convolve(a, a)
    max_val = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_val / (s * s))


def _initial_arcsine(n: int) -> np.ndarray:
    """U‑shaped (arcsine‑like) start that puts more mass at the ends."""
    i = np.arange(n, dtype=float)
    # 1/√((i+½)*(n‑½‑i)) is finite for all i and symmetric.
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _gradient_top_peaks(a: np.ndarray, conv: np.ndarray, top_frac: float = 0.02):
    """
    Compute a sub‑gradient of max(conv) using a weighted sum of the
    largest convolution entries.  The weight for a peak k is
    (conv[k]‑average), guaranteeing only positive contributions.
    """
    n = a.shape[0]
    L = conv.shape[0]                     # = 2·n‑1
    avg = 1.0 / L                         # because sum(conv)=1 when sum(a)=1
    k_top = max(1, int(top_frac * L))

    # indices of the largest entries
    top_idx = np.argpartition(conv, -k_top)[-k_top:]
    weights = conv[top_idx] - avg
    # keep only strictly positive weights
    positive = weights > 0
    top_idx = top_idx[positive]
    weights = weights[positive]

    if top_idx.size == 0:                 # fall back to the single maximum
        k = int(conv.argmax())
        top_idx = np.array([k], dtype=int)
        weights = np.array([conv[k] - avg], dtype=float)

    grad = np.zeros_like(a)
    for k, w in zip(top_idx, weights):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        idx_j = k - idx_i
        # derivative of a_i a_{k‑i} w.r.t. a_i is 2·a_{k‑i}
        grad[idx_i] += 2.0 * w * a[idx_j]
    return grad


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a step‑function with a low evaluation value.

    The function runs for roughly ``budget_s`` seconds and returns the
    best sequence (list of non‑negative floats) found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(1.0, budget_s - 2.0)   # leave a tiny safety margin

    # ------------------------------------------------------------------
    # Choose a length.  Larger n gives a theoretical lower bound close to 1,
    # but also costs more per iteration.
    # ------------------------------------------------------------------
    if "n" in kwargs:
        n = int(kwargs["n"])
    else:
        n = np.random.randint(2000, 6000)   # a compromise between quality and speed

    # ------------------------------------------------------------------
    # Build the initial candidate.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=float)
        if init.shape[0] != n:
            if init.shape[0] > n:
                init = init[:n]
            else:
                # pad with zeros (they do not affect the score)
                init = np.concatenate([init, np.zeros(n - init.shape[0], dtype=float)])
        init = np.maximum(init, 0.0)
        init_sum = init.sum()
        if init_sum < 1e-12:
            init = _initial_arcsine(n)
        else:
            init = init / init_sum
    else:
        init = _initial_arcsine(n)

    a = init.copy()
    best_seq = a.copy()
    best_score = _score_of(best_seq)
    curr_score = best_score
    curr_conv = np.convolve(a, a)

    # Hyper‑parameters
    step0 = 0.8            # initial step magnitude
    top_frac = 0.02        # proportion of convolution entries used for the sub‑gradient
    stagnation = 0
    iter_cnt = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) Compute a sub‑gradient using the currently largest peaks.
        # ------------------------------------------------------------------
        grad = _gradient_top_peaks(a, curr_conv, top_frac=top_frac)

        # ------------------------------------------------------------------
        # 2) Projected sub‑gradient step with a simple back‑tracking line search.
        # ------------------------------------------------------------------
        # step size decays like 1/√t
        step = step0 / np.sqrt(iter_cnt + 1.0)

        # Try a few reductions of the step if the move does not improve.
        for _ in range(10):
            cand = a - step * grad
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-12:          # numerical safety – reinitialise with a random point
                cand = np.random.rand(n)
                cand = cand / cand.sum()
            else:
                cand = cand / s

            cand_conv = np.convolve(cand, cand)
            cand_score = 2.0 * n * cand_conv.max()   # sum(cand)=1 → denominator = 1

            if cand_score < curr_score:
                # Accept the move.
                a = cand
                curr_score = cand_score
                curr_conv = cand_conv
                stagnation = 0
                break
            else:
                step *= 0.5               # shrink and retry

        else:
            # No improvement after line‑search → treat as stagnation.
            stagnation += 1

        # ------------------------------------------------------------------
        # 3) Occasionally apply a random mass‑transfer move to escape plateaus.
        # ------------------------------------------------------------------
        if stagnation >= 30:
            i_src = np.random.randint(n)
            i_dst = np.random.randint(n)
            while i_dst == i_src:
                i_dst = np.random.randint(n)

            if a[i_src] > 1e-12:
                delta = a[i_src] * np.random.rand() * 0.4   # move up to 40 % of the source mass
                a[i_src] -= delta
                a[i_dst] += delta

"""
Simple projected sub‑gradient optimizer for the step‑function problem.

The algorithm builds a non‑negative sequence (interpreted as a probability
distribution, i.e. sum = 1) whose auto‑convolution is as flat as possible.
It starts from a U‑shaped (arcsine‑like) initialization (or from the
previously found `height_sequence_1` if that variable exists) and then
repeatedly

    1. computes the convolution,
    2. builds a sub‑gradient using the largest convolution entries,
    3. takes a small projected gradient step,
    4. normalises to keep the sum equal to one,
    5. occasionally performs a random mass‑transfer move to escape plateaus.

The best sequence seen during the run is returned.  The whole search is
constrained to the supplied time budget (default ≈ 1000 s).
"""

import time
import numpy as np

def _score_of(a: np.ndarray) -> float:
    """Return the value of `evaluate_sequence` for a non‑negative array `a`."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = np.convolve(a, a)
    max_val = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_val / (s * s))


def _initial_arcsine(n: int) -> np.ndarray:
    """U‑shaped (arcsine‑like) start that concentrates mass near the ends."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _gradient_top_peaks(a: np.ndarray, conv: np.ndarray, top_frac: float = 0.02):
    """
    Compute a sub‑gradient of `max(conv)` using a weighted sum of the
    largest convolution entries.  Each peak `k` contributes a weight
    `w = conv[k] - avg`, where `avg = 1/(len(conv))` because the
    convolution sums to one when `sum(a)=1`.  The gradient entry for
    index `i` is `2 * w * a[k-i]` whenever the term is defined.
    """
    n = a.shape[0]
    L = conv.shape[0]                 # L = 2·n‑1
    avg = 1.0 / L                     # average value of the convolution
    k_top = max(1, int(top_frac * L))

    # indices of the largest entries
    top_idx = np.argpartition(conv, -k_top)[-k_top:]
    weights = conv[top_idx] - avg
    mask = weights > 0
    top_idx = top_idx[mask]
    weights = weights[mask]

    # fallback to a single maximum if everything else failed
    if top_idx.size == 0:
        k = int(conv.argmax())
        top_idx = np.array([k], dtype=int)
        weights = np.array([conv[k] - avg], dtype=float)

    grad = np.zeros_like(a)
    for k, w in zip(top_idx, weights):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        idx_j = k - idx_i
        grad[idx_i] += 2.0 * w * a[idx_j]
    return grad


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation value.

    The function runs for roughly ``budget_s`` seconds and returns the
    best sequence (list of non‑negative floats) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(1.0, budget_s - 2.0)   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a sequence length.
    # Larger lengths give a theoretical lower bound closer to 1,
    # but also increase the per‑iteration cost.
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", np.random.randint(2000, 6000)))

    # ------------------------------------------------------------------
    # Initialise the search point.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=float)
        if init.shape[0] != n:
            if init.shape[0] > n:
                init = init[:n]
            else:
                init = np.concatenate([init, np.zeros(n - init.shape[0])])
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s < 1e-12:
            init = _initial_arcsine(n)
        else:
            init = init / s
    else:
        init = _initial_arcsine(n)

    a = init.copy()
    best_seq = a.copy()
    best_score = _score_of(best_seq)
    curr_score = best_score
    curr_conv = np.convolve(a, a)

    # Hyper‑parameters
    step0 = 0.8          # initial step scale
    top_frac = 0.02      # start with the top 2 % of convolution entries
    stagnation = 0
    it = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Sub‑gradient from the current highest peaks.
        # --------------------------------------------------------------
        grad = _gradient_top_peaks(a, curr_conv, top_frac=top_frac)

        # --------------------------------------------------------------
        # 2) Projected sub‑gradient descent with back‑tracking line search.
        # --------------------------------------------------------------
        step = step0 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(10):                     # try shrinking the step if needed
            cand = a - step * grad
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-12:
                cand = np.random.rand(n)
                cand = cand / cand.sum()
            else:
                cand = cand / s

            cand_conv = np.convolve(cand, cand)
            cand_score = 2.0 * n * cand_conv.max()   # sum(cand)=1 → denominator = 1

            if cand_score < curr_score:
                a = cand
                curr_score = cand_score
                curr_conv = cand_conv
                stagnation = 0
                improved = True
                break
            step *= 0.5

        if not improved:
            stagnation += 1

        # --------------------------------------------------------------
        # 3) Random mass‑transfer move to break possible plateaus.
        # --------------------------------------------------------------
        if stagnation >= 30:
            i_src = np.random.randint(n)
            i_dst = np.random.randint(n)
            while i_dst == i_src:
                i_dst = np.random.randint(n)

            if a[i_src] > 1e-12:
                delta = a[i_src] * np.random.rand() * 0.4   # move up to 40 % of source mass
                a[i_src] -= delta
                a[i_dst] += delta
                a = a / a.sum()
                curr_conv = np.convolve(a, a)
                curr_score = 2.0 * n * curr_conv.max()
                stagnation = 0

        # --------------------------------------------------------------
        # 4) Record the best sequence so far.
        # --------------------------------------------------------------
        if curr_score < best_score:
            best_score = curr_score
            best_seq = a.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.2f}s] iter {it:5d} – new best {best_score:.6f}")

        # --------------------------------------------------------------
        # 5) Slowly tighten the focus on the most extreme peaks.
        # --------------------------------------------------------------
        if it % 200 == 0 and it > 0:
            top_frac = max(0.005, top_frac * 0.9)   # concentrate on fewer peaks
            step0 = max(0.05, step0 * 0.95)        # smaller steps later on

        it += 1

    print(f"Search finished after {it} iterations, best score {best_score:.6f}")
    return best_seq.tolist()
