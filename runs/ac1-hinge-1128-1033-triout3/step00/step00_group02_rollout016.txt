"""
Propose a candidate step‑function by exploiting the known near‑optimal
“inverse‑square‑root” shape (f_i ≈ 1/√(i+1)) and then improving it with a
simple stochastic hill‑climbing that transfers a tiny amount of mass
between two random positions while keeping the total sum constant.
The algorithm works for any sequence length (the search uses several
lengths, from a few hundred up to a few thousand).  It never violates
the non‑negativity or scaling checks of ``evaluate_sequence`` and
returns the best sequence found before the time budget expires.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper constructors for a few sensible initial shapes
# --------------------------------------------------------------

def shape_power_law(n: int, p: float = 0.5) -> np.ndarray:
    """f_i = (i+1)^(-p)  (p=0.5 gives the classic 1/√i shape)."""
    i = np.arange(1, n + 1, dtype=float)
    return i ** (-p)


def shape_exponential(n: int, decay: float = 0.001) -> np.ndarray:
    """
    f_i = r**i with r = 1 - decay.
    ``decay`` is chosen small so that the sequence is almost flat,
    yet the convolution peak is reduced below the uniform value.
    """
    r = 1.0 - decay
    i = np.arange(n, dtype=float)
    return r ** i


def shape_arcsine(n: int) -> np.ndarray:
    """
    Symmetric arcsine‑like shape:
          f_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    This concentrates mass near the ends and mimics the continuous
    arcsine density on [0,1].
    """
    i = np.arange(n, dtype=float) + 0.5
    return 1.0 / np.sqrt(i * (n - i))


# --------------------------------------------------------------
# Core stochastic improvement routine
# --------------------------------------------------------------

def hill_climb(
    seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_no_improve: int = 2000,
) -> (np.ndarray, float):
    """
    Tiny random mass transfers between two indices.
    The total sum is preserved, the vector stays non‑negative,
    and a new candidate is accepted only if it improves the score.
    """
    n = seq.shape[0]
    cur = seq.copy()
    cur_score = best_score

    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        # pick two distinct positions
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # amount to move: a random fraction of the source entry
        # (drawn from an exponential to favour very small steps)
        delta = np.random.exponential(scale=0.01) * cur[i]
        if delta <= 0.0:
            continue
        delta = min(delta, cur[i])  # cannot move more than we own

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cur_score < best_score:
                best_score = cur_score
        else:
            no_improve += 1

    return cur, best_score

# Main entry point required by the evaluation harness
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑scoring step‑function.
    The routine tries a handful of initial shapes for several
    lengths, normalises each to unit sum (scaling does not affect
    the evaluation), then refines the best candidate with a fast
    stochastic hill‑climber.  The best sequence found before the time
    budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a little safety margin

    best_seq = None
    best_score = np.inf

    # A few lengths to explore – larger lengths give a better asymptotic
    # bound but are slower to optimise.
    lengths = [300, 600, 1200, 2400]                # total ≈ 4530 items, similar to SOTA size

    # Shape factories that will be tried for every length
    shape_funs = [
        lambda n: shape_power_law(n, p=0.5),
        lambda n: shape_power_law(n, p=0.45),       # slight deviation from ½
        lambda n: shape_exponential(n, decay=0.002),
        lambda n: shape_arcsine(n),
    ]

    for n in lengths:
        for make_shape in shape_funs:
            # ----- initialise candidate ---------------------------------
            f = make_shape(n).astype(float)

            # Normalise to unit sum (any scaling leaves the score unchanged)
            total = f.sum()
            if total == 0.0:

            f /= total

            # Evaluate the raw shape
            cur_score = evaluate_sequence(f.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.tolist()
                print(f"[init] n={n}, shape={make_shape.__name__ if hasattr(make_shape,'__name__') else 'lambda'} "
                      f"=> score={cur_score:.6f}")

            # ----- stochastic refinement ---------------------------------
            # Give each start a modest amount of time (share the budget)
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            # limit the hill‑climb to a fraction of the remaining budget
            hill_deadline = time.time() + min(remaining * 0.2, 15.0)

                best_score = refined_score

                print(f"[refine] n={n}, shape={make_shape.__name__ if hasattr(make_shape,'__name__') else 'lambda'} "

    if best_seq is None:

        best_score = evaluate_sequence(best_seq)

    print(f"Final best score: {best_score:.6f} using length {len(best_seq)}")
    return best_seq
