```
def convolve_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << (n-1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    fz = fx * fy
    z = np.fft.irfft(fz, size)[:n]
    return z
```

**Simplex projection algorithm** (Duchi et al., 2008):

```
def simplex_projection(v):
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Works for sum=1.

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Gradient‐based search for a non‑negative sequence that minimises
    evaluate_sequence.  The objective is to flatten the self‑convolution
    C = a*a, i.e. to make all convolution entries close to the average.
    Starting from random Dirichlet vectors (and optionally the known
    construction) we perform projected gradient descent on the
    squared‑deviation loss L = Σ (C_k - avg)^2, where avg = (∑a)^2/(2n-1).
    Each step is projected onto the probability simplex (∑a=1, a≥0).
    The best sequence seen before the time budget expires is returned.
    """
    import numpy as np, time, random

    # Helper: projection onto simplex (sum=1, non‑negative)
    def simplex_projection(v):
        v = np.asarray(v, dtype=float)
        # algorithm from Duchi et al. (2008)
        if np.sum(v) == 1 and np.all(v >= 0):
            return v
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(1, len(v) + 1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # all non‑positive, return uniform
            return np.full_like(v, 1.0 / len(v))
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # FFT‑based convolution (real inputs)
    def conv_fft(x, y):
        n = len(x) + len(y) - 1
        m = 1 << (n - 1).bit_length()
        fx = np.fft.rfft(x, m)
        fy = np.fft.rfft(y, m)
        fz = fx * fy
        z = np.fft.irfft(fz, m)[:n]
        return z

    # Ratio for a given vector (assumes sum = 1)
    def ratio_from_seq(a):
        n = len(a)
        c = conv_fft(a, a)
        return 2.0 * n * np.max(c)

    # ------------------------------------------------------------------
    # Setup
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 5.0   # keep a few seconds margin

    # Length to use – take from the known construction if present
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        base_seq = np.array(height_sequence_1, dtype=float)
        base_seq = np.maximum(base_seq, 0.0)
        if np.sum(base_seq) == 0:
            base_seq = np.ones(n)
    else:
        n = 2000  # fallback length
        base_seq = np.ones(n)

    # Normalise base sequence (probability simplex)
    base_seq = base_seq / np.sum(base_seq)

    # Global best
    best_seq = base_seq.copy()
    best_val = ratio_from_seq(best_seq)

    print(f"Initial length {n}, starting best ratio {best_val:.6f}")

    # Parameters for the inner gradient loop
    max_iter_per_restart = 500
    # start step size – will be adapted
    step_size = 0.5
    min_step = 1e-8
    max_step = 5.0

    # Main loop – many random restarts
    restart_counter = 0
    while time.time() < deadline:
        # ----- choose initial point for this restart --------------------
        # Alternate between the known construction and a fresh Dirichlet draw
        if restart_counter % 2 == 0:
            a = base_seq.copy()
        else:
            # Dirichlet with all parameters = 1 (uniform on simplex)
            a = np.random.dirichlet(np.ones(n))
        # Ensure exact sum = 1 after possible numeric noise
        a = a / np.sum(a)

        # inner optimisation
        cur_step = step_size
        for it in range(max_iter_per_restart):
            # time guard
            if time.time() >= deadline:
                break

            # convolution and current ratio
            conv = conv_fft(a, a)
            cur_max = np.max(conv)
            cur_ratio = 2.0 * n * cur_max

            if cur_ratio < best_val:
                best_val = cur_ratio
                best_seq = a.copy()
                print(f"Iter {restart_counter}-{it}: new best {best_val:.6f}")

            # compute gradient of squared‑deviation loss
            # average of the convolution (sum a = 1)
            avg = (np.sum(a) ** 2) / (2 * n - 1.0)
            diff = conv - avg
            grad = 4.0 * conv_fft(a[::-1], diff)[n-1:n-1+n]   # length n

            # descent direction
            direction = -grad
            norm = np.linalg.norm(direction)
            if norm == 0:
                # random direction to escape possible saddle
                direction = np.random.randn(n)
                norm = np.linalg.norm(direction)

            # propose a new point
            step = cur_step * direction / (norm + 1e-12)
            a_candidate = simplex_projection(a + step)

            # evaluate candidate ratio (no need to call evaluate_sequence)
            conv_cand = conv_fft(a_candidate, a_candidate)
            cand_ratio = 2.0 * n * np.max(conv_cand)

            # simple Armijo‑style acceptance
            if cand_ratio < cur_ratio:
                a = a_candidate
                # increase step a little (to speed up when we are going downhill)
                cur_step = min(cur_step * 1.02, max_step)
            else:
                # try a smaller step – if it still does not improve we keep the old point
                cur_step = max(cur_step * 0.5, min_step)

            # Occasionally inject a tiny random jitter to avoid stagnation
            if it % 50 == 0 and it > 0:
                jitter = np.random.randn(n) * 1e-4
                a = simplex_projection(a + jitter)

        restart_counter += 1

    # final check – evaluate with the official function (in case any clipping occurs)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Finished search, best score according to evaluate_sequence: {final_score:.6f}")
    except Exception as e:
        print("Evaluation failed:", e)

    return best_seq.tolist()

    construction (``height_sequence_1``) or a fresh Dirichlet draw, we run
    projected gradient descent on the loss

        L(a) = Σ_k (C_k – avg)²

    iteration costs only a few FFTs (≈ O(n log n)).  After each step the
    vector is projected onto the probability simplex (∑a = 1, a ≥ 0) to keep
    the scaling invariance required by ``evaluate_sequence``.
