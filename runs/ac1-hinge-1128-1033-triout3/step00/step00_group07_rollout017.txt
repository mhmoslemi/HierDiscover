"""
Propose a step‑function (non‑negative list) that minimizes
evaluate_sequence(sequence).

Algorithm
---------
* The evaluation metric is  2 * n * max(conv(a,a)) / (sum(a)**2).
  The objective is invariant to scaling, therefore we work with a
  probability vector (sum == 1).  Lower max‑convolution → lower score.

* We minimise the L2‑norm of the auto‑convolution (`sum_k (a*a)_k**2`).
  This objective is smooth and its gradient can be computed by a single
  convolution using the identity

        ∂/∂a_i  Σ_k (b_k)²  = 4 Σ_j a_j b_{i+j}
  where b = a * a (full linear convolution).

  Minimising the L2‑norm pushes the convolution toward a flat shape,
  which in turn reduces the L∞‑norm that is actually measured.

* A stochastic projected gradient descent with an annealing acceptance
  rule is used:
    - start from several diverse seeds (uniform, arcsine‑shaped,
      exponential, random Dirichlet, and the historic `height_sequence_1`);
    - at each step take a step opposite to the gradient,
      clip negatives, renormalise to a probability vector;
    - if the new point improves the true evaluation score it is kept,
      otherwise it may be accepted with a temperature‑dependent probability
      (simulated‑annealing style) to escape shallow minima;
    - the step size slowly decays and the temperature is tied to the
      remaining time budget.

* The whole search runs for the allotted time (default 1000 s) and
  returns the best sequence found.

The method needs only `numpy` and the already‑imported `evaluate_sequence`
function; it never performs I/O and respects the 1‑CPU restriction.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of a non‑negative vector onto the
    probability simplex (sum == 1).  Zero vectors are replaced by a uniform
    distribution."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size, dtype=float)
    return v / s


def _gradient_l2(a: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k ( (a*a)_k )²  with respect to a.
    Using the formula  g_i = 4 * Σ_j a_j * b_{i+j},
    where b = a * a (full linear convolution).
    """
    n = a.size
    b = np.convolve(a, a)               # length 2n‑1
    # convolution of a with the reversed b yields the needed sums:
    conv = np.convolve(a, b[::-1])      # length 3n‑2
    offset = 2 * n - 2
    # g_i = 4 * conv[offset - i]  for i = 0 .. n‑1
    g = 4.0 * conv[offset - np.arange(n)]
    return g


def _initial_shapes(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a small collection of diverse initial probability vectors of
    length ``n``.  They are used as starting points for the optimiser.
    """
    shapes = []

    # 1) uniform
    shapes.append(np.full(n, 1.0 / n))

    # 2) exponential decay
    exp_decay = np.exp(-np.arange(n) / (n / 5.0))
    shapes.append(_project_to_simplex(exp_decay))

    # 3) power‑law (U‑shaped, similar to the continuous arcsine density)
    eps = 1e-12
    idx = np.arange(n) + 0.5
    power = 1.0 / np.sqrt(idx * (n + 0.5 - idx) + eps)
    shapes.append(_project_to_simplex(power))

    # 4) random Dirichlet (alpha=1)
    random_vec = rng.random(n)
    shapes.append(_project_to_simplex(random_vec))

    # 5) if a historic solution is available, reuse it as a seed
    if 'height_sequence_1' in globals():
        old = np.array(height_sequence_1, dtype=float)
        if old.size != n:
            # truncate or pad with zeros to the required length
            if old.size > n:
                old = old[:n]
            else:
                pad = np.zeros(n - old.size)
                old = np.concatenate([old, pad])
        shapes.append(_project_to_simplex(old))

    return shapes


# ----------------------------------------------------------------------
# Core optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine runs a stochastic projected gradient
    descent with several random restarts, always respecting the 1000 s
    time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin
    best_score = float("inf")
    best_seq = None

    # parameters for the optimiser – they are deliberately generous
    # because the search is cheap for vectors of a few hundred entries.
    min_len = 200
    max_len = 800
    base_step = 0.1            # initial step size (will be decayed)
    step_decay = 0.9999        # multiplicative decay per inner iteration
    temp_start = 1.0           # starting temperature for annealing
    temp_end = 1e-3            # final temperature when time runs out

    # ------------------------------------------------------------------
    # Main outer loop – random restarts until time is exhausted.
    # ------------------------------------------------------------------
    outer_iter = 0
    while time.time() < deadline:
        outer_iter += 1

        # Choose a random length for this restart (helps diversify the
        # search space while keeping the runtime per iteration modest).
        n = rng.integers(min_len, max_len + 1)

        # Generate a handful of different starting points.
        candidates = _initial_shapes(n, rng)

        for start_vec in candidates:
            a = start_vec.copy()
            cur_score = evaluate_sequence(a.tolist())
            # keep track of the best ever found
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] "
                      f"Initial best {best_score:.6f} (len={n})")

            step = base_step
            inner_iter = 0

            # ----------------------------------------------------------------
            # Inner optimisation loop – gradient descent with annealing.
            # ----------------------------------------------------------------
            while time.time() < deadline:
                inner_iter += 1

                # 1) compute the gradient of the L2‑norm of the convolution
                grad = _gradient_l2(a)

                # 2) take a step opposite to the gradient, clip and renormalise
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                a_new = _project_to_simplex(a_new)

                # 3) evaluate the true objective (the expensive call)
                new_score = evaluate_sequence(a_new.tolist())

                # 4) accept if better, otherwise accept with a temperature‑
                #    dependent probability (simulated annealing).
                if new_score < cur_score:
                    a, cur_score = a_new, new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_seq = a_new.copy()
                        print(f"[{time.time():.1f}] "
                              f"New best {best_score:.6f} "
                              f"(len={n}, outer={outer_iter}, inner={inner_iter})")
                else:
                    # linear temperature schedule based on remaining fraction
                    frac = max(0.0, (deadline - time.time()) /
                               (budget_s - 2.0))
                    temperature = temp_end + (temp_start - temp_end) * frac
                    # Metropolis acceptance
                    if rng.random() < np.exp(-(new_score - cur_score) / temperature):
                        a, cur_score = a_new, new_score

                # 5) decay step size slowly
                step *= step_decay

                # 6) occasional restart of the inner loop if progress stalls
                if inner_iter % 1500 == 0:
                    # tiny random perturbation to escape possible plateaus
                    perturb = rng.normal(scale=0.005, size=n)
                    a = _project_to_simplex(a + perturb)
                    cur_score = evaluate_sequence(a.tolist())
                    step = base_step
                # stop inner loop early if we have already spent most of the budget
                if time.time() > deadline - 5.0:
                    break

    # ----------------------------------------------------------------------
    # Return the best sequence encountered (as a plain Python list)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # As a last resort return a simple uniform vector (never happens
        # because the loop always generates at least one candidate).
        best_seq = np.full(min_len, 1.0 / min_len)
    return best_seq.tolist()
