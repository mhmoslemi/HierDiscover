"""
Propose a step‑function (list of non‑negative heights) that minimises
evaluate_sequence().  The algorithm is a fast variant of the alternating
LP method described in the statement:

* start from a random (or the existing) sequence,
* normalise it to the standard sum = sqrt(2 n),
* compute the current maximal self‑convolution M,
* solve a linear programme that finds a non‑negative sequence g with the
  largest possible sum under the linear constraints conv(f,g) ≤ M,
* mix the current sequence with g (line‑search over a few mixing factors),
* accept the mixture only if the evaluation improves,
* otherwise apply a small random perturbation,
* repeat until the allotted wall‑clock time expires.

The LP is small (n ≤ 600) so it can be solved many times within the
1000 s budget.  Random restarts and a few different lengths are tried;
the best sequence encountered over all runs is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog


def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve the LP:
        maximise   sum(b_i)
        subject to  (f_seq * b)_k ≤ rhs   for all k = 0 … 2n‑2
                    b_i ≥ 0
    Returns the optimal b vector (as a NumPy array) or None if the solver fails.
    """
    n = f_seq.shape[0]
    # build the convolution constraint matrix A_ub @ b ≤ rhs
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)

    # each row k corresponds to Σ_i f_i * b_{k‑i} where indices outside [0,n) are ignored
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i runs over the overlapping part; the corresponding b‑index is j = k‑i
        for i in range(i_min, i_max + 1):
            j = k - i
            A[k, j] = f_seq[i]

    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=float)

    # bounds enforce b_i ≥ 0
    bounds = [(0, None)] * n
    # RHS vector
    b_ub = np.full(rows, rhs, dtype=float)

    # use HiGHS (default) – fast and reliable
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True})
    if result.success:
        return result.x
    return None


def run_optimizer_for_length(n: int, time_budget: float):
    """
    Optimise a sequence of length n within the supplied time budget (seconds).
    Returns (best_sequence_as_list, best_score).
    """
    deadline = time.time() + time_budget
    # initialise: a random start (uniform is a trivial feasible point)
    cur = np.random.rand(n).astype(float) + 1e-6
    # normalise to the conventional sum √(2n) – scaling does not affect the score
    cur = cur * (np.sqrt(2.0 * n) / cur.sum())

    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # some parameters
    mix_alphas = [0.0, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0]
    stagnation = 0          # counts iterations without improvement

    while time.time() < deadline:
        # normalise current sequence (required for the LP)
        s = cur.sum()
        if s < 1e-12:
            break
        f_norm = cur * (np.sqrt(2.0 * n) / s)

        # current maximal convolution M
        conv_f = np.convolve(f_norm, f_norm)
        M = float(conv_f.max())

        # solve the LP to obtain a direction g
        g = solve_convolution_lp(f_norm, M)

        if g is not None:
            # normalise g in the same way
            g = g * (np.sqrt(2.0 * n) / g.sum())

            # try several mixing factors and keep the best improvement
            local_best_seq = cur
            local_best_score = best_score
            for alpha in mix_alphas:
                candidate = (1.0 - alpha) * cur + alpha * g
                score = evaluate_sequence(candidate.tolist())
                if score < local_best_score:
                    local_best_score = score
                    local_best_seq = candidate

            # accept the improvement (if any)
            if local_best_score < best_score - 1e-12:
                cur = local_best_seq
                best_seq = cur.copy()
                best_score = local_best_score
                stagnation = 0
                print(f"[n={n}] new best: {best_score:.6f}")
                continue   # go to the next iteration
        # -----------------------------------------------------------------
        # if we reach here either LP failed or no improvement – do a random move
        stagnation += 1
        if stagnation >= 200:          # too many non‑improving steps → restart
            cur = np.random.rand(n).astype(float) + 1e-6
            cur *= np.sqrt(2.0 * n) / cur.sum()
            stagnation = 0
            continue

        # small random perturbation of a single coordinate (keeps the sum roughly)
        i = np.random.randint(n)
        delta = np.random.randn() * cur[i] * 0.05   # 5 % of the current value
        cur[i] = max(0.0, cur[i] + delta)
        # (optional) renormalise to avoid drift in scale
        cur *= np.sqrt(2.0 * n) / cur.sum()

    return best_seq.tolist(), best_score


def propose_candidate(seed: int = 42, budget_s: int = 990, **kwargs):
    """
    Main entry point used by the evaluation harness.
    Runs several optimisation blocks with increasing lengths and returns the
    best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)

    overall_best = None
    overall_score = float('inf')
    start_time = time.time()
    deadline = start_time + budget_s

    # lengths to explore – small enough for fast LP solves, large enough for a good bound
    lengths = [250, 400, 600]

    for n in lengths:
        # give each length a fair share of the remaining time
        now = time.time()
        if now >= deadline:
            break
        remaining = deadline - now
        # at least a few seconds per block
        block_time = max(5.0, remaining / (len(lengths) - lengths.index(n)))
        block_time = min(block_time, remaining)

        print(f"--- Optimising length {n} for {block_time:.1f}s ---")
        seq, score = run_optimizer_for_length(n, block_time)

        if score < overall_score:
            overall_best = seq
            overall_score = score
            print(f"## New overall best: {overall_score:.6f} (length {n})")

    # Fallback: if nothing succeeded we fall back to a simple uniform sequence
    if overall_best is None:
        overall_best = [1.0] * 300
        overall_score = evaluate_sequence(overall_best)

    print(f"Finished search. Best score {overall_score:.6f} with length {len(overall_best)}")
    return overall_best
