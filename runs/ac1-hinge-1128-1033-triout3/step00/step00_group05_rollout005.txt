"""
Sparse Sidon (B2) construction.

The idea: pick a set of indices whose pairwise sums are all distinct.
If all non‑zero heights have the same value w, then every entry of the
self‑convolution contains at most one product w·w = w², so

    max_b = w²   and   sum_a = k·w

The evaluation function becomes

    2·n·max_b / (sum_a)² = 2·(L+1) / k²

where L is the largest index in the Sidon set and n = L+1.
The greedy construction of a Sidon (B2) set – the Mian‑Chowla sequence –
produces L ≈ 0.5·k² for moderate k, yielding an evaluation around 1.0
and well below the target 1.503.  This routine builds such a set,
creates the corresponding step function (uniform weight on the marks,
zeros elsewhere) and returns it.
"""

import time
import numpy as np

def generate_sidon_set(k: int) -> list[int]:
    """
    Greedy construction of a B2 (Sidon) set of size k.
    Returns a list of increasing non‑negative integers such that all
    unordered sums a_i + a_j (i ≤ j) are distinct.
    """
    marks = [0]                     # start with 0
    sums = {0}                      # current set of sums (i+j)
    while len(marks) < k:
        cand = marks[-1] + 1
        while True:
            # compute all new sums that would appear with cand
            conflict = False
            new_sums = set()
            for m in marks:
                s = m + cand
                if s in sums:
                    conflict = True
                    break
                new_sums.add(s)
            # also need to check the double sum cand + cand
            if not conflict and (cand + cand) in sums:
                conflict = True
            if not conflict:
                # accept cand
                sums.update(new_sums)
                sums.add(cand + cand)
                marks.append(cand)
                break
            cand += 1
    return marks


def build_sequence_from_marks(marks: list[int], weight: float) -> list[float]:
    """Create a zero‑filled sequence with the given marks set to `weight`."""
    n = max(marks) + 1
    seq = np.zeros(n, dtype=float)
    for m in marks:
        seq[m] = weight
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function sequence with a very low evaluation score.

    The function:
      * builds a Sidon set (using the greedy Mian‑Chowla construction)
        of a chosen size,
      * assigns equal weight to each mark,
      * evaluates the sequence,
      * and returns the best one found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2  # small safety margin

    # Try a few different sizes; the smallest size that beats the target
    # is kept.  Sizes 20‑30 are plenty for the target 1.503.
    best_seq = None
    best_score = float("inf")
    target = 1.503

    for k in range(20, 31):                 # 20 ≤ k ≤ 30
        if time.time() >= deadline:
            break

        marks = generate_sidon_set(k)
        weight = 1.0 / k                     # sum of sequence will be 1.0
        seq = build_sequence_from_marks(marks, weight)

        # evaluate – we use the provided evaluate_sequence function
        score = evaluate_sequence(seq)

        print(f"k={k:2d}, marks={len(marks)}, max_idx={max(marks):4d}, "
              f"len={len(seq):4d}, score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq

        # if we already beat the target comfortably we can stop early
        if best_score < target:
            # continue a little to see if a slightly better one appears
            pass

    # If for some reason none of the constructed sequences improve,
    # fall back to a simple uniform sequence (worst case still returns
    # a valid candidate).
    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    print(f"Best score obtained: {best_score:.6f} with length {len(best_seq)}")
    return best_seq
