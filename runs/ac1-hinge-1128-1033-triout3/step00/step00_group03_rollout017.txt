def run_gradient_descent(n, time_limit):
    # set parameters
    a = np.random.rand(n)
    a = a / a.sum()
    # compute initial metric
    conv = np.convolve(a, a)
    best = a.copy()
    best_score = 2 * n * conv.max() / (a.sum()**2)  # a.sum =1 -> 2*n*max

    start = time.time()
    iteration = 0
    # parameters for schedule
    while time.time() - start < time_limit:
        # compute conv and score
        conv = np.convolve(a, a)
        max_conv = conv.max()
        cur_score = 2 * n * max_conv / (a.sum()**2)  # sum=1
        # compute weighting
        # increase beta gradually
        beta = 2.0 + 0.001 * iteration
        # compute weights
        exp_term = np.exp(beta * (conv - max_conv))
        # avoid overflow by using float64
        weights = exp_term / exp_term.sum()
        # gradient via correlation
        corr = np.correlate(weights, a, mode='full')
        grad = 2.0 * corr[n-1:n-1+n]  # shape n
        # line search for step size
        eta = 0.5 / (1 + iteration * 0.001)  # adapt step decreasing
        # try few step sizes
        accepted = False
        for trial in range(5):
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0.0)
            sum_new = a_new.sum()
            if sum_new == 0:
                # reinitialize
                a_new = np.random.rand(n)
                sum_new = a_new.sum()
            a_new = a_new / sum_new
            conv_new = np.convolve(a_new, a_new)
            score_new = 2 * n * conv_new.max() / (a_new.sum()**2)
            if score_new < cur_score:
                a = a_new
                cur_score = score_new
                accepted = True
                break
            else:
                eta *= 0.5
        if not accepted:
            # random jitter
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = a[i] * 0.01 * (np.random.rand() - 0.5)  # small perturb
                # ensure a[i] stays nonneg
                if a[i] - delta >= 0:
                    a[i] -= delta

"""
Title: Gradient–soft‑max optimizer for the step‑function evaluation metric

Summary
-------
We search for a non‑negative sequence `a` (the step heights) that minimises

    E(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

where `conv` is the linear convolution.  Because the metric is scale‑invariant
we normalise the sequence to `sum(a)=1`; then `E(a)=2·n·max(conv(a,a))`.
The optimal uniform sequence yields `E=2`.  Using an exponential
distribution as a principled starting point (the analytically optimal
member of the geometric family) gives `E≈1.80`.  We then improve this
solution by a projected gradient descent that smooths the max‑operator
with a soft‑max (controlled by a slowly increasing temperature `beta`).
Each iteration consists of:

1. Compute the full convolution and its current maximum.
2. Build soft‑max weights concentrated on the largest convolution entries.
3. Compute the gradient of the weighted sum w.r.t. `a`
   (`grad = 2 * correlate(weights, a)`).
4. Perform a back‑tracking line‑search, project onto the non‑negative simplex
   (clip, renormalise) and accept the step if the metric improves.
5. Occasionally inject a tiny random perturbation to escape plateaus.

The routine runs for the allotted time budget, keeps the best sequence
encountered, and finally returns it as a plain Python list.  If the global
`height_sequence_1` is present we fall back to its length; otherwise we use
the standard length 5808.

The algorithm is pure NumPy, needs no external solvers and respects all
security checks performed by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: metric (exactly the value returned by evaluate_sequence for sum=1)
def _metric(a: np.ndarray) -> float:
    """Return the evaluation metric for a non‑negative sequence a (sum(a) assumed >0)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Initialise a geometric (exponential) sequence.
# Parameter y solves e^y = 1 + 2 y ≈ 1.25643 which minimises the metric
# within the exponential family.
def _init_exponential(n: int, y: float = 1.25643) -> np.ndarray:
    """Return an n‑vector a_i = (1‑r) r^i / (1‑r^n) with r = exp(-y/n)."""
    r = np.exp(-y / n)
    # avoid division by zero when r is extremely close to 1
    denom = 1.0 - r ** n
    a = (1.0 - r) * (r ** np.arange(n)) / denom
    # Normalise to sum = 1 (numerical safety)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimiser: projected gradient descent with soft‑max smoothing
def _optimize(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Improve a0 until `deadline` seconds have elapsed.
    Returns (best_sequence, best_metric).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _metric(a)

    it = 0
    no_improve = 0

    while time.time() < deadline:
        # ----- current convolution and metric -----
        conv = np.convolve(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        # ----- soft‑max weights over the 2n‑1 convolution entries -----
        beta = 2.0 + 0.001 * it          # slowly increase focus on the peak
        # shift by max to avoid overflow
        exp_arg = beta * (conv - max_conv)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum

        # ----- gradient of weighted sum (∂/∂a_i of Σ w_k conv_k) -----
        # correlation gives Σ_k w_k * a_{k‑i}
        corr = np.correlate(w, a, mode='full')
        grad = 2.0 * corr[n - 1:n - 1 + n]

        # ----- back‑tracking line search -----
        eta = 0.5 / (1.0 + 0.001 * it)   # starting step, decays with time
        improved = False

        for _ in range(5):
            a_new = a - eta * grad
            # projection onto non‑negative orthant + simplex
            a_new = np.maximum(a_new, 0.0)
            total = a_new.sum()
            if total == 0.0:
                break
            a_new /= total

            new_score = _metric(a_new)
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                improved = True
                break
            else:
                eta *= 0.5   # shrink the step and retry

        # ----- if no descent, apply a tiny random jiggle -----
        if not improved:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = a[i] * 0.01 * (np.random.rand() - 0.5)
                if a[i] - delta > 0.0:
                    a[i] -= delta
                    a[j] += delta
                    a /= a.sum()

        # ----- record best observed -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # optional progress output (once every 1000 iterations)
        if (it & 1023) == 0:
            print(
                f"[Optim] it={it:5d}  cur={cur_score: .6f}  best={best_score: .6f}",
                flush=True,
            )

        # stop if we have stalled for a long stretch
        if no_improve > 5000:
            break

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑height sequence that (hopefully) beats the current bound.
    The function respects the time budget, uses an exponential start‑point,
    refines it with a soft‑max gradient descent and falls back to random
    restarts if time permits.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # leave a small safety margin

    # ------------------------------------------------------------------
    # Choose a length.  Prefer the length of the previously known candidate.
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]
    except Exception:
        n = 5808

    # ------------------------------------------------------------------
    # 1) Start from the analytically motivated exponential distribution.
    a0 = _init_exponential(n)

    # 2) Refine it with the gradient optimiser.
    best_a, best_metric = _optimize(a0, deadline)

    # ------------------------------------------------------------------
    # 3) If time remains, attempt a few random restarts (cheap cheap).
    while time.time() < deadline - 2:
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        a_opt, score_opt = _optimize(a_rand, deadline)
        if score_opt < best_metric:
            best_metric = score_opt
            best_a = a_opt.copy()
            print(f"[Restart] improved metric to {best_metric:.6f}", flush=True)

    # ------------------------------------------------------------------
    # Final conversion to a plain list and final validation via the provided
    # `evaluate_sequence` function (the official scorer will also call it).
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)   # noqa: F821  (provided externally)

    print(
        f"[Result] length={len(final_seq)}  evaluate_sequence={final_score:.6f}",
        flush=True,
    )
    return final_seq
