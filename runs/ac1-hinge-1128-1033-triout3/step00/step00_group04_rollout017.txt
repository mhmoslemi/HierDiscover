def solve_convolution_lp(f, M):
    n = len(f)
    # Build A_ub matrix of size (2*n -1) x n
    rows = []
    for k in range(2*n -1):
        row = np.zeros(n)
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        # For i in [i_min,i_max], j = k - i
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f[i]  # f_i coefficient for b_j
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2*n -1, M)
    # bounds for variables
    bounds = [(0, None)] * n
    # objective: maximize sum b_j => minimize -sum b_j
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 10.0})

    else:

"""
Propose a step‐function (non‑negative sequence) that reduces the
evaluation metric

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm starts from a uniform sequence (which gives a value of 2)
and iteratively improves it using the linear‑programming direction
described in the reference: for the current sequence f we solve

    max   Σ b_j
    s.t.  (f * b)_k ≤ M          for all k = 0 … 2n‑2
          b_j ≥ 0

where M = max_k (f * f)_k.  The obtained g is renormalised to the same
ℓ₁‑norm as f and the new sequence is taken as a convex combination

    f ← (1‑t)·f + t·g ,

with an adaptive step size t.  If the LP cannot be solved we perform a
small random perturbation.  The process repeats until the time budget
expires (≈ 1000 s).  The best sequence seen so far is returned.

The LP is solved with SciPy’s HiGHS interior‑point method; the problem
size (n ≈ 600) is chosen so that each LP solves comfortably within a
few seconds, allowing many improvement steps.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _normalize_sum(seq: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals sqrt(2·len)."""
    n = len(seq)
    target = np.sqrt(2.0 * n)
    total = seq.sum()
    if total == 0:
        # avoid division by zero – fall back to uniform
        return np.full(n, target / n)
    return seq * (target / total)


def _solve_convolution_lp(f: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Solve the LP
        max Σ b_j
        s.t.  (f * b)_k ≤ bound   (for all shifts k)
              b_j ≥ 0
    where (f * b)_k = Σ_i f_i · b_{k‑i}.
    Returns the optimal b (as a NumPy array) or None on failure.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the constraint matrix A_ub.
    # A_ub[k, j] = f_{k‑j}  if 0 ≤ k‑j < n, else 0.
    A_ub = np.zeros((rows, n))
    idx = np.arange(n)                # column indices 0 … n‑1
    for i in range(n):
        # For a fixed i the coefficient f_i appears in rows i … i+n‑1,
        # always paired with column j = row‑i.
        rows_i = i + idx               # i, i+1, …, i+n‑1
        A_ub[rows_i, idx] = f[i]       # element‑wise assignment

    b_ub = np.full(rows, bound)

    # Bounds for the variables (non‑negative)
    bounds = [(0, None)] * n

    # Objective (maximize Σ b_j) → minimize –Σ b_j.
    c = -np.ones(n)

    # Solve with the HiGHS interior‑point method (fast and robust).
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 8.0},
        )
    except Exception:                # pragma: no cover – safety net
        return None

    if res.success:
        return res.x
    return None


def _max_convolution(seq: np.ndarray) -> float:
    """Return max(conv(seq, seq))."""
    return np.max(np.convolve(seq, seq))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence with a low evaluation score.
    The routine runs for roughly ``budget_s`` seconds and returns the
    best sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.
    # If a previous construction is available we start from its length,
    # otherwise we use a moderate size that keeps the LP cheap.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:                       # pragma: no cover
        n = 600                              # a comfortable size for the LP

    # Limit n a bit to guarantee the LP stays fast.
    if n > 800:
        n = 750

    # Uniform initial sequence (this yields evaluation = 2.0)
    a = np.full(n, np.sqrt(2.0 * n) / n)

    # Track the best sequence seen so far.
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Adaptive step size for the convex combination.
    step = 0.20
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Current bound M = max_k (a * a)_k  (the quantity we try to lower)
        # --------------------------------------------------------------
        M = _max_convolution(a)

        # --------------------------------------------------------------
        # Compute a direction g by solving the linear programme.
        # --------------------------------------------------------------
        g = _solve_convolution_lp(a, M)

        if g is None:
            # LP failed – fall back to a tiny random jitter.
            idx = np.random.randint(n)
            delta = a[idx] * 0.03 * (np.random.rand() - 0.5)
            a[idx] = max(0.0, a[idx] + delta)
            a = _normalize_sum(a)
            continue

        # --------------------------------------------------------------
        # Normalise g to the same ℓ₁‑norm as a.
        # --------------------------------------------------------------
        g = _normalize_sum(g)

        improved = False

        # Try a few step sizes (the current one and slight variations).
        for tt in (step, step * 0.5, step * 1.5):
            cand = (1.0 - tt) * a + tt * g
            cand = _normalize_sum(cand)           # keep the ℓ₁‑norm exact
            cand_score = evaluate_sequence(cand.tolist())

            if cand_score < best_score - 1e-9:
                # Accept the improvement.
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                # If we improved, slightly increase the step for faster travel.
                step = min(tt * 1.2, 0.55)
                no_improve = 0
                break

        if not improved:
            # No progress → shrink the step; after many failures do a reset.
            step *= 0.5
            no_improve += 1
            if step < 5e-4 or no_improve >= 30:
                # Random restart to escape a possible plateau.
                a = np.random.rand(n)
                a = _normalize_sum(a)
                step = 0.20
                no_improve = 0

        # Occasionally print status.
        if iteration % 10 == 0:
            print(
                f"[{iteration:4d}] best={best_score:.6f}  step={step:.5f}"
            )
            # Early exit if we have already crossed the target.
            if best_score < 1.505:
                print("[info] Target reached – terminating early.")
                break

    print(f"[done] iterations={iteration}, best_score={best_score:.6f}")
    return best_seq.tolist()
