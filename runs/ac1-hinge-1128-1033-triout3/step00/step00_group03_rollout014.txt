def project_onto_simplex(v):
    # ensure non-negative and sum = 1
    # standard algorithm: sort descending, compute cumulative sum minus 1, find rho, compute lambda.
    v = np.maximum(v, 0) # but algorithm deals with negative for generic.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    # choose length n

    else:

    a /= a.sum()
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    start_time = time.time()
    deadline = start_time + budget_s - 2 # keep margin

    # parameters
    step = 0.5
    alpha = 5000.0
    # in each iteration compute
    while time.time() < deadline:
        # Compute conv and gradient
        c = np.convolve(a, a)
        # compute weights
        c_max = c.max()
        # to avoid overflow
        w = np.exp(alpha * (c - c_max))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback lower alpha
            alpha = max(alpha/2, 1.0)
            continue
        w /= w_sum
        # gradient
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # gradient descent step
        a_tmp = a - step * grad
        # project onto simplex
        a_new = project_onto_simplex(a_tmp)
        # evaluate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception as e:
            # if any error (should not)
            a_new = a.copy()
            continue

        if score < best_score:

            # optionally increase step?

            step *= 0.9  # gradually shrink step?

            a = np.random.rand(n)
            a /= a.sum()
    # after time done, return best seq as list of floats
    return best_seq.tolist()

import time

def project_onto_simplex(v):
    """
    Projects a vector onto the probability simplex:
        { w | w_i >= 0 , Σ w_i = 1 }.
    Implements the algorithm from
    Duchi et al., "Efficient Projections onto the l1‑Ball for
    Learning in High Dimensions", ICML 2008.
    """
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v.copy()
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find the rho index
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def beta_sequence(length):
    """
    Generates the un‑normalised β‑distribution coefficients
        b_k = comb(2k, k) / 4^k
    using the stable recurrence
        b_0 = 1,
        b_{k+1} = b_k * (2k+1) / (2(k+1)).
    Returns a NumPy array of length `length`.
    """
    b = np.empty(length, dtype=np.float64)
    b[0] = 1.0
    for k in range(1, length):
        b[k] = b[k - 1] * (2 * (k - 1) + 1) / (2.0 * k)
    return b
