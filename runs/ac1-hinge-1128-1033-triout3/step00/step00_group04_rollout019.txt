"""
propose_candidate
-----------------
Searches for a step‑function (non‑negative list of heights) that minimizes the
evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three phases:

1. **Parametric grid search** –  generates candidates of the form
       a_i = (i + beta) ** (-p) * sin(π * (i + 0.5) / n) ** r
   (beta ≥ 0, p ≈ 0.5, r ≥ 0).  After normalising the sum to 1 the candidate is
   evaluated.  The best candidate of this coarse search is kept.

2. **Local hill‑climbing** – starting from the best grid candidate the algorithm
   repeatedly transfers a small amount of mass between two random indices while
   preserving the total sum.  Only moves that improve the score are kept.
   The step size slowly decays; the loop runs until the time budget is almost
   exhausted.

3. **Return** – the best sequence found (as a plain Python list) is returned.
   All numbers are forced into the interval [0, 1000] by ``evaluate_sequence``,
   so no additional clipping is required.

The method is completely self‑contained, uses only NumPy and the provided
``evaluate_sequence`` function, and respects the 1000‑second time limit.
"""

import time
import numpy as np

def _normalise(arr):
    """Scale ``arr`` so that its sum is exactly 1 (preserving non‑negativity)."""
    arr = np.maximum(arr, 0.0)           # guard against negative values
    total = arr.sum()
    if total == 0.0:
        # fallback to a tiny uniform distribution
        arr[:] = 1.0 / len(arr)
        return arr
    return arr / total


def _evaluate(arr):
    """Thin wrapper around the provided evaluate_sequence."""
    # ``evaluate_sequence`` expects a plain list
    return evaluate_sequence(arr.tolist())


def propose_candidate(seed=0, budget_s=1000, **kwargs):

    the evaluation function.  The function respects the time budget and returns
    the best sequence discovered.
    """
    np.random.seed(seed)

    deadline = start_time + budget_s - 0.5           # a safety margin

    # 0.  Basic data
    # ----------------------------------------------------------------------

        best_score = _evaluate(best_seq)
        print(f"[init] Starting from height_sequence_1, score = {best_score:.6f}")
    except Exception:

        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _evaluate(best_seq)
        print(f"[init] Uniform start, score = {best_score:.6f}")

    while time.time() < deadline - 0.1:
        iteration += 1
        # Randomly pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # Propose moving a small fraction of mass from i to j
        delta = step_factor * seq[i] * (np.random.rand() - 0.5)   # signed change
        if delta == 0.0:
            continue
        # Ensure non‑negativity
        if seq[i] - delta < 0.0:
            continue

        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # No need to re‑normalise – the total sum is preserved exactly

        # Evaluate the tentative move
        new_score = _evaluate(new_seq)
        if new_score < best_score:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = new_score
            print(f"[hill] iter {iteration:5d} improved → {best_score:.6f}")

        # Gradually shrink the step size to fine‑tune
        if iteration % 2000 == 0:
            step_factor *= 0.7
            step_factor = max(step_factor, 1e-5)

    print(f"[final] Best score achieved: {best_score:.6f}")
    return best_seq.tolist()
