def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    from scipy.optimize import linprog
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # Function to evaluate safely
    def safe_eval(seq):
        try:
            return evaluate_sequence(seq)
        except Exception:
            return float('inf')

    # Choose initial length
    n = np.random.randint(300, 800)
    # initial sequence uniform with random perturbation to break symmetry
    base = np.full(n, 1.0)
    # perturb
    for _ in range(5):
        i, j = np.random.randint(n, size=2)
        eps = np.random.rand() * base[i] * 0.5
        base[i] -= eps
        base[j] += eps
    sequence = base.tolist()
    best_seq = sequence[:]
    best_score = safe_eval(best_seq)

    # add some deterministic seeds to maybe get better start
    def try_candidates(candidates):
        nonlocal best_seq, best_score
        for cand in candidates:
            s = safe_eval(cand)
            if s < best_score:
                best_score = s
                best_seq = cand[:]

    # generate some pattern seeds
    patterns = []
    # uniform
    patterns.append([1.0] * n)
    # linear ramp
    ramp = np.linspace(1.0, 2.0, n)  # scaling will be normalized by eval anyway
    patterns.append(ramp.tolist())
    # quadratic ramp
    quad = np.arange(1, n+1) ** 2
    patterns.append(quad.tolist())
    # cosine
    cos = np.sin(np.pi * np.arange(n) / (n-1))
    patterns.append(cos.tolist())
    # Gaussian
    sigma = n / 6.0
    x = np.arange(n)
    gauss = np.exp(- (x - (n-1)/2)**2 / (2 * sigma**2))
    patterns.append(gauss.tolist())
    # random binary sets (sparse)
    for _ in range(3):
        binary = np.zeros(n)
        k = int(np.sqrt(2*n))  # approximate target sum
        idx = np.random.choice(n, k, replace=False)
        binary[idx] = 1.0
        patterns.append(binary.tolist())

    try_candidates(patterns)

    # main iteration loop
    t_vals = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]
    no_improve = 0
    while time.time() < deadline:
        n = len(sequence)
        sum_seq = np.sum(sequence)
        if sum_seq < 0.01:
            # restart
            sequence = np.random.rand(np.random.randint(300, 800)).tolist()
            continue
        # normalize for LP
        factor = np.sqrt(2 * n) / sum_seq
        f_norm = [x * factor for x in sequence]
        # compute M
        conv_f = np.convolve(f_norm, f_norm)
        M = float(conv_f.max())
        # solve LP for improving direction
        g = solve_convolution_lp(f_norm, M)
        improved = False
        if g is not None:
            sum_g = np.sum(g)
            if sum_g > 0:
                # scale g to have same target sum sqrt(2n)
                g_scaled = [x * np.sqrt(2 * n) / sum_g for x in g]
                # try several mixing fractions
                for t in t_vals:
                    cand = [(1 - t) * a + t * b for a, b in zip(sequence, g_scaled)]
                    score = safe_eval(cand)
                    if score < best_score:
                        best_score = score
                        best_seq = cand[:]
                        sequence = cand[:]
                        improved = True
                        no_improve = 0
                        break  # accept first improvement
        if not improved:
            # perform small random redistribution
            i, j = np.random.choice(n, 2, replace=False)
            epsilon = np.random.rand() * 0.02 * np.mean(sequence)
            # ensure we don't go negative
            delta = min(epsilon, sequence[i])
            sequence[i] -= delta
            sequence[j] += delta
            # optional evaluate improvement
            cur_score = safe_eval(sequence)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = sequence[:]
                no_improve = 0
            else:
                no_improve += 1
        # Occasionally restart if stagnation
        if no_improve > 200:
            # random restart
            n_new = np.random.randint(300, 800)
            sequence = np.random.rand(n_new).tolist()
            no_improve = 0

    return best_seq

""" 
search algorithm for minimizing evaluate_sequence 
using LP direction + line search + gradient swaps. 
"""

import time
import numpy as np
from scipy.optimize import linprog

def _build_conv_constraints(f):
    """Build matrix A for constraints conv(f, b) <= M."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # indices j where b[j] appears
        js = np.arange(j_start, j_end + 1)
        is_ = k - js
        A[k, js] = f[is_]
    return A

def solve_convolution_lp(f, M, time_limit=5.0):
    """
    Linear program: maximise sum(b) subject to conv(f,b) <= M and b >= 0.
    Returns the optimal b (numpy array) or None if LP fails.
    """
    n = len(f)
    A = _build_conv_constraints(f)
    b = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n)        # minimise -sum(b)  <=> maximise sum(b)

    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'time_limit': time_limit})
    if res.success:
        return res.x
    return None

def _gradient_from_max(conv, seq):
    """
    Compute a simple sub‑gradient of the max‑entry of the convolution
    with respect to each entry of seq.
    """
    n = len(seq)
    max_val = conv.max()
    max_idxs = np.where(conv == max_val)[0]

    grad = np.zeros(n)
    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            grad[i] += seq[j]
    return grad

def propose_candidate(seed=42, budget_s=1000, **_):
    """
    Heuristic search for a sequence of non‑negative heights that
    minimises `evaluate_sequence`.  The algorithm mixes three ideas:

    1.  A handful of handcrafted seeds (uniform, ramp, cosine, Gaussian,
        sparse binary) to obtain a decent initial value.
    2.  An LP that, for the current (normalised) sequence,
        finds a direction `g` with a larger ℓ1‑norm while keeping the
        convolution bound unchanged.  A short line‑search over
        convex combinations (1‑t)·f + t·g is performed and a better
        candidate is kept.
    3.  If the LP does not give improvement a cheap gradient‑swap
        (move a small amount from a heap‑contributing index to a light‑
        contributing index) is applied; this breaks symmetry and lets
        the LP make progress in later rounds.

    The loop runs until approximately ``budget_s`` seconds have elapsed,
    keeping the best sequence seen.
    """
    import numpy as np
    import time

    # the evaluation routine is expected to be available in the global namespace
    safe_eval = lambda seq: evaluate_sequence(seq)

    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a tiny safety margin

    # ---- generate a few deterministic seeds -------------------------------------------------
    def _as_list(arr):
        return [float(v) for v in arr]

    # Choose a length in a comfortable range (keeps LP cheap but still allows low bound)
    n = np.random.randint(300, 800)

    candidates = []

    # uniform
    candidates.append(_as_list(np.ones(n)))
    # linear ramp
    candidates.append(_as_list(np.linspace(1.0, 2.0, n)))
    # quadratic ramp
    candidates.append(_as_list(np.arange(1, n + 1) ** 2))
    # cosine bell
    candidates.append(_as_list(np.sin(np.pi * np.arange(n) / (n - 1))))
    # Gaussian
    sigma = n / 6.0
    x = np.arange(n)
    gauss = np.exp(- (x - (n - 1) / 2) ** 2 / (2 * sigma ** 2))
    candidates.append(_as_list(gauss))
    # sparse binary sets (size ≈ sqrt(2n))
    target_k = int(np.sqrt(2 * n))
    for _ in range(3):
        vec = np.zeros(n)
        idx = np.random.choice(n, target_k, replace=False)
        vec[idx] = 1.0
        candidates.append(_as_list(vec))

    # optional: use the global SOTA sequence if it exists and has comparable length
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple)) and len(height_sequence_1) <= 2000:
            candidates.append(list(map(float, height_sequence_1)))
    except NameError:
        pass

    # keep the best of the seeds
    best_seq = None
    best_score = float('inf')
    for cand in candidates:
        sc = safe_eval(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand[:]

    # start the iterative phase from the current best seed
    sequence = best_seq[:]
    no_improve_counter = 0
    t_values = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]

    while time.time() < deadline:
        n = len(sequence)
        sum_seq = float(np.sum(sequence))
        if sum_seq < 0.01:      # degenerate – restart with a fresh random vector
            sequence = np.random.rand(np.random.randint(300, 800)).tolist()
            continue

        # normalise to the standard sum sqrt(2n) required by the LP formulation
        target_sum = np.sqrt(2.0 * n)
        factor = target_sum / sum_seq
        f_norm = (np.array(sequence) * factor).tolist()

        # current convolution bound M (also the evaluation value for the normalised vector)
        conv_f = np.convolve(f_norm, f_norm)
        M = float(conv_f.max())

        # --- 1) LP direction -------------------------------------------------
        g = solve_convolution_lp(f_norm, M, time_limit=5.0)

        improved = False
        if g is not None:
            sum_g = float(np.sum(g))
            if sum_g > 0.0:
                # scale g to have the same target ℓ1‑norm as the normalised vector
                g_scaled = (g * (target_sum / sum_g)).tolist()

                # try a short line search
                for t in t_values:
                    cand = [(1.0 - t) * a + t * b for a, b in zip(sequence, g_scaled)]
                    sc = safe_eval(cand)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand[:]
                        sequence = cand[:]
                        improved = True
                        no_improve_counter = 0
                        # small logging
                        print(f"Iter LP: t={t:.3f}, new score={sc:.6f}")
                        break   # accept the first improvement found

        # --- 2) Gradient swap (fallback) --------------------------------------
        if not improved:
            # compute the convolution of the *original* (non‑normalised) sequence
            conv_orig = np.convolve(sequence, sequence)
            grad = _gradient_from_max(conv_orig, sequence)
            # indices with largest/smallest contribution to the current max
            i_max = int(np.argmax(grad))
            i_min = int(np.argmin(grad))
            if grad[i_max] > grad[i_min]:
                # move a modest amount from the heavy index to the light one
                delta = 0.02 * sequence[i_max]   # 2 % of the larger entry
                delta = min(delta, sequence[i_max])   # keep non‑negative
                if delta > 0:
                    sequence[i_max] -= delta
                    sequence[i_min] += delta
                    sc = safe_eval(sequence)
                    if sc < best_score:
                        best_score = sc
                        best_seq = sequence[:]
                        improved = True
                        no_improve_counter = 0
                        print(f"Grad swap: i_max={i_max}, i_min={i_min}, delta={delta:.4f}, score={sc:.6f}")

        # --- 3) Random nudges if nothing moved ---------------------------------
        if not improved:
            # pick two random positions and transfer a tiny amount (keeps sum unchanged)
            i, j = np.random.choice(len(sequence), 2, replace=False)
            eps = np.random.rand() * 0.01 * max(1.0, np.mean(sequence))
            delta = min(eps, sequence[i])
            sequence[i] -= delta
            sequence[j] += delta
            sc = safe_eval(sequence)
            if sc < best_score:
                best_score = sc
                best_seq = sequence[:]
                no_improve_counter = 0
                print(f"Nudge: i={i}, j={j}, delta={delta:.6f}, score={sc:.6f}")
            else:
                no_improve_counter += 1
        else:
            # even when we improved, count a success as resetting the stagnation counter
            pass

        # occasional restart when we have been idle for a while
        if no_improve_counter > 250:
            n_new = np.random.randint(300, 800)
            sequence = np.random.rand(n_new).tolist()
            no_improve_counter = 0
            print(f"Restarting with new length {n_new}")

    # final sanity: ensure no negative numbers survive (evaluate_sequence already clamps)
    best_seq = [max(0.0, float(v)) for v in best_seq]
    return best_seq

    no_improve = 0
    t_vals = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]

    while time.time() < deadline:
        n = len(sequence)
        sum_seq = float(np.sum(sequence))
        if sum_seq < 0.01:                     # degenerate – restart randomly
            sequence = np.random.rand(np.random.randint(300, 800)).tolist()
            continue

        # normalise to the standard ℓ₁‑norm √(2n) required by the LP
        target_sum = np.sqrt(2.0 * n)
        factor = target_sum / sum_seq
        f_norm = (np.array(sequence) * factor).tolist()

        # current convolution bound M (also the evaluation value for the normalised vector)
        conv_f = np.convolve(f_norm, f_norm)
        M = float(conv_f.max())

        # --------------------------------------------------------------
        # LP direction
        # --------------------------------------------------------------
        g = solve_convolution_lp(f_norm, M, time_limit=5.0)

        improved = False
        if g is not None:
            sum_g = float(np.sum(g))
            if sum_g > 0.0:
                # scale g to have the same ℓ₁‑norm as the normalised vector
                g_scaled = (g * (target_sum / sum_g)).tolist()

                # short line‑search over mixing coefficients
                for t in t_vals:
                    cand = [(1.0 - t) * a + t * b for a, b in zip(sequence, g_scaled)]
                    sc = safe_eval(cand)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand[:]
                        sequence = cand[:]
                        improved = True
                        no_improve = 0
                        print(
                            f"[LP] t={t:.3f} → score={sc:.6f} (best={best_score:.6f})"
                        )
                        break

        # --------------------------------------------------------------
        # Gradient swap (fallback)
        # --------------------------------------------------------------
        if not improved:
            conv_orig = np.convolve(sequence, sequence)
            grad = _gradient_from_max(conv_orig, sequence)
            i_max = int(np.argmax(grad))
            i_min = int(np.argmin(grad))

            if grad[i_max] > grad[i_min]:
                delta = 0.02 * sequence[i_max]          # move ~2 % of the large entry
                delta = min(delta, sequence[i_max])     # keep non‑negative
                if delta > 0:
                    sequence[i_max] -= delta
                    sequence[i_min] += delta
                    sc = safe_eval(sequence)
                    if sc < best_score:
                        best_score = sc
                        best_seq = sequence[:]
                        improved = True
                        no_improve = 0
                        print(
                            f"[Grad] i_max={i_max}, i_min={i_min}, Δ={delta:.4f} → {sc:.6f}"
                        )

        # --------------------------------------------------------------
        # Small random nudge if everything else failed
        # --------------------------------------------------------------
        if not improved:
            i, j = np.random.choice(len(sequence), 2, replace=False)
            eps = np.random.rand() * 0.01 * max(1.0, np.mean(sequence))
            delta = min(eps, sequence[i])
            sequence[i] -= delta
            sequence[j] += delta
            sc = safe_eval(sequence)
            if sc < best_score:
                best_score = sc
                best_seq = sequence[:]
                no_improve = 0
                print(f"[Nudge] i={i}, j={j}, Δ={delta:.6f} → {sc:.6f}")
            else:
                no_improve += 1

        # --------------------------------------------------------------
        # Stagnation handling – occasional restart
        # --------------------------------------------------------------
        if no_improve > 250:
            n_new = np.random.randint(300, 800)
            sequence = np.random.rand(n_new).tolist()
            no_improve = 0
            print(f"[Restart] new length {n_new}")

    # final sanity: remove accidental negatives (evaluation already clamps)
    best_seq = [max(0.0, float(v)) for v in best_seq]
    return best_seq
