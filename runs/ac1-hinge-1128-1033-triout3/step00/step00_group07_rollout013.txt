def simplex_projection(v):
    # v is a numpy array
    n = len(v)
    u = np.sort(v)[::-1]

    return w

"""
Propose a candidate step‑function (list of non‑negative heights).

The objective to minimise is

    M = 2·n·max(conv(a,a)) / (sum(a)**2)

Because the evaluation is homogeneous, we can normalise the
distribution to sum to 1 and minimise the maximum entry of the
self‑convolution.  This is exactly the problem

    minimise   max_k  (p * p)_k   subject to   p_i ≥ 0 ,  Σ p_i = 1

where * denotes discrete convolution.  The code performs a
projected gradient descent on a smooth approximation of the
max‑operator (log‑sum‑exp).  The gradient of the smoothed
objective can be computed by a single correlation (implemented
with `np.convolve`).  After each step the vector is projected
onto the probability simplex, guaranteeing feasibility.
Occasional random perturbations help to escape shallow local
minima.  The algorithm runs until the allotted time expires and
returns the best distribution found – a list of floats that can
be used directly with `evaluate_sequence`.
"""

import numpy as np
import time


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0 , Σ x_i = 1}.
    Reference: Duchi et al., “Efficient Projections onto the l1‑Ball
    for Learning in High Dimensions”, ICML 2008.
    """
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)                # 1‑based indices
    # Find the largest ρ satisfying u_ρ > (cssv_ρ / ρ)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                         # all entries are zero or negative
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    """

    the evaluation function.  The routine runs a projected gradient
    descent on a smoothed version of the objective, using the
    probability simplex as feasible set.  It returns the best
    sequence discovered before the time budget expires.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop a few seconds
        before this limit to avoid termination).
    kwargs : dict
        Ignored – kept for compatibility with the surrounding framework.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size (use the length of the known construction
    #    if it exists, otherwise fall back to a reasonable default).
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)               # type: ignore[name‑defined]
    except Exception:
        n = 1153                                   # default length used by the benchmark

    # 2. Initialise a probability vector p (sum(p) == 1).
    #    Occasionally we start from the existing construction.

    if rng.random() < 0.3 and 'height_sequence_1' in globals():
        # use the provided sequence as a starting point (normalise it)
        seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name‑defined]

    best_p = p.copy()

    best_score = 2.0 * n * np.max(np.convolve(p, p))

    # --------------------------------------------------------------
    # 3. Parameters for the optimisation
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # leave a small safety margin
    iteration = 0

    alpha_mul = 1.003                         # slowly tighten the approximation

    while time.time() < deadline:

        w = exp_z / exp_z.sum()                # soft‑max weights, sum to 1
