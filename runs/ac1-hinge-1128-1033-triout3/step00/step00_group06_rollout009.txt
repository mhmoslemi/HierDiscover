        init_seq += 1.0
    init_seq = init_seq / init_seq.sum()
    # choose start: 50% chance from init_seq, else random
    if rng.random() < 0.5:

        cur = cur / cur.sum()

"""
Simulated‑annealing optimizer for the step‑function evaluation problem.

The objective (for a probability vector p with sum = 1) is

    f(p) = 2 * n * max_k (p * p)_k

where “*” denotes the discrete convolution.
Because the provided `evaluate_sequence` is scale‑invariant,
any vector with the same relative shape yields the same score.

The algorithm:
* Start from the known construction (`height_sequence_1`) or from a random
  probability vector of the same length.
* Repeatedly move a small amount of probability mass from a random index i
  to another random index j while keeping the total mass equal to 1.
* The convolution of the candidate vector is obtained with a single FFT
  (zero‑padded to the next power of two) and the maximum entry is measured.
* Moves that improve the score are always accepted; worse moves are accepted
  with a Metropolis probability that gradually cools down.
* The step size (`move_scale`) adapts to the observed acceptance ratio.
* Periodically the current vector is renormalised to avoid drift due to
  rounding errors and the best vector seen so far is stored.

The routine runs for the allotted time (default ≈ 1000 s) and returns the
best coefficient list it discovered.
"""

import time
import math
import numpy as np


def _conv_max(p: np.ndarray, L: int) -> float:
    """
    Return the maximum entry of the linear convolution of p with itself.
    The convolution is computed with a real FFT of length L (L >= 2*len(p)-1).
    """
    # forward FFT of the zero‑padded vector
    fft = np.fft.rfft(p, n=L)
    # element‑wise product (p̂)^2 gives the convolution in the frequency domain
    conv = np.fft.irfft(fft * fft, n=L)
    # only the first 2*n-1 components correspond to the full linear convolution
    return conv[: 2 * len(p) - 1].max()


def _score(p: np.ndarray, L: int) -> float:
    """
    Compute the evaluation function for a probability vector p.
    Since sum(p) == 1, the denominator simplifies and the score is
        2 * n * max(p * p)
    """
    n = len(p)
    return 2.0 * n * _conv_max(p, L)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation sequence using a simple simulated‑annealing
    scheme.  The function respects the time budget, prints progress and finally
    returns a list of non‑negative floats (the best sequence found).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # a small safety margin

    # -------------------------------------------------------------
    # Initialise a probability vector (sum == 1) of the same length as the
    # best known construction `height_sequence_1` if it exists.
    # -------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        n = len(base)
    else:
        # fallback: a modest length if the global is not present
        n = 2000
        base = np.abs(rng.normal(size=n))

    # make sure the starting vector is non‑negative and normalised
    base = np.maximum(base, 0.0)
    if base.sum() == 0.0:
        base = np.ones_like(base)
    base /= base.sum()

    # choose between the known construction and a fresh random start
    if rng.random() < 0.5:
        cur = base.copy()
    else:
        cur = np.abs(rng.normal(size=n))
        cur /= cur.sum()

    # pre‑compute a convenient FFT length (next power of two)
    L = 1 << ((2 * n - 1).bit_length())

    # current score and best‑ever record
    cur_score = _score(cur, L)
    best = cur.copy()
    best_score = cur_score

    print(f"[init] length={n}, start_score={cur_score:.6f}")

    # -------------------------------------------------------------
    # Simulated‑annealing parameters
    # -------------------------------------------------------------
    start_time = time.time()
    total_time = deadline - start_time
    move_scale = 0.1                # typical fraction of mass moved in one step
    attempts = 0
    accepted = 0
    renorm_interval = 10000        # re‑normalise to avoid drift

    while time.time() < deadline:
        attempts += 1

        # ---- propose a move -------------------------------------------------
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or cur[i] == 0.0:
            continue

        # fraction of the source mass to transfer (bounded by move_scale)
        delta = cur[i] * move_scale * rng.random()
        old_i = cur[i]
        old_j = cur[j]
        cur[i] = old_i - delta
        cur[j] = old_j + delta

        # ---- evaluate -------------------------------------------------------
        new_score = _score(cur, L)

        # ---- acceptance test ------------------------------------------------
        improve = new_score < cur_score
        if improve:
            accepted += 1
        else:
            # linear cooling schedule; temperature proportional to remaining time
            elapsed = time.time() - start_time
            temperature = max(best_score, 1e-12) * (1.0 - elapsed / total_time)
            # Metropolis rule
            prob = math.exp(-(new_score - cur_score) / (temperature + 1e-12))
            if rng.random() < prob:
                improve = True
                accepted += 1

        if improve:
            cur_score = new_score
            # keep the new vector as the current state
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                print(f"[{time.time() - start_time:6.1f}s] new_best = {best_score:.6f}")
        else:
            # revert the move
            cur[i] = old_i
            cur[j] = old_j

        # ---- adaptive move size ---------------------------------------------
        if attempts % 5000 == 0:
            acc_ratio = accepted / attempts
            if acc_ratio > 0.25:
                move_scale = min(move_scale * 1.5, 0.4)
            elif acc_ratio < 0.05:
                move_scale = max(move_scale * 0.7, 0.001)
            # occasional heavy perturbation to escape plateaus
            if rng.random() < 0.005:
                # select ~5 % of positions and redistribute their total mass randomly
                mask = rng.random(n) < 0.05
                mass = cur[mask].sum()
                if mass > 0:
                    cur[mask] = rng.random(mask.sum())
                    cur[mask] /= cur[mask].sum()
                    cur[mask] *= mass
            # renormalise to mitigate accumulated rounding errors
            if attempts % renorm_interval == 0:
                cur /= cur.sum()

    # final normalisation (the evaluator will clip anyway)
    result = best.tolist()
    print(f"[done] best_score = {best_score:.6f} after {time.time() - start_time:.1f}s")
    return result
