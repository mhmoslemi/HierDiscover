"""
Propose a step function that analytically gives a very low evaluation score.

The evaluation metric is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

which is *scale‑invariant*: multiplying every entry of `a` by a constant does
not change the score.  For a monotone power shape

    a_i = ( i / (n‑1) ) ** p ,   i = 0 … n‑1 ,

the convolution at the centre (the maximal entry) is approximately

    max(conv) ≈ n * ∫_0^1 x^p (1‑x)^p dx = n * B(p+1, p+1)

and   sum(a) ≈ n * ∫_0^1 x^p dx = n / (p+1).

Consequently the score tends to

    score(p) = 2 * B(p+1, p+1) * (p+1)^2

which decreases rapidly as `p` grows (e.g. ≈0.6 for p=2, ≈0.08 for p=4,
≈0.008 for p=6, ≈7·10⁻⁴ for p=8).  By trying a few lengths `n` and a few
exponents `p` we can quickly find a sequence with a score far below the
target 1.503.

The function below enumerates a modest set of `(n, p)` pairs, evaluates each
candidate with the provided `evaluate_sequence`, keeps the best one and
returns it.  The whole search finishes in far less than the allotted
budget.
"""

import time
import numpy as np

def _make_power_sequence(n: int, p: int) -> list[float]:
    """Return a monotone power‑law sequence a_i = (i/(n‑1))**p."""
    if n <= 1:
        # Degenerate case – just a single element set to 1.0
        return [1.0]
    idx = np.arange(n, dtype=float)
    denom = float(n - 1)
    seq = (idx / denom) ** p
    # Convert to plain Python floats for the evaluator
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate and return a step‑function sequence with a very low score.

    The search is deterministic (the seed is ignored) and scans a small
    handcrafted set of lengths and powers.  It stops well before the time
    budget expires.
    """
    # ------------------------------------------------------------
    # Parameters for the handcrafted search
    # ------------------------------------------------------------
    lengths = [500, 1000, 2000, 5000]          # different sequence sizes
    powers  = [1, 2, 3, 4, 5, 6, 7, 8]        # exponents p to try

    deadline = time.time() + budget_s - 0.5   # keep a small safety margin
    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------
    # Exhaustive scan (tiny – at most 32 candidates)
    # ------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        for p in powers:
            if time.time() > deadline:
                break

            seq = _make_power_sequence(n, p)
            try:
                score = evaluate_sequence(seq)
            except Exception as e:
                # Any unexpected error – treat as infinite score
                print(f"Evaluation failed for n={n}, p={p}: {e}")
                score = float('inf')

            print(f"Tested n={n}, p={p:2d} → score={score:.6g}")

            if score < best_score:
                best_score = score
                best_seq = seq

    # ------------------------------------------------------------
    # Safety net – if nothing was found (should never happen), fall back
    # to a simple uniform sequence.
    # ------------------------------------------------------------
    if best_seq is None:
        print("No valid candidate found; using a uniform sequence as fallback.")
        best_seq = [1.0] * 1000
        best_score = evaluate_sequence(best_seq)

    print(f"Best candidate: length={len(best_seq)}, score={best_score:.6g}")
    return best_seq
