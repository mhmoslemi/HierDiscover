We implement projection onto simplex following algorithm described by Duchi et al.:

def proj_simplex(v):
    """Project onto simplex {x >=0, sum x = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np, time, math

def proj_simplex(v):
    # Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # all zeros
        return np.zeros_like(v)
    rho = ind[mask][-1]
    theta = cssv[mask][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0

    try:

    best_score = evaluate_sequence(best_a.tolist())
    print(f"Init score: {best_score:.6f}")

    beta = 5.0
    lr = 0.2
    iteration = 0

    while time.time() < deadline:
        # compute convolution via FFT
        # next power of two
        m = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(A * A, n=m)[:2*n-1]

        max_b = conv.max()

        # smooth weights
        # to avoid overflow, shift by max
        exp_term = np.exp(beta * (conv - max_b))
        w = exp_term / exp_term.sum()

        # gradient of smoothed max w.r.t a: g = 2 * correlate(w, a, mode='valid')
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # we keep sum=1 by projecting onto simplex after the step
        # take step
        step = lr * grad
        a_new = a - step
        a_new = proj_simplex(a_new)

        # compute real score occasionally
        if iteration % 20 == 0:
            score = evaluate_sequence(a_new.tolist())
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                print(f"Iter {iteration}: new best {best_score:.6f}")

        a = a_new
        iteration += 1

        # schedules
        if iteration % 200 == 0:
            beta = min(beta * 1.2, 100.0)
            lr = max(lr * 0.9, 1e-4)

        # occasional random perturbation to escape flat regions
        if iteration % 500 == 0:
            noise = np.random.normal(scale=0.01, size=n)
            a = proj_simplex(a + noise)

    print(f"Finished, best score {best_score:.6f}")
    return best_a.tolist()

"""
Simple (yet surprisingly effective) optimizer for the step‑function inequality.

The objective to minimise is

    F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

where `conv(a,a)` is the linear (non‑circular) convolution of the
non‑negative vector `a` with itself.  The function is homogeneous,
so we can enforce `sum(a) = 1` and work on the probability simplex.
The algorithm uses a smooth approximation of the max (soft‑max)
to obtain a usable gradient, then performs projected gradient descent
with an adaptive step‑size and an increasing “sharpness’’ parameter.
Occasional random perturbations help escape plateaus.
The best sequence found before the time budget expires is returned.

No external files are accessed, only NumPy and the already‑available
`evaluate_sequence` function.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection of a vector onto the simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (Duchi et al., 2008)."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)           # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical safety – renormalise exactly to 1
    w /= w.sum()
    return w

# Core optimiser
# ------------------------------------------------------------

    list[float]
        The best sequence found within the allocated time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a safety margin

    # --------------------------------------------------------

    # --------------------------------------------------------

    a = np.random.rand(n)

    best_a = a.copy()
    best_obj = 2.0 * n * np.max(np.convolve(a, a)) / (np.sum(a) ** 2)

    print(f"[init] n={n}, initial obj={best_obj:.6f}")

    # --------------------------------------------------------

    # --------------------------------------------------------
    beta = 5.0          # soft‑max sharpness

    it   = 0
    # FFT size for the convolution a*a (length 2n‑1)

        # -----------------------------------------------------------------

        # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())

    return best_a.tolist()
