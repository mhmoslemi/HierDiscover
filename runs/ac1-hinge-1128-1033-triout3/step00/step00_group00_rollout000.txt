"""
propose_candidate
=================

A robust, time‑bounded search for a step‑function (non‑negative sequence) that
minimises the evaluation function

    2 * n * max(conv(a, a)) / (sum(a)**2)

The theoretical lower bound for non‑negative sequences is 2 (attained by the
uniform sequence).  The algorithm therefore starts from the uniform
distribution – which is optimal – and then performs a cheap random‑swap local
search for a short time; any discovered improvement replaces the current best.
If a previously found construction (`height_sequence_1`) is available it is used
as the initial candidate, otherwise a fresh uniform sequence is created.

The function respects the required 1000 s budget, always returns a valid
(non‑negative, non‑empty) list and prints diagnostic information.
"""

import time
import numpy as np

def _uniform_sequence(length: int, value: float = 1.0) -> list[float]:
    """Return a uniform non‑negative sequence of the given length."""
    return [float(value)] * length


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Total time budget in seconds (default 1000 s).
    **kwargs
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found within the allotted time.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Reserve a tiny margin for the final return.
    deadline = start_time + max(0.0, budget_s - 0.5)

    # ------------------------------------------------
    # 1. Initialise a starting sequence.
    # ------------------------------------------------
    try:
        # If the notebook provides a previous construction, reuse its length.
        init_len = len(height_sequence_1)  # type: ignore[name-defined]
        # Use the provided construction if it already has a good score.
        current = list(height_sequence_1)  # type: ignore[name-defined]
        current_score = evaluate_sequence(current)
        print(f"Starting from existing construction (len={init_len}) → score={current_score:.6f}")
    except Exception:
        # Fallback: a fresh uniform sequence.
        init_len = 1742                     # the length used in the reference solution
        current = _uniform_sequence(init_len, value=1.0)
        current_score = evaluate_sequence(current)
        print(f"Starting from uniform sequence (len={init_len}) → score={current_score:.6f}")

    best_seq = current.copy()
    best_score = current_score

    # Pre‑compute the total sum – it stays constant during the simple swap moves.
    total_sum = float(np.sum(current))

    # ------------------------------------------------
    # 2. Simple local search: move a tiny amount from one index to another.
    #    The L1 norm (total_sum) is preserved, so the evaluation ratio
    #    depends only on the change of the maximal convolution.
    # ------------------------------------------------
    n = init_len
    avg_val = total_sum / n
    # Step size is a small fraction of the average value.
    step_frac = 0.1

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # Choose two distinct indices.
        i, j = np.random.choice(n, size=2, replace=False)

        # Determine a feasible epsilon to transfer from i to j.
        eps = step_frac * avg_val * np.random.rand()
        eps = min(eps, current[i])           # keep a_i ≥ 0

        if eps == 0.0:
            continue

        # Apply the swap.
        new_seq = current.copy()
        new_seq[i] -= eps
        new_seq[j] += eps

        # Evaluate the new candidate.
        try:
