def simplex_projection(v):
    # v is a vector, return projection onto {x >=0, sum x =1}
    # Use algorithm by Wang & Carreira-Perpiñán
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_step(a, step):
    # Compute convolution
    b = np.convolve(a, a)
    max_idx = np.argmax(b)
    # Compute gradient: derivative of max b w.r.t a
    # g_i = 2 * a_{max_idx - i} if index valid else 0.
    n = len(a)
    grad = np.zeros_like(a)
    for i in range(n):
        j = max_idx - i
        if 0 <= j < n:
            grad[i] = 2 * a[j]
    # Move opposite direction (minus step*grad)
    a_new = a - step * grad
    # Project to simplex
    a_proj = simplex_projection(a_new)
    return a_proj, b[max_idx]

"""
Propose a step‑function (non‑negative heights) that minimizes

    2 * n * max(conv(a,a)) / (sum(a)**2)

where `conv` is the full linear convolution.  

The algorithm works in three stages:

1. **Convex relaxation (Geometric Program).**  
   By writing a_i = exp(x_i) the quadratic constraints
   `sum_i a_i a_{k-i} <= t` become convex:
   `sum_i exp(x_i + x_{k-i}) <= t`.  
   For a fixed length `n` we solve the GP

        minimise   t
        subject to sum_i exp(x_i) = 1
                   sum_i exp(x_i + x_{k-i}) <= t   (for all k)

   using CVXPY/ECOS.  The optimal value `t` is the smallest possible
   maximal convolution value for sequences of length `n` with unit sum.
   The corresponding heights are `a_i = exp(x_i)`.

2. **Local stochastic refinement.**  
   Starting from the GP solution we perform a cheap hill‑climbing
   on the simplex: repeatedly move a tiny amount of mass from one
   index to another and keep the change if the evaluation improves.
   This often pushes the score a few percent further down.

3. **Multi‑scale search.**  
   The procedure is repeated for several lengths (≈50 … 400).  The
   best sequence found within the allotted time (≈1000 s) is returned.

The whole routine respects the 1000 s wall‑time limit, never
produces negative entries, and works completely within the provided
`evaluate_sequence` function.
"""

import time
import numpy as np

# CVXPY is optional – if it cannot be imported we fall back to a
# simple random‑search heuristic.
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVXPY_AVAILABLE = False


def _project_simplex(v):
    """Project a vector onto the probability simplex {x >= 0, sum x = 1}."""
    # Implementation of the classic O(n log n) algorithm.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho = max { j | u_j - cssv_j / j > 0 }
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


def _local_refinement(a, budget, score_fn, rng):
    """
    Cheap stochastic hill‑climbing on the simplex.

    Parameters
    ----------
    a : np.ndarray
        Starting sequence (already normalized, sum == 1).
    budget : float
        Seconds allocated to the refinement.
    score_fn : callable
        Function that returns the evaluation score for a given sequence.
    rng : np.random.Generator
        Random number generator.

    Returns
    -------
    (np.ndarray, float)
        Best sequence found and its score.
    """
    start = time.time()
    best_seq = a.copy()
    best_score = score_fn(best_seq.tolist())

    n = len(a)
    # step size is a small fraction of the current average mass
    eps = 0.02 / n

    while time.time() - start < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 2 * eps   # in [-eps, eps]
        # move mass from i to j
        if best_seq[i] - delta < 0 or best_seq[j] + delta < 0:
            continue
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum is unchanged, only need to enforce non‑negativity
        cand = np.clip(cand, 0, None)
        # re‑normalise (tiny numerical drift)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum
        cand_score = score_fn(cand.tolist())
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
    return best_seq, best_score


def _solve_gp(n, time_limit):
    """
    Solve the geometric program for a given length `n`.

    Returns
    -------
    tuple(np.ndarray or None, float)
        (optimal sequence, optimal t)  –  both ``None`` if the solver fails
        or exceeds the allotted time.
    """
    if not _CVXPY_AVAILABLE:      # pragma: no cover
        return None, None

    # Variables: x_i (log of a_i) and scalar t (max convolution)
    x = cp.Variable(n)
    t = cp.Variable(nonneg=True)

    constraints = [cp.sum(cp.exp(x)) == 1]            # unit sum
    # Build the convolution constraints
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        terms = [cp.exp(x[i] + x[k - i]) for i in range(i_min, i_max + 1)]
        constraints.append(cp.sum(terms) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # ECOS handles exponential cones; we give it a modest iteration cap.
    try:
        prob.solve(solver=cp.ECOS,
                    max_iters=5000,
                    abstol=1e-7,
                    reltol=1e-7,
                    verbose=False)
    except Exception:          # pragma: no cover
        return None, None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, None

    # Recover the positive sequence a_i = exp(x_i)
    a = np.exp(x.value)
    # Numerical cleanup
    a = np.clip(a, 0, 1000.0)
    a_sum = a.sum()
    if a_sum == 0:
        return None, None
    a /= a_sum                         # ensure exact unit sum
    return a, float(t.value)

        # 2. Local refinement (takes at most a few seconds)
        # --------------------------------------------------------
        remaining = deadline - time.time()
        # give at most 4 seconds per refinement
        refine_time = min(4.0, remaining, gp_budget * 0.3)
        if refine_time > 0.5:          # need some time for a worthwhile run
            print(f"   Refinement for n={n} (≈{refine_time:.1f}s)")
            a_ref, score_ref = _local_refinement(
                a,
                budget=refine_time,
                score_fn=lambda seq: evaluate_sequence(seq),
                rng=rng,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.tolist()
                print("   ** New global best (refined) **")
        # keep a little slack for the next length
        if time.time() + 2.0 > deadline:
            break

    # ------------------------------------------------------------
    # 3. Fallback: simple random Dirichlet if nothing succeeded
    # ------------------------------------------------------------
    if best_seq is None:
        print("GP phase failed – falling back to a random Dirichlet.")
        n = rng.integers(200, 800)
        a = rng.dirichlet(np.ones(n))
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)

    # final log
    print(f"Finished search – best score = {best_score:.6f} "
          f"with length {len(best_seq)}")
    return best_seq
