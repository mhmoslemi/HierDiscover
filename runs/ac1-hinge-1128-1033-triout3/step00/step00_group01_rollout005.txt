def simplex_projection(v):
    # input: vector v (any shape)
    # output: projected onto simplex {x: sum x = 1, x >= 0}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    # ensure v sum to 1, nonnegative
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0]
    if len(rho) == 0:
        theta = 0
    else:

        theta = cssv[rho_idx] / (rho_idx + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
A simple mirror‑descent optimizer that tries to flatten the
self‑convolution of a non‑negative sequence.

The algorithm works on the simplex (sum of coefficients = 1) and
minimises a smooth approximation of max_k (a*a)[k] using an
entropy‑regularised (soft‑max) objective.
At each iteration we:

  * evaluate the convolution a*a,
  * build a soft‑max distribution over the convolution values,
  * compute the gradient of the soft‑max objective w.r.t. a,
  * perform an exponentiated‑gradient (mirror‑descent) update,
  * renormalise to the simplex.

The step‑size decays and the soft‑max temperature (α) is increased
gradually so that the optimisation first smoothes the landscape and
later focuses on the true maximum.  Several random restarts (including
a uniform start and an “arcsine‑shaped” start) are performed until the
time budget runs out.  The best sequence found (as a plain Python list)
is returned.

The score for a candidate a with sum(a)=S is simply
    2 * n * max(conv(a, a)) / S**2,
because evaluate_sequence() applies the same formula after clipping.
All numbers are kept non‑negative and far below the 1000‑limit, so the
wrapper checks never interfere.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions (top‑level, no closures)
# --------------------------------------------------------------

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    # Implementation of the algorithm from
    # "Efficient Projections onto the l1-Ball for Learning in High Dimensions"
    # (Duchi et al., 2008)
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # python index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _gradient_softmax_max(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth approximation
        L(a) = (1/α) log Σ_k exp(α * (a*a)[k])
    evaluated at a.
    Returns (grad, conv_max, conv_full) where grad has shape (n,).
    """
    n = a.shape[0]

    # full convolution a*a (length 2n-1)
    conv_full = np.convolve(a, a)

    # soft‑max weights over the convolution entries
    # p_k = exp(α * conv_k) / Σ_j exp(α * conv_j)
    w = _softmax(alpha * conv_full)

    # gradient: g_i = a_i + Σ_j w_{i+j} a_j
    # Compute the weighted sum efficiently via a convolution.
    w_rev = w[::-1]
    # convolution of w_rev (len 2n-1) with a (len n) → length 3n-2
    conv_rev = np.convolve(w_rev, a)
    # index mapping: conv_rev[ (len(w)-1) - i ] = Σ_j w_{i+j} a_j
    idx = (len(w) - 1) - np.arange(n)
    weighted_sum = conv_rev[idx]
    grad = a + weighted_sum
    return grad, np.max(conv_full), conv_full


def _score_from_conv(conv: np.ndarray, n: int, total: float) -> float:
    """Compute the evaluation score from a convolution array."""
    # evaluate_sequence returns 2 * n * max(conv) / (sum(a)**2)
    max_b = np.max(conv)
    if total < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (total ** 2))


def _run_optimisation(
    a_init: np.ndarray,
    deadline: float,
    max_iters: int = 5000,
    alpha_start: float = 0.5,
    alpha_max: float = 200.0,
    alpha_mul: float = 1.5,
    alpha_step: int = 200,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimisation for a single start vector.
    Returns (best_a, best_score).
    """
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score_from_conv(np.convolve(a, a), n, a.sum())

    alpha = alpha_start
    # time‑guarded optimisation loop
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # gradient of the smoothed max‑convolution
        grad, conv_max, conv_full = _gradient_softmax_max(a, alpha)

        # Mirror‑descent (exponentiated‑gradient) update
        # step size decays like η_t = η0 / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1)

        # shift gradient to avoid overflow in exp
        shift = grad - grad.max()
        factor = np.exp(-eta * shift)
        a = a * factor

        # renormalise onto the simplex (entropy projection)
        s = a.sum()
        if s <= 0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # evaluate current score (no need to call evaluate_sequence)
        score = _score_from_conv(conv_full, n, a.sum())
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(
                f"Iter {it:5d} | score {score:.6f} | max_conv {conv_max:.6e} | "
                f"α {alpha:.2f}"
            )

        # increase the soft‑max temperature periodically
        if (it + 1) % alpha_step == 0:
            alpha = min(alpha * alpha_mul, alpha_max)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) with a small
    evaluation score.  The function runs a simple mirror‑descent
    optimiser (with several random restarts) for up to `budget_s`
    seconds and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # keep a little margin
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a problem size.  Use the existing construction if available,
    # otherwise fall back to a moderate size.
    # ------------------------------------------------------------------
    global_height_seq = globals().get("height_sequence_1")
    if global_height_seq is not None and isinstance(global_height_seq, (list, tuple)):
        n = len(global_height_seq)
        # normalise the existing sequence to sum = 1 (keep the shape)
        base = np.array(global_height_seq, dtype=float)
        base = np.clip(base, 0.0, 1000.0)
        total = base.sum()
        if total > 0:
            a_start = base / total
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        # No external sequence – choose a fixed length that is fast enough
        n = 2000
        a_start = np.full(n, 1.0 / n)

    best_seq = a_start.copy()
    best_score = _score_from_conv(np.convolve(best_seq, best_seq), n, best_seq.sum())

    print(f"Starting optimisation with n = {n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to run a single optimisation pass and possibly update best.
    # ------------------------------------------------------------------
    def _try_start(a0: np.ndarray, iters: int):
        nonlocal best_seq, best_score
        seq, sc = _run_optimisation(
            a_init=a0,
            deadline=deadline,
            max_iters=iters,
        )
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"*** New global best: {sc:.6f}")

    # ------------------------------------------------------------------
    # 1) Try the uniform start (already stored in a_start)
    # ------------------------------------------------------------------
    _try_start(a_start, iters=4000)

    # ------------------------------------------------------------------
    # 2) Try an arcsine‑like start (more mass at the ends)
    # ------------------------------------------------------------------
    inds = np.arange(n, dtype=float)
    arcsine_shape = 1.0 / np.sqrt((inds + 0.5) * (n - 0.5 - inds))
    arcsine_shape /= arcsine_shape.sum()
    _try_start(arcsine_shape, iters=3000)

    # ------------------------------------------------------------------
    # 3) Random Dirichlet restarts while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Dirichlet(1) → uniform on the simplex; add a tiny Gaussian jitter
        rand_vec = np.random.dirichlet(np.ones(n))
        _try_start(rand_vec, iters=1500)

        # occasionally insert a small perturbation of the current best
        if np.random.rand() < 0.2:
            pert = best_seq * (1.0 + 0.01 * np.random.randn(n))
            pert = np.clip(pert, 0, None)
            if pert.sum() > 0:
                pert = pert / pert.sum()
                _try_start(pert, iters=800)

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = _score_from_conv(np.convolve(best_seq, best_seq), n, best_seq.sum())
    print(f"Finished. Best score = {final_score:.6f} (n = {n})")
    return final_list
