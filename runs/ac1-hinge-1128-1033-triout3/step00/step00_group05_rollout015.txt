    n = len(a)

def propose_candidate():
    import numpy as np, time, random
    start = time.time()
    deadline = start + 970  # reserve a few seconds
    best_seq = None
    best_score = float('inf')
    # optionally start from height_sequence_1
    try:
        init_seq = list(height_sequence_1)
        score = evaluate_sequence(init_seq)
        if score < best_score:
            best_seq = init_seq[:]
            best_score = score
    except Exception:
        pass
    # random search
    MIN_N = 5
    MAX_N = 50
    # many iterations
    while time.time() < deadline:
        n = np.random.randint(MIN_N, MAX_N+1)
        # choose concentration parameter
        alpha = np.random.choice([0.2,0.5,0.8,1.0,2.0])
        raw = np.random.gamma(alpha, 1.0, size=n)
        if raw.sum() == 0:
            raw = np.ones(n)
        p = raw / raw.sum()
        seq = p.tolist()
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"New best random: n={n}, alpha={alpha:.2f}, score={score:.6f}")
        # occasional local improvement on best_seq if exists
        if best_seq is not None and np.random.rand() < 0.01:
            # small hill climb steps starting from best_seq
            a = np.array(best_seq, dtype=float)
            n = len(a)
            for _ in range(100):
                i, j = np.random.randint(n, size=2)
                if i==j:
                    continue
                # move a fraction from i to j
                if a[i] <= 1e-12:
                    continue
                delta = np.random.rand() * a[i] * 0.5
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # renorm not needed
                new_score = evaluate_sequence(a_new.tolist())
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.tolist()
                    a = a_new
                    print(f"Improved via hillclimb: n={n}, score={new_score:.6f}")
    print(f"Best final score: {best_score:.6f}, length={len(best_seq)}")
    return best_seq

def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, random

    start = time.time()
    deadline = start + budget_s - 5
    np.random.seed(int(time.time()) % 2**32)

    best_seq = None
    best_score = float('inf')

    # optionally start from known sequences
    try:
        init = list(height_sequence_1)
        s = evaluate_sequence(init)
        best_seq, best_score = init, s
    except Exception:
        pass

    # functions for generating candidate
    def uniform_seq(n):
        return [1.0/n]*n

    def gamma_seq(n, alpha):
        raw = np.random.gamma(alpha, 1.0, size=n)
        if raw.sum() == 0:
            raw = np.ones(n)
        return (raw / raw.sum()).tolist()

    def beta_like_seq(n, power):
        idx = np.arange(n)
        # U-shaped: inverse of a smooth curve
        w = 1.0 / ( ( (idx+1)*(n-idx) ) ** power + 1e-12 )
        w /= w.sum()
        return w.tolist()

    def polynomial_seq(n, a):
        # a in (0,1) controlling shape: w_i = (i+1)^a * (n-i)^a
        idx = np.arange(n)
        w = ((idx+1)*(n-idx))**a
        w /= w.sum()
        return w.tolist()

    # main loop
    while time.time() < deadline:
        # choose random length
        n = np.random.randint(5, 101)  # up to 100
        # choose generation method
        method = np.random.choice(['gamma','beta','poly','uniform'])
        if method == 'gamma':
            alpha = np.random.choice([0.2,0.5,0.8,1.0,2.0,5.0])
            seq = gamma_seq(n, alpha)
        elif method == 'beta':
            power = np.random.uniform(0.5, 3.0)
            seq = beta_like_seq(n, power)
        elif method == 'poly':
            a = np.random.uniform(-2, 2)  # allow negative to produce U shape
            seq = polynomial_seq(n, a)
        else: # uniform
            seq = uniform_seq(n)
        # evaluate
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"New best: n={n}, method={method}, score={score:.6f}")

        # occasional local hill climb
        if best_seq is not None and np.random.rand() < 0.02:
            a = np.array(best_seq, dtype=float)
            n2 = len(a)
            # perform a few small perturbations
            for _ in range(50):
                i, j = np.random.randint(n2, size=2)
                if i == j:
                    continue
                # delta up to small fraction
                delta = np.random.rand() * a[i] * 0.3
                if delta == 0:
                    continue
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # ensure non-negative

"""
Search for a non‑negative step‑function (a list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm mixes three cheap heuristics:

1. **Randomised initialization** – Dirichlet / Gamma draws with various
   concentration parameters to obtain a wide variety of probability vectors.
2. **LP‑based direction improvement** – given a current vector `f` (scaled to
   have L1‑norm √(2n)), solve a linear programme that finds a vector `g` with a
   larger L1‑norm while keeping the mixed convolution `f⋆g` bounded by the
   current peak.  A few mixing weights `t` are tried and the best improvement
   is kept.  This is the idea from the published SOTA but with an adaptive
   step‑size search and a tighter time budget per LP.
3. **Simple hill‑climbing** – occasional random transfers of mass between two
   coordinates; accepted only if they improve the evaluation.

The whole search runs for the allotted time (≈1000 s) and finally returns the
best sequence found.  All numbers are non‑negative; the evaluator normalises
them internally, so the algorithm keeps the vectors normalised to L1‑norm = 1
(which is also convenient for the LP).
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: solve the LP  max sum(b)  s.t.  conv(f, b) ≤ M  and  b ≥ 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, bound):
    """
    Solve:
        maximize    sum(b_j)
        subject to  (f_seq * b)_k ≤ bound   for all k
                    b_j ≥ 0
    Returns the optimal b as a dense numpy array, or None if the LP fails.
    """
    n = len(f_seq)
    # Objective (minimise negative sum to emulate maximise)
    c = -np.ones(n, dtype=float)

    # Build A_ub * b ≤ bound  (one row per convolution entry)
    A = np.zeros((2 * n - 1, n), dtype=float)
    for k in range(2 * n - 1):
        # indices i + j = k  →  j = k - i
        low_i = max(0, k - (n - 1))
        high_i = min(k, n - 1)
        i_vals = np.arange(low_i, high_i + 1, dtype=int)
        j_vals = k - i_vals
        A[k, j_vals] = f_seq[i_vals]

    # Bounds for variables: b_j ≥ 0
    bounds = [(0, None)] * n

    # Solve using HiGHS (the default).  Give a small per‑solve time limit.
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, bound),
                  bounds=bounds, method='highs',
                  options={'time_limit': 2.0, 'disp': False})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Improve a single candidate using the LP step and tiny hill‑climbs
# ----------------------------------------------------------------------
def improve_sequence(seq, max_iters, deadline):
    """
    Starting from `seq` (a list of non‑negative numbers, sum ≈ 1),
    perform up to `max_iters` refinement steps or until `deadline`.
    Returns a (list, score) pair with the best sequence discovered.
    """
    a = np.array(seq, dtype=float)
    n = len(a)
    # Normalise to the scale expected by the LP (L1 = sqrt(2n))
    target_norm = np.sqrt(2.0 * n)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    a = a * target_norm / a.sum()

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # mixing coefficients to try after solving the LP
    mix_vals = (0.01, 0.05, 0.1, 0.2, 0.4)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break

        # current maximal self‑convolution value
        cur_max = np.max(np.convolve(best_seq, best_seq))

        # --- LP direction -------------------------------------------------
        g = solve_convolution_lp(best_seq, cur_max)
        if g is None:
            # fallback: tiny random push on a random coordinate
            i = np.random.randint(n)
            best_seq[i] = max(0.0, best_seq[i] + np.random.randn() * 0.01)
            # re‑normalise to the prescribed L1 norm
            best_seq = best_seq * target_norm / best_seq.sum()
            continue

        # normalise g to the same L1‑norm
        sumg = g.sum()
        if sumg > 0:
            g = g * target_norm / sumg

        # try several convex combinations (1‑t)·f + t·g
        for t in mix_vals:
            cand = (1.0 - t) * best_seq + t * g
            # small numerical safety – clip tiny negatives
            cand = np.clip(cand, 0.0, None)
            cand = cand * target_norm / cand.sum()
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

        # --- tiny hill‑climb on the current best ------------------------
        # (occasionally, to escape possible plateaus)
        if np.random.rand() < 0.05:
            i, j = np.random.randint(n, size=2)
            if i != j and best_seq[i] > 1e-12:
                delta = np.random.rand() * best_seq[i] * 0.2
                cand = best_seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = cand * target_norm / cand.sum()
                sc = evaluate_sequence(cand.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()

    # final re‑scale to a plain probability vector (sum = 1)
    final_seq = best_seq / best_seq.sum()
    return final_seq.tolist(), best_score

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Random length (small enough for cheap LPs but large enough to beat 2.0)
        n = np.random.randint(30, 151)           # 30 ≤ n ≤ 150

        # Random generation method
        method = np.random.choice(['gamma', 'beta'])
        if method == 'gamma':
            # Gamma (Dirichlet) with a random concentration
            alpha = np.random.choice([0.2, 0.5, 0.8, 1.0, 2.0, 5.0])
            raw = np.random.gamma(alpha, 1.0, size=n)
            if raw.sum() == 0:
                raw = np.ones(n)
            init_seq = raw / raw.sum()
        else:  # 'beta' – U‑shaped deterministic family
            power = np.random.uniform(0.3, 3.0)
            idx = np.arange(n)
            # w_i = 1 / ((i+1)*(n-i))**power   gives larger weight at the edges
            w = 1.0 / ((idx + 1) * (n - idx)) ** power
            init_seq = w / w.sum()

        # Refine this candidate with the LP‑based routine
        refined_seq, refined_score = improve_sequence(
            init_seq.tolist(),
            max_iters=80,                # enough iterations for a noticeable improvement
            deadline=deadline
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq
            print(f"New global best → n={n}, method={method}, score={refined_score:.6f}")

        # Occasionally launch a pure hill‑climb on the current global best
        if best_seq is not None and np.random.rand() < 0.07:
            a = np.array(best_seq, dtype=float)
            n_best = len(a)
            for _ in range(120):
                i, j = np.random.randint(n_best, size=2)
                if i == j or a[i] <= 0:
                    continue
                delta = np.random.rand() * a[i] * 0.25
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # keep non‑negative, renormalise
                a_new = np.clip(a_new, 0.0, None)
                a_new = a_new / a_new.sum()
                sc = evaluate_sequence(a_new.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a_new.tolist()
                    a = a_new
                    print(f"  Hill‑climb improved global best to {sc:.6f}")

    # ------------------------------------------------------------------
    # Safety net – if nothing was found, fall back to a simple uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / 10] * 10

    print(f"Finished search – best score {best_score:.6f}, length {len(best_seq)}")
    return best_seq
