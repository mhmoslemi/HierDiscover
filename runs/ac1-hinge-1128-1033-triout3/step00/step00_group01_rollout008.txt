"""
Propose a step‐function (non‑negative sequence) with a low self‑convolution peak.

The algorithm starts from a uniform sequence (which gives a value of 2.0) and
repeatedly solves a linear program:

    maximise    Σ b_i
    subject to  (f * b)_k ≤ M      for all k
                b_i ≥ 0

where `f` is the current sequence, `M = max(f * f)`.  The LP finds a direction `g`
that can increase the L1‑norm without increasing the current worst convolution
value.  A convex combination `f ← (1‑t) f + t g` (with a modest step size `t`)
strictly decreases the evaluation function

    evaluate = 2·n·max(f*f) / (Σ f_i)²

because the denominator grows while the numerator does not.  The process is
repeated until the time budget expires, occasionally performing a tiny random
perturbation if the LP fails.  The best sequence seen is returned.

The code uses only NumPy and SciPy (linprog) and respects the 1000 s budget.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def solve_convolution_lp(f_seq: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise    Σ b_i
        subject to  (f * b)_k ≤ rhs   for all k
                    b_i ≥ 0

    Returns the optimal b (as a NumPy array) or None if the solver fails.
    """
    n = f_seq.size
    # objective: maximise sum(b) → minimise -sum(b)
    c = -np.ones(n)

    # Build convolution constraints (2·n‑1 rows)
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i ranges over indices where 0 ≤ i < n and 0 ≤ k‑i < n
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # indices i where the term f_i * b_{k‑i} appears
        # coefficient of b_{j} is f_i with j = k‑i
        # = f_i where i = k‑j  → fill row[j] = f_seq[k‑j]
        j_vals = np.arange(i_min, i_max + 1)
        row[j_vals] = f_seq[k - j_vals]
        rows.append(row)

    a_ub = np.vstack([rows, -np.eye(n)])           # ≤ rhs  and  b_i ≥ 0
    b_ub = np.hstack([np.full(len(rows), rhs), np.zeros(n)])

    # Use the high‑performance HiGHS solver (default)
    res = linprog(c,
                  A_ub=a_ub,
                  b_ub=b_ub,
                  bounds=(0, None),
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    return None


def get_good_direction(f_seq: np.ndarray, t: float) -> np.ndarray | None:
    """
    One iteration of the LP‑based improvement.
    Returns a new sequence (convex combination) or None if the LP was infeasible.
    """
    n = f_seq.size
    # normalise to the standard L1‑norm √(2n)
    target_sum = math.sqrt(2.0 * n)
    sum_f = f_seq.sum()
    if sum_f == 0:
        return None
    f_norm = f_seq * (target_sum / sum_f)

    # current worst convolution value M
    conv_f = np.convolve(f_norm, f_norm)
    M = float(conv_f.max())

    # solve LP to obtain a vector with a larger L1‑norm (if possible)
    g = solve_convolution_lp(f_norm, M)
    if g is None:
        return None

    sum_g = g.sum()
    if sum_g == 0:
        return None

    # scale g back to the same normalisation (does not change feasibility)
    g_norm = g * (target_sum / sum_g)

    # convex combination
    new_seq = (1.0 - t) * f_norm + t * g_norm
    return new_seq

    """
    Search for a sequence with a low evaluation value.
    The routine runs until the time budget expires and returns the best
    sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a problem size.  800‑1000 gives a good trade‑off between
    #    quality and speed of the LP solves.
    # ------------------------------------------------------------------
    n = 800
    target_sum = math.sqrt(2.0 * n)

    # initial sequence: uniform (the known 2.0 bound)
    cur_seq = np.ones(n, dtype=float) * (target_sum / n)

    # bookkeeping
    best_seq = cur_seq.copy()
    best_val = evaluate_sequence(best_seq.tolist())
    cur_val = best_val
    print(f"[init] n={n}, value={best_val:.6f}")

    # mixing parameter – start relatively aggressive and decay slowly
    t = 0.07
    decay_t = 0.9998

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ------------------------------------------------------------------
        # Main improvement step – LP based direction
        # ------------------------------------------------------------------
        new_seq = get_good_direction(cur_seq, t)

        if new_seq is not None:
            # evaluate the candidate
            try:
                new_val = evaluate_sequence(new_seq.tolist())
            except Exception:
                new_val = float('inf')

            # accept if it improves the current sequence
            if new_val < cur_val:
                cur_seq = new_seq
                cur_val = new_val

            # keep the global best
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()
                print(f"[{iteration:05d}] improved to {best_val:.6f}")

        else:
            # ------------------------------------------------------------------
            # Occasionally fall back to a tiny random perturbation (helps if LP

            # ------------------------------------------------------------------

                cur_seq[i] += eps
                cur_seq[j] -= eps
                # re‑normalise to keep the L1‑norm exactly √(2n)

            cur_val = val
            best_seq = cur_seq.copy()
            best_val = val
            print(f"[refine] new best {best_val:.6f}")

    print(f"[done] best value = {best_val:.6f} after {iteration} LP iterations")
    return best_seq.tolist()
