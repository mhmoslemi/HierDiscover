def propose_candidate(seed=42, budget_s=990, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    # Determine length n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000
    # initialize a uniform distribution
    a = np.full(n, 1.0/n, dtype=np.float64)
    # normalize to sum 1
    a /= a.sum()
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    curr_score = best_score
    curr_a = a.copy()
    # FFT size
    def next_pow2(x):
        return 1 << (int(x)-1).bit_length()
    L = next_pow2(3 * n)
    # parameters
    lr = 1.0   # learning rate for gradient step
    temp = 0.1
    decay = 0.999
    start = time.time()
    deadline = start + budget_s - 2.0
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # FFT of current a
        A = np.fft.rfft(curr_a, n=L)
        # convolution a*a
        conv_self = np.fft.irfft(A * A, n=L)[:2 * n - 1]
        max_b = conv_self.max()

        # sum of a is 1 (by construction)
        score = 2 * n * max_b  # since sum_a = 1
        # record best
        if score < best_score:
            best_score = score
            best_seq = curr_a.copy()
            print(f"Iter {iteration}: new best score {best_score:.6f} time {time.time() - start:.1f}s")
            # early exit if good enough
            if best_score <= 1.5:
                break
        # compute target average
        avg = 1.0 / (2 * n - 1)   # same as (sum_a**2)/(2n-1)
        diff = conv_self - avg

        diff_rev = diff[::-1]
        D = np.fft.rfft(diff_rev, n=L)
        conv_grad_full = np.fft.irfft(A * D, n=L)
        # extract c_i = conv_grad_full[2n-2 - i]
        idx_start = 2 * n - 2
        c = conv_grad_full[idx_start - np.arange(n)]
        grad = 4.0 * c
        # propose update
        step = lr * grad
        a_new = curr_a - step
        # enforce non-negativity
        a_new = np.where(a_new > 0, a_new, 0.0)
        total = a_new.sum()
        if total == 0.0:

            a_new = np.random.rand(n)
            a_new = a_new / a_new.sum()
        else:
            a_new /= total

"""
Propose a step‑function (non‑negative sequence) that lowers the value of
evaluate_sequence.  The idea is to treat the evaluation
    score = 2·n·max(conv(a,a)) / (sum(a)²)
as a surrogate for the maximal entry of the convolution.  For a fixed sum
(e.g. 1) the score is proportional to max(conv).  We minimise the
squared deviation of the full convolution from its average value,
which flattens the convolution and consequently reduces its maximum.

The algorithm:
  * Initialise a uniform vector a (sum = 1) of the same length as the
    previously known construction (height_sequence_1) or a default size.
  * In each iteration compute the self‑convolution via an FFT
    (cost O(n log n)).
  * Build the residual `diff = conv - avg`, where `avg = sum(a)²/(2n‑1)`.
  * Compute the gradient of ‖diff‖² w.r.t. a using another FFT
    (cross‑correlation of `diff` with a).  The gradient is
    `4 * c`, where `c_i = Σ_l diff[i+l]·a[l]`.
  * Take a projected gradient step:
        a ← a – lr·gradient
    clip negatives to zero and renormalise to keep the total sum equal to
    one.
  * Accept the step if it improves the score, otherwise keep the old
    vector and shrink the learning rate.  A tiny simulated‑annealing
    mechanism (temperature that slowly cools) allows occasional uphill
    moves to escape plateaus.
  * Occasionally apply a tiny random redistribution of mass between two
    coordinates to inject stochasticity.
  * Track the best vector seen so far and stop early as soon as a score
    below the target (1.5) is achieved or the time budget expires.

All heavy work is done with NumPy’s FFT which is fast enough to perform
many thousands of iterations within the 1000 s limit.
"""

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 990, **kwargs):
    """Return a list of non‑negative heights with a low evaluation score.

    The function runs a projected gradient descent on the squared deviation
    of the self‑convolution from a constant, using FFTs for speed.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Determine the length of the sequence.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # use the existing construction if present
    except NameError:
        n = 2000                            # fallback length

    # ------------------------------------------------------------------
    # 2. Initialise a uniform distribution (sum = 1).
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)
    a /= a.sum()                           # guarantee exact sum = 1

    # Best solution seen so far
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    current_score = best_score

    # ------------------------------------------------------------------
    # 3. FFT helper: smallest power of two ≥ x
    # ------------------------------------------------------------------
    def next_pow2(x):
        x = int(x)
        return 1 << (x - 1).bit_length()

    # Length sufficient for the two convolutions we need:
    #   a * a  (length 2n‑1)   and   a * diff_rev (length 3n‑2)
    L = next_pow2(3 * n)

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    # ------------------------------------------------------------------
    lr = 1.0                # learning rate for the gradient step
    temp = 0.1              # temperature for occasional uphill moves
    decay = 0.999           # annealing schedule for the temperature
    min_lr = 1e-6
    max_lr = 5.0

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin
    iteration = 0

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- self‑convolution a * a (full linear convolution) -----
        A = np.fft.rfft(a, n=L)
        conv_self = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

        max_b = conv_self.max()
        # Since we keep sum(a) == 1, the score is simply 2*n*max_b
        score = 2.0 * n * max_b

        # ----- keep the best vector found -----
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"Iter {iteration:5d}: new best score {best_score:.6f}  "
                f"time {time.time() - start_time:.1f}s"
            )
            if best_score <= 1.5:                     # target reached
                break

        # ----- target constant (average of the convolution) -----
        sum_a = a.sum()                     # should be 1.0
        avg = (sum_a * sum_a) / (2 * n - 1) # = 1/(2n‑1) when sum_a == 1

        # ----- residual from the desired constant -----
        diff = conv_self - avg
        diff_rev = diff[::-1]

        # ----- gradient of ||diff||² w.r.t. a (via FFT) -----
        D = np.fft.rfft(diff_rev, n=L)
        conv_grad = np.fft.irfft(A * D, n=L)          # linear conv a * diff_rev
        # c_i = conv_grad[2n‑2‑i]   for i = 0 … n‑1
        idx_start = 2 * n - 2
        c = conv_grad[idx_start - np.arange(n)]
        grad = 4.0 * c

        # ----- propose a gradient step -----
        step = lr * grad
        a_new = a - step
        a_new = np.where(a_new > 0.0, a_new, 0.0)     # enforce non‑negativity
        total = a_new.sum()
        if total == 0.0:                               # degenerate case → random restart
            a_new = np.random.rand(n)
            a_new /= a_new.sum()
        else:
            a_new /= total                             # renormalise to sum = 1

        # ----- evaluate the candidate -----
        A_new = np.fft.rfft(a_new, n=L)
        conv_self_new = np.fft.irfft(A_new * A_new, n=L)[: 2 * n - 1]
        max_b_new = conv_self_new.max()
        score_new = 2.0 * n * max_b_new

        # ----- acceptance rule (simulated annealing) -----
        if (score_new < score) or (np.random.rand() < math.exp(-(score_new - score) / max(temp, 1e-12))):
            a = a_new
            current_score = score_new
            # modestly increase learning rate after a successful move
            lr = min(lr * 1.02, max_lr)
        else:
            # No improvement – shrink the step size
            lr = max(lr * 0.9, min_lr)

        # ----- temperature cooling -----
        temp *= decay

        # ----- occasional random jitter to avoid stagnation -----
        if iteration % 50 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.02 * a[i]   # up to ±1 % of a[i]
            if a[i] + delta >= 0.0:
                a[i] += delta
                a[j] -= delta
                a = np.clip(a, 0.0, None)
                if a.sum() > 0.0:
                    a /= a.sum()

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
