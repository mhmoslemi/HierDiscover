def simplex_projection(v):
    """Project v onto simplex {x >=0, sum x = 1}"""
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all zero? just set uniform
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
A simple but effective optimizer for the step‑function bound.

The idea:
* The evaluation metric is 2·n·max(conv(p,p)) where p is a
  probability vector (sum(p)=1, p_i≥0).  Uniform p yields the value 2.
* To push the bound below 2 we need a distribution whose self‑convolution
  is as flat as possible – i.e. we minimise the maximum entry of the
  convolution.
* The continuous analogue suggests a density that is large at the edges
  and small in the centre (an arcsine‑type shape).  Starting from this
  shape we perform a projected gradient descent on the functional
  max(conv(p,p)).  The gradient of a single convolution entry
  (p*p)[t] is simply 2·shift(p,t), so a cheap sub‑gradient is available.
* After each gradient step we project back on the simplex (non‑negative,
  sum = 1).  Small random perturbations are occasionally injected to
  avoid stagnation.
* The algorithm runs for the allotted time budget (≈ 1000 s) and returns
  the best sequence found.  If the optimisation does not improve on the
  provided baseline, the baseline is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex
# ------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]                 # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Helper: shift vector needed for the sub‑gradient of a convolution entry
# ------------------------------------------------------------
def shift_by(p: np.ndarray, t: int) -> np.ndarray:
    """
    Return a vector q such that q[j] = p[t-j] if 0≤t-j<n, otherwise 0.
    This is precisely the gradient of (p*p)[t] with respect to p (up to the
    factor 2).
    """
    n = p.shape[0]
    q = np.zeros_like(p)
    lo = max(0, t - (n - 1))
    hi = min(t, n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        q[idx] = p[t - idx]
    return q


# ------------------------------------------------------------
# Main entry point required by the evaluator
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The method starts from an arcsine‑shaped
    distribution and performs a projected sub‑gradient descent with a
    simple simulated‑annealing style perturbation scheme.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine the length n to work with.
    # Prefer the length of the provided baseline (height_sequence_1) if
    # it exists, otherwise fall back to a modest size.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # provided by the judge
        baseline = np.array(height_sequence_1, dtype=float)
    except Exception:
        n = 2000
        baseline = None

    # ------------------------------------------------------------------
    # Initialise a probability vector p with an arcsine‑type shape:
    #   p_i ∝ 1 / sqrt( (i+½)*(n‑i‑½) )
    # This makes the edges heavy and the centre light, flattening the
    # self‑convolution.
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=float)
    denom = np.sqrt((i + 0.5) * (n - i - 0.5))
    p = 1.0 / denom
    p /= p.sum()                     # now sum(p) == 1

    # keep the best ever seen (objective = 2·n·max(conv))
    best_p = p.copy()
    best_val = evaluate_sequence(best_p.tolist())
    if baseline is not None:
        base_val = evaluate_sequence(baseline.tolist())
        if base_val < best_val:
            best_p = baseline.copy()
            best_val = base_val

    # ------------------------------------------------------------------
    # Parameters for the optimisation loop
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # leave a tiny safety margin
    max_iters = 200_000                       # generous upper bound
    it = 0

    # step size decays linearly from 0.5 to 0.001
    step_start = 0.5
    step_end = 0.001

    # occasional random resets after long periods without improvement
    last_improve = start_time

    while time.time() < deadline and it < max_iters:
        # --------------------------------------------------------------
        # current objective (no need to call evaluate_sequence each step)
        # --------------------------------------------------------------
        conv = np.convolve(p, p)
        maxc = conv.max()
        cur_val = 2.0 * n * maxc

        # record improvement
        if cur_val < best_val:
            best_val = cur_val
            best_p = p.copy()
            last_improve = time.time()
            print(f"[{time.time() - start_time:6.1f}s] new best: {best_val:.6f}")

        # --------------------------------------------------------------
        # Sub‑gradient step for the maximal convolution entry
        # --------------------------------------------------------------
        t_max = int(conv.argmax())
        grad = 2.0 * shift_by(p, t_max)   # sub‑gradient of max(conv)

        # linear step‑size schedule
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 2.0)
        step = step_start * (1.0 - frac) + step_end * frac

        # candidate after a projected gradient step
        cand = simplex_projection(p - step * grad)

        # evaluate candidate
        conv_cand = np.convolve(cand, cand)
        maxc_cand = conv_cand.max()
        if maxc_cand <= maxc:
            p = cand
        else:
            # occasional Metropolis‑type acceptance to escape shallow traps
            if np.random.rand() < 0.0005:
                p = cand

        # --------------------------------------------------------------
        # Small random mass‑transfer perturbation (helps to explore)
        # --------------------------------------------------------------
        if np.random.rand() < 0.001:
            i1, i2 = np.random.randint(n, size=2)
            if i1 != i2:
                delta = (np.random.rand() - 0.5) * 0.01
                delta = np.clip(delta, -p[i2], p[i1])  # keep non‑negative
                p[i1] -= delta
                p[i2] += delta

        # --------------------------------------------------------------
        # If no improvement for a while, re‑seed around the best point
        # --------------------------------------------------------------
        if time.time() - last_improve > 30.0:
            # add a tiny amount of noise to the best solution found so far
            p = best_p + 0.005 * np.random.rand(n)
            p = simplex_projection(p)
            last_improve = time.time()

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (convert to plain Python list).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_p, 0.0, 1000.0)   # respect the evaluator's caps
    return best_seq.tolist()
