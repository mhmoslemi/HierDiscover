"""
Propose a candidate step‐function (non‑negative sequence) that minimizes
the evaluation function:

    score = 2 * n * max(conv(a, a)) / (sum(a)^2)

The algorithm is a simple continuous optimisation on the simplex.
It repeatedly smooths the max‑convolution via a soft‑max (log‑sum‑exp)
and takes a projected gradient step that reduces the weighted
convolution. Random noise and occasional restarts help escape the
uniform stationary point.  Multiple sequence lengths are tried; the best
sequence found before the time budget expires is returned.
"""

import time
import numpy as np
import random

def _softmax_weights(conv, alpha):
    """
    Return a soft‑max weight vector w such that w_k ≈ 1 for the
    largest convolution entry and quick to compute.
    """
    # shift by the maximum for numerical stability
    max_c = conv.max()
    exp_w = np.exp(alpha * (conv - max_c))
    return exp_w / exp_w.sum()


def _gradient_step(a, eta, alpha):
    """
    Perform one projected gradient step:
        a <- a - eta * grad  (grad of soft‑max weighted convolution)
    The gradient of sum_k w_k * (a * a)_k w.r.t. a is
        g_i = 2 * sum_k w_k * a_{k-i}  (valid indices only)
    which can be written as a correlation.
    """
    # full convolution a*a
    conv = np.convolve(a, a)                     # length 2n-1
    # soft‑max over convolution entries
    w = _softmax_weights(conv, alpha)            # length 2n-1, sum(w)=1
    # gradient = 2 * correlate(w, a, mode='valid')
    grad = 2.0 * np.correlate(w, a, mode='valid')
    # gradient step
    a_new = a - eta * grad
    # add a tiny random perturbation to avoid exact symmetry
    a_new += np.random.randn(a.shape[0]) * (eta * 0.02)
    # project onto the feasible set: non‑negative + sum = 1
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:                 # safety fallback (should not happen)
        a_new = np.full_like(a_new, 1.0 / a_new.size)
    else:
        a_new /= s
    return a_new


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score sequence within the given time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    random.seed(seed)

    # time management
    end_time = time.time() + budget_s - 0.5   # small safety margin

    best_score = float("inf")
    best_seq = None

    # a few different lengths – short ones allow many iterations,
    # longer ones give more flexibility.
    lengths = [50, 100, 200, 400, 800, 1200]

    # try each length while we still have time left
    for n in lengths:
        if time.time() >= end_time:
            break

        # initialization: Dirichlet with concentration <1 gives diverse shapes
        a = np.random.dirichlet(np.ones(n) * 0.5)

        # optimisation hyper‑parameters (will be adapted)
        eta = 0.8          # initial step size
        alpha = 5.0        # initial soft‑max sharpness
        last_improve = 0   # iteration index of last improvement

        # a generous but still bounded number of iterations per length
        for it in range(15000):
            if time.time() >= end_time:
                break

            # one gradient step
            a = _gradient_step(a, eta, alpha)

            # evaluate current candidate
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score

                last_improve = it
