    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum = 1, v_i >= 0)."""
    # reference: https://doi.org/10.1016/j.neunet.2007.09.003
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero, return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def smooth_max_gradient(a, alpha=5.0):
    """Return gradient of the smooth soft‑max of a∗a and the raw convolution."""
    b = np.convolve(a, a)
    max_b = np.max(b)
    # soft‑max weights, stable with subtraction of max_b
    w = np.exp(alpha * (b - max_b))
    w_sum = np.sum(w)
    p = w / w_sum
    # gradient = (p * a) convolution with reversed a, sliced appropriately
    grad = np.convolve(p, a[::-1])[len(a)-1 : len(a)-1 + len(a)]
    return grad, b

def propose_candidate(seed=42, budget_s=1000):
    """Return a non‑negative step‑function (list of heights) that tries to minimise
    evaluate_sequence.  The method uses an exponentiated‑gradient scheme that
    directly reduces the maximum entry of the auto‑convolution while keeping the
    L1‑norm fixed (projection onto the simplex).  It runs until the time budget
    expires and returns the best sequence seen."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # leave a tiny margin
    
    # length of the candidate sequence – a few thousand gives a good trade‑off
    n = 2000
    
    # initialise with a random point on the simplex
    a = np.random.rand(n)
    a = project_simplex(a)
    
    best_a = a.copy()
    # fast evaluation because sum(a)=1 => score = 2 * n * max(conv)
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    print(f'initial score (fast): {best_score:.6f}')
    
    # hyper‑parameters (they will be annealed)
    eta = 0.5          # step size for exponentiated update
    alpha = 5.0        # soft‑max sharpness
    it = 0
    
    while time.time() < deadline:
        grad, b = smooth_max_gradient(a, alpha=alpha)
        # exponentiated‑gradient update, preserving positivity
        a = a * np.exp(-eta * grad)
        # project back onto the simplex (L1 = 1)
        a = project_simplex(a)
        
        # occasional evaluation of the true objective (cheap fast version)
        # because sum is exactly 1 we can compute the score directly.
        cur_score = 2.0 * n * np.max(b)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f'iter {it:6d} – new best score {best_score:.6f}')
            # early exit if we already beat the target
            if best_score <= 1.503:
                break
        
        # annealing schedule

"""Optimise a step‑function to minimise `evaluate_sequence`.

The routine keeps the L¹‑norm of the heights fixed (projected onto the
probability simplex) and repeatedly applies an exponentiated‑gradient
step that reduces the (smooth) maximum of the auto‑convolution `a*a`.
A soft‑max with a controllable sharpness `alpha` provides a differentiable
approximation of the max, and the gradient is obtained via a second
convolution.  The learning rate `eta` and the soft‑max temperature are
annealed, occasional jitter avoids stagnation and the best sequence found
within the time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex
    (non‑negative entries that sum to 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):               # all zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float = 5.0):
    """Gradient of the smooth soft‑max of the auto‑convolution `a*a`.

    Returns
    -------
    grad : np.ndarray
        ∂/∂a  ( (1/α)·log Σ exp(α·(a*a)) )
    b : np.ndarray
        The raw convolution a*a (length 2·n‑1).
    """
    b = np.convolve(a, a)                     # linear convolution
    max_b = np.max(b)

    # stable soft‑max weights (max_b is subtracted)
    w = np.exp(alpha * (b - max_b))
    w_sum = np.sum(w)
    p = w / w_sum                              # probability vector, length 2n‑1

    # gradient = (p * a) convolved with reversed a, sliced to length n
    #   grad_i = Σ_j p_{i+j} a_j
    conv = np.convolve(p, a[::-1])             # length 3n‑2
    n = a.shape[0]
    grad = conv[n - 1 : n - 1 + n]              # exactly n entries
    return grad, b


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) achieve a low
    value of `evaluate_sequence`.

    The algorithm runs until the supplied time budget (in seconds) expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size – a few thousand entries give a good trade‑off between
    # solution quality and per‑iteration cost.
    n = 2000

    # initialise a random point on the simplex (L¹‑norm = 1)
    a = np.random.rand(n)
    a = _project_simplex(a)

    # keep the best vector seen so far
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))   # fast proxy (sum=1)
    print(f'initial fast score: {best_score:.6f}')

    # hyper‑parameters (they will be annealed during the run)
    eta = 0.5          # step size for the exponentiated‑gradient update
    alpha = 5.0        # soft‑max sharpness (larger → closer to true max)
    it = 0

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 1) compute gradient of the smooth max of the convolution
        grad, b = _smooth_max_gradient(a, alpha=alpha)

        # 2) exponentiated‑gradient step (keeps entries positive)
        a = a * np.exp(-eta * grad)

        # 3) project back onto the simplex (preserves Σ a_i = 1)
        a = _project_simplex(a)

        # cheap evaluation (exact because Σ a_i = 1)
        cur_score = 2.0 * n * np.max(b)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f'iter {it:6d} – new best score {best_score:.6f}')
            if best_score <= 1.503:      # stop early if target already met
                break

        # anneal parameters every few hundred iterations
        if it % 200 == 0 and it > 0:
            eta *= 0.9            # slower steps as we approach a minimum
            alpha = min(alpha * 1.05, 50.0)   # sharpen the soft‑max slowly

        if it % 500 == 0 and it > 0:
            jitter = np.random.randn(n) * 0.001
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

        it += 1

    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    print(f'finished after {it} iterations, best score {best_score:.6f}')
    return final_seq
