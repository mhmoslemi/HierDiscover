import numpy as np
import time

def propose_candidate(budget_s=1000, seed=0, **kwargs):
    """Simple simulated annealing optimizer for the step‑function evaluation.

    The algorithm starts from a random Dirichlet vector whose sum is
    sqrt(2·n) and then performs a large number of pairwise mass‑transfer
    moves while keeping the total sum constant.  Moves that improve the
    evaluation are always accepted; worsening moves are accepted with a
    Metropolis probability that slowly cools down.  The best sequence seen
    before the time limit is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # length of the sequence – use the provided construction if present,
    # otherwise fall back to 1153 (the length used by the benchmark).
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 1153

    target_sum = np.sqrt(2.0 * n)          # the normalisation used by the evaluator

    # start from a random Dirichlet distribution (guarantees non‑negative,
    # sum == target_sum).  Adding a small amount of the uniform vector helps
    # to stay away from pathological spikes.
    base = np.random.dirichlet(np.ones(n))
    uniform = np.full(n, 1.0 / n)
    cur = (0.9 * base + 0.1 * uniform) * target_sum

    # initial evaluation
    cur_score = evaluate_sequence(cur.tolist())
    best_seq = cur.copy()
    best_score = cur_score

    # temperature schedule
    T0 = 0.1
    it = 0
    while time.time() < deadline:
        it += 1
        # pick two distinct indices and move a small amount from i to j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # proportion of mass to move – limited to at most 5 % of the donor amount
        max_move = min(cur[i], 0.05 * cur[i])
        if max_move <= 0:
            continue
        delta = np.random.rand() * max_move
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # occasionally perform a bigger random perturbation to explore
        if np.random.rand() < 0.001:
            # replace a random block of the vector by a fresh Dirichlet sample
            block_start = np.random.randint(0, n)
            block_len = min(n - block_start, np.random.randint(1, 20))
            block = np.random.dirichlet(np.ones(block_len))
            new[block_start:block_start+block_len] = block * target_sum * block_len / n

        # keep the total sum exactly target_sum (numerical drift)
        # (the move preserves the sum, the block replacement may not)
        new_sum = new.sum()
        if new_sum != 0:
            new *= target_sum / new_sum

        # evaluate the candidate
        try:
            new_score = evaluate_sequence(new.tolist())
        except Exception:
            continue

        # accept if better, otherwise accept with Metropolis probability
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()
                print(f"[{time.time():.2f}] New best score: {best_score:.6f} (iter {it})")
        else:
            # cooling schedule
            elapsed = time.time() - (deadline - budget_s + 5.0)
            frac = min(1.0, elapsed / (budget_s - 5.0))
            T = T0 * (1.0 - frac)
            # Metropolis acceptance
            prob = np.exp((cur_score - new_score) / (T + 1e-12))
            if np.random.rand() < prob:
                cur = new
                cur_score = new_score

    print(f"Finished optimisation. Best score found: {best_score:.6f}")
    # Return the best sequence as a plain Python list of floats
    return best_seq.tolist()
