import time
import numpy as np

# ----------------------------------------------------------------------
#  Propose a sequence that minimises `evaluate_sequence`.
#
#  The algorithm works with a probability vector `p` (sum(p)=1).  The
#  objective  max_k (p * p)_k  is approximated by a smooth p‑norm:
#
#        f(p) = ( Σ_k ( (p * p)_k )^q )^{1/q}       (q >> 1)
#
#  Its gradient can be written in closed form and is obtained by a
#  correlation between the current convolution and `p`.  Projected
#  gradient descent (with a projection onto the non‑negative simplex)
#  is used, together with occasional random perturbations and several
#  random restarts.  The best vector found during the allotted time
#  (default 1000 s) is returned as a plain Python list; scaling does not
#  affect the score, therefore the unit‑sum vector can be used directly.
# ----------------------------------------------------------------------


def _softmax_weights(conv, q):
    """Return w_k ∝ (conv_k)^{q‑1} used in the gradient of the p‑norm."""
    eps = 1e-15
    # log‑space avoids underflow for large q
    logb = np.log(conv + eps)
    w = np.exp((q - 1) * logb)
    return w


def _gradient(p, conv, q):
    """
    Gradient of the smoothed max‑convolution objective.
    g_i = 2 Σ_k w_k * p_{k-i}
    where w_k = conv_k^{q-1} (up to a common factor).
    """
    w = _softmax_weights(conv, q)
    # correlation (w ⋆ p)[k] = Σ_i w_{i+k} p_i
    corr = np.correlate(w, p, mode="full")
    n = p.shape[0]
    # take the n entries that correspond to k = 0 … n‑1
    g = 2.0 * corr[(n - 1):(n - 1 + n)]
    return g


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """

    `evaluate_sequence`.  The function respects the time budget and
    returns the best sequence found (as a list of floats).
    """

    # ------------------------------------------------------------------
    # Length of the sequence: use the length of the previously supplied
    # construction if it exists, otherwise fall back to the benchmark size.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # provided by the surrounding environment
    except NameError:
        n = 1153                          # default length used in the benchmark

    deadline = time.time() + budget_s - 5.0   # a few seconds margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Small helper to compute the score for a unit‑sum vector.
    # ------------------------------------------------------------------
    def _score(p_vec):
        conv = np.convolve(p_vec, p_vec)
        return float(2 * n * conv.max())   # sum(p)=1  ⇒  score = 2·n·max(conv)

    # ------------------------------------------------------------------
    # Main optimisation loop – several random restarts.
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        restart += 1

        # ---- initialise ------------------------------------------------
        if restart == 1:
            # a deterministic start – the uniform distribution (score 2.0)
            p = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            # Dirichlet‑like random start (α = 0.5 gives a fairly spread vector)
            alpha = np.full(n, 0.5, dtype=np.float64)
            p = np.random.gamma(alpha, 1.0)
            p /= p.sum()

        # ---- parameters for the inner optimisation --------------------
        q = 12                # exponent for the soft‑max (larger → sharper)
        lr = 0.5              # initial (unnormalised) step size
        decay = 0.9995        # geometric decay of lr each iteration

        # ---- projected gradient descent --------------------------------
        while time.time() < deadline:
            cur_score = _score(p)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = p.copy()
                print(f"[{time.time():.2f}] restart {restart} – new best: {best_score:.6f}")

                # stop early if we already beat the target
                if best_score <= 1.503:
                    deadline = time.time()   # break outer loops as well
                    break

            # convolution of the current probability vector
            conv = np.convolve(p, p)

            # gradient with respect to the smoothed objective
            g = _gradient(p, conv, q)

            # normalise the step size so that the update magnitude stays bounded
            grad_norm = np.linalg.norm(g)
            if grad_norm == 0.0:
                break
            step = lr / grad_norm

            # projected gradient step onto the non‑negative simplex

                s = 1.0
            p = p_new / s
