Implementation of projection onto simplex:

def project_onto_simplex(v):
    """Projects a vector onto the probability simplex (nonnegative sum=1)."""
    # v shape (n,)
    # reference: algorithm from https://arxiv.org/abs/1309.1541
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def compute_score(seq):
    n = len(seq)
    # seq sum is S
    S = np.sum(seq)
    if S < 1e-9:
        return np.inf
    m = 1 << ((2*n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft*fft, n=m)[:2*n-1]
    max_conv = conv.max()
    return float(2*n * max_conv / (S**2))

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = len(height_sequence_1)
    # initial sequence

    seq = np.maximum(seq, 0)

    cur_score = compute_score(seq)
    best_seq = seq.copy()
    best_score = cur_score
    iter_count = 0
    temp = 0.1 * cur_score  # initial temperature

    while time.time() < deadline:
        iter_count += 1

        # compute k_max for current sequence
        conv = compute_conv(seq)  # returns array
        k_max = conv.argmax()

        # compute contributions
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        shifted = np.zeros_like(seq)
        shifted[mask] = seq[j[mask]]
        contrib = seq * shifted

        source_candidates = np.where(contrib > 0)[0]
        if source_candidates.size == 0:
            source = np.random.randint(n)
        else:
            source = source_candidates[np.argmax(contrib[source_candidates])]

            # pick random source with non-zero

"""
Propose a candidate sequence for the step‑function minimisation problem.

Algorithm
---------
1. Initialise a near‑uniform distribution of non‑negative heights (sum = 1).
2. Repeatedly perform a simulated‑annealing / greedy hill‑climbing move:
   * Compute the auto‑convolution (via FFT) and locate the index `k_max`
     where the convolution attains its maximum.
   * For this `k_max` compute the per‑coordinate contribution
     `c_i = a_i * a_{k_max-i}` (zero if the partner index is out of range).
   * Choose a “source’’ index with large contribution and a “target’’ index
     with small contribution and transfer a small fraction of the source
     mass to the target (preserving the total sum).
   * Accept the move if it improves the objective, otherwise accept
     with a temperature‑dependent probability (simulated annealing).
3. Keep the best sequence seen so far.
4. Stop after the allotted time (≈budget_s – few seconds) and return the best
   sequence as a plain Python list.

The objective for a normalised sequence (∑a_i = 1) is simply
    f(a) = 2·n·max(conv(a,a)),
so minimising the maximum entry of the auto‑convolution directly reduces
the score returned by `evaluate_sequence`.  FFT is used for fast O(n log n)
convolution, enabling thousands of iterations within the time budget.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper functions (top‑level as required)
# -------------------------------------------------------------

def compute_conv(seq: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution of `seq` with itself using FFT."""
    n = len(seq)
    # length needed for linear convolution = 2*n-1, pad to next power of two
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    return conv


def compute_score(seq: np.ndarray) -> float:
    """
    Exact value of the evaluation function (without the safety checks)
    for a given sequence.  This is faster than calling `evaluate_sequence`
    because it avoids Python‑level list conversion and extra clipping.
    """
    S = seq.sum()
    if S < 1e-9:
        return float("inf")
    conv = compute_conv(seq)
    max_conv = conv.max()
    n = len(seq)
    return float(2.0 * n * max_conv / (S * S))

    perturb_interval = 5000

    # -----------------------------------------------------------------
    seq = np.full(n, 1.0 / n) + 0.001 * np.random.rand(n)
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()                     # keep the sum = 1 (scale‑invariant)

    cur_score = compute_score(seq)
    best_seq = seq.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # temperature for the Metropolis acceptance rule
    temperature = 0.1 * cur_score if cur_score > 0 else 1.0

    deadline = time.time() + budget_s - 5  # keep a few seconds margin
    iteration = 0

        # -------------------------------------------------------------
        # 1) Compute the current convolution and locate the worst lag k_max
        # -------------------------------------------------------------

        source_candidates = np.where(contrib > 0)[0]

            cur_score = compute_score(seq)

    # Return the best sequence found (as a plain Python list of floats)
    # -----------------------------------------------------------------
    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    return best_seq.tolist()
