"""
Propose candidate step‑function heights.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the linear convolution implemented by ``np.convolve``.
For a fixed L¹‑norm ``sum(a)`` the quantity to minimise is the
ℓ∞‑norm of the self‑convolution.  The trivially optimal construction is the
uniform vector (all entries equal) which yields a score of exactly 2.
Below we perform a fast simulated‑annealing / random‑walk search starting
from a uniform vector (or from the provided ``height_sequence_1`` if it is
available).  The routine tries to lower the score; if it cannot improve
the uniform benchmark it simply returns the uniform candidate.

The algorithm works under the 1000 s time budget, keeps all entries
non‑negative, respects the clipping limits of ``evaluate_sequence`` and
periodically reports the current best score.
"""

import time
import random
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: tiny deterministic copy of the LP routine used by the SOTA code.
# It is kept here because the optimisation occasionally needs a
# “good direction’’ that respects the current convolution bound.
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, rhs, time_limit=5.0):
    """
    Linear programme:
        maximise  Σ b_j
        subject to  (f_seq * b)_k ≤ rhs   for all k
                    b_j ≥ 0
    Returns the optimal vector b or None on failure.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # maximise Σ b → minimise -Σ b
    # Build A_ub, b_ub for each convolution inequality
    A = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        A.append(row)
        b.append(rhs)
    # non‑negativity constraints  -b ≤ 0
    A.append(-np.eye(n))
    b.append(np.zeros(n))
    A = np.vstack(A)
    b = np.hstack(b)

    # SciPy's linprog respects a time limit via the `options` dict.
    res = optimize.linprog(c, A_ub=A, b_ub=b,
                           method='highs',
                           options={'time_limit': time_limit, 'disp': False})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best bound.  The routine runs for at most ``budget_s`` seconds
    and prints progress information.
    """
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 5.0   # a safety margin

    # ------------------------------------------------------------------
    # 1) Initialise the sequence.
    #    Prefer the provided SOTA construction if it exists, otherwise start
    #    from a uniform vector of moderate length.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is expected to be defined in the global scope.
        init_seq = list(height_sequence_1)  # type: ignore[name-defined]
        n = len(init_seq)
        print(f"Starting from provided sequence (len={n})")
    except Exception:
        # Fall back to a uniform vector; length between 500‑1000 is a good trade‑off.
        n = random.randint(500, 1000)
        init_seq = [1.0] * n
        print(f"No external sequence found – using uniform init (len={n})")

    # Normalise to a convenient L¹‑norm (the actual scale does not affect the score).
    init_seq = np.array(init_seq, dtype=float)
    init_seq = init_seq / np.sum(init_seq)   # now sum == 1.0

    best_seq = init_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Main stochastic optimisation loop.
    #    A simple “move a tiny amount of mass from i → j’’ neighbourhood
    #    is used.  Occasionally a LP‑derived direction (the SOTA idea) is
    #    injected to escape plateaus.
    # ------------------------------------------------------------------
    current_seq = best_seq.copy()
    temperature = 1.0      # annealing temperature – controls step size
    it = 0

    while time.time() < deadline:
        it += 1

        # Occasionally try a large LP‑step (every ~200 iterations)
        if it % 200 == 0:
            # Normalise the current sequence (required by the LP routine)
            sum_cur = np.sum(current_seq)
            if sum_cur == 0:
                continue
            norm_cur = current_seq * np.sqrt(2 * n) / sum_cur
            rhs = float(np.max(np.convolve(norm_cur, norm_cur)))
            g = solve_convolution_lp(norm_cur, rhs, time_limit=2.0)
            if g is not None:
                # Scale g back to the same L¹‑norm as the current sequence
                sum_g = np.sum(g)
                if sum_g > 0:
                    g = g * np.sqrt(2 * n) / sum_g
                    # A small convex combination with the current vector
                    t = 0.02
                    new_seq = (1 - t) * current_seq + t * g
                    new_seq = np.maximum(new_seq, 0.0)
                    # Re‑normalise to keep sum = 1
                    new_seq /= np.sum(new_seq)
                    current_seq = new_seq
                    cur_score = evaluate_sequence(current_seq.tolist())
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = current_seq.copy()
                        print(f"[{it}] LP step improved score to {best_score:.6f}")
                    continue    # skip the random move for this iteration

        # ------------------------------------------------------------------
        # Random small perturbation: move a tiny fraction of mass from i → j.
        # ------------------------------------------------------------------
        i, j = random.sample(range(n), 2)
        if current_seq[i] <= 1e-12:
            continue
        # Fraction of the source mass to transfer (scaled by temperature)
        delta = temperature * random.random() * current_seq[i]
        # Apply move
        new_seq = current_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # Numerical safety – clip to the allowed range
        new_seq = np.clip(new_seq, 0.0, 1000.0)

        # Normalise back to sum == 1 (scale invariance of the score)
        new_seq_sum = np.sum(new_seq)
        if new_seq_sum < 1e-12:
            continue
        new_seq /= new_seq_sum

        # ------------------------------------------------------------------
        # Evaluate and keep the move if it improves the objective.
        # ------------------------------------------------------------------
        cur_score = evaluate_sequence(new_seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            current_seq = new_seq
            # Cool down a little when we succeed
            temperature *= 0.99
            if it % 50 == 0:
                print(f"[{it}] Improved score = {best_score:.6f}")
        else:
            # Occasionally accept a worse move (simulated‑annealing spirit)
            # to avoid being trapped in a flat region.
            prob = np.exp(-(cur_score - best_score) / (temperature + 1e-12))
            if random.random() < prob:
                current_seq = new_seq
                temperature *= 0.995

        # Gradually lower the temperature – ensures convergence.
        if it % 500 == 0:
            temperature = max(temperature * 0.9, 1e-4)

    # ------------------------------------------------------------------
    # 3) Return the best sequence discovered (as a plain Python list).
    # ------------------------------------------------------------------
    print(f"Finished after {it} iterations – best score = {best_score:.6f}")
    # Scale back to a convenient magnitude (the evaluator clips anyway).
    final_seq = (best_seq * 100.0).tolist()   # any positive scale works
    return final_seq
