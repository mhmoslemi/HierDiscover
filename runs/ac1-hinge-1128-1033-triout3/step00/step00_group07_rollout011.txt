"""
Propose a step–function (non‑negative sequence) that minimises the
evaluate_sequence objective.

The algorithm builds several candidate sequences of moderate length,
adds a small random perturbation and then performs a projected gradient
descent on a *smoothed* version of the max‑convolution term.
The smoothed max is   S_λ(a)= (log Σ exp(λ·(a*a))) / λ   (log‑sum‑exp),
whose gradient can be written in closed form using a single convolution.
For a fixed λ the full objective is

    F(a)= 2·n·S_λ(a) / (∑a)² .

The gradient of F is obtained analytically, a step is taken, the
result is clipped to non‑negative values and re‑normalised to keep the
ℓ₁‑norm equal to one (scaling does not affect the objective).
A simple back‑tracking line‑search guarantees decrease of the true
objective (max‑convolution, not the smoothed version).  Random restarts
and occasional noise injections avoid getting stuck in poor local minima.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, lam: float):
    """
    Compute a smooth approximation of max(convolve(a,a)) together with its
    gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Non‑negative coefficients, sum(a) > 0.
    lam : float
        Smoothing parameter (larger → closer to the true max).

    Returns
    (S, grad_S, maxb) where
        S      – smooth max value,
        grad_S – ∇_a S,
        maxb   – the exact max of the convolution (needed for the true
                 objective).
    """
    n = a.shape[0]
    # exact convolution and its max
    b = np.convolve(a, a)                 # length 2n‑1
    maxb = b.max()

    # smooth max via log‑sum‑exp
    lam_b = lam * b
    lse = logsumexp(lam_b)                # log Σ exp(λ·b_k)
    S = lse / lam                         # smooth approximation of max(b)

    # weights of the soft‑max (they sum to 1)
    w = np.exp(lam_b - lse)               # length 2n‑1

    # gradient of the smooth max:
    #   ∇S = Σ_k w_k ∇b_k,   ∇b_k[i] = 2·a_{k‑i} (if index in range)
    # This weighted sum can be computed by a single convolution.
    w_rev = w[::-1]                       # reversed weight vector
    conv = np.convolve(a, w_rev)          # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)      # positions that give Σ_j a_j w_{i+j}
    grad_S = 2.0 * conv[idx]              # shape (n,)

    return S, grad_S, maxb

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (the best sequence found before the
    time budget runs out).  The routine runs a multi‑restart projected
    gradient descent using the smoothed max‑convolution objective.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # leave a small safety margin

    # ------------------------------------------------------------------
    # helper to compute the true objective (no need to call evaluate_sequence)
    def objective(a):
        n = a.shape[0]
        s = a.sum()
        if s < 1e-12:
            return np.inf
        maxb = np.convolve(a, a).max()
        return 2.0 * n * maxb / (s * s)

    # ------------------------------------------------------------------
    # Initialise a few candidates (uniform, random, and possibly the
    # previously known construction `height_sequence_1`).
    candidates = []

    # uniform candidates of several lengths
    for n in (64, 128, 256, 512):
        a = np.full(n, 1.0 / n, dtype=float)
        candidates.append(a)

    # random candidates
    for _ in range(6):
        n = np.random.randint(200, 800)
        a = np.random.rand(n)
        a = a / a.sum()
        candidates.append(a)

    # include the SOTA construction if it exists in the global namespace
    if "height_sequence_1" in globals():
        try:
            a = np.array(height_sequence_1, dtype=float)
            if a.ndim == 1 and a.size > 0:
                a = np.maximum(a, 0.0)
                a = a / max(a.sum(), 1e-12)
                candidates.append(a)
        except Exception:
            pass

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # pick a candidate and give it a slight random kick
        a = candidates[np.random.randint(len(candidates))].copy()
        if np.random.rand() < 0.3:
            # perturb a small random subset
            nn = a.shape[0]
            idx = np.random.choice(nn, size=max(1, nn // 50), replace=False)

            if s < 1e-12:

            S, grad_S, maxb = _smooth_max_and_grad(a, lam)

            # true objective (used for acceptance checks)
            cur_score = 2.0 * n * maxb / (s * s)

            # store the best sequence seen so far
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[{time.time() - start_time:5.1f}s] new best "
                      f"{best_score:.6f} (n={n}, it={it})")

            # gradient of the real objective using the smoothed max gradient
            factor1 = 2.0 * n / (s * s)
            factor2 = 4.0 * n * S / (s * s * s)
            grad = factor1 * grad_S - factor2 * np.ones_like(a)

            # ----- back‑tracking line search -----
            step = eta
            improved = False
            for _ in range(12):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                sum_new = a_new.sum()

            if (it + 1) % 25 == 0 and lam < 80.0:
                lam *= 1.4

        # keep the last iterate as a possible future start (limit list size)
        candidates.append(a)
        if len(candidates) > 20:
            candidates.pop(0)

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a simple uniform sequence
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n

    return best_seq
