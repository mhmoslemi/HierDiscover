import numpy as np, time

def project_simplex(v):
    # ensure v nonnegative may be negative; will project onto simplex.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def random_shape(n):
    # sample exponent p

    seq = mix * np.ones(n) + (1 - mix) * base

"""
Propose a candidate sequence of non‑negative heights that (hopefully) improves on the
baseline value 2.0.  The algorithm works in three stages:

1.  Initialise a random probability vector (sum = 1) of the same length as the
    provided `height_sequence_1` (or 2000 if it is not available).
2.  Repeatedly improve the vector while time permits.
    * Gradient‑type updates:  for the index k where the convolution a*a is maximal,
      the derivative of the maximal entry is proportional to a_{k‑i}.
      We either multiply the vector by exp(‑η·grad) or perform an additive step.
    * Random “shape” proposals:   a_i ∝ ((i+1)*(n-i))**p with a negative exponent p
      (large values near the edges, small values in the centre) optionally mixed
      with a uniform component.
    * Small random mass‑transfer perturbations.
    All candidates are projected onto the probability simplex (non‑negative,
      sum = 1) before evaluation.
3.  The best candidate found (according to a cheap FFT based estimate of the
    evaluation function) is finally evaluated with the official
    `evaluate_sequence` routine and returned as a plain Python list.

The whole search runs for ``budget_s`` seconds (default 1000) and constantly
prints progress information.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper functions (all top‑level as required)
# -------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i >= 0, sum_i x_i = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    condition = u - cssv / ind > 0
    if not np.any(condition):
        # all entries are zero; return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[condition][-1] - 1  # zero‑based index
    theta = cssv[condition][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(seq: np.ndarray) -> float:
    """
    Compute max of the linear convolution seq * seq using FFT (O(n log n)).
    """
    n = seq.shape[0]
    # size for zero‑padded convolution (next power of two)
    size = 1 << ((2 * n - 1).bit_length())
    spec = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(spec * spec, n=size)[: 2 * n - 1]
    return float(np.max(conv))


def _quick_score(seq: np.ndarray) -> float:
    """
    Fast surrogate for evaluate_sequence:
    2 * n * max(conv) / (sum(seq)^2)   (the denominator is 1 after simplex proj)
    """
    n = seq.shape[0]
    max_conv = _max_convolution_fft(seq)
    return 2.0 * n * max_conv / (seq.sum() ** 2)


def _exp_gradient_update(seq: np.ndarray, kmax: int, eta: float) -> np.ndarray:
    """
    Multiplicative (exponentiated‑gradient) update that reduces the entries
    participating in the maximal convolution entry.
    """
    n = seq.shape[0]
    idx = kmax - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(seq)
    grad[mask] = 2.0 * seq[idx[mask]]
    new_seq = seq * np.exp(-eta * grad)
    return _project_to_simplex(new_seq)


def _additive_gradient_update(seq: np.ndarray, kmax: int, step: float) -> np.ndarray:
    """
    Simple additive gradient step (seq ← seq – step·grad) followed by simplex
    projection.
    """
    n = seq.shape[0]
    idx = kmax - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(seq)
    grad[mask] = 2.0 * seq[idx[mask]]
    new_seq = seq - step * grad
    return _project_to_simplex(new_seq)


def _random_edge_heavy_shape(n: int) -> np.ndarray:
    """
    Produce a candidate where values are larger near the two ends and smaller
    in the middle.  The shape is ((i+1)*(n-i))**p with a negative exponent p.
    A fraction of a uniform component and a tiny amount of Gaussian noise are
    added before projection onto the simplex.
    """
    p = np.random.uniform(-4.0, -0.2)          # negative → edge heavy
    i = np.arange(1, n + 1, dtype=float)      # 1 … n
    prod = i * (n - i + 1)                     # (i)*(n‑i+1)
    base = prod ** p                           # edge‑heavy base
    mix = np.random.rand()                    # weight of uniform component
    seq = mix * np.ones(n) + (1.0 - mix) * base
    # add a touch of noise
    seq += np.random.randn(n) * np.random.uniform(0.0, 0.05)
    seq = np.maximum(seq, 0.0)
    return _project_to_simplex(seq)


def _small_mass_transfer(seq: np.ndarray, frac: float = 0.05) -> np.ndarray:
    """
    Transfer a tiny random amount of mass from one index to another while
    preserving total sum.  The amount is a fraction of the donor entry.
    """
    n = seq.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    if seq[i] == 0:
        return seq.copy()
    eps = np.random.rand() * frac * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] += eps
    # projection is cheap but kept for safety against tiny negative values
    return _project_to_simplex(new_seq)

        rnd = np.random.rand()

            spec = np.fft.rfft(cur_seq, n=conv_fft_size)
            conv_full = np.fft.irfft(spec * spec, n=conv_fft_size)[: 2 * n - 1]
            kmax = int(np.argmax(conv_full))
