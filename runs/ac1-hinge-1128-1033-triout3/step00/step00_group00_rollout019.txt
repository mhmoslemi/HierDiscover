def propose_candidate(budget_s=1000):
    import time, numpy as np, random

    deadline = time.time() + budget_s - 0.5
    # Choose initial n
    n = 2000  # moderate length
    # start from random decreasing sequence
    seq = np.linspace(0.1,1.0,n)  # increase from small to large
    # random shuffle maybe to unsymmetrize
    # Ensure sum >0.01
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    step = 1.0
    while time.time() < deadline:
        # compute conv and max index
        conv = np.convolve(seq, seq)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        s = np.sum(seq)
        # compute gradient of objective
        # dM/da_i = 2 * seq[max_idx - i] if valid else 0
        grad = np.zeros_like(seq)
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                grad[i] = 2 * seq[j]
        # gradient of objective
        # dO = 2*n/(s**2) * grad - 4*n*max_val/(s**3) * seq
        dO = (2.0 * n / (s * s)) * grad - (4.0 * n * max_val / (s ** 3)) * seq
        # update seq with small step in negative gradient direction
        lr = step / (1.0 + 0.001 * (np.linalg.norm(dO)))  # adapt
        seq = seq - lr * dO
        # clip non-negative and max 1000
        np.maximum(seq, 0, out=seq)
        np.minimum(seq, 1000, out=seq)
        # evaluate
        cur_score = evaluate_sequence(seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"Improved: {best_score:.6f}")
        # adjust step size
        if cur_score > best_score:
            # maybe increase step?
            step *= 0.95
        else:
            step *= 1.01
        # maybe random perturb occasionally
        if random.random() < 0.01:
            seq += np.random.randn(*seq.shape) * 0.01
            seq = np.clip(seq, 0, 1000)
    return best_seq.tolist()

"""
Simple linear-step construction.

The evaluation function is
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

For a linearly increasing non‑negative sequence,
    a_i = c * i   (i = 0 … n‑1)
the maximal convolution occurs at shift n‑1 and equals
    max_b = c² * n (n‑1) (n‑2) / 6 .
With the sum   S = c * n (n‑1) / 2

the ratio simplifies to
    4 * (n‑2) / (3 * (n‑1))  → 4/3 ≈ 1.3333
as n grows.  This is well below the target bound 1.503.

The implementation creates a linearly increasing sequence whose
maximum height is limited to 1000 (as required by the evaluator) and
returns it directly – no iterative search is needed.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a linear increasing step function.
    The sequence length is chosen large enough (default 2000) that the
    evaluation value is close to the theoretical limit 4/3.
    """
    # The length of the construction.  A few thousand points give a stable
    # value (~1.333) while keeping the computation cheap.
    n = 2000

    # Scale so that the last height is exactly 1000 (the evaluator clamps
    # values to at most 1000).  The first height is 0.
    scale = 1000.0 / (n - 1)
    sequence = [i * scale for i in range(n)]

    # Quick sanity check (optional, does not affect the returned result)
    # Ensure the sequence respects the bounds required by evaluate_sequence.
    # These assertions are cheap and safe.
    assert all(x >= 0 for x in sequence), "Negative height detected"
    assert all(x <= 1000 for x in sequence), "Height exceeds the allowed maximum"

    # The function does not need to spend the whole budget – the construction
    # is instantaneous.  Returning early is fine; the evaluator will compute
    # the score afterwards.
    return sequence
