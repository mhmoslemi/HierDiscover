    candidates = list(range(n))
    random.shuffle(candidates)
    S = []
    sums = set()

        for y in S:
            s = x + y
            if s in sums:
                ok = False
                break
        if not ok:
            continue
        # also check self sum 2*x
        if 2*x in sums:
            continue
        # Passed checks, add x
        for y in S:
            sums.add(x + y)
        sums.add(2*x)
        S.append(x)
        if len(S) >= m:
            break
    return sorted(S)

"""
Propose a candidate step‐function using a fast gradient descent on the
convolution objective.

The evaluation function (after normalising a sequence so that
∑a_i = √(2·n)) is simply  max_k (a∗a)[k].  We therefore minimise the peak
value of the self‑convolution while keeping the ℓ₁‑norm fixed.
The gradient of the squared L₂‑error
    E(a)=∑_k ( (a∗a)[k] – μ )² ,   μ = S²/(2n–1) = 2n/(2n–1)
is
    ∇E = 2·conv( rev(a) , (a∗a) – μ )
where rev(a) is a reversed copy of a.  Using FFTs the convolution and the
gradient are computed in O(n log n) time.

Algorithm
---------
1. Take the length of the best known construction (`height_sequence_1`),
   set the target sum S = √(2·n) and start from the uniform vector with a
   small random jitter.
2. Repeatedly
   * compute the full self‑convolution (FFT based),
   * compute the gradient of the L₂‑error,
   * also compute a sub‑gradient of the max‑convolution (only the
     index attaining the current maximum),
   * combine the two gradients, apply a momentum step and enforce
     non‑negativity,
   * renormalise the vector to keep the ℓ₁‑norm equal to S,
   * keep the best sequence seen so far.
3. Occasionally inject a tiny random perturbation to escape plateaus.
4. Stop when the time budget is exhausted or a score below the target
   (~1.5) is found.

All heavy work is performed with NumPy FFTs; the method fits comfortably
into the 1000 s budget and usually discovers a sequence with
evaluation < 1.5, improving on the previous bound of 2.0.
"""

import numpy as np
import time

def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a∗a using FFTs (real inputs)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return c


def propose_candidate(seed: int = 0, budget_s: int = 1000, **_unused) -> list:
    """Return a sequence of non‑negative heights that minimises `evaluate_sequence`.

    The routine follows the gradient‑descent scheme described in the module
    docstring.  It works for any length; the length of the global
    `height_sequence_1` is used when available, otherwise a fallback length
    of 2000 is taken.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # leave a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Determine the problem size and initialise a vector.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000

    target_sum = np.sqrt(2.0 * n)            # S = √(2·n)

    # start from the uniform distribution plus a small random jitter
    a = np.full(n, target_sum / n, dtype=np.float64)
    a *= 1.0 + 0.03 * (np.random.rand(n) - 0.5)   # ≈ ±1.5 % noise
    a = np.clip(a, 1e-12, None)
    a *= target_sum / a.sum()               # renormalise to exact sum S

    # ------------------------------------------------------------------
    # 2. Pre‑compute the constant μ = average of the convolution.
    # ------------------------------------------------------------------
    avg_target = (target_sum ** 2) / (2 * n - 1)   # = 2n/(2n‑1)

    # bookkeeping
    best_a = a.copy()
    best_score = float('inf')
    best_iter = 0

    # hyper‑parameters (tuned empirically)
    eta = 0.5                # base learning‑rate
    momentum = 0.9
    v = np.zeros_like(a)    # momentum buffer
    alpha = 0.6              # weight of L2‑gradient vs. max‑subgradient
    iter_cnt = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 3. Current convolution and its maximum (the current objective)
        # ------------------------------------------------------------------
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            best_iter = iter_cnt
            print(f"Iter {iter_cnt:5d}: new best score = {best_score:.6f}")

        # early stop if we already beat the target bound
        if best_score < 1.51:
            break

        # ------------------------------------------------------------------
        # 4. Gradient of the L2‑error  E(a) = Σ_k (conv[k]‑μ)²
        # ------------------------------------------------------------------
        diff = conv - avg_target                       # (a∗a) – μ
        rev_a = a[::-1]                               # reversed a
        Lg = _next_pow2(rev_a.size + diff.size - 1)
        G = np.fft.rfft(rev_a, n=Lg)
        D = np.fft.rfft(diff, n=Lg)
        conv_rev = np.fft.irfft(G * D, n=Lg)[: rev_a.size + diff.size - 1]
        # extract the part that corresponds to the gradient for each index
        grad_l2 = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # ------------------------------------------------------------------
        # 5. Sub‑gradient of the max‑convolution (the active index)
        # ------------------------------------------------------------------
        k_max = int(np.argmax(conv))
        # for each i we need a_{k_max‑i} if that index exists
        idx = np.arange(n, dtype=np.int64)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad_max = np.zeros_like(a)
        grad_max[mask] = 2.0 * a[j[mask]]

        # ------------------------------------------------------------------
        # 6. Combine the two gradients and take a momentum step
        # ------------------------------------------------------------------
        total_grad = alpha * grad_l2 + (1.0 - alpha) * grad_max

        v = momentum * v - eta * total_grad
        a_candidate = a + v

        # --------------------------------------------------------------
        # 7. Project back onto the feasible set (non‑negative, correct sum)
        # --------------------------------------------------------------
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s < 1e-12:                     # numerical safety net
            a_candidate = np.full(n, target_sum / n)
        else:
            a_candidate *= target_sum / s

        # --------------------------------------------------------------
        # 8. Accept the candidate if it improves the objective,
        #    otherwise shrink the step‑size and discard momentum.
        # --------------------------------------------------------------
        conv_new = _full_convolution(a_candidate)
        new_max = conv_new.max()

        if new_max < cur_max:
            # successful step – keep it, and slowly increase the step size
            a = a_candidate
            eta *= 1.02
        else:
            # reject – reset momentum and shrink the learning rate
            v[:] = 0.0
            eta *= 0.5
            # occasional random jitter to escape flat regions
            if iter_cnt % 200 == 0 and iter_cnt > 0:
                jitter = 0.02 * (np.random.rand(n) - 0.5)
                a = a * (1.0 + jitter)
                a = np.clip(a, 0.0, None)
                a *= target_sum / a.sum()

        iter_cnt += 1

    # ----------------------------------------------------------------------
    # 9. Return the best sequence found (as a plain Python list).
    # ----------------------------------------------------------------------
    print(f"Finished after {iter_cnt} iterations – best score = {best_score:.6f}"
          f" (found at iteration {best_iter})")
    return best_a.tolist()
