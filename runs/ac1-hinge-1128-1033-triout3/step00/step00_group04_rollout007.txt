"""
search_1: Mini‑max optimisation of a non‑negative step function.

The evaluation metric is

    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

which is homogeneous (scale‑invariant).  Hence we can keep the
ℓ¹‑norm of the sequence equal to 1 during the whole search and only
minimise the peak of the self‑convolution.

The algorithm works as follows

1.  Choose a length n (a few thousand works well with the time limit).
2.  Initialise a with the (continuous) extremiser a_i ∝ 1/√(i+1) –
    this shape already gives a score of ≈π/2≈1.57, better than the
    naïve constant sequence (score = 2).
3.  Repeatedly improve the sequence by a projected (sub‑)gradient step.
    The gradient of the smoothed maximum
          p_k  ∝  exp(α·conv_k)
    of the convolution is a convolution of the weights p with the
    reversed sequence.  The update is

        a ← a – η·∇   (η = learning‑rate)

    After the step we clip negatives to zero and renormalise the
    ℓ¹‑norm to 1 (projection onto the simplex).

4.  The smoothing parameter α is increased slowly (α≈1 → 30) so that
    the algorithm first moves in a gentle direction and finally
    concentrates on the true maximum.
5.  Every few hundred iterations we evaluate the current sequence with
    `evaluate_sequence`.  The best sequence found across all restarts
    (different lengths, random perturbations) is returned.

The whole optimisation runs for the allotted budget (default 1000 s)
and never exceeds it.  Simple prints give a live view of the best
score.
"""

import time
import math
import numpy as np

def _project_onto_simplex(v):
    """Project a non‑negative vector onto the ℓ¹‑simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _softmax_weights(conv, alpha):
    """Return the soft‑max weights for a convolution vector."""
    # numerical stability – shift by the maximum
    m = conv.max()
    # exp(alpha*(conv - m)) ≤ 1, never overflows
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0.0:
        # extremely unlikely, but guard against division by zero
        return np.full_like(conv, 1.0 / conv.size)
    return w / w_sum


def _gradient_of_softmax_max(a, conv, alpha):
    """
    Gradient of the smoothed max (soft‑max) of a*a.
    Uses the identity
        ∂/∂a_i  sum_j p_j (a*a)_j = 2 * (p * a_rev)[i]
    where p are the soft‑max weights and a_rev = a[::-1].
    """
    n = a.size
    p = _softmax_weights(conv, alpha)

    # convolution of p with the reversed a
    # full length = (2n-1) + n - 1 = 3n-2
    rev = a[::-1]
    full = np.convolve(p, rev, mode='full')
    # the needed slice is centred at index n‑1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # leave a small safety margin

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # Helper to evaluate and keep the best sequence
    # ------------------------------------------------------------------
    def maybe_update(seq):
        nonlocal best_seq, best_score
        try:
            sc = evaluate_sequence(seq.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = seq.tolist()
                print(f"[{time.time():.2f}] New best score {sc:.6f} (len={len(seq)})")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Main optimisation loop – a few independent restarts
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2400, 3200]   # all comfortably < 5400
    restart = 0
    while time.time() < deadline and restart < 10:
        n = int(np.random.choice(lengths))
        # --------------------------------------------------------------
        # 1. Initial shape: a_i ∝ 1/√(i+1)  (the known analytic approx.)
        # --------------------------------------------------------------
        a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
        a = _project_onto_simplex(a)

        # Hyper‑parameters (they are annealed during the run)
        lr = 0.05                # initial learning rate
        alpha = 1.0              # smoothing for the soft‑max
        it = 0

        # --------------------------------------------------------------
        # 2. Iterative improvement
        # --------------------------------------------------------------
        while time.time() < deadline:
            conv = np.convolve(a, a)           # length 2n‑1
            grad = _gradient_of_softmax_max(a, conv, alpha)

            # gradient step + projection onto the simplex
            a = a - lr * grad
            a = _project_onto_simplex(a)

            # ----------------------------------------------------------
            # 3. Schedule – slowly tighten the approximation of the max
            # ----------------------------------------------------------
            it += 1
            if it % 500 == 0:
                maybe_update(a)

                # decay learning rate, increase smoothing
                lr *= 0.90
                alpha = min(alpha * 1.30, 30.0)

            # stop inner loop early if we have performed many iterations
            if it >= 20000:
                break

        # final check for this restart
        maybe_update(a)
        restart += 1

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 200
        best_score = evaluate_sequence(best_seq)
        print("Fallback to constant sequence, score:", best_score)

    print("Best final score:", best_score, "length:", len(best_seq))
    return best_seq
