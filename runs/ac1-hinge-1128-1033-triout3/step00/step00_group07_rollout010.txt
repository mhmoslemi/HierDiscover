"""
A fast heuristic for minimizing ``evaluate_sequence``.

The algorithm works on a fixed length ``n`` (starting at 256) and
performs a projected gradient descent on a smooth approximation of the
``max`` in the evaluation.  The smoothing is a soft‑max with a temperature
parameter ``beta`` that is increased during optimisation.  After each
gradient step the vector is projected onto the feasible set:
* non‑negative entries,
* fixed ℓ₁‑norm Σ a_i = √(2 n).

Multiple random restarts are performed until the time budget expires.
Occasionally a tiny random “kick” is added to avoid being trapped in a
flat region.  The best sequence seen so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the objective (identical to evaluate_sequence for a
# correctly normalised vector) without the python‑level validation.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, target_sum: float) -> float:
    """Return the value of evaluate_sequence for a normalised vector `a`.

    The vector is assumed to be non‑negative and Σ a_i = target_sum.
    """
    conv = np.convolve(a, a)                # full convolution
    max_val = conv.max()
    # 2 * n * max / (sum_a**2) ; here sum_a == target_sum
    return 2.0 * n * max_val / (target_sum * target_sum)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that yields a low
    value of ``evaluate_sequence``.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.

    Returns
    -------
    list[float]
        The sequence of heights (non‑negative) achieving the smallest
        objective found within the time limit.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Settings that may evolve during the run
    # ------------------------------------------------------------------
    n = 256                     # start with a modest size
    target_sum = np.sqrt(2.0 * n)

    # pre‑computed index matrix for the soft‑max gradient (i+j)
    i_mat = np.arange(n)[:, None]
    j_mat = np.arange(n)[None, :]
    sum_mat = i_mat + j_mat                       # shape (n, n)

    best_seq = None
    best_score = float("inf")
    restart_cnt = 0

    deadline = time.time() + budget_s - 0.1   # a little safety margin

    # ------------------------------------------------------------------
    # Main outer loop – keep launching fresh random starts while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Random feasible start (ℓ₁‑norm = target_sum)
        # --------------------------------------------------------------
        a = rng.random(n)
        a = a / a.sum() * target_sum

        # optimisation hyper‑parameters for this start
        lr = 0.05                     # learning rate

        beta_inc = 0.05
        beta_max = 30.0
        max_iters = 3000

        # keep a local copy of the current objective (exact, cheap version)
        cur_score = _objective(a, n, target_sum)

            best_score = cur_score
            best_seq = a.copy()
            print(f"[restart {restart_cnt}] initial best {best_score:.6f}")

        # --------------------------------------------------------------
        # 2) Inner gradient‑descent loop
        # --------------------------------------------------------------
        for it in range(max_iters):
            # stop early if the global budget is exhausted
            if time.time() >= deadline:

            # exact objective (fast) – used for bookkeeping
            max_val = np.convolve(a, a).max()
            cur_score = 2.0 * n * max_val / (target_sum * target_sum)
