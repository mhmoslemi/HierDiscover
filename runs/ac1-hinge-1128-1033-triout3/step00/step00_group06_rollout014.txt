import numpy as np
import time

def compute_score(seq):
    seq = np.array(seq, dtype=np.float64)
    n = len(seq)
    # FFT length power of two >= 2n-1
    L = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def arcsine_coeffs(n):
    """Generate the first n coefficients of (1 - z)^(-1/2) = sum_{i>=0} binom(2i,i)/4^i z^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0 * i - 1) / (2.0 * i)
    return coeffs.tolist()

def propose_candidate(seed=0, budget_s=1000, length=5000):
    """
    Generate a sequence that improves the evaluation function.
    The baseline construction uses the coefficients of (1 - z)^(-1/2),
    which give a convolution that is constant (value 1) on the first
    ``length`` entries and lead to an evaluation score close to π/2 ≈ 1.57.
    A short stochastic refinement is then performed while time permits.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a margin

    # ---- Initial deterministic construction ---------------------------------
    seq = arcsine_coeffs(length)
    best_seq = seq.copy()
    best_score = compute_score(best_seq)
    print(f"Initial arcsine sequence length={length} score={best_score:.6f}")

    # ---- Simple stochastic refinement ---------------------------------------
    # Randomly transfer a tiny amount of mass between two positions.
    # The total sum is kept (approximately) constant, aiming to reduce the
    # maximal convolution entry.
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(length, size=2)
        if i == j:
            continue
        # amount to move (relative to the source value)
        delta = np.random.exponential(scale=1e-3) * seq[i]
        if delta <= 0 or seq[i] < delta:
            continue
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # enforce non‑negativity (already guaranteed by construction)
        # compute a cheap score using FFT; this is equivalent to evaluate_sequence
        cur_score = compute_score(new_seq)
        # accept only if improvement
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            seq = new_seq
            # occasional progress report
            if int(best_score * 1e5) % 1000 == 0:
                print(f"Improved score: {best_score:.6f}")
        else:
            # optional: accept with small probability to escape plateaus
            if np.random.rand() < 0.001:
                seq = new_seq

    print(f"Final best score: {best_score:.6f}")
    return best_seq

"""
Propose a candidate step‑function (list of non‑negative heights).

The core idea is to use the coefficients of the power series  

    (1 – z)^{-1/2} = Σ_{i≥0} binom(2 i, i) / 4^i * z^i ,

which satisfy  

    Σ_{i=0}^{k} a_i a_{k-i} = 1   for every k ≥ 0.

Hence the discrete convolution `a * a` is equal to 1 on the first
`len(a)` entries, giving a maximal convolution value `max_b = 1`.
For a sequence of length `n` the evaluation function  

    2 * n * max_b / ( Σ a_i )^2

converges to `π / 2 ≈ 1.5708`, a large improvement over the uniform
sequence (value 2).  The routine builds such a sequence, then performs a
very cheap stochastic refinement (mass transfer between two positions)
while time permits, always keeping the sum of the coefficients constant
and trying to lower the maximal convolution entry.  The refinement is
optional – the plain arcsine sequence already beats the current upper
bound.

The function returns the best sequence found before the allotted time
expires.
"""

import numpy as np
import time


def _compute_score(seq):
    """
    Fast computation of the evaluation score.
    Uses an FFT based convolution (real FFT) which is O(N log N) and
    matches the behaviour of the supplied `evaluate_sequence`.
    """
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    # length for FFT: next power of two ≥ 2*n-1
    L = 1 << ((2 * n - 1).bit_length())
    # real FFT (more efficient for real data)
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _arcsine_coeffs(n):
    """
    Return the first `n` coefficients of (1 - z)^(-1/2) = Σ binom(2i,i)/4^i z^i.
    The recurrence
        a_0 = 1,
        a_{i+1} = a_i * (2*i + 1) / (2*(i+1))
    is used for numerical stability.
    """
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs.tolist()


def propose_candidate(seed=0, budget_s=1000, length=5000):
    """
    Generate a step function that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int, optional
        Random seed for the stochastic refinement phase.
    budget_s : float, optional
        Time budget in seconds.  The function stops a few tenths of a
        second before the limit to avoid a forced termination.
    length : int, optional
        Length of the initial arcsine sequence.  Values around 5000 give a
        good trade‑off between quality and speed.

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a small safety margin

    # ----- 1. deterministic start with the arcsine (inverse‑square‑root) series
    seq = _arcsine_coeffs(length)
    best_seq = seq.copy()
    best_score = _compute_score(best_seq)
    print(f"[init] length={length}, score={best_score:.6f}")

    # ----- 2. cheap stochastic refinement (mass‑transfer moves)
    # We keep the total sum roughly constant, hoping to lower the peak of the
    # convolution without sacrificing the sum too much.
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(length, size=2)
        if i == j:
            continue

        # transfer a tiny amount of mass from i to j
        # delta is a small fraction of the source entry to keep changes smooth
        src = seq[i]
        if src <= 0.0:
            continue
        delta = np.random.exponential(scale=1e-3) * src
        if delta > src:
            continue

        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # fast evaluation
        cur_score = _compute_score(new_seq)

        # accept only improvements
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            seq = new_seq
            # occasional progress report
            if int(best_score * 1e5) % 1000 == 0:
                print(f"[improve] score={best_score:.6f}")
        else:
            # occasional random acceptance to escape local plateaus
            if np.random.rand() < 0.001:
                seq = new_seq

    print(f"[final] best score={best_score:.6f}")
    return best_seq
