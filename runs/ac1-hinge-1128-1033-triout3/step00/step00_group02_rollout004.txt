"""
Arcsine (Beta(½,½)) based construction.

The evaluation function is
    2 * n * max(conv(a,a)) / (sum(a)**2)

It is scale‑invariant, so we may fix sum(a)=1 and minimise
max(conv(a,a)).  For a continuous Beta(½,½) density f on [0,1] the
self‑convolution f*f is exactly uniform on [0,2]; discretising this
distribution yields a sequence whose discrete self‑convolution is
almost flat, i.e. the maximal entry is close to the average 1/(2n).
Consequently the score approaches 1 (the theoretical lower bound).

The code builds the discretised arcsine density for several
lengths, evaluates each candidate and returns the best one found
within the allotted time budget.
"""

import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function height sequence based on the discretised
    arcsine distribution (Beta(½,½)).  Several lengths are tried and
    the sequence with the smallest evaluation score is returned.

    Parameters
    ----------
    seed : int
        Random seed (unused, kept for API compatibility).
    budget_s : int
        Time budget in seconds.  The routine stops a few seconds
        before the deadline to guarantee a return.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)                     # keep API compatibility
    deadline = time.time() + budget_s - 5    # safety margin

    best_score = float('inf')
    best_seq = None

    # Lengths to try – they fit comfortably into the 1000 s budget.
    lengths = [500, 1000, 2000, 4000, 8000]

    for n in lengths:
        if time.time() > deadline:
            break

        # Discretise the Beta(½,½) density:
        # x_i = (i+0.5)/n \in (0,1)  avoids the singularities at 0 and 1.
        i = np.arange(n, dtype=np.float64)
        x = (i + 0.5) / n
        # weight proportional to 1/sqrt(x*(1-x))
        w = 1.0 / np.sqrt(x * (1.0 - x))

        # Normalise to sum = 1 (any scaling leaves the score unchanged).
        a = w / w.sum()

        # Evaluate the candidate.
        try:
            score = evaluate_sequence(a.tolist())
        except Exception:
            # In the unlikely event evaluate_sequence crashes,
            # treat this candidate as infinitely bad.
            score = float('inf')

        print(f"Length={n:5d}, score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = a.tolist()

    # If for some reason no sequence was accepted, fall back to a simple uniform one.
    if best_seq is None:
        n = 1000
        a = np.full(n, 1.0 / n)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"Best score={best_score:.6f} with length={len(best_seq)}")
    return best_seq
