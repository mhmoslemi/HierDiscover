def project_simplex(v):
    v = np.maximum(v, 0)
    if v.sum() == 0: return np.full_like(v, 1.0 / len(v))
    # Using method from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Propose a non‑negative step function that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

The evaluation is scale‑invariant, therefore we work with probability vectors
(sum = 1).  The algorithm consists of

1.  Warm‑start from a possible external variable ``height_sequence_1``.
2.  A deterministic sweep over several analytic families (U‑shaped,
    sum‑of‑tails, exponential, product‑power‑law, symmetric Beta and a few
    asymmetric power‑law/exponential families).  For each length we try a
    fine grid of shape parameters and keep the best approximate score
    ``2·n·max(conv)`` (computed with a real FFT – O(n log n)).
3.  Local refinement:
       – a projected gradient step on the current maximal‑convolution index,
       – random “mass‑transfer” moves evaluated with an O(n) incremental
         update of the convolution peak,
       – standard simulated‑annealing acceptance.
   The loop runs until the time budget expires or the target score 1.503 is
   reached.
4.  The best vector is finally evaluated with the official
   ``evaluate_sequence`` routine and returned as a plain Python list.

All helper functions are top‑level; the code respects the 1000 s time limit,
uses only NumPy and SciPy (which are available in the execution environment)
and prints intermediate progress.
"""

import time
import numpy as np
from scipy.optimize import minimize_scalar

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def max_conv_fft(a: np.ndarray) -> float:
    """
    Maximum entry of the self‑convolution a*a using a real FFT.
    a is assumed to be a 1‑D array of floats.
    """
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a (real part) using a real FFT.
    Returns a length‑(2n‑1) array.
    """
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al.
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return a normalized probability vector)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^(‑p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Two‑sided power‑law tails: a[i] ∝ (i+1)^(‑p) + (n‑i)^(‑p)."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """Product power‑law shape: a[i] ∝ ((i+1)*(n‑i))^(‑p)."""
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-12
    a /= a.sum()
    return a


def beta_distribution(n: int, a_param: float, b_param: float) -> np.ndarray:
    """
    Discrete Beta( a , b ) – points are taken at the mid‑points of the
    intervals to avoid singularities at 0/1.
    """
    i = np.arange(n) + 0.5  # shift to avoid exact 0/1
    x = i / n
    a = (x ** (a_param - 1)) * ((1 - x) ** (b_param - 1))
    a += 1e-12
    a /= a.sum()
    return a


def asymmetric_powerlaw_distribution(n: int, p_left: float, p_right: float) -> np.ndarray:
    """Asymmetric power‑law: left side exponent p_left, right side exponent p_right."""
    i = np.arange(n)
    a = (i + 1) ** (-p_left) + (n - i) ** (-p_right) + 1e-12
    a /= a.sum()
    return a


def asymmetric_exponential_distribution(n: int, a_left: float, a_right: float) -> np.ndarray:
    """Asymmetric exponential decay from both ends."""
    i = np.arange(n)
    a = np.exp(-a_left * i) + np.exp(-a_right * (n - 1 - i)) + 1e-12
    a /= a.sum()
    return a

    max_frac = 0.30            # maximal fraction of source mass to move
    grad_step_factor = 0.10    # base step size for gradient step
    batch_moves = 40           # candidates examined per iteration
    iteration = 0

    print("\n=== Stochastic refinement ===")
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 3a. Choose move type
        # --------------------------------------------------------------
        if np.random.rand() < 0.20:
            # ------- Gradient (mirror‑descent) step -------
            kmax = int(conv.argmax())
            i = np.arange(n)
            j = kmax - i
            mask = (j >= 0) & (j < n)
            grad = np.zeros_like(current)
            grad[mask] = 2.0 * current[j[mask]]

            step = grad_step_factor * temperature
            cand = current - step * grad
            cand = np.clip(cand, 0.0, None)
            cand = project_simplex(cand)

            cand_conv = full_conv_fft(cand)
            cand_score = 2.0 * n * cand_conv.max()

            # ---- SA acceptance ----
            accept = False
            if cand_score < current_score:
                accept = True
            else:
                prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                current = cand
                conv = cand_conv
                conv_max = conv.max()
                current_score = cand_score
                if current_score < best_score:
                    best_score = current_score
                    best_seq = current.copy()
                    print(f"Iter {iteration:6d}: new global best = {best_score:.6f}")

        else:
            # ------- Random mass‑transfer (batch evaluation) -------
            # Pad once – allows O(1) extraction of the shift‑vectors needed
            pad = np.concatenate([np.zeros(n - 1), current, np.zeros(n - 1)])

            best_local_score = current_score
            best_move = None

            for _ in range(batch_moves):
                if np.all(current == 0):
                    break
                # source index proportional to its current mass
                src = np.random.choice(n, p=current / current.sum())
                if current[src] <= 0.0:
                    continue
                # destination must differ from source
                dst = np.random.randint(0, n - 1)
                if dst >= src:
                    dst += 1

                delta = current[src] * np.random.rand() * max_frac
                if delta <= 0.0:
                    continue

                # incremental update of the convolution peak
                offset_src = n - 1 - src
                offset_dst = n - 1 - dst
                shift_src = pad[offset_src: offset_src + 2 * n - 1]
                shift_dst = pad[offset_dst: offset_dst + 2 * n - 1]

                new_max = np.max(conv + 2.0 * delta * (shift_dst - shift_src))
                cand_score = 2.0 * n * new_max

                if cand_score < best_local_score:
                    best_local_score = cand_score
                    best_move = (src, dst, delta, new_max)

            if best_move is not None:
                src, dst, delta, new_max = best_move
                # Apply the chosen transfer
                current[src] -= delta
                current[dst] += delta

                # Exact recomputation (cheap enough because move is accepted)
                conv = full_conv_fft(current)
                conv_max = conv.max()
                current_score = 2.0 * n * conv_max

                if current_score < best_score:
                    best_score = current_score
                    best_seq = current.copy()
                    print(f"Iter {iteration:6d}: new global best = {best_score:.6f}")

            # If no improving random move was found we simply continue;
            # the temperature will gradually cool and the gradient branch
            # will dominate later on.

        # --------------------------------------------------------------
        # 3c. Temperature cooling
        # --------------------------------------------------------------
        temperature = max(temperature * cooling_rate, 1e-7)

        # Early exit if we have already beaten the target
        if best_score < 1.503:
            print(f"Target achieved (score {best_score:.6f}); exiting early.")
            break

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (once) and return
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())  # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()
