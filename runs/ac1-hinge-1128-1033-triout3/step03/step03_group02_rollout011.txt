def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, scipy.optimize, scipy.sparse ...
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # choose n
    n = 2000
    # init a
    x = np.arange(n)
    a = np.maximum(1.0 - np.sin(np.pi * x / (n - 1)), 0.0)
    a = a / a.sum()

    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    conv_a = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv_a.max())
    iteration = 0

    # schedule lengths
    # we allocate 80% time for Adam, then LP phases.
    while time.time() < deadline:
        # compute conv and M
        conv_a = _full_convolution_fft(a, a)
        M = conv_a.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration}] new best {best_val:.6f}")

        # progress ratio for schedule
        t = (time.time() - start) / (deadline - start)
        # compute tau schedule
        tau = M * (0.25 * (1 - t) + 1e-5 * t) 

        # compute weights
        exp_args = (conv_a - M) / (tau if tau > 1e-12 else 1e-12)
        # shift to avoid overflow
        max_exp = np.max(exp_args) # negative or zero.
        exp_vals = np.exp(exp_args - max_exp) # safe
        sum_exp = np.sum(exp_vals)
        if sum_exp == 0 or not np.isfinite(sum_exp):
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0
        else:
            w = exp_vals / sum_exp

        # compute gradient
        grad = _gradient_from_weights(a, w)

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        # schedule step size
        alpha_start = 0.5
        alpha_end = 1e-5
        alpha = alpha_start * (1 - t) + alpha_end * t
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step

        # add jitter
        if np.random.rand() < 0.02:
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i: j += 1
            d = np.random.rand() * a_candidate[i] * 0.5
            a_candidate[i] -= d
            a_candidate[j] += d

        # project onto simplex
        a = _project_onto_simplex(a_candidate)

        # periodic local search after some iterations
        if iteration % 5000 == 0:
            # attempt a few random transfers
            for _ in range(200):
                i = np.random.randint(n)
                if a[i] == 0: continue
                j = np.random.randint(n-1)
                if j >= i: j += 1
                delta = np.random.rand() * a[i] * 0.1
                a_new = a.copy()

                a_new[j] += delta
                conv_new = _full_convolution_fft(a_new, a_new)
                val_new = 2.0 * n * conv_new.max()
                if val_new < best_val:
                    best_val = val_new
                    best_a = a_new.copy()
                    a = a_new

Implementation details:

```
def _linear_program_update(f, M, n, time_allow):
    import scipy.optimize, scipy.sparse as sp
    # Build matrix
    rows = np.empty(n*n, dtype=int)
    cols = np.empty(n*n, dtype=int)
    data = np.empty(n*n, dtype=float)
    idx = 0
    for i in range(n):
        # compute row indices
        rows[idx: idx+n] = i + np.arange(n)
        cols[idx: idx+n] = np.arange(n)
        data[idx: idx+n] = f[i]   # f_i repeated
        idx += n
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M, dtype=float)
    c = -np.ones(n)  # maximize sum b => minimize -sum
    # Solve using scipy.optimize.linprog
    # Use method='highs-ds' or 'highs' whichever available.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                 bounds=[(0, None)]*n,
                                 method='highs',
                                 options={'time_limit': time_allow,
                                          'presolve': True})
    if not res.success:
        return None
    b0 = res.x
    sum_b0 = b0.sum()
    if sum_b0 <= 0:
        return None
    # scale to sum=1
    b_scaled = b0 / sum_b0
    return b_scaled

"""propose_candidate
A two‑phase optimizer for the step‑function problem.

Phase 1 – Adam‑based soft‑max gradient descent on the simplex.
Phase 2 – occasional exact linear‑programming updates (if SciPy is
available) followed by a short stochastic local‑search polish.
Both phases respect the non‑negativity and sum‑to‑one constraints,
which makes the objective 2·n·max_k (a*a)[k] (scale‑invariant).
The routine runs until the supplied time budget expires and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al., 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Normalise to guard against rounding errors
    s = w.sum()
    if s == 0:
        return np.full_like(v, 1.0 / n)
    return w / s


# ----------------------------------------------------------------------
# Linear convolution via real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full (linear) convolution x*y using a power‑of‑two FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Small negative rounding artefacts are removed
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of Σ_k w_k·(a*a)[k] = 2·Σ_k w_k·a_{k‑i}
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Return g_i = 2·Σ_k w_k·a_{k‑i} for all i (implemented by a single FFT)."""
    n = a.shape[0]
    u = w[::-1]                 # reverse the weight vector
    conv = _full_convolution_fft(u, a)      # length = n + (2n‑1) – 1 = 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)          # positions that correspond to g_i
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Optional linear‑programming step (requires SciPy)
# ----------------------------------------------------------------------
def _lp_update(f: np.ndarray, M: float):
    """Solve  max Σ b_j  s.t.  f*b ≤ M, b ≥ 0   (returns normalized b)."""
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception:                     # SciPy not available – skip LP
        return None

    n = f.shape[0]
    # Build the (2n‑1) × n convolution matrix A where (A b)[k] = Σ_i f_i b_{k‑i}
    # Using vectorised construction: each f_i appears in every column of its
    # shifted row.
    i = np.repeat(np.arange(n), n)                 # i‑index (row shift)
    j = np.tile(np.arange(n), n)                   # column index
    rows = i + j                                    # convolution row (0 … 2n‑2)
    cols = j
    data = np.repeat(f, n)                         # each f_i repeats n times
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                                 # minimise –Σ b_j  → maximise Σ b_j
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0.0, None)] * n

    # Solve with HiGHS (fast for medium‑size dense‑ish LPs)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    b = res.x
    s = b.sum()
    if s <= 0:
        return None
    # Normalise to sum‑to‑one (our global normalisation)
    return b / s


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a non‑negative step‑function to minimise
    evaluate_sequence().  The algorithm consists of:
      1. Adam‑based soft‑max gradient descent on the probability simplex.
      2. Periodic exact LP updates (if SciPy is available) that enlarge the
         feasible mass without increasing the current convolution bound.
      3. A short stochastic mass‑transfer local search for final polishing.
    The best sequence encountered before the budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the framework
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Choose an initial length n and an initial a
    # --------------------------------------------------------------
    init_a = globals().get("height_sequence_1", None)
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        n = a.shape[0]
    else:
        # modestly large, edge‑heavy shape – works well as a seed
        n = 2000
        x = np.arange(n, dtype=float)
        a = 1.0 - np.sin(np.pi * x / (n - 1))
        a = np.maximum(a, 0.0)

    # Normalise to a probability distribution (Σ a = 1)
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total

    # --------------------------------------------------------------
    # Adam bookkeeping
    # --------------------------------------------------------------
    m = np.zeros_like(a)          # first moment estimate
    v = np.zeros_like(a)          # second moment estimate
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-12

    # Global best tracking
    conv_a = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv_a.max())
    best_a = a.copy()
    iteration = 0

    # Hyper‑parameter schedules
    eta_start = 0.5               # initial Adam step‑size (effective learning rate)
    eta_end = 1e-5                # final step‑size
    tau_factor_start = 0.25       # relative to current max for the first τ
    tau_factor_end = 1e-5

    # ------------------------------------------------------------------
    # Phase 1 – Adam with soft‑max weighting
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- convolution of current a with itself ----
        conv_a = _full_convolution_fft(a, a)          # length 2n‑1
        M = conv_a.max()
        cur_val = float(2.0 * n * M)

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # ---- progress for schedules ----
        now = time.time()
        prog = (now - start_time) / (deadline - start_time)

        # ---- soft‑max temperature (τ) ----
        tau = M * (tau_factor_start * (1 - prog) + tau_factor_end * prog)
        if tau <= 0:
            tau = 1e-12

        # ---- soft‑max weights over the convolution ----
        # to avoid overflow we shift by the maximal exponent (which is 0)
        exp_args = (conv_a - M) / tau               # ≤ 0
        # numerical stabilisation
        max_exp = np.max(exp_args)                  # = 0
        exp_vals = np.exp(exp_args - max_exp)
        sum_exp = exp_vals.sum()
        if sum_exp == 0 or not np.isfinite(sum_exp):
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0
        else:
            w = exp_vals / sum_exp

        # ---- gradient of the soft‑max weighted objective ----
        grad = _gradient_from_weights(a, w)

        # ---- Adam moments update ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # ---- learning‑rate schedule ----
        eta = eta_start * (1 - prog) + eta_end * prog

        # ---- Adam step (project back onto the simplex) ----
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step

        # ---- occasional random jitter (helps escape plateaux) ----
        if np.random.rand() < 0.02:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a_candidate[src] * 0.5
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        a = _project_onto_simplex(a_candidate)

        # ------------------------------------------------------------------
        # Periodic exact LP update – helps to push the bound down systematically
        # ------------------------------------------------------------------
        if iteration % 20000 == 0:
            lp_b = _lp_update(a, M)
            if lp_b is not None:
                # mix a little bit towards the LP solution
                t_mix = 0.15
                a = (1.0 - t_mix) * a + t_mix * lp_b
                a = _project_onto_simplex(a)      # re‑project to be safe
                print(f"[{iteration}] LP mix performed (t={t_mix})")

        # ------------------------------------------------------------------
        # Light weight local search every few thousand iterations
        # ------------------------------------------------------------------
        if iteration % 5000 == 0:
            # try a handful of random mass‑transfer moves
            improved = False
            for _ in range(200):
                i = np.random.randint(n)
                if a[i] <= 0:
                    continue
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
                delta = np.random.rand() * a[i] * 0.1
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                conv_tmp = _full_convolution_fft(a_tmp, a_tmp)
                val_tmp = float(2.0 * n * conv_tmp.max())
                if val_tmp < best_val:
                    best_val = val_tmp
                    best_a = a_tmp.copy()
                    a = a_tmp
                    improved = True
                    print(f"[{iteration}] local tweak improved to {best_val:.6f}")
                    break
            if not improved:
                # no improvement – continue as usual
                pass

    # ----------------------------------------------------------------------
    # Final reporting (optional – the outer framework may also call it)
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
