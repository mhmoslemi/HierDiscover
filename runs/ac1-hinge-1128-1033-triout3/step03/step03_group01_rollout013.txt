"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max_k (a * a)[k] / (sum a)^2

The algorithm proceeds in three stages:

1. **Deterministic sweep** – a set of simple parametric families
   (U‑shaped, product‑U, exponential) is evaluated for several
   lengths (500‒2000).  The best vector for each length is kept.

2. **Linear‑programming refinement** – for a chosen moderate length
   (default 2000) we iteratively solve a LP:

        maximise   Σ b_i
        subject to   b_i ≥ 0,
                    (f * b)_k ≤ M   for all k,
                    Σ b_i free

   where `f` is the current vector and `M` is its present maximal
   convolution value.  The optimal `b` is blended with `f`
   (`f ← (1‑t) f + t b̂`) to strictly reduce the bound.
   A handful of these LP steps quickly drives the autocorrelation
   towards a flat profile.

3. **Stochastic sub‑gradient / simulated‑annealing polishing** – the
   refined vector may be up‑sampled to larger lengths, then a projected
   sub‑gradient descent with occasional random mass‑transfer moves,
   temperature decay and re‑heating is run until the time budget is
   exhausted.  The best vector seen throughout the whole run is returned.

All heavy work respects the global time budget (default 1000 s).  The
implementation uses only NumPy and SciPy (linprog with the HiGHS
solver) and runs on a single CPU.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w ≥ 0, Σw = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len‑1)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a (used in SGD)."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Parametric families (symmetric “U‑shape”, product‑U and exponential)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] = (min(i, n‑1‑i) + 1)^{‑p}, normalised to sum‑1."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] = ((i+1)*(n‑i))^{‑p}, normalised."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] = exp(-lam*i) + exp(-lam*(n‑1‑i)), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# Helpers for LP refinement
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray):
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that
        A @ b = conv(f, b)
    for any non‑negative vector b.
    """
    from scipy.sparse import coo_matrix

    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs from i_start .. i_end inclusive
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals               # column indices
        rows.extend([k] * i_vals.size)
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())
    A = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_refine(f: np.ndarray, time_deadline: float,
              max_iters: int = 12, t_initial: float = 0.25):
    """
    Perform a few LP‑based refinement steps on the vector f.
    Each step solves:

        maximise   Σ b_i
        subject to b_i ≥ 0,
                   conv(f, b) ≤ M   (M = max conv of f)

    The solution b̂ is blended into f with mixing weight t.
    """
    from scipy.optimize import linprog

    n = f.shape[0]
    best_f = f.copy()
    best_score = score_from_vector(best_f)
    t = t_initial

    for it in range(max_iters):
        if time.time() > time_deadline:
            break
        # current bound
        cur_conv = full_convolution_fft(f)
        M = cur_conv.max()
        # build constraint matrix
        A = build_convolution_matrix(f)
        ub = np.full(2 * n - 1, M)
        c = -np.ones(n)                     # maximise Σb_i  ⇒ minimise -Σb_i
        res = linprog(c, A_ub=A, b_ub=ub,
                      bounds=(0, None), method='highs')
        if not res.success or res.x is None:
            break
        b = res.x
        total = b.sum()
        if total <= 0:
            break
        g = b / total                     # normalise to sum‑1
        # convex combination
        f = (1.0 - t) * f + t * g
        # ensure exact normalisation (numerical drift)
        f /= f.sum()
        # evaluate
        sc = score_from_vector(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
        # shrink mixing weight – we need ever smaller steps
        t = max(0.01, t * 0.85)
    return best_f, best_score


def interpolate_to_length(a_src: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of a_src to length n_target, then normalise."""
    if a_src.shape[0] == n_target:
        return a_src.copy()
    x_src = np.arange(a_src.shape[0])
    x_tgt = np.linspace(0, a_src.shape[0] - 1, n_target)
    a_tgt = np.interp(x_tgt, x_src, a_src)
    a_tgt = np.maximum(a_tgt, 0.0)
    a_tgt /= a_tgt.sum()
    return a_tgt

        print(f"LP refinement improved score to {sc_refined:.6f}")

    # --------------------------------------------------------------
    # 3. Upscaling to larger lengths and quick evaluation
    # --------------------------------------------------------------

    for n_target in upscale_lengths:
        if time.time() > deadline:
            break
        a_up = interpolate_to_length(f_small, n_target)
        sc = score_from_vector(a_up)
        if sc < best_score:
            best_score, best_seq = sc, a_up.copy()
            print(f" upscale candidate  n={n_target:5d}  score={sc:.6f}")

    # --------------------------------------------------------------
    # 4. Stochastic refinement (sub‑gradient + simulated annealing)
    # --------------------------------------------------------------
    curr = best_seq.copy()
    n = curr.shape[0]
    curr /= curr.sum()
    conv_curr = full_convolution_fft(curr)
    curr_score = float(2.0 * n * conv_curr.max())

    # hyper‑parameters
    temperature = 1.0
    temp_decay = 0.99992
    step_size = 0.3
    step_decay = 0.99985
    iteration = 0
    last_improve = 0

    print("\n=== Stochastic refinement start ===")
    while time.time() < deadline:
        iteration += 1

        grad = compute_subgradient(curr, conv_curr)
        cand = curr - step_size * grad
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)

        # b) occasional random mass‑transfer move
        if iteration % 97 == 0:

            if curr[i] > 0.0:
                delta = curr[i] * np.random.rand() * 0.5
                cand_rand = curr.copy()

                cand_rand[j] += delta
                cand_rand = np.maximum(cand_rand, 0.0)
                cand_rand = simplex_projection(cand_rand)
                cand = cand_rand

        step_size *= step_decay

        # f) re‑heat after long stagnation
        if iteration - last_improve > 250000:
            bump = np.random.rand(n) * 1e-5
            curr = simplex_projection(curr + bump)
            curr /= curr.sum()
            conv_curr = full_convolution_fft(curr)
            curr_score = float(2.0 * n * conv_curr.max())
            temperature = 1.0
            step_size = 0.3
            last_improve = iteration
            print(f"Iter {iteration:8d}: re‑heat after stagnation")

    # --------------------------------------------------------------
    # 5. Return best vector found
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score achieved: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator rescales, so we simply return the raw list.
    return best_seq.tolist()
