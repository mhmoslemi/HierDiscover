#!/usr/bin/env python
"""
Implementation of propose_candidate to minimize evaluation bound.
Algorithm:
1. Deterministic sweep over several parametric families (U-shaped
   distributions) for multiple lengths and parameter values.
2. If cvxpy is available, perform a few iterations of a linear‑program
   based refinement (maximising sum(g) under the convolution bound),
   followed by an optional convex combination.
3. Finish with a short projected sub‑gradient descent to polish the
   solution.
All vectors are kept normalised to sum=1, i.e. they lie on the probability
simplex.
"""
import time
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a unit‑sum vector."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a (only active max positions)."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

def u_min_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric “U‑shape’’ based on distance to the nearest edge."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric “U‑shape’’ based on product of distances to both edges."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a

def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""

"""
Implementation of propose_candidate to minimise the evaluation bound.

Algorithm
---------
1. Deterministic sweep over several parametric “U‑shaped” families
   (inverse‑distance, inverse‑product‑distance, power‑law and exponential
   decay) for a handful of lengths and parameter values.  The best vector
   from this sweep becomes the initial seed.

2. If CVXPY is available, perform a few iterations of a linear‑programming
   refinement.  For the current vector f we solve

        maximise   sum(g)
        subject to f * g ≤ M   (elementwise)
                  g ≥ 0

   where M = max_k (f * f)[k].
   The optimal g is normalised to sum(g)=1 and either replaces f
   (if it improves the score) or is blended with f.

3. A short projected sub‑gradient descent finishes the search,
   polishing the best candidate found.

All vectors are kept on the probability simplex (non‑negative, sum = 1)
so the denominator of the evaluation function is always 1 and the score
is simply `2·n·max(conv)`.  The routine respects the given time budget
(`budget_s` seconds) and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w≥0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)

def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] with respect to a (active max entries)."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad
