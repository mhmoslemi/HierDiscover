def propose_candidate(seed=..., budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # choose n
    n = 8000
    # initialize a0: try to load from height_sequence_1 and interpolate; else uniform.
    ...

    w = np.log(a0 + 1e-12)

    beta1,beta2 = 0.9,0.999
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    # hyperparameters
    lr_start = 0.05
    lr_end = 0.001

    eps_start = 5e-3
    eps_end = 0.0   # will go to zero

    best_a = a0.copy()
    best_val = float('inf')
    no_improve = 0
    IMPROVE_LIMIT = 50000

    while time.time() < deadline:
        # compute a via softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        conv = _full_convolution(a)  # maybe use FFT full conv
        # objective:
        cur_val = 2.0 * n * conv.max()
        # track best
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[{t}] best = {best_val}")
        else:
            no_improve += 1

        progress = (time.time() - start_time) / (deadline - start_time)
        # step-size schedule
        lr = lr_start * (1.0 - progress) + lr_end * progress
        eps = max(eps_end, eps_start * (1.0 - progress) + eps_end * progress)

        # smoothing (if epsilon > small tolerance)
        if eps > 1e-12:
            # softmax p
            max_c = conv.max()
            shifted = (conv - max_c) / eps
            weights = np.exp(shifted)
            p = weights / weights.sum()
        else:
            # hard mask: select indices where conv approx max (within tolerance)
            max_c = conv.max()
            tol = max_c * 1e-12 + 1e-15
            mask = (conv >= max_c - tol).astype(float)
            # ensure mask sums to 1 as probability distribution
            p = mask / mask.sum()

        # gradient of smoothed objective w.r.t a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n-1:2*n-1]
        grad_a = 4.0 * n * segment[::-1]

        # gradient wrt w via softmax Jacobian
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturb
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.02, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[{t}] restart after {no_improve} steps")
            w = np.random.randn(n) * 0.1
            m.fill(0)
            v.fill(0)
            t = 0
            no_improve = 0

    final_score = evaluate_sequence(best_a.tolist())
    print("[final] score", final_score)
    return best_a.tolist()
