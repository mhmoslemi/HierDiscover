"""
Propose a step‑function (non‑negative vector) that minimises the
evaluation metric used by `evaluate_sequence`.

Algorithm
---------

1.  **Initial candidates** – For several lengths (8 k … 30 k) we generate
    simple *U‑shaped* power‑law distributions
    a_i ∝ ((i+1)∧(n−i))^(−p) and mixtures with a uniform component.
    All candidates are normalised to sum 1 and their exact objective
    2·n·max(a∗a) is evaluated by an FFT based convolution.  The best
    candidate (lowest score) is kept as a seed.

2.  **Sub‑gradient refinement** – Starting from the seeded vector we run
    a projected sub‑gradient descent on the *maximum* of the
    self‑convolution:

        * The active set consists of all indices where the convolution
          attains (or almost attains) its current maximum.
        * The sub‑gradient of the maximum is the average of the gradients
          of those active convolutions; it can be obtained efficiently
          by a pair of FFT‑based convolutions (the same routine used for
          the soft‑max gradient in the reference code).
        * A step `a ← a – η·grad` is taken and the result is projected back
          onto the probability simplex (non‑negative, sum = 1).
        * The step size η is adapted: it is increased after a successful
          step and shrunk after a failure.  If the algorithm stalls for
          a few hundred iterations a small random perturbation is added
          to escape flat regions.

3.  **Length up‑sampling** – The refined vector is up‑sampled (linear
    interpolation) to a larger length and the sub‑gradient phase is
    repeated.  Larger n gives a larger factor 2·n in the objective, but
    a well‑flattened convolution can keep the product 2·n·max low.

4.  The best vector found in the whole run is finally handed to the
    official `evaluate_sequence` function and returned.

The whole procedure is limited by the supplied time budget (default
1000 s) and never performs any file or network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a supplied FFT size."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k (conv = a*a) with respect to a.
    Implements the same formula used for the smooth‑max gradient.
    """
    n = a.shape[0]
    L = 2 * n - 1               # length of w / conv
    # FFT of a
    A = np.fft.rfft(a, n=size)

    # FFT of weights and of the reversed weight vector
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # a * w  (length n+L‑1 = 3·n‑2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    # gradient formula
    idx = L - 1 - np.arange(n)          # indices to pick from rev‑convolution
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho_arr = u - cssv / ind > 0
    if not np.any(rho_arr):
        return np.full_like(v, 1.0 / n)
    rho = ind[rho_arr][-1] - 1
    theta = cssv[rho_arr][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law “U‑shape”: a_i ∝ ((i+1)∧(n−i))^(−p).
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)                     # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14                     # avoid exact zero
    a /= a.sum()
    return a


def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length old_n to new_n, then re‑normalise."""
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new += 1e-14
    new /= new.sum()
    return new

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """Search for a non‑negative step‑function that minimises `evaluate_sequence`."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # 0.  Try to seed from a previously supplied construction, if any.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            if external.sum() < 1e-12:
                external = np.ones_like(external) / external.size
            external = external / external.sum()
            cur_score = 2.0 * external.size * max_conv_fft(external)
            best_seq = external.copy()
            best_score = cur_score
            print(f"Seeded from height_sequence_1: n={external.size}, score≈{cur_score:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Scan simple parametric families (U‑shape + uniform mixture).
    # ------------------------------------------------------------------
    lengths_to_try = [8000, 12000, 20000, 30000]   # moderate set, later we may enlarge
    p_grid = np.linspace(0.4, 1.2, 9)             # 0.4 … 1.2
    uniform_weights = [0.0, 0.02, 0.05, 0.10]

    for n in lengths_to_try:
        if time.time() >= deadline:
            break
        print(f"\n--- Scanning candidates for length n={n} ---")
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = u_shaped_distribution(n, p)
            # pure U‑shape
            cur_max = max_conv_fft(base)
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = base.copy()
                print(f"p={p:.3f}, w=0.0 → score≈{cur_score:.6f}")

            # mixtures with a uniform component
            for w in uniform_weights:
                if w == 0.0:
                    continue
                cand = (1 - w) * base + w * np.ones(n) / n
                cand /= cand.sum()
                cur_max = max_conv_fft(cand)
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cand.copy()
                    print(f"p={p:.3f}, w={w:.3f} → score≈{cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Sub‑gradient refinement on the current best seed.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback uniform
        n0 = 8000
        best_seq = np.ones(n0) / n0
        best_score = 2.0 * n0 * max_conv_fft(best_seq)
        print(f"Fallback initialisation: n={n0}, score≈{best_score:.6f}")

    def refine(a: np.ndarray, max_iters: int = 5000) -> tuple[np.ndarray, float]:
        """
        Projected sub‑gradient descent on max(conv(a,a)).
        Returns the best vector seen and its (approximate) score.
        """
        n = a.shape[0]
        fft_size = 1 << ((3 * n - 2).bit_length())
        eta = 0.08          # initial step size
        eta_min = 1e-6
        eta_max = 1.0
        best_local = a.copy()
        best_local_score = 2.0 * n * max_conv_fft(a)
        no_improve = 0

        for it in range(max_iters):
            if time.time() >= deadline:
                break

            conv = full_conv_fft(a, fft_size)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep local best
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a.copy()
                no_improve = 0
                # update global best as well
                nonlocal best_seq, best_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"Iter {it:04d} – new global best: {cur_score:.6f} (n={n})")
            else:
                no_improve += 1

            # ------- active set (indices with value ≈ max) -------
            eps = max(1e-14, cur_max * 1e-12)
            active = np.where(conv >= cur_max - eps)[0]
            w = np.zeros_like(conv)
            if active.size:
                w[active] = 1.0 / active.size
            else:                               # safety fallback
                w[np.argmax(conv)] = 1.0

            # ------- gradient of the active (sub‑gradient) -------
            grad = gradient_softmax(a, w, fft_size)

            # ------- candidate step -------
            a_cand = a - eta * grad
            a_cand = project_to_simplex(a_cand)

            # ------- evaluate candidate -------
            conv_cand = full_conv_fft(a_cand, fft_size)
            cand_max = conv_cand.max()
            cand_score = 2.0 * n * cand_max

            # ------- accept / reject -------
            if cand_score < cur_score:
                a = a_cand
                eta = min(eta * 1.07, eta_max)   # become slightly more aggressive
            else:
                eta = max(eta * 0.73, eta_min)   # shrink step

            # ------- occasional random perturbation to escape plateaus -------
            if no_improve >= 300:
                noise = np.random.randn(n) * (eta * 0.5)
                a = project_to_simplex(a + noise)
                no_improve = 0

        return best_local, best_local_score

    # initial refinement on the currently best length
    print("\n=== Refinement phase (first length) ===")
    best_seq, _ = refine(best_seq, max_iters=4000)

    # ------------------------------------------------------------------
    # 3.  Upsample to larger lengths and refine again (if time permits).
    # ------------------------------------------------------------------
    upsample_factors = [1.2, 1.5, 2.0]          # relative growth
    for factor in upsample_factors:
        if time.time() >= deadline:
            break
        n_old = best_seq.shape[0]
        n_new = int(n_old * factor)
        if n_new > 50000:            # keep FFT sizes reasonable
            n_new = 50000
        print(f"\n=== Upsampling to n={n_new} (factor {factor:.2f}) ===")
        a_up = resample_distribution(best_seq, n_new)
        a_up, _ = refine(a_up, max_iters=3000)
        best_seq = a_up

    # ------------------------------------------------------------------
    # 4.  Final official evaluation & return.
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:
        print("Could not call evaluate_sequence:", exc)

    print("\n=== Search finished ===")
    print(f"Best approximate score : {best_score:.6f}")
    print(f"Best length            : {best_seq.shape[0]}")
    return best_seq.tolist()
