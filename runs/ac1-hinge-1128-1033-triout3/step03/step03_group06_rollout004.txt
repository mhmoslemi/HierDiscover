import time, numpy as np

def max_conv_fft(a):
  n=a.shape[0]
  size = 1 << ((2*n-1).bit_length())
  A=np.fft.rfft(a,n=size)
  conv = np.fft.irfft(A*A, n=size)[:2*n-1]
  return conv.max()

def full_conv_fft(a,size):
  A=np.fft.rfft(a,n=size)
  conv = np.fft.irfft(A*A, n=size)[:2*a.shape[0]-1]
  return conv

def softmax_weights(conv, alpha):
  mx=conv.max()
  exps=np.exp(alpha*(conv-mx))
  s=exps.sum()
  if s==0:
    return np.full_like(conv, 1.0/conv.size)
  return exps/s

def gradient_softmax(a,w,size):
  n=a.shape[0]
  L = 2*n-1
  A=np.fft.rfft(a, n=size)
  W=np.fft.rfft(w, n=size)
  W_rev=np.fft.rfft(w[::-1], n=size)
  conv_aw=np.fft.irfft(A*W, n=size)[: n+L-1]
  conv_aw_rev = np.fft.irfft(A*W_rev, n=size)[: n+L-1]
  idx = L-1 - np.arange(n)
  grad = conv_aw[:n] + conv_aw_rev[idx]
  return grad

def softmax_logits(x):
    m = x.max()
    ex = np.exp(x - m)
    return ex / ex.sum()

"""
Propose a step‑function (non‑negative vector) that minimises the objective
used by `evaluate_sequence`.

Algorithm
---------
* Initialise from the previously supplied construction if it exists,
  otherwise from a simple U‑shaped distribution.
* For several candidate lengths (4000 – 20000) run a few independent
  Adam‑optimisations in the *log‑probability* (soft‑max) parametrisation.
* The smooth approximation `Sα(a)= (1/α)·log Σ exp(α·(a*a))` is used as a
  surrogate for `max(a*a)`.  Its gradient is obtained by FFT‑based
  convolution (three FFTs per iteration) and then back‑propagated to the
  logits.
* The smoothing parameter `α` is increased slowly; once it becomes large
  (`α≈α_max`) the soft‑max is replaced by a hard max (weights are a
  normalised indicator of the maximal convolution entries) to obtain an
  exact sub‑gradient.
* Adam’s per‑coordinate learning‑rates adapt automatically, and the best
  vector seen across all runs is kept.
* When the time budget (≈1000 s) is exhausted the best vector is finally
  evaluated with `evaluate_sequence` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using supplied FFT size."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Normalised soft‑max weights w_k = exp(α·conv_k)/Σexp."""
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:          # numerical safeguard
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution with respect to a.
    Implements the formula grad_i = Σ_k w_k (a_{k-i}+a_{k-i}) using FFTs.
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of w
    # FFT of a
    A = np.fft.rfft(a, n=size)
    # FFT of w and its reversal
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # Convolution a * w   (length n+L‑1 = 3·n‑2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    # Convolution a * rev(w)
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    # Assemble gradient (two symmetric contributions)
    idx = L - 1 - np.arange(n)                     # 2·n‑2, …, n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def softmax_logits(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max of a vector of logits."""
    m = x.max()
    ex = np.exp(x - m)
    return ex / ex.sum()


def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new += 1e-14            # avoid exact zeros
    new /= new.sum()
    return new


def u_shaped_distribution(n: int, p: float = 0.8) -> np.ndarray:
    """Simple symmetric power‑law “U‑shape”."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)               # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def arcsine_distribution(n: int, delta: float = 0.5) -> np.ndarray:
    """
    Discrete approximation of an arcsine‑like density
    ρ(x) ∝ 1/√(x(1−x)+δ²) on {0,…,n−1}.
    The small `delta` regularises the singularities at the ends.
    """
    i = np.arange(n)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x) + delta * delta)
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects the given time budget and
    returns the best list of floats it found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # --------------------------------------------------------------
    # 0.  Initialise from a previously supplied construction if possible
    # --------------------------------------------------------------
    best_seq = None
    global_best_score = np.inf
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external = external / s
            cur_max = max_conv_fft(external)
            cur_score = 2.0 * external.size * cur_max
            best_seq = external.copy()
            global_best_score = cur_score
            print(f"Seeded from height_sequence_1: n={external.size}, score≈{cur_score:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 1.  Fallback initialisation if no seed was available
    # --------------------------------------------------------------
    if best_seq is None:
        n0 = 8000
        best_seq = u_shaped_distribution(n0, p=0.8)
        global_best_score = 2.0 * n0 * max_conv_fft(best_seq)
        print(f"Fallback initialisation: n={n0}, score≈{global_best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Lengths to explore (larger n ⇒ larger factor 2·n in the score)
    # --------------------------------------------------------------
    lengths_to_try = [4000, 6000, 8000, 12000, 16000, 20000]
    lengths_to_try = sorted(set(lengths_to_try))

    # --------------------------------------------------------------
    # 3.  Core optimisation: Adam on logits with a smooth max surrogate
    # --------------------------------------------------------------
    global_best_seq = best_seq.copy()

    # hyper‑parameters (these are generous but respect the time budget)
    max_restarts_per_len = 3
    max_iters_per_restart = 3000   # will be cut by the time budget
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.5
    alpha_start = 1.0
    alpha_max = 400.0
    alpha_mul = 1.001

    for n in lengths_to_try:
        if time.time() >= deadline:
            break
        fft_size = 1 << ((3 * n - 2).bit_length())   # ≥ 3n‑2 for the gradient step
        print(f"\n=== Optimising length n={n} (FFT size={fft_size}) ===")
        for restart in range(max_restarts_per_len):
            if time.time() >= deadline:
                break

            # ------------------------------------------------------
            # 3.1  Choose an initial distribution
            # ------------------------------------------------------
            if best_seq.shape[0] != n:
                a0 = resample_distribution(best_seq, n)
            else:
                a0 = best_seq.copy()

            # Mix several seeds: U‑shape, arcsine, random Dirichlet
            if restart == 0:
                a0 = u_shaped_distribution(n, p=0.8)
            elif restart == 1:
                a0 = arcsine_distribution(n, delta=0.5)
            elif restart == 2:
                a0 = np.random.rand(n)
                a0 /= a0.sum()

            # Add a tiny log‑space noise to avoid symmetry traps
            x = np.log(a0 + 1e-18) + 0.1 * np.random.randn(n)

            # Adam state
            m = np.zeros_like(x)
            v = np.zeros_like(x)

            # Tracking the best vector for this restart
            best_local_a = None
            best_local_score = np.inf

            alpha = alpha_start

            for it in range(1, max_iters_per_restart + 1):
                if time.time() >= deadline:
                    break

                # ---- (a) probability vector from logits
                a = softmax_logits(x)

                # ---- (b) true convolution and its max (exact objective)
                conv = full_conv_fft(a, fft_size)
                max_conv_val = conv.max()
                cur_score = 2.0 * n * max_conv_val

                # update the best seen so far (both locally and globally)
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_a = a.copy()
                if cur_score < global_best_score:
                    global_best_score = cur_score
                    global_best_seq = a.copy()
                    print(f"Iter {it:4d} – new global best (n={n}): {cur_score:.6f}")

                # ---- (c) smooth‑max weights
                if alpha >= alpha_max * 0.85:
                    # Switch to an exact max sub‑gradient when α is large
                    mask = (conv == max_conv_val)
                    w = mask.astype(float) / mask.sum()
                else:
                    w = softmax_weights(conv, alpha)

                # ---- (d) gradient w.r.t. a
                grad_a = gradient_softmax(a, w, fft_size)

                # ---- (e) chain rule to logits (x‑space)
                dot = np.dot(grad_a, a)
                grad_x = a * (grad_a - dot)

                # ---- (f) Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                t = it
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                lr_t = base_lr * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
                x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

                # ---- (g) increase smoothing parameter α
                alpha = min(alpha * alpha_mul, alpha_max)

                # occasional progress report
                if it % 500 == 0:
                    print(f"   n={n:5d} restart={restart} it={it:5d} "
                          f"score={cur_score:.6f} α={alpha:.1f}")

            # after restart: seed the next length with the best we found here
            if best_local_a is not None:
                best_seq = best_local_a.copy()

            # early exit if we already beat a generous target
            if global_best_score < 1.48:
                print("Target comfortably reached – exiting early.")
                break
        if global_best_score < 1.48:
            break

    # --------------------------------------------------------------
    # 4.  Final official evaluation
    # --------------------------------------------------------------
    try:
        exact = evaluate_sequence(global_best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact:.6f}")
    except Exception as exc:
        print(f"Evaluation failed: {exc}")
        exact = global_best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {global_best_score:.6f}")
    print(f"Best length            : {len(global_best_seq)}")
    return global_best_seq.tolist()
