"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` metric.

Algorithm
---------

1.  Load the optional seed `height_sequence_1`.  If it exists, normalise it
    to a probability vector and store it as the current best.

2.  For a selection of lengths `n` (up to 50 000) scan a one‑parameter
    family of “U‑shaped’’ vectors
        a_i ∝ (i+1)^(-p) + (n−i)^(-p)   (p>0)
    on a coarse grid of exponents `p`.  The objective
        f(a) = 2·n·max(conv(a,a))
    is evaluated by an FFT based convolution (`max_conv_fft`).  The best
    candidate from this coarse scan is kept.

3.  Starting from the best coarse candidate for each length we run a
    mirror‑descent / exponentiated‑gradient optimisation on the smooth
    approximation of the max‑convolution
        S_α(a) = (1/α)·log Σ exp(α·conv_k(a))
    using the soft‑max weights `w_k` and the gradient obtained via FFT.
    An Adam‑style adaptation of the per‑coordinate step is employed and
    the learning‑rate is increased on successful steps and decreased on
    failures.  The smoothing parameter `α` is slowly increased (making
    the bound tighter) until a maximum of `α_max` is reached.

4.  All updates keep the vector on the probability simplex (non‑negative,
    sum = 1) by a multiplicative update followed by a Euclidean projection.

5.  The best vector seen over the whole search (across all lengths and
    iterations) is finally evaluated with the official
    `evaluate_sequence` and returned.

The routine respects a wall‑clock budget (default 1000 s) and prints
progress information.  No file or network I/O is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """
    Return the maximum entry of the self‑convolution a*a using a real FFT.
    a must be a 1‑D non‑negative array (normally sum(a)=1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·n‑1) using FFT of given length.
    """
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max of the convolution and its gradient
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Normalised soft‑max weights: w_k = exp(alpha*conv_k) / Σ exp(...)
    Stabilised by subtracting the maximum entry.
    """
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k(a) w.r.t. a.
    Implements the expression via FFTs (O(n log n)).
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of w
    A = np.fft.rfft(a, n=size)

    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # a * w
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    # a * w_rev
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    idx = L - 1 - np.arange(n)  # indices aligning the reversed part
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = 1}.
    Implementation based on the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Input must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper to build a symmetric U‑shaped distribution for a given exponent
# ----------------------------------------------------------------------


def u_shape_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric U‑shape: a_i ∝ (i+1)^(-p) + (n-i)^(-p).
    Returned vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    a += 1e-14                     # avoid exact zeros
    a /= a.sum()
    return a

            # occasional progress report
            if itr % 1000 == 0:
                print(f"Iter {itr:5d} – obj≈{cur_obj:.6f}, α={alpha:.1f}, η={eta:.4f}")

        # keep the last a as the seed for the next length
        best_seq = a.copy()

        if best_obj < 1.503:
            break

    # --------------------------------------------------------------
    # 4.  Final official evaluation & return
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: single entry with sum=1
        best_seq = np.array([1.0])

    # make sure the vector is a plain Python list of floats
    result = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(result)   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")

    print("\n=== Search finished ===")
    print(f"Best approx objective: {best_obj:.6f}")
    print(f"Best length: {best_n}")
    return result
