def golden_section(func, a, b, tol=1e-5, max_iter=30):
    invphi = (np.sqrt(5) - 1) / 2  # 1/phi
    invphi2 = (3 - np.sqrt(5)) / 2  # 1/phi^2
    # c and d inside interval
    c = a + invphi2 * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = a + invphi2 * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
    # best point is min(fc, fd)
    if fc < fd:
        return c, fc
    else:
        return d, fd

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. Deterministic sweep over several simple parametric families
   (U‑shapes, Beta‑shapes, exponential decay, piecewise constant).
   A fine grid of parameters is scanned for several lengths
   (8000, 12000, 20000).  The best normalised vector found is kept as a seed.

2. Stochastic refinement.
   Starting from the best seed we repeatedly
       – compute the self‑convolution (FFT),
       – build a sub‑gradient from all convolution entries that are within
         a tiny relative tolerance of the current maximum,
       – apply one of three projected moves:
            * additive projected sub‑gradient step,
            * multiplicative “mirror‑descent” step,
            * directed random mass transfer from a high‑gradient index to a low‑gradient index,
       – accept the move by a simulated‑annealing rule,
       – slowly cool the temperature and the step size,
       – re‑heat if no improvement for a long while.
   The loop runs as long as the 1000 s budget allows.

Both stages work with normalised vectors (sum = 1); the evaluation
function `evaluate_sequence` is scale‑invariant, so the returned list may be
used directly.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w ≥ 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# FFT‑based convolution utilities
# ------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length 2*len(a)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_vector(vec: np.ndarray, n: int) -> float:
    """Score of a (non‑negative) vector according to `evaluate_sequence`
    – the vector is normalised to sum = 1 before evaluation."""
    total = np.sum(vec)
    if total <= 1e-12:
        return np.inf
    a = vec / total
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Parametric families (all return un‑normalised vectors)
# ------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    return a

def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the two edges."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    return a

def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Symmetric beta(α,β) on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    return a

def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    return a

def piecewise_constant_distribution(n: int, w_frac: float, ratio: float) -> np.ndarray:
    """Two‑level shape:
       edge region of width w_frac·n gets weight `ratio`,
       interior gets weight 1."""
    edge_len = max(1, int(round(w_frac * n)))
    a = np.ones(n, dtype=np.float64)
    a[:edge_len] = ratio
    a[-edge_len:] = ratio
    return a


# ------------------------------------------------------------
# Gradient of max‑convolution (sub‑gradient)
# ------------------------------------------------------------
def subgradient_max_conv(a: np.ndarray, rel_tol: float = 1e-5):
    """Return (grad, max_val) where grad is a sub‑gradient of
    max_k (a*a)[k] at the current vector a."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_val = conv.max()
    active = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad, max_val

# ------------------------------------------------------------
