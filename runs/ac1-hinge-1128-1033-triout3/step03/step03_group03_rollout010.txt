"""
Optimisation via soft‑max smoothed sub‑gradient descent.

The objective (after normalising ∑a_i = 1) is
    J(a) = 2·n·max(conv(a,a))

The non‑smooth max is replaced by a temperature‑scaled soft‑max:
    Sτ(a) = Σ_k exp(conv_k/τ)·conv_k / Σ_k exp(conv_k/τ)

∇Sτ can be written as a convolution, so each iteration uses only FFT‑based
convolutions (O(n·log n)).  After a gradient step we project back onto the
probability simplex.  The temperature τ is annealed from a relatively large
value (smooth landscape) to a tiny one (focus on the true maximum).  The
step size is also decayed and occasional random restarts are performed if
no improvement is observed for a while.

The routine runs for the allotted wall‑clock budget (≈ 1000 s) and returns
the best sequence found.  It works with any initial sequence; if the global
variable ``height_sequence_1`` is present it is used as a warm start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x | x ≥ 0, sum(x) = 1} using Duchi et al."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _next_pow2(x: int) -> int:
    """Small utility – next power of two not smaller than *x*."""
    return 1 << (x - 1).bit_length()


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm is a soft‑max smoothed gradient
    descent on the maximal self‑convolution, using FFTs for speed.
    """
    # ------------------------------------------------------------------
    # 1) time budget
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0  # a safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 2) problem size and initial point
    # ------------------------------------------------------------------
    # try to use the previously found construction if available
    global height_sequence_1
    if isinstance(globals().get("height_sequence_1"), list):
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = min(len(init_seq), 2000)          # keep n moderate for speed
        a = init_seq[:n].copy()
    else:
        # fall back to the Catalan‑type (central binomial) coefficients
        n = 2000
        a = np.empty(n, dtype=np.float64)
        cur = 1.0
        a[0] = cur
        for k in range(1, n):
            cur *= (2 * k - 1) / (2 * k)
            a[k] = cur
    # make sure everything is non‑negative and on the simplex
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    a = _project_simplex(a)           # guarantees ∑a_i = 1

    # ------------------------------------------------------------------
    # 3) bookkeeping
    # ------------------------------------------------------------------
    # evaluate the initial candidate exactly (cheap – once)
    best_seq = a.copy()
    best_val = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial_score={best_val:.10f}")

    # parameters for the annealed gradient descent
    eta = 0.5                     # base step‑size (will be normalised)
    eta_decay = 0.9999
    min_eta = 1e-6

    tau = 0.5                     # temperature for the soft‑max
    tau_decay = 0.9995
    min_tau = 1e-5

    no_improve = 0
    max_no_improve = 5000

    it = 0

    # pre‑compute FFT lengths (powers of two) – they stay constant
    L1 = _next_pow2(2 * n - 1)            # for a * a
    L2 = _next_pow2(3 * n - 2)            # for a * w_rev

    # ------------------------------------------------------------------
    # 4) main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- self‑convolution (via FFT) -----
        A = np.fft.rfft(a, L1)
        conv = np.fft.irfft(A * A, L1)[: 2 * n - 1]
        M = conv.max()

        # ----- soft‑max weights -----
        # w_i ∝ exp((conv_i - M)/tau)  – numerically stable
        w = np.exp((conv - M) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:          # should never happen, but guard against it
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # ----- gradient of the smoothed objective -----
        # g_j = 2 * Σ_i a_i * w_{i+j}
        w_rev = w_norm[::-1]                     # reverse for the convolution
        A2 = np.fft.rfft(a, L2)
        W2 = np.fft.rfft(w_rev, L2)
        conv2 = np.fft.irfft(A2 * W2, L2)        # length ≥ 3n‑2
        idxs = (2 * n - 2) - np.arange(n)       # map to required entries
        g = 2.0 * conv2[idxs]

        # ----- adaptive additive step (projected onto the simplex) -----
        max_g = g.max()
        if max_g > 0:
            step = eta / max_g            # guarantees max change ≈ eta
            a_new = a - step * g
        else:
            a_new = a.copy()
        a_new = _project_simplex(a_new)

        # ----- quick evaluation of the new point (approximate) -----
        A_new = np.fft.rfft(a_new, L1)
        conv_new = np.fft.irfft(A_new * A_new, L1)[: 2 * n - 1]
        cand_val = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        # ----- update best if improvement observed -----
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_seq = a_new.copy()
            no_improve = 0
            print(f"[iter {it:6d}] score={best_val:.10f}   eta={eta:.6f}   tau={tau:.6f}")
        else:
            no_improve += 1

        # ----- accept the move (even if not improving) -----
        a = a_new

        # ----- schedules -----
        eta = max(min_eta, eta * eta_decay)
        tau = max(min_tau, tau * tau_decay)

        # occasional renormalisation to avoid accumulation of tiny numerical drift
        if it % 1000 == 0:
            a = a / a.sum()

        # ----- random restart if we are stuck -----
        if no_improve > max_no_improve:
            # perturb the best sequence a little, otherwise start from uniform
            if best_seq is not None and rng.random() < 0.5:
                a = best_seq + rng.normal(scale=0.02, size=n)
                a = np.maximum(a, 0.0)
            else:
                a = rng.random(n)
            a = a / a.sum()
            eta = 0.5
            tau = 0.5
            no_improve = 0
            print(f"[restart] iteration {it}")

    # ------------------------------------------------------------------
    # 5) finalise and return
    # ------------------------------------------------------------------
    # clamp to the limits respected by the evaluator
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(best_seq)
    print(f"[result] final_score={final_score:.10f}, length={len(best_seq)}")
    return best_seq
