    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] norm val = {best_val:.6f}")

    # define projection to simplex
    def project_simplex(v):
        # returns projection onto simplex sum=1, v>=0
        # Using algorithm: sort descending, find rho, theta.
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    # Helper to compute max conv fast with FFT (real)
    def full_conv(x):
        n_local = x.shape[0]
        out_len = 2*n_local - 1
        size = 1 << (out_len-1).bit_length()
        fx = np.fft.rfft(x, size)
        conv = np.fft.irfft(fx * fx, size)[:out_len]
        # Clip possible -0.0: conv = np.maximum(conv, 0, out=conv)
        return conv

    # Subgradient descent loop
    # parameters
    max_iters = 1000000  # but will break earlier based on time
    eta0 = 1.0  # initial step size
    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = full_conv(a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val  # normalized sum=1
        if cur_val < best_val - 1e-9:  # small epsilon
            best_val = cur_val
            best_a = a.copy()
            print(f"[{t}] new best {best_val:.6f}")

        # Find indices where conv is close to max (within tolerance)
        # tolerance maybe 1e-12 relative
        tol = max_val * 1e-12
        max_idxs = np.where(conv >= max_val - tol)[0]
        # Compute subgradient as average of gradients for each max index
        grad = np.zeros_like(a)
        for k in max_idxs:
            # for i in [max(0,k-n+1), min(n-1, k)]
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            # contributions: grad_i += a_{k-i}

            grad[idxs] += a[k - idxs]

    return best_a.tolist()

"""
Propose a step‑function (vector of non‑negative heights) that minimizes
evaluate_sequence().  The algorithm works in three phases:

1.  Initialise from the best known construction (height_sequence_1) if
    available, otherwise from a uniform distribution, and up‑sample to
    n = 8000 entries.

2.  Run a smoothed‑max Adam optimisation on the soft‑max parametrisation
    a = softmax(w).  The objective 2·n·max(conv(a,a)) is approximated by
    a log‑sum‑exp with temperature ε that is linearly annealed from
    1e‑2 to 1e‑9.  Gradients are computed via FFT‑based convolutions.
    Adaptive learning‑rate (linear decay) and Adam moments are used.
    Random restarts and occasional perturbations keep the search from
    stagnating.  The best exact objective seen is stored.

3.  With any remaining time a cheap hill‑climbing stage moves a small
    amount of mass between two random positions and accepts the move if
    it improves the exact objective.  This fine‑tunes the solution.

The final best sequence (list of floats) is returned.

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (real inputs)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Return the full convolution x*x (length 2·len(x)‑1).
    Results are clipped to non‑negative values to avoid tiny negative
    round‑off errors.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power‑of‑two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)                # clip very small negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Problem size – a larger n gives a richer search space.
    # --------------------------------------------------------------
    n = 8000                                   # feel free to adjust

    # --------------------------------------------------------------
    # Initialise a from the previous best construction if possible.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0, 1, old.shape[0])
        new_x = np.linspace(0, 1, n)
        a0 = np.interp(new_x, old_x, old)
    except Exception:
        a0 = np.full(n, 1.0 / n)

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()                         # ensure Σa = 1

    # soft‑max parameters (unconstrained)
    w = np.log(a0 + 1e-12)

    # --------------------------------------------------------------
    # Adam hyper‑parameters and annealing schedules
    # --------------------------------------------------------------
    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.07
    lr_end = 0.0005

    eps_start = 1e-2
    eps_end = 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    # bookkeeping
    best_a = a0.copy()
    best_val = float('inf')
    no_improve_cnt = 0
    IMPROVE_LIMIT = 20000   # steps without improvement before a restart

    # initial exact evaluation
    conv0 = _full_convolution(best_a)
    best_val = 2.0 * n * conv0.max()
    print(f"[init] n={n}, initial exact value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smoothed‑max Adam optimisation
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → feasible a (non‑negative, sum = 1)
        w_shift = w - np.max(w)                # for numerical stability
        a = np.exp(w_shift)
        a = a / a.sum()

        # exact convolution and objective
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve_cnt = 0
            print(f"[{t:7d}] new best = {best_val:.6f}")
        else:
            no_improve_cnt += 1

        # ------------------------------------------------------------------
        # Smooth‑max setup (ε annealing)
        # ------------------------------------------------------------------
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # p_k = soft‑max(conv_k / ε)
        shifted = (conv - max_conv) / eps          # ≤ 0
        weights = np.exp(shifted)
        p = weights / weights.sum()                # length 2n‑1, sums to 1

        # ------------------------------------------------------------------
        # Gradient of the smoothed objective w.r.t. a
        #   g = 4·n·(a * p)   (implemented via convolution)
        # ------------------------------------------------------------------
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)              # length 3n‑2
        segment = conv_ab[n - 1: 2 * n - 1]         # middle n entries
        grad_a = 4.0 * n * segment[::-1]           # shape (n,)

        # ------------------------------------------------------------------
        # Chain‑rule through the soft‑max (gradient w.r.t. w)
        # ------------------------------------------------------------------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ------------------------------------------------------------------
        # Adam update
        # ------------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional perturbation to kick the search out of plateaus
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.01, size=n)

        # random restart if we have been idle too long
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[{t}] Restarting after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

    # --------------------------------------------------------------
    # Phase 2 – simple hill‑climbing with exact objective
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        hc_deadline = time.time() + remaining * 0.5
        a = best_a.copy()
        cur_val = best_val
        while time.time() < hc_deadline:
            # propose moving a small amount of mass from i to j
            i, j = np.random.choice(n, size=2, replace=False)
            if a[i] <= 0:
                continue
            delta = a[i] * np.random.uniform(0.0, 0.05)   # ≤5 % of a[i]
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # exact evaluation
            conv_new = _full_convolution(a_new)
            val_new = 2.0 * n * conv_new.max()
            if val_new < cur_val - 1e-12:
                a = a_new
                cur_val = val_new
                best_a = a_new.copy()
                best_val = val_new
                print(f"[hc] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final check using the supplied evaluate_sequence (extra safety)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
