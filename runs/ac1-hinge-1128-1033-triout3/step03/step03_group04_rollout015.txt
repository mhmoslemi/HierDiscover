"""
propose_candidate
-----------------

Search for a non‑negative step‑function (vector of heights) that minimises
the objective used by `evaluate_sequence`.  The algorithm works in two
phases:

1. **Deterministic sweep** – a set of parametric families (U‑shapes,
   beta, exponential, cosine, triangular, etc.) is sampled for several
   lengths.  The best vector from this sweep becomes the initial seed.

2. **Smooth‑max gradient descent** – the objective
   ``max_k (a*a)[k]`` is approximated by a soft‑max
   ``(1/λ)·log Σ exp(λ·conv_k)``.  The gradient of this smooth surrogate
   can be computed efficiently with three FFTs per iteration (self‑
   convolution, a convolution with the soft‑max weights, and the FFT of the
   weight vector).  The iterates are projected onto the probability simplex
   (non‑negative and sum = 1).  An annealing schedule gradually increases the
   soft‑max sharpness λ and decreases the step size, while a simulated‑annealing
   acceptance rule allows occasional uphill moves.  Stagnation triggers a
   small random “re‑heat”.

The routine runs for the whole time budget (default ≈ 1000 s) and returns
the best vector found.  All numbers are non‑negative; the external evaluator
will clip and renormalise again, so we return the raw simplex‑projected vector.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper utilities for FFT based convolutions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def full_self_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT length L (L >= 2·len(a))."""
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def max_self_convolution(a: np.ndarray, L: int) -> float:
    """Maximum entry of a*a, using L‑point FFT."""
    return float(full_self_convolution(a, L).max())


def grad_softmax(a: np.ndarray,
                 Fa_rev: np.ndarray,
                 conv: np.ndarray,
                 lam: float,
                 L: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate
        G(a) = (1/lam) * log Σ_k exp(lam * conv_k)
    where conv = a*a (full linear conv).
    Returns ∂G/∂a as a vector of length len(a).
    """
    M = conv.max()
    # soft‑max weights (shifted by M for numerical stability)
    w = np.exp(lam * (conv - M))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    # correlation of w with a (linear, not circular)
    Fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(Fw * Fa_rev, n=L)
    n = a.shape[0]
    # slice corresponds to indices n-1 … n-1+n-1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad.real


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector, i.e. sum=1)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def harmonic_left_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n) + 1
    a = i.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def triangular_distribution(n: int) -> np.ndarray:
    i = np.arange(n)
    a = np.minimum(i + 1, n - i) + 1e-14
    a = a.astype(np.float64)
    a /= a.sum()
    return a


def cosine_distribution(n: int, amp: float = 0.9) -> np.ndarray:
    """a_i ∝ 1 + amp·cos(2πi/(n‑1)), 0 ≤ amp ≤ 1."""
    i = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) discovered before the budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over several families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # lengths to explore (modest – sweep must finish quickly)
    lengths = [2000, 4000, 6000, 8000, 10000, 12000, 16000, 20000]

    # parameter grids
    p_grid = np.arange(0.1, 2.01, 0.1)            # for U‑shapes & harmonic
    beta_grid = np.arange(0.2, 2.01, 0.2)        # symmetric beta
    gamma_grid = np.arange(0.001, 0.021, 0.001) # exponential decay
    amp_grid = np.arange(0.2, 1.01, 0.2)        # cosine amplitude

    print("=== Deterministic sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # pre‑allocate FFT length for quick scoring
        L = next_pow2(2 * n)

        # ----- U‑shape families -----
        for p in p_grid:
            a = u_min_distribution(n, p)
            sc = 2.0 * n * max_self_convolution(a, L)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_min)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = 2.0 * n * max_self_convolution(a, L)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_prod) n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # ----- Beta family -----
        for alpha in beta_grid:
            a = beta_distribution(n, alpha, alpha)  # symmetric
            sc = 2.0 * n * max_self_convolution(a, L)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (beta)   n={n:5d} α={alpha:4.2f}  score={sc:.6f}")

        # ----- Exponential decay -----
        for gamma in gamma_grid:
            a = exp_decay_distribution(n, gamma)
            sc = 2.0 * n * max_self_convolution(a, L)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (exp)    n={n:5d} γ={gamma:5.3f}  score={sc:.6f}")

        # ----- Harmonic left -----
        for p in p_grid:
            a = harmonic_left_distribution(n, p)
            sc = 2.0 * n * max_self_convolution(a, L)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (harm)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # ----- Triangular -----
        a = triangular_distribution(n)
        sc = 2.0 * n * max_self_convolution(a, L)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f" new best (tri)    n={n:5d}                 score={sc:.6f}")

        # ----- Cosine family -----
        for amp in amp_grid:
            a = cosine_distribution(n, amp)
            sc = 2.0 * n * max_self_convolution(a, L)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (cos)    n={n:5d} amp={amp:3.1f}   score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Include an external construction if present
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            if external.sum() > 0.0:
                external /= external.sum()
                L_ext = next_pow2(2 * external.shape[0])
                sc_ext = 2.0 * external.shape[0] * max_self_convolution(external, L_ext)
                if sc_ext < best_score:
                    best_score, best_seq = sc_ext, external.copy()
                    print(f" using external height_sequence_1 score={sc_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback random seed (very unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        L = next_pow2(2 * n)
        best_score = 2.0 * n * max_self_convolution(best_seq, L)
        print(f" fallback random seed n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement via smooth‑max gradient descent
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]
    L = next_pow2(3 * n)                 # enough for convolution with weight vector
    current_score = 2.0 * n * max_self_convolution(current, L)

    # hyper‑parameters
    temperature = 1.0
    temp_decay = 0.9995
    lam = 0.5                # start with a smooth surrogate
    lam_factor = 1.005
    step = 0.4               # initial additive step size
    step_decay = 0.9998

    iteration = 0
    last_improve = 0
    stagnation_limit = 200_000   # re‑heat after this many iterations without improvement

    print("\n=== Starting stochastic refinement ===")
    while time.time() < deadline:
        iteration += 1

        # ---- compute current convolution and gradient ----
        Fa = np.fft.rfft(current, n=L)
        conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
        M = conv.max()
        # current_score is already 2*n*M, but recompute occasionally for safety
        current_score = 2.0 * n * M

        # gradient of the smooth‑max surrogate
        Fa_rev = np.fft.rfft(current[::-1], n=L)
        grad = grad_softmax(current, Fa_rev, conv, lam, L)

        # ---- propose a candidate ----
        cand = current - step * grad
        cand = simplex_projection(cand)

        # evaluate candidate (cheap, uses same L)
        Fa_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(Fa_c * Fa_c, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_c.max()

        # ---- simulated‑annealing acceptance ----
        accept = False
        if cand_score < current_score - 1e-12:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:8d}: new best score = {best_score:.6f}")

        # ---- schedule updates ----
        temperature *= temp_decay
        lam *= lam_factor
        step *= step_decay

        # ---- reheating on stagnation ----
        if iteration - last_improve > stagnation_limit:
            # small random perturbation and reset of schedules
            jitter = np.random.rand(n) * 1e-4
            current = simplex_projection(current + jitter)
            current_score = 2.0 * n * max_self_convolution(current, L)
            temperature = 1.0
            lam = 0.5
            step = 0.4
            last_improve = iteration
            print(f" * Re‑heat at iter {iteration}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # external evaluator will re‑apply clipping and normalisation,
    # so we simply return the raw vector.
    return best_seq.tolist()
