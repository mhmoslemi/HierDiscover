"""
Propose a low‑score step‑function.

Algorithm
---------
1. Deterministic sweep over several parametric families (U‑shape, beta,
   exponential, edge‑constant and offset‑U).  The best candidate found is
   kept as a seed.

2. If the global variable ``height_sequence_1`` exists, it is also
   considered.

3. Stochastic refinement:
   * Compute the self‑convolution with a fast FFT.
   * Build the sub‑gradient of the current maximal convolution entries.
   * Apply an Adam‑style projected update (additive on the probability
     simplex).  The update is accepted if it improves the score, otherwise
     it may be accepted with a simulated‑annealing probability.
   * When progress stalls the algorithm “re‑heats’’ by adding a tiny random
     bump and resetting the temperature.

The whole routine runs until the supplied time budget (≈1000 s) expires
and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length 2*len(a)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is non‑negative, sum=1)."""
    conv = full_convolution_fft(a)
    # Numerical noise may produce tiny negatives – clamp.
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a probability vector."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge: (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the two edges."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def edge_constant_distribution(n: int, edge_width: int, edge_ratio: float) -> np.ndarray:
    """Two‑level shape: heavy mass on the outer `edge_width` entries."""
    if edge_width * 2 >= n:
        edge_width = n // 2 - 1
    a = np.ones(n, dtype=np.float64)
    a[:edge_width] = edge_ratio
    a[-edge_width:] = edge_ratio
    a /= a.sum()
    return a


def u_min_offset_distribution(n: int, p: float, eps: float) -> np.ndarray:
    """U‑shape with a small uniform offset `eps` (to avoid a huge a0)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + eps
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) discovered before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over several families
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000, 12000, 16000, 20000, 30000, 40000]
    p_grid = np.arange(0.1, 3.01, 0.2)               # for U‑families
    beta_grid = np.arange(0.3, 3.01, 0.3)            # symmetric Beta
    gamma_grid = np.arange(0.001, 0.051, 0.005)      # exponential decay
    edge_width_grid = [int(frac * n) for frac in (0.01, 0.02, 0.04, 0.07, 0.1) for n in [1]]
    # we will compute actual width later per n
    edge_ratio_grid = [2, 5, 10, 20, 40, 80]        # how many times the interior weight
    eps_grid = [0.0, 5e-4, 1e-3, 5e-3, 1e-2]        # offset for U‑min

    print("=== Deterministic sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # ----- uniform baseline (for sanity) -----
        a = np.full(n, 1.0 / n, dtype=np.float64)
        sc = score_from_vector(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f" new best (uniform) n={n:5d}  score={sc:.6f}")

        # ----- U‑min family & offset version -----
        for p in p_grid:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

            for eps in eps_grid[1:]:          # skip eps=0 (already done)
                a_off = u_min_offset_distribution(n, p, eps)
                sc = score_from_vector(a_off)
                if sc < best_score:
                    best_score, best_seq = sc, a_off.copy()
                    print(f" new best (u_min+ε) n={n:5d} p={p:4.2f} eps={eps:.5f}  score={sc:.6f}")

        # ----- U‑prod family -----
        for p in p_grid:
            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # ----- Beta family (symmetric) -----
        for a_par in beta_grid:
            a = beta_distribution(n, a_par, a_par)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (beta)    n={n:5d} α={a_par:4.2f}  score={sc:.6f}")

        # ----- Exponential decay family -----
        for gamma in gamma_grid:
            a = exp_decay_distribution(n, gamma)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (exp)    n={n:5d} γ={gamma:5.3f}  score={sc:.6f}")

        # ----- Edge‑constant two‑level family -----
        for frac in (0.01, 0.02, 0.04, 0.07, 0.10):
            edge_width = max(1, int(round(frac * n)))
            for ratio in edge_ratio_grid:
                a = edge_constant_distribution(n, edge_width, ratio)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f" new best (edge)   n={n:5d} w={edge_width:4d} r={ratio:5.1f}  score={sc:.6f}")

        # occasional time check
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. Incorporate external construction if available
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            total = external.sum()
            if total > 0.0:
                external /= total
                sc_ext = score_from_vector(external)
                if sc_ext < best_score:
                    best_score, best_seq = sc_ext, external.copy()
                    print(f" using external height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback random seed (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement (Adam‑style projected descent + SA)
    # ------------------------------------------------------------------
    current = best_seq.copy()
    current_score = best_score
    n = len(current)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.25                     # base learning rate
    m = np.zeros_like(current)    # first moment
    v = np.zeros_like(current)    # second moment

    # Simulated‑annealing parameters
    temperature = 1.0
    cooling_rate = 0.9999
    rel_tol = 1e-4                # active‑set tolerance
    iteration = 0
    last_improve_iter = 0
    beta1_pow = beta1
    beta2_pow = beta2

    print("\n=== Starting stochastic refinement ===")
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 4a – self‑convolution and active set
        # --------------------------------------------------------------
        conv = full_convolution_fft(current)
        max_val = conv.max()
        # indices whose value is within a relative tolerance of the maximum
        active_idx = np.where(conv >= max_val * (1.0 - rel_tol))[0]

        # --------------------------------------------------------------
        # 4b – sub‑gradient (sum of gradients of all active entries)
        # --------------------------------------------------------------
        grad = np.zeros_like(current)
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_vals = k - i_vals
            grad[i_vals] += 2.0 * current[j_vals]

        # --------------------------------------------------------------
        # 4c – Adam update (additive) and projection onto simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand = current - step
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)

        # --------------------------------------------------------------
        # 4d – evaluate candidate
        # --------------------------------------------------------------
        cand_score = score_from_vector(cand)

        # --------------------------------------------------------------
        # 4e – simulated‑annealing acceptance
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve_iter = iteration
                print(f"Iter {iteration:9d}: new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 4f – update annealing / Adam powers
        # --------------------------------------------------------------
        temperature *= cooling_rate
        beta1_pow *= beta1
        beta2_pow *= beta2

        # --------------------------------------------------------------
        # 4g – stagnation handling (re‑heat)
        # --------------------------------------------------------------
        if iteration - last_improve_iter > 500000:
            # add a tiny random bump and reset temperature
            bump = np.random.rand(n) * 1e-4
            current = simplex_projection(current + bump)
            current_score = score_from_vector(current)
            last_improve_iter = iteration
            temperature = 1.0
            print(f" * Re‑heat at iter {iteration}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
