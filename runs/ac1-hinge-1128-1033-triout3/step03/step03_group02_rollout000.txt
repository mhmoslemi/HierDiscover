"""
Optimisation of a non‑negative step‑function via multiple Adam‑styled
restarts and a soft‑max (log‑sum‑exp) surrogate for the L∞ objective.

* The objective (for Σ a_i = 1) is 2·n·max_k (a * a)[k].
* We approximate max by a temperature‑controlled log‑sum‑exp:
      Lτ(a) = τ·log Σ_k exp(( (a*a)[k] – M ) / τ)  where M = max_k (a*a)[k].
  As τ → 0, Lτ → M.  The soft‑max weights w_k = exp(( (a*a)[k] – M)/τ)/Σ exp(... )
  give a smooth sub‑gradient.
* Adam is used on the simplex (additive update + Euclidean projection).
* After each restart a short local random‑mass‑transfer hill‑climb is performed.
* Several restarts with different lengths and random Dirichlet initialisations
  increase the chance to escape local minima.
* The best sequence seen during the whole budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σ x = 1}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0:
        return np.full_like(v, 1.0 / n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y using FFT (real inputs)."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate Lτ(a)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented by a single convolution: reverse w and convolve with a.
    """
    n = a.shape[0]
    u = w[::-1]                                      # reverse weights
    conv = _full_convolution_fft(u, a)               # length 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)                       # indices yielding g_i
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Small local improvement (random mass transfer hill‑climb)
# ----------------------------------------------------------------------
def _local_improvement(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       max_trials: int = 300) -> np.ndarray:
    """
    Try a few random transfers of probability mass.
    Keep the first improvement that yields a lower objective.
    """
    n = a.shape[0]
    cur_val = best_val
    for _ in range(max_trials):
        if time.time() >= deadline - 0.2:
            break
        src = np.random.randint(n)
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1
        # size of transferred mass – up to 20 % of src
        delta = np.random.rand() * a[src] * 0.2
        if delta <= 0:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_onto_simplex(cand)
        conv = _full_convolution_fft(cand, cand)
        val = 2.0 * n * conv.max()
        if val < cur_val:
            return cand, val
    return a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------

    Adam‑based soft‑max descent with multiple restarts and a final
    random hill‑climb.  Returns the best non‑negative step‑function
    (list of floats) found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the outer harness
    deadline = start_time + budget_s - 2.0

    best_val = np.inf
    best_seq = None

    # ------------------------------------------------------------------
    # Settings for the restart loop
    # ------------------------------------------------------------------
    # A handful of lengths – larger n gives a smaller theoretical bound
    # but costs more per iteration.  The list is ordered from cheap to
    # expensive; we cycle through it.
    length_pool = [1200, 1600, 2000, 2400, 2800, 3200]
    restart_counter = 0

    # ------------------------------------------------------------------
    # Main restart loop – continues until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Choose a length (cycle through the pool)
        n = length_pool[restart_counter % len(length_pool)]
        restart_counter += 1

        # Initialise a:
        #   * If we already have a good sequence of exactly this length,
        #     warm‑start from it.
        #   * Otherwise draw a sparse Dirichlet (γ‑shape < 1 encourages
        #     many near‑zero entries, which helps flatten the convolution).
        if best_seq is not None and len(best_seq) == n:
            a = np.array(best_seq, dtype=float)
        else:
            # Gamma with shape 0.3 → Dirichlet(alpha=0.3) – sparse start
            gamma = np.random.gamma(shape=0.3, scale=1.0, size=n)

        # Adam moments

        beta2 = 0.999             # second‑moment decay

            # ---------- Adam update ----------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

                a_candidate[src] -= delta
                a_candidate[dst] += delta

            # ---------- projection onto simplex ----------

    # ------------------------------------------------------------------
