def proj_simplex(v):
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - 1
    rho = np.nonzero(v_sorted - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:

"""
Optimise a non‑negative step‑function (simplex‑constrained vector) so as to
minimise ``evaluate_sequence``.  The implementation uses a smooth
log‑sum‑exp surrogate for the max‑convolution, Adam updates on the
log‑soft‑max parameters and a linear schedule that drives the smoothing
parameter ε down to 1e‑12 – i.e. the gradient becomes the true sub‑gradient
of the max.  Random restarts and occasional Gaussian perturbations avoid
getting stuck in local minima.  When the time budget is almost exhausted a
tiny coordinate‑wise hill‑climber performs a final polish using the exact
objective.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full convolution x*x (length 2·len(x)‑1) using a real FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())           # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)                      # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

    """Numerically stable soft‑max returning a probability vector (sum=1)."""
    w_shift = w - np.max(w)                # for stability
    a = np.exp(w_shift)
    a /= a.sum()
    return a

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin

    # problem size – 4000 gives a good trade‑off between flexibility and speed
    n = 4000

    # Adam hyper‑parameters
    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.30
    lr_end   = 0.0005

    # smoothing (ε) schedule – finish at a very small value so that we
    # effectively optimise the true max after the first half of the run.
    eps_start = 5e-3
    eps_end   = 1e-12

    # restart criterion
    stagnation_limit = 150_000          # steps without improvement → restart

    # bookkeeping
    best_a = None
    best_val = float('inf')
    total_iters = 0

    # ------------------------------------------------------------------
    # Optional: start from a previously discovered construction if present
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if init.shape[0] != n:
            # linearly interpolate to the desired length
            old_x = np.linspace(0, 1, init.shape[0])
            new_x = np.linspace(0, 1, n)
            init = np.interp(new_x, old_x, init)
        init = np.maximum(init, 0.0)
        init = init / init.sum()
    except Exception:
        init = None

    # ------------------------------------------------------------------
    # Exact objective (sum of a is held at 1 → 2·n·max(conv) is the score)
    # ------------------------------------------------------------------
    def exact_obj(conv_arr):
        """Exact objective for a vector whose sum is (approximately) 1."""
        return 2.0 * n * conv_arr.max()

    # ------------------------------------------------------------------
    # Outer optimisation loop – keep restarting until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Initialise a new start point
        # --------------------------------------------------------------
        if best_a is not None and np.random.rand() < 0.5:
            # Perturb the current best and use it as a fresh start
            a0 = best_a + np.random.normal(scale=0.01, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            if init is not None and np.random.rand() < 0.3:
                a0 = init.copy()
            else:
                a0 = np.random.rand(n)
                a0 /= a0.sum()

        w = np.log(a0 + 1e-12)               # logits for soft‑max
        m = np.zeros_like(w)                 # Adam first moment
        v = np.zeros_like(w)                 # Adam second moment
        t_iter = 0
        no_improve_cnt = 0

        # --------------------------------------------------------------
        # Single Adam run (stops on stagnation or time‑out)
        # --------------------------------------------------------------
        while True:
            now = time.time()
            if now >= deadline:
                break

            # ----- soft‑max to obtain a feasible point -----
            a = _softmax_from_logits(w)                 # Σa = 1, a_i ≥ 0

            # ----- exact objective for bookkeeping -----
            conv_aa = _full_convolution(a)
            cur_val = exact_obj(conv_aa)

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                no_improve_cnt = 0
                print(f"[{total_iters:8d}] new best = {best_val:.6f}")
            else:
                no_improve_cnt += 1

            # ----- schedule for ε and learning‑rate -----
            progress = (now - start_time) / (deadline - start_time)
            eps = eps_start * (1.0 - progress) + eps_end * progress
            lr  = lr_start * (1.0 - progress) + lr_end  * progress

            # ----- smoothed max (log‑sum‑exp) and its gradient w.r.t. a -----
            max_conv = conv_aa.max()
            shifted = (conv_aa - max_conv) / eps          # max → 0, others ≤0
            weights = np.exp(shifted)                     # safe: all ≤1
            p = weights / weights.sum()                   # shape (2n‑1,)

            # gradient dL/da = 4·n·(a * rev(p)) restricted to centre n entries
            p_rev = p[::-1]                               # reverse
            conv_ab = _conv_fft(a, p_rev)                 # length 3n‑2
            segment = conv_ab[n - 1: 2 * n - 1]            # middle n values
            grad_a = 4.0 * n * segment[::-1]              # shape (n,)

            # ----- chain rule through the soft‑max (gradient wrt w) -----
            dot = np.dot(grad_a, a)                       # ⟨grad_a , a⟩
            grad_w = a * (grad_a - dot)                   # shape (n,)

            # ----- Adam update on the logits -----
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_w
            v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            total_iters += 1

            # ----- occasional random perturbation to shake out of plateaus -----
            if np.random.rand() < 0.01:
                w += np.random.normal(scale=0.02, size=n)

            # ----- stagnation => break to a fresh restart -----
            if no_improve_cnt > stagnation_limit:
                print(f"[{total_iters:8d}] stagnation, restarting...")
                break

        # end of a single Adam run; loop back for another start if time permits

    # ------------------------------------------------------------------
    # Final fine‑tuning – tiny coordinate moves using the exact objective
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_a is not None and remaining > 5.0:
        a = best_a.copy()
        conv = _full_convolution(a)
        cur_best = exact_obj(conv)
        # relative amount of mass shifted in each trial
        move_factor = 0.005

        while time.time() < deadline:
            # pick a donor with non‑negligible mass
            i = np.random.randint(n)
            if a[i] < 1e-8:
                continue
            delta = min(move_factor * a[i], 1e-4)
            if delta <= 0:
                continue
            j = np.random.randint(n)
            if i == j:
                continue

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta   # sum stays equal to 1

            conv_new = _full_convolution(a_new)
            val_new = exact_obj(conv_new)

            # accept only strict improvements (tiny tolerance to avoid noise)
            if val_new + 1e-12 < cur_best:
                a = a_new
                conv = conv_new
                cur_best = val_new
                if cur_best < best_val:
                    best_val = cur_best
                    best_a = a.copy()
                    print(f"[refine] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Report the final score using the provided safe evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
