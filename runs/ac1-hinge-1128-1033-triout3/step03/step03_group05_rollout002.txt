"""
Propose a high‑quality step‑function (probability vector) that minimises
evaluate_sequence().  The algorithm is a multi‑stage simplex‑constrained
optimiser:

1. Build several promising initial vectors (arcsine‑shaped, a few
   gamma‑variants, random Dirichlet draws and, if present,
   the historic construction ``height_sequence_1``).
2. Pick the initial vector with the lowest raw score.
3. Sub‑gradient descent on the *actual* maximal convolution entry
   (few thousand cheap iterations).
4. Adam optimisation on a smooth log‑sum‑exp approximation of the max.
   The smoothing parameter α is increased gradually up to 200 and the
   Adam loop runs for many tens of thousands of iterations.
5. A short exponentiated‑gradient (multiplicative) phase that further
   reduces the max entry.
6. Final tiny random “swap” hill‑climber.
All updates keep the vector on the probability simplex by simple
clipping + renormalisation (much faster than a full Euclidean projection).
The routine respects a 1000 s wall‑time budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (used only as a fallback)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Normalise a vector onto the simplex by clipping negatives and renorm.
# ----------------------------------------------------------------------
def _normalize_simplex(v: np.ndarray) -> np.ndarray:
    """Clip negative entries to zero and renormalise to sum 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# FFT‑based self‑convolution utilities
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Linear self‑convolution of p using a pre‑computed FFT size."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _score(p: np.ndarray, m_fft: int) -> float:
    """Exact evaluation 2·n·max(conv_self) for a (already normalised) vector."""
    conv = _conv_self(p, m_fft)
    return float(2.0 * p.size * np.max(conv))


# ----------------------------------------------------------------------
# Gradient of the smoothed max (log‑sum‑exp) w.r.t. p.
# ----------------------------------------------------------------------
def _smooth_max_gradient(p: np.ndarray, m_fft: int, alpha: float) -> np.ndarray:
    """
    Return the gradient of  (1/α)·log Σ exp(α·(p*p))  w.r.t. p.
    Uses the same FFT trick as the original code.
    """
    n = p.size
    p_fft = np.fft.rfft(p, n=m_fft)

    # self‑convolution (full length 2·n‑1)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

    # soft‑max weights
    z = alpha * conv
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    w = exp_z / exp_z.sum()

    # correlation p * rev(w)  (full length 3·n‑2)
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]

    # gradient indices mapping (see original implementation)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * corr_full[idx]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the actual max entry (used in the first stage)
# ----------------------------------------------------------------------
def _max_entry_subgradient(p: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    ``conv`` must be the full self‑convolution (length 2·n‑1).
    """
    n = p.size
    max_idx = int(conv.argmax())
    g = np.zeros_like(p)

    if max_idx < n:
        # overlapping part: indices 0 … max_idx
        g[: max_idx + 1] = 2.0 * p[max_idx :: -1]
    else:
        # tail part
        start = max_idx - n + 1
        g[start:] = 2.0 * p[n - 1 : start - 1 : -1]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)  # type: ignore[name-defined]
        historic = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        historic = np.clip(historic, 0.0, 1000.0)
        hist_sum = historic.sum()
        if hist_sum > 0.0:
            historic = historic / hist_sum
        else:
            historic = None
    else:
        # fallback length – a few thousand entries give a nice trade‑off
        n = 2000
        historic = None

    # FFT size: next power‑of‑two that can contain a convolution of length 2·n‑1
    m_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Build a pool of promising start vectors
    # --------------------------------------------------------------
    i = np.arange(n, dtype=float)

    init_vectors = []

    # 1) Classical arcsine shape (γ = 0.5)
    inv = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + 1e-12)
    init_vectors.append(inv / inv.sum())

    # 2) A few gamma‑variants around 0.5
    for gamma in (0.35, 0.40, 0.45, 0.55, 0.60, 0.65):
        invg = 1.0 / np.power((i + 0.5) * (n - 0.5 - i) + 1e-12, gamma)
        init_vectors.append(invg / invg.sum())

    # 3) Random Dirichlet draws (exponential weights give Dirichlet(1))
    for _ in range(4):
        rnd = rng.random(n)
        init_vectors.append(rnd / rnd.sum())

    # 4) Historic sequence, if available
    if historic is not None:
        init_vectors.append(historic)

    # --------------------------------------------------------------
    # Pick the best raw start (cheapest way to avoid wasting time)
    # --------------------------------------------------------------
    best_init = None
    best_init_score = float("inf")
    for vec in init_vectors:
        sc = _score(vec, m_fft)
        if sc < best_init_score:
            best_init_score = sc
            best_init = vec.copy()
    if best_init is None:                     # should never happen
        best_init = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------------
    TARGET = 1.503            # stop early if we get below this value
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Phase A – short sub‑gradient descent on the real max entry
    # ------------------------------------------------------------------
    p = best_init.copy()
    step = 0.25
    decay = 0.9996
    min_step = 1e-6
    best_score = _score(p, m_fft)
    best_p = p.copy()

    for it in range(12_000):
        if time.time() >= deadline:
            break
        conv = _conv_self(p, m_fft)
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_p = p.copy()
            print(f"[Subgrad] iter {it:5d} → new best {best_score:.6f}")
            if best_score <= TARGET:
                break

        g = _max_entry_subgradient(p, conv)
        p = p - step * g
        p = _normalize_simplex(p)
        step = max(step * decay, min_step)

    # ------------------------------------------------------------------
    # Phase B – Adam on a log‑sum‑exp smooth approximation of the max
    # ------------------------------------------------------------------
    m_vec = np.zeros_like(p)
    v_vec = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.07

    alpha = 1.0           # smoothing factor
    alpha_mul = 1.001
    alpha_max = 200.0

    max_adam_iters = 120_000
    for it in range(max_adam_iters):
        if time.time() >= deadline:
            break

        # gradient of the smoothed max
        grad = _smooth_max_gradient(p, m_fft, alpha)

        # Adam update (additive)
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad ** 2)

        m_hat = m_vec / (1.0 - beta1 ** (it + 1))
        v_hat = v_vec / (1.0 - beta2 ** (it + 1))

        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        p = p - step_vec
        p = _normalize_simplex(p)

        # occasional real‑score check
        if (it % 1500) == 0 or it == max_adam_iters - 1:
            cur_score = _score(p, m_fft)
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_p = p.copy()
                print(f"[Adam]   iter {it:6d} → new best {best_score:.6f}")
                if best_score <= TARGET:
                    break

        # tighten the smooth approximation
        alpha = min(alpha * alpha_mul, alpha_max)

    # ------------------------------------------------------------------
    # Phase C – exponentiated‑gradient (multiplicative) refinement
    # ------------------------------------------------------------------
    p = best_p.copy()
    eta = 0.7                     # initial multiplicative step size
    for it in range(60_000):
        if time.time() >= deadline:
            break

        conv = _conv_self(p, m_fft)
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_p = p.copy()
            print(f"[ExpGrad] iter {it:5d} → new best {best_score:.6f}")
            if best_score <= TARGET:
                break

        # sub‑gradient of the max entry (same as Phase A)
        g = _max_entry_subgradient(p, conv)

        # multiplicative update: p_i ← p_i * exp(−η·g_i)
        p = p * np.exp(-eta * g)
        p = _normalize_simplex(p)

        # slowly decay the step size
        eta *= 0.99997

    # ------------------------------------------------------------------
    # Phase D – tiny random swap hill‑climber (local fine‑tuning)
    # ------------------------------------------------------------------
    p = best_p.copy()
    cur_score = _score(p, m_fft)
    for _ in range(12_000):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_move = 1e-4 * rng.random()
        if p[i] < eps_move:
            continue
        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move
        new_score = _score(new_p, m_fft)
        if new_score < cur_score:
            p = new_p
            cur_score = new_score
            best_p = p.copy()
            best_score = cur_score
            print(f"[Swap]   improvement → {best_score:.6f}")
            if best_score <= TARGET:
                break

    print(f"Final best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to the limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    return np.clip(best_p, 0.0, 1000.0).tolist()
