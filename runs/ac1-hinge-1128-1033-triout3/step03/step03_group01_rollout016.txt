def correlated_fft(w, a):
    n = len(a)
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    # conv length = len(w) + len(a_rev) - 1 = (2n-1)+n-1 = 3n-2
    # extract valid part: indices n-1 to n-1 + n - 1 = 2n-2 inclusive.
    start = n - 1
    end = start + n
    return conv[start:end]
```

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises

    2 * n * max_k (a * a)[k]      ( * – linear convolution )

The algorithm combines two phases:

1. **Deterministic sweep** – evaluates several parametric families
   (U‑shaped power‑law, exponential, symmetric Beta) for a few
   lengths.  The best vector from this sweep is taken as a seed.

2. **Smoothed‑max Adam descent** – the objective
   ``L_T(a)=T·log Σ_k exp( (a*a)[k]/T )`` approximates the true
   max‑convolution.  Its gradient is
   ``∇_a L_T = 2·(w ⋆ a)`` where the weight vector
   ``w_k ∝ exp((a*a)[k]/T)``.
   By slowly decreasing the temperature ``T`` (i.e. increasing the
   scaling ``β=1/T``) we obtain a smooth landscape at the beginning
   and converge to a true max‑convolution minimiser.
   The search is performed on the log‑softmax parameters and uses the
   Adam optimiser.  The best sequence found during the run is returned.

All heavy work is wrapped in a single time‑budget check (default 1000 s)
so the function always finishes in time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors a and b using real FFT.
    Result length = len(a)+len(b)-1.
    """
    size = next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid cross‑correlation of w (length 2n‑1) with a (length n):
    result[i] = Σ_j w[i+j] * a[j]    for i = 0 … n‑1.
    Implemented via FFT as convolution of w with a reversed.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = next_pow2(len(w) + len(a_rev) - 1)
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(fw * fa_rev, n=size)
    start = n - 1
    end = start + n
    return conv[start:end]


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a vector that already satisfies
    Σ a_i = 1 (the denominator of ``evaluate_sequence`` becomes 1).
    """
    n = a.shape[0]
    conv = fft_convolve(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families for the deterministic sweep
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] = (min(i, n‑1‑i)+1)^(-p)  (U‑shape based on distance to edge)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] = ((i+1)*(n‑i))^{-p}  (U‑shape based on product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] = exp(-lam*i) + exp(-lam*(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric (or asymmetric) Beta‑like shape on a discrete grid.
    x in (0,1) and a[i] ∝ x^{alpha-1} (1-x)^{beta-1}.
    """
    x = (np.arange(1, n + 1) - 0.5) / n  # centre of each bin
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max gradient (used in the Adam refinement)
# ----------------------------------------------------------------------


def compute_weighted_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Given a (probability vector) and its convolution conv = a*a,
    compute the gradient of the smoothed objective with parameter beta:
        w_k ∝ exp(beta * (conv_k - max(conv)))
        grad_a = 2 * (w ⋆ a)
    """
    max_c = conv.max()
    # weights (avoid overflow by shifting with max_c)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient w.r.t. a
    grad_a = 2.0 * correlation_fft(w, a)
    return grad_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for the best non‑negative step‑function within the given
    time budget.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over several parametric families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000]

    # power‑law exponents for the U‑shaped families
    exponents = np.arange(0.1, 4.01, 0.2)           # fewer steps than before

    # exponential decay parameters
    lam_vals = np.logspace(-4, -0.3, num=20)       # ~1e‑4 … 0.5

    # Beta‑distribution shape parameters (edge‑heavy when <1)
    alphas = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]
    betas = alphas                               # symmetric sweep

    print("=== Deterministic sweep start ===")
    for n in lengths:
        if time.time() > deadline:
            break
        # ----- U‑shapes -----
        for p in exponents:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (U‑min)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (U‑prod) n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # ----- exponential -----
        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (exp)     n={n:5d} λ={lam:.5f}  score={sc:.6f}")

        # ----- Beta -----
        for alpha in alphas:
            for beta in betas:
                a = beta_distribution(n, alpha, beta)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f" new best (beta)   n={n:5d} α={alpha:3.1f} β={beta:3.1f}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Use an externally supplied construction if present
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0.0:
                external = external / s
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score = sc
                    best_seq = external.copy()
                    print(f" using external height_sequence_1  score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smoothed‑max Adam refinement (log‑softmax parametrisation)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # initialise logits y such that softmax(y) = a
    eps = 1e-12
    y = np.log(a + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr = 0.15
    lr_decay = 0.99995

    # smoothed‑max temperature schedule (β = 1/T)
    beta = 0.0
    beta_max = 200_000.0
    # we aim at ~200 k iterations → linear increase
    beta_inc = beta_max / 200_000.0

    iteration = 0
    last_improve = 0

    print("\n=== Smoothed‑max Adam refinement start ===")
    while time.time() < deadline:
        iteration += 1

        # ---- a from logits (softmax) ----
        y_shift = y - np.max(y)                # avoid overflow
        a = np.exp(y_shift)
        a /= a.sum()

        # ---- convolution of a with itself ----
        conv = fft_convolve(a, a)
        max_conv = conv.max()
        curr_score = float(2.0 * n * max_conv)

        # ---- keep global best ----
        if curr_score < best_score:
            best_score = curr_score
            best_seq = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:7d}: new global best = {best_score:.6f}")

        # ---- update smoothing parameter β ----
        beta = beta + beta_inc
        if beta > beta_max:
            beta = beta_max

        # ---- weighted gradient of the smoothed max ----
        grad_a = compute_weighted_gradient(a, conv, beta)

        # ---- gradient w.r.t. logits (softmax chain rule) ----
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        y -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ---- learning‑rate decay ----
        lr *= lr_decay

        # ---- occasional re‑heat if stuck for a long time ----
        if iteration - last_improve > 120_000:
            # inject a tiny random bump to logits and reset Adam momentum
            y += np.random.randn(*y.shape) * 1e-4
            m.fill(0.0)
            v.fill(0.0)
            lr = 0.15
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
