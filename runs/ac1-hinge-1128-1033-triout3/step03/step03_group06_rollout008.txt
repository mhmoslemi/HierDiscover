It searches a few simple parametric families (power‑law, exponential and a
product‑law) over several lengths, keeps the best normalised vector it has
found and then performs a short mirror‑descent refinement on the smooth‑max
upper bound of the convolution.  All work stays inside the 1000 s budget,
uses only NumPy and the already‑available `evaluate_sequence`, and finally
returns the best list of heights.

```python
"""
Propose a step‑function (non‑negative vector) that minimises the metric
used by `evaluate_sequence`.

Algorithm
---------

1.  Initialise from the previous best construction if it exists,
    otherwise from a uniform distribution.

2.  Global parametric search:
    * For several lengths n (4000 … 30000) generate three families of
      symmetric vectors:
        – power‑law   a_i ∝ (dist_to_edge+1)^(-p)
        – exponential a_i ∝ r^(dist_to_edge)
        – product‑law a_i ∝ ((dist+1)*(n‑dist))^(-p)
      Scan a modest grid of the parameters (p, r) and keep the best
      candidate (approximate score = 2·n·max(conv) because the vectors are
      normalised to sum‑one).

3.  Local refinement:
    Starting from the best candidate run a mirror‑descent optimiser on a
    soft‑max upper bound of the self‑convolution (the same machinery that
    was used in the previous solution) for a few thousand iterations or
    until the time budget is exhausted.

4.  Evaluate the final vector with the official `evaluate_sequence`
    and return it as a plain Python list.

All helper functions are defined at module level (no closures) and only
NumPy and the standard library are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a (real FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())           # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution (length 2·n‑1) using a supplied FFT length."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max of the convolution and its gradient (mirror‑descent helper)
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities w_k = exp(α·conv_k) / Σ exp(α·conv_k)."""
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of a*a w.r.t. a.
    a : current probability vector (length n)
    w : soft‑max weights (length 2·n‑1, sum to 1)
    size : FFT length, must be ≥ 3·n‑2
    Returns a vector of length n.
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of w
    A = np.fft.rfft(a, n=size)

    # FFT of w and its reversal
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # a * w and a * w_rev (both of length n+L‑1 = 3·n‑2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    # Gradient assembling – see comment in the original code
    idx = L - 1 - np.arange(n)                     # 2·n‑2, …, n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad

# Small utility functions
# ----------------------------------------------------------------------


def normalise_positive(v: np.ndarray) -> np.ndarray:
    """Clip to ≥0 and renormalise to sum‑one (avoids division by zero)."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        # fall back to uniform
        return np.full_like(v, 1.0 / v.size)
    return v / s

            if it_no_improve >= 2000:
                break

        return best_local, best_local_score

    # Run the refinement if we still have some time left
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Starting local mirror‑descent refinement ---")
        refined_seq, refined_score = refine_with_mirror_descent(best_seq, deadline)
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"Refinement succeeded – improved approx. score to {refined_score:.6f}")
        else:
            print("Refinement finished – no further improvement found.")
    else:
        print("No time left for refinement.")

    # ------------------------------------------------------------------
    # 4.  Final official evaluation & return
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:      # the function may not be present in a test run
        print(f"Could not call evaluate_sequence (error: {exc})")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {best_score:.6f}")
    print(f"Best exact score       : {exact_score:.6f}")
    print(f"Best length (n)        : {best_seq.shape[0]}")
    return best_seq.tolist()
```
