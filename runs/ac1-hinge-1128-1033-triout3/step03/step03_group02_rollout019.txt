def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    best_seq = None
    best_val = float('inf')
    # Predefine base length maybe 1500
    while time.time() < deadline:
        # randomly choose n
        n = np.random.randint(500, 2500) # ensure not too large >5000 maybe slow
        # maybe also choose n from set [700, 1100, 1500, 2000] but random is fine.

        # initialization: sometimes use previous best sequence truncated/padded, sometimes random.
        init_type = np.random.rand()
        if best_seq is not None and init_type < 0.3:
            # Use best seq as base but maybe rescaled to chosen n (interpolation)
            base = np.array(best_seq)
            # Interpolate to length n
            a = np.interp(np.linspace(0, len(base)-1, n), np.arange(len(base)), base)
        elif init_type < 0.6:
            # sine shape similar to earlier
            x = np.arange(n, dtype=float)
            a = np.sin(np.pi * x / (n - 1))  # shape maybe peaked middle
            a = np.maximum(a, 0)
        else:
            # random Dirichlet
            alpha = np.random.rand() * 2 + 0.1  # random concentration
            a = np.random.gamma(alpha, 1.0, size=n)
        # Normalize to sum 1
        a = a / a.sum()
        # Adam parameters
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        # schedule of learning rate
        # set initial lr inversely proportional to sqrt(n) maybe
        lr_start = 0.3
        lr_end = 0.001
        steps = np.random.randint(500, 2000) # number of iterations for this run
        for t in range(1, steps+1):
            # compute conv
            conv = _full_convolution_fft(a, a)
            M = conv.max()
            # optional compute obj for early exit
            cur_val = 2.0 * n * M  # sum a =1
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # Optionally print
                print(f"[{time.time()-start:.1f}s] new best overall {best_val:.6f} n={n}")

            # compute softmax weight; use temperature schedule decreasing
            prog = t / steps
            tau = max(conv.max() * (0.5 * (1-prog) + 0.01 * prog), 1e-12)  # or some schedule
            # using exponential weighting
            exp_args = (conv - M) / tau
            # clip to avoid overflow
            # maybe use stable exp
            max_exp = np.max(exp_args)
            exp_vals = np.exp(exp_args - max_exp)
            w = exp_vals / np.sum(exp_vals)

            # gradient
            grad = _gradient_from_weights(a, w)
            # Adam update
            lr = lr_start * (1-prog) + lr_end * prog
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # jitter: random mass transfer
            if np.random.rand() < 0.01:
                i = np.random.randint(n)
                j = np.random.randint(n-1)
                if j >= i:
                    j += 1
                delta = np.random.rand() * a[i] * 0.5
                a[i] -= delta
                a[j] += delta

            # projection
            a = _project_onto_simplex(a)

            if time.time() >= deadline:
                break

    # final evaluation using evaluate_sequence on best_seq
    if best_seq is None:
        # fallback default: uniform
        n = 1000
        a = np.full(n, 1.0 / n)
        best_seq = a

    # Ensure list of floats
    return best_seq.tolist()

    beta2 = 0.999

        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine {t}] new best {best_val:.6f}")

        # compute softmax weight with temperature schedule
        tau = max(conv.max() * 0.1, 1e-8)
        w = np.exp((conv - M) / tau)
        w /= w.sum()
        grad = _gradient_from_weights(a, w)
        lr = 0.1 / (np.sqrt(t) + 1)   # decaying
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # jitter random mass transfer occasionally
        if np.random.rand() < 0.01:
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i: j += 1
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta
        a = _project_onto_simplex(a)

        if time.time() >= deadline:
            break

    return best_seq.tolist()

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    best_val = float('inf')
    best_seq = None

    # 1. random search over parametric families (first part)
    # allocate e.g., 15% of time for random search
    random_search_end = start + (budget_s - 2.0) * 0.15

    while time.time() < random_search_end:
        # pick n from a range
        n = np.random.randint(800, 2501)   # inclusive
        # pick shape family
        shape_type = np.random.choice(['beta', 'exp', 'lin', 'rand'])
        if shape_type == 'beta':
            alpha = 10**np.random.uniform(-2, 1)   # 0.01 to 10
            beta = 10**np.random.uniform(-2, 1)
            a = _generate_beta_shape(n, alpha, beta)
        elif shape_type == 'exp':
            lam = np.random.uniform(0.9, 0.9999)   # close to 1
            a = _generate_exponential_shape(n, lam)
        elif shape_type == 'lin':
            # decreasing linear shape
            a = np.linspace(1.0, 0.0, n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:  # random Dirichlet
            conc = np.random.uniform(0.1, 5.0)
            a = np.random.gamma(conc, 1.0, size=n)
            a = a / a.sum()
        # compute objective
        obj = _compute_objective(a)
        if obj < best_val:
            best_val = obj
            best_seq = a.copy()
            print(f"[{time.time() - start:.1f}s] new best from random search = {best_val:.6f} n={n}")
    # 2. refine best sequence with Adam optimizer
    if best_seq is None:
        # fallback: uniform
        n = 1500
        best_seq = np.full(n, 1.0/n)
        best_val = _compute_objective(best_seq)
    a = best_seq.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iteration = 0
    # hyperparameters schedule
    eta_start = 0.2
    eta_end = 1e-4
    tau_start = None  # will be set first iteration
    tau_end_factor = 1e-6  # relative to initial max

    while True:
        now = time.time()
        if now >= deadline:
            break
        # convolution
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_obj = float(2.0 * n * M / (a.sum() ** 2))
        if cur_obj < best_val:
            best_val = cur_obj
            best_seq = a.copy()
            print(f"[{now - start:.1f}s] new best after refinement = {best_val:.6f}")

        progress = (now - start) / (deadline - start)
        # temperature schedule for softmax
        if tau_start is None:
            tau_start = M * 0.2
            tau_end = max(tau_start * 1e-4, 1e-12)
        tau = tau_start * (1.0 - progress) + tau_end * progress
        # compute softmax weights
        if tau > 0:
            exp_args = (conv - M) / tau
            # shift for numerical stability
            max_arg = np.max(exp_args)
            exp_vals = np.exp(exp_args - max_arg)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # gradient
        grad = _gradient_from_weights(a, w)

        # learning rate schedule (linear decay)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random jump
        if np.random.rand() < 0.02:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:

            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta

        # projection onto simplex
        a = _project_onto_simplex(a)
        iteration += 1

    # final check
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports {final_val:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence error:", exc)

    return best_seq.tolist()

#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Propose a candidate step‑function sequence that minimises the provided
evaluation function.  The algorithm works in two phases:

1.  **Random parametric search** – generate many candidate shapes using
    a few simple families (Beta‑pdf, power‑law, exponential, linear and
    random Dirichlet).  Each candidate is normalised to sum to 1 and
    evaluated with a fast FFT‑based convolution.  The best candidate
    found is kept.

2.  **Refinement** – starting from the best candidate, run a projected
    Adam optimiser on the simplex.  The objective is
        F(a) = 2·n·max_k (a*a)[k] / (sum a)^2 .
    A soft‑max weighting of the convolution entries yields a smooth
    approximation to the sub‑gradient of the max‑operator; momentum,
    an annealing temperature and occasional random “jumps” help escape
    plateaux.  The optimiser runs until the time budget (≈ 1000 s) is
    exhausted and the best sequence ever seen is returned.

All helper functions are defined at module top‑level (no closures, no
lambdas) and only NumPy and the supplied ``evaluate_sequence`` are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al., 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / s


# ----------------------------------------------------------------------
# Linear convolution via real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return full convolution x*y (length len(x)+len(y)-1) using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negatives from rounding
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max weighted max‑convolution
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i}
    using a single convolution (reverse w, convolve with a).
    """
    n = a.shape[0]
    u = w[::-1]                     # reverse the weight vector
    conv = _full_convolution_fft(u, a)   # length 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)          # indices n‑1,…,0
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Objective computation (fast, matches evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv) / (sum a)^2 for a real vector a."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    s = a.sum()
    if s <= 0.0:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Shape generators (always return a non‑negative vector that sums to 1)
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf on (0,1) evaluated at mid‑points, normalised."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    tot = a.sum()
    if tot == 0.0:
        return np.full(n, 1.0 / n)
    return a / tot


def _exponential_shape(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    """
    a_i ∝ lam**i   (or lam**(n-1-i) if reverse=True), then normalised.
    lam ∈ (0,1) controls the decay speed.
    """
    i = np.arange(n, dtype=float)
    if reverse:
        i = n - 1 - i
    a = np.power(lam, i)
    tot = a.sum()
    if tot == 0.0:
        return np.full(n, 1.0 / n)
    return a / tot


def _power_shape(n: int, p: float, reverse: bool = False) -> np.ndarray:
    """
    a_i ∝ (i+1)**p   (or (n-i)**p if reverse=True), then normalised.
    Positive p concentrates mass at the right, negative p at the left.
    """
    i = np.arange(1, n + 1, dtype=float)
    if reverse:
        i = np.arange(n, 0, -1, dtype=float)
    a = np.power(i, p)
    tot = a.sum()
    if tot == 0.0:
        return np.full(n, 1.0 / n)
    return a / tot


def _linear_shape(n: int, reverse: bool = False) -> np.ndarray:
    """Simple decreasing linear shape."""
    if reverse:
        a = np.linspace(0.0, 1.0, n)
    else:
        a = np.linspace(1.0, 0.0, n)
    a = np.maximum(a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        return np.full(n, 1.0 / n)
    return a / tot


def _random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Sample a Dirichlet(α,…,α) vector and normalise to sum 1."""
    a = np.random.gamma(concentration, 1.0, size=n)
    tot = a.sum()
    if tot == 0.0:
        return np.full(n, 1.0 / n)
    return a / tot


# ----------------------------------------------------------------------
# Main entry point expected by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.  The function returns
    the best sequence found before the budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_val = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – quick random exploration of simple parametric families
    # ------------------------------------------------------------------
    # allocate roughly 15 % of the total time to this phase
    phase1_end = start_time + (budget_s - 2.0) * 0.15

    while time.time() < phase1_end:
        n = np.random.randint(800, 2501)          # 800 … 2500
        shape_type = np.random.choice(['beta',
                                       'exp',
                                       'exp_rev',
                                       'pow',
                                       'pow_rev',
                                       'lin',
                                       'lin_rev',
                                       'rand'])

        if shape_type == 'beta':
            # log‑uniform sampling of shape parameters
            alpha = 10 ** np.random.uniform(-2.0, 1.0)   # 0.01 … 10
            beta = 10 ** np.random.uniform(-2.0, 1.0)
            a = _beta_shape(n, alpha, beta)

        elif shape_type == 'exp':
            lam = np.random.uniform(0.90, 0.9999)       # decay factor
            a = _exponential_shape(n, lam, reverse=False)

        elif shape_type == 'exp_rev':
            lam = np.random.uniform(0.90, 0.9999)
            a = _exponential_shape(n, lam, reverse=True)

        elif shape_type == 'pow':
            p = np.random.uniform(-3.0, 3.0)            # power exponent
            a = _power_shape(n, p, reverse=False)

        elif shape_type == 'pow_rev':
            p = np.random.uniform(-3.0, 3.0)
            a = _power_shape(n, p, reverse=True)

        elif shape_type == 'lin':
            a = _linear_shape(n, reverse=False)

        elif shape_type == 'lin_rev':
            a = _linear_shape(n, reverse=True)

        else:   # 'rand'
            concentration = np.random.uniform(0.1, 5.0)
            a = _random_dirichlet(n, concentration)

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}s] "
                  f"new best from random search = {best_val:.6f}, n={n}")

    # ------------------------------------------------------------------
    # Phase 2 – projected Adam optimisation starting from the best so far
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely, but provide a fallback uniform vector
        n = 1500
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)

    a = best_seq.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8

    iteration = 0
    eta_start = 0.20          # initial learning rate
    eta_end   = 1e-4          # final learning rate
    # temperature (soft‑max) parameters – will be initialised on first loop
    tau_start = None
    tau_end_factor = 1e-6     # final temperature relative to first max

    while True:
        now = time.time()
        if now >= deadline:
            break

        # convolution of the current candidate
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_obj = float(2.0 * n * M / (a.sum() ** 2))

        if cur_obj < best_val:
            best_val = cur_obj
            best_seq = a.copy()
            print(f"[{now - start_time:.1f}s] new best after refinement = {best_val:.6f}")

        # schedule (0 … 1) over the remaining time
        progress = (now - start_time) / (deadline - start_time)

        # initialisation of the soft‑max temperature
        if tau_start is None:
            tau_start = M * 0.30                     # relatively high at start
            tau_end = max(tau_start * tau_end_factor, 1e-12)

        tau = tau_start * (1.0 - progress) + tau_end * progress

        # soft‑max weights over the convolution entries
        if tau > 0.0:
            exp_args = (conv - M) / tau
            # numeric stabilisation
            max_arg = np.max(exp_args)
            exp_vals = np.exp(exp_args - max_arg)
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # gradient (softened sub‑gradient)
        grad = _gradient_from_weights(a, w)

        # learning‑rate schedule (linear decay)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random mass‑transfer to improve exploration
        if np.random.rand() < 0.02:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta

        # projection back onto the probability simplex
        a = _project_onto_simplex(a)

        iteration += 1

    # ------------------------------------------------------------------
    # Final verification (optional – the outer framework will also call it)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_seq.tolist()
