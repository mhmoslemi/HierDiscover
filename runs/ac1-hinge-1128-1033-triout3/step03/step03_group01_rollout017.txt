def refine_sequence(a, max_iters=5000):
    n = len(a)
    a = a.copy()
    # ensure sum 1
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999
    eps=1e-8
    step0 = 0.2
    t=0
    beta = 0.5
    beta_max = 5000.0
    beta_factor = 1.01 # after each iteration, increase beta multiplicatively until max

    best_local_a = a.copy()
    best_local_score = score_from_vector(a)

    while time.time() < deadline and t < max_iters:
        t+=1
        # compute conv of a*a
        conv = full_convolution_self(a)  # length 2n-1
        # compute smoothed gradient
        max_val = conv.max()
        # weight vector w
        w = np.exp(beta * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback: use indicator at max
            w = (conv >= max_val - 1e-12).astype(float)
            w_sum = w.sum()
        w = w / w_sum

        # compute gradient via convolution w and a_rev
        a_rev = a[::-1]
        grad_full = full_convolution_fft_custom(w, a_rev)  # length 3n-2
        grad = 2 * grad_full[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = -step0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = a + step
        cand = simplex_projection(cand)
        cand = cand / cand.sum()
        # evaluate
        cand_score = score_from_vector(cand)

        # acceptance if improvement or occasional SA (maybe always accept if improvement)
        if cand_score < best_local_score or np.random.rand() < 0.001:
            a = cand
            # we'll accept any candidate with improvement to proceed
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local_a = cand.copy()
        # else maybe stay with a unchanged

        # update best global
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"New best global score {best_score:.6f} at length {n}")

        # update beta
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # optional random perturbation after many steps without improvement
        if t % 5000 == 0:
            # random jitter
            jitter = np.random.randn(n) * 1e-5
            a = simplex_projection(a + jitter)
            a = a / a.sum()
            # reset Adam accumulators maybe
            m = np.zeros_like(a); v = np.zeros_like(a)
    return best_seq, best_score

"""
Propose a step‑function sequence that minimises the evaluation score
2 * n * max_k (a*a)[k] / (sum a)^2.

Algorithm
---------

1. **Deterministic candidate sweep** – For several lengths (up to 20 000)
   generate parametric families (U‑shaped, exponential, cosine windows,
   random) and keep the best few candidates.

2. **Soft‑max Adam refinement** – Starting from each top candidate, run a
   projected Adam optimiser.  The non‑smooth max is approximated by a
   soft‑max (log‑sum‑exp) with a temperature β that is increased during
   optimisation, turning the problem gradually into the original
   max‑objective.  After each Adam step the vector is projected onto the
   probability simplex.  Occasional “re‑heat’’ jitter and a tiny SA
   acceptance probability are used to escape shallow minima.

3. Return the best sequence found (as a plain list of floats).  All
   operations respect the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {w >= 0, sum(w) = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full convolution via real FFT (generic for two vectors)
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of real vectors a and b using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    size = 1 << ((len(a) + len(b) - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def full_convolution_self(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return full_convolution_fft(a, a)


# ----------------------------------------------------------------------
# Exact score (assuming sum(a)=1)
# ----------------------------------------------------------------------


def score_from_vector(a: np.ndarray) -> float:
    """2 * n * max_k (a*a)[k]  (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    conv = full_convolution_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (U‑shapes, exponential, cosine windows)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def hann_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def hamming_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the smoothed objective (soft‑max approximation)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, beta: float):
    """
    Returns (grad, max_val) where grad is the (sub‑)gradient of the
    smooth approximation 2*n*logsumexp(beta*conv)/beta and max_val is the
    current maximum of conv = a*a.
    """
    n = a.shape[0]
    conv = full_convolution_self(a)          # length 2n-1
    max_val = conv.max()

    # soft‑max weights (numerically stable)
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum <= 0.0 or np.isnan(w_sum):
        mask = (conv >= max_val - 1e-12).astype(float)
        w = mask / mask.sum()
    else:
        w /= w_sum

    # gradient = 2 * (w * a_rev)  (full convolution, then slice)
    a_rev = a[::-1]
    grad_full = full_convolution_fft(w, a_rev)   # length 3n-2
    start = n - 1
    grad = 2.0 * grad_full[start : start + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Adam‑style refinement with temperature scheduling
# ----------------------------------------------------------------------


def refine(a_init: np.ndarray, deadline: float, max_iters: int = None):
    """
    Refines a candidate vector a_init using projected Adam optimisation.
    Returns (best_vector, best_score) found before the deadline.
    """
    n = a_init.shape[0]
    a = a_init.copy()
    a /= a.sum()

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.5

    # temperature (soft‑max) schedule
    beta = 0.5
    beta_max = 5000.0
    beta_mul = 1.02

    best_a = a.copy()
    best_score = score_from_vector(a)
    last_improve = 0
    t = 0

    while time.time() < deadline - 0.01:
        if max_iters is not None and t >= max_iters:
            break
        t += 1

        # gradient of the smoothed objective
        grad, _ = smoothed_gradient(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = -step0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = a + step
        cand = simplex_projection(cand)
        cand /= cand.sum()

        cand_score = score_from_vector(cand)

        # accept improvements (always) and occasional worsening moves
        if cand_score < best_score:
            a = cand
            best_a = cand.copy()
            best_score = cand_score
            last_improve = t
            print(
                f"Iter {t:7d}: new local best {best_score:.6f} (beta={beta:.2f})"
            )
        else:
            if np.random.rand() < 0.001:  # tiny SA acceptance
                a = cand

        # temperature annealing
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        # re‑heat if stuck for a long time
        if t - last_improve > 20000:
            jitter = np.random.rand(n) * 1e-5
            a = simplex_projection(a + jitter)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = t
            print(f"Iter {t:7d}: re‑heat due to stagnation")
    return best_a, best_score


# ----------------------------------------------------------------------
# Generate a handful of diverse seeds for a given length
# ----------------------------------------------------------------------


def generate_candidates(n: int):
    """Return a list of candidate vectors for length n."""
    candidates = []

    # uniform
    a = np.full(n, 1.0)
    a /= a.sum()
    candidates.append(a)

    # U‑shapes (min‑distance)
    for p in [0.5, 1, 2, 3, 4, 5, 7, 10]:
        candidates.append(u_min_distribution(n, p))

    # U‑shapes (product‑distance)
    for p in [0.5, 1, 2, 3, 4, 5, 7, 10]:
        candidates.append(u_prod_distribution(n, p))

    # Exponential decay
    for lam in [1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1, 2e-1]:
        candidates.append(exp_distribution(n, lam))

    # Cosine‑based windows
    candidates.append(hann_distribution(n))
    candidates.append(hamming_distribution(n))
    candidates.append(blackman_distribution(n))

    # Randomised seeds
    for _ in range(3):
        r = np.random.rand(n)
        r /= r.sum()
        candidates.append(r)

    return candidates


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score within the given time budget.  Returns the best sequence found
    as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # small safety margin
    print(f"Search started – budget {budget_s}s (deadline at {deadline:.2f})")

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over parametric families
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000]
    top_seeds = []  # list of (score, vector)

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\nGenerating candidates for length n={n}")
        cand_list = generate_candidates(n)
        for a in cand_list:
            sc = score_from_vector(a)
            top_seeds.append((sc, a))
        # keep only the best 20 seeds overall (to limit later work)
        top_seeds.sort(key=lambda x: x[0])
        top_seeds = top_seeds[:20]
        print(f"  current best score among seeds: {top_seeds[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate external construction if present
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        if external.sum() > 0.0:
            external /= external.sum()
            sc = score_from_vector(external)
            top_seeds.append((sc, external))
            top_seeds.sort(key=lambda x: x[0])
            top_seeds = top_seeds[:20]
            print(f"  added external candidate – score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    if not top_seeds:
        # fallback: random vector of moderate size
        n = 5000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = score_from_vector(a)
        print(f"Fallback random init – length {n}, score {best_score:.6f}")
    else:
        best_score, best_seq = top_seeds[0]
        print(f"\nInitial best seed – length {best_seq.shape[0]}, score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement of the top seeds
    # ------------------------------------------------------------------
    print("\n=== Refinement phase ===")
    for idx, (seed_score, seed_vec) in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(
            f"\nRefining seed {idx+1}/{len(top_seeds)} (len={seed_vec.shape[0]}, "
            f"init_score={seed_score:.6f})"
        )
        refined_vec, refined_score = refine(seed_vec, deadline)
        print(f"  refined score = {refined_score:.6f}")
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_vec.copy()
            print(f"  <<< New global best! score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final report
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
