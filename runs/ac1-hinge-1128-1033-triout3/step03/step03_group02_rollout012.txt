A mixture of two gradients is used:

* a soft‑max weight vector focuses on the current maximal convolution entry;
* the L2‑gradient (weights = 2·(a*a – avg)) tries to flatten the entire
  convolution towards the uniform target avg = 1/(2n‑1).

Early in the budget the soft‑max term dominates, afterwards the L2 term
takes over.  Random “jitter’’ moves occasionally transfer a small amount
of mass between two positions to escape plateaus.  The algorithm runs
several short restarts with different lengths and initialisations and
keeps the best sequence seen.

All helper functions are top‑level, no closures are used and the code
conforms to the required `propose_candidate` signature.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al.)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0:
        return np.full_like(v, 1.0 / n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (length len(x)+len(y)-1) using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    # next power of two for efficient FFT
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of Σ_k w_k·(a*a)[k] (valid for arbitrary real w)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k-i} for all i via a single convolution.
    Works for any real weight vector w (positive, negative or zero).
    """
    n = a.shape[0]
    u = w[::-1]                     # reverse the weight vector
    conv = _full_convolution_fft(u, a)   # length = 3n-2
    start = 2 * n - 2
    idx = start - np.arange(n)      # pick entries n..2n-1 in reverse order
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Fast evaluation of the objective for a probability vector (sum=1)
# ----------------------------------------------------------------------
def _fast_value(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialise a sequence of length n with a few different heuristics
# ----------------------------------------------------------------------
def _init_sequence(n: int, mode: str, rng: np.random.Generator) -> np.ndarray:
    if mode == "sin":
        x = np.arange(n, dtype=float)
        a = 1.0 - np.sin(np.pi * x / (n - 1))
        a = np.maximum(a, 0.0)
    elif mode == "dirichlet":
        # Gamma(α) is equivalent to Dirichlet(α) after normalisation
        a = rng.gamma(0.6, size=n)
    elif mode == "two_peaks":
        a = np.zeros(n, dtype=float)
        i, j = rng.choice(n, size=2, replace=False)
        a[i] = a[j] = 0.5
    else:                     # fallback: uniform
        a = np.ones(n, dtype=float)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Core optimisation loop for a single (length, initial) run
# ----------------------------------------------------------------------
def _optimise_one_run(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise sequence a until `deadline` and return the best (a, value)
    observed during this run.
    """
    n = a.shape[0]
    avg = 1.0 / (2 * n - 1)           # target uniform value of the convolution
    best_a = a.copy()
    best_val = _fast_value(a)

    # Adam‑style moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    iteration = 0
    start = time.time()
    # parameters for schedules
    eta_start, eta_end = 0.30, 1e-5          # step size
    tau_ratio_start, tau_ratio_end = 0.30, 1e-5  # relative to current max
    jitter_prob = 0.015

    while True:
        now = time.time()
        if now >= deadline:
            break

        # -------- convolution and current objective ----------
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)   # Σa = 1 by construction
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[run {n}] iter={iteration:,d} new best={best_val:.6f}")

        # ---------- progress based schedules ----------
        total_run = deadline - start
        progress = (now - start) / total_run   # 0..1

        # learning rate (linear decay)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # temperature for soft‑max (relative to the current max)
        tau = M * (tau_ratio_start * (1.0 - progress) + tau_ratio_end * progress)

        # ---------- gradient from soft‑max (focus on max) ----------
        if tau > 0:
            # stability: shift by the maximum (already non‑positive)
            exp_args = (conv - M) / tau
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()
            if sum_exp > 0 and np.isfinite(sum_exp):
                w_soft = exp_vals / sum_exp
            else:
                w_soft = np.zeros_like(conv)
                w_soft[np.argmax(conv)] = 1.0
        else:
            w_soft = np.zeros_like(conv)
            w_soft[np.argmax(conv)] = 1.0
        grad_max = _gradient_from_weights(a, w_soft)

        # ---------- gradient from L2 flattening (global) ----------
        w_err = 2.0 * (conv - avg)          # weight = 2·(b‑avg)
        grad_l2 = _gradient_from_weights(a, w_err)

        # blend the two gradients: early half focuses on max,
        # later half shifts weight to the L2 term.
        if progress < 0.5:
            blend = 0.0
        else:
            blend = (progress - 0.5) * 2.0   # linearly 0 → 1 over [0.5,1]
        grad = (1.0 - blend) * grad_max + blend * grad_l2

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step

        # ---------- occasional random jitter ----------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a_candidate[src] * 0.5
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        # ---------- projection onto simplex ----------
        a = _project_onto_simplex(a_candidate)

        iteration += 1

        # occasional full recomputation to keep numerical drift tiny
        if iteration % 3000 == 0:
            _ = _full_convolution_fft(a, a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The function runs for at most `budget_s`
    seconds and returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_val = float("inf")

    # Prepare a modest pool of lengths and initialisation modes.
    # The pool is deliberately small so that each run gets a reasonable
    # time slice; the loop will keep restarting with fresh random seeds
    # until the overall budget expires.
    length_pool = [1500, 1700, 1900, 2100, 2400]
    mode_pool = ["sin", "dirichlet", "two_peaks"]

    # Pre‑compute an even split of the remaining time among the runs.
    # We'll start a new optimisation run whenever the current one
    # finishes or when we have not seen any improvement for a while.
    run_index = 0
    while time.time() < deadline:
        # pick length & initialisation (cycling through the pools)
        n = length_pool[run_index % len(length_pool)]
        mode = mode_pool[(run_index // len(length_pool)) % len(mode_pool)]

        # try to reuse the previous best as a warm start when possible
        if global_best_a is not None and len(global_best_a) == n:
            a0 = np.array(global_best_a, copy=True)
            # add a small random perturbation to escape possible stagnation
            jitter = rng.normal(scale=0.01, size=n)
            a0 = np.maximum(a0 + jitter, 0.0)
            if a0.sum() == 0:
                a0 = _init_sequence(n, mode, rng)
            else:
                a0 = a0 / a0.sum()
        else:
            a0 = _init_sequence(n, mode, rng)

        # allocate a slice of the remaining budget to this run
        remaining = deadline - time.time()
        # give at least 5 s per run, otherwise stop
        slice_time = max(5.0, remaining * 0.12)   # ~12 % of what is left
        slice_deadline = min(deadline, time.time() + slice_time)

        # run the optimiser
        best_a, best_val = _optimise_one_run(a0, slice_deadline)

        # update the global best
        if best_val < global_best_val:
            global_best_val = best_val
            global_best_a = best_a.copy()
            print(f"[global] new overall best = {global_best_val:.6f} (n={n}, mode={mode})")

        run_index += 1

    # final safety evaluation with the official function
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
