      (log‑sum‑exp) with a temperature β.  Its gradient can be
      written analytically as  g = 2·(w⋆a) where w are the soft‑max
      weights and “⋆” denotes linear cross‑correlation.
      FFTs give both the convolution and the cross‑correlation in
      O(L log L) time (L≈3·n, a power of two).
    • β starts small (smooth) and is multiplied by a factor >1 each
      iteration, gradually turning the smooth objective into the true
      max.  A simple projected gradient step with simplex projection
      (Wang & Carreira‑Perpiñán, O(n log n)) is used, together with
      an annealed acceptance rule (simulated annealing) and occasional
      reheating to escape plateaus.
    • The loop stops when the global time budget (≈ 1000 s) is exhausted.
      The best vector seen is returned.

The algorithm stays fully on‑CPU, uses only NumPy and no file I/O,
and respects the required API.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length 2·len(a).
    Returns the first 2·len(a)‑1 entries (the full convolution).
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score that `evaluate_sequence` would return for a normalized vector."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_conv = conv.max()
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Parametric families (same definitions as in the reference code)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge: (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the two edges: ((i+1)*(n-i))^(-p)."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete symmetric Beta(α,β) on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max descent utilities
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution entries (max‑approximation)."""
    shift = conv - conv.max()          # non‑positive numbers
    exp_shift = np.exp(beta * shift)   # safe; max term is 1
    w = exp_shift / exp_shift.sum()
    return w


def gradient_via_correlation(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Compute g = 2·(w ⋆ a) where ⋆ denotes linear cross‑correlation.
    Implemented with FFTs of length L >= len(a)+len(w)-1.
    Returns a real vector of length len(a).
    """
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    # cross‑correlation: conj(W) * A   (linear, zero‑padded)
    corr = np.fft.irfft(np.conj(W) * A, n=L)
    # first n entries correspond to k = 0 … n‑1
    return 2.0 * corr[: a.shape[0]].real


def smoothed_descent(
    init: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Smoothed max‑convolution descent starting from `init`.
    Returns the best vector seen and its score.
    """
    n = init.shape[0]
    L = next_pow2(3 * n)          # large enough for correlation (≈ 3 n)

    # hyper‑parameters (tuned empirically)
    beta = 2.0                    # soft‑max temperature
    beta_inc = 1.003              # slowly sharpen
    step = 0.5                    # initial step magnitude
    step_decay = 0.9997
    temperature = 1.0
    temp_decay = 0.9999

    cur = init.copy()
    cur_score = score_from_vector(cur)

    best_vec = cur.copy()
    best_score = cur_score
    last_improve = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # ---- convolution & soft‑max weights ---------------------------------
        A_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(A_fft * A_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        w = softmax_weights(conv_full, beta)

        # ---- gradient (cross‑correlation) ----------------------------------
        grad = gradient_via_correlation(cur, w, L)

        # ---- projected gradient step ---------------------------------------
        cand = cur - step * grad
        cand = np.clip(cand, 0.0, None)
        cand = simplex_projection(cand)

        cand_score = score_from_vector(cand)

        # ---- simulated‑annealing acceptance --------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_vec = cand.copy()
                last_improve = it
                print(f"Iter {it:7d}: new best score = {best_score:.6f}")

        # ---- schedule updates -----------------------------------------------
        step *= step_decay
        beta *= beta_inc
        temperature *= temp_decay

        # ---- occasional re‑heat ---------------------------------------------
        if it - last_improve > 30_000:
            noise = np.random.randn(n) * 1e-4
            cur = np.clip(cur + noise, 0.0, None)
            cur = simplex_projection(cur)
            cur_score = score_from_vector(cur)
            temperature = 1.0
            step = 0.5
            beta = 2.0
            last_improve = it
            print(f" * Re‑heat at iter {it}")

    return best_vec, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) discovered before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0     # safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over several families (broader grids)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    lengths = [4000, 8000, 12000]               # modest sizes for quick sweep
    p_grid = np.concatenate(
        (np.arange(0.1, 2.01, 0.2), np.arange(2.2, 6.01, 0.5))
    )
    beta_grid = np.arange(0.2, 5.01, 0.4)
    gamma_grid = np.concatenate(
        (np.arange(0.001, 0.021, 0.001), np.arange(0.025, 0.151, 0.025))
    )

    print("=== Deterministic sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break
        # U‑shape (min distance)
        for p in p_grid:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_min) n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # U‑shape (product of distances)
        for p in p_grid:
            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_prod) n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # Symmetric Beta family
        for alpha in beta_grid:
            a = beta_distribution(n, alpha, alpha)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (beta) n={n:5d} α={alpha:4.2f}  score={sc:.6f}")

        # Symmetric exponential decay
        for gamma in gamma_grid:
            a = exp_decay_distribution(n, gamma)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (exp) n={n:5d} γ={gamma:5.3f}  score={sc:.6f}")

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. External construction if it exists (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        ext = np.array(height_sequence_1, dtype=float)
        if ext.size > 0:
            ext = np.maximum(ext, 0.0)
            total = ext.sum()
            if total > 0.0:
                ext = ext / total
                sc_ext = score_from_vector(ext)
                if sc_ext < best_score:
                    best_score, best_seq = sc_ext, ext.copy()
                    print(f" using external height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement with smoothed‑max descent
    # ------------------------------------------------------------------
    print("\n=== Stochastic refinement ===")
    remaining = deadline - time.time()
    # allocate the majority of the remaining time to the descent
    max_iters = int(min(300_000, remaining * 300))   # crude scaling
    best_seq, best_score = smoothed_descent(best_seq, deadline, max_iter=max_iters)

    # ------------------------------------------------------------------
    # 5. Return result
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # the external evaluator will re‑normalise, so we return the raw vector
    return best_seq.tolist()
