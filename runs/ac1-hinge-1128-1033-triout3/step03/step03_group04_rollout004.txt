"""

   (convolution of a with the soft‑max weights).  An Adam optimiser
   on the probability simplex (projection after each step) is used.
   λ is increased during optimisation so that the surrogate
   becomes tighter.

The method differs from the reference code by using a smooth surrogate,

import time

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------
def _next_pow_of_two(x: int) -> int:
    """Small helper to get the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT (real‑to‑real)."""
    size = _next_pow_of_two(len(u) + len(v) - 1)
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fu * fv, n=size)
    return conv[: len(u) + len(v) - 1]


def self_convolution(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    return conv_fft(a, a)


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a (a is non‑negative, sum=1)."""
    return float(self_convolution(a).max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a probability vector."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def harmonic_left_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n) + 1
    a = i.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def sum_power_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (i+1)^{-p} + (n-i)^{-p} (symmetric sum of power‑law edges)."""
    i = np.arange(n) + 1
    left = i.astype(np.float64) ** (-p)
    right = (n - np.arange(n)).astype(np.float64) ** (-p)
    a = left + right + 1e-14
    a /= a.sum()
    return a


def edge_uniform_distribution(n: int, K: int) -> np.ndarray:
    """Mass placed uniformly on the first and last K positions (2·K ≤ n)."""
    if K <= 0 or 2 * K > n:
        raise ValueError("Invalid K for edge_uniform_distribution")
    a = np.zeros(n, dtype=np.float64)
    mass = 1.0 / (2 * K)
    a[:K] = mass
    a[-K:] = mass
    return a

def adam_softmax_optimize(
    a0: np.ndarray,
    max_iters: int,

    lam_end: float,
    lr: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,

):

    Minimise  log‑sum‑exp( λ·(a*a) )  with Adam on the probability simplex.
    Returns the best vector and its exact score (via score_from_vector).
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = score_from_vector(a)

    # pre‑allocate array for soft‑max weights to avoid repeated allocations?
    for t in range(1, max_iters + 1):
        # optional early exit if we run out of time
        if time_deadline is not None and time.time() > time_deadline:
            break

        # linearly increase λ
        lam = lam_start + (lam_end - lam_start) * (t - 1) / max_iters

        # self‑convolution and soft‑max weights
        conv = self_convolution(a)                     # length 2n‑1
        c_max = conv.max()
        w = np.exp(lam * (conv - c_max))
        w_sum = w.sum()

                if len(top_candidates) > max_top:

        # ------- Exponential decay ----------
        for gamma in gamma_grid:
            a = exp_decay_distribution(n, gamma)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (exp)    n={n:5d} γ={gamma:5.3f}  score={sc:.6f}")
            if len(top_candidates) < max_top or sc < max(top_candidates, key=lambda x: x[0])[0]:
                top_candidates.append((sc, a.copy()))
                top_candidates.sort(key=lambda x: x[0])
                if len(top_candidates) > max_top:
                    top_candidates.pop()

        # ------- Sum‑power (two‑sided) ----------
        for p in p_grid:
            a = sum_power_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (sum)    n={n:5d} p={p:4.2f}  score={sc:.6f}")
            if len(top_candidates) < max_top or sc < max(top_candidates, key=lambda x: x[0])[0]:
                top_candidates.append((sc, a.copy()))
                top_candidates.sort(key=lambda x: x[0])
                if len(top_candidates) > max_top:
                    top_candidates.pop()

        # ------- Edge‑uniform ----------
        for K in K_grid:
            if 2 * K > n:
                continue
            a = edge_uniform_distribution(n, K)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (edgeU)  n={n:5d} K={K:3d}  score={sc:.6f}")
            if len(top_candidates) < max_top or sc < max(top_candidates, key=lambda x: x[0])[0]:
                top_candidates.append((sc, a.copy()))
                top_candidates.sort(key=lambda x: x[0])
                if len(top_candidates) > max_top:
                    top_candidates.pop()

    # ------------------------------------------------------------------
    # 2. Include external construction if present (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            total = external.sum()
            if total > 0.0:
                external /= total
                sc_ext = score_from_vector(external)
                if sc_ext < best_score:
                    best_score, best_seq = sc_ext, external.copy()
                    print(f" using external height_sequence_1  score={sc_ext:.6f}")
                if len(top_candidates) < max_top or sc_ext < max(top_candidates, key=lambda x: x[0])[0]:
                    top_candidates.append((sc_ext, external.copy()))
                    top_candidates.sort(key=lambda x: x[0])
                    if len(top_candidates) > max_top:
                        top_candidates.pop()
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – random seeds (if everything else failed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")
        top_candidates.append((best_score, best_seq.copy()))

    # ------------------------------------------------------------------
    # 4. Stochastic refinement (Adam on soft‑max surrogate)
    # ------------------------------------------------------------------
    print("\n=== Stochastic refinement (Adam + soft‑max surrogate) ===")
    # Determine remaining time and split roughly equally among candidates
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        # time exhausted – return what we have
        print("Time budget exhausted before refinement.")
        return best_seq.tolist()

    # allocate at most 200 seconds per candidate (adjustable)
    max_time_per = min(200.0, remaining_time / max(1, len(top_candidates)))
    for idx, (init_score, seed_vec) in enumerate(top_candidates):
        if time.time() > deadline:
            break
        n = seed_vec.shape[0]
        # heuristic for number of iterations: about 4000 * (200 / n) but cap
        max_iters = int(15000 * (20000 / n))   # roughly 15k iterations for n≈20000
        max_iters = max(3000, min(max_iters, 25000))

        # tighten deadline for this run
        sub_deadline = time.time() + max_time_per

        print(f"\nRefining candidate {idx + 1}/{len(top_candidates)}: n={n}, init_score={init_score:.6f}, iters≈{max_iters}")

        refined_vec, refined_score = adam_softmax_optimize(
            a0=seed_vec,
            max_iters=max_iters,
            lam_start=2.0,
            lam_end=30.0,
            lr=0.03,
            time_deadline=sub_deadline,
        )
        print(f" → candidate refined score = {refined_score:.6f}")

        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_vec.copy()
            print(f" *** New global best! score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator will clip and renormalise again, but we already
    # respect the required constraints (non‑negative, finite, reasonable magnitude).
    return best_seq.tolist()
