Propose a non‑negative step function minimizing the provided evaluation metric.

Algorithm Overview
------------------
1. **Warm‑start** – If a global variable ``height_sequence_1`` exists,
   use it (normalized to sum‑1) as an initial candidate.

2. **Deterministic parametric sweep** – Generate a few thousand vectors
   from several symmetric families (U‑shape, double‑tail, exponential,
   product‑power‑law, sine‑shape).  For each length in a growing set
   (up to 32 000) and each parameter value, compute a cheap proxy
   ``2·n·max(conv)`` using a real FFT.  Keep the best proxy as the start
   for the local optimizer.

3. **Multiphase stochastic refinement** –  Perform a simulated‑annealing
   search on the simplex with three move types:
   * **Multiplicative gradient step** – Identify the current maximal
     convolution entry, compute its sub‑gradient (g_i = 2·a_{k−i}) and
     update the vector multiplicatively:  a_i ← a_i·exp(−η·g_i).  The
     step size η follows the temperature schedule.
   * **Random mass‑transfer** – Move a random fraction of the mass from
     one index to another.
   * **Noise perturbation** – Apply a tiny multiplicative Gaussian
     perturbation to all entries (useful for escaping plateaus).

   After each move the vector is symmetrised (a←(a+rev(a))/2) and
   projected back onto the probability simplex.
   Acceptance follows a Metropolis rule with a slowly cooling temperature.

4. **Exact evaluation** – The best vector found is finally evaluated
   with ``evaluate_sequence`` and returned.

The routine respects a total time budget (default 1000 s) and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Return max entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Parametric families (all return a normalized vector with sum == 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^(-p) + (n‑i)^(-p)."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges: a[i] ∝ exp(-α·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """Product‑power‑law shape: a[i] ∝ ((i+1)*(n‑i))^(-p)."""
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sine_distribution(n: int, p: float) -> np.ndarray:
    """Sine‑shaped U‑distribution: a[i] ∝ sin(π·(i+0.5)/n)^p."""
    i = np.arange(n) + 0.5
    a = np.sin(np.pi * i / n) ** p + 1e-12
    a /= a.sum()
    return a


def logistic_distribution(n: int, scale: float) -> np.ndarray:
    """Logistic (sigmoid) shape, symmetric around centre."""
    mid = (n - 1) / 2.0
    i = np.arange(n)
    a = 1.0 / (1.0 + np.exp((np.abs(i - mid) - mid) / scale)) + 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Helper to keep a vector on the probability simplex and symmetric
# ----------------------------------------------------------------------


def project_onto_simplex_sym(a: np.ndarray) -> np.ndarray:
    """
    Symmetrise, clip to non‑negative and renormalise to sum‑1.
    """
    # symmetrise
    a = (a + a[::-1]) * 0.5
    # clip negatives (should not happen but guard)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        # fallback to uniform
        a[:] = 1.0 / a.shape[0]
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimizing ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin for final eval

    best_seq = None          # will hold a numpy array (sum == 1)
    best_score = np.inf      # proxy: 2·n·max(conv)

    # ------------------------------------------------------------------
    # 0. Warm‑start from external variable if present
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)  # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external /= s
            approx = 2.0 * external.size * max_conv_fft(external)
            if approx < best_score:
                best_score = approx
                best_seq = external.copy()
                print(f"[warm‑start] external seed approx_score={approx:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Deterministic parametric sweep (quick coarse search)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000, 24000, 28000, 32000]
    p_vals = np.linspace(0.2, 3.0, 27)          # exponents for power‑law families
    alphas = np.linspace(0.0002, 0.010, 15)    # exponential decay rates
    scales = np.linspace(0.01, 0.3, 14)        # logistic scales

    print("\n=== Parametric sweep (coarse) ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # U‑shaped power‑law
        for p in p_vals:
            if time.time() > deadline:
                break
            a = u_shaped_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[sweep] n={n:5d}, U‑pow p={p:4.2f}, approx={score:.6f}")

        # Double‑tail power‑law
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sum_tails_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[sweep] n={n:5d}, tails p={p:4.2f}, approx={score:.6f}")

        # Exponential decay
        for alpha in alphas:
            if time.time() > deadline:
                break
            a = exponential_distribution(n, alpha)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[sweep] n={n:5d}, exp α={alpha:5.4f}, approx={score:.6f}")

        # Product power‑law
        for p in p_vals:
            if time.time() > deadline:
                break
            a = product_powerlaw_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[sweep] n={n:5d}, prod‑pow p={p:4.2f}, approx={score:.6f}")

        # Sine‑shape
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sine_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[sweep] n={n:5d}, sine p={p:4.2f}, approx={score:.6f}")

        # Logistic shape
        for scale in scales:
            if time.time() > deadline:
                break
            a = logistic_distribution(n, scale)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[sweep] n={n:5d}, logistic s={scale:5.3f}, approx={score:.6f}")

    # ------------------------------------------------------------------
    # 2. If no candidate found (unlikely), fall back to a random vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = 2.0 * n * max_conv_fft(a)
        print(f"[fallback] random init n={n}, approx={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local stochastic refinement (simulated‑annealing + gradient)
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]

    conv = full_conv_fft(current)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    temperature = 1.0
    cooling_rate = 0.996
    iteration = 0

    # Move‑type probabilities
    prob_grad = 0.35
    prob_transfer = 0.55
    prob_noise = 0.10

    print("\n=== Stochastic refinement ===")
    while time.time() < deadline:
        iteration += 1
        move_rnd = np.random.rand()

        # ------------------------------------------------------------------
        # a) Gradient (multiplicative) step
        # ------------------------------------------------------------------
        if move_rnd < prob_grad:
            # Identify the index attaining the current convolution maximum
            kmax = int(conv.argmax())
            # compute sub‑gradient g_i = 2·a_{kmax−i} where defined
            idx = np.arange(n)
            partner = kmax - idx
            mask = (partner >= 0) & (partner < n)
            grad = np.zeros_like(current)
            grad[mask] = 2.0 * current[partner[mask]]

            # step size adapts to temperature
            eta = temperature * 0.07   # small multiplier; tuned experimentally
            # multiplicative update (mirror descent on simplex)
            cand = current * np.exp(-eta * grad)
            cand = project_onto_simplex_sym(cand)

        # ------------------------------------------------------------------
        # b) Random mass‑transfer move (batch evaluation)
        # ------------------------------------------------------------------
        elif move_rnd < prob_grad + prob_transfer:
            # Take a batch of proposals and keep the best one (greedy)
            batch = 12
            best_local_score = cur_score
            best_candidate = None

            for _ in range(batch):
                src = np.random.choice(n, p=current)          # source weighted by mass
                # ensure destination != source
                dst = np.random.randint(0, n - 1)
                if dst >= src:
                    dst += 1

                # fraction to move, shrinking with temperature
                max_frac = 0.4 * temperature
                delta = current[src] * np.random.rand() * max_frac
                if delta <= 0.0:
                    continue

                cand = current.copy()
                cand[src] -= delta
                cand[dst] += delta
                cand = project_onto_simplex_sym(cand)

                cand_score = 2.0 * n * max_conv_fft(cand)
                if cand_score < best_local_score:
                    best_local_score = cand_score
                    best_candidate = cand

            if best_candidate is not None:
                cand = best_candidate
                cand_score = best_local_score
            else:
                # No improving move found – do a tiny SA acceptance attempt
                cand = current
                cand_score = cur_score + np.inf

        # ------------------------------------------------------------------
        # c) Small global noise perturbation
        # ------------------------------------------------------------------
        else:
            # multiplicative Gaussian noise, magnitude tied to temperature
            sigma = 0.025 * temperature
            noise = np.random.randn(n) * sigma
            cand = current * (1.0 + noise)
            cand = project_onto_simplex_sym(cand)
            cand_score = 2.0 * n * max_conv_fft(cand)

        # ------------------------------------------------------------------
        # Acceptance test (Metropolis)
        # ------------------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability to accept a worsening move
            prob = math.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            conv = full_conv_fft(current)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            if cur_score < best_score:
                best_score = cur_score
                best_seq = current.copy()
                print(f"[iter {iteration:5d}] new best approx={best_score:.6f}")

            # early stop if we have already breached the target bound
            if best_score < 1.503:
                print("[target] Approximate bound reached, exiting early.")
                break

        # temperature schedule
        temperature = max(temperature * cooling_rate, 1e-6)

    # ------------------------------------------------------------------
    # 4. Final exact evaluation
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())  # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {best_score:.6f}")
    print(f"Best exact score (if computed) : {exact_score:.6f}")
    print(f"Best length            : {len(best_seq)}")
    return best_seq.tolist()
