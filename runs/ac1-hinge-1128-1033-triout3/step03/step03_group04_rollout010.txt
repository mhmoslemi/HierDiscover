def gradient_refinement(a, time_budget):
    n = len(a)
    current = a.copy()
    current_score = 2 * n * max_convolution(current)
    best = current.copy()
    best_score = current_score
    start = time.time()
    step = 0.2
    T = 0.5
    lambda_err = 0.5
    iteration = 0
    last_best_iter = 0
    while time.time() < time_budget:
        iteration +=1
        # compute conv, max, etc.
        conv = full_convolution_fft(current)
        max_val = conv.max()
        # active indices:
        active_idx = np.where(np.isclose(conv, max_val, rtol=1e-12, atol=0))[0]
        # grad_sub
        grad_sub = np.zeros_like(current)
        for k in active_idx:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            i_arr = np.arange(i_start, i_end+1)
            j_arr = k - i_arr
            grad_sub[i_arr] += 2.0 * current[j_arr]
        # gradient error
        avg = 1.0/(2*n-1)
        err = conv - avg
        # compute cross-correlation gradient
        size = 1
        while size < 3*n:
            size <<=1
        a_rev = current[::-1]
        fa_rev = np.fft.rfft(a_rev, n=size)
        ferr = np.fft.rfft(err, n=size)
        conv_rev = np.fft.irfft(fa_rev * ferr, n=size)
        grad_err = 2.0 * conv_rev[n-1:n-1+n]
        
        grad = grad_sub + lambda_err * grad_err
        
        # update
        cand = current - step * grad
        cand = np.maximum(cand, 0)
        tot = cand.sum()
        if tot > 0:
            cand /= tot
        else:
            cand = np.full_like(cand, 1.0/n)
        
        # evaluate
        cand_score = 2.0 * n * max_convolution(cand)
        # acceptance
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(T, 1e-12))
            accept = np.random.rand() < prob
        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score - 1e-12:
                best = cand.copy()
                best_score = cand_score
                last_best_iter = iteration
                print(f"Iter {iteration:6d}: new best {best_score:.6f}")
        # cooling
        T *= 0.99995
        step *= 0.9999
        # reheat
        if iteration - last_best_iter > 20000:
            # add small jitter
            jitter = np.random.rand(n) * 1e-5
            current = np.maximum(current + jitter, 0)
            current /= current.sum()
            current_score = 2.0*n*max_convolution(current)
            T = 0.5
            step = 0.2
            last_best_iter = iteration
            print(f"Reheat at iteration {iteration}")
    return best, best_score
```

"""
Propose a low‑score step‑function.

The algorithm has two phases:

1. **Deterministic sweep** over several parametric families (U‑shape,
   beta‑distribution, exponential decay, Gaussian) for a handful of
   lengths.  Each candidate is normalised to unit sum and its exact
   score `2·n·max(a*a)` is computed via a single FFT.  The best vector
   from this sweep is kept as a seed.

2. **Gradient‑based refinement** of the seed.  For the current vector
   `a` we compute
      * the self‑convolution `c = a*a` and its maximum,
      * the *sub‑gradient* of the max‐convolution (active set = indices
        where `c` attains its maximum),
      * the gradient of the L2‑error `‖c‑mean(c)‖₂²`, which is the
        cross‑correlation of `a` with the error vector.
   The combined gradient points towards an increase of the max;
   we therefore move **opposite** to it, project back onto the simplex,
   and accept the move according to a simple simulated‑annealing rule.
   Temperature and step size are slowly cooled; occasional reheating
   adds a tiny random jitter when progress stalls.

The whole search stops after the allotted time (≈ 1000 s) and returns the
best sequence discovered.  All numbers are non‑negative; the returned
list is ready to be fed to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (fallback, not used for the main updates)
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w ≥ 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray, size: int = None) -> np.ndarray:
    """
    Linear convolution a*a using an FFT.
    If ``size`` is provided, the FFT length is forced to that value.
    Returns the full convolution (length 2·len(a)‑1).
    """
    n = a.shape[0]
    if size is None:
        size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(fa * fa, n=size)
    return conv_full[: 2 * n - 1]


def max_convolution(a: np.ndarray, size: int = None) -> float:
    """Maximum entry of the self‑convolution of a (a is non‑negative)."""
    conv = full_convolution_fft(a, size=size)
    return float(conv.max())


def score_from_vector(a: np.ndarray, size: int = None) -> float:
    """Exact score returned by evaluate_sequence for a unit‑sum vector."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a, size=size)


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge: (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the two edges."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Discrete (truncated) Gaussian centred in the middle."""
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------
def subgradient_active(a: np.ndarray, conv: np.ndarray,
                       active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    ``active_idx`` contains the indices where conv attains its maximum.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def gradient_l2_error(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the L2‑error ‖conv‑mean(conv)‖₂² w.r.t. a.
    Uses an FFT of length ≥ 3·n for the cross‑correlation.
    """
    n = a.shape[0]
    avg = 1.0 / (2 * n - 1)
    err = conv - avg

    # size for FFT – the next power of two >= 3·n
    size = 1 << ((3 * n - 1).bit_length())

    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, n=size)
    ferr = np.fft.rfft(err, n=size)
    conv_rev = np.fft.irfft(fa_rev * ferr, n=size)
    # slice that corresponds to indices [n‑1 … n‑1+n‑1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
