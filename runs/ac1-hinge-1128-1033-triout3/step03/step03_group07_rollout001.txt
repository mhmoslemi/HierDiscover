Function:

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def gradient_cross_corr(a, w):
    # w length = 2*n-1
    # compute cross-correlation of a (len n) with w (len 2n-1)
    # g_i = 2 * Σ_j a_j w_{i+j}
    # Use FFT: treat w_rev = w[::-1]; conv = np.convolve(a, w_rev, mode='full')
    # Then slice
    n = a.shape[0]
    size = 1 << ((n + (2*n-1) -1).bit_length())
    # size = next power of two >= 3n-2
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(w[::-1], n=size)
    conv = np.fft.irfft(A * B, n=size)[: 3*n -2]
    # cross-correlation: relevant slice indices: (len(w)-1) to (len(w)-1)+n-1 inclusive
    start = (2*n-1) - 1 # actually len(w)-1 = 2n-2
    # conv indices start.. start + n -1
    grad = 2.0 * conv[start : start + n]
    return grad

"""
Propose a non‑negative step function minimizing the metric
    2·n·max(a∗a) / (∑a)²  (the denominator is fixed by normalising ∑a=1)

Algorithm
---------
1.   Initialise a distribution on the simplex.  If a global variable
     ``height_sequence_1`` exists it is used (resampled / padded to the
     target length).  Otherwise a random vector or a simple exponential tail
     is taken.

2.   Optimise the smooth approximation of the max‑convolution:
        Fβ(a) = (1/β)·log∑k exp(β·(a∗a)[k])
     Its gradient is
        ∇Fβ(a) = 2·(a ⋆ w)   with   w_k ∝ exp(β·(a∗a)[k])
     (⋆ denotes cross‑correlation).  All convolutions are carried out with a
     real FFT (O(n log n)).

3.   Projected gradient descent with a continuation on β.
     * β is increased in stages [2, 5, 10, 20, 40, 80]; larger β makes the
       approximation tighter.
     * Step size is scaled as η / (β·√t) (η≈0.2, t = iteration counter).
     * After each step the vector is projected onto the probability simplex
       (non‑negative, sum = 1) using the O(n) algorithm of Duchi et al.
     * If the exact objective improves, the best sequence is stored.
     * Occasionally a tiny random perturbation is added to escape shallow
       local minima.

4.   The loop runs until the allotted time (≈ 1000 s) expires or the target
     score 1.503 is reached.  Finally the exact ``evaluate_sequence`` routine
     is called for the best vector and the list of floats is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index (largest i satisfying u_i > cssv_i / i)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero -> return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights proportional to exp(beta·conv)."""
    max_val = conv.max()
    exp_vals = np.exp(beta * (conv - max_val))
    return exp_vals / exp_vals.sum()


def gradient_cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·(a ⋆ w) where ⋆ is cross‑correlation.
    a has length n, w has length 2n‑1.
    Returns a vector of length n.
    """
    n = a.shape[0]
    size = 1 << ((3 * n - 2).bit_length())  # next power of two ≥ 3n‑2
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(w[::-1], n=size)          # reverse w for cross‑correlation
    conv_full = np.fft.irfft(A * B, n=size)[: 3 * n - 2]
    start = 2 * n - 2                         # len(w)-1
    grad = 2.0 * conv_full[start:start + n]
    return grad


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Edge‑decaying exponential a[i] ∝ exp(‑α·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a
