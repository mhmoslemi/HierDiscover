def improve_via_lp(p, max_iters, t, deadline):
    n = len(p)
    while time.time() < deadline and iters < max_iters:
        # compute conv of p with itself
        conv = np.convolve(p, p)
        M = conv.max()
        # Build A matrix
        A = np.zeros((2*n-1, n))
        for j in range(n):
            A[j:j+n, j] = p
        # Solve LP: maximize sum(b) s.t. A @ b <= M, b>=0
        # Equivalent: minimize -sum(b)
        # Use scipy.optimize.linprog
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            print('LP failed', res.message)
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        # Normalize to sum 1
        b_norm = b / sum_b
        # Mix with current p: p = (1-t)*p + t*b_norm
        p = (1 - t) * p + t * b_norm
        # Optionally reduce t adaptively if improvement small? We'll keep constant.
        iters+=1
        # Compute new objective
        new_conv = np.convolve(p, p)
        new_score = 2.0 * n * new_conv.max()
        # keep track best
        # if improvement else break maybe...
    return p

"""
Propose Candidate – LP‑based iterative smoothing.

The routine builds a few symmetric probability vectors (arcsine, uniform,
a few random draws and the historic construction if available).
For each start it repeats:

1.  Compute the self‑convolution of the current vector p and its
    maximum value M = max(p * p).

2.  Solve a linear program

        maximise    Σ b_i
        subject to  (p * b)_k ≤ M   for all k = 0 … 2n‑2
                    b_i ≥ 0

    The convolution (p * b) is linear in b, therefore the constraints are
    linear.  The LP returns a vector b that spreads the mass as much as
    possible without exceeding the current worst‑case bound M.

3.  Normalise b to a probability vector (b̂ = b / Σb) and move a small
    fraction t towards it:
        p ← (1‑t)·p + t·b̂ .
    The mixture keeps the bound M (the quadratic term t²·b̂*b̂ is tiny for
    small t) while allowing M to shrink in subsequent iterations.

4.  After a few hundred LP steps the progress stalls; a fast
    pair‑wise‑swap hill‑climber with tiny mass transfers finishes the
    optimisation.

All vectors are kept symmetric (p ← (p+p[::-1])/2) and normalised to sum
to one, which is without loss of generality because the evaluation
function is invariant to a uniform scaling of the whole vector.
The final sequence is clipped to the limits enforced by
``evaluate_sequence`` and returned as a plain list of floats.
"""

import time
import numpy as np

# Try to import the LP solver from SciPy.  Fall back to a no‑op if unavailable.
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _objective(p: np.ndarray, n: int) -> float:
    """Return the exact value that ``evaluate_sequence`` would give for a
    probability vector (sum(p) == 1)."""
    conv = np.convolve(p, p)
    return float(2.0 * n * np.max(conv))


def _build_conv_matrix(p: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A ∈ ℝ^{(2n‑1)×n} with entries
        A[k, j] = p[k‑j]  if 0 ≤ k‑j < n,
        A[k, j] = 0      otherwise.
    For a given p this matrix satisfies (p * b) = A @ b.
    """
    n = p.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)

    # each column j contains p starting at row j
    for j in range(n):
        A[j:j + n, j] = p
    return A


def _lp_step(p: np.ndarray, M: float) -> np.ndarray:
    """
    One LP step: maximise Σ b_i under the constraints (p * b) ≤ M.
    Returns the optimal b (unscaled).  If the LP fails, returns None.
    """
    if not _SCIPY_AVAILABLE:  # pragma: no cover
        return None

    A = _build_conv_matrix(p)
    rows = A.shape[0]
    c = -np.ones(p.shape[0])               # maximise sum(b)  ⇔ minimise -sum(b)
    b_ub = np.full(rows, M, dtype=float)   # right‑hand side of A·b ≤ M
    bounds = [(0.0, None)] * p.shape[0]    # b_i ≥ 0

    # Use the HiGHS interior‑point solver – fast for medium‑size LPs.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def _symmetrize(p: np.ndarray) -> np.ndarray:
    """Force symmetry p_i = p_{n‑1‑i} and renormalise to sum 1."""
    p = (p + p[::-1]) * 0.5
    s = p.sum()
    if s == 0:
        return p
    return p / s
