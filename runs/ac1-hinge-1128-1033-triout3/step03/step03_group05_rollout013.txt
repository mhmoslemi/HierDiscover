def lp_improvement(f, n, m_fft, max_iter=80, t=0.2, deadline ):

    best_f = f
    best_score = _score(f)

    for it in range(max_iter):
        # compute conv via FFT
        conv = _conv_self(f, m_fft)
        M = conv.max()
        # build convolution matrix
        col = np.concatenate([f, np.zeros(n-1)])
        row = np.concatenate([f[:1], np.zeros(n-1)])
        A = toeplitz(col, row)  # shape (2n-1, n)

        # linear constraints: A @ g <= M
        # variable bounds: g >= 0
        # objective: maximize sum(g) -> minimize -sum(g)

        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failed at iter {it}: {res.message}")
            break
        g0 = res.x

        # Normalize g to sum = 1
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 / sum_g0

        f = (1 - t) * f + t * g

        # Ensure sum=1 (due to rounding)
        f /= f.sum()

        cur_score = _score(f, m_fft)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_f = f.copy()
            print(f" LP iter {it}: new best score {best_score:.6f}")

        # optionally reduce t (like annealing)
        t = max(t * 0.97, 0.01)  # gradually smaller steps

        if time.time() >= deadline:
            break

    return best_f, best_score

def subgradient_phase(p, n, m_fft, deadline, max_iters=5000):
    best_score = _score(p, m_fft)
    best_p = p.copy()
    step = 0.2
    decay = 0.9995
    min_step = 1e-6
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self_fft(p, m_fft)
        max_val = conv.max()
        max_idx = conv.argmax()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
        # compute subgradient:
        g = np.zeros_like(p)
        if max_idx < n:
            # overlap size = max_idx + 1
            # indices of p that contribute: p[0..max_idx]
            # gradient for position i (i <= max_idx) is 2 * p[max_idx - i]
            g[:max_idx+1] = 2.0 * p[max_idx::-1]
        else:
            start = max_idx - n + 1
            # For positions start..n-1
            # i index corresponds to p[i] with (i - start)?? gradient 2 * p[n-1 - (i-start)] = 2 * p[ (n - 1) - (i - start) ]
            rev_slice = p[n-1:start-1:-1] if start > 0 else p[n-1::-1]
            g[start:] = 2.0 * rev_slice
        # update:
        p = p - step * g
        # projection onto simplex:
        p = _simplex_projection(p)
        # step decay:
        step = max(step * decay, min_step)
    return best_p, best_score

def lp_flatten_phase(p, n, m_fft, deadline, max_iter=20, t_start=0.4):
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz

    best_score = _score(p, m_fft)
    best_p = p.copy()

    t = t_start
    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # compute current M
        conv = _conv_self_fft(p, m_fft)
        M = conv.max()
        # build convolution matrix A (2n-1 x n)
        col = np.concatenate([p, np.zeros(n - 1, dtype=float)])  # length 2n-1
        row = np.concatenate([p[:1], np.zeros(n - 1, dtype=float)])
        A = toeplitz(col, row)  # shape (2n-1, n)

        # linear program: maximize sum(g) s.t. A @ g <= M, g >= 0
        c = -np.ones(n)  # minimize -sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0.0, None)] * n

        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print(f" LP iteration {it} failed: {res.message}")
            break

        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 0:
            break
        g = g0 / sum_g

        # convex combination
        p = (1.0 - t) * p + t * g
        # ensure sum = 1
        p /= p.sum()

        cur_score = _score(p, m_fft)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            print(f" LP iter {it} – new best {best_score:.6f}")

        # anneal t
        t = max(t * 0.95, 0.01)

    return best_p, best_score
```

def adam_refine_phase(p, n, m_fft, deadline, max_iters=5000):
    # parameters
    m1 = np.zeros_like(p)
    m2 = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05
    alpha = 1.0
    alpha_mul = 1.002
    alpha_max = 50.0

    best_score = _score(p, m_fft)
    best_p = p.copy()

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute conv via FFT (full)
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
        # smoothing via log-sum-exp
        z = alpha * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        w = exp_z / np.sum(exp_z)  # softmax weights over 2n-1 entries

        # correlation between p and reversed w to get gradient of smoothed max
        w_rev = w[::-1]
        # compute convolution p * w_rev (same size)
        # Could use FFT again:
        w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
        corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]  # length 3n-2 (full linear conv)
        idx = 2 * n - 2 - np.arange(n)  # mapping to indices for gradient entries
        grad = 2.0 * corr_full[idx]  # shape n

        # Adam update
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * (grad ** 2)
        m1_hat = m1 / (1 - beta1 ** (it + 1))
        m2_hat = m2 / (1 - beta2 ** (it + 1))
        step_vec = lr * m1_hat / (np.sqrt(m2_hat) + eps)
        p = p - step_vec
        # simplex projection
        p = _simplex_projection(p)

        # evaluate
        cur_score = _score(p, m_fft)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            # optionally print
            # print(f" Adam iter {it} – new best {best_score:.6f}")

        # increase alpha
        alpha = min(alpha * alpha_mul, alpha_max)

    return best_p, best_score

def swap_refinement(p, n, m_fft, deadline, attempts=5000):
    best_score = _score(p, m_fft)
    best_p = p.copy()
    for _ in range(attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer small epsilon of mass from i to j
        eps_move = 1e-4 * rng.random()
        if p[i] < eps_move:
            continue
        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move
        # projection not needed because sum unchanged and nonnegative
        new_score = _score(new_p, m_fft)
        if new_score < best_score - 1e-12:
            p = new_p
            best_score = new_score
            best_p = new_p.copy()
            # print improvement
            print(f"Swap improvement → {best_score:.6f}")
            if best_score <= 1.503:
                break
    return best_p, best_score

"""
Propose a candidate step‑function sequence minimizing
    2 * n * max(self‑convolution) / (sum^2)

The algorithm combines several ideas:
* a family of arcsine‑type initial shapes (different exponents);
* a cheap sub‑gradient descent on the current maximal convolution entry;
* a linear‑programming “flattening” step (maximize sum(g) s.t. f∗g ≤ M);
* Adam optimisation on a smooth log‑sum‑exp approximation of the max;
* a tiny random‑swap hill‑climber.

All steps respect a hard 1000 s wall‑clock budget and always return a
non‑negative sequence whose sum is positive (the evaluator normalises
the scale automatically).  The code works even if SciPy is unavailable
(the LP phase is simply skipped).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (onto {x ≥ 0, sum(x)=1})
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_self_fft(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution of p using a pre‑computed FFT size."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _score(p: np.ndarray, m_fft: int) -> float:
    """Return 2·n·max(conv_self) for a probability vector p."""
    n = p.size
    conv = _conv_self_fft(p, m_fft)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient descent on the actual max entry
# ----------------------------------------------------------------------
def _subgradient_phase(p0: np.ndarray, n: int, m_fft: int,
                       deadline: float, max_iters: int = 5_000):
    """Move probability mass away from the current peak."""
    p = p0.copy()
    best_p = p.copy()
    best_score = _score(p, m_fft)

    step = 0.2
    decay = 0.9995
    min_step = 1e-6

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self_fft(p, m_fft)
        max_idx = conv.argmax()
        cur_score = 2.0 * n * conv[max_idx]

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            print(f"  Sub‑grad iter {it:5d} – new best {best_score:.6f}")
            if best_score <= 1.503:
                break

        # sub‑gradient of the maximal entry
        g = np.zeros_like(p)
        if max_idx < n:
            # indices 0 … max_idx are active
            g[: max_idx + 1] = 2.0 * p[max_idx :: -1]
        else:
            start = max_idx - n + 1
            # indices start … n‑1 are active
            rev = p[n - 1 : start - 1 : -1] if start > 0 else p[::-1]
            g[start:] = 2.0 * rev

        p = p - step * g
        p = _simplex_projection(p)

        step = max(step * decay, min_step)

    return best_p, best_score


# ----------------------------------------------------------------------
# Linear‑programming flattening step (requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz
    _HAVE_SCIPY = True
except Exception:               # pragma: no cover
    _HAVE_SCIPY = False
    print("SciPy not available – LP flattening phase will be skipped.")


def _lp_flatten_phase(p0: np.ndarray, n: int, m_fft: int,
                      deadline: float, max_iter: int = 12,
                      t_start: float = 0.5):
    """Iteratively mix f with a maximising g (linear programme)."""
    if not _HAVE_SCIPY:
        return p0.copy(), _score(p0, m_fft)

    p = p0.copy()
    best_p = p.copy()
    best_score = _score(p, m_fft)
    t = t_start

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _conv_self_fft(p, m_fft)
        M = conv.max()

        # build convolution matrix A (2n‑1 rows, n cols)
        col = np.concatenate([p, np.zeros(n - 1, dtype=float)])
        row = np.concatenate([p[:1], np.zeros(n - 1, dtype=float)])
        A = toeplitz(col, row)                     # shape (2n‑1, n)

        # Linear programme: maximise sum(g)  ⇔ minimise –1·g
        c = -np.ones(n, dtype=float)
        b_ub = np.full(2 * n - 1, M, dtype=float)  # f ∗ g ≤ M
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"  LP iter {it} failed: {res.message}")
            break

        g0 = res.x
        s = g0.sum()
        if s <= 0:
            break
        g = g0 / s                                 # normalise to sum‑1

        # convex mixing
        p = (1.0 - t) * p + t * g
        p /= p.sum()                               # enforce exact simplex

        cur_score = _score(p, m_fft)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            print(f"  LP iter {it} – new best {best_score:.6f}")
            if best_score <= 1.503:
                break

        t = max(t * 0.95, 0.01)                    # slowly shrink step size

    return best_p, best_score


# ----------------------------------------------------------------------
# Adam optimisation on a log‑sum‑exp smooth max
# ----------------------------------------------------------------------
def _adam_phase(p0: np.ndarray, n: int, m_fft: int,
                deadline: float, max_iters: int = 20_000):
    """Soft‑max (log‑sum‑exp) smoothing + Adam optimisation."""
    p = p0.copy()
    best_p = p.copy()
    best_score = _score(p, m_fft)

    # Adam state
    m1 = np.zeros_like(p)
    m2 = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    # smoothing schedule
    alpha = 1.0
    alpha_mul = 1.001
    alpha_max = 50.0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # full self‑convolution
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

        # soft‑max weights over the 2n‑1 entries
        z = alpha * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)                     # stability
        w = exp_z / np.sum(exp_z)

        # gradient of the smooth approximation
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
        corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]
        idx = 2 * n - 2 - np.arange(n)                # map to gradient indices
        grad = 2.0 * corr_full[idx]

        # Adam update
        m1 = beta1 * m1 + (1.0 - beta1) * grad
        m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

        m1_hat = m1 / (1.0 - beta1 ** (it + 1))
        m2_hat = m2 / (1.0 - beta2 ** (it + 1))

        step_vec = lr * m1_hat / (np.sqrt(m2_hat) + eps_adam)
        p = p - step_vec
        p = _simplex_projection(p)

        cur_score = _score(p, m_fft)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            # optional progress print
            # print(f"  Adam iter {it:5d} – new best {best_score:.6f}")
            if best_score <= 1.503:
                break

        alpha = min(alpha * alpha_mul, alpha_max)

    return best_p, best_score

# Tiny random‑swap hill‑climber
# ----------------------------------------------------------------------
def _swap_refinement(p0: np.ndarray, n: int, m_fft: int,
                     deadline: float, rng: np.random.Generator,
                     attempts: int = 5_000):
    """Move a tiny amount of probability between two indices if it helps."""
    p = p0.copy()
    best_score = _score(p, m_fft)
    best_p = p.copy()

    for _ in range(attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_move = 1e-4 * rng.random()
        if p[i] < eps_move:
            continue
        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move
        new_score = _score(new_p, m_fft)
        if new_score < best_score - 1e-12:
            p = new_p
            best_score = new_score
            best_p = new_p.copy()
            print(f"Swap improvement → {best_score:.6f}")
            if best_score <= 1.503:
                break

    return best_p, best_score

        for start_id, p0 in enumerate(start_vectors):
            if time.time() >= deadline:
                break

            print(f"\n-- Start {start_id + 1}/{len(start_vectors)} --")

            # ---------------------------------------------------------
            # (1) Sub‑gradient descent on the actual max entry
            # ---------------------------------------------------------
            p, score = _subgradient_phase(p0, n, m_fft, deadline,
                                          max_iters=8_000)
            if score < best_overall_score - 1e-12:
                best_overall_score = score
                best_overall_seq = p.copy()
                print(f"  After sub‑grad: new overall best {best_overall_score:.6f}")

            # ---------------------------------------------------------
            # (2) LP flattening step (if SciPy is present)
            # ---------------------------------------------------------
            p, score = _lp_flatten_phase(p, n, m_fft, deadline,
                                         max_iter=12, t_start=0.5)
            if score < best_overall_score - 1e-12:
                best_overall_score = score
                best_overall_seq = p.copy()
                print(f"  After LP: new overall best {best_overall_score:.6f}")

            # ---------------------------------------------------------
            # (3) Adam refinement on smooth max
            # ---------------------------------------------------------
            p, score = _adam_phase(p, n, m_fft, deadline, max_iters=15_000)
            if score < best_overall_score - 1e-12:
                best_overall_score = score
                best_overall_seq = p.copy()
                print(f"  After Adam: new overall best {best_overall_score:.6f}")

            # ---------------------------------------------------------
            # (4) Final tiny swap hill‑climber
            # ---------------------------------------------------------
            p, score = _swap_refinement(p, n, m_fft, deadline, rng,
                                        attempts=4_000)
            if score < best_overall_score - 1e-12:
                best_overall_score = score
                best_overall_seq = p.copy()
                print(f"  After swaps: new overall best {best_overall_score:.6f}")

            # early stopping if we hit the target
            if best_overall_score <= 1.503:
                print("Target reached – terminating early.")
                break

        print(f"Best score for n={n}: {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        n = 1153
        m_fft = 1 << ((2 * n - 2).bit_length())
        best_overall_seq = np.full(n, 1.0 / n)
        best_overall_score = _score(best_overall_seq, m_fft)

    print(f"\nFinal best score: {best_overall_score:.6f}")

    # Clip to the limits enforced by evaluate_sequence (0 ≤ x ≤ 1000) and
    # return a plain Python list.
    return np.clip(best_overall_seq, 0.0, 1000.0).tolist()
