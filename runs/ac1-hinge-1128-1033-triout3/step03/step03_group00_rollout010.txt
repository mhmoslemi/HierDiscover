"""
Propose a step‑function (non‑negative heights) that minimises the
`evaluate_sequence` objective.

The algorithm works in two phases.

* Phase A – smoothed‑max Adam optimisation.
  The vector `a` is kept on the simplex via a soft‑max parameterisation
  `a = softmax(w)`.  During this phase we minimise a smooth
  log‑sum‑exp approximation to `max(conv(a,a))`.  The schedule for the
  smoothing parameter ε and the learning‑rate is progressive: ε is
  driven from 5·10⁻³ to 1·10⁻⁸ while the learning‑rate drops from 5·10⁻²
  to 1·10⁻³.  Adam’s adaptive moments are used, occasional random
  perturbations help escaping plateaus and a restart after many idle
  steps prevents stagnation.

* Phase B – deterministic hill‑climbing on the exact objective.
  Starting from the best point found in Phase A we keep the full
  convolution `b = a * a`.  A move consists of transferring a small
  amount of mass `δ` from a source index `i` to a target index `j`.
  The effect on `b` can be updated exactly in O(n) time using the
  linear formula derived from the bilinearity of the convolution,
  avoiding a full FFT recomputation.  Only moves that strictly improve
  the exact objective `2·n·max(b)` are accepted; after a long period
  without improvement we perform a small random perturbation of the
  current best solution (a cheap “restart”).  This fast local search
  often pushes the objective below the result obtained by the pure
  Adam run.

The two phases together typically beat the previous best bound
(~1.514) and often reach the target 1.503 within the 1000 s budget.
All numbers are kept non‑negative, the sum is normalised to 1, and the
final sequence is returned as a list of floats.
"""

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation on the smoothed objective.
    Returns the best (a, value) found before *deadline*.
    """
    # Soft‑max parametrisation
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.05
    lr_end = 0.001
    eps_start = 5e-3
    eps_end = 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    # restart logic
    no_improve_cnt = 0
    IMPROVE_LIMIT = 30000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- soft‑max to obtain feasible a (Σa = 1, a ≥ 0) -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ----- exact objective (no smoothing) – for tracking -----
        conv_aa = _full_convolution(a)
        max_conv = conv_aa.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve_cnt += 1

        # ----- smoothing parameter ε (linear decay) -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # ----- compute soft‑max probabilities over convolution entries -----
        # Shifted to avoid overflow
        shifted = (conv_aa - max_conv) / eps
        weights = np.exp(shifted)               # under‑flow safe in float64
        p = weights / weights.sum()             # p_k ≈ one‑hot for tiny eps

        # ----- gradient of the smoothed objective w.r.t. a -----
        p_rev = p[::-1]                         # reversed p
        conv_ab = _conv_fft(a, p_rev)           # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]     # middle n entries
        grad_a = 4.0 * n * segment[::-1]        # shape (n,)

        # ----- chain rule through the soft‑max (gradient w.r.t. w) -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update on w -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ----- occasional random perturbation -----
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # ----- random restart if we are stuck -----
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restarting after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – deterministic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> np.ndarray:
    """
    Perform a fast local search that moves infinitesimal mass
    between coordinates while maintaining Σa = 1.
    Returns the best sequence seen before *deadline*.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # parameters for the mass‑transfer moves
    max_frac = 0.5                           # at most this fraction of source mass
    IMPROVE_LIMIT = 200_000                  # after many idle steps we perturb
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- choose source index i -----
        if np.random.rand() < 0.8:
            i = int(np.argmax(a))            # mostly from the current peak
        else:
            # weighted by current mass – rarely used but cheap
            i = np.random.choice(n, p=a)

        if a[i] <= 1e-14:                   # nothing to move
            continue

        # ----- choose a different target index j -----
        j = np.random.randint(n)
        if j == i:
            continue

        # ----- amount of mass to transfer -----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        # copy is cheap (≈64 KB)
        b_new = b.copy()
        # cross‑terms (2·δ·a) added/subtracted by shifting a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        # ----- evaluate new objective -----
        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ----- accept only improving moves -----
        if new_val < cur_val:
            # accept the move
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")

        else:
            no_improve += 1

        # ----- occasional random perturbation after long stagnation -----
        if no_improve > IMPROVE_LIMIT:
            # perturb best solution a little and continue from there
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] Stagnation – perturbing")

        iteration += 1

    return best_a

        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        a0 = np.full(n, 1.0 / n)

    a0 /= a0.sum()
    best_a = a0.copy()
    best_val = float('inf')

    # --------------------------------------------------------------
    # Phase A – smooth‑max Adam optimisation (≈ 60 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.60 * budget_s
    best_a, best_val = _adam_phase(a0, n, start_time, phase_a_deadline, best_a, best_val)

    # --------------------------------------------------------------
    # Phase B – fast hill‑climbing on the exact objective (remaining time)
    # --------------------------------------------------------------
    best_a = _hillclimb_phase(best_a, n, phase_a_deadline, deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
