def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0

    best_val = float('inf')
    best_seq = None

    # Helper functions
    def conv_max(a):
        # compute max conv using FFT
        n = a.shape[0]
        size = 2*n - 1
        m = 1 << ((size - 1).bit_length())
        A = np.fft.rfft(a, n=m)
        C = np.fft.irfft(A * A, n=m)[:size]
        # maybe small negative due to precision
        return np.max(C)

    def evaluate_and_update(a):
        nonlocal best_val, best_seq
        # compute objective
        val = 2.0 * a.shape[0] * conv_max(a)  # sum=1
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[new best] n={a.shape[0]}, val={best_val:.10f}")
        return val

    # Phase 1: sample Beta-shaped distributions
    # Predefine n=2000 for speed
    n0 = 2000
    for _ in range(2000):
        # random α,β in some ranges
        alpha = rng.uniform(0.1, 5.0)
        beta = rng.uniform(0.1, 5.0)

"""
Hybrid optimizer for the step‑function convolution bound.

The algorithm works in three stages:

1. **Initialisation** – try to use a previously stored sequence
   (`height_sequence_1`).  If unavailable, fall back to the classic
   Catalan (constant‑convolution) construction or a random Dirichlet
   vector.

2. **Momentum sub‑gradient descent** – at each iteration the self‑
   convolution is computed with an FFT, the (up to three) indices where
   the convolution attains its maximum are selected and the sub‑gradient
   of the maximal entry is accumulated.  A heavy‑ball momentum term is
   added and the updated vector is projected back onto the probability
   simplex (∑a = 1, a_i ≥ 0).  The step size decays slowly; after a long
   stretch without improvement the algorithm restarts from a fresh random
   point.

3. **Finalisation** – the best sequence seen is clamped to the limits
   imposed by the judge (0 ≤ height ≤ 1000) and evaluated once with the
   official `evaluate_sequence` routine.

The whole search respects the 1000 s wall‑clock budget and prints
intermediate progress.  The returned list of heights should achieve a
lower bound than the previous 1.519724, aiming for the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x | x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full self‑convolution via FFT (real valued).
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution a*a (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 2 * n - 1
    # use the next power of two for speed
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    C = np.fft.irfft(A * A, n=m)[:size]
    # tiny negative values caused by rounding are harmless after clipping
    return np.maximum(C, 0.0)


# ----------------------------------------------------------------------
# Helper: sub‑gradient for a set of maximal convolution indices.
# ----------------------------------------------------------------------
def _gradient_at_indices(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of max_k (a*a)_k for the given maximal indices.
    The gradient is the average over the selected indices.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for idx in idxs:
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i_range                     # always in [0, n-1]
        g[i_range] += 2.0 * a[j]
    if len(idxs) > 0:
        g /= float(len(idxs))
    return g


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function (list of heights) that minimises the
    evaluation function within the given time budget.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Wall‑clock time budget in seconds.
    **kwargs : dict
        Ignored – present only for interface compatibility.

    Returns
    -------
    list[float]
        The best height sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    # safety margin for the harness
    deadline = time.time() + budget_s - 4.0

    # --------------------------------------------------------------
    # 1) Choose a problem size.
    # --------------------------------------------------------------
    n = 5000   # length of the step‑function (kept constant for simplicity)

    # --------------------------------------------------------------
    # 2) Initialise sequence.
    # --------------------------------------------------------------
    a = None
    # Try to use a previously discovered construction if it exists.
    try:
        # `height_sequence_1` should be a list/array of floats if supplied.
        h1 = np.asarray(height_sequence_1, dtype=np.float64)
        if h1.size == n:
            a = np.maximum(h1, 0.0)
            a = a / a.sum()
            print("[init] using global height_sequence_1")
        else:
            # length mismatch – ignore it
            pass
    except Exception:  # noqa: BLE001
        pass

    # Fall back to Catalan (constant‑convolution) if needed.
    if a is None:
        coeffs = np.empty(n, dtype=np.float64)
        cur = 1.0
        coeffs[0] = cur
        for k in range(1, n):
            cur = cur * (2.0 * k - 1.0) / (2.0 * k)
            coeffs[k] = cur
        a = coeffs / coeffs.sum()
        print("[init] Catalan start, n={}".format(n))

    # --------------------------------------------------------------
    # 3) Track the best solution.
    # --------------------------------------------------------------
    conv = _conv_full(a)
    best_val = 2.0 * n * conv.max()          # sum(a)=1 ⇒ objective = 2·n·max(conv)
    best_seq = a.copy()
    print(f"[init] initial_score={best_val:.10f}")

    # --------------------------------------------------------------
    # 4) Optimisation parameters.
    # --------------------------------------------------------------
    eta = 0.20                # initial step size
    min_eta = 1e-7
    decay = 0.9998            # geometric decay
    beta = 0.9                # momentum coefficient
    v = np.zeros_like(a)      # momentum buffer
    it = 0
    no_improve = 0
    TARGET = 1.503            # stop early if we reach this bound

    # --------------------------------------------------------------
    # 5) Main optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # Full self‑convolution and maximal entry
        conv = _conv_full(a)
        max_val = conv.max()

        # Find (up to) three indices where the convolution is maximal.
        # A tiny tolerance avoids missing numerically equal peaks.
        tol = max_val * 1e-12 + 1e-15
        max_idxs_all = np.flatnonzero(conv >= max_val - tol)
        # pick at most three distinct indices uniformly at random
        k = min(3, max_idxs_all.size)
        chosen = rng.choice(max_idxs_all, size=k, replace=False)

        # Sub‑gradient (average over the chosen indices) and momentum update
        g = _gradient_at_indices(a, chosen)
        v = beta * v + (1.0 - beta) * g

        # Gradient step + projection onto simplex
        a = a - eta * v
        a = _project_simplex(a)

        # Periodically evaluate the true objective
        if it % 10 == 0:
            conv = _conv_full(a)
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_seq = a.copy()
                no_improve = 0
                print(f"[iter {it:6d}] new best={best_val:.10f}  step={eta:.6f}")
            else:
                no_improve += 1

            # early exit if we already beat the target
            if best_val <= TARGET:
                print(f"[info] target reached ({best_val:.10f}) at iteration {it}")
                break

        # Decay step size, but never let it become smaller than `min_eta`
        eta = max(min_eta, eta * decay)

        # Restart if we are stuck for a while
        if no_improve > 3000:
            a = rng.random(n)
            a = a / a.sum()
            v = np.zeros_like(a)
            eta = 0.20
            no_improve = 0
            print(f"[restart] iteration {it}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (clamped to evaluator limits).
    # --------------------------------------------------------------
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(best_seq)
    print(f"[result] final_score={final_score:.10f}, length={len(best_seq)}")
    return best_seq
