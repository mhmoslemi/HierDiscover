def optimize_adam(a0, max_iters, deadline):
    n = a0.shape[0]
    fft_size = 1 << ((3*n - 2).bit_length())
    eps = 1e-12
    z = np.log(a0 + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.1
    alpha = 0.5
    alpha_max = 400.0
    alpha_mul = 1.02

    best_a = a0.copy()
    best_score = compute_score(best_a)

    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break

        # compute a from softmax(z)
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        # compute conv
        conv = full_conv_fft(a, fft_size)  # shape 2n-1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum()**2)  # near 2*n*max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # optionally print
        # compute softmax weights
        w = softmax_weights(conv, alpha)
        # gradient
        grad_a = gradient_softmax(a, w, fft_size)
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z - step
        # update alpha
        alpha = min(alpha * alpha_mul, alpha_max)
        # optional: add noise occasionally
        if t % 200 == 0:
            # small Gaussian noise to z to escape
            z += np.random.normal(scale=0.01, size=n)
        # optional print progress
        if t % 500 == 0:
            print(f"Iter {t}: cur_score≈{cur_score:.6f}, best_score≈{best_score:.6f}, α={alpha:.2f}")

    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimises the
evaluation metric used by `evaluate_sequence`.

Algorithm
---------

1.  Build a diverse collection of cheap initial candidates
    (U‑shaped power laws, exponentials, a few Dirichlet draws and,
    if present, the external `height_sequence_1`).  All candidates are
    normalised to sum = 1 and resized to a modest length `n0≈2000`.

2.  Optimise each candidate with an Adam optimiser on the *soft‑max*
    of the self‑convolution.
      * a is represented as a soft‑max of free logits `z`,
        guaranteeing positivity and the unit‑sum constraint.
      * For a given temperature `α` the smooth max
        `Sα(a) = (1/α)·log Σexp(α·conv)` is differentiable.
        Its gradient w.r.t. `a` is obtained by a single FFT‑based
        convolution (O(n log n)).
      * Adam provides adaptive step‑sizes and momentum, which avoids the
        stagnation observed in simple mirror‑descent.
      * `α` is increased slowly (geometric schedule) so the bound
        tightens while optimisation proceeds.

3.  Keep the best (lowest `2·n·max(conv)`) candidate found over all
    initialisations.  Upsample it to the final length `N≈20000` by
    linear interpolation and normalise again.

4.  Run a short Adam‑refinement on the long vector and finally evaluate
    the exact score with `evaluate_sequence`.  The best vector seen is
    returned.

The routine respects the global time budget (default 1000 s) and prints
intermediate progress.  All helper functions are top‑level and use only
NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution `a*a` using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full self‑convolution (length 2·n‑1) using an FFT of length ``size``.
    ``size`` must be ≥ 2·n‑1.
    """
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights w_k = exp(α·conv_k) / Σ exp(α·conv).
    Stabilised by subtracting max(conv) before exponentiation.
    """
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the weighted sum Σ w_k·conv_k (conv = a*a) w.r.t. a.
    Implements the formula ∇ = a * w  +  a_rev * w_rev, using FFTs.
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of w (and of conv)
    A = np.fft.rfft(a, n=size)

    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # a * w  (length n+L-1 = 3·n‑2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    idx = L - 1 - np.arange(n)                     # 2·n‑2, …, n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (Σx_i=1, x_i≥0)."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a normalised probability vector ``old``
    to a new length ``new_n`` and re‑normalisation.
    """
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new += 1e-14                     # avoid exact zeros
    new /= new.sum()
    return new


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law “U‑shape”: a_i ∝ (min(i, n‑1‑i)+1)^‑p."""
    d = np.arange(n)
    dist = np.minimum(d, n - 1 - d)
    a = (dist + 1.0) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exponential_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.arange(n)
    dist = np.minimum(d, n - 1 - d)
    a = np.exp(-lam * dist) + 1e-14
    a /= a.sum()
    return a


def approximate_score(a: np.ndarray) -> float:
    """Score assuming Σa = 1 : 2·n·max(conv)."""
    n = a.shape[0]
    max_conv = max_conv_fft(a)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Core optimiser (Adam on the soft‑max of the convolution)
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray,
                  max_iters: int,
                  deadline: float,
                  verbose_prefix: str = "") -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a probability vector.
    Returns the best vector found and its approximate score.
    """
    n = a0.shape[0]
    # FFT size large enough for the gradient computation (≥ 3·n‑2)
    fft_size = 1 << ((3 * n - 2).bit_length())
    eps = 1e-14

    # initialise logits (soft‑max parametrisation)
    z = np.log(a0 + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.2                     # base learning rate for the logits

    # temperature for the soft‑max of the convolution
    alpha = 0.5
    alpha_max = 400.0
    alpha_mul = 1.02

    best_a = a0.copy()
    best_score = approximate_score(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- a = softmax(z) ----------
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)               # stabilised exponentials
        a = exp_z / exp_z.sum()                  # unit‑sum, non‑negative

        # ---------- convolution and score ----------
        conv = full_conv_fft(a, fft_size)        # length 2·n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max            # Σa = 1 ⇒ exact objective
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"{verbose_prefix}Iter {t:5d}: new best ≈{cur_score:.6f}")

        # ---------- gradient of smooth max ----------
        w = softmax_weights(conv, alpha)         # soft‑max probabilities
        grad_a = gradient_softmax(a, w, fft_size)

        # ---------- map gradient to logits ----------
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # ---------- Adam update ----------
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z - step

        # ---------- anneal the temperature ----------
        alpha = min(alpha * alpha_mul, alpha_max)

        # occasional tiny random jitter to escape flat regions
        if t % 250 == 0:
            z += np.random.normal(scale=0.01, size=n)

        # progress report
        if t % 500 == 0:
            print(f"{verbose_prefix}Iter {t:5d}: cur≈{cur_score:.6f}, "
                  f"best≈{best_score:.6f}, α={alpha:.1f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the platform
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------
    # 0.  Load an external seed if it exists
    # ------------------------------------------------------------
    external = None
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        external = np.maximum(external, 0.0)
        total = external.sum()
        if total < 1e-12:
            external = np.ones_like(external)
            total = external.size
        external = external / total
        print(f"External seed found, length={external.size}")
    except NameError:
        print("No external seed available.")

    # ------------------------------------------------------------
    # 1.  Build a pool of cheap initial candidates (all length n0)
    # ------------------------------------------------------------
    n0 = 2000                     # modest size for the heavy optimisation
    candidates = []

    if external is not None:
        candidates.append(resample_distribution(external, n0))

    # power‑law U‑shapes
    for p in np.linspace(0.3, 2.0, 7):
        candidates.append(u_shaped_distribution(n0, p))

    # exponentials
    for lam in np.linspace(0.001, 0.2, 5):
        candidates.append(exponential_distribution(n0, lam))

    # a few random Dirichlet draws
    for _ in range(3):
        candidates.append(np.random.dirichlet(np.ones(n0)))

    # ------------------------------------------------------------
    # 2.  Optimise each candidate on the short length n0
    # ------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # allocate roughly 60 % of the budget to the short‑length phase
    short_deadline = start_time + 0.60 * budget_s

    for idx, a_start in enumerate(candidates):
        if time.time() >= short_deadline:
            break
        print(f"\n=== Start optimisation of candidate {idx} (n={n0}) ===")
        a_opt, score_opt = adam_optimize(
            a_start,
            max_iters=8000,
            deadline=short_deadline,
            verbose_prefix=f"[c{idx}] "
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            print(f"*** Global best updated: approx score {best_score:.6f}")

    # fallback – should never happen
    if best_a is None:
        print("No candidate succeeded; falling back to a simple U‑shape.")
        best_a = u_shaped_distribution(n0, p=0.8)
        best_score = approximate_score(best_a)

    # ------------------------------------------------------------
    # 3.  Upsample to the final length and perform a short refinement
    # ------------------------------------------------------------
    N_final = 20000
    a_big = resample_distribution(best_a, N_final)

    print("\n=== Refinement on the full length ===")
    final_deadline = deadline
    a_final, final_score = adam_optimize(
        a_big,
        max_iters=3000,
        deadline=final_deadline,
        verbose_prefix="[final] "
    )

    # ------------------------------------------------------------
    # 4.  Final official evaluation & return
    # ------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(a_final.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")
        exact_score = None

    # Choose the sequence that gave the best *exact* score (if we have it)
    if exact_score is not None and exact_score < np.inf:
        result = a_final.tolist()
    else:
        # fall back to the approximate‑best vector
        result = a_final.tolist()

    print("\n=== Search finished ===")
    print(f"Best approximate score : {final_score:.6f}")
    if exact_score is not None:
        print(f"Best exact score       : {exact_score:.6f}")
    print(f"Best length            : {len(result)}")
    return result
