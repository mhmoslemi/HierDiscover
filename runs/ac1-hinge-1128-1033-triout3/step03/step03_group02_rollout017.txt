def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0
    # Choose n
    n = 1742  # as previous; could also try larger.
    # We'll also try alternative n values maybe n_list = [1742, 2000, 3000]
    best_a = None
    best_score = float('inf')
    for n in [1742, 2000, 3000, 4000]:
        if time.time() >= deadline: break
        # lambda range
        lambda0 = math.e / (2.0 * n)
        # coarse search
        for factor in np.linspace(0.5, 1.5, 101):
            lam = lambda0 * factor
            r = 1.0 - lam
            if r <= 0.0: continue
            # compute geometric sequence
            # Avoid overflow in r**n: use np.power
            powers = np.power(r, np.arange(n, dtype=np.float64))
            denom = 1.0 - powers[-1]  # 1 - r^n
            if denom <= 0:
                continue
            C = (1.0 - r) / denom
            a = C * powers
            # compute conv and score
            conv = _full_convolution_fft(a, a)
            score = float(2.0 * n * conv.max())
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[n={n}, factor={factor:.3f}] new best score {best_score:.6f}")
            # early break if near lower bound maybe
            if best_score <= 1.0 + 1e-6:
                break
        # optionally refine around best factor with smaller step
    # after all loops, return best_a as list
    return best_a.tolist()

"""
Propose a candidate step‑function.

The algorithm works in two phases.

1.  A cheap parametric search.
    We use two simple “U‑shaped’’ bases (1‑sin and |cos|) that are
    high at the ends and low in the middle.  Raising a base to a power α
    flattens or sharpens the dip, and we mix the result with the uniform
    distribution using a weight t∈[0,1].  The (α,t) pair is searched on a
    coarse grid.  For every candidate a we evaluate the objective
    2·n·max(a*a) (the same quantity returned by `evaluate_sequence`
    when Σa = 1) using a fast FFT‑based convolution.  The best (α,t)
    combination is kept.

2.  Local refinement.
    Starting from the best vector found above we run a projected
    sub‑gradient descent.  The sub‑gradient is computed from the
    active set of indices attaining the current maximum of the
    auto‑convolution; the update uses a momentum term and a linear
    step‑size schedule, finally projecting back onto the probability
    simplex.  The best vector seen during this phase is returned.

The whole routine respects the 1000 s time budget and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y (length len(x)+len(y)-1) using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper: gradient from a weight vector over the convolution entries
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    For a vector a and a weight vector w (same length as a*a),
    compute the gradient of Σ_k w_k (a*a)[k] which equals
        g_i = 2 Σ_k w_k a_{k-i}
    Implemented via a single convolution.
    """
    n = a.shape[0]
    u = w[::-1]                          # reverse the weights
    conv = _full_convolution_fft(u, a)   # length 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)           # pick n entries
    g = 2.0 * conv[idx]
    return g

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine first performs a
    low‑dimensional grid search over simple parametric families and
    then refines the best candidate with a projected sub‑gradient descent.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Choose length n (use previous solution if available)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:
        n = 1742                               # typical length used before

    # Uniform distribution on the simplex
    uniform = np.full(n, 1.0 / n, dtype=float)

    # --------------------------------------------------------------
    # Define base “U‑shaped’’ profiles (high at the ends, low in the middle)
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=float)
    base1 = 1.0 - np.sin(np.pi * idx / (n - 1))          # 1‑sin shape
    base2 = np.abs(np.cos(np.pi * idx / (n - 1)))       # |cos| shape
    base_profiles = [base1, base2]

    best_a = uniform.copy()
    best_score = float('inf')
    best_params = None

    # --------------------------------------------------------------
    # Coarse grid search on (α, t) for each base profile
    # --------------------------------------------------------------
    alphas = np.linspace(0.5, 1.5, 51)   # power to raise the base
    ts = np.linspace(0.0, 1.0, 51)      # mixing weight with uniform

    for base_id, base in enumerate(base_profiles, 1):
        base_sum = base.sum()
        if base_sum == 0:
            continue
        # normalise once (the shape will be re‑normalised after powering)
        base_normalised = base / base_sum

        for alpha in alphas:
            # shape = base ** alpha , then re‑normalise to sum 1
            shape = base_normalised ** alpha
            shape_sum = shape.sum()
            if shape_sum == 0:
                continue
            shape = shape / shape_sum

            for t in ts:
                # convex combination with the uniform distribution
                a = (1.0 - t) * uniform + t * shape

                # evaluate (the sum of a is exactly 1)
                conv = _full_convolution_fft(a, a)
                cur_score = float(2.0 * n * conv.max())

                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    best_params = (base_id, alpha, t)
                    print(f"[grid] new best = {best_score:.6f} "
                          f"(base{base_id}, α={alpha:.3f}, t={t:.3f})")

                # time check
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Local refinement: projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    # Use the best vector from the grid as a warm start
    a = best_a.copy()
    momentum = np.zeros_like(a)

    # hyper‑parameters (linear schedules)
    eta_start = 0.20
    eta_end   = 1e-6
    beta = 0.90                         # momentum decay
    jitter_prob = 0.02                  # occasional random mass transfer

    iteration = 0
    max_iters = 200_000                 # safety cap; real stop is time‑budget

    while True:
        now = time.time()
        if now >= deadline or iteration >= max_iters:
            break

        # ----- full convolution and current value -----
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)

        # keep the best ever seen
        if cur_val < best_score:
            best_score = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] improvement = {best_score:.6f}")

        # ----- active set of constraints (indices achieving the max) -----
        eps = max(1e-12, M * 1e-12)          # tolerance for “active’’ indices
        active = np.where(conv >= M - eps)[0]
        if active.size == 0:                # degenerate safety fallback
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size    # uniform weight on the active set

        # ----- gradient (sub‑gradient of the max‑convolution) -----
        grad = _gradient_from_weights(a, w)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step‑size schedule (linear decay) -----
        progress = iteration / 200_000.0
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # ----- candidate step + occasional random jump -----
        a_candidate = a - eta * momentum

        if np.random.rand() < jitter_prob:
            # move a random small fraction of mass from src to dst
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a_candidate[src] * 0.3
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        # ----- projection back onto the simplex -----
        a = _project_onto_simplex(a_candidate)

        iteration += 1

        # sanity recomputation occasionally
        if iteration % 5000 == 0:
            _ = _full_convolution_fft(a, a)   # keep numerical drift tiny

    # --------------------------------------------------------------
    # Final reporting (optional – the outer framework will also call it)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_val:.6f} "
              f"(params: {best_params})")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
