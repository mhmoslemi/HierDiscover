"""Step‑function optimizer using power‑law shapes.

The evaluation objective is  
    score = 2·n·max(conv(a,a)) / (sum(a)**2)

Because the score is homogeneous, we can fix Σa = 1.  For a vector
a_i ∝ (i+1)^p (1 ≤ i ≤ n) the convolution has an analytically known
asymptotic maximum:
    max(conv) = ((p+1)! )² / ( (2(p+1)−1)! ) / n    (large n)

Hence the score approaches
    2·((p+1)! )² / ( (2(p+1)−1)! ),
which decreases very fast as p grows.
The algorithm simply enumerates a handful of integer exponents p,
builds the corresponding normalised power‑law sequences,
evaluates them with the supplied ``evaluate_sequence`` routine, and
returns the best one found before the time budget expires.
No external optimisation loops are required – the construction itself
is essentially optimal for the chosen family of sequences.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The method searches over power‑law shapes
    a_i ∝ (i+1)^p for a few integer exponents p and picks the best.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # leave a safety margin

    # problem size – large enough that the asymptotic formula holds.
    n = 4000

    # pre‑compute the index vector (1‑based for the power law)
    idx = np.arange(1, n + 1, dtype=np.float64)

    best_seq = None
    best_score = float('inf')

    # exponents to try – the score drops dramatically with p.
    # p = 0 gives the uniform distribution (score = 2).
    # p = 1 gives a linear increase (score ≈ 4/3).
    # Larger p quickly drive the score towards zero.
    exponents = list(range(0, 11))   # 0 … 10 inclusive

    for p in exponents:
        # stop if we run out of time
        if time.time() >= deadline:
            break

        # build the raw power‑law vector and normalise to sum 1
        a_raw = np.power(idx, p, dtype=np.float64)
        a_raw_sum = a_raw.sum()
        if a_raw_sum == 0.0:
            continue
        a = a_raw / a_raw_sum

        # quick surrogate score (exact for Σa=1): 2·n·max(conv)
        conv = np.convolve(a, a)          # full convolution, length 2n‑1
        max_conv = conv.max()
        surrogate = 2.0 * n * max_conv

        # we keep the best according to the real, safety‑checked evaluator.
        # The surrogate is monotone with the true score, but we still call the
        # official function for correctness and to honour the clipping logic.
        if surrogate < best_score:
            try:
                real_score = evaluate_sequence(a.tolist())
            except Exception:
                real_score = float('inf')

            if real_score < best_score:
                best_score = real_score
                best_seq = a.tolist()
                print(f"[p={p}] new best score = {best_score:.12f}")

    # fallback – if, for any reason, no sequence was accepted, use uniform.
    if best_seq is None:
        best_seq = (np.full(n, 1.0 / n)).tolist()
        best_score = evaluate_sequence(best_seq)
        print("[fallback] using uniform distribution, score =",
              f"{best_score:.12f}")

    print(f"[final] best score after search = {best_score:.12f}")
    return best_seq
