Implementation of conv_fft:

```
def conv_fft(x, y):
    size = len(x) + len(y) - 1
    fsize = 1 << (size-1).bit_length()  # next power of two
    X = np.fft.rfft(x, fsize)
    Y = np.fft.rfft(y, fsize)
    z = np.fft.irfft(X*Y, fsize)
    return z[:size]
```

    size = len(x) + len(y) - 1
    fsize = 1 << (size-1).bit_length()
    X = np.fft.rfft(x, fsize)
    Y = np.fft.rfft(y, fsize)

def smooth_max_grad(a, beta):

    conv = conv_fft(a, a)

"""
Propose a low‑score step‑function by smooth‑max gradient descent.

The algorithm:
1. Initialise a non‑negative vector on the simplex (U‑shape or a provided
   external sequence).
2. At each iteration compute the self‑convolution a*a.
3. Approximate the max‑norm of the convolution by a smooth
   log‑sum‑exp (soft‑max) with a temperature β.
4. Obtain the gradient of this smooth objective via FFT‑based
   convolutions (O(n log n)).
5. Perform a mirror‑descent / multiplicative update and project back
   onto the probability simplex.
6. Increase β slowly so the smooth approximation converges to the true
   max, decay the step size, and occasionally re‑heat to escape plateaus.
7. Keep the best vector seen and stop when the time budget expires.

The method uses only FFTs (via numpy) and runs entirely in memory,
making it fast enough to explore many thousands of candidates for
large n (≈2·10⁴) within the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (real signals)
# ----------------------------------------------------------------------


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    size = x.shape[0] + y.shape[0] - 1
    fsize = 1 << (size - 1).bit_length()        # next power of two
    X = np.fft.rfft(x, fsize)
    Y = np.fft.rfft(y, fsize)
    Z = X * Y
    conv = np.fft.irfft(Z, fsize)
    return conv[:size]


# ----------------------------------------------------------------------
# Parametric families (used for a quick deterministic seed)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape a[i] ∝ (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    For a simplex‑normalised vector a (sum(a)=1, a≥0) compute

        f(a) = (1/beta) * log Σ_k exp(beta * (a*a)[k])

    and its gradient ∇f(a).  Returns (grad, max_conv) where max_conv
    is the exact maximum entry of the self‑convolution (used for the
    true score 2·n·max_conv).
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = conv_fft(a, a)                 # length 2n‑1
    max_conv = conv.max()

    # 2) soft‑max weights (numerically stable)
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3) gradient of the smooth max:
    #    ∂f/∂a_i = 2 * Σ_k w_k a_{k-i}
    a_rev = a[::-1]
    conv2 = conv_fft(a_rev, w)            # length 3n‑2
    grad = 2.0 * conv2[n - 1 : n - 1 + n]  # slice of length n
    return grad, max_conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) discovered before the time budget.
    """

    # Choose a length – larger n gives a tighter theoretical bound.
    # --------------------------------------------------------------

        n = 20000          # ambitious length if we have enough time
    else:

    # --------------------------------------------------------------

        else:

        a = simplex_projection(a)

    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
