"""
Propose a non‑negative step function that minimizes the evaluation metric.

Algorithm
---------
We treat the coefficients as a probability distribution (sum == 1).  The
objective  f(a) = 2·n·max_k (a∗a)[k]  is invariant to scaling, therefore we
optimise a on the unit simplex.

*Smooth approximation* –  The non‑smooth max is replaced by a
log‑sum‑exp (soft‑max) with a temperature β:
    gβ(a) = (1/β)·log Σ_k exp(β·(a∗a)[k]).
When β→∞, gβ → max_k (a∗a)[k].

*Gradient* –  For a fixed β the gradient of gβ is
    ∇_a gβ = Σ_k w_k ∂(a∗a)[k]/∂a,
where w_k = softmax(β·(a∗a)[k]).
The derivative of the convolution entry is
    ∂(a∗a)[k]/∂a_i = a_{k-i} + a_{i-k}
(with out‑of‑range indices taken as 0).  Using two FFT‑based
convolutions we obtain the whole gradient in O(n log n).

*Simplex parametrisation* –  We store the variables in the
log‑soft‑max space:  a_i = exp(z_i) / Σ_j exp(z_j).  Updating
z automatically respects non‑negativity and the unit‑sum constraint.
The update direction is
    Δz = a ⊙ (∇_a gβ  –  ⟨∇_a gβ , a⟩).

*Continuation* –  β is increased slowly (β←β·β_factor) so that the
optimization first smooths the landscape and later focuses on the true
maximum.  A simple decaying learning‑rate schedule is used.

The loop runs until the time budget expires (≈ 1000 s).  The best
candidate seen (according to the exact max‑convolution value) is
returned as a plain Python list.
"""

import time
import numpy as np

# Helper utilities – FFT based convolutions
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT of given size."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """Convolution a*b (full) using FFT of given size."""
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)[: a.shape[0] + b.shape[0] - 1]
    return conv

    # Choose a length.  Larger n gives a better theoretical lower bound.
    # ------------------------------------------------------------------
    n = 20000                     # can be changed – 2·10⁴ works comfortably
    size_fft = _next_pow2(3 * n)   # large enough for all convolutions

    # ------------------------------------------------------------------
    # Warm start – use the external sequence if it exists and has the right length.
    # ------------------------------------------------------------------
    try:
        ext = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if ext.size:
            ext = np.maximum(ext, 0.0)
            if ext.sum() < 1e-12:
                ext = np.ones_like(ext)
            ext = ext / ext.sum()
            if ext.shape[0] != n:
                # simple linear interpolation to the required length
                x_old = np.linspace(0, 1, ext.shape[0])
                x_new = np.linspace(0, 1, n)
                ext = np.interp(x_new, x_old, ext)
                ext = ext / ext.sum()
            # initialise z from the warm start
            z = np.log(ext + 1e-12)
            print(f"Warm‑started from height_sequence_1 (length {ext.shape[0]})")
    except NameError:
        # no external seed – start from uniform distribution
        a0 = np.full(n, 1.0 / n, dtype=float)
        z = np.log(a0 + 1e-12)

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    beta = 0.5            # start temperature for the soft‑max
    beta_factor = 1.20    # geometric increase of β
    beta_max = 60.0
    lr = 2.0              # learning rate for the z‑variables
    lr_decay = 0.998      # decay per iteration
    best_score = np.inf
    best_seq = None
    it = 0
    last_improve = 0

    print("=== Gradient‑descent optimisation (smooth max) ===")
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) Recover a from the current logits z (stable soft‑max)
        # --------------------------------------------------------------
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()                     # shape (n,)

        # --------------------------------------------------------------
        # 2) Compute the full self‑convolution (needed for the objective)
        # --------------------------------------------------------------
        conv = _full_conv_fft(a, size_fft)          # length 2n‑1

        # --------------------------------------------------------------
        # 3) Soft‑max weights w_k = exp(β·conv_k) / Σ exp(β·conv)
        # --------------------------------------------------------------
        beta_conv = beta * conv
        max_bc = np.max(beta_conv)
        exp_bc = np.exp(beta_conv - max_bc)
        w = exp_bc / exp_bc.sum()                   # shape (2n‑1,)

        # --------------------------------------------------------------
        # 4) Gradient of the smooth max (see derivation in the docstring)
        # --------------------------------------------------------------
        # term2 = (a * w)  – convolution a * w (no reversal)
        W = np.fft.rfft(w, n=size_fft)
        A = np.fft.rfft(a, n=size_fft)
        conv_aw = np.fft.irfft(A * W, n=size_fft)[: 3 * n - 2]   # a * w
        term2 = conv_aw[:n]                                    # (a*w)[i]

        # term1 = Σ_j a_j w_{i+j}  – use reversed w
        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, n=size_fft)
        conv_aw_rev = np.fft.irfft(A * W_rev, n=size_fft)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)                       # indices for term1
        term1 = conv_aw_rev[idx]

        grad_a = term1 + term2                                 # ∇_a gβ

        # --------------------------------------------------------------
        # 5) Convert to gradient w.r.t. the logits z (simplex‑preserving)
        # --------------------------------------------------------------
        # dL/dz_i = a_i * (grad_a_i - <grad_a, a>)
        inner = np.dot(grad_a, a)
        grad_z = a * (grad_a - inner)

        # --------------------------------------------------------------
        # 6) Gradient step on z, with decaying learning‑rate
        # --------------------------------------------------------------
        z -= lr * grad_z
        lr *= lr_decay

        # --------------------------------------------------------------
        # 7) Update β (increase the focus on the true maximum)
        # --------------------------------------------------------------
        if beta < beta_max:
            beta *= beta_factor

        # --------------------------------------------------------------
        # 8) Track the best exact score seen so far (occurs every few iters)
        # --------------------------------------------------------------
        if it % 15 == 0:
            # conv already computed – its maximum is the actual max‑convolution value
            max_b = conv.max()
            score = 2.0 * n * max_b
            if score < best_score:
                best_score = float(score)
                best_seq = a.copy()
                last_improve = it
                print(f"Iter {it:6d}: β={beta:.2f}, lr={lr:.4e}, "
                      f"approx_score={best_score:.6f}")
                # early stop if we already beat the target
                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    break
            # occasional random nudge if stagnating
            if it - last_improve > 300:
                # small Gaussian kick in logits
                z += 0.01 * np.random.randn(n)
                last_improve = it

    # ------------------------------------------------------------------
    # 9) Final exact evaluation (once) and return the best sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = a.copy()
    try:
        exact = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact:.6f}")
        best_score = float(exact)
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    print("\n=== Search finished ===")
    print(f"Best score (exact) : {best_score:.6f}")
    print(f"Best length        : {len(best_seq)}")
    return best_seq.tolist()
