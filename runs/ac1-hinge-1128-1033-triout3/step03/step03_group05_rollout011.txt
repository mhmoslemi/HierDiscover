"""
Optimising step‑function heights for the L∞ self‑convolution objective.

The programme builds a few analytically‑motivated initial vectors (arcsine
and a grid of symmetric beta‑shapes), evaluates them and keeps the
promising ones.  Each start is then refined with a long Adam optimiser on
a log‑sum‑exp smooth approximation of the max‑convolution.  The gradient
of the smooth objective is computed exactly via FFT using the
cross‑correlation identity
    ∂/∂p_i Σ_k w_k (p*p)_k = 2·∑_j w_{i+j} p_j .
The smoothness parameter α is increased slowly (α≈0.5 → 300) so that the
optimiser finally attacks the true max.  After Adam we perform a short
sub‑gradient descent on the actual maximal entry and finish with a tiny
pairwise‑swap hill‑climber.  All updates are projected onto the
probability simplex (simplex projection).  The routine respects the
overall time budget (default 1000 s) and always returns a valid
non‑negative sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σ x = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast self‑convolution (via FFT) – returns the linear convolution
# of p with itself (length 2·n‑1).
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)
    return conv[: 2 * p.size - 1]


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max (log‑sum‑exp).  Returns the best
# vector and its score.
# ----------------------------------------------------------------------
def _adam_opt(
    p0: np.ndarray,
    n: int,
    m_fft: int,
    deadline: float,
    best_score: float,
    best_p: np.ndarray,
    target: float = 1.503,
) -> tuple[np.ndarray, float]:
    """Run Adam from p0, returning possibly improved (p, score)."""
    p = p0.copy()
    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.03

    # smoothing schedule for log‑sum‑exp
    alpha = 0.5
    alpha_mul = 1.005
    alpha_max = 300.0

    conv_len = 2 * n - 1

    it = 0
    while time.time() < deadline:
        # ----- FFT of current p -----
        p_fft = np.fft.rfft(p, n=m_fft)

        # ----- self‑convolution (real) -----
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[:conv_len]

        # ----- true objective (for bookkeeping) -----
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_p = p.copy()
            print(f"  Adam iter {it:6d} – new best {best_score:.6f}")
            if best_score <= target:
                break

        # ----- smooth max (log‑sum‑exp) -----
        z = alpha * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)        # stabilise
        w = exp_z / np.sum(exp_z)        # soft‑max weights (sum to 1)

        # ----- gradient of the smoothed objective -----
        #   grad_i = 2 * Σ_j w_{i+j} p_j
        w_fft = np.fft.rfft(w, n=m_fft)
        # cross‑correlation via conj(p_fft) * w_fft
        cross = np.fft.irfft(np.conj(p_fft) * w_fft, n=m_fft)[:n]
        grad = 2.0 * cross

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        p = p - step
        p = _simplex_projection(p)

        # ----- increase smoothing -----
        alpha = min(alpha * alpha_mul, alpha_max)

        it += 1

    return best_p, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the actual max entry.
# ----------------------------------------------------------------------
def _subgradient_descent(
    p0: np.ndarray,
    n: int,
    m_fft: int,
    deadline: float,
    best_score: float,
    best_p: np.ndarray,
    target: float = 1.503,
) -> tuple[np.ndarray, float]:
    p = p0.copy()
    step = 0.2
    decay = 0.9995
    min_step = 1e-5
    conv_len = 2 * n - 1

    it = 0
    while time.time() < deadline:
        # self‑convolution
        conv = _conv_self(p, m_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_p = p.copy()
            print(f"  Subgrad iter {it:5d} – new best {best_score:.6f}")
            if best_score <= target:
                break

        max_idx = conv.argmax()

        # sub‑gradient of the maximal entry
        g = np.zeros_like(p)
        # the set of i for which p_i participates in (p*p)_{max_idx}
        start_i = max(0, max_idx - (n - 1))
        end_i = min(n - 1, max_idx)
        i_range = np.arange(start_i, end_i + 1, dtype=int)
        j_range = max_idx - i_range
        g[i_range] = 2.0 * p[j_range]

        # gradient step + projection
        p = p - step * g
        p = _simplex_projection(p)

        step = max(step * decay, min_step)
        it += 1

    return best_p, best_score


# ----------------------------------------------------------------------
# Random tiny swaps (hill‑climber).
# ----------------------------------------------------------------------
def _swap_hillclimb(
    p0: np.ndarray,
    n: int,
    m_fft: int,
    deadline: float,
    best_score: float,
    best_p: np.ndarray,
    target: float = 1.503,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    p = p0.copy()
    cur_score = 2.0 * n * _conv_self(p, m_fft).max()
    attempts = 0
    while time.time() < deadline and attempts < 100_000:
        i, j = rng.integers(0, n, size=2)
        if i == j or p[i] < 1e-12:
            attempts += 1
            continue
        eps_move = 1e-4 * rng.random()
        eps_move = min(eps_move, p[i])  # avoid negativity
        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move
        new_score = 2.0 * n * _conv_self(new_p, m_fft).max()
        if new_score < cur_score:
            p = new_p
            cur_score = new_score
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_p = p.copy()
                print(f"Swap improvement → {best_score:.6f}")
                if best_score <= target:
                    break
        attempts += 1
    return best_p, best_score
