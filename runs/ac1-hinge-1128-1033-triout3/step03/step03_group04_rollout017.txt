   These moves are accepted by a simple simulated‑annealing test.

4. **Re‑heating & decay** – If no improvement happens for a long stretch
   we add a tiny random bump to escape plateaus.  Temperature, Adam learning
   rate and the tolerance for the active set are exponentially decayed.

The routine stops shortly before the 1000 s budget and returns the best
vector found (as a plain Python list).  All numbers are non‑negative, and
the evaluator will renormalise them automatically.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def normalize_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length 2·len(a)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is assumed to sum to 1)."""
    return float(full_convolution_fft(a).max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a distribution with sum = 1."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Parametric families (all return a *sum‑1* vector)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    return a / a.sum()


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    return a / a.sum()


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    return a / a.sum()


# ----------------------------------------------------------------------
# Random sorted families
# ----------------------------------------------------------------------


def random_sorted_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Sample Dirichlet(α) and return the sorted (ascending) version."""
    g = np.random.gamma(concentration, 1.0, size=n)
    a = g / g.sum()
    return np.sort(a)


def random_sorted_uniform(n: int) -> np.ndarray:
    """Uniform random numbers, sorted ascending and normalised."""
    a = np.random.rand(n)
    a = np.sort(a)
    return a / a.sum()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic / stochastic sweep over families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    lengths = [8000, 12000, 16000]               # powers of two are FFT‑friendly
    p_grid = np.arange(0.2, 2.01, 0.4)           # U‑shape exponent
    beta_grid = np.arange(0.5, 5.01, 0.5)        # symmetric beta
    gamma_grid = np.arange(0.003, 0.021, 0.004)  # exponential decay

    print("=== Initial parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # Uniform baseline
        a = np.full(n, 1.0 / n)
        sc = score_from_vector(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f" uniform            n={n:5d}  score={sc:.6f}")

        # U‑shape families
        for p in p_grid:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" u_min (p={p:4.2f})    n={n:5d}  score={sc:.6f}")

        # Beta families (symmetric)
        for a_par in beta_grid:
            a = beta_distribution(n, a_par, a_par)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" beta (α={a_par:4.2f})   n={n:5d}  score={sc:.6f}")

        # Exponential decay families
        for gamma in gamma_grid:
            a = exp_decay_distribution(n, gamma)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" exp (γ={gamma:5.3f})    n={n:5d}  score={sc:.6f}")

        # Random sorted Dirichlet (several draws)
        for concentration in [0.3, 0.8, 2.0, 5.0]:
            for _ in range(30):    # 30 draws per concentration
                a = random_sorted_dirichlet(n, concentration)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f" dirichlet(α={concentration}) n={n:5d}  score={sc:.6f}")

        # Random sorted uniform (several draws)
        for _ in range(30):
            a = random_sorted_uniform(n)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" sorted‑uniform    n={n:5d}  score={sc:.6f}")

    # 2. Include external construction if present
    # ------------------------------------------------------------------

        if external.size > 0:
            external = np.maximum(external, 0.0)
            total = external.sum()
            if total > 0.0:
                external = external / total
                sc_ext = score_from_vector(external)
                if sc_ext < best_score:
                    best_score, best_seq = sc_ext, external.copy()
                    print(f" external height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback if nothing was found (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8000
        best_seq = np.full(n, 1.0 / n)
        best_score = score_from_vector(best_seq)
        print(f" fallback uniform  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement – projected Adam + occasional moves
    # ------------------------------------------------------------------
    current = best_seq.copy()
    current_score = best_score
    n = len(current)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_initial = 0.05                     # base learning rate
    m = np.zeros_like(current)            # first moment
    v = np.zeros_like(current)            # second moment

    temperature = 1.0
    cooling = 0.99985                     # slower cooling for robustness
    rel_tol = 0.01                        # start with a generous active‑set tolerance
    rel_tol_decay = 0.9999                # slow decay towards 1e‑5

    iteration = 0
    last_improve_iter = 0

    print("\n=== Starting stochastic refinement (Adam) ===")
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Convolution & active set
        # --------------------------------------------------------------
        conv = full_convolution_fft(current)
        max_val = conv.max()
        # gradually tighten tolerance
        rel_tol = max(1e-5, rel_tol * rel_tol_decay)
        active_idx = np.where(conv >= max_val * (1.0 - rel_tol))[0]

        # --------------------------------------------------------------
        # Sub‑gradient of the maximal convolution entries
        # --------------------------------------------------------------
        grad = np.zeros_like(current)
        for k in active_idx:
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            i_vals = np.arange(start_i, end_i + 1)
            grad[i_vals] += 2.0 * current[k - i_vals]   # 2 * a_{k-i}

        # --------------------------------------------------------------
        # Adam update (projected onto the simplex)
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr_initial * m_hat / (np.sqrt(v_hat) + eps)
        cand = current - step
        cand = normalize_to_simplex(cand)

        # --------------------------------------------------------------
        # Occasionally try alternative move types
        # --------------------------------------------------------------
        r = np.random.rand()
        if r < 0.12:
            # sorting (ascending) – often reduces the central peak
            cand = np.sort(current)
            cand = normalize_to_simplex(cand)
        elif r < 0.22:
            # small random Gaussian perturbation
            sigma = 0.02 / np.sqrt(iteration + 1)   # diminishing noise
            cand = current + sigma * np.random.randn(n)
            cand = normalize_to_simplex(cand)
        elif r < 0.28:
            # random permutation (keeps the multiset, changes pairings)
            perm = np.random.permutation(n)
            cand = current[perm]
            cand = normalize_to_simplex(cand)
        # else keep Adam candidate

        # --------------------------------------------------------------
        # Evaluate candidate (cheap internal score)
        # --------------------------------------------------------------
        cand_score = score_from_vector(cand)

        # --------------------------------------------------------------
        # Simulated‑annealing acceptance
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            # global best update
            if cand_score < best_score - 1e-12:
                best_score, best_seq = cand_score, cand.copy()
                last_improve_iter = iteration
                print(f"Iter {iteration:7d}: new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # Annealing & decay
        # --------------------------------------------------------------
        temperature *= cooling
        lr_initial *= cooling

        # --------------------------------------------------------------
        # Re‑heat if stuck for a long time
        # --------------------------------------------------------------
        if iteration - last_improve_iter > 60000:
            bump = np.random.rand(n) * 1e-4
            current = normalize_to_simplex(current + bump)
            current_score = score_from_vector(current)
            last_improve_iter = iteration
            temperature = 1.0
            lr_initial = 0.05
            print(f" * Re‑heat at iter {iteration}")

        # occasional progress report
        if iteration % 20000 == 0:
            elapsed = time.time() - start_time
            print(f"[{elapsed:5.1f}s] iter {iteration} – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return best sequence found
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score: {best_score:.6f}  (length {len(best_seq)})")
    # The external evaluator will clip and renormalise again, so we return raw values.
    return best_seq.tolist()
