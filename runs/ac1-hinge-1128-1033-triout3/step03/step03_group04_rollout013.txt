"""
Propose a step‑function (non‑negative vector) with a low value of
evaluate_sequence().  The algorithm consists of three stages:

1.  Parametric sweep.  For several lengths we optimise a few
    one‑parameter families (U‑shapes, beta‑shapes, exponential decay)
    by a cheap bounded scalar optimisation (scipy.optimize.minimize_scalar).
    The best vector seen is kept as a seed.

2.  Stochastic refinement.  Starting from the seed we run a long
    projected‑gradient / mirror‑descent loop.  The sub‑gradient is
    obtained from the (tiny) active set of convolution entries that are
    within a relative tolerance of the current maximum.  The update
    combines three move types:
        – Adam‑style additive Adam update,
        – multiplicative mirror‑descent update,
        – random directed mass‑transfer between a high‑gradient and a
          low‑gradient index.
    Simple simulated‑annealing acceptance and occasional re‑heating are
    employed to escape plateaus.

3.  The best vector encountered before the time budget expires is
    returned.  The vector is normalised to sum 1 – the external
    evaluate_sequence() is scale‑invariant, so any non‑zero scaling would
    give the same score.
"""

import time
import numpy as np
from scipy.optimize import minimize_scalar
from functools import partial

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities (linear convolution)
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length 2*len(a)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # keep only the first 2*n-1 entries (linear convolution length)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is non‑negative, sum=1)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector already normalised."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Parametric families (all return a normalised non‑negative vector)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n-i))^(-p)."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric discrete Beta(α,α) on {0,…,n‑1}: a[i] ∝ (i+1)^{α‑1}(n‑i)^{α‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (alpha - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges: exp(-γ·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring wrappers for scalar optimisation (required by minimise_scalar)
# ----------------------------------------------------------------------


def _score_u_min(p: float, n: int) -> float:
    a = u_min_distribution(n, p)
    return score_from_vector(a)


def _score_u_prod(p: float, n: int) -> float:
    a = u_prod_distribution(n, p)
    return score_from_vector(a)


def _score_beta(alpha: float, n: int) -> float:
    a = beta_distribution(n, alpha)
    return score_from_vector(a)


def _score_exp(logg: float, n: int) -> float:
    """logg is log10(gamma) – optimisation is easier on the log‑scale."""
    gamma = 10 ** logg
    a = exp_decay_distribution(n, gamma)
    return score_from_vector(a)


# ----------------------------------------------------------------------
# Sub‑gradient (active‑set) computation
# ----------------------------------------------------------------------


def compute_subgradient(a: np.ndarray, rel_tol: float = 1e-5) -> np.ndarray:
    """
    Compute a sub‑gradient of max_k (a*a)[k] using the active set.
    rel_tol: relative tolerance for defining the active set.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_val = conv.max()
    active_idx = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    grad = np.zeros_like(a)
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

# Main search routine
# ----------------------------------------------------------------------

            best_score, best_seq = sc, exp_decay_distribution(n, gamma_opt).copy()
            print(f" new best (exp)     n={n:5d} γ={gamma_opt:.6f}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate the external construction if it exists (height_sequence_1)
    # ------------------------------------------------------------------
    try:

            total = external.sum()
            if total > 0.0:

                if sc_ext < best_score:

        best_seq /= best_seq.sum()

        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------

    current_score = best_score
    n = current.shape[0]

    beta2 = 0.999

    while time.time() < deadline:

        # --------------------------------------------------------------

            cand = current - step
            cand = np.maximum(cand, 0.0)
            cand = simplex_projection(cand)

        elif r < 0.70:
            # ---- Mirror‑descent (multiplicative) update ----
            cand = current * np.exp(-lr * grad)
            cand = simplex_projection(cand)

        else:
            # ---- Directed random mass transfer (high‑grad → low‑grad) ----
            top_k = 8
            low_k = 8
            src_indices = np.argpartition(-grad, top_k)[:top_k]
            dst_indices = np.argpartition(grad, low_k)[:low_k]
            src = np.random.choice(src_indices)
            dst = np.random.choice(dst_indices)
            if src != dst and current[src] > 1e-12:
                delta = current[src] * np.random.rand() * 0.5
                cand = current.copy()
                cand[src] -= delta
                cand[dst] += delta
                cand = simplex_projection(cand)
            else:
                # If the random choice does nothing, skip this iteration
                continue

        # --------------------------------------------------------------
        # 4d – evaluate candidate (cheap internal score)
        # --------------------------------------------------------------
        cand_score = score_from_vector(cand)

        # --------------------------------------------------------------
        # 4e – simulated‑annealing acceptance
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            # Update global best if improved enough
            if cand_score < best_score - 1e-12:
                best_score, best_seq = cand_score, cand.copy()
                last_improve_iter = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 4f – anneal temperature and learning‑rate
        # --------------------------------------------------------------
        temperature *= cooling_rate
        lr *= cooling_rate

        # --------------------------------------------------------------
        # 4g – stagnation handling (re‑heat occasionally)
        # --------------------------------------------------------------
        if iteration - last_improve_iter > 60000:
            # Small random bump to escape flat regions
            bump = np.random.rand(n) * 1e-4
            current = current + bump
            current = simplex_projection(current)
            current_score = score_from_vector(current)
            last_improve_iter = iteration
            temperature = 1.0
            lr = base_lr
            print(f" * Re‑heat at iter {iteration}")

    # ------------------------------------------------------------------
    # 5. Return the best vector discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator will clip and normalise again, so we return the raw vector.
    return best_seq.tolist()
