assistantfinal**Solution Overview**

The task is to find a non‑negative step‑function (a probability vector) that
minimises  

\[
\text{score}(a)=2\,n\;\max_k (a*a)[k] \qquad 
\bigl(a\in\mathbb{R}_{\ge0}^n,\;\sum a_i=1\bigr)
\]

where “\(*\)” denotes the full linear convolution.  
The search is performed in two phases:

1. **Deterministic sweep** – a quick exhaustive scan over several parametric
   families (`U‑shape`, `product‑U‑shape`, exponential, edge‑power) for a
   handful of lengths.  
   The best‑scoring vectors are stored as *seeds*.

2. **Stochastic evolutionary refinement** – a small population of vectors is
   evolved under a combination of crossover, Gaussian mutation and a
   handcrafted sub‑gradient step (the exact sub‑gradient of the maximal
   convolution entry).  Elitism guarantees that the current best vector is never
   lost.  The mutation strength decays slowly; when the algorithm stagnates a
   “re‑heat” step re‑initialises the non‑elite part of the population.

All heavy work is done inside `propose_candidate`, which respects the 1000 s
budget.  The implementation only uses `numpy` (and `time` for the budget
control); no external files or network access are required.

```python
"""
propose_candidate
-----------------

Search for a non‑negative vector a (∑a = 1) that minimises

    2 * n * max_k (a * a)[k]

where * denotes full linear convolution.  The algorithm consists of:

1.  A fast deterministic sweep over several parametric families
    (U‑shape, product‑U‑shape, exponential decay, edge‑power) for a few
    lengths.  The best few candidates are kept as seeds.

2.  An evolutionary refinement with a population of size ~30:
    – elitism (the current best individuals survive unchanged),
    – crossover (convex combination of two elites),
    – Gaussian mutation of an elite,
    – occasional exact sub‑gradient step on the best individual,
    – re‑heat when stagnation is detected.

All vectors are kept inside the simplex (non‑negative, sum = 1) by simple
clipping and renormalisation.  The loop runs until the time budget is exhausted
and finally returns the best vector found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level)
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise so that sum(v) == 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform
        v = np.ones_like(v) / v.size
    else:
        v = v / s
    return v


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (length 2*n‑1)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a already normalised vector."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices attaining (or almost attaining) the maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ---------------------------- parametric families ----------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on distance to the nearest edge:
        a[i] = (min(i, n‑1‑i) + 1)^{-p}
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    return simplex_normalise(a)


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on the product of distances to both edges:
        a[i] = ((i+1)*(n‑i))^{-p}
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    return simplex_normalise(a)


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """
    Symmetric exponential decay from the edges:
        a[i] = exp(-lam*i) + exp(-lam*(n‑1‑i))
    """
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    return simplex_normalise(a)


def edge_power_distribution(n: int, p: float) -> np.ndarray:
    """
    Power growth with the distance to the centre:
        a[i] = (|i-(n-1)/2| + 1)^{p}
    Normalised to the simplex.
    """
    i = np.arange(n, dtype=np.float64)
    d = np.abs(i - (n - 1) / 2.0) + 1.0
    a = d ** p + 1e-14
    return simplex_normalise(a)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the given time budget.
    The returned list contains ordinary Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep – keep the K best seeds
    # ------------------------------------------------------------------
    K_SEEDS = 12
    seed_list = []               # (score, vector) tuples, kept sorted ascending

    def try_seed(v):
        sc = score_from_vector(v)
        if len(seed_list) < K_SEEDS:
            seed_list.append((sc, v.copy()))
            seed_list.sort(key=lambda x: x[0])
        else:
            if sc < seed_list[-1][0]:
                seed_list[-1] = (sc, v.copy())
                seed_list.sort(key=lambda x: x[0])

    lengths = [2000, 4000, 6000, 8000, 12000]   # a few representative lengths
    exponents = np.arange(0.1, 6.01, 0.5)       # coarser grid – speed vs quality
    lam_vals = np.logspace(-4, -0.5, num=20)   # exponential decay parameters

    print("=== Deterministic sweep start ===")
    for n in lengths:
        if time.time() > deadline:
            break
        # U‑min family
        for p in exponents:
            a = u_min_distribution(n, p)
            try_seed(a)
        # U‑product family
        for p in exponents:
            a = u_prod_distribution(n, p)
            try_seed(a)
        # exponential family
        for lam in lam_vals:
            a = exp_distribution(n, lam)
            try_seed(a)
        # edge‑power family (high edge, low centre)

            try_seed(a)

        # occasional time check
        if time.time() > deadline:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    else:

    # ------------------------------------------------------------------

    ELITE = 5

    for i in range(POPSIZE):

    # keep global best
    global_best_vec = best_vec.copy()
    global_best_score = best_score
    last_improve_iter = 0
    iteration = 0
    sigma = 0.05            # initial mutation strength (std dev)

    print("\n=== Evolutionary refinement start ===")
    while time.time() < deadline:
        iteration += 1

        # ---- 4a. elitism + generation of offspring ----
        order = np.argsort(scores)
        elite_idx = order[:ELITE]
        new_pop = np.empty_like(population)

            last_improve_iter = iteration

    # ------------------------------------------------------------------
```
