def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------


def exponential_sym_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric exponential: a[i] ∝ exp(-α·min(i, n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-16
    a /= a.sum()
    return a


def exponential_one_sided_distribution(n: int, alpha: float) -> np.ndarray:
    """One‑sided exponential: a[i] ∝ exp(-α·i)."""
    i = np.arange(n)
    a = np.exp(-alpha * i) + 1e-16
    a /= a.sum()
    return a


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = (d + 1) ** (-p) + 1e-16
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-16
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """Product‑power‑law shape: a[i] ∝ ((i+1)*(n‑i))^{‑p}."""
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-16
    a /= a.sum()
    return a

def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # Implementation of the algorithm from
    # https://arxiv.org/pdf/1101.6081.pdf (section 3)
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def contributions_to_max(a: np.ndarray, k_max: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (indices, contributions) of the terms that build up the
    maximal convolution entry at position k_max.
    """
    n = a.shape[0]
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    contrib = a[idx] * a[k_max - idx]
    return idx, contrib

            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external /= s
            approx = 2.0 * external.size * max_conv_fft(external)
            if approx < best_score:
                best_score = approx
                best_seq = external.copy()
                print(f"Seeded from height_sequence_1: approx_score={approx:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 1. Deterministic parametric sweep – refined exponential grid + other families
    # --------------------------------------------------------------
    # length candidates (increase granularity for larger budgets)
    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000, 24000, 30000, 40000]
    # power‑law parameters
    p_vals = np.linspace(0.2, 3.0, 25)

    # exponential target metric (used to place α near the analytical optimum)
    target_metric = 1.503

    print("\n=== Parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # ---- symmetric exponential -------------------------------------------------
        # α ≈ e·T / n ; sweep log‑space around this value
        alpha_opt = np.e * target_metric / n
        factors = np.logspace(np.log10(0.3), np.log10(3.0), 20)
        for f in factors:
            alpha = alpha_opt * f
            # keep α in a sensible interval
            if not (1e-6 <= alpha <= 0.02):
                continue
            a = exponential_sym_distribution(n, alpha)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"New best (sym‑exp): n={n:5d}, α={alpha:.6e}, approx_score={score:.6f}")

        # ---- one‑sided exponential -------------------------------------------------
        alpha_opt_one = np.e * target_metric / n   # same analytical expression works
        for f in factors:
            alpha = alpha_opt_one * f
            if not (1e-6 <= alpha <= 0.02):
                continue
            a = exponential_one_sided_distribution(n, alpha)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"New best (one‑exp): n={n:5d}, α={alpha:.6e}, approx_score={score:.6f}")

        # ---- U‑shaped power‑law ----------------------------------------------------
        for p in p_vals:
            if time.time() > deadline:
                break
            a = u_shaped_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"New best (U‑pow):    n={n:5d}, p={p:4.2f}, approx_score={score:.6f}")

        # ---- sum‑of‑tails ----------------------------------------------------------
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sum_tails_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"New best (tails):    n={n:5d}, p={p:4.2f}, approx_score={score:.6f}")

        # ---- product power‑law ------------------------------------------------------
        for p in p_vals:
            if time.time() > deadline:
                break
            a = product_powerlaw_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"New best (prod‑pow): n={n:5d}, p={p:4.2f}, approx_score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random fallback if nothing succeeded (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = 2.0 * n * max_conv_fft(a)
        print(f"Fallback random init: n={n}, approx_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Targeted stochastic refinement (simulated annealing)
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]

    # full convolution of the current vector (exact)
    conv = full_conv_fft(current)
    conv_max = conv.max()
    k_max = int(conv.argmax())
    current_score = 2.0 * n * conv_max

    # annealing parameters
    temperature = 1.0
    cooling_rate = 0.997
    max_frac = 0.35          # max fraction of source mass moved
    grad_step_factor = 0.04  # mirror‑descent step size factor
    it = 0

    # pre‑compute a long zero‑padded view for fast _incremental_ updates
    pad = np.concatenate([np.zeros(n - 1), current, np.zeros(n - 1)])

    print("\n=== Stochastic refinement ===")
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 3a. Decide which move type to try
        # --------------------------------------------------------------
        if np.random.rand() < 0.14:
            # ---- Gradient (mirror‑descent) step ---------------------------------
            # sub‑gradient of the maximal entry
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            idx_j = k_max - idx_i
            grad = np.zeros_like(current)
            grad[idx_i] = 2.0 * current[idx_j]

            step = grad_step_factor * temperature
            cand = current - step * grad
            cand = np.clip(cand, 0.0, None)
            cand = project_simplex(cand)

            cand_score = 2.0 * n * max_conv_fft(cand)

        else:
            # ---- Targeted random mass‑transfer move -----------------------------
            # Compute contributions to the current maximal entry
            idx_range, contrib = contributions_to_max(current, k_max)
            if contrib.sum() == 0.0:
                # fallback to a plain random source if something degenerate
                src = np.random.randint(0, n)
            else:
                prob_src = contrib / contrib.sum()
                src = np.random.choice(idx_range, p=prob_src)

            # Destination – any other index (uniform)
            dst = np.random.randint(0, n - 1)
            if dst >= src:
                dst += 1

            # Transfer a random fraction of the source mass
            delta = current[src] * np.random.rand() * max_frac
            if delta <= 1e-15:
                continue  # nothing to do

            # Incremental update of the convolution peak
            offset_src = n - 1 - src
            offset_dst = n - 1 - dst
            shift_src = pad[offset_src: offset_src + 2 * n - 1]
            shift_dst = pad[offset_dst: offset_dst + 2 * n - 1]

            # new maximal entry after the move (linearised)
            new_max = np.max(conv + 2.0 * delta * (shift_dst - shift_src))
            cand_score = 2.0 * n * new_max

            # Apply the move only after the acceptance decision
            apply_move = True

        # --------------------------------------------------------------
        # 3b. Acceptance decision (SA)
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if apply_move:
                # perform the selected mass transfer
                current[src] -= delta
                current[dst] += delta
                # keep the vector on the simplex (numerical drift)
                current = project_simplex(current)
                # recompute full convolution and related info
                conv = full_conv_fft(current)
                conv_max = conv.max()
                k_max = int(conv.argmax())
                current_score = 2.0 * n * conv_max
                # update the padded view used for incremental updates
                pad = np.concatenate([np.zeros(n - 1), current, np.zeros(n - 1)])
            else:
                # gradient move – we already have a full candidate array
                current = cand
                conv = full_conv_fft(current)
                conv_max = conv.max()
                k_max = int(conv.argmax())
                current_score = cand_score
                pad = np.concatenate([np.zeros(n - 1), current, np.zeros(n - 1)])

            # Keep the global best
            if current_score < best_score:
                best_score = current_score
                best_seq = current.copy()
                print(f"Iter {it:6d}: new global best ≈ {best_score:.6f}")
                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    break

        # --------------------------------------------------------------
        # 3c. Temperature cooling
        # --------------------------------------------------------------
        temperature = max(temperature * cooling_rate, 1e-6)

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (once) and report
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())  # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best (approx.) score : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()
