def simplex_projection(v):
    """projects onto simplex {x: x>=0, sum x = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho: largest i s.t. u_i > cssv_i / (i+1)
    rho = np.where(u > cssv / np.arange(1, n+1))[0]
    if len(rho) == 0:
        # all zeros
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient descent with smooth max objective for minimizing 2*n*max(conv(a,a)).
"""

import time
import numpy as np

def max_conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())

def full_conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv

def proj_simplex(v):
    # Duchi projection onto probability simplex
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def grad_smoothed(v, beta):
    """Gradient of (1/beta) log sum exp(beta * (v*v conv))."""
    n = v.shape[0]
    # compute convolution
    conv = full_conv_fft(v) # length 2n-1
    max_c = conv.max()
    # softmax weights (stable)
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    # reverse w for convolution
    w_rev = w[::-1]
    # compute convolution of v with w_rev
    size = 1 << ((n + (2 * n - 1) - 1).bit_length())
    V = np.fft.rfft(v, n=size)
    W = np.fft.rfft(w_rev, n=size)
    grad_full = np.fft.irfft(V * W, n=size)
    # slice part n-1 to 2n-2 (inclusive)
    slice_part = grad_full[n-1:2 * n - 1]  # length n
    grad = 2.0 * slice_part[::-1]
    return grad

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Gradient‑descent search for a non‑negative step function minimizing
    2·n·max(conv(a,a)).  The objective is smoothed with a log‑sum‑exp
    barrier and each gradient step is projected onto the probability simplex.
    The routine respects a time budget, falls back to a random start if no
    external sequence is supplied, and returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Initialise from external variable if present, otherwise random.
    # ------------------------------------------------------------------
    try:
        a0 = np.array(height_sequence_1, dtype=float)  # noqa: F821
        a0 = np.maximum(a0, 0.0)
        if a0.size == 0:
            raise ValueError
        s = a0.sum()
        if s < 1e-12:
            a0 = np.ones_like(a0) / a0.size
            s = 1.0
        a0 = a0 / s
        print(f"Starting from external height_sequence_1, n={len(a0)}")
    except Exception:
        n = 8000
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum()
        print(f"No external sequence, random start, n={n}")

    n = len(a0)
    current = a0.copy()

    # compute initial score (approx)
    conv_curr = full_conv_fft(current)
    cur_max = conv_curr.max()
    current_score = 2.0 * n * cur_max
    best_seq = current.copy()
    best_score = current_score
    print(f"Initial approx. score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optimization hyper‑parameters
    # ------------------------------------------------------------------
    beta = 5.0
    beta_target = 200.0
    # we will increase beta exponentially; after about 1500 iterations reach target
    beta_growth = (beta_target / beta) ** (1 / 1500.0)

    step0 = 0.2          # initial step size
    step_decay = 0.9995  # geometric decay

    iteration = 0
    print("\n=== Begin smoothed gradient descent ===")
    while time.time() < deadline:
        iteration += 1

        # gradient of smooth max
        grad = grad_smoothed(current, beta)

        # step size
        step = step0 * (step_decay ** iteration)

        # trial move
        cand = current - step * grad
        cand = proj_simplex(cand)

        # evaluate approx score
        conv_cand = full_conv_fft(cand)
        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        # accept if improvement, otherwise maybe accept with small prob.
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # simple simulated‑annealing acceptance using current step as temperature
            temp = max(step, 1e-12)

        exact = best_score

    print("\n=== Search finished ===")
    print(f"Best approx. score: {best_score:.6f}")
    print(f"Best length: {len(best_seq)}")
    return best_seq.tolist()

"""
Gradient‑descent search for a non‑negative step function minimising
    2·n·max(conv(a,a))

The objective is smoothed with a log‑sum‑exp barrier so that its gradient
is well defined.  Each gradient step is projected onto the probability
simplex (non‑negative entries summing to 1).  The smoothing parameter β is
progressively increased, driving the solution towards minimising the true
sup‑norm of the auto‑convolution.  The routine respects a 1000 s time
budget, falls back to a random initial vector if no external seed is
available, and finally evaluates the best candidate with the provided
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the simplex {x ≥ 0, sum(x) = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # In the degenerate case return the uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth approximation of the sup‑norm
# ----------------------------------------------------------------------


def grad_smoothed(v: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ_k exp(β·(v*v conv)[k]).
    Uses FFTs to evaluate both the convolution and the gradient efficiently.
    """
    n = v.shape[0]

    # 1) self‑convolution
    conv = full_conv_fft(v)                     # length 2n‑1
    max_c = conv.max()

    # 2) softmax weights (stable version)
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()               # length 2n‑1

    # 3) gradient = 2·(v * w_rev) (reversed slice)
    w_rev = w[::-1]
    size = 1 << ((n + (2 * n - 1) - 1).bit_length())   # ≥ 3n‑2
    V = np.fft.rfft(v, n=size)
    W = np.fft.rfft(w_rev, n=size)
    grad_full = np.fft.irfft(V * W, n=size)
    # slice indices n‑1 … 2n‑2 (inclusive) → length n, then reverse
    slice_part = grad_full[n - 1 : 2 * n - 1]
    grad = 2.0 * slice_part[::-1]
    return grad

# ----------------------------------------------------------------------
