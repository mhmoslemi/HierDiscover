import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    conv_fg = cp.conv(f, g)  # full convolution length 2*n-1
    constraints = [conv_fg <= M]
    # sum(g) is objective
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    g_opt = g.value
    return g_opt
```

def smoothed_objective_and_grad(w, eps):
    # compute a = softmax(w)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum
    # compute full convolution via FFT
    conv = _full_convolution(a)  # length 2n-1
    max_conv = conv.max()
    # compute softmax probs p over conv
    shifted = (conv - max_conv) / eps
    # Clip to avoid overflow; compute exp
    weights = np.exp(shifted)
    p = weights / weights.sum()
    # gradient wrt a
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)
    segment = conv_ab[n-1 : 2*n-1]
    grad_a = 4.0 * n * segment[::-1]
    # gradient wrt w
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
* Represent the step heights with a soft‑max:  a_i = exp(w_i)/∑exp(w).
  This guarantees a_i ≥ 0 and Σ a_i = 1 (the evaluation formula is
  scale‑invariant).
* Minimise a smooth approximation of the max‑convolution:
        L_ε(a) = 2·n·ε·log(∑_k exp(conv_k/ε))
  The gradient ∂L/∂a is known in closed form and can be evaluated with
  two FFT‑based convolutions.
* Optimise the smoothed objective with a sequence of L‑BFGS‑B solves,
  gradually decreasing ε (continuation method).  The L‑BFGS‑B solver
  uses the analytical gradient, so only a few hundred gradient
  evaluations are required per ε‑stage.
* After the continuation phase a cheap hill‑climbing refinement
  (random mass transfers) is performed until the time budget expires.
* The best exact value seen (via ``evaluate_sequence``) is returned.

The whole procedure runs under the 1000 s budget and prints progress.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective and its gradient (w.r.t. the soft‑max parameters)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Return (L_ε, ∂L_ε/∂w) for a given parameter vector w and smoothing ε.
    The gradient is computed by the chain rule through the soft‑max.
    """
    # ------- soft‑max (feasible a, Σa = 1) ----------
    w_shift = w - np.max(w)          # numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                    # a_i ≥ 0, Σa_i = 1

    # ------- full auto‑convolution a*a ------------
    conv = _full_convolution(a)      # length 2·n‑1
    max_conv = conv.max()

    # ------- smoothed max (log‑sum‑exp) ----------
    # L = 2·n·ε·log(∑_k exp(conv_k/ε))
    # we use the shifted formulation to avoid overflow.
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)     # = log∑_k exp((conv_k‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ------- gradient w.r.t. a (closed form) -----
    # p_k = exp(conv_k/ε) / Σ_j exp(conv_j/ε)
    # we already have shifted = (conv‑max)/ε, so
    # exp(conv/ε) = exp(max/ε) * exp(shifted); the constant cancels.
    weights = np.exp(shifted)
    p = weights / weights.sum()      # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}  (indices outside [0,n‑1] give 0)
    p_rev = p[::-1]                           # reversed p
    conv_ap = _conv_fft(a, p_rev)              # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]            # middle n entries
    grad_a = 4.0 * n * seg[::-1]               # shape (n,)

    # ------- chain rule (soft‑max) ----------------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)                # ∂L/∂w

    return L, grad_w


# ----------------------------------------------------------------------
# Hill‑climbing refinement (random mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """Small random perturbations that keep Σa constant."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a fraction of mass from i to j
        delta = cur_a[i] * 0.05 * rng.random()
        if delta == 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # evaluate with the official function (exact objective)

            val = evaluate_sequence(new_a.tolist())
        except Exception:               # safety – ignore bad candidates
            continue
        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val
