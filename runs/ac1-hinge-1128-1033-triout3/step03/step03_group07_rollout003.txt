def random_transfer(a, conv, kmax, max_frac):
    n = len(a)
    w = np.zeros(n)
    for i in range(max(0, kmax - (n - 1)), min(kmax, n - 1) + 1):
        j = kmax - i
        w[i] = a[i] * a[j]
    total_w = w.sum()
    if total_w == 0:
        return a, conv, False
    src = np.random.choice(n, p=w / total_w)
    # Choose destination uniformly among indices where shift reduces conv like maybe where w is minimal.
    # simply random not equal src.
    dst = src
    while dst == src:
        dst = np.random.randint(0, n)
    delta = a[src] * np.random.rand() * max_frac
    if delta <= 0:
        return a, conv, False
    b = a.copy()
    b[src] -= delta
    b[dst] += delta
    # ensure nonneg current
    if b[src] < 0:
        b[src] = 0.0
    # renorm
    s = b.sum()
    if s > 0:
        b /= s
    # compute new conv
    conv_new = full_conv_fft(b)
    new_score = 2.0 * n * conv_new.max()
    return b, conv_new, new_score

Propose a non‑negative step function that minimizes the given evaluation metric.

Algorithm (≈5 min total runtime)
---------------------------------
1. **Warm‑start** – If a global variable ``height_sequence_1`` exists,
   evaluate it and keep the best one.

2. **Beta‑family sweep** –  The continuous optimal shape for the problem is
   the arcsine density (Beta(½,½)) because the sum of two independent
   arcsine variables is exactly uniform, i.e. the self‑convolution is flat.
   We therefore generate symmetric Beta‑distributed vectors
   a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1} for several (α,β) pairs (including the
   optimal (½,½)) and for a few lengths (2000…12000).  The best candidate
   according to the fast‐FFT estimate 2·n·max(conv) is kept.

3. **Local stochastic refinement** –  Starting from the best candidate we
   run a simulated‑annealing / gradient‑descent loop.  In each iteration
   we (a) identify the index *k* where the convolution attains its maximum,
   (b) either perform a small gradient‐step (a ← a – η·∇_k a) or a random
   mass‑transfer move that moves a fraction of the mass from an index that
   contributes heavily to the peak to a random destination.
   Moves are accepted if they improve the score or with a Metropolis
   probability that decays with temperature.  The temperature is cooled
   geometrically; the loop stops when the time budget is exhausted or when
   a score below the target 1.503 is found.

4. **Exact evaluation** –  The best vector found is evaluated with the
   official ``evaluate_sequence`` routine and returned.

All helper functions are defined at module level; no I/O or external

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, zero‑padded self‑convolution)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Parametric families (all return a normalized vector with sum == 1)
# ----------------------------------------------------------------------


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric (or not) Beta density discretised on n points.
    a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1}.
    """
    # avoid overflow for extreme exponents by adding a tiny epsilon
    eps = 1e-12
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta - 1.0))
    # Prevent all‑zero vectors if a bad exponent is chosen
    if a.sum() == 0.0:
        a = np.ones_like(a)
    a /= a.sum()
    return a


def arcsine_distribution(n: int) -> np.ndarray:
    """Beta(½,½) – the continuous optimal shape (arcsine law)."""
    return beta_distribution(n, 0.5, 0.5)

    (negative gradient) and the vector is kept on the simplex.

    i_start = max(0, kmax - (n - 1))
    i_end = min(n - 1, kmax)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    idx_j = kmax - idx_i
    grad[idx_i] = 2.0 * a[idx_j]
    cand = a - step * grad
    cand = np.clip(cand, 0.0, None)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    else:
        cand = a.copy()
    return cand


def random_mass_transfer(a: np.ndarray, kmax: int, max_frac: float) -> tuple:
    """
    Propose a move that transfers a random fraction of mass from an index
    that contributes heavily to the current worst convolution entry (kmax)
    to a uniformly chosen destination.
    Returns (candidate_vector, new_score) where new_score is the
    2·n·max(conv) value (computed exactly via FFT).  If no valid move can
    be generated, the original vector and an infinite score are returned.
    """
    n = a.shape[0]
    # contribution weights w_i = a_i * a_{kmax-i} (zero where out‑of‑range)
    w = np.zeros_like(a)
    i_start = max(0, kmax - (n - 1))
    i_end = min(n - 1, kmax)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    j_range = kmax - i_range
    w[i_range] = a[i_range] * a[j_range]

    total_w = w.sum()
    if total_w == 0.0:
        return a.copy(), np.inf

    src = np.random.choice(n, p=w / total_w)
    if a[src] <= 0.0:
        return a.copy(), np.inf

    # choose a destination different from src
    dst = src
    while dst == src:
        dst = np.random.randint(0, n)

    delta = a[src] * np.random.rand() * max_frac
    if delta <= 0.0:
        return a.copy(), np.inf

    cand = a.copy()
    cand[src] -= delta
    cand[dst] += delta
    # renormalise onto the simplex
    s = cand.sum()
    if s > 0.0:
        cand /= s
    else:
        return a.copy(), np.inf

    # exact new score
    conv = full_conv_fft(cand)
    new_score = 2.0 * n * conv.max()
    return cand, new_score


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step function that minimises the evaluation
    metric.  Returns a plain Python list of floats suitable for
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0. Warm‑start from an externally supplied sequence (if any)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)  # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external /= s
            approx = 2.0 * external.size * max_conv_fft(external)
            best_seq = external.copy()
            best_score = approx
            print(f"Seeded from height_sequence_1 – approx_score={approx:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over symmetric Beta families (incl. arcsine)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000, 12000]
    # symmetric alphas (beta = alpha) – 0.2 .. 0.9 (includes 0.5)
    alphas = np.linspace(0.2, 0.9, 15)

    print("\n=== Beta‑family sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break
        for a_val in alphas:
            if time.time() > deadline:
                break
            a = beta_distribution(n, a_val, a_val)
            approx = 2.0 * n * max_conv_fft(a)
            if approx < best_score:
                best_score, best_seq = approx, a.copy()
                print(f"New best (beta={a_val:.3f}) – n={n:5d} – approx={approx:.6f}")

        # also test the exact arcsine shape (Beta(0.5,0.5)) if not already covered
        if time.time() > deadline:
            break
        a = arcsine_distribution(n)
        approx = 2.0 * n * max_conv_fft(a)
        if approx < best_score:
            best_score, best_seq = approx, a.copy()
            print(f"New best (arcsine) – n={n:5d} – approx={approx:.6f}")

    # ------------------------------------------------------------------
    # 2. Fallback random initialisation (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = 2.0 * n * max_conv_fft(a)
        print(f"Fallback random init – n={n} – approx={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local stochastic refinement
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]
    conv = full_conv_fft(current)
    conv_max = conv.max()
    current_score = 2.0 * n * conv_max

    temperature = 1.0
    cooling_rate = 0.998
    max_frac = 0.30          # up to 30 % of source mass may be moved
    grad_step_factor = 0.04  # fraction of temperature used for gradient steps
    it = 0

    print("\n=== Stochastic refinement ===")
    while time.time() < deadline:
        it += 1

        # recompute current metric (in case we accepted a move without updating)
        current_score = 2.0 * n * conv_max

        # record global best
        if current_score < best_score:
            best_score = current_score
            best_seq = current.copy()
            print(f"Iter {it:6d}: new global best = {best_score:.6f}")
            if best_score < 1.503:
                print("Target reached – exiting early.")
                break

        # identify the worst convolution position
        kmax = int(conv.argmax())

        # Choose move type
        if np.random.rand() < 0.25:
            # ----- Gradient step -----
            step = grad_step_factor * temperature
            cand = gradient_step(current, kmax, step)
            cand_score = 2.0 * n * max_conv_fft(cand)
        else:
            # ----- Random mass‑transfer move -----
            cand, cand_score = random_mass_transfer(current, kmax, max_frac)

        # ---- Simulated‑annealing acceptance ----
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # Metropolis rule (temperature never reaches zero)
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            # recompute full convolution for the accepted vector
            conv = full_conv_fft(current)
            conv_max = conv.max()
        # temperature decay
        temperature = max(temperature * cooling_rate, 1e-8)

    # ------------------------------------------------------------------
    # 4. Exact evaluation (once) and final report
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()
