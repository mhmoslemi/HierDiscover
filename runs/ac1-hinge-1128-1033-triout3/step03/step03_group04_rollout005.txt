"""
Improved search for a low‑score non‑negative step function.

The algorithm consists of three stages:

1. **Deterministic parametric sweep** – explore several families
   (U‑shapes, beta, exponential decay, mixed power‑laws) on a handful
   of moderate lengths.  The best ~5 candidates are retained.

2. **Adam‑style projected optimisation** – starting from the best
   candidate we run a large number of iterations of a stochastic
   sub‑gradient descent with per‑coordinate adaptive step sizes
   (Adam).  After each update the iterate is projected onto the
   probability simplex.  A tiny simulated‑annealing acceptance rule
   helps escape plateaus, and an occasional “re‑heat” adds random
   jitter.

3. **Upsampling refinement** – if time permits the current best
   vector is interpolated to larger lengths and refined again,
   giving the algorithm a chance to exploit the extra freedom of a
   finer discretisation.

All convolutions are performed with a single FFT, making a single
iteration fast enough to explore millions of candidates within the
1000 s time budget.  The returned list contains raw non‑negative
values (the external checker will normalise and clip them again).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length 2*len(a)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is non‑negative, sum=1)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a vector already normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * max_convolution(a))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge: (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the two edges: ((i+1)*(n-i))^(-p)."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}.  a[i] ∝ (i+1)^{α‑1} (n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges: exp(-γ·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def powerlaw_edge_distribution(n: int, p_left: float, p_right: float) -> np.ndarray:
    """Edge‑biased power law: a[i] ∝ (i+1)^{-p_left}·(n‑i)^{-p_right}."""
    i = np.arange(n)
    a = (i + 1) ** (-p_left) * (n - i) ** (-p_right) + 1e-14
    a /= a.sum()
    return a


def mixture_u_min(n: int, p1: float, p2: float, w: float) -> np.ndarray:
    """Convex mixture of two U‑min shapes."""
    a1 = u_min_distribution(n, p1)
    a2 = u_min_distribution(n, p2)
    a = w * a1 + (1.0 - w) * a2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def subgradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int = 3) -> np.ndarray:
    """
    Sub‑gradient of max_{k} conv_k (where conv = a*a) obtained from the
    `top_k` largest convolution entries.
    """
    n = a.shape[0]
    if top_k >= conv.size:
        indices = np.arange(conv.size)
    else:
        # indices of the `top_k` largest elements
        part = np.argpartition(-conv, top_k - 1)
        indices = part[:top_k]

    grad = np.zeros_like(a)
    for k in indices:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector onto a larger length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total > 0.0:
        new_seq /= total
    return new_seq


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (a small safety margin is kept).

    Returns
    -------
    list[float]
        The best sequence discovered (raw, non‑negative, not required
        to sum to any particular value – the external checker will
        renormalise it).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # keep a few seconds margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over parametric families
    # ------------------------------------------------------------------
    best_score = np.inf
    best_seq = None
    best_n = None

    # store a small set of elite candidates (length, vector, score)
    elite = []          # each entry is (score, n, vector)
    elite_capacity = 5

    # lengths to explore (moderate sizes – large enough to exhibit the
    # asymptotic behaviour but cheap enough for the sweep)
    lengths = [2000, 4000, 6000, 8000, 10000, 12000, 16000, 20000]

    print("=== Deterministic parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # ------------------------------------------------------------------
        # Simple families with grid search
        # ------------------------------------------------------------------
        # U‑min and U‑prod (p from 0.05 to 5.0)
        p_grid = np.arange(0.05, 5.01, 0.15)
        for p in p_grid:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a.copy(), n
                print(f" new best (u_min) n={n:5d} p={p:4.2f} score={sc:.6f}")
            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a.copy(), n
                print(f" new best (u_prod) n={n:5d} p={p:4.2f} score={sc:.6f}")

        # Beta (symmetric)
        alpha_grid = np.arange(0.2, 5.01, 0.2)
        for alpha in alpha_grid:
            a = beta_distribution(n, alpha, alpha)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a.copy(), n
                print(f" new best (beta) n={n:5d} α={alpha:4.2f} score={sc:.6f}")

        # Exponential decay
        gamma_grid = np.arange(0.001, 0.041, 0.002)
        for gamma in gamma_grid:
            a = exp_decay_distribution(n, gamma)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a.copy(), n
                print(f" new best (exp) n={n:5d} γ={gamma:5.3f} score={sc:.6f}")

        # ------------------------------------------------------------------
        # Randomised families (few samples to diversify the pool)
        # ------------------------------------------------------------------
        # Mixed power‑law edge distribution
        for _ in range(30):
            p_left = np.random.uniform(0.05, 5.0)
            p_right = np.random.uniform(0.05, 5.0)
            a = powerlaw_edge_distribution(n, p_left, p_right)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a.copy(), n
                print(f" new best (edge) n={n:5d} pl={p_left:4.2f} pr={p_right:4.2f} score={sc:.6f}")

        # Mixture of two U‑min shapes
        for _ in range(30):
            p1 = np.random.uniform(0.05, 5.0)
            p2 = np.random.uniform(0.05, 5.0)
            w = np.random.rand()
            a = mixture_u_min(n, p1, p2, w)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a.copy(), n
                print(f" new best (mix) n={n:5d} p1={p1:4.2f} p2={p2:4.2f} w={w:4.2f} score={sc:.6f}")

        # keep a few elite candidates (simple insertion)
        # we push the current candidate if it's among the top few seen so far
        # for speed we only consider the best candidate of this length (best_seq)
        # and maybe a few random ones
        # Here we simply keep the global best and a few randomly sampled good ones
        if len(elite) < elite_capacity:
            elite.append((best_score, best_n, best_seq.copy()))
        else:
            # replace worst if current is better
            worst_idx = np.argmax([e[0] for e in elite])
            if best_score < elite[worst_idx][0]:
                elite[worst_idx] = (best_score, best_n, best_seq.copy())

    # ------------------------------------------------------------------
    # 2. Incorporate external construction if present
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            total = external.sum()
            if total > 0.0:
                external /= total
                sc_ext = score_from_vector(external)
                if sc_ext < best_score:
                    best_score, best_seq, best_n = sc_ext, external.copy(), external.shape[0]
                    print(f" using external height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback random seed (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_n = n
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam‑style projected refinement on the elite candidates
    # ------------------------------------------------------------------
    # Hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.08
    cooling_rate = 0.9997
    temperature = 1.0
    temp_cool = 0.9999
    rel_topk = 3                # how many of the largest convolution entries to use
    reheat_iter = 40000
    jitter_scale = 1e-4

    # helper to run a refinement session on a given start vector
    def refine_adam(start_vec: np.ndarray, n: int, time_left: float):
        """
        Refine `start_vec` (already normalised to sum=1) using Adam
        and return the best vector found together with its score.
        """
        a = start_vec.copy()
        best_local = a.copy()
        best_local_score = score_from_vector(a)

        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        last_improve = 0

        while time.time() + 0.001 < deadline:   # tiny safety margin
            t += 1
            # compute convolution once
            conv = full_convolution_fft(a)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val

            # sub‑gradient from the top‑k largest entries
            grad = subgradient_topk(a, conv, top_k=rel_topk)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = simplex_projection(cand)

            # evaluate candidate
            conv_cand = full_convolution_fft(cand)
            cand_score = 2.0 * n * conv_cand.max()

            # simulated‑annealing acceptance
            accept = False
            if cand_score < cur_score:
                accept = True
            else:
                prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                a = cand
                cur_score = cand_score
                if cand_score < best_local_score - 1e-12:
                    best_local_score = cand_score
                    best_local = cand.copy()
                    last_improve = t
                    print(f"  [refine] iter {t:7d}  new local best = {best_local_score:.6f}")

            # anneal temperature and learning‑rate.
            temperature *= temp_cool
            base_lr *= cooling_rate

            # occasional re‑heat if we have been stagnant for long
            if t - last_improve > reheat_iter:
                noise = np.random.rand(n) * jitter_scale
                a = a + noise
                a = simplex_projection(a)
                m[:] = 0.0
                v[:] = 0.0
                t = 0
                temperature = 1.0
                base_lr = 0.08
                last_improve = 0
                print(f"   ** Re‑heat at iter {t}")

            # break if time is about to run out
            if time.time() > deadline - 2.0:
                break

        return best_local, best_local_score

    # ------------------------------------------------------------------
    # Run refinement on each elite candidate (most promising first)
    # ------------------------------------------------------------------
    # sort elite by score (ascending)
    elite_sorted = sorted(elite, key=lambda x: x[0])
    print("\n=== Adam refinement phase ===")
    for idx, (sc, n, vec) in enumerate(elite_sorted):
        if time.time() > deadline - 10.0:
            break
        print(f" Refine elite #{idx+1} (len={n}, start score={sc:.6f})")
        refined_vec, refined_score = refine_adam(vec, n, deadline - time.time())
        if refined_score < best_score - 1e-12:
            best_score, best_seq, best_n = refined_score, refined_vec.copy(), n
            print(f"  *** Updated global best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upsampling to larger lengths (if we still have time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        larger_lengths = [ln for ln in lengths if ln > best_n]
        for new_n in larger_lengths:
            if time.time() > deadline - 5.0:
                break
            print(f"\n Upsampling to length {new_n}")
            up_seq = upsample_sequence(best_seq, new_n)
            up_score = score_from_vector(up_seq)
            print(f"  initial upsampled score = {up_score:.6f}")
            # a short refinement (few thousand iterations) on the larger size
            refined_up, refined_up_score = refine_adam(up_seq, new_n, deadline - time.time())
            if refined_up_score < best_score - 1e-12:
                best_score, best_seq, best_n = refined_up_score, refined_up.copy(), new_n
                print(f"  *** New global best after upsampling: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Return the best sequence discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {best_n})")
    # The external evaluator will clip and renormalise again,
    # so we can return the raw probability vector.
    return best_seq.tolist()
