"""
Search for a non‑negative step‑function (probability vector) that minimizes

    f(a) = 2·n·max(a*a) / (sum(a))²

The objective is scale‑invariant, so we restrict the sum of the vector to 1.
The algorithm combines:

* a set of simple analytical initialisations (U‑shape, exponential, random spikes);
* an Adam optimiser applied to the smooth approximation
      Sα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)   of the ‖a*a‖∞ term;
* an increasing smoothing parameter α (soft‑max → hard max);
* projection onto the probability simplex after each update;
* a cheap stochastic polishing phase after the gradient phase.

The whole search respects a configurable time budget (default 1000 s) and
returns the best sequence found.  All helper functions are top‑level, no
closures or file I/O are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (real FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a given FFT size."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# 2.  Soft‑max weights for a smooth ∞‑norm approximation
# ----------------------------------------------------------------------
def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_k = exp(α·conv_k) / Σ exp, numerically stabilized."""
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


# ----------------------------------------------------------------------
# 3.  Gradient of Σ w_k·conv_k with respect to the base vector a
# ----------------------------------------------------------------------
def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k·(a*a)_k using FFTs.
    a : length n, w : length 2·n‑1 (weights, sum to 1), size ≥ 3·n‑2.
    """
    n = a.shape[0]
    L = 2 * n - 1                       # length of w
    # FFT of a
    A = np.fft.rfft(a, n=size)

    # FFT of w and its reversal
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # a * w  (length n+L‑1 = 3·n‑2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    # Gradient = conv_aw[:n] + conv_aw_rev[L‑1‑np.arange(n)]
    idx = L - 1 - np.arange(n)
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


# ----------------------------------------------------------------------
# 4.  Simple geometric helpers
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto {x | x_i ≥ 0, Σ x_i = 1}.
    O(n log n) algorithm (Wang & Carreira‑Perpiñán, 2013).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new += 1e-14                     # avoid exact zeros
    new /= new.sum()
    return new


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law “U” shape (larger p ⇒ more weight at the edges)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)               # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exponential_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    x = np.arange(n)
    a = np.exp(-lam * x) + np.exp(-lam * (n - 1 - x))
    a += 1e-14
    a /= a.sum()
    return a


def random_spike_distribution(n: int, k: int = 12, sigma: float = None) -> np.ndarray:
    """
    A few Gaussian‑like spikes placed at random locations.
    k   – number of spikes,
    sigma – width of each spike (defaults to n/400, at least 5.0).
    """
    if sigma is None:
        sigma = max(5.0, n / 400.0)
    positions = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n)
    x = np.arange(n, dtype=float)
    for pos in positions:
        weight = np.random.rand() + 0.1
        a += weight * np.exp(-0.5 * ((x - pos) / sigma) ** 2)
    a += 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 5.  Optimisation core (Adam on the soft‑max of the convolution)
# ----------------------------------------------------------------------
def adam_optimize(a_init: np.ndarray,
                  n: int,
                  size: int,
                  deadline: float,
                  max_iters: int = 6000,
                  lr: float = 0.2,
                  alpha_start: float = 2.0,
                  alpha_max: float = 2000.0,
                  alpha_mul: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth max‑convolution objective.
    Returns the best vector seen and its (approximate) score 2·n·max(conv).
    """
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    alpha = alpha_start
    best_score = np.inf
    best_a = a.copy()

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and score
        conv = full_conv_fft(a, size)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max of the convolution (smooth approx. to the max)
        w = softmax_weights(conv, alpha)

        # gradient of Σ w_k·conv_k w.r.t. a
        grad = gradient_softmax(a, w, size)

        # Adam update (gradient descent direction)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(v_hat) + eps)        # per‑coordinate step
        a = a - step * m_hat

        # enforce feasibility (non‑negative + sum‑to‑1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = a_init.copy()

        # tighten the soft‑max
        alpha = min(alpha * alpha_mul, alpha_max)

        # occasional progress report
        if t % 1000 == 0:
            print(f"    n={n:6d}  iter={t:5d}  cur≈{cur_score:.6f}  best≈{best_score:.6f}  α={alpha:.1f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 6.  Light polishing – random local moves that only keep improvements
# ----------------------------------------------------------------------
def polish_sequence(a: np.ndarray,
                    n: int,
                    size: int,
                    deadline: float,
                    max_iters: int = 2000,
                    eps_move: float = 5e-5) -> tuple[np.ndarray, float]:
    """
    Small stochastic hill‑climbing to further reduce max(conv).
    Moves a tiny amount of mass from one index to another.
    """
    best_a = a.copy()
    best_score = 2.0 * n * max_conv_fft(best_a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most eps_move or the whole mass at i
        move = min(eps_move, best_a[i])
        if move <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= move
        cand[j] += move
        # keep feasibility (sum unchanged, non‑negative)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        cand_score = 2.0 * n * max_conv_fft(cand)
        if cand_score < best_score:
            best_a = cand
            best_score = cand_score
    return best_a, best_score


# ----------------------------------------------------------------------
# 7.  Main entry point – sequence proposal
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects a time budget (seconds) and
    returns the best list of floats it found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 0.  Load external seed if provided
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            external_sum = external.sum()
            if external_sum < 1e-12:
                external = np.ones_like(external) / external.size
                external_sum = 1.0
            external = external / external_sum
            cur_score = 2.0 * external.size * max_conv_fft(external)
            best_seq = external.copy()
            best_score = cur_score
            print(f"Seeded from height_sequence_1: n={external.size}, score≈{cur_score:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 1.  Fallback initialisation if no seed
    # --------------------------------------------------------------
    if best_seq is None:
        n0 = 12000
        best_seq = u_shaped_distribution(n0, p=1.0)
        best_score = 2.0 * n0 * max_conv_fft(best_seq)
        print(f"Fallback init: n={n0}, score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Lengths to explore (larger n ⇒ smaller theoretical bound)
    # --------------------------------------------------------------
    lengths_to_try = [20000, 40000, 60000, 80000]
    lengths_to_try = [n for n in lengths_to_try if n >= best_seq.shape[0]]
    lengths_to_try = sorted(set(lengths_to_try))

    global_best_seq = best_seq.copy()
    global_best_score = best_score

    # --------------------------------------------------------------
    # 3.  Main optimisation loop over lengths
    # --------------------------------------------------------------
    for n in lengths_to_try:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT size needed for gradient (≥ 3·n‑2)
        fft_size = 1 << ((3 * n - 2).bit_length())

        # ------------------------------------------------------------------
        # 3a.  Build a collection of diverse seeds for this length
        # ------------------------------------------------------------------
        seed_vectors = []

        # (i)  Resample the current global best
        seed_vectors.append(resample_distribution(global_best_seq, n))

        # (ii) Uniform distribution (theoretical lower bound reference)
        seed_vectors.append(np.full(n, 1.0 / n))

        # (iii) U‑shapes with several exponents
        for p in (0.4, 0.8, 1.2, 2.0):
            seed_vectors.append(u_shaped_distribution(n, p))

        # (iv) Symmetric exponentials
        for lam in (0.001, 0.003, 0.008):
            seed_vectors.append(exponential_distribution(n, lam))

        # (v) Random spike patterns
        for _ in range(3):
            seed_vectors.append(random_spike_distribution(n, k=12))

        # ------------------------------------------------------------------
        # 3b.  Optimise each seed with Adam + polishing
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(seed_vectors):
            if time.time() >= deadline:
                break
            print(f"  Seed {idx + 1}/{len(seed_vectors)} (n={n})")
            # Adam optimisation
            a_opt, score_opt = adam_optimize(seed_vec,
                                             n,
                                             fft_size,
                                             deadline,
                                             max_iters=6000,
                                             lr=0.25,
                                             alpha_start=4.0,
                                             alpha_max=2000.0,
                                             alpha_mul=1.001)

            # Polishing phase (cheap local moves)
            a_opt, score_opt = polish_sequence(a_opt,
                                               n,
                                               fft_size,
                                               deadline,
                                               max_iters=1500,
                                               eps_move=1e-5)

            if score_opt < global_best_score:
                global_best_score = score_opt
                global_best_seq = a_opt.copy()
                print(f"    >>> New global best: score≈{global_best_score:.6f}")

            # early exit if we have already reached near‑theoretical bound
            if global_best_score < 1.05:   # practically impossible, but a guard
                print("    Target approached – ending early.")
                break

    # --------------------------------------------------------------
    # 4.  Final official evaluation & return
    # --------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(global_best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")
        exact_score = global_best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {global_best_score:.6f}")
    print(f"Best exact score       : {exact_score:.6f}")
    print(f"Best length            : {global_best_seq.shape[0]}")
    return global_best_seq.tolist()
