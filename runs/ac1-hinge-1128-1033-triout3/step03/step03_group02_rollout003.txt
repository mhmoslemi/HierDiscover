"""
Optimise a non‑negative step‑function sequence for the objective

    F(a) = 2·n·max_k (a*a)[k]      with   a_i ≥ 0 , Σ a_i = 1 .

The optimisation is performed with a projected Adam optimiser working on the
log‑softmax (which automatically satisfies the simplex constraints).  The
max‑convolution term is approximated by a soft‑max (temperature τ) that is
annealed from a relatively smooth regime to an almost hard max.  The search
runs for the whole time budget, first on a modest length (≈ 1700) to obtain a
good shape quickly and then on a larger length (≈ 4000) obtained by interpolating
the best small‑n solution.  The best sequence seen at any time is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper: gradient from soft‑max weights
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented via a single convolution: reverse w, convolve with a,
    then pick the appropriate slice.
    """
    n = a.shape[0]
    # reverse the weight vector (full convolution length = 2n‑1)
    u = w[::-1]
    # full convolution u * a   -> length 3n‑2
    conv = _full_convolution_fft(u, a)
    # indices 2n‑2, 2n‑3, … , n‑1  (total n entries)
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Helper: stable soft‑max used both for the probability simplex and
# the temperature‑softened max of the convolution.
# ----------------------------------------------------------------------
def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    max_log = np.max(logits)
    exp_shift = np.exp(logits - max_log)
    return exp_shift / np.sum(exp_shift)


# ----------------------------------------------------------------------
# Core optimiser for a fixed length n
# ----------------------------------------------------------------------
def _optimize_fixed_length(
    n: int,
    a_init: np.ndarray,
    deadline: float,
    phase_name: str = "phase"
) -> tuple[np.ndarray, float]:
    """
    Run Adam on the log‑softmax parametrisation for a given length `n`
    until `deadline`.  Returns the best sequence (as a numpy array) and its
    objective value 2·n·max(conv).
    """
    # initialise logits (log‑softmax parameters)
    eps = 1e-14
    a = np.maximum(a_init, eps)
    a = a / a.sum()
    l = np.log(a + eps)

    # Adam state
    m = np.zeros_like(l)
    v = np.zeros_like(l)
    beta1 = 0.9
    beta2 = 0.999
    lr0 = 0.2           # base learning rate
    eps_adam = 1e-8

    # bookkeeping
    best_a = a.copy()
    conv_a = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv_a.max())
    iteration = 0
    no_improve = 0

    # temperature schedule (soft‑max over the convolution)
    M0 = conv_a.max()
    tau_start = max(M0 * 0.5, 1e-9)
    tau_end = max(M0 * 1e-6, 1e-12)

    # probability of a tiny random jitter on the logits
    jitter_prob = 0.015
    jitter_scale = 0.25

    while True:

        if now >= deadline:

            best_a = a.copy()

        else:

        progress = (now - (deadline - (deadline - start_time))) / (deadline - start_time)
        tau = tau_start * (1.0 - progress) + tau_end * progress

        # ----- Adam update on logits -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_l
        v = beta2 * v + (1.0 - beta2) * (grad_l ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)

        lr_t = lr0 * (np.sqrt(1.0 - beta2 ** iteration) /

        if np.random.rand() < jitter_prob:
            i, j = np.random.choice(n, size=2, replace=False)
            delta = (np.random.rand() - 0.5) * jitter_scale
            l[i] += delta
            l[j] -= delta

        # ----- restart if stuck for a long time -----
        if no_improve >= 5000:
            # re‑initialise around the current best but add noise
            a = best_a.copy()
            # perturb a few coordinates
            for _ in range(10):
                idx = np.random.randint(n)
                a[idx] = max(a[idx] + np.random.randn() * 0.01, eps)
            a = a / a.sum()
            l = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{phase_name}] iter {iteration:7d}  restart triggered")

    return best_a, best_val

    the length (≈ 4000) by interpolating the best small‑n solution and continues
    optimisation.  The best sequence encountered before the time budget expires

    # ------------------------------------------------------------------

        final_best = best_a_phase1
        final_val = best_val_phase1
        final_n = n1
    else:

        final_val = best_val_phase2

        score = evaluate_sequence(final_best.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        score = float('inf')

    print(f"[summary] best length = {final_n}, best objective = {final_val:.6f}")

    return final_best.tolist()
