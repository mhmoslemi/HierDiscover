    best_a = a.copy()

    # We'll maintain several hyperparameters

    # Temperature for softmax for first phase
    tau0 = 0.2  # relative to max conv
    tau_end = 1e-5

    # Use Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps=1e-8

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute conv and max
        conv = _full_convolution_fft(a,a)
        M = conv.max()
        # compute objective (2*n*M/(sum_a**2)) but sum_a = 1
        cur_val = float(2.0 * n * M)  # sum_a^2 = 1
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(...)

        # schedule progress
        progress = (now - start) / (deadline - start)
        eta = eta0 * (1 - progress) + eta_end * progress
        tau = tau0 * (1 - progress) + tau_end * progress

        # compute gradient using softmax or hard
        if tau > 0.01:
            # softmax weights
            exp_args = (conv - M) / tau
            w = np.exp(exp_args - np.max(exp_args))  # subtract max for stability
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                # fallback hard
                idx = np.argmax(conv)
                w = np.zeros_like(conv)
                w[idx] = 1.0
            else:
                w = w / w_sum
        else:
            # hard: use only max indices
            max_idxs = np.where(conv == M)[0]
            w = np.zeros_like(conv)
            w[max_idxs] = 1.0 / len(max_idxs)  # average

        grad = _gradient_from_weights(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (iteration + 1))

"""
Propose a candidate step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

The optimizer is a two‑phase projected Adam / sub‑gradient descent.
* Phase 1 (high temperature) uses a soft‑max over the full auto‑convolution
  to obtain a smooth gradient that helps escape plateaux.
* Phase 2 (low temperature) switches to the exact sub‑gradient of the
  active maximal entry (or a small active set) for fine tuning.

Both phases operate on the probability simplex (Σ a_i = 1); the
objective value returned by ``evaluate_sequence`` is invariant under
scaling, so this normalisation is harmless.
A small amount of random “jitter’’ (mass transfer) and occasional full
re‑initialisations are added to improve exploration.

The routine runs for the allotted time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: re‑normalise if rounding error accumulates
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full linear convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y (length len(x)+len(y)-1) using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    # next power of two for efficient FFT
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negatives due to rounding
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient from a soft‑max weight vector (soft sub‑gradient)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i}   for all i.
    Implemented via a single convolution: reverse w, convolve with a,
    and pick the appropriate slice.
    """
    n = a.shape[0]
    u = w[::-1]                               # reverse weight vector
    conv = _full_convolution_fft(u, a)        # length = 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)                # indices that give required slice
    grad = 2.0 * conv[idx]
    return grad


# ----------------------------------------------------------------------
# Exact sub‑gradient for the active maximal convolution entries
# ----------------------------------------------------------------------
def _hard_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] when the set of active indices is
    ``active_idx`` (all k with maximal value). Returns a vector of length n.
    """
    n = a.shape[0]
    g = np.zeros(n, dtype=float)
    # vector of indices 0..n-1 for broadcasting
    i = np.arange(n)

    for k in active_idx:
        # contribution from a_{k-i}
        idx1 = k - i
        mask1 = (0 <= idx1) & (idx1 < n)
        if mask1.any():
            g[mask1] += a[idx1[mask1]]

        # contribution from a_{i-k}
        idx2 = i - k
        mask2 = (0 <= idx2) & (idx2 < n)
        if mask2.any():
            g[mask2] += a[idx2[mask2]]

    return 2.0 * g


# ----------------------------------------------------------------------
# Internal fast evaluation (Σ a = 1) – used during optimisation
# ----------------------------------------------------------------------
def _objective_from_conv(n: int, max_conv: float) -> float:
    """Given n and the maximal auto‑convolution value, return the objective."""
    # sum_a = 1 after our normalisation
    return 2.0 * n * max_conv

        a = np.maximum(a, 0.0)

    # Normalise to the probability simplex (Σ a = 1)

    if total <= 0:

    # --------------------------------------------------------------

    print(f"[init] n={n}, initial objective={best_val:.6f}")

    # Adam optimiser state

    beta1 = 0.90
    beta2 = 0.999

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step

            dst = np.random.randint(n - 1)

            a_candidate[dst] += delta

                m.fill(0.0)
                v.fill(0.0)
