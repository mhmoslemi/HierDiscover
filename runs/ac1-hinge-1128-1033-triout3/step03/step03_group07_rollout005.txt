def proj_simplex(v):
    # Project onto simplex {x >=0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-indexed
    theta = cssv[cond][-1] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def proj_simplex(v):
    """Exact Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all negative? then return uniform?
        # But should not happen; just return zeros?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def refine_candidate(a, deadline, max_iters=500, start_beta_factor=0.2, beta_factor=1.05, max_beta_factor=5.0, lr=0.5):

    a = a.copy()
    # ensure normalized
    a = a / a.sum()
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # beta for smooth max
    beta_cur = start_beta_factor * n
    max_beta = max_beta_factor * n
    best_a = a.copy()
    best_score = 2.0 * n * full_conv_fft(a).max()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # Compute conv and max
        conv = full_conv_fft(a)
        maxc = conv.max()
        # Update beta
        if beta_cur < max_beta:

        # Compute w and denom

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_score = np.inf

    # 0: seed from height_sequence_1
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            external = external / external.sum()
            score = 2.0 * external.size * max_conv_fft(external)
            best_seq = external.copy()
            best_score = score
            print(f"Seeded from external: approx_score={score:.6f}")
    except NameError:
        pass

    # 1: parametric sweep
    lengths = [8000, 12000, 20000]  # choose moderate lengths
    # families and parameter grids
    p_vals = np.linspace(0.3, 2.0, 12)
    alphas = np.linspace(0.0005, 0.005, 10)
    candidates = []  # list of (score, a)
    print("=== Parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break
        for p in p_vals:
            a = u_shaped_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            candidates.append((score, a))
        for p in p_vals:
            a = sum_tails_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            candidates.append((score, a))
        for alpha in alphas:
            a = exponential_distribution(n, alpha)
            score = 2.0 * n * max_conv_fft(a)
            candidates.append((score, a))
        for p in p_vals:
            a = product_powerlaw_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            candidates.append((score, a))
    # Keep top K
    candidates.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = candidates[:top_k]
    # incorporate external if better
    if best_seq is not None:
        ext_score = 2.0 * len(best_seq) * max_conv_fft(best_seq)
        top_candidates.append((ext_score, best_seq.copy()))

    # 2: refine each top candidate
    print("=== Refinement phase ===")
    for idx, (approx_score, a) in enumerate(top_candidates):
        if time.time() > deadline:
            break
        n = a.shape[0]
        print(f"Refine candidate {idx+1}/{len(top_candidates)}: n={n}, approx_score={approx_score:.6f}")
        # call refine function with remaining time
        cand_deadline = deadline
        a_refined, refined_score = refine_candidate(a, cand_deadline, max_iters=5000)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = a_refined.copy()
            print(f"New global best from candidate {idx+1}: score={best_score:.6f}")

    # 3: final evaluation
    if best_seq is None:
        # fallback random
        n = 8000
        a = np.random.rand(n)
        a = a / a.sum()
        best_seq = a
        best_score = 2.0 * n * max_conv_fft(a)

    # exact evaluation
    try:
        exact_score = evaluate_sequence(best_seq.tolist())
        print(f"\nExact evaluation: {exact_score:.6f}")
    except Exception as e:
        print(f"Failed exact evaluation: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score: {best_score:.6f}")
    print(f"Length: {len(best_seq)}")
    return best_seq.tolist()

"""
Propose a non‑negative step function that minimizes the metric
    2·n·max(conv(a,a))  (with a normalized to sum 1).

Algorithm
---------
1. **Warm‑start** – use the global variable ``height_sequence_1`` if it
   exists (normalised to sum‑1).

2. **Deterministic parametric sweep** – generate a few hundred candidates
   from simple analytic families (U‑shaped power‑law, sum‑of‑tails,
   exponential, product‑power‑law) for several lengths (up to 48 000).
   Keep the best few candidates (according to the fast FFT based
   approximation).

3. **Smooth‑max gradient refinement** – for each kept candidate run a
   projected Adam optimiser that minimises a smooth approximation of the
   ``max`` (log‑sum‑exp) of the convolution.  The smoothness parameter
   β is increased gradually so that the optimisation first sees a smooth
   landscape and then focuses on the true peak.  After each Adam step the
   vector is projected onto the probability simplex.

   The routine stops when the global time budget (≈1000 s) is exhausted
   or the target score 1.503 is reached.

4. **Exact evaluation** – the best vector obtained is finally evaluated
   with the provided ``evaluate_sequence`` function and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple convolution utilities (real FFT via complex FFT for clarity)
# ----------------------------------------------------------------------


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors using a complex FFT."""
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1).bit_length())
    X = np.fft.fft(x, size)
    Y = np.fft.fft(y, size)
    conv = np.fft.ifft(X * Y)[: n + m - 1].real
    return conv


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a."""
    return conv_fft(a, a)


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (fast via FFT)."""
    return float(full_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = len(v)
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return a normalised vector, sum == 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """Product‑power‑law shape: a[i] ∝ ((i+1)*(n-i))^{‑p}."""
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max + Adam optimisation on the simplex
# ----------------------------------------------------------------------


def smooth_max_and_grad(a: np.ndarray, smooth_beta: float):
    """
    Returns (smooth_max, gradient, conv) for the current a.
    Gradient is the derivative of the smooth max
    (log‑sum‑exp) with respect to a.
    """
    n = len(a)
    conv = full_conv_fft(a)                     # length 2n‑1
    maxc = conv.max()
    # stable log‑sum‑exp
    exp_arg = smooth_beta * (conv - maxc)       # ≤ 0
    w = np.exp(exp_arg)                         # w[max] = 1
    denom = w.sum()
    smooth = maxc + np.log(denom) / smooth_beta

    # gradient: (2/denom) * (w * a_rev) convolution slice
    a_rev = a[::-1]
    conv_w_rev = conv_fft(w, a_rev)            # length 3n‑2
    grad_slice = conv_w_rev[n - 1: n - 1 + n]   # length n
    grad = (2.0 / denom) * grad_slice
    return smooth, grad, conv


def refine_candidate(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 5000,
    start_beta_factor: float = 0.2,
    beta_mult: float = 1.07,
    max_beta_factor: float = 5.0,
    lr: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation that minimises the smooth‑max of a*a.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    n = len(a)
    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # smooth‑max parameters
    smooth_beta = max(start_beta_factor * n, 1.0)
    max_smooth_beta = max_beta_factor * n

    # current objective
    conv = full_conv_fft(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # compute smooth max and gradient
        smooth, grad, conv = smooth_max_and_grad(a, smooth_beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat
        a_candidate = proj_simplex(a_candidate)

        conv_new = full_conv_fft(a_candidate)
        score_new = 2.0 * n * conv_new.max()

        # greedy acceptance with a tiny simulated‑annealing tail
        if score_new < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - score_new) / max(smooth_beta, 1e-8))
            accept = np.random.rand() < prob

        if accept:
            a = a_candidate
            conv = conv_new
            cur_score = score_new

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"  Refine iter {it:5d}: beta={smooth_beta:.1f}, "
                    f"score={best_score:.6f}"
                )
                if best_score < 1.503:
                    print("  Target reached inside refinement.")
                    break

        # increase smoothness gradually (approach true max)
        if smooth_beta < max_smooth_beta:
            smooth_beta = min(smooth_beta * beta_mult, max_smooth_beta)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the challenge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step function that minimises the provided evaluation metric.
    Returns a plain Python list of floats (non‑negative, sum irrelevant).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0. Warm‑start from external variable if present
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)  # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            if external.sum() == 0.0:
                external = np.ones_like(external)
            external /= external.sum()
            approx = 2.0 * external.size * max_conv_fft(external)
            best_seq = external.copy()
            best_score = approx
            print(f"Seeded from height_sequence_1: approx_score={approx:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Deterministic parametric sweep
    # ------------------------------------------------------------------
    lengths = [8000, 16000, 32000, 48000]
    p_vals = np.linspace(0.3, 2.2, 12)      # 12 values
    alphas = np.linspace(0.0004, 0.004, 10)  # 10 values

    candidate_pool = []      # (approx_score, vector)

    print("\n=== Parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break
        # U‑shaped power‑law
        for p in p_vals:
            a = u_shaped_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            candidate_pool.append((score, a))
        # Sum‑of‑tails
        for p in p_vals:
            a = sum_tails_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            candidate_pool.append((score, a))
        # Exponential decay
        for alpha in alphas:
            a = exponential_distribution(n, alpha)
            score = 2.0 * n * max_conv_fft(a)
            candidate_pool.append((score, a))
        # Product power‑law
        for p in p_vals:
            a = product_powerlaw_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            candidate_pool.append((score, a))

    # keep the top few candidates for refinement
    candidate_pool.sort(key=lambda x: x[0])
    top_k = 4
    top_candidates = candidate_pool[:top_k]

    # also make sure the current best (maybe from external) is present
    if best_seq is not None:
        ext_score = 2.0 * len(best_seq) * max_conv_fft(best_seq)
        top_candidates.append((ext_score, best_seq.copy()))

    # ------------------------------------------------------------------
    # 2. Smooth‑max Adam refinement
    # ------------------------------------------------------------------
    print("\n=== Refinement phase ===")
    for idx, (approx_score, vec) in enumerate(top_candidates, start=1):
        if time.time() > deadline:
            break
        n = vec.shape[0]
        print(f"\nRefining candidate {idx}/{len(top_candidates)}: n={n}, approx={approx_score:.6f}")
        # allocate as many iterations as time permits (rough heuristic)
        remaining = deadline - time.time()
        iters = max(100, int( (remaining / (len(top_candidates) - idx + 1)) / 0.02 ))
        refined_vec, refined_score = refine_candidate(
            vec,
            deadline=deadline,
            max_iters=iters,
            start_beta_factor=0.2,
            beta_mult=1.07,
            max_beta_factor=5.0,
            lr=0.6,
        )
        print(f"  Candidate {idx} best score after refinement: {refined_score:.6f}")
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_vec.copy()
            print(f"  ** New global best! **")

        if best_score < 1.503:
            print("Target reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # 3. Fallback – random vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = 2.0 * n * max_conv_fft(a)
        print(f"Fallback random initialisation: n={n}, approx_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact evaluation (once) and reporting
    # ------------------------------------------------------------------
    try:
        exact = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation (evaluate_sequence): {exact:.6f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")
        exact = best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {best_score:.6f}")
    print(f"Exact score (if evaluated) : {exact:.6f}")
    print(f"Best vector length : {len(best_seq)}")
    return best_seq.tolist()
