    score = 2 * n * max_k (a * a)[k] / (sum a)^2

The algorithm:

1. **Fast parametric sweep** – explore several families of U‑shaped
   sequences (inverse distance, inverse‑product distance, exponential,
   inverse‑sqrt, logarithmic) for a few lengths and a coarse grid of
   parameters.  The best vector from this sweep seeds the refinement.

2. **Smoothed‑max gradient descent** – minimise a smooth approximation of
   the maximum convolution entry using a soft‑max (log‑sum‑exp) surrogate.
   The gradient is obtained analytically via FFT‑based convolutions.
   An Adam‑style update together with projection onto the probability
   simplex (Wang‑Carreira‑Perpiñán) drives the optimisation.
   The temperature of the soft‑max (parameter ``alpha``) is increased
   gradually so the optimisation first sees a smooth landscape and then
   focuses on the true maximum.  Occasional random re‑heats escape
   stagnation.

The routine respects the given time‑budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {w >= 0, sum w = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using real FFT.
    Returns array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def full_convolution_fft_generic(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * b (both non‑negative) using real FFT.
    Returns array of length len(a)+len(b)-1.
    """
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector with sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def gradient_smoothed_max(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate:
        F(a) = (1/alpha) * log sum_k exp(alpha * (a*a)[k])
    w.r.t. a (assuming sum(a)=1).  Returns a vector of same length as a.
    """
    n = a.shape[0]
    # 1) convolution a*a
    conv = full_convolution_fft(a)                     # length 2n-1
    # 2) soft‑max weights
    max_c = conv.max()
    exp_vals = np.exp(alpha * (conv - max_c))
    w = exp_vals / exp_vals.sum()                     # shape (2n-1,)

    # 3) compute b_i = sum_j a_j * w_{i+j} for i = 0..n-1
    a_rev = a[::-1]
    b_full = full_convolution_fft_generic(a_rev, w)   # length 3n-2
    b = b_full[n - 1: n - 1 + n]                      # slice of length n

    # 4) extra diagonal term for i=j (k=2i)
    w2i = w[2 * np.arange(n)]

    # gradient of the surrogate (descent direction is -grad)
    grad = 2.0 * b + w2i * a
    return grad


# ----------------------------------------------------------------------
# Parametric families used in the deterministic sweep
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on min distance to an edge: (min(i,n-1-i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def inv_sqrt_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape with inverse square‑root power."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a = np.sqrt(a)                     # equivalent to power -p/2
    a /= a.sum()
    return a


def log_distribution(n: int, scale: float) -> np.ndarray:
    """Very slow decay using logarithm."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d) + 1.0
    a = 1.0 / np.log(scale * d + np.e) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the time budget.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Fast deterministic sweep over parametric families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    lengths = [2000, 4000, 8000, 12000, 20000]          # trade‑off quality / speed
    exponents = np.linspace(0.1, 8.0, 80)               # for power‑law families
    lam_vals = np.logspace(-5, -0.3, num=30)           # for exponential family
    sqrt_exps = np.linspace(0.1, 4.0, 40)              # for inverse‑sqrt family
    log_scales = np.logspace(0.5, 3.0, num=20)        # for log family

    print("=== Deterministic sweep start ===")
    for n in lengths:
        if time.time() > deadline:
            break
        for p in exponents:
            # min‑distance family
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (min)  n={n:5d} p={p:5.2f}  score={sc:.6f}")

            # product‑distance family
            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (prod) n={n:5d} p={p:5.2f}  score={sc:.6f}")

        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (exp)  n={n:5d} λ={lam:.5f}  score={sc:.6f}")

        for p in sqrt_exps:
            a = inv_sqrt_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (sqrt) n={n:5d} p={p:5.2f}  score={sc:.6f}")

        for scale in log_scales:
            a = log_distribution(n, scale)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (log)  n={n:5d} s={scale:.5f}  score={sc:.6f}")

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. Incorporate external construction if available
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0.0:
                external /= s
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score = sc
                    best_seq = external.copy()
                    print(f" using external height_sequence_1  score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smoothed‑max gradient descent with Adam & re‑heat
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]
    current /= current.sum()                  # ensure exact normalisation
    current_conv = full_convolution_fft(current)
    current_score = float(2.0 * n * current_conv.max())

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(current)
    v = np.zeros_like(current)
    beta1_pow, beta2_pow = 1.0, 1.0

    # soft‑max temperature schedule
    alpha = 1.0
    alpha_max = 2000.0
    alpha_mul = 1.001

    # base learning rate (will be scaled by Adam)
    base_lr = 0.15

    iteration = 0
    last_improve = 0
    print("\n=== Smoothed‑max gradient descent start ===")
    while time.time() < deadline:
        iteration += 1

        # ----- gradient of the smoothed max -----
        grad = gradient_smoothed_max(current, alpha)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = current - step
        cand = simplex_projection(cand)          # project onto simplex

        # ----- evaluate candidate -----
        cand_conv = full_convolution_fft(cand)
        cand_score = float(2.0 * n * cand_conv.max())

        # ----- accept if improvement (or occasional uphill move) -----
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # simulated‑annealing style acceptance
            temp = max(1.0, 0.5 * (iteration / 10000.0))
            prob = np.exp((current_score - cand_score) / temp)
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_conv = cand_conv
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ----- schedule updates -----
        alpha = min(alpha * alpha_mul, alpha_max)

        # ----- stagnation handling -----
        if iteration - last_improve > 60000:
            # small random bump + reset of Adam moments
            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            current /= current.sum()
            # reset Adam statistics
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0
            # reset temperature schedule
            alpha = 1.0
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # Ensure non‑negative and finite values before returning
    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()
