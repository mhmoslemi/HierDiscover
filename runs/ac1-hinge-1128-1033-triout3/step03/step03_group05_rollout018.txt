Propose a step‐function (non‑negative sequence) that minimizes
   2 * n * max(conv_self) / (sum(a)^2)

The algorithm builds several initial candidates (arcsine, geometric,
uniform, random Dirichlet and the historic sequence if available) and
optimises each of them with three stages:

1. Sub‑gradient descent on the true maximal convolution entry using
   an active‑set of all currently maximal indices.
2. Adam optimisation on a smooth log‑sum‑exp approximation of the
   max.  The smoothing parameter α is increased gradually up to a
   large value, turning the smooth objective into the exact one.
3. A final short sub‑gradient phase on the exact max + a tiny random
   swap hill‑climber.

All updates are projected onto the probability simplex so the sum of
the coefficients stays equal to 1 and the vector stays non‑negative.
The best vector found within the allotted time budget is returned.

import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex.
# --------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Convolution utilities (FFT based)
# --------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution of p using a pre‑computed FFT size."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _gradient_from_weights(p: np.ndarray, p_fft: np.ndarray,
                          w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smoothed max.
    w – soft‑max weights (length 2·n‑1) that sum to 1.
    """
    # reverse the weight vector and convolve with p (via FFT)
    w_rev = w[::-1]

    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * p.size - 2]
    # mapping to the gradient indices (see original reference implementation)
    idx = 2 * p.size - 2 - np.arange(p.size)
    grad = 2.0 * corr[idx]
    return grad

"""
Propose a step‑function (non‑negative sequence) that minimises

    2 · n · max(conv_self) / (sum(a)²)

The algorithm builds several initial candidates (arcsine, a geometric decay,
uniform, a few random Dirichlet draws and the historic sequence if it exists)
and refines each of them in four stages:

1. Sub‑gradient descent on the *exact* maximal convolution entry using an
   active‑set of all indices that are numerically maximal.
2. Adam optimisation on a smooth log‑sum‑exp approximation of the max.
   The smoothing parameter α is increased gradually up to a large value,
   turning the smooth objective into the true one.
3. A short polishing phase with the exact max sub‑gradient.
4. A tiny random‑swap hill‑climber.

All updates are projected onto the probability simplex, guaranteeing a
non‑negative vector with sum = 1, which is exactly what `evaluate_sequence`
expects (the clipping to [0, 1000] never changes the values).  The best
vector found before the time budget expires is returned.
"""

import numpy as np
import time

# --------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# --------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Convolution utilities (FFT based)
# --------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution of p using a pre‑computed FFT size."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _gradient_from_weights(p: np.ndarray, p_fft: np.ndarray,
                          w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smoothed max.
    w – soft‑max weights (length 2·n‑1) that sum to 1.
    """
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * p.size - 2]
    idx = 2 * p.size - 2 - np.arange(p.size)
    grad = 2.0 * corr[idx]
    return grad

# --------------------------------------------------------------

    # --------------------------------------------------------------
    # Determine problem size (length of the step function)

    except Exception:

    n = len(base_seq) if base_seq is not None else 1153

    # FFT size – next power‑of‑two that can hold any convolution we need
    m_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Exact objective (no extra normalisation needed)
    # --------------------------------------------------------------
    def _objective(p: np.ndarray) -> float:
        conv = _conv_self(p, m_fft)
        return float(2.0 * n * np.max(conv))

    # --------------------------------------------------------------
    # Build a pool of promising initial vectors
    # --------------------------------------------------------------
    init_vectors = []

    # 1) Arcsine shape (known to flatten the autocorrelation)
    i = np.arange(n, dtype=float)
    inv = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + 1e-12)
    p_arcsine = inv / inv.sum()
    init_vectors.append(p_arcsine)

    # 2) Geometric decay (r < 1)
    r = 0.99
    geom = r ** i
    p_geom = geom / geom.sum()
    init_vectors.append(p_geom)

    # 3) Uniform distribution
    init_vectors.append(np.full(n, 1.0 / n))

    # 4) Random Dirichlet draws
    for _ in range(3):
        rnd = rng.random(n)
        rnd /= rnd.sum()
        init_vectors.append(rnd)

    # 5) Historic construction, if available
    if base_seq is not None:
        seq = np.clip(base_seq, 0.0, 1000.0)
        s = seq.sum()
        if s > 0:
            init_vectors.append(seq / s)

    # --------------------------------------------------------------
    # Optimisation state
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # tiny safety margin
    best_score = float("inf")
    best_p = None

    # Core optimisation loops
    # --------------------------------------------------------------
    for start_id, p0 in enumerate(init_vectors):
        if time.time() >= deadline:

        print(f"\n=== Start {start_id + 1}/{len(init_vectors)} =====================")
        p = np.maximum(p0, 0.0)

        # ----------------------------------------------------------

        decay = 0.9997

        m2 = np.zeros_like(p)

            max_z = np.max(z)

            # Adam update

            p = _simplex_projection(p)

                if cur_score < best_score - 1e-12:

                    if best_score <= 1.503:
                        break

        # ----------------------------------------------------------
        # (C) Fine‑grained sub‑gradient phase on the true max
        # ----------------------------------------------------------
        step_c = 0.1
        decay_c = 0.9999
        min_step_c = 1e-8
        for it in range(50_000):

            max_val = conv.max()

                best_p = p.copy()

                if best_score <= 1.503:
                    break

            active = np.where(conv >= max_val - tol)[0]

            g = np.zeros_like(p)
            for idx in active:
                if idx < n:
                    g[: idx + 1] += 2.0 * p[idx::-1]
                else:
                    start = idx - n + 1
                    g[start:] += 2.0 * p[n - 1 : start - 1 : -1]
            g /= float(len(active))

            p = p - step_c * g
            p = _simplex_projection(p)

            step_c = max(step_c * decay_c, min_step_c)

        # ----------------------------------------------------------
        # (D) Tiny random‑swap hill‑climber
        # ----------------------------------------------------------
        p_swap = best_p.copy() if best_p is not None else p.copy()
        cur_score = _objective(p_swap)

        for _ in range(30_000):

        best_score = _objective(best_p)

    return np.clip(best_p, 0.0, 1000.0).tolist()
