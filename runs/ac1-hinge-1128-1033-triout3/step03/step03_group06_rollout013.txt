def propose_candidate(budget_s=1000, seed=0):
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_metric = np.inf

    # Try prior solution if any
    try:
        prev = np.array(height_sequence_1, dtype=float)
        if prev.size:
            # ensure positivity
            prev = np.maximum(prev, 0.0)
            # normalize sum to 1
            if prev.sum() == 0:
                prev = np.ones_like(prev) / prev.size
            else:
                prev = prev / prev.sum()
            metric = 2 * prev.size * max_conv_fft(prev)
            best_seq = prev.copy()
            best_metric = metric
            print(f"Seeded from height_sequence_1: n={prev.size}, metric={metric:.6f}")
    except NameError:
        pass

    # Set target length (maybe use 20000)
    target_lengths = [20000]  # maybe add 25000 as alternative
    # you can try multiples based on time

    for n in target_lengths:
        # if time is low break
        if time.time() >= deadline:
            break

        # param sweep
        # generate list of p values
        p_vals = np.linspace(0.1, 2.0, 20)  # 0.1..2.0 step 0.1 ~ 20
        lam_vals = np.linspace(0, 1, 11)   # 0..1 step 0.1

        local_best_a = None
        local_best_metric = np.inf
        for p in p_vals:
            # compute U shape
            u = u_shaped_distribution(n, p)
            # uniform shape
            uniform = np.full(n, 1.0/n)
            for lam in lam_vals:
                a = lam * uniform + (1 - lam) * u
                # ensure nonnegative
                a = np.maximum(a, 0.0)
                a /= a.sum()
                cur_metric = 2 * n * max_conv_fft(a)
                if cur_metric < local_best_metric:
                    local_best_metric = cur_metric
                    local_best_a = a.copy()
        print(f"Length {n}: best parametric metric = {local_best_metric:.6f}")

        # If parametric best is better than global best, update.
        if local_best_metric < best_metric:
            best_metric = local_best_metric
            best_seq = local_best_a.copy()
        # Now apply Adam optimization starting from local_best_a
        # config
        max_iters = 5000    # adjust based on time
        # Adam hyperparams
        lr = 0.2
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        # initialize logit vector
        x = np.log(local_best_a + 1e-12)
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        t = 0
        # Smoothing parameter
        alpha = 2.0
        alpha_max = 500.0
        alpha_mul = 1.005

        # Precompute FFT size for gradient (need size >= 3n-2)
        fft_size = 1 << ((3 * n - 2).bit_length())
        print(f"FFT size: {fft_size}")

        while time.time() < deadline and t < max_iters:
            t += 1
            # compute a = softmax(x)
            max_x = np.max(x)
            exps = np.exp(x - max_x)
            a = exps / exps.sum()
            # compute conv
            conv = full_conv_fft(a, fft_size)
            max_conv = conv.max()
            metric = 2 * n * max_conv
            # update global best
            if metric < best_metric:
                best_metric = metric
                best_seq = a.copy()
                print(f"Iter {t:5d}, new best metric {metric:.6f}")

            # compute softmax weights for convex approximation
            w = softmax_weights(conv, alpha)

            # compute gradient wrt a
            grad_a = gradient_softmax(a, w, fft_size)

            # gradient wr.t. logits
            grad_dot = np.dot(grad_a, a)
            grad_x = a * (grad_a - grad_dot)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_x

            m_hat = m / (1 - beta1 ** t)

    print(f"Best length: {best_seq.shape[0]}")

"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` metric.

Approach
--------
1.  Use the previously supplied construction (if any) as a seed.
2.  Perform a cheap parametric sweep over simple symmetric “U‑shaped”
    families (mixture of uniform and a power‑law edge distribution) to
    obtain a good starting point.
3.  Refine the candidate with an Adam optimiser that minimises a
    smooth approximation of `max(conv)` (soft‑max of the self‑convolution).
    The variables are logits of a soft‑max, guaranteeing positivity and
    unit sum.
4.  If optimisation stalls, inject a small random perturbation and
    continue.
5.  Finally, run a cheap random‑swap local search to polish the result.
6.  Return the best list found.  All work respects the given time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·n‑1) using a supplied FFT size.
    Returned array is real and has length 2·n‑1.
    """
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities w_k = exp(α·conv_k) / Σ exp(α·conv)."""
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution w.r.t. a.
    a : vector of length n
    w : soft‑max weights of length 2·n‑1 (sum to 1)
    size : FFT length (≥ 3·n‑2)
    Returns a vector of length n.
    """
    n = a.shape[0]
    L = 2 * n - 1
    A = np.fft.rfft(a, n=size)

    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    idx = L - 1 - np.arange(n)                # positions for the reversed part
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law “U‑shape”.
    a_i ∝ (min(i, n‑1‑i) + 1)^{‑p}.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)               # distance to nearest edge
    a = (d + 1.0) ** (-p) + 1e-14
    a /= a.sum()
    return a


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Symmetric arcsine density: a_i ∝ 1 / sqrt(i+0.5) / sqrt(n-0.5-i).
    """
    i = np.arange(n) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i)) + 1e-14
    a /= a.sum()
    return a

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns the best sequence found within the time
    budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_metric = np.inf

    # --------------------------------------------------------------
    # 0.  Seed from a previously supplied construction, if any
    # --------------------------------------------------------------
    try:

        if external.size > 0:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external = external / s
            cur_metric = 2.0 * external.size * max_conv_fft(external)
            best_seq = external.copy()
            best_metric = cur_metric
            print(f"[seed] length={external.size:,d}, metric≈{cur_metric:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 1.  Choose a working length (20000 is a good trade‑off)
    # --------------------------------------------------------------
    n = 20000
    if best_seq is not None and best_seq.shape[0] != n:
        # resample the seeded vector to length n (linear interpolation)
        old = best_seq
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        best_seq = np.interp(x_new, x_old, old)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq /= best_seq.sum()
        # update metric for the resampled vector
        best_metric = 2.0 * n * max_conv_fft(best_seq)
        print(f"[resample] seeded vector resampled to n={n}")

    # --------------------------------------------------------------
    # 2.  Simple parametric sweep (U‑shape + uniform + arcsine)
    # --------------------------------------------------------------
    param_best_seq = None
    param_best_metric = np.inf

    # families to test
    families = []

    # family 0: pure U‑shape
    families.append(("U", None))

    # family 1: mixture of uniform and U‑shape
    families.append(("mixU", None))

    # family 2: arcsine (no extra parameter)
    families.append(("arc", None))

    # sweep parameters
    p_vals = np.linspace(0.1, 2.2, 22)          # 0.1 … 2.2
    lam_vals = np.linspace(0.0, 1.0, 11)       # mixture weight

        for p in p_vals:
            # basic U‑shape for this exponent
            u = u_shaped_distribution(n, p)

            if fam == "U":
                a = u
                metric = 2.0 * n * max_conv_fft(a)
                if metric < param_best_metric:
                    param_best_metric = metric
                    param_best_seq = a.copy()

            elif fam == "mixU":
                uniform = np.full(n, 1.0 / n)
                for lam in lam_vals:
                    a = lam * uniform + (1.0 - lam) * u
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    metric = 2.0 * n * max_conv_fft(a)
                    if metric < param_best_metric:
                        param_best_metric = metric
                        param_best_seq = a.copy()

                metric = 2.0 * n * max_conv_fft(a)

    # optimiser hyper‑parameters

    lr = 0.18
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # smoothing (soft‑max) parameters
    alpha = 2.0
    alpha_max = 500.0
    alpha_mul = 1.005

    # FFT size needed for the gradient (≥ 3·n‑2)
    fft_size = 1 << ((3 * n - 2).bit_length())

    # initialise logits from the current best sequence
    if best_seq is None:

    x = np.log(best_seq + 1e-14)   # logits (real numbers)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # stagnation tracking
    last_improvement = 0
    stagnation_counter = 0

    print(f"[opt] starting Adam optimisation, n={n}, fft_size={fft_size}")

    while time.time() < deadline and max_iters > 0:
        # ----- iteration bookkeeping -----
        max_iters -= 1
        # compute probability vector a = softmax(x)
        mx = x.max()
        exps = np.exp(x - mx)
        a = exps / exps.sum()

        # ----- convolution and current metric -----
        conv = full_conv_fft(a, fft_size)
        cur_max = conv.max()
        cur_metric = 2.0 * n * cur_max

        if cur_metric < best_metric:
            best_metric = cur_metric
            best_seq = a.copy()
            last_improvement = time.time()
            stagnation_counter = 0
            print(f"[opt] iter {max_iters:5d} – new best metric = {cur_metric:.6f}")

        # ----- soft‑max (smooth) of the convolution -----
        w = softmax_weights(conv, alpha)

        # ----- gradient w.r.t. a (soft‑max loss) -----
        grad_a = gradient_softmax(a, w, fft_size)

        # ----- gradient w.r.t. logits (log‑softmax parametrisation) -----
        # ∂L/∂x = a * (∂L/∂a – 〈∂L/∂a , a〉)
        inner = np.dot(grad_a, a)
        grad_x = a * (grad_a - inner)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        t = max_iters + 1  # iteration counter (decreasing, but only a scale)
        m_hat = m / (1.0 - beta1 ** (t))
        v_hat = v / (1.0 - beta2 ** (t))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        x -= step
        # keep logits bounded (prevents overflow in the soft‑max)
        np.clip(x, -60.0, 60.0, out=x)

        # ----- increase smoothness -----
        alpha = min(alpha * alpha_mul, alpha_max)

        # ----- stagnation handling -----
        stagnation_counter += 1
        if stagnation_counter % 2000 == 0:
            # small random jitter to shake a possible plateau
            jitter = np.random.normal(scale=0.02, size=n)
            x += jitter
            print(f"[opt] iteration {max_iters:5d} – added jitter, alpha={alpha:.2f}")

        # safety break
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 4.  Light random‑swap local polishing
    # --------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        # pre‑compute FFT size for this vector (can reuse existing)

            if time.time() >= deadline:

            a_candidate[i1] += delta
            a_candidate[i2] -= delta
            # renormalise to keep sum exactly 1 (tiny numerical drift)
            a_candidate /= a_candidate.sum()
            cand_max = max_conv_fft(a_candidate)
            cand_metric = 2.0 * n * cand_max
            if cand_metric < best_metric:
                best_metric = cand_metric
                best_seq = a_candidate.copy()
                a = a_candidate
                print(f"[local] swap #{i+1} – improved metric = {cand_metric:.6f}")

    # --------------------------------------------------------------
    # 5.  Final official evaluation
    # --------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluate_sequence result: {exact_score:.6f}")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")
        exact_score = best_metric

    print("\n=== Search finished ===")
    print(f"Best metric (approx.) : {best_metric:.6f}")
    print(f"Best length           : {best_seq.shape[0]}")
    return best_seq.tolist()
