"""
Optimisation of a non‑negative step function to minimise

    F(a) = 2·n·max_k (a*a)[k] / ( Σ a_i )²

where n = len(a) and “*” is the full linear convolution.
All a_i are forced to be ≥0 and the vector is projected onto the
probability simplex (Σ a_i = 1), so the denominator is constant and the
goal is simply to minimise the peak of the auto‑convolution.

The optimiser is a multi‑restart projected Adam optimiser with a
soft‑max (temperature‑controlled) approximation of the ∞‑norm.
Each restart begins from a random Dirichlet draw mixed with a smooth
sinusoidal shape (or from the previously discovered `height_sequence_1`
if it exists).  The learning‑rate and soft‑max temperature are linearly
decayed, a small random “jitter’’ is applied occasionally, and the best
sequence seen over the whole time budget is returned.

The implementation uses FFT‑based convolutions for speed and a fast
Euclidean projection onto the simplex (Duchi et al.).
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σ x = 1} using the O(n) algorithm of Duchi et al."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise if rounding errors broke the sum
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full linear convolution by real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Return the full convolution x*y (length len(x)+len(y)-1) using FFT.
    The result is forced to be non‑negative to avoid tiny negatives from rounding.
    """
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Eliminate possible negative rounding errors
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max weighted convolution objective
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient g_i = 2 Σ_k w_k * a_{k-i}
    using a single convolution.  The weight vector w has length 2·n‑1.
    """
    n = a.shape[0]
    # reverse the weight vector (as required by the formula)
    u = w[::-1]
    # full convolution u * a has length (2n‑1) + n - 1 = 3n‑2
    conv = _full_convolution_fft(u, a)
    # The needed slice corresponds to indices: 2n‑2, 2n‑3, … , n‑1 (total n entries)
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g
