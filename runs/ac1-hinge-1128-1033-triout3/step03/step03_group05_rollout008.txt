def subgradient_descent(p0, n, m_fft, step_init, decay, min_step, max_iter, deadline):
    p = p0.copy()
    best_score = score(p, n, m_fft)
    best_p = p.copy()

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        max_val = conv.max()
        max_idx = conv.argmax()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
        # gradient accumulation for potentially multiple maxima:
        # find all indices where conv is close to max_val
        # but use simple one.
        g = np.zeros_like(p)

def adam_optimize(p0, n, m_fft, max_iter, deadline, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8, alpha_init=0.5, alpha_mul=1.005, alpha_max=200.0):
    p = p0.copy()
    best_score = score(p, n, m_fft)
    best_p = p.copy()
    m1 = np.zeros_like(p)
    m2 = np.zeros_like(p)
    alpha = alpha_init
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute fft
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2*n-1]
        # softmax weights
        z = alpha * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        w = exp_z / np.sum(exp_z)
        # gradient
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
        corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[:3*n-2]
        idx = 2*n-2 - np.arange(n)
        grad = 2.0 * corr_full[idx]
        # Adam update
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * (grad ** 2)
        m_hat = m1 / (1 - beta1 ** (it + 1))
        v_hat = m2 / (1 - beta2 ** (it + 1))
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
        p = p - step_vec
        p = simplex_projection(p)
        # increase alpha gradually
        if alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)
        # optional decreasing lr schedule? Could keep constant.

"""
propose_candidate
-----------------

A multi‑restart optimizer for the self‑convolution L∞‑objective.
For each promising initial shape we perform

1. Projected sub‑gradient descent on the *actual* maximal convolution entry.
2. Adam optimisation on a smooth log‑sum‑exp approximation of the max.
3. Tiny stochastic mass‑transfer hill‑climber (local swaps).

A broad family of initial vectors (arcsine, Beta‑shapes, triangular,
exponential, uniform and random Dirichlet draws) is explored.
All steps respect the probability‑simplex, keep the vector non‑negative,
and stop as soon as the 1000 s budget is exhausted or a score ≤ 1.503
is found.  The best vector encountered is returned (clipped to the limits
enforced by ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x | x_i ≥ 0, Σ x_i = 1}
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution of p (length 2·len(p)‑1) using FFT."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def objective_score(p: np.ndarray, n: int, m_fft: int) -> float:
    """2·n·max(conv_self(p)) – the quantity returned by evaluate_sequence."""
    conv = conv_self(p, m_fft)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max entry
# ----------------------------------------------------------------------
def subgradient_descent(p0: np.ndarray,
                        n: int,
                        m_fft: int,
                        step_init: float = 0.3,
                        decay: float = 0.9995,
                        min_step: float = 1e-6,
                        max_iter: int = 15000,
                        deadline: float = 1e20):
    """Return (p_end, best_p, best_score) after a projected sub‑gradient phase."""
    p = p0.copy()
    best_p = p.copy()
    best_score = objective_score(p, n, m_fft)
    step = step_init

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = conv_self(p, m_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            # print progress only on improvement
            print(f"  Subgrad it {it:5d} – new best {best_score:.6f}")

        # indices where the max is attained (handle ties)
        max_idxs = np.where(conv >= max_val - 1e-12)[0]
        g = np.zeros_like(p)
        for k in max_idxs:
            if k < n:
                # slice 0 .. k
                g[: k + 1] += 2.0 * p[k::-1]
            else:
                start = k - n + 1
                # slice start .. n‑1
                g[start:] += 2.0 * p[n - 1 : start - 1 : -1]

        # gradient step + projection
        p = p - step * g
        p = simplex_projection(p)

        # step‑size decay
        step = max(step * decay, min_step)

        if best_score <= 1.503:
            break

    return p, best_p, best_score


# ----------------------------------------------------------------------
# Adam optimiser on a log‑sum‑exp (soft‑max) surrogate of the max
# ----------------------------------------------------------------------
def adam_optimize(p0: np.ndarray,
                  n: int,
                  m_fft: int,
                  max_iter: int = 100000,
                  deadline: float = 1e20,
                  lr: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps_adam: float = 1e-8,
                  alpha_init: float = 0.5,
                  alpha_mul: float = 1.005,
                  alpha_max: float = 200.0):
    """Return (p_end, best_p, best_score) after Adam on the smooth max."""
    p = p0.copy()
    best_p = p.copy()
    best_score = objective_score(p, n, m_fft)

    m1 = np.zeros_like(p)
    m2 = np.zeros_like(p)

    alpha = alpha_init

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # FFT of current p
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

        # soft‑max weights over conv entries
        z = alpha * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)          # stability shift
        w = exp_z / np.sum(exp_z)          # sum(w) = 1

        # gradient of Σ_i w_i conv_i  == 2 * (p * w_rev)  (restricted to first n elements)
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
        corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]
        idx = 2 * n - 2 - np.arange(n)    # map to appropriate slice
        grad = 2.0 * corr_full[idx]

        # Adam state updates
        m1 = beta1 * m1 + (1.0 - beta1) * grad
        m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

        m_hat = m1 / (1.0 - beta1 ** (it + 1))
        v_hat = m2 / (1.0 - beta2 ** (it + 1))

        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        p = p - step_vec
        p = simplex_projection(p)

        # increase smoothing parameter
        if alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # evaluate true objective
        cur_score = objective_score(p, n, m_fft)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            print(f"  Adam it {it:5d} – new best {best_score:.6f}")

        if best_score <= 1.503:
            break

    return p, best_p, best_score


# ----------------------------------------------------------------------
# Tiny local refinement: move a tiny amount of mass between two indices
# ----------------------------------------------------------------------
def local_swap_refinement(p0: np.ndarray,
                          n: int,
                          m_fft: int,
                          attempts: int = 20000,
                          deadline: float = 1e20,
                          rng: np.random.Generator = None):
    """Perform a greedy pairwise‑swap hill‑climber."""
    if rng is None:
        rng = np.random.default_rng()
    p = p0.copy()
    cur_score = objective_score(p, n, m_fft)
    best_p = p.copy()
    best_score = cur_score

    for _ in range(attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a tiny random amount from i to j
        delta = min(p[i], 0.001 * rng.random())
        if delta <= 0.0:
            continue
        new_p = p.copy()
        new_p[i] -= delta
        new_p[j] += delta
        new_score = objective_score(new_p, n, m_fft)
        if new_score < cur_score - 1e-12:
            p = new_p
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_p = p.copy()
                print(f"Swap improvement → {best_score:.6f}")
                if best_score <= 1.503:
                    break
    return best_p, best_score

    rng = np.random.default_rng(seed)
    start_time = time.time()

    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------

    try:

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    init_vectors = []

    eps = 1e-12

    p_arcsine = inv / inv.sum()
    init_vectors.append(p_arcsine)

    # 2) A handful of Beta‑shaped families (symmetric Beta(α,α))
    for α in (0.3, 0.8, 1.0, 1.5, 2.0):

        beta = np.power(x, α - 1.0) * np.power(1.0 - x, α - 1.0)
        beta += eps
        beta /= beta.sum()
        init_vectors.append(beta)

    # 3) Triangular (linear ramp up then down)
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(float)
    tri /= tri.sum()
    init_vectors.append(tri)

    # 4) Symmetric exponential decay
    for lam in (0.001, 0.003, 0.01, 0.02):

    # --------------------------------------------------------------
    # Global best trackers
    # --------------------------------------------------------------
    global_best_score = float("inf")
    global_best_p = None

    # --------------------------------------------------------------
    # Core optimisation loops – one pass per starting vector
    # --------------------------------------------------------------
    for start_id, p0 in enumerate(init_vectors):
