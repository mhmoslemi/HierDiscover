def project_simplex(v):
    # ensure sum(v) = 1, v >= 0.
    # Implementation from Duchi et al. (2008).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

import time, numpy as np

def simplex_projection(v):
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all non-positive, return uniform
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def beta_distribution(n, a, b):
    i = np.arange(1, n+1)

def subgradient_step(a, kmax, step):
    n = len(a)
    grad = np.zeros_like(a)
    i_start = max(0, kmax - (n - 1))
    i_end = min(n - 1, kmax)
    idx_i = np.arange(i_start, i_end+1)
    idx_j = kmax - idx_i
    grad[idx_i] = 2.0 * a[idx_j]
    cand = a - step * grad
    cand = np.clip(cand, 0.0, None)
    s = cand.sum()
    if s > 0:
        cand /= s
    else:
        cand = np.full_like(a, 1.0/len(a))
    return cand
```

        # ------------------------------------------------------------------

            cand_score = best_local_score

            current[dst] += delta

            conv_max = conv.max()

                best_seq = current.copy()
                print(f"Iter {it:6d}: new global best = {best_score:.6f}")
                if best_score < target_score:
                    print("Target reached – exiting early.")
                    break
            # continue to next iteration
            # temperature annealing happens below
            accept = False  # already applied
        else:

            # using the candidate that gave the smallest increase (if any)
            # Here we simply keep the current state and rely on SA acceptance for the
            # sub‑gradient candidate (which may be worse).
            if best_cand is not None:

                cand_score = best_local_score
            else:

                break

            else:

                if np.random.rand() < prob:

            conv = full_conv_fft(current)

            current_score = 2.0 * n * conv_max

                best_seq = current.copy()
                print(f"Iter {it:6d}: new global best = {best_score:.6f}")

        # 3d. Temperature annealing
        # ------------------------------------------------------------------

    # 4.  Final exact evaluation
    # ----------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())  # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:

        exact_score = best_score

    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")

"""
Hybrid parametric sweep + projected‑gradient / simulated‑annealing refinement.

The program first explores a variety of symmetric parametric families
(U‑shaped power‑law, sum‑of‑tails, exponential, product‑power‑law and a
Beta‑distribution) for several lengths (up to 50 000).  The best
candidate (according to the cheap proxy 2 n·max(a∗a)) seeds a local
search.  The local phase repeatedly performs:

* a sub‑gradient descent step on the currently active convolution entry
  (with a small line‑search over several step sizes), and
* a batch of random “mass‑transfer” moves evaluated by an O(n) incremental
  update of the convolution peak.

Improving moves are always accepted; occasional uphill moves are accepted
according to a slowly cooling temperature.  The loop stops when the
budget (default 1000 s) is exhausted or a target score of 1.503 is
reached.  Finally the best vector is evaluated with the official
``evaluate_sequence`` routine and returned as a plain Python list.

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (L2‑norm Euclidean projection onto {x≥0, sum x = 1})
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find the largest rho where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – return uniform distribution
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges: a[i] ∝ exp(-α·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """
    Product‑power‑law shape:
        a[i] ∝ ((i+1)*(n-i))^{‑p}
    (very small values in the middle, larger at the ends).
    """
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-12
    a /= a.sum()
    return a


def beta_distribution(n: int, a_param: float, b_param: float) -> np.ndarray:
    """
    Symmetric beta‑distribution on {0,…,n‑1}:
        a[i] ∝ ( (i+1)/(n+1) )^{a‑1} · ( 1‑(i+1)/(n+1) )^{b‑1}
    Small epsilons avoid infinities for a,b<1.

    a_vec = np.maximum(x, eps) ** (a_param - 1) * np.maximum(1 - x, eps) ** (b_param - 1)
    a_vec = np.maximum(a_vec, eps)
    a_vec /= a_vec.sum()
    return a_vec

    cooling_rate = 0.995
    max_frac = 0.4                # largest fraction of mass that may be moved
    batch_moves = 80              # candidates examined for a random move
    subgrad_steps = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]  # step fractions
    target_score = 1.503
    it = 0

    print("\n=== Stochastic refinement ===")
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 3a. Sub‑gradient descent on the active convolution entry
        # --------------------------------------------------------------
        kmax = int(conv.argmax())
        grad = np.zeros_like(current)

        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        idx_j = kmax - idx_i
        grad[idx_i] = 2.0 * current[idx_j]

        best_local_score = current_score
        best_cand = None

        for frac in subgrad_steps:
            step = frac * temperature
            cand = current - step * grad
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s > 0:
                cand /= s
            else:
                cand = np.full_like(current, 1.0 / n)

            cand_score = 2.0 * n * max_conv_fft(cand)

            if cand_score < best_local_score:
                best_local_score = cand_score
                best_cand = cand

        # --------------------------------------------------------------
        # 3b. Random mass‑transfer batch (linear update of the peak)
        # --------------------------------------------------------------
        pad = np.concatenate([np.zeros(n - 1), current, np.zeros(n - 1)])
        best_move_score = current_score
        best_move = None

        for _ in range(batch_moves):
            if np.all(current == 0):
                break
            src = np.random.choice(n, p=current / current.sum())
            if current[src] <= 0.0:
                continue
            dst = np.random.randint(0, n - 1)
            if dst >= src:
                dst += 1
            delta = current[src] * np.random.rand() * max_frac
            if delta <= 0.0:
                continue

            offset_src = n - 1 - src
            offset_dst = n - 1 - dst
            shift_src = pad[offset_src: offset_src + 2 * n - 1]
            shift_dst = pad[offset_dst: offset_dst + 2 * n - 1]

            new_max = np.max(conv + 2.0 * delta * (shift_dst - shift_src))
            cand_score = 2.0 * n * new_max

            if cand_score < best_move_score:
                best_move_score = cand_score
                best_move = (src, dst, delta, new_max)

        # --------------------------------------------------------------
        # 3c. Choose and apply the best improving move (if any)
        # --------------------------------------------------------------
        accept = False

        # Prefer a sub‑gradient improvement if it exists
        if best_cand is not None and best_local_score < current_score:
            cand = best_cand
            cand_score = best_local_score
            accept = True

        # Otherwise fall back to the best random move
        elif best_move is not None and best_move_score < current_score:
            src, dst, delta, new_max = best_move
            current[src] -= delta
            current[dst] += delta
            conv = full_conv_fft(current)
            conv_max = conv.max()
            current_score = 2.0 * n * conv_max
            if current_score < best_score:
                best_score = current_score
                best_seq = current.copy()
                print(f"Iter {it:6d}: new global best = {best_score:.6f}")
                if best_score < target_score:
                    print("Target reached – exiting early.")
                    break
            accept = False   # already applied
        else:
            # No improving move – attempt a probabilistic uphill step using the
            # best (still possibly worse) sub‑gradient candidate.
            if best_cand is not None:
                cand = best_cand
                cand_score = best_local_score
                if cand_score < current_score:
                    accept = True
                else:
                    prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
                    if np.random.rand() < prob:
                        accept = True

        if accept:
            current = cand
            conv = full_conv_fft(current)
            conv_max = conv.max()
            current_score = 2.0 * n * conv_max
            if current_score < best_score:
                best_score = current_score
                best_seq = current.copy()
                print(f"Iter {it:6d}: new global best = {best_score:.6f}")
                if best_score < target_score:
                    print("Target reached – exiting early.")
                    break

        # --------------------------------------------------------------
        # 3d. Temperature cooling
        # --------------------------------------------------------------
        temperature = max(temperature * cooling_rate, 1e-6)

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (once) and report
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())  # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()
