"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises the evaluation score

    2 * n * max_k (a * a)[k]

where ``*`` is the full linear convolution.
The algorithm works in three stages:

1.  Generate a handful of deterministic seeds from simple parametric
    “U‑shaped’’ families and, if available, from the global variable
    ``height_sequence_1`` (interpolated to the required length).

2.  Run a smooth‑gradient descent on the soft‑max approximation of the
    max‑operator.  The objective is

        fγ(a) = log Σ_k exp( γ·(a*a)[k] )

    whose gradient can be evaluated by one FFT for the convolution
    ``a*a`` and one FFT for the cross‑correlation of the soft‑max
    weights with ``a``.  The parameter γ is increased slowly (homotopy)
    so that the optimisation starts very smooth and gradually tightens
    to the true max‑norm.  After each step the iterate is projected
    onto the probability simplex.

3.  When the time budget is almost exhausted a short sub‑gradient
    descent (the same idea as the previous baseline) is performed to
    polish the best solution found.

The whole routine respects the given time budget (default 1000 s) and
always returns a list of non‑negative numbers that sums to an arbitrary
positive value (the external evaluator will re‑scale it automatically).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto { w >= 0 , sum(w) = 1 }."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (real FFT). Return length 2*len(a)-1."""
    n = a.shape[0]
    size = _next_pow2(2 * n)          # ≥ 2n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def cross_correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute (w ⋆ a)[i] = Σ_k w[k+i] a[k] for i=0..len(a)-1.
    Implemented via convolution w * reverse(a) and a slice.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = _next_pow2(len(w) + n - 1)   # length of the convolution
    fw = np.fft.rfft(w, n=size)
    fa = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(fw * fa, n=size)
    start = n - 1
    return conv[start:start + n]       # length n


# ----------------------------------------------------------------------
# Parametric families (used for deterministic seeds)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on min distance to the nearest edge."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the two edges."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def softmax_gradient(a: np.ndarray, gamma: float):
    """
    Returns (grad, conv) where
        conv = a * a  (full convolution)
        grad = ∂/∂a log Σ_k exp(γ·conv_k)
    both evaluated at the current a.
    """
    # 1) convolution a*a
    conv = full_convolution_fft(a)                     # length 2n‑1
    # 2) soft‑max weights
    scaled = gamma * conv
    m = scaled.max()
    w = np.exp(scaled - m)
    w /= w.sum()                                       # w_k ≥ 0, Σ w = 1
    # 3) gradient = 2·γ·(w ⋆ a)
    grad = 2.0 * gamma * cross_correlation_fft(w, a)
    return grad, conv


# ----------------------------------------------------------------------
# Classic sub‑gradient (used for a final polishing phase)
# ----------------------------------------------------------------------
def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence inside the given
    time budget.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Determine a list of problem sizes to try.
    #    Larger n gives more freedom but each iteration is a bit slower.
    # ------------------------------------------------------------------
    lengths = [8000, 12000, 16000, 20000]

    # ------------------------------------------------------------------
    # 2. Retrieve external construction if present (global variable).
    # ------------------------------------------------------------------
    external_seq = None
    try:
        external_seq = np.array(height_sequence_1, dtype=float)
        if external_seq.ndim != 1:
            external_seq = None
    except Exception:
        external_seq = None

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Helper: evaluate the (normalized) score quickly.
    # ------------------------------------------------------------------
    def score_from_conv(conv, n):
        """Given a*a convolution (full) and length n, return the exact score."""
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # 3. Loop over chosen lengths.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Processing length n = {n} ===")

        # --------------------------------------------------------------
        # 3a. Build a modest collection of seed vectors.
        # --------------------------------------------------------------
        seeds = []

        # a) Interpolated external sequence (if we have one)
        if external_seq is not None and external_seq.size > 0:
            old_n = external_seq.size
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, external_seq)
            interp = np.maximum(interp, 0.0)
            interp /= interp.sum()
            seeds.append(interp)

        # b) Small grid over the three parametric families
        for p in (0.5, 1.0, 1.5, 2.0, 3.0):
            seeds.append(u_min_distribution(n, p))
            seeds.append(u_prod_distribution(n, p))
        for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
            seeds.append(exp_distribution(n, lam))

        # c) A few purely random points (helps escaping local minima)
        for _ in range(3):
            rnd = np.random.rand(n)
            rnd /= rnd.sum()
            seeds.append(rnd)

        # --------------------------------------------------------------
        # 3b. Run the smooth‑gradient descent for each seed.
        # --------------------------------------------------------------
        for seed_vec in seeds:
            if time.time() > deadline:
                break

            a = seed_vec.copy()
            a = np.maximum(a, 0.0)
            a /= a.sum()

            # hyper‑parameters (can be tuned)
            gamma = 1.0
            gamma_inc = 1.005          # slowly increase to tighten the max‑norm
            max_gamma = 200.0

            step = 0.5                 # step size for the projected gradient
            step_decay = 0.9995

            # simple momentum (exponential moving average of gradients)
            beta = 0.9
            momentum = np.zeros_like(a)

            # keep track of the best vector for THIS (n, seed) pair
            conv_a = full_convolution_fft(a)
            cur_score = score_from_conv(conv_a, n)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"initial seed n={n}   score={cur_score:.6f}")

            # allocate a modest number of iterations; we will break early
            # if the global time budget runs out.
            max_iters = 4000
            it = 0
            last_improve = 0

            while time.time() < deadline and it < max_iters:
                it += 1

                # ---- gradient of the log‑sum‑exp surrogate ----
                grad, conv_a = softmax_gradient(a, gamma)

                # ---- momentum update ----
                momentum = beta * momentum + (1.0 - beta) * grad

                # ---- projected gradient step ----
                cand = a - step * momentum
                cand = np.maximum(cand, 0.0)
                cand = simplex_projection(cand)
                cand /= cand.sum()

                # ---- evaluate candidate ----
                conv_cand = full_convolution_fft(cand)
                cand_score = score_from_conv(conv_cand, n)

                # ---- accept if better, otherwise Metropolis‑type acceptance ----
                accept = False
                if cand_score < cur_score:
                    accept = True
                else:
                    # probability of uphill move (temperature‑free, very mild)
                    prob = np.exp((cur_score - cand_score) / max(step, 1e-12))
                    if np.random.rand() < prob:
                        accept = True

                if accept:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = cand.copy()
                        last_improve = it
                        print(f"Iter {it:6d} (n={n})  new best = {best_score:.6f}")

                # ---- schedule updates ----
                gamma = min(gamma * gamma_inc, max_gamma)
                step *= step_decay

                # ---- re‑heat if stagnation persists ----
                if it - last_improve > 2000:
                    noise = np.random.rand(n) * 1e-6
                    a = simplex_projection(a + noise)
                    a /= a.sum()
                    conv_a = full_convolution_fft(a)
                    cur_score = score_from_conv(conv_a, n)
                    gamma = 1.0
                    step = 0.5
                    momentum.fill(0.0)
                    last_improve = it
                    print(f"Iter {it:6d} (n={n})  reheated after stagnation")

        # ------------------------------------------------------------------
        # 4. If we still have time left, perform a short classic
        #    sub‑gradient polishing on the current best vector of this n.
        # ------------------------------------------------------------------
        if time.time() < deadline and best_seq is not None:
            a = best_seq.copy()
            a = np.maximum(a, 0.0)
            a /= a.sum()
            n_local = a.shape[0]
            conv_a = full_convolution_fft(a)
            cur_score = score_from_conv(conv_a, n_local)

            step = 0.3
            step_decay = 0.9999
            it = 0
            last_improve = 0
            max_iters = 3000

            while time.time() < deadline and it < max_iters:
                it += 1
                grad = compute_subgradient(a, conv_a)
                cand = a - step * grad
                cand = np.maximum(cand, 0.0)
                cand = simplex_projection(cand)
                cand /= cand.sum()
                conv_cand = full_convolution_fft(cand)
                cand_score = score_from_conv(conv_cand, n_local)

                if cand_score < cur_score:
                    a = cand
                    conv_a = conv_cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = cand.copy()
                        last_improve = it
                        print(f"PolishIter {it:5d} n={n_local}  new best = {best_score:.6f}")

                step *= step_decay
                if it - last_improve > 1000:
                    # small random perturbation to restart
                    noise = np.random.rand(n_local) * 1e-7
                    a = simplex_projection(a + noise)
                    a /= a.sum()
                    conv_a = full_convolution_fft(a)
                    cur_score = score_from_conv(conv_a, n_local)
                    step = 0.3
                    last_improve = it
                    print(f"PolishIter {it:5d} n={n_local}  reheated (polish)")

    # ----------------------------------------------------------------------
    # 5. Return the best sequence discovered.
    # ----------------------------------------------------------------------
    print("\n=== Search finished ===")
    if best_seq is not None:
        print(f"Best score found: {best_score:.6f}  (length {len(best_seq)})")
    else:
        print("No feasible sequence was generated.")
    return best_seq.tolist() if best_seq is not None else []  # fallback empty list
