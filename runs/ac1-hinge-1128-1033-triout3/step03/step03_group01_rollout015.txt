"""
Propose a step‑function (non‑negative vector) that minimises
    2 * n * max_k (a * a)[k]

The algorithm works in three stages:

1. Deterministic search over three simple symmetric families
   (inverse distance, inverse product distance, symmetric exponential)
   for a moderately large length (default 20 000).  The best vector from
   this sweep is taken as the seed.

2. Smooth optimisation of the objective using a soft‑max
   approximation of the maximum.  The gradient of the soft‑max
   (a convex combination of all convolution entries) is obtained by a
   single FFT‑based correlation.  An Adam optimiser with projection onto
   the probability simplex drives the updates.  The soft‑max temperature
   β is increased slowly so that the optimisation gradually focuses on
   the true maximum.

3. A few simulated‑annealing style re‑heat steps are performed if
   progress stalls.

The whole procedure respects the supplied time budget (default 1000 s)
and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex { w >= 0 , sum(w) = 1 }.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft_self(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT of length `size`
    (size must be >= 2*len(a)-1).  Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def full_convolution_fft_general(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear convolution a*b using a real FFT of length `size`
    (size must be >= len(a)+len(b)-1).  Returns an array of length
    len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact value returned by ``evaluate_sequence`` for a vector whose
    sum equals 1 (the denominator becomes 1).  Uses the fast FFT.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    conv = full_convolution_fft_self(a, size)
    return float(2.0 * n * conv.max())


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to nearer edge: (min(i,n‑1‑i)+1)^{-p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the edges."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

                    best_seq = external.copy()
                    print(f"  external seed accepted, score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – should never happen
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = score_from_vector(a)
        print(f"fallback random seed, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth optimisation using soft‑max + Adam + simplex projection
    # ------------------------------------------------------------------
    current = best_seq.copy()
    # enforce exact normalisation (helps numerics)
    current /= current.sum()
    # FFT sizes (powers of two)
    size_self = 1 << ((2 * n - 1).bit_length())          # >= 2n-1
    size_corr = 1 << ((3 * n - 2).bit_length())          # >= 3n-2

    # Adam hyper‑parameters
    lr = 0.10                     # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1_pow, beta2_pow = 1.0, 1.0

    # soft‑max temperature schedule
    temper = 5.0                 # start temperature
    temper_inc = 1.001
    temper_max = 200.0

    # simulated‑annealing temperature for acceptance
    sa_temp = 1.0
    sa_decay = 0.9999

    iteration = 0
    last_improve = 0

    print("\n=== Smooth optimisation start ===")
    while time.time() < deadline:
        iteration += 1

        # ---- 4a. self‑convolution, current score ----
        fa = np.fft.rfft(current, n=size_corr)          # reuse size_corr for everything
        conv_self = np.fft.irfft(fa * fa, n=size_corr)[: 2 * n - 1]
        max_val = conv_self.max()
        current_score = 2.0 * n * max_val

        # keep global best
        if current_score < best_score:
            best_score = current_score
            best_seq = current.copy()
            last_improve = iteration
            print(f"Iter {iteration:7d}: new global best score = {best_score:.6f}")

        # ---- 4b. soft‑max weights ----
        shifted = conv_self - max_val                     # stability shift
        w = np.exp(temper * shifted)
        w /= w.sum()                                      # normalise to a probability vector

        # ---- 4c. gradient of soft‑max (correlation) ----
        a_rev = current[::-1]                             # reversed copy
        fw = np.fft.rfft(w, n=size_corr)
        far = np.fft.rfft(a_rev, n=size_corr)
        conv_grad = np.fft.irfft(fw * far, n=size_corr)[: 3 * n - 2]
        # extract the valid part: indices N‑1 .. N‑1+N‑1
        grad = 2.0 * conv_grad[n - 1 : n - 1 + n]

        # ---- 4d. Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = current - step
        cand = np.maximum(cand, 0.0)                     # keep non‑negative
        cand = simplex_projection(cand)                  # project onto simplex

        # ---- 4e. occasional random mass‑transfer move ----
        if iteration % 256 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            if cand[i] > 0.0:
                delta = cand[i] * np.random.rand() * 0.5
                cand[i] -= delta
                cand[j] += delta
                cand = simplex_projection(cand)

        # ---- 4f. evaluate candidate (exact max) ----
        fa_c = np.fft.rfft(cand, n=size_corr)
        conv_c = np.fft.irfft(fa_c * fa_c, n=size_corr)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_c.max()

        # ---- 4g. SA acceptance ----
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(sa_temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            # (conv_self will be recomputed next loop)

        # ---- 4h. update schedules ----
        sa_temp *= sa_decay
        temper = min(temper_max, temper * temper_inc)

        # ---- 4i. stagnation handling (re‑heat) ----
        if iteration - last_improve > 200_000:
            # tiny random bump and reset temperatures
            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            current /= current.sum()
            sa_temp = 1.0
            temper = 5.0
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
