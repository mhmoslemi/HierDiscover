"""
Propose a step‑function that minimises
    2 * n * max(self‑convolution) / (sum^2)

The algorithm is based on the linear‑programming refinement described in the
statement:

1. Start from a few promising probability vectors (arcsine shape, random
   Dirichlet draws and the historic best if available).
2. For a vector `p` (∑p = 1) compute the current maximal convolution value `M`.
3. Solve the LP
        maximise   Σ g_j
        subject to A(p)·g ≤ M   (self‑convolution of p with g bounded)
                  g_j ≥ 0
   where `A(p)` is the (2n‑1)×n convolution matrix of `p`.
   The optimal sum `S = Σ g` is always ≥ 1.  Re‑scale `g` to a probability
   vector `p' = g / S`.  Because the objective scales quadratically,
   the new score becomes the old one divided by `S²`, i.e. it never increases.
4. Repeat the LP step until `S` is very close to 1 or the time budget expires.
5. Optionally run a short sub‑gradient / projection loop to polish the result.
6. Return the best vector found (clipped to the limits required by
   `evaluate_sequence`).

If SciPy is unavailable the code falls back to the previous gradient‑based
heuristic and returns the arcsine distribution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (used in the fallback gradient stage)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution(p: np.ndarray, m_fft: int) -> float:
    """Return max(self‑convolution) of p using an FFT of size m_fft."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Build the (2n‑1)×n convolution matrix A(p) such that A(p)·g = p∗g.
# ----------------------------------------------------------------------
def _conv_matrix(p: np.ndarray):
    """Sparse convolution matrix for vector p (shape (2n‑1, n))."""
    try:
        import scipy.sparse as sp
    except Exception as e:
        raise RuntimeError("SciPy sparse required for LP step.") from e
    n = p.shape[0]
    # each diagonal offset k (0 … n‑1) contains the constant value p[k]
    data = [np.full(n, p[k], dtype=float) for k in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


# ----------------------------------------------------------------------
# One LP iteration: maximise Σ g subject to p∗g ≤ M, g≥0.
# Returns the optimal g and its sum.
# ----------------------------------------------------------------------
def _lp_optimize(p: np.ndarray, M: float, n: int):
    """Solve the LP for a given p and bound M.  Returns (g, sum(g))."""
    try:
        import scipy.sparse as sp
        import scipy.optimize as opt
    except Exception as e:
        raise RuntimeError("SciPy required for LP optimisation.") from e

    # Build A(p) and scale it so that the RHS is the all‑ones vector.
    A = _conv_matrix(p)
    A_scaled = A / M
    b = np.ones(2 * n - 1, dtype=float)

    # Objective: maximise Σ g  <=> minimise - Σ g
    c = -np.ones(n, dtype=float)

    # Bounds: g_j ≥ 0
    bounds = [(0, None)] * n

    # Solve with HiGHS (fast interior‑point)
    res = opt.linprog(
        c,
        A_ub=A_scaled,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    g = np.maximum(res.x, 0.0)
    return g, float(g.sum())


# ----------------------------------------------------------------------
# Construct a classic arcsine‑shaped distribution (used as a strong start).
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 1e-12
    inv = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    p = inv / inv.sum()
    return p
