"""
Propose a high‑quality step sequence.
-------------------------------------------------
This implementation builds on the previous sub‑gradient + Adam scheme but
adds several practical improvements:

* A richer set of initial vectors (arcsine, several symmetric Beta
  distributions and a few random Dirichlet draws).
* Longer sub‑gradient and Adam phases (more iterations, finer step‑size
  schedules) while always respecting the 1000 s time budget.
* A slowly increasing soft‑max temperature (α) up to a much larger value
  (≈500) so that the smoothed objective approaches the true max more
  closely.
* A final hill‑climber that performs many tiny mass‑transfer swaps.

All operations stay on the probability simplex (Σ aᵢ = 1, aᵢ ≥ 0) and
the returned list is clipped to the bounds enforced by ``evaluate_sequence``.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex.
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities – FFT based.
# ----------------------------------------------------------------------
def conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Self‑convolution of p using an FFT of size m_fft."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv

def objective(p: np.ndarray, m_fft: int) -> float:
    """Exact evaluation 2·n·max(p*p).  p is assumed to sum to 1."""
    conv = conv_self(p, m_fft)
    return float(2.0 * p.size * np.max(conv))

# ----------------------------------------------------------------------
# Gradient of the smoothed max (log‑sum‑exp) objective.
# ----------------------------------------------------------------------
def softmax_gradient(p: np.ndarray, m_fft: int, alpha: float):
    """
    Returns gradient of  (1/α)·log∑exp(α·conv)  w.r.t. p,
    together with the current convolution and its raw max.
    """
    n = p.size
    p_fft = np.fft.rfft(p, n=m_fft)

    # full self‑convolution
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

    # soft‑max weights over the convolution entries
    z = alpha * conv
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)  # stable exponentiation
    w = exp_z / np.sum(exp_z)  # Σ w = 1

    # correlation needed for the gradient (see derivation in analysis)
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]

    # mapping to gradient indices (same as the sub‑gradient case)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * corr_full[idx]   # gradient of the smoothed max
    return grad, conv, np.max(conv)

# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (use historic length if available)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        n = 1153

    # FFT size – next power of two that can hold a convolution of length 2·n‑1
    m_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Timing
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_score = float("inf")
    best_p = None

    # --------------------------------------------------------------
    # Build a pool of promising initial vectors
    # --------------------------------------------------------------
    init_vectors = []

    # 1) Arcsine shape (a = 0.5)
    i = np.arange(n, dtype=float)
    eps = 1e-12
    inv = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    p_arcsine = inv / inv.sum()
    init_vectors.append(p_arcsine)

    # 2) Symmetric Beta‑like shapes (α in a range)
    beta_alphas = [0.3, 0.4, 0.5, 0.6, 0.8, 1.2, 1.5, 2.0]
    for a in beta_alphas:
        w = ((i + 0.5) * (n - 0.5 - i) + eps) ** (a - 1.0)
        if w.sum() == 0:
            continue
        p_beta = w / w.sum()
        init_vectors.append(p_beta)

    # 3) Random Dirichlet draws (moderate concentration)
    for _ in range(6):
        p_dir = rng.dirichlet(np.ones(n))
        init_vectors.append(p_dir)

    # 4) Historic construction if present
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        seq = np.clip(seq, 0.0, 1000.0)
        s = seq.sum()
        if s > 0:
            init_vectors.append(seq / s)

    # --------------------------------------------------------------
    # Core optimisation loops
    # --------------------------------------------------------------
    for start_id, p0 in enumerate(init_vectors):
        if time.time() >= deadline:
            break

        print(f"=== Start {start_id + 1}/{len(init_vectors)} =====================")
        p = p0.copy()

        # ----------------------------------------------------------
        # (a) Sub‑gradient descent on the *actual* max entry
        # ----------------------------------------------------------
        step = 0.2
        decay = 0.9995
        min_step = 1e-8
        for it in range(8000):
            if time.time() >= deadline:
                break

            conv = conv_self(p, m_fft)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                print(f"  Sub‑grad iter {it:5d} – new best {best_score:.6f}")

            # gradient of the maximal entry (sub‑gradient)
            max_idx = conv.argmax()
            g = np.zeros_like(p)
            if max_idx < n:
                g[: max_idx + 1] = 2.0 * p[max_idx :: -1]
            else:
                start = max_idx - n + 1
                g[start:] = 2.0 * p[n - 1 : start - 1 : -1]

            p = p - step * g
            p = simplex_projection(p)

            step = max(step * decay, min_step)

        # ----------------------------------------------------------
        # (b) Adam optimisation on a smooth log‑sum‑exp approximation
        # ----------------------------------------------------------
        m1 = np.zeros_like(p)
        m2 = np.zeros_like(p)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lr = 0.05

        alpha = 0.5
        alpha_mul = 1.0015
        alpha_max = 500.0

        max_adam_iters = 120_000
        for it in range(max_adam_iters):
            if time.time() >= deadline:
                break

            grad, conv, max_val = softmax_gradient(p, m_fft, alpha)

            # Adam state update
            m1 = beta1 * m1 + (1.0 - beta1) * grad
            m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

            m_hat = m1 / (1.0 - beta1 ** (it + 1))
            v_hat = m2 / (1.0 - beta2 ** (it + 1))

            step_vec = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            p = p - step_vec
            p = simplex_projection(p)

            # increase smoothing temperature slowly
            alpha = min(alpha * alpha_mul, alpha_max)

            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                print(f"  Adam iter {it:6d} – new best {best_score:.6f}")

        # ----------------------------------------------------------
        # (c) Tiny local swap hill‑climber (already done later on global best)
        # ----------------------------------------------------------

    # ------------------------------------------------------------------
    # Final local refinement on the *global* best vector
    # ------------------------------------------------------------------
    if best_p is None:
        # Fallback: uniform distribution (should never happen)
        best_p = np.full(n, 1.0 / n)
        best_score = objective(best_p, m_fft)

    p = best_p.copy()
    cur_score = objective(p, m_fft)

    swap_attempts = 30_000
    for _ in range(swap_attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a small random chunk of mass
        eps_move = 1e-4 * rng.random()
        if p[i] <= eps_move:
            continue
        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move
        new_score = objective(new_p, m_fft)
        if new_score < cur_score:
            p = new_p
            cur_score = new_score
            best_p = p.copy()
            best_score = cur_score
            print(f"Swap improvement → {best_score:.6f}")
            # stop early if we already beat the target
            if best_score <= 1.503:
                break

    print(f"Final best score: {best_score:.6f}")

    # Clip to match the limits of evaluate_sequence and return as a list
    return np.clip(best_p, 0.0, 1000.0).tolist()
