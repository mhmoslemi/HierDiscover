def smoothed_max_and_grad(x, eps):
    n = len(x)
    size = 1 << ((2*n-1).bit_length())
    # compute convolution
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[:2*n-1]
    max_conv = conv.max()
    # compute smoothed max weight
    # sub max for numerical stability
    exp_vals = np.exp((conv - max_conv) / eps)
    sum_exp = exp_vals.sum()
    # f = max_conv + eps * log(sum_exp)
    f = max_conv + eps * np.log(sum_exp)

"""
Propose a candidate step‑function (non‑negative vector) that minimises
evaluate_sequence().

Algorithm
---------

1.  Initialise from the best known global vector (height_sequence_1 if
    present) and upscale it to a target length.  If none is present start
    from a simple “U‑shaped’’ family.

2.  Optimise the vector on the probability simplex with a projected Adam
    optimiser.  The objective is a smooth approximation of the
    max‑convolution entry:
        f_ε(x) = ε·log Σ_k exp( (x*x)_k / ε )
    whose gradient can be computed by two FFTs (convolution + a
    cross‑correlation).  ε is decreased slowly (annealing).

3.  Every few thousand steps a tiny random jitter is added and the vector
    is re‑projected – this helps escape shallow plateaus.

4.  The optimisation runs for the whole time budget (≈1000 s), possibly
    on several lengths; the best vector found (according to the exact
    evaluation score) is returned.

The implementation is fully vectorised, uses only NumPy and respects the
time limit.  No file/network I/O is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum w = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper families for deterministic initialisation (optional)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric U‑shape based on distance to the nearest edge."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric U‑shape based on product of distances to both edges."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Upscaling a vector to a different length (linear interpolation)
# ----------------------------------------------------------------------
def upscale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a probability vector to length new_n using linear interpolation."""
    if seq.ndim != 1:
        raise ValueError("seq must be 1‑D")
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    # protect against tiny negatives caused by floating point noise
    new_seq = np.maximum(new_seq, 0.0)
    new_seq_sum = new_seq.sum()
    if new_seq_sum > 0:
        new_seq /= new_seq_sum
    else:
        new_seq = np.full(new_n, 1.0 / new_n)
    return new_seq


# ----------------------------------------------------------------------
# Smoothed max of the convolution and its gradient (FFT‑based)
# ----------------------------------------------------------------------
def smoothed_max_and_grad(x: np.ndarray, eps: float):
    """
    Compute a smooth approximation of max_k (x*x)[k] and its gradient.

    Parameters
    ----------
    x : np.ndarray, shape (n,)
        Probability vector (sum = 1, non‑negative).
    eps : float
        Smoothing parameter (ε > 0).  Smaller eps -> closer to the true max.

    Returns
    -------
    f : float
        Smooth approximation of max_k (x*x)[k].
    grad : np.ndarray, shape (n,)
        Gradient of f w.r.t. x.
    max_conv : float
        Exact maximum entry of the linear convolution (x*x).
    """
    n = x.shape[0]

    # FFT length that accommodates both the 2n‑1 convolution and the
    # (2n‑1)+n‑1 = 3n‑2 cross‑correlation.
    size = 1 << ((3 * n - 2).bit_length())

    # ----- convolution x * x (linear, not circular) -----
    fx = np.fft.rfft(x, n=size)
    conv_full = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
    max_conv = conv_full.max()

    # ----- smooth max (log‑sum‑exp) -----
    # subtract max for numerical stability
    exp_vals = np.exp((conv_full - max_conv) / eps)
    sum_exp = exp_vals.sum()
    f = max_conv + eps * np.log(sum_exp)          # smooth max value
    w_norm = exp_vals / sum_exp                   # softmax weights (sum = 1)

    # ----- gradient: 2 * (w_norm cross‑correlated with x) -----
    w_rev = w_norm[::-1]                         # reverse for cross‑correlation
    fw = np.fft.rfft(w_rev, n=size)
    # reuse fx (FFT of x)
    corr_full = np.fft.irfft(fw * fx, n=size)[: len(w_rev) + n - 1]
    start = len(w_rev) - 1
    grad = 2.0 * corr_full[start:start + n]

    return f, grad, max_conv


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (projected Adam)
# ----------------------------------------------------------------------
def optimise_on_simplex(
    n: int,
    init: np.ndarray,
    time_budget: float,
    target_score: float = 1.503,
):
    """
    Run projected Adam for a fixed length n starting from init.

    Returns
    -------
    best_vec : np.ndarray, shape (n,)
        Vector that achieved the lowest exact evaluation score.
    best_score : float
        Corresponding evaluation score.
    """
    # Normalise initial vector to a probability simplex
    x = np.maximum(init, 0.0)
    sx = x.sum()
    if sx == 0:
        x = np.full(n, 1.0 / n)
    else:
        x /= sx

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05

    # smoothing schedule
    eps_smooth = 0.02          # start fairly smooth
    eps_decay = 0.9995

    # keep track of best solution
    _, _, max_conv = smoothed_max_and_grad(x, eps_smooth)
    best_score = 2.0 * n * max_conv
    best_vec = x.copy()
    print(f"[{n}] start score = {best_score:.6f}")

    # Adam bias‑correction accumulators
    beta1_pow = 1.0
    beta2_pow = 1.0

    it = 0
    last_improve = 0
    deadline = time.time() + time_budget

    while time.time() < deadline:
        it += 1
        # gradually tighten the smoothing
        eps_smooth = max(eps_smooth * eps_decay, 1e-6)

        # compute smooth objective + gradient
        _, grad, max_conv = smoothed_max_and_grad(x, eps_smooth)

        # exact evaluation score (what the final judge uses)
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_vec = x.copy()
            last_improve = it
            print(f"[{n}] iter {it:7d}  new best = {best_score:.6f}")

        # stop early if we already beat the target
        if best_score < target_score:
            print(f"[{n}] target met (score {best_score:.6f}) – terminating early.")
            break

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand = x - step

        # projection onto simplex (ensures non‑negativity & sum = 1)
        cand = simplex_projection(cand)

        # tiny random jitter every few thousand iterations to avoid dead‑ends
        if it % 5000 == 0:
            jitter = (np.random.rand(n) - 0.5) * 1e-4
            cand = simplex_projection(cand + jitter)

        # renormalise exactly (projection may introduce minute drift)
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            cand = np.full(n, 1.0 / n)

        x = cand

        # simple re‑heat if no improvement for a long while
        if it - last_improve > 150000:
            x = simplex_projection(x + np.random.rand(n) * 1e-5)
            x /= x.sum()
            m[:] = 0.0
            v[:] = 0.0
            beta1_pow, beta2_pow = 1.0, 1.0
            last_improve = it
            print(f"[{n}] iter {it:7d}  re‑heat (stagnation)")

    return best_vec, best_score

    # ------------------------------------------------------------------
