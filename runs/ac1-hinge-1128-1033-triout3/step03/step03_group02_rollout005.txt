approximation of the ∞‑norm.  The algorithm works on the
probability simplex (a_i ≥ 0, Σ a_i = 1).  At each iteration we

* compute the full auto‑convolution a*a by a real FFT,

* renormalise to the simplex.

The temperature τ is annealed from a sizeable fraction of the
current maximum toward a tiny value, turning the smooth objective

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # tiny negative values caused by round‑off are set to zero
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient from soft‑max weights (same as the baseline)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute  g_i = 2 * Σ_k w_k * a_{k-i}
    via a single convolution: reverse w, convolve with a,
    then take the proper slice.
    """

    # start = 2n‑2, then step backwards
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g

    eta_end   = 1e-4                # final step size
    tau_factor_start = 0.30         # τ ≈ factor * max_conv at the beginning
    tau_factor_end   = 1e-5         # final factor (practically zero)

    beta = 0.90                     # momentum decay
    jitter_prob = 0.015             # probability of a random mass‑transfer

    iteration = 0
    eps = 1e-15                     # to avoid divide‑by‑zero in logs

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- convolution and current objective value ----
        conv_a = _full_convolution_fft(a, a)      # length 2n‑1
        M = conv_a.max()
        cur_val = float(2.0 * n * M)

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # progress ∈ [0,1] for schedules
        progress = (now - start_time) / (deadline - start_time)

        # ---- temperature for the soft‑max (annealed) ----
        tau = (tau_factor_start * (1.0 - progress) +
               tau_factor_end   * progress) * M
        if tau <= 0:
            # fallback to a hard sub‑gradient
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0
        else:
            # stable soft‑max: shift by the maximum to avoid overflow
            exp_args = (conv_a - M) / tau
            # clip extreme values for safety
            np.clip(exp_args, -80, 80, out=exp_args)
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv_a)
                w[np.argmax(conv_a)] = 1.0
            else:
                w = exp_vals / sum_exp

        # ---- gradient of the smoothed objective (before scaling) ----
        grad = _gradient_from_weights(a, w)      # shape (n,)

        # ---- momentum update (exponential moving average) ----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- step‑size schedule (multiplicative) ----
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # ---- multiplicative (exponentiated) update ----
        # limit the exponent to avoid overflow/underflow
        step = -eta * momentum
        np.clip(step, -50, 50, out=step)
        a = a * np.exp(step)

        # ---- occasional random jitter (helps escape plateaux) ----
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # move a random fraction of the mass at src to dst
            delta = np.random.rand() * a[src] * 0.5
            a[src] -= delta
            a[dst] += delta

        # ---- projection back onto the simplex (simple renormalisation) ----
        # keep a strictly positive to stay in the interior of the simplex
        a = np.maximum(a, eps)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        iteration += 1

        # occasional full recomputation to keep numerical drift tiny
        if iteration % 2500 == 0:
            conv_a = _full_convolution_fft(a, a)   # sanity check, ignored

    # --------------------------------------------------------------
    # Final reporting (optional – the outer framework will also call it)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
