"""
Propose a candidate step-function sequence that minimizes the 
maximal entry of its self‑convolution (equivalently the 
`evaluate_sequence` objective).  
The algorithm starts from the classic central‑binomial (Catalan) 
construction, then repeatedly solves a linear program that maximises 
the ℓ₁‑norm of a partner sequence while keeping the convolution bound 
fixed.  The partner is rescaled to the required normalisation and mixed 
with the current sequence; the mixing weight is chosen by a cheap line
search that directly minimises the maximal convolution entry.
Occasional random restarts avoid stagnation.  All operations respect the
security checks of `evaluate_sequence`.
"""

import time
import numpy as np

# Try importing the optimisation tools; fall back to a simple sub‑gradient
# method if they are unavailable.
try:
    import scipy.optimize as opt
    from scipy.sparse import coo_matrix
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


def _project_to_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project a non‑negative vector onto the affine set {x ≥ 0, sum(x)=target_sum}."""
    # First truncate negatives
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0:
        # Return uniform distribution if everything became zero.
        return np.full_like(v, target_sum / v.size)
    return v * (target_sum / total)


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current best bound (≈ 1.5197).  The routine runs for at most
    ``budget_s`` seconds and prints progress information.
    """
    # ------------------------------------------------------------------
    # 0) housekeeping
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 2.0  # a small safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) pick a problem size and initialise the sequence.
    # ------------------------------------------------------------------
    # If a previous construction is available we can re‑use its length.
    global height_sequence_1
    if 'height_sequence_1' in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            if init.size > 0:
                a = init.copy()
                n = a.shape[0]
                # normalise to the required sum S = sqrt(2n)
                S = np.sqrt(2.0 * n)
                a = _project_to_sum(a, S)
                print(f"[init] Using external sequence of length {n}.")
            else:
                raise ValueError
        except Exception:
            # fall back to the classic construction
            height_sequence_1 = None
            a = None
    else:
        a = None

    if a is None:
        n = 2000                     # default length – fits comfortably in memory
        S = np.sqrt(2.0 * n)         # target ℓ₁‑norm (the evaluator uses it internally)
        # central binomial coefficients a_k = C(2k,k)/4**k, truncated at n
        coeffs = np.empty(n, dtype=np.float64)
        cur = 1.0
        coeffs[0] = cur
        for k in range(1, n):
            cur = cur * (2 * k - 1) / (2 * k)
            coeffs[k] = cur
        a = coeffs / coeffs.sum() * S
        print(f"[init] Central‑binomial start, n={n}")

    # ------------------------------------------------------------------
    # 2) helper data for the linear programme (rows/cols are constant)
    # ------------------------------------------------------------------
    # Build the index arrays that describe the convolution matrix.
    # For each column i (0 ≤ i < n) the value a[i] appears on rows i … i+n‑1.
    idx_i = np.repeat(np.arange(n, dtype=np.int32), n)          # column index, repeated
    idx_offset = np.tile(np.arange(n, dtype=np.int32), n)       # 0 … n‑1 repeated for each column
    rows = idx_i + idx_offset                                   # row index for each non‑zero entry
    cols = idx_i                                                # column index (same as idx_i)
    shape = (2 * n - 1, n)                                      # full convolution size

    # ------------------------------------------------------------------
    # 3) core optimisation loop – LP + line‑search mixing
    # ------------------------------------------------------------------
    best_seq = a.copy()
    best_conv = np.convolve(a, a)
    best_M = best_conv.max()          # with our normalisation, this *is* the objective
    best_val = best_M
    print(f"[state] initial objective = {best_val:.9f}")

    iteration = 0
    no_improve = 0
    max_no_improve = 25               # after this many failed mixes we restart
    # coarse grid for mixing parameter t
    t_grid = np.linspace(0.0, 1.0, 21)

    while time.time() < deadline:
        iteration += 1

        # ------------------------------------------------------------------
        # 3a) compute the current max‑convolution value M
        # ------------------------------------------------------------------
        conv_a = np.convolve(a, a)
        M = conv_a.max()

        # ------------------------------------------------------------------
        # 3b) build the LP that maximises sum(g) under the bound (a * g) ≤ M
        # ------------------------------------------------------------------
        # data vector for the sparse matrix – simply a repeated for each row of its block
        data = np.repeat(a, n)  # length n*n
        A = coo_matrix((data, (rows, cols)), shape=shape, dtype=np.float64).tocsr()
        b_ub = np.full(shape[0], M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)   # minimise –Σg  ⇔ maximise Σg

        # solve
        if _SCIPY_OK:
            res = opt.linprog(c, A_ub=A, b_ub=b_ub,
                              bounds=[(0.0, None)] * n,
                              method='highs',
                              options={'presolve': True,
                                       'time_limit': max(deadline - time.time(), 0.0)})
        else:
            # If SciPy is unavailable fall back to a trivial “do nothing’’ step.
            res = None

        if res is None or not res.success:
            # The LP should always be feasible (a itself works).  If something went
            # wrong we simply skip this iteration.
            print(f"[lp] iteration {iteration} failed – skipping")
            no_improve += 1
            if no_improve >= max_no_improve:
                a = rng.random(n)
                a = _project_to_sum(a, S)
                no_improve = 0
                print(f"[restart] LP failure at iter {iteration}")
            continue

        g0 = np.maximum(res.x, 0.0)          # numerical safety
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            # Degenerate solution – restart
            a = rng.random(n)
            a = _project_to_sum(a, S)
            no_improve = 0
            print(f"[restart] zero‑sum LP at iter {iteration}")
            continue

        # ------------------------------------------------------------------
        # 3c) rescale g0 so that it has the required ℓ₁‑norm S
        # ------------------------------------------------------------------
        scale = S / sum_g0
        g = g0 * scale

        # ------------------------------------------------------------------
        # 3d) line‑search for the best convex combination of a and g
        # ------------------------------------------------------------------
        conv_g = np.convolve(g, g)
        conv_ag = np.convolve(a, g)

        # quadratic coefficients for each convolution entry
        #   h(t) = (1‑t)²·conv_a + 2t(1‑t)·conv_ag + t²·conv_g
        a_arr = conv_a
        b_arr = conv_ag
        c_arr = conv_g

        best_t = 0.0
        best_t_M = M

        for t in t_grid:
            mixed = (1 - t) ** 2 * a_arr + 2 * t * (1 - t) * b_arr + t ** 2 * c_arr
            cur_M = mixed.max()
            if cur_M < best_t_M - 1e-12:
                best_t_M = cur_M
                best_t = t

        # optional refinement around the best coarse point
        if 0.0 < best_t < 1.0:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.05)
            for _ in range(8):
                t1 = lo + (hi - lo) / 3.0
                t2 = hi - (hi - lo) / 3.0
                m1 = ((1 - t1) ** 2 * a_arr + 2 * t1 * (1 - t1) * b_arr + t1 ** 2 * c_arr).max()
                m2 = ((1 - t2) ** 2 * a_arr + 2 * t2 * (1 - t2) * b_arr + t2 ** 2 * c_arr).max()
                if m1 < m2:
                    hi = t2
                else:
                    lo = t1
            t_opt = (lo + hi) / 2.0
            val_opt = ((1 - t_opt) ** 2 * a_arr + 2 * t_opt * (1 - t_opt) * b_arr + t_opt ** 2 * c_arr).max()
            if val_opt < best_t_M - 1e-12:
                best_t = t_opt
                best_t_M = val_opt

        # ------------------------------------------------------------------
        # 3e) accept the new mixture if it improves the objective
        # ------------------------------------------------------------------
        if best_t_M < M - 1e-12:
            a = (1 - best_t) * a + best_t * g
            # re‑normalise (tiny drift may appear)
            a = _project_to_sum(a, S)

            if best_t_M < best_val - 1e-12:

"""
Propose a candidate step‑function sequence that minimises the maximal entry
of its self‑convolution (equivalently the `evaluate_sequence` objective).
The algorithm starts from the classic central‑binomial (Catalan) construction,
then repeatedly solves a linear programme that maximises the ℓ₁‑norm of a
partner sequence while keeping the convolution bound fixed.  The partner is
rescaled to the required normalisation and mixed with the current sequence;
the mixing weight is chosen by a cheap line‑search that directly minimises
the maximal convolution entry.  Occasional random restarts avoid stagnation.
All numbers are non‑negative and respect the limits imposed by the evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports – fall back to a simple sub‑gradient method if missing
# ----------------------------------------------------------------------
try:
    import scipy.optimize as opt
    from scipy.sparse import coo_matrix
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


def _project_to_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project a non‑negative vector onto the affine set {x ≥ 0, sum(x)=target_sum}."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, target_sum / v.size)
    return v * (target_sum / total)

    Return a list of heights (non‑negative) that (hopefully) improves on the
    previous best bound (~1.5197).  The routine runs for at most
    ``budget_s`` seconds and prints progress information.

    # ------------------------------------------------------------------
    # 1) Choose problem size and initialise the sequence
    # ------------------------------------------------------------------

    else:

    if a is None:
        n = 2000                      # default length – comfortably fits in memory

        # Central binomial coefficients a_k = C(2k,k)/4**k, truncated at n
        coeffs = np.empty(n, dtype=np.float64)

    idx_off = np.tile(np.arange(n, dtype=np.int32), n)          # 0 … n‑1 repeated for each column
    rows = idx_i + idx_off                                      # row indices of the non‑zeros
    cols = idx_i                                                # column indices (same as idx_i)

    while time.time() < deadline:
