"""
Propose a step‑function (non‑negative, sum‑normalised) that minimises
`evaluate_sequence`.  

The search consists of three stages:

1. **Deterministic sweep** over several parametric families
   (U‑shapes, beta‑shapes, exponential, Gaussian).  The best vector
   found becomes the seed for the refinement.

2. **Smooth‑max gradient descent**:  we minimise a smooth approximation
   of the maximum entry of the self‑convolution.
   For a probability vector `a` (sum = 1) let  
       `c = a * a`  (linear convolution).  
   The quantity to minimise is `max(c)`.  We replace `max` by a
   log‑sum‑exp with temperature `α` and compute its gradient analytically
   (using FFT for the correlation terms).  The update is a
   multiplicative (exponentiated‑gradient) step followed by normalisation,
   i.e. a mirror‑descent on the simplex.

3. **Stochastic reheating**: if no improvement is observed for a long
   time we add a tiny random jitter and reset the step‑size.

All heavy operations are performed with FFTs, allowing us to explore
millions of candidates within the 1000 s budget.  The final vector is
returned as a plain Python list; `evaluate_sequence` will clip and
renormalise it automatically.
"""

# Utility functions (all top‑level, no closures)

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x >= 0, sum(x)=1}.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full convolution of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation r[i] = Σ_j a[j] * w[i+j] for i = 0…len(a)-1.
    Both inputs are real, non‑negative and zero‑padded outside their support.
    """
    n = a.shape[0]
    size = next_pow2(n + w.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w, n=size)
    corr_full = np.fft.irfft(fa * np.conj(fw), n=size)
    return corr_full[:n]  # only the first n lags are needed


def grad_softmax_max(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth max (log‑sum‑exp) of the self‑convolution.
    Returns a vector of the same length as `a`.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_val = conv.max()
    # stable soft‑max weights
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    w /= w_sum

    # term Σ_j w[i+j] * a[j]  (cross‑correlation)
    term1 = correlation_fft(a, w)

    # term a[i] * Σ_{j=0}^{n-1} w[i+j]  (moving‑window sum)
    window_sum = np.convolve(w, np.ones(n), mode='valid')
    term2 = a * window_sum

    return term1 + term2

def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Symmetric discrete Gaussian centred at the middle."""
    centre = (n - 1) / 2.0
    x = np.arange(n) - centre
    a = np.exp(-0.5 * (x / sigma) ** 2) + 1e-14
    a /= a.sum()
    return a

        conv_new = full_convolution_fft(a_new)
        score_new = 2.0 * n * conv_new.max()

        # ---- acceptance & bookkeeping ---------------------------------
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            last_improve = iter_count
            print(f"Iter {iter_count:7d}: new best score = {best_score:.6f}")

        # always move forward (the update is a descent direction in expectation)
        a = a_new

        # ---- schedule updates -----------------------------------------
        lr *= lr_decay
        alpha *= alpha_mul

        # ---- stagnation handling --------------------------------------
        if iter_count - last_improve > 30000:
            # add a tiny random jitter and reset the step parameters
            jitter = np.random.rand(n) * 1e-5
            a = a + jitter
            a = simplex_projection(a)   # re‑project to the simplex
            lr = 0.5
            alpha = 5.0
            last_improve = iter_count
            print(f" * Re‑heat at iter {iter_count}")

    # ------------------------------------------------------------------
    # 5. Return the best vector discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_a)})")
    # `evaluate_sequence` applies its own clipping/normalisation, so we return
    # the raw normalised vector.
    return best_a.tolist()
```
