"""
propose_candidate
-----------------

A hybrid optimizer that mixes a *soft‑max* approximation of the
max‑convolution objective with a gradient of the L2‑norm of the
convolution.  The smooth max is
      fτ(a)= (1/τ)·log Σk exp(τ·(a*a)k) .
Its gradient is 2·(p⊛arev) where p is the soft‑max of the
convolution entries and ⊛ denotes convolution.  Adding the L2‑gradient
(∇‖a*a‖₂² = 4·((a*a)⊛arev)) encourages a flatter convolution
profile.  The update uses Adam on the combined gradient and each
iteration projects back onto the probability simplex (∑a=1, a≥0).
A simple schedule raises τ during the run, while the learning rate
decays.  The best sequence seen before the 1000 s deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x | x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth‑max of the convolution.
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of   (1/τ)·log Σ exp(τ·conv_k)   w.r.t. a.
    It equals 2 * (p ⊛ a_rev) where p is the soft‑max of conv.
    """
    # soft‑max weights, numerically stable
    exp_term = np.exp(tau * conv - np.max(tau * conv))
    p = exp_term / np.sum(exp_term)
    # convolution p * reversed a, then slice the middle n entries
    grad = 2.0 * np.convolve(p, a[::-1])[len(a) - 1:len(a) - 1 + len(a)]
    return grad


# ----------------------------------------------------------------------
# Gradient of the L2‑norm of the convolution (∥a*a∥₂²).
# ----------------------------------------------------------------------
def _grad_l2_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k (conv_k)²   w.r.t. a.
    It equals 4 * ((conv) ⊛ a_rev).
    """
    grad = 4.0 * np.convolve(conv, a[::-1])[len(a) - 1:len(a) - 1 + len(a)]
    return grad


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a step‑function (list of non‑negative heights) that
    minimises the evaluation function within the allotted time.
    """
    deadline = time.time() + budget_s - 5.0          # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # 2) Initialise – use the classic Catalan coefficients or a given
    #    global sequence if it is long enough.
    # --------------------------------------------------------------
    global height_sequence_1
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, list) \
            and len(height_sequence_1) >= n:
        a = np.array(height_sequence_1[:n], dtype=np.float64)
        a = _project_simplex(a)                     # force simplex
    else:
        # Catalan (central‑binomial) coefficients, normalized to sum = 1
        coeffs = np.empty(n, dtype=np.float64)
        cur = 1.0
        coeffs[0] = cur
        for k in range(1, n):
            cur = cur * (2 * k - 1) / (2 * k)
            coeffs[k] = cur
        a = coeffs / coeffs.sum()                   # simplex projection

    # best solution seen so far
    conv = np.convolve(a, a)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_seq = a.copy()
    print(f"[init] n={n}, initial_score={best_val:.10f}")

    # --------------------------------------------------------------
    # 3) Adam hyper‑parameters and schedules
    # --------------------------------------------------------------
    lr = 0.05               # base learning rate
    min_lr = 1e-6
    decay = 0.9995          # geometric decay per iteration

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)    # first moment
    v = np.zeros_like(a)    # second moment
    beta1_pow = 1.0
    beta2_pow = 1.0

    # smoothing‑parameter τ for the soft‑max (starts small, grows)
    tau = 0.5
    tau_max = 30.0
    tau_inc = (tau_max - tau) / 20000.0    # reach τ≈max after ~20 k iters

    # relative weight of the L2‑gradient (flatness encouragement)
    alpha = 0.30          # 0 → pure soft‑max, 1 → pure L2‑gradient

    it = 0
    no_improve = 0

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- convolution of the current iterate -----
        conv = np.convolve(a, a)
        max_val = conv.max()

        # ----- gradients -------------------------------------------------
        # smooth‑max gradient
        grad_max = _grad_smooth_max(a, conv, tau)

        # L2‑gradient (flattening term)
        grad_l2 = _grad_l2_conv(a, conv)

        # blended gradient
        g = (1.0 - alpha) * grad_max + alpha * grad_l2

        # ----- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)   # back to the simplex

        # ----- evaluate the candidate ------------------------------------
        conv_new = np.convolve(a_new, a_new)
        val = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a_new.copy()
            no_improve = 0
            print(f"[iter {it:6d}] score={best_val:.10f}   step={lr:.6f}   tau={tau:.2f}")
        else:
            no_improve += 1

        a = a_new

        # ----- schedules --------------------------------------------------
        lr = max(min_lr, lr * decay)
        tau = min(tau_max, tau + tau_inc)

        # occasionally re‑normalise to avoid drift of the sum
        if it % 1000 == 0:
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0

        # ----- restart if we are stuck -----------------------------------
        if no_improve > 8000:
            # perturb the best sequence slightly and resume
            a = best_seq + 0.02 * rng.normal(size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            lr = 0.05
            tau = 0.5
            no_improve = 0
            print(f"[restart] iteration {it}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clamped to the limits of the evaluator)
    # --------------------------------------------------------------
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(best_seq)
    print(f"[result] final_score={final_score:.10f}, length={len(best_seq)}")
    return best_seq
