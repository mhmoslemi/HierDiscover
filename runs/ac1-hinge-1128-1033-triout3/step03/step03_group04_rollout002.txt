"""
Propose a low‑score step‑function.

The algorithm works in two phases:

1️⃣  Deterministic sweep – several parametric families (U‑shape,
    beta, exponential, one‑sided power law and a few random Dirichlet
    draws) are evaluated for a set of lengths.  The best vector found
    becomes the seed for the next phase.  Optional external construction
    `height_sequence_1` is also considered.

2️⃣  Stochastic refinement – a projected Adam‑style gradient descent on
    the objective  f(a)=max(a∗a)  (self‑convolution) is performed.
    The sub‑gradient is built from the *active* convolution entries
    (those within a tiny relative tolerance of the current maximum).
    The gradient is obtained efficiently with an FFT‑based convolution
    between the reversed current vector and a binary mask of the active
    indices.  After each Adam step the iterate is projected onto the
    probability simplex.  Simulated‑annealing acceptance and occasional
    “re‑heat” (small random jitter) help escape plateaus.

The whole search respects the 1000 s wall‑clock budget; the routine
returns the best sequence discovered (as a plain list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities (linear convolution)
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length 2*len(a)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is non‑negative,
    sum=1)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a vector already
    normalised to sum‑one (2·n·max(a*a))."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector, i.e. sum=1)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge:
    (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the two edges:
    ((i+1)*(n-i))^(-p)."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def harmonic_left_distribution(n: int, p: float) -> np.ndarray:
    """One‑sided power law (mass concentrated on the left)."""
    i = np.arange(n) + 1
    a = i.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
