"""
Propose candidate sequence for the convolution‐max problem.

Algorithm
---------

1. **Initialisation** –  Use the globally available `height_sequence_1`
   (if present) as a starting point; otherwise fall back to a simple
   U‑shaped distribution.  All vectors are normalised to sum to 1.

2. **Smooth L‑BFGS optimisation** –  Replace the nonsmooth objective
   `2·n·max(a*a)` with the smooth surrogate
       (2·n/α)·log‑sum‑exp(α·(a*a)) .
   For a schedule of increasing smoothing parameters `α` we run a
   projected L‑BFGS optimisation on the simplex by parametrising the
   vector `a` with a soft‑max (`a_i = exp(u_i)/∑exp(u)`).  The gradient
   of the surrogate is obtained analytically using FFT‑based cross
   correlation, yielding fast O(n log n) iterations even for n≈10⁴.

3. **Sub‑gradient refinement** –  After the smooth phases a short
   projected sub‑gradient descent with simulated‑annealing acceptance
   cleans up any remaining slack in the true objective.

4. **Return the best sequence** –  The routine keeps the best (lowest)
   exact score seen during the whole run and returns it as a plain
   Python list before the time budget expires.
"""

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities (all top‑level)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {w≥0, sum w = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    # length of linear convolution is 2*n-1, pad to 2*n for FFT convenience
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector with sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices attaining the maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric “U‑shape’’ based on distance to nearest edge."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1.0) ** (-p) + 1e-14
    return a / a.sum()


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric “U‑shape’’ based on product of distances to both edges."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    return a / a.sum()


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    return a / a.sum()


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Smooth surrogate for the max‑convolution objective
# ----------------------------------------------------------------------


class SmoothObjective:
    """
    Callable object that evaluates the smooth surrogate
        (2·n/α)·log‑sum‑exp(α·(a*a))
    and its gradient w.r.t. the soft‑max parameters u (a = softmax(u)).
    """

    def __init__(self, n: int, alpha: float):
        self.n = n
        self.alpha = alpha
        self.len_w = 2 * n - 1  # length of convolution a*a
        # size for FFT when computing cross‑correlation (w ⋆ a_rev)
        self.fft_size = next_pow2(self.len_w + n - 1)

    def __call__(self, u: np.ndarray):
        # --- a = softmax(u) -------------------------------------------
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        a = exp_u / exp_u.sum()                     # sum(a)=1, a≥0

        # --- convolution a*a -----------------------------------------
        conv = full_convolution_fft(a)              # length 2*n-1

        # --- log‑sum‑exp of α·conv -----------------------------------
        y = self.alpha * conv
        max_y = np.max(y)
        exp_y = np.exp(y - max_y)
        sumexp = np.sum(exp_y)
        logsumexp = max_y + np.log(sumexp)
        obj = (2.0 * self.n / self.alpha) * logsumexp

        # --- weights w = softmax(α·conv) -------------------------------
        w = exp_y / sumexp                          # sum(w)=1, length 2*n-1

        # --- gradient w.r.t a (via cross‑correlation) ---------------
        # compute cross‑correlation w ⋆ a_rev  (linear convolution)
        a_rev = a[::-1]
        wf = np.fft.rfft(w, self.fft_size)
        af = np.fft.rfft(a_rev, self.fft_size)
        conv_full = np.fft.irfft(wf * af, n=self.fft_size)
        conv_full = conv_full[: self.len_w + n - 1]

        # sum_term[j] = Σ_k w_k * a_{k-j}
        sum_term = conv_full[self.n - 1 :: -1][: self.n]

        # g = 2 * sum_term  (∂/∂a of the smooth max)
        g = 2.0 * sum_term

        # grad_a = (2·n) * g
        grad_a = 2.0 * self.n * g

        # chain rule through soft‑max: ∂/∂u = diag(a)·grad_a - a·(a·grad_a)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        return obj, grad_u

                cand_rand[j] += delta
                cand_rand = simplex_projection(cand_rand)
                cand_rand /= cand_rand.sum()
                cand = cand_rand

        # ---- evaluate candidate ----
        cand_conv = full_convolution_fft(cand)
        cand_score = float(2.0 * n * cand_conv.max())

        # ---- simulated‑annealing acceptance ----
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_conv = cand_conv
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best sub‑grad score = {best_score:.6f}")

        # ---- schedule updates ----
        temperature = max(temperature * temp_decay, 1e-12)
        step_size = max(step_size * step_decay, 1e-12)

        # ---- re‑heat if stagnating for a long time ----
        if iteration - last_improve > 200000:
            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            current /= current.sum()
            current_conv = full_convolution_fft(current)
            current_score = float(2.0 * n * current_conv.max())
            temperature = 0.5
            step_size = 0.2
            last_improve = iteration
            print(f"Iter {iteration:7d}: reheated after stagnation")

    # ------------------------------------------------------------------
    # 4.  Return best sequence found
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}  (length {len(best_seq)})")
    return best_seq.tolist()
