    return res.x

def optimal_b_simple(f, M):
    n = f.shape[0]
    eps = 1e-12
    f = np.maximum(f, eps)
    b = np.empty(n, dtype=np.float64)
    suffix_sum = 0.0
    for i in range(n-1, -1, -1):
        max_b = (M - suffix_sum) / f[i]
        if max_b <= 0:
            b[i] = 0.0
        else:
            b[i] = max_b
        suffix_sum += f[i] * b[i]
    return b
```

def convolve(f, b):

def solve_lp_b(f, M):
    import scipy.optimize as opt
    import scipy.sparse as sp

    n = len(f)
    # Build sparse matrix for convolution constraints
    # as described: rows = i + j, cols = i
    idx_i = np.arange(n, dtype=np.int64)
    idx_j = np.arange(n, dtype=np.int64)
    rows = (idx_i[:, None] + idx_j).ravel()
    cols = np.tile(idx_i, n)
    data = np.repeat(f, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        print("LP failed: ", res.message)
        return None
```

import time, numpy as np, scipy.optimize as opt, scipy.sparse as sp

def max_conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n-1]
    return float(conv.max())

def full_conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n-1]
    return conv

def build_conv_matrix(f):
    n = f.shape[0]
    i = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + i).ravel()
    cols = np.repeat(i, n)
    data = np.repeat(f.astype(np.float64), n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp_max_sum(f, M):
    n = f.shape[0]
    A = build_conv_matrix(f)
    c = -np.ones(n, dtype=np.float64)  # maximize sum => minimize -sum
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    # Use HiGHS via scipy
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return zeros
        print("LP failed:", res.message)
        return None
    b = res.x
    # Clip tiny negatives due to numerical errors
    b = np.maximum(b, 0.0)
    return b

def parametric_candidates():
    candidates = []
    lengths = [1500, 2000, 2500, 3000]
    p_vals = np.linspace(0.3, 2.0, 9)
    alphas = np.linspace(0.001, 0.02, 10)
    for n in lengths:
        # U‑shaped power‑law
        for p in p_vals:
            d = np.arange(n)
            d = np.minimum(d, n-1 - d)
            a = (d + 1) ** (-p) + 1e-12
            a = a / a.sum()
            candidates.append(a)
        # exponential decay

            candidates.append(a)

            a = ((i+1)*(n-i)) ** (-p) + 1e-12
            a = a / a.sum()
            candidates.append(a)

   (≈ 2·n·max(f*f)/sum(f)²) is kept as a starting point.  An externally
   supplied `height_sequence_1` is also considered if present.

2. Main loop – repeat until the time budget expires or no improvement
   is possible:
   * Let `M = max(f*f)`.  Solve the linear programme

         maximise   Σ b_j
         subject to  f * b ≤ M       (convolution, entry‑wise)
                     b ≥ 0

     This LP can be written with a Toeplitz‑like sparse matrix and
     solved with SciPy's HiGHS backend.
   * The LP solution `b` is a direction that can increase the total
     mass without violating the current convolution bound.
   * Scale `b` to keep its *self*‑convolution below `M`:
         scale = min(1, sqrt(M / max(b*b)) ).
     The scaled vector `g = scale·b` satisfies both `f·g ≤ M` and
     `g·g ≤ M`.
   * Replace the current vector by `g`.  Because all three convolutions
     (`f·f`, `f·g`, `g·g`) are bounded by `M`, the new bound does not
     increase, and it strictly decreases unless `f` is already optimal.
   * Track the best vector seen (lowest exact evaluation).

3. After the loop, the best vector is evaluated once with the official
   `evaluate_sequence` routine and returned as a plain Python list.

The implementation respects a 1000 s time budget, prints progress, and
uses only NumPy, SciPy and the supplied `evaluate_sequence` function.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Fast convolution utilities (self‑convolution)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Linear programme: maximise Σb  subject to f*b ≤ M, b≥0
# ----------------------------------------------------------------------


def build_conv_matrix_indices(n: int):
    """Pre‑compute row/column index arrays for the convolution matrix A (size (2n‑1)×n)."""
    i = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + i).ravel()               # i + j for all i,j
    cols = np.repeat(i, n)                         # column i repeated n times
    return rows, cols


def solve_lp_max_sum(f: np.ndarray, M: float, rows, cols) -> np.ndarray | None:
    """
    Solve  max Σb_j  s.t.  f * b ≤ M , b ≥ 0.
    Returns the optimal vector b (non‑negative) or None on failure.
    """
    n = f.shape[0]
    # data for the sparse matrix – each column i is filled with the constant f[i]
    data = np.repeat(f.astype(np.float64), n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n, dtype=np.float64)                     # minimise -sum(b)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)       # right‑hand side
    bounds = [(0.0, None)] * n

    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        print(f"LP failed (status {res.status}): {res.message}")
        return None
    b = np.maximum(res.x, 0.0)      # clip tiny negatives caused by numerical noise
    return b


# ----------------------------------------------------------------------
# Parametric families (all return a normalised vector with sum == 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-12
    a /= a.sum()
    return a


def generate_warm_start_candidates():
    """Yield a sequence of candidate vectors for the warm‑start phase."""
    # include external sequence if present
    try:
        ext = np.array(height_sequence_1, dtype=float)  # noqa: F821
        if ext.size > 0:
            ext = np.maximum(ext, 0.0)
            ext /= ext.sum()
            yield ext
    except NameError:
        pass

    lengths = [1500, 1800, 2100, 2500, 3000]
    p_vals = np.linspace(0.3, 2.0, 9)          # 0.3 … 2.0
    alphas = np.linspace(0.001, 0.02, 10)      # 0.001 … 0.02

    for n in lengths:
        # U‑shaped power‑law
        for p in p_vals:
            yield u_shaped_distribution(n, p)
        # exponential decay
        for a in alphas:
            yield exponential_distribution(n, a)
        # product power‑law
        for p in p_vals:
            yield product_powerlaw_distribution(n, p)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step function that minimizes the provided
    evaluation metric.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    best_seq = None
    best_eval = np.inf

    # ------------------- 0. Warm‑start (parametric sweep) -------------------
    print("=== Warm‑start parametric sweep ===")
    for a in generate_warm_start_candidates():
        if time.time() > deadline:
            break
        n = a.shape[0]
        M = max_conv_fft(a)
        # evaluation is 2·n·M / (sum(a)²); sum(a)=1 here, so it simplifies to 2·n·M
        eval_approx = 2.0 * n * M
        if eval_approx < best_eval:
            best_eval = eval_approx
            best_seq = a.copy()
            print(f"  new warm‑start: n={n:4d}, approx_eval={eval_approx:.6f}")

    # If nothing was found (should not happen), initialise at random
    if best_seq is None:
        n = 2000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_eval = 2.0 * n * max_conv_fft(best_seq)
        print(f"Fallback random start: n={n}, approx_eval={best_eval:.6f}")

    # ------------------------------------------------------------------
    # 1. Frank‑Wolfe iteration using the LP
    # ------------------------------------------------------------------
    f = best_seq.copy()
    n = f.shape[0]
    eps = 1e-12
    f = np.maximum(f, eps)            # avoid zero coefficients (LP would be unbounded)

    # Pre‑compute sparse‑matrix indices (they depend only on the length)
    rows_idx, cols_idx = build_conv_matrix_indices(n)

    # Initial bound and evaluation
    M = max_conv_fft(f)
    best_score = 2.0 * n * M / (f.sum() ** 2)
    best_seq = f.copy()
    print(f"\nInitial: n={n}, M={M:.6f}, eval≈{best_score:.6f}")

    it = 0
    while time.time() < deadline:
        it += 1
        # ---- solve the LP for the current f ----
        b = solve_lp_max_sum(f, M, rows_idx, cols_idx)
        if b is None:
            print("LP failed – terminating search.")
            break

        # ---- check the self‑convolution of the LP solution ----
        Mb = max_conv_fft(b)

        # Scale b so that its own self‑convolution does not exceed M
        if Mb > M * (1 + 1e-12):
            scale = np.sqrt(M / Mb)
        else:
            scale = 1.0
        g = b * scale

        # ---- compute the new bound after replacement ----
        M_new = max_conv_fft(g)
        eval_new = 2.0 * n * M_new / (g.sum() ** 2)

        # ---- accept if we made progress ----
        if eval_new < best_score - 1e-9:
            best_score = eval_new
            best_seq = g.copy()
            f = g
            M = M_new
            print(f"Iter {it:4d}: improved eval≈{best_score:.6f}, M={M:.6f}")
            if best_score < 1.503:
                print("Target reached – exiting early.")
                break
        else:
            # No further improvement – stop the loop
            print(f"Iter {it:4d}: no improvement, stop.")
            break

    # ------------------------------------------------------------------
    # 2. Final exact evaluation (once) and report
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {best_score:.6f}")
    print(f"Best exact score       : {exact_score:.6f}")
    print(f"Best length            : {len(best_seq)}")
    return best_seq.tolist()
