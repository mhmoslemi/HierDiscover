propose_candidate
-----------------

Search for a non‑negative sequence that minimizes the evaluation score.
The algorithm consists of three phases:

   exponential, linear and mixed families).  Large parameter ranges and

3. Fine‑tuning with a smooth approximation of the max‑convolution
   (log‑sum‑exp) and a back‑tracking line‑search.  The smoothing
   parameter α is increased continuously, turning the smooth objective
   into the true max‑objective.

import time
import numpy as np

# helper functions
def simplex_projection(v):
    """Project vector v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def full_convolution_fft(a):
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

def score_from_vector(a):
    """Score = 2 * n * max(conv) assuming sum(a) == 1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

def compute_subgradient(a, conv):
    """Sub‑gradient of max_k (a * a)[k] w.r.t. a."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

def smooth_gradient(a, conv, alpha):
    """Gradient of log‑sum‑exp(alpha * conv) (smooth max)."""
    n = a.shape[0]
    # compute weights w = exp(alpha*conv) / sum(exp(alpha*conv))
    # use stable exp
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()
    # reverse w for convolution trick
    w_rev = w[::-1]
    size = n + w_rev.shape[0] - 1   # = 3n - 2
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w_rev, n=size)
    conv_full = np.fft.irfft(fa * fw, n=size)
    # gradient = 2 * conv_full[2*n-2 - i] for i=0..n-1
    indices = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv_full[indices]
    return grad

def u_min_distribution(n, p):
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

def u_prod_distribution(n, p):
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a

def exp_distribution(n, lam):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

def linear_distribution(n):
    # decreasing linearly from left to right
    a = np.arange(n, 0, -1, dtype=np.float64)
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

def quadratic_distribution(n, p):
    # a_i = (i+1)^{-p} + (n-i)^{-p}, but squared exponent for sharper shape
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-14
    a /= a.sum()
    return a

def mixed_distribution(n, p, w):
    """Mixture of an inverse‑distance U‑shape and a uniform background."""
    u = u_min_distribution(n, p)
    uniform = np.ones(n, dtype=np.float64) / n
    a = w * u + (1.0 - w) * uniform
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises the evaluation score

    2 * n * max_k (a * a)[k]   (the denominator of ``evaluate_sequence`` is
    cancelled by the normalisation Σ a_i = 1).

The algorithm works in three stages:

1. **Deterministic sweep** – many parametric families (U‑shapes based on
   distance or product, exponentials, linear, quadratic and mixtures) are
   examined for several lengths up to 32000.  Every candidate is
   normalised, its exact score is computed (via an FFT‑based convolution)
   and the best vector is kept.

2. **Fast stochastic refinement** – a projected sub‑gradient descent with
   simulated‑annealing acceptance, adaptive step‑size, occasional random
   mass‑transfer moves and a re‑heat strategy.  This phase quickly reduces
   the maximal convolution entry.

3. **Fine‑tuning** – a smooth approximation of the max (log‑sum‑exp with a
   growing smoothing parameter α) is minimised using a back‑tracking line
   search.  The smooth objective converges to the true max as α → ∞,
   providing a precise final adjustment.

All heavy work respects the supplied time budget (default 1000 s) and the
returned list is a plain Python list of non‑negative floats.
"""

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex { w≥0 , Σ w = 1 }."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score for a vector with Σa = 1 (i.e. 2·n·max(conv))."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices attaining (or almost attaining) the maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth max f(a)=log Σ_k exp(α·conv_k).
    Uses the identity ∂conv_k/∂a_i = 2·a_{k‑i} (when the index is valid).
    The computation is performed via FFT for O(n log n) cost.
    """
    n = a.shape[0]

    # stable soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                     # length 2n‑1

    # convolution trick: g_i = 2·Σ_k w_k·a_{k‑i}
    # can be written as 2·(a * w_rev)[2n‑2‑i]
    w_rev = w[::-1]
    size = n + w_rev.shape[0] - 1          # = 3n‑2
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w_rev, n=size)
    conv_full = np.fft.irfft(fa * fw, n=size)
    idx = 2 * n - 2 - np.arange(n)          # map to required positions
    grad = 2.0 * conv_full[idx]
    return grad

        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement – Phase A (sub‑gradient descent)
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]
    current /= current.sum()
    current_conv = full_convolution_fft(current)
    current_score = float(2.0 * n * current_conv.max())

    temperature = 1.0
    temp_decay = 0.99995
    step_size = 0.8
    step_decay = 0.99997
    iteration = 0
    last_improve = 0

    print("\n=== Sub‑gradient refinement start ===")
    half_deadline = start_time + budget_s / 2.0   # switch to phase B after half the budget
    while time.time() < deadline:
        iteration += 1

        # ---- sub‑gradient step -------------------------------------------------
        grad = compute_subgradient(current, current_conv)
        cand = current - step_size * grad
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)
        cand /= cand.sum()
        cand_conv = full_convolution_fft(cand)
        cand_score = float(2.0 * n * cand_conv.max())

        # ---- simulated‑annealing acceptance ------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_conv = cand_conv
            current_score = cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ---- occasional random mass‑transfer move --------------------------------
        if iteration % 256 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] > 0.0:
                delta = current[i] * np.random.rand() * 0.4
                cand_rand = current.copy()
                cand_rand[i] -= delta
                cand_rand[j] += delta
                cand_rand = np.maximum(cand_rand, 0.0)
                cand_rand = simplex_projection(cand_rand)
                cand_rand /= cand_rand.sum()
                cand_rand_conv = full_convolution_fft(cand_rand)
                cand_rand_score = float(2.0 * n * cand_rand_conv.max())
                # acceptance rule identical to the sub‑gradient one
                accept_rand = False
                if cand_rand_score < current_score:
                    accept_rand = True
                else:
                    prob_rand = np.exp((current_score - cand_rand_score) /
                                       max(temperature, 1e-12))
                    if np.random.rand() < prob_rand:
                        accept_rand = True
                if accept_rand:
                    current = cand_rand
                    current_conv = cand_rand_conv
                    current_score = cand_rand_score
                    if cand_rand_score < best_score:
                        best_score, best_seq = cand_rand_score, cand_rand.copy()
                        last_improve = iteration
                        print(f"Iter {iteration:7d}: new best (rand) = {best_score:.6f}")

        # ---- schedule updates ---------------------------------------------------
        temperature *= temp_decay
        step_size *= step_decay

        # ---- re‑heat after prolonged stagnation ---------------------------------
        if iteration - last_improve > 40000:
            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            current /= current.sum()
            current_conv = full_convolution_fft(current)
            current_score = float(2.0 * n * current_conv.max())
            temperature = 1.0
            step_size = 0.8
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

        # ---- switch to phase B after half the allotted time --------------------
        if time.time() > half_deadline:
            break

    # ------------------------------------------------------------------
    # 5. Stochastic refinement – Phase B (smooth‑max gradient descent)
    # ------------------------------------------------------------------
    alpha = 10.0                # start with a gentle smoothing
    alpha_max = 2000.0
    alpha_growth = 1.001
    step_size = 0.2

    print("\n=== Smooth‑max refinement start ===")
    while time.time() < deadline:
        # compute current convolution (needed for both score and gradient)
        conv = full_convolution_fft(current)
        grad = smooth_gradient(current, conv, alpha)

        # back‑tracking line search (max 10 reductions)
        t = step_size
        improved = False
        for _ in range(10):
            cand = current - t * grad
            cand = np.maximum(cand, 0.0)
            cand = simplex_projection(cand)
            cand /= cand.sum()
            cand_conv = full_convolution_fft(cand)
            cand_score = float(2.0 * n * cand_conv.max())
            if cand_score < current_score:
                improved = True
                break
            t *= 0.5

        if improved:
            current = cand
            current_conv = cand_conv
            current_score = cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                print(f"Smooth iter: new best = {best_score:.6f}")
        else:
            # no improvement → increase smoothing (approach true max)
            alpha = min(alpha * alpha_growth, alpha_max)

        # always increase smoothing slowly, steering towards the real objective
        alpha = min(alpha * alpha_growth, alpha_max)

    # ------------------------------------------------------------------
    # 6. Return the best sequence discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
