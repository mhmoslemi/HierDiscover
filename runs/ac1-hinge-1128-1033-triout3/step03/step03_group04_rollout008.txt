def golden_section_search(func, a, b, tol=1e-4, max_iter=50):
    phi = (1 + np.sqrt(5)) / 2
    resphi = 2 - phi
    c = b - resphi * (b - a)
    d = a + resphi * (b - a)
    fc = func(c)
    fd = func(d)
    for i in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - resphi * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + resphi * (b - a)
            fd = func(d)
    # return best
    if fc < fd:
        return c, fc
    else:
        return d, fd
```

        a_opt = u_prod_distribution(n, p_opt)
        if sc_opt < best_score:

    # optionally include external height_sequence_1

        if external.size:

            if external.sum() > 0:

                sc_ext = score_from_vector(external)
                if sc_ext < best_score:

                    print(f"External height_sequence_1 score={sc_ext:.6f}")
    except NameError:
        pass

"""
Propose a step‑function with a very low evaluation score.

Algorithm
---------
1. **Fast parametric sweep** – For several lengths (2000 … 32000) we
   optimise a handful of 1‑parameter families (U‑shape, product‑U,
   symmetric Beta, exponential decay).  For each family a golden‑section
   line‑search finds the parameter that minimises the score
   `2·n·max(a*a)`.  The best vector discovered becomes the seed.

2. **Optional external seed** – If a global `height_sequence_1` exists it
   is examined and used if it improves the score.

3. **Deterministic refinement** – Starting from the seed we perform
   a projected gradient‑descent on the smoothed maximum of the
   self‑convolution.
   * The self‑convolution `c = a*a` is computed by a single FFT.
   * A soft‑max weighting `w ∝ exp(β·(c‑max(c)))` (β≈300) yields a
     smooth sub‑gradient `g = 2·(w_rev * a)[L‑1‑i]`.
   * Mass is transferred from the index with largest gradient to the
     one with smallest gradient.  The transferred fraction is
     `δ = step_frac·a[src]` (step size decays exponentially).
   * The candidate is projected back onto the probability simplex.
   * If the candidate improves the score it is accepted; otherwise the
     step size is reduced.  After long stagnation a tiny random
     perturbation (“re‑heat”) is applied.

The whole routine stops after the allocated time budget (default 1000 s)
and returns the best sequence found.  All numbers are non‑negative;
the external `evaluate_sequence` function is scale‑invariant, so the
returned vector is normalised to sum 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {w ≥ 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length 2·len(a)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a normalised vector."""
    n = a.shape[0]
    return 2.0 * n * full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances: ((i+1)*(n‑i))^(-p)."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete symmetric Beta(α,β) on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Utility: golden‑section line search on a scalar interval
# ----------------------------------------------------------------------
def golden_section_search(func, lo, hi, tol=1e-4, max_iter=40):
    """Return (x_opt, f_opt) minimizing unimodal func on [lo, hi]."""
    phi = (1 + np.sqrt(5.0)) / 2.0
    resphi = 2.0 - phi
    a, b = lo, hi
    c = b - resphi * (b - a)
    d = a + resphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - resphi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + resphi * (b - a)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------

    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) discovered before the time budget expires.
    """

    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------ 1. Deterministic sweep over families ------------------
    best_seq = None
    best_score = np.inf

    # lengths to explore – keep a reasonable spread
    lengths = [2000, 4000, 8000, 16000, 32000]

    for n in lengths:

            break

        def obj_u_min(p):
