def project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
* Choose a relatively large problem size (default n=8000).  The theoretical
  lower bound of the objective is 2·n·(1/(2n-1)) ≈ 1, so a larger n gives room
  for a smaller value.
* Initialise the vector from the best known construction ``height_sequence_1``
  (if present) by linear interpolation, otherwise start from the uniform
  distribution.
* Optimise the probability vector a (∑a_i = 1, a_i ≥ 0) by minimising a
  smooth Log‑Sum‑Exp approximation of the max‑convolution:
        L_ε(a) = 2·n·ε·log(∑_k exp(conv_k/ε)).
  The gradient of L_ε with respect to a is
        ∂L/∂a = 4·n·conv(a, rev(p))[n‑1:2n‑1][::-1] ,
  where p_k = softmax(conv_k/ε) and rev denotes reversal.
* Parameters are updated with the Adam optimiser on the *soft‑max* of the
  variables (w = log a).  The smoothing parameter ε is linearly annealed
  from 5·10⁻³ to 10⁻⁹, the learning rate from 0.12 to 0.001.
* Occasionally a small Gaussian perturbation is added and if the optimiser
  stalls we perform a random restart.
* The best exact objective (computed with ``evaluate_sequence``) is tracked
  throughout the run and returned at the end of the allotted time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions (real valued)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full convolution x * x  (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (used for occasional restarts)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    # Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation entry point
# ----------------------------------------------------------------------

    before the time budget expires.
    """

    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – larger n gives a lower theoretical bound.
    # --------------------------------------------------------------
    n = 8000                     # feel free to enlarge if time permits

    # --------------------------------------------------------------
    # Initialise a from the previous best construction if available.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        a0 = np.full(n, 1.0 / n)

    a0 = a0 / a0.sum()                     # enforce Σa = 1
    w = np.log(a0 + 1e-12)                 # soft‑max parameters (log‑domain)

    # --------------------------------------------------------------
    # Adam hyper‑parameters
    # --------------------------------------------------------------
    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.12
    lr_end = 0.001

    m = np.zeros_like(w)
    v_vec = np.zeros_like(w)
    t = 0

    # --------------------------------------------------------------
    # Smoothing schedule for the log‑sum‑exp approximation
    # --------------------------------------------------------------
    eps_start = 5e-3
    eps_end = 1e-9

    # Book‑keeping for the best solution seen so far
    best_a = a0.copy()
    best_val = float("inf")
    no_improve = 0
    IMPROVE_LIMIT = 35000   # random restart after many idle steps

    iteration = 0
    print(f"[init] n={n}, starting optimisation...")

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---------------------------------------------------------
        # Soft‑max to obtain a feasible probability vector a
        # ---------------------------------------------------------
        w_shift = w - np.max(w)                # for numerical stability
        a = np.exp(w_shift)
        a /= a.sum()                           # Σa = 1, a_i ≥ 0

        # ---------------------------------------------------------
        # Exact objective (no smoothing) – also used for best‑track
        # ---------------------------------------------------------
        conv_aa = _full_convolution(a)         # length 2n‑1
        max_conv = conv_aa.max()
        cur_val = 2.0 * n * max_conv           # equals evaluate_sequence for Σa=1

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[{iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ---------------------------------------------------------
        # Update smoothing parameter ε (linear decay)
        # ---------------------------------------------------------
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # ---------------------------------------------------------
        # Compute the probability vector p_k = soft‑max(conv_k / ε)
        # ---------------------------------------------------------
        shifted = (conv_aa - max_conv) / eps    # shift to avoid overflow
        weights = np.exp(shifted)
        p = weights / weights.sum()            # length 2n‑1, Σp = 1

        # ---------------------------------------------------------
        # Gradient of the smoothed objective w.r.t. a
        # g = 4·n·conv(a, rev(p))[n‑1 : 2n‑1][::-1]
        # ---------------------------------------------------------
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3n‑2
        segment = conv_ab[n - 1 : 2 * n - 1]    # middle n entries
        grad_a = 4.0 * n * segment[::-1]       # shape (n,)

        # ---------------------------------------------------------
        # Chain rule through the soft‑max (gradient w.r.t. w)
        # ---------------------------------------------------------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---------------------------------------------------------
        # Adam update on w
        # ---------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)

        # Learning‑rate schedule (linear)
        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ---------------------------------------------------------
        # Small random perturbations to escape plateaus
        # ---------------------------------------------------------
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.015, size=n)

        # ---------------------------------------------------------
        # Random restart if stagnation is detected
        # ---------------------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            print(f"[{iteration:7d}] Restarting after {no_improve} idle steps")
            # Randomly initialise w but keep a reasonable distribution
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v_vec.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

        # Occasionally recompute the exact score (already done above)
        if iteration % 5000 == 0:
            pass  # placeholder – keeping the loop lightweight

    # ------------------------------------------------------------------
    # Final report – evaluate the best sequence with the provided safety
    # wrapper (guards against accidental clipping/invalid values).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
