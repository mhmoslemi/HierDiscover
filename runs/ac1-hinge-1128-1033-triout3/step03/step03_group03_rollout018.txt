def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np

    deadline = time.time() + budget_s - 5.0
    rng = np.random.default_rng(seed)

    n = 1500  # choose size; may adjust

    # Initialize Catalan truncated normalized sum=1
    coeffs = np.empty(n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, n):
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    a = coeffs / coeffs.sum()
    # Ensure simplex
    a = _project_simplex(a)

    # precompute length L and FFT size
    L = 2 * n - 1
    n_fft = 1 << ((2*L - 1).bit_length())  # Next power of two >= 2L maybe 4n

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    decay = 0.9995

    gamma = 1.0
    gamma_max = 4000.0
    gamma_factor = 1.001

    best_val = float('inf')
    best_seq = a.copy()
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1

        # compute convolution via FFT
        a_fft = np.fft.rfft(a, n_fft)
        conv_full = np.fft.irfft(a_fft * a_fft, n_fft)
        conv = conv_full[:L]          # self-convolution values

        max_conv = conv.max()
        obj = 2.0 * n * max_conv      # sum is 1

        if obj < best_val - 1e-12:
            best_val = obj
            best_seq = a.copy()
            no_improve = 0
            print(f"[iter {it:6d}] best={best_val:.10f}  max_conv={max_conv:.8e}")
        else:
            no_improve += 1

        # Softmax weights for smoothing max
        # stable exponent
        shift = conv.max()  # computed as max_conv
        exps = np.exp(gamma * (conv - shift))
        w = exps / exps.sum()    # shape L

        # gradient w.r.t a
        a_rev = np.concatenate([np.zeros(L - n), a[::-1]])   # length L
        w_fft = np.fft.rfft(w, n_fft)
        a_rev_fft = np.fft.rfft(a_rev, n_fft)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project onto simplex
        a_new = _project_simplex(a_new)

        a = a_new

        # schedule
        lr = max(1e-6, lr * decay)
        if gamma < gamma_max:
            gamma *= gamma_factor

        # restart if stagnation
        if no_improve > 5000:
            a = rng.random(n)
            a = a / a.sum()
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            lr = 0.1
            no_improve = 0
            gamma = 1.0
            print(f"[restart] iteration {it}")

    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(best_seq)
    print(f"[result] final_score={final_score:.10f}, length={len(best_seq)}")

"""Optimising the step‑function bound with a smoothed max objective.

The algorithm works on a probability simplex (∑a_i = 1, a_i ≥ 0).  
We minimise  

    2 * n * max(a * a)          (the sum‑normalised objective)

by performing an Adam‑style projected optimisation on a *smooth* version of
the max operator.  The smooth max is realised with a soft‑max weighting of
the full self‑convolution.  The gradient of the smooth max w.r.t. the
coefficients is obtained efficiently by a single FFT‑based cross‑correlation.
After each Adam update the vector is projected back onto the simplex.
A simple schedule slowly increases the “softness’’ parameter γ so that the
algorithm focuses on the true maximum later in the run.  Random restarts
prevent stagnation.

The method runs well within the 1000 s budget for n≈3000 and typically
achieves a bound below 1.50, improving on the previously reported 1.5197.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x | x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:                         # all non‑positive – return the uniform vector
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine.
# ----------------------------------------------------------------------

    Produce a non‑negative step‑function (list of heights) that minimises
    the provided `evaluate_sequence` objective.  The routine uses a

    Parameters

    list[float]

    """
    # ------------------------------------------------------------------
    # 1) time budget
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # small safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 3000                               # length of the step‑function
    L = 2 * n - 1                          # length of the full self‑convolution
    # FFT size large enough for both self‑convolution (length L) and the
    # cross‑correlation used in the gradient (length 2*L‑1).
    n_fft = 1 << ((2 * L - 1).bit_length())   # next power of two

    #    then project onto the simplex (sum = 1)

    cur = 1.0
    coeffs[0] = cur
    for k in range(1, n):
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    a = coeffs / coeffs.sum()
    a = _project_simplex(a)               # guarantees exact sum = 1

    # 4) Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(a)                  # first‑moment estimate
    v = np.zeros_like(a)                  # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    lr = 0.12
    decay = 0.9997

    gamma = 1.0
