"""
Propose a low‑score non‑negative step function.

Algorithm outline
-----------------
1. **Deterministic sweep** over a rich set of parametric families
   (U‑shapes, β‑distributions, exponential, polynomial, sine,
   half‑circle, triangular, cosine).  A modest grid of lengths
   `[2000, 4000, 8000, 12000, 20000, 40000]` is explored; the best
   vector (normalized to sum = 1) is kept as a seed.

2. **Upsample** the seed to larger lengths (up to 80000) by linear
   interpolation and refine each size separately.

3. **Stochastic refinement** – a hybrid of Adam‑style projected
   gradient descent, multiplicative mirror‑descent, and directed
   random mass‑transfer moves.  The gradient is the sub‑gradient of the
   current maximum convolution entry, computed only on an enlarged
   active set (all entries within 1e‑6 of the maximum).  Simulated‑annealing
   acceptance with exponential cooling and periodic “re‑heat” steps
   help escape plateaus.

All updates are projected onto the probability simplex (non‑negative,
unit‑sum) using the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
The routine respects the 1000 s budget; if time expires it returns the
best sequence found so far.
"""

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w >= 0, sum w = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1.0) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities (linear convolution a*a)
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length 2*len(a)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a non‑negative, sum≈1)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact score that evaluate_sequence would return for a (sum≈1)."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def harmonic_left_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n) + 1
    a = i.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def sine_distribution(n: int, power: float = 1.0) -> np.ndarray:
    x = np.linspace(0, np.pi, n)
    a = (np.sin(x) ** power) + 1e-14
    a /= a.sum()
    return a


def half_circle_distribution(n: int, power: float = 1.0) -> np.ndarray:
    """Discrete half‑circle (semicircle) shape."""
    t = np.linspace(-1.0, 1.0, n)
    a = np.sqrt(np.maximum(0.0, 1.0 - t ** 2)) ** power + 1e-14
    a /= a.sum()
    return a


def polynomial_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n) + 1
    rev = np.arange(n, 0, -1)
    a = (i * rev) ** (-p) + 1e-14
    a /= a.sum()
    return a


def triangular_distribution(n: int) -> np.ndarray:
    """Symmetric tent shape peaking at the centre."""
    i = np.arange(n)
    a = np.minimum(i + 1, n - i) + 1e-14
    a = a.astype(np.float64)
    a /= a.sum()
    return a


def cosine_distribution(n: int) -> np.ndarray:
    """Cosine bump centred in the middle (non‑negative)."""
    x = np.linspace(0, np.pi, n)
    a = (1.0 + np.cos(x)) + 1e-14
    a /= a.sum()
    return a

def interpolate_to_length(vec: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_len = vec.shape[0]
    if old_len == new_len:
        return vec.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_vec = np.interp(new_x, old_x, vec)
    # renormalise (tiny numerical drift may appear)
    new_vec = np.maximum(new_vec, 0.0)
    s = new_vec.sum()
    if s == 0.0:
        # fallback to uniform
        new_vec[:] = 1.0 / new_len
    else:
        new_vec /= s
    return new_vec


def subgradient_active_set(a: np.ndarray, rel_tol: float = 1e-6) -> np.ndarray:
    """Return the sub‑gradient of the max‑convolution w.r.t. a."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_val = conv.max()
    active_idx = np.where(conv >= max_val * (1.0 - rel_tol))[0]
    grad = np.zeros_like(a)
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad
