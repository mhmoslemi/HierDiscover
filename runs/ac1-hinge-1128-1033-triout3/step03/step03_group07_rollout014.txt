"""
Propose a non‑negative step function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a))**2

The core idea
-------------
* The ratio only depends on the *shape* of the vector, not on its scale.
* For a monotone exponential shape  a_i ∝ exp(‑α·i)  with α = c/n
  (c is a constant of order 1) one can compute the asymptotic value of the
  objective analytically:

        R(c) = 4·c²·e^(‑2c) / (1‑e^(‑c))²          for c < 1
               = 2·c / (e·(1‑e^(‑c))²)            for c ≥ 1

  The minimum is attained just below c = 1 and gives
  R_min ≈ 1.354.  Hence picking α ≈ 1/n (i.e. c≈0.9‑0.99) yields a huge
  improvement over the uniform distribution (R=2).

* The program sweeps a few lengths (8000 … 40000) and a dense grid of
  constants c ∈ [0.5, 0.99] to find the best monotone exponential vector.
  The convolution peak is evaluated with a fast FFT (O(n log n)).
* After the deterministic sweep a lightweight simulated‑annealing /
  random‑mass‑transfer loop tries to improve the best candidate a little
  further.
* Finally the best vector is passed to the official ``evaluate_sequence``
  routine and returned as a plain Python list.

The whole search respects the 1000 s budget (a 5 s safety margin is kept).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, returns the maximal coefficient)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    # Length required for the linear convolution (2n‑1).  Pad to next power of two.
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def exp_decreasing_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Monotone exponential shape: a[i] ∝ exp(-α·i) , i = 0 … n‑1.
    The vector is normalised to sum == 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

# Main search routine
# ----------------------------------------------------------------------

    """
    Search for a step function with a very low evaluation score.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0.  Warm‑start from a globally supplied sequence, if present.
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)  # noqa: F821

            if s < 1e-12:

                s = 1.0
            external /= s
            approx = 2.0 * external.size * max_conv_fft(external)
            if approx < best_score:
                best_score = approx
                best_seq = external.copy()

    except NameError:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    lengths = [8000, 12000, 16000, 20000, 24000, 28000, 32000, 40000]

    c_vals = np.linspace(0.50, 0.99, 250)
