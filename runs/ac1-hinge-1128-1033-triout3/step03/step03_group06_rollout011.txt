def propose_candidate(budget_s=1000, seed=0):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    n = 20000
    # Uniform vector for mixing
    uniform = np.full(n, 1/n)
    best_seq = None
    best_score = np.inf

    # initial seed - try previous best if available
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        # Normalize
        s = external.sum()
        if s == 0:
            external = uniform.copy()
        else:
            external = external / s
        maxconv = max_conv_fft(external)
        cur_score = 2 * n * maxconv
        best_seq = external.copy()
        best_score = cur_score
        print(f"Seed from height_sequence_1: score {cur_score:.6f}")
    except NameError:
        pass

    # fallback
    if best_seq is None:
        best_seq = uniform.copy()
        best_score = 2 * n * max_conv_fft(best_seq)
        print(f"Start from uniform: score {best_score:.6f}")

    # random param search
    iter_count = 0
    while time.time() < deadline:
        # sample parameters
        p = np.random.uniform(0.4, 0.8)  # exponent
        lam = np.exp(np.random.uniform(np.log(1e-3), np.log(10)))  # shift >0
        mix = np.random.uniform(0.2, 1.0)  # weight of power law
        # generate power-law vector
        idx = np.arange(n, dtype=float)
        pw = (idx + lam) ** (-p)
        pw_sum = pw.sum()
        if pw_sum == 0:
            continue
        pw = pw / pw_sum
        # mix with uniform
        a = mix * pw + (1 - mix) * uniform
        # optional small epsilon to avoid zeros
        # a already sum=1, because pw and uniform each sum 1, weight sum to 1
        # compute conv max
        conv_max = max_conv_fft(a)
        cur_score = 2 * n * conv_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iter_count:6d}: new best {cur_score:.6f} (p={p:.3f}, λ={lam:.3f}, mix={mix:.3f})")
        iter_count += 1

        # occasional local gradient descent from best_seq
        if iter_count % 2000 == 0 and time.time() < deadline:
            # run a few mirror descent steps
            a_local = best_seq.copy()
            n_local = n
            fft_size = 1 << ((3 * n_local - 2).bit_length())
            step_eta = 0.3
            alpha = 5.0
            for inner in range(500):
                conv = full_conv_fft(a_local, fft_size)
                cur_max = conv.max()
                cur_obj = 2 * n_local * cur_max
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a_local.copy()
                    print(f"  Local descent iter {inner}: new best {cur_obj:.6f}")
                w = softmax_weights(conv, alpha)
                grad = gradient_softmax(a_local, w, fft_size)
                exp_arg = -step_eta * grad
                exp_arg = np.clip(exp_arg, -30, 30)
                a_candidate = a_local * np.exp(exp_arg)
                a_candidate /= a_candidate.sum()
                conv_cand = full_conv_fft(a_candidate, fft_size)
                cand_max = conv_cand.max()
                cand_obj = 2 * n_local * cand_max
                if cand_obj <= cur_obj:
                    a_local = a_candidate
                    step_eta = min(step_eta * 1.02, 5.0)
                else:
                    step_eta = max(step_eta * 0.90, 1e-6)
                alpha = min(alpha * 1.001, 200.0)
                if time.time() >= deadline:
                    break

    # final evaluation with given evaluate_sequence
    try:
        exact_score = evaluate_sequence(best_seq.tolist())
        print(f"Exact evaluate_sequence score: {exact_score:.6f}")
    except Exception as e:
        print(f"evaluate_sequence failed: {e}")
        exact_score = best_score

    print(f"Search finished after {iter_count} random trials.")
    return best_seq.tolist()

The evaluation function is  
\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]
and it is scale‑invariant, so we may normalise the vector to a
probability distribution (sum = 1).  The problem therefore reduces to

*find a non‑negative vector on the simplex whose self‑convolution has the
smallest possible maximum entry.*

A useful analytical observation is that a **monotonically decreasing**
distribution minimises the central entry  
\(\sum_i a_i a_{n-1-i}\) (the rearrangement inequality).  If the tail is
too heavy, the first entry \(a_0^2\) becomes the bottleneck, therefore a
good shape is a **power‑law** that is not too steep, possibly mixed with a
small uniform component to keep \(a_0\) moderate.

The algorithm consists of two phases that comfortably fit into the 1000 s
budget.

1. **Randomised parametric search** – we generate vectors of the form  

   \[
   a_i = \alpha\;\frac{(i+\lambda)^{-p}}{\sum_j (j+\lambda)^{-p}}
          \;+\;(1-\alpha)\;\frac1n,
   \qquad
   p\in(0,1),\;
   \lambda>0,\;
   \alpha\in[0,1].
   \]

   The three parameters are sampled (log‑uniform for `\lambda` and uniform
   for the others).  For each candidate the convolution is obtained with a
   single FFT, the approximate score `2*n*max(conv)` is computed, and the
   best vector is stored.

2. **Mirror‑descent refinement** – the best vector from phase 1 is used as a
   warm‑start for a few thousand steps of the soft‑max mirror‑descent that
   was used in the reference solution.  The step‑size and the soft‑max
   temperature are adapted so that the true objective keeps decreasing.

The whole search is performed on the CPU only, uses no I/O and respects the
time limit.  At the end the official `evaluate_sequence` is called once to

"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` objective.

Algorithm
---------

1. Randomly explore three‑parameter families of monotone decreasing
   distributions:
        a_i = α·(i+λ)^(-p) / Z(p,λ)  +  (1-α)/n
   where p∈(0,1), λ>0 (log‑uniform) and α∈[0,1] (uniform‑mix weight).
   For each candidate the convolution is computed by a single FFT,
   the approximate objective 2·n·max(conv) is evaluated and the best
   candidate is remembered.

2. Starting from the best candidate, run a short mirror‑descent on the
   soft‑max of the convolution (the same technique used in the reference
   implementation) to fine‑tune the vector.

3. Return the best vector found (as a plain Python list).  The final,
   exact score is obtained by a single call to `evaluate_sequence`.

The routine respects a 1000 s budget, prints progress information and
never uses any file or network I/O.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ------------------------------------------------------------

def max_conv_fft(a: np.ndarray) -> float:
    """
    Maximum entry of the full self‑convolution a*a using a real FFT.
    a must be a 1‑D array (non‑negative, preferably normalised to sum=1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·n‑1) using a supplied FFT size.
    Returned array is real.
    """
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Normalised soft‑max weights w_k = exp(α·conv_k) / Σ exp().
    The exponent is stabilised by subtracting the maximum entry.
    """
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:                     # should never happen
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution w.r.t. a.
    a : vector of length n.
    w : weight vector of length 2·n‑1 (sums to 1).
    size : FFT length (>= 3·n‑2).
    Returns a vector of length n.
    """
    n = a.shape[0]
    L = 2 * n - 1                           # length of w
    A = np.fft.rfft(a, n=size)

    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # a * w  (length n+L‑1 = 3·n‑2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    idx = L - 1 - np.arange(n)               # 2·n‑2, …, n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the O(n log n) algorithm
    from Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("input must be 1‑D")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
