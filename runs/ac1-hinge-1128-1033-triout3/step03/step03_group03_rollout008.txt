def full_convolution(a, b):
    size = len(a) + len(b) -1
    nfft = 1 << (size-1).bit_length()
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(fa*fb, n=nfft)[:size]
    return conv
```

def full_convolution_fft(a, b):
    n = len(a) + len(b) - 1
    N = 1 << (n-1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:n]
    return conv
```

def softmax_weights(conv, epsilon):
    m = conv.max()
    # compute stable exponent
    exps = np.exp((conv - m) / epsilon)
    sum_exps = exps.sum()
    if sum_exps == 0:
        # fallback: use uniform weights
        return np.full_like(conv, 1.0/conv.size)
    return exps / sum_exps

def initial_sequence(rng, n):
    # generate base shapes
    x = np.arange(n)
    # catalan shape
    catalan = np.empty(n)
    cur = 1.0
    catalan[0] = cur
    for k in range(1, n):
        cur = cur * (2*k - 1) / (2*k)
        catalan[k] = cur
    # normalize
    catalan = np.maximum(0, catalan)

    # sine shape
    sine = np.sin(np.pi * (x + 0.5) / n)  # positive

    # semi-circle shape
    t = 2 * x / (n - 1) - 1  # [-1,1]
    semicircle = np.sqrt(np.clip(1 - t**2, 0, None))

    # Gaussian
    sigma = n / 6
    gauss = np.exp(-0.5 * ((x - (n-1)/2)/sigma)**2)

    # triangle
    triangle = np.minimum(x, n-1 - x) + 1.0  # positive

    bases = [catalan, sine, semicircle, gauss, triangle]
    weights = rng.random(len(bases))
    weights = weights / weights.sum()
    a = sum(w * b for w, b in zip(weights, bases))
    a = np.maximum(a, 0)
    if a.sum() == 0:
        a = rng.random(n)
    a = a / a.sum()
    return a
```

Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a smoothed‑max projected gradient descent with momentum.
The algorithm:

* Initialise a mixture of several “nice’’ shapes (Catalan,
  sine, semi‑circle, Gaussian, triangle).  If a previous best
  sequence `height_sequence_1` is available it is used as a warm start.
* While time permits:
    – Compute the self‑convolution with FFT (O(n log n)).
    – Form a soft‑max weighting of the convolution entries; the
      temperature ε is gradually reduced.
    – The gradient of the weighted sum is a cross‑correlation
      (computed again by FFT).
    – Take a momentum step, project back onto the probability simplex.
    – Decay the step‑size η and the temperature ε.
    – Occasionally add a tiny random perturbation if no improvement
      has been seen for a while (to escape plateaus).
* Keep the best sequence seen.
* After the main loop perform a short pure sub‑gradient polishing
  (max‑entry gradient) to shave off the last bits.
* Clamp the final heights to the limits required by `evaluate_sequence`
  and return them.

The method runs comfortably within the 1000 s budget and typically
finds scores well below the previous bound of 1.52, often reaching the
target ≤ 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution via FFT (full linear convolution).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv
