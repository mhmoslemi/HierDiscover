def compute_gradient(a, k):
    n = len(a)
    grad = np.zeros_like(a)
    # a[i] present only if k-i in [0, n-1]
    # So loop over i where k-i in [0, n-1] => i in [k-(n-1), k]
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    # For i in lo..hi inclusive, compute grad[i] = a[i] + a[k-i]
    grad[lo:hi+1] = a[lo:hi+1] + a[k - np.arange(lo, hi+1)]
    return grad

def simplex_projection(v):
    """ Project onto simplex sum=1, v>=0. Return w."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Hybrid optimizer for the step‑function problem.

Phase 1 – 80 % of the time budget:
    Adam on a smooth log‑sum‑exp approximation of
        2·n·max(conv(a,a)).
    The variables are re‑parameterised by a soft‑max (w) so that the
    probability simplex (a_i ≥ 0, Σa_i = 1) is always satisfied.

Phase 2 – remaining time:
    Direct sub‑gradient descent on the exact objective
        2·n·max(conv(a,a)).
    For the current maximal convolution index k* we use the exact
    sub‑gradient   ∂/∂a_i max(conv) = a_i + a_{k*-i}   (zero outside the
    overlap) and project back onto the simplex.

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # numerical artefacts may produce tiny negatives – clip them
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {a≥0, Σa = 1})
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    # translation of the algorithm from
    #   Wang & Carreira‑Perpiñán, “Projection onto the probability simplex”
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
