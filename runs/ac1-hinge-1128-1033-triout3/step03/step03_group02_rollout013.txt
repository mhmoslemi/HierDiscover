"""
Propose a candidate sequence minimizing evaluate_sequence using
a hybrid Adam‑based soft‑max gradient descent interleaved with
linear‑programming “mass re‑allocation” steps.
The algorithm works on the probability simplex (∑a_i = 1) because
the objective is scale‑invariant.  It repeatedly reduces the peak
of the auto‑convolution a∗a, which directly lowers the evaluation
value 2·n·max(a∗a).  A simple Euclidean projection onto the simplex
keeps the iterates feasible.  Occasionally a linear program is solved
to find a distribution that spreads the convolution mass as much as
possible under the current peak bound; the current iterate is blended
towards this distribution.  The whole procedure runs until the time
budget is exhausted and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    total = w.sum()
    if total == 0:
        return np.full_like(v, 1.0 / n)
    return w / total


# ----------------------------------------------------------------------
# Helper: Full (linear) convolution using real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    np.maximum(conv, 0, out=conv)  # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Compute g_i = 2 * Σ_k w_k * a_{k-i} using a single FFT‑based convolution."""
    n = a.shape[0]
    # reverse the weight vector
    u = w[::-1]
    # full convolution u * a -> length 3n‑2
    conv = _full_convolution_fft(u, a)
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Build the sparse constraint matrix for the linear‑programming step.
# For a given length n the pattern of non‑zero entries does not depend
# on the actual values of a, only on n.
# ----------------------------------------------------------------------
def _build_convolution_ub_matrix(n: int):
    """Return row indices, column indices suitable for constructing a CSR matrix
    that represents the linear map b ↦ a * b (full convolution)."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        pos += n
    return rows, cols


# ----------------------------------------------------------------------
# Solve the LP: maximise Σ b_j  subject to  a * b ≤ M (component‑wise) , b ≥ 0 .
# ----------------------------------------------------------------------
def _solve_lp_max_sum(a: np.ndarray, M: float, rows, cols) -> np.ndarray | None:
    """Return the optimal b (as a dense ndarray) or None if LP fails."""
    n = a.shape[0]
    # data = a_i repeated n times each
    data = np.empty(n * n, dtype=np.float64)
    for i in range(n):
        data[i * n:(i + 1) * n] = a[i]
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n, dtype=np.float64)               # maximise sum b ↔ minimise -sum b
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    # use HiGHS interior‑point method – fast and reliable for this size
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None

"""
Hybrid Adam‑soft‑max descent with occasional LP re‑balancing.

The objective   F(a) = 2·n·max_k (a*a)[k]   is invariant under scaling, so we
work on the probability simplex (∑a_i = 1, a_i ≥ 0).  The algorithm repeatedly
reduces the peak of the auto‑convolution a∗a:

* a soft‑max over the convolution yields a smooth approximation of the max;
  its gradient can be obtained by a single FFT‑based convolution.
* An Adam‑style update (adaptive step sizes) followed by projection onto the
  simplex provides fast, stable descent.
* Every few thousand iterations a small linear program is solved:
  maximise Σ b_j subject to a∗b ≤ M (current peak) and b ≥ 0.
  The optimiser returns a distribution that spreads the convolution mass as
  much as possible without raising the peak.  The current iterate is blended
  towards this distribution, which empirically pushes the peak further down.
* A tiny random “jitter” (mass transfer) helps escape plateaus.

The routine runs until the time budget expires and returns the best sequence
found (as a plain list of floats).  It works for any `height_sequence_1`
global that may already be available; otherwise it starts from a smooth
half‑sine shape.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    total = w.sum()
    if total == 0:
        return np.full_like(v, 1.0 / n)
    return w / total


# ----------------------------------------------------------------------
# Helper: Full (linear) convolution using real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    np.maximum(conv, 0, out=conv)                # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i via a single FFT‑based convolution.
    """
    n = a.shape[0]
    u = w[::-1]                                   # reverse the weight vector
    conv = _full_convolution_fft(u, a)            # length 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)                    # indices 2n‑2,2n‑3,…,n‑1
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Build the sparse constraint matrix for the LP step (a * b ≤ M)
# ----------------------------------------------------------------------
def _build_convolution_ub_matrix(n: int):
    """Return row/col index arrays for the CSR matrix that maps b → a * b."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        pos += n
    return rows, cols


# ----------------------------------------------------------------------
# Solve the LP: maximise Σ b_j  subject to a * b ≤ M, b ≥ 0.
# ----------------------------------------------------------------------
def _solve_lp_max_sum(a: np.ndarray, M: float, rows, cols) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise   sum(b)
        subject to a * b ≤ M  (full convolution, component‑wise)
                   b ≥ 0
    Returns the optimal b as a dense ndarray, or None on failure.
    """
    n = a.shape[0]
    # each a_i appears n times, once for every row where column i participates
    data = np.repeat(a, n)                         # shape (n*n,)
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n, dtype=np.float64)              # minimise -sum(b)  ⇒ maximise sum(b)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # HiGHS is the default method for recent SciPy versions; it handles this size well.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None
