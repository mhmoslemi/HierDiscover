"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

The method consists of two phases:

1. **Mass‑addition LP phase** – given a current distribution `a`
   (non‑negative, sum = 1), compute the largest possible total mass
   `sum(g)` that can be added while keeping the cross‑convolution
   `a * g` bounded by the current maximal autocorrelation `M`.
   This is a linear program:
       max  Σ_j g_j
       s.t. A(g) ≤ M·1,   g ≥ 0,
   where A(g) = a * g (full convolution).  The optimal `g`
   is normalised back to a probability vector and mixed into `a`
   (large mixing factor first, falling back to smaller factors if the
   objective does not improve).  The loop stops when no further mass
   can be added or the time budget is almost exhausted.

2. **Smoothed soft‑max descent** – the final refinement uses the
   momentum‑based soft‑max sub‑gradient method (as in the previous
   baseline) for the remaining time.

Both phases keep the sequence on the probability simplex, therefore
the evaluation is simply `2·n·max(conv(a,a))`.  The algorithm respects
the 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x >= 0, sum(x) = 1} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny sum error
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full (linear) convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper: build the full convolution matrix A such that A @ g = a * g
# ----------------------------------------------------------------------
def _build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """Return matrix A (shape (2n‑1, n)) with rows representing a * g."""
    n = a.shape[0]
    rows = 2 * n - 1
    i = np.arange(n)
    k = np.arange(rows)[:, None]          # shape (rows, 1)
    j = k - i                              # broadcasting: (rows, n)
    mask = (j >= 0) & (j < n)
    # a[i] broadcasts over rows; where mask is True, place a[i]
    A = np.where(mask, a[i], 0.0)
    return A


# ----------------------------------------------------------------------
# Helper: gradient from soft‑max weights (used in the final descent)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k‑i} for all i.
    Implemented via a single convolution: reverse w, convolve with a,
    then pick the appropriate slice.
    """
    n = a.shape[0]
    u = w[::-1]                     # reverse weights
    conv = _full_convolution_fft(u, a)   # length 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)      # descending indices
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Phase‑2: smoothed soft‑max descent with momentum (refinement)
# ----------------------------------------------------------------------
def _softmax_descent(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Refine a candidate distribution a_start until the deadline.
    Returns the best sequence observed in this phase.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    conv_a = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv_a.max())
    momentum = np.zeros_like(a)

    # hyper‑parameters (linear schedules)
    eta_start = 0.25
    eta_end   = 1e-6
    tau_start = conv_a.max() * 0.25
    tau_end   = conv_a.max() * 1e-5
    beta = 0.90
    jitter_prob = 0.02

    iteration = 0
    while time.time() < deadline:
        conv_a = _full_convolution_fft(a, a)
        M = conv_a.max()
        cur_val = float(2.0 * n * M)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[softmax] new best = {best_val:.6f}")

        progress = (deadline - time.time()) / (deadline - (deadline - (deadline - time.time())))
        # linear schedules based on elapsed fraction
        elapsed = max(0.0, (time.time() - (deadline - (deadline - time.time()))))
        total = max(1e-9, (deadline - (deadline - (deadline - time.time()))))
        frac = elapsed / total

        tau = tau_start * (1.0 - frac) + tau_end * frac
        if tau > 0:
            exp_args = (conv_a - M) / tau
            exp_vals = np.exp(exp_args - np.max(exp_args))  # improve stability
            w = exp_vals / exp_vals.sum()
        else:
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0

        grad = _gradient_from_weights(a, w)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        eta = eta_start * (1.0 - frac) + eta_end * frac

        a_candidate = a - eta * momentum

        # occasional random jitter
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a_candidate[src] * 0.5
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        a = _project_onto_simplex(a_candidate)
        iteration += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The search consists of a linear‑programming mass‑addition phase
    followed by a soft‑max gradient‑descent refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # Choose length n and initialise a
    # --------------------------------------------------------------
    seq1 = globals().get("height_sequence_1")
    if isinstance(seq1, list) and len(seq1) > 0:
        a = np.asarray(seq1, dtype=float)
        n = a.shape[0]
        if a.sum() == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
        print(f"[init] Using provided height_sequence_1 of length {n}")
    else:
        # modest length, random Dirichlet start (α=0.5 gives a sparse start)
        n = 1500
        a = np.random.dirichlet(np.full(n, 0.5))
        a = a / a.sum()
        print(f"[init] Random start, n={n}")

    # initial evaluation
    conv_a = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv_a.max())
    best_a = a.copy()
    print(f"[init] initial bound = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase‑1: LP mass‑addition loop
    # --------------------------------------------------------------
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline - 30.0:          # reserve ~30 s for refinement phase
            break

        # current max autocorrelation
        conv_a = _full_convolution_fft(a, a)
        M = conv_a.max()
        cur_val = float(2.0 * n * M)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[LP {iteration:5d}] new best = {best_val:.6f}")

        # construct convolution matrix A (shape (2n‑1, n))
        A = _build_convolution_matrix(a)

        # Linear program: max Σ g_j  ⇔  min −Σ g_j
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        lp_res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                         method='highs', options={'presolve': True})

        if not lp_res.success:
            print(f"[LP {iteration}] linprog failed: {lp_res.message}")
            break

        sum_g = -lp_res.fun
        # If we cannot increase total mass, we are at a stationary point
        if sum_g <= 1.0 + 1e-8:
            print(f"[LP {iteration}] No mass increase possible (sum_g={sum_g:.6f})")
            break

        g = lp_res.x / sum_g          # scale to a probability vector

        # Try to mix g into a with decreasing step sizes until improvement
        improved = False
        for t in (1.0, 0.5, 0.25, 0.125, 0.0625):
            a_cand = (1.0 - t) * a + t * g
            a_cand = _project_onto_simplex(a_cand)
            conv_cand = _full_convolution_fft(a_cand, a_cand)
            val_cand = float(2.0 * n * conv_cand.max())
            if val_cand < best_val - 1e-9:
                best_val = val_cand
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                print(f"[LP {iteration}] t={t:.3f} improved to {best_val:.6f}")
                break

        if not improved:
            # No improvement via mixing; stop the LP phase.
            print(f"[LP {iteration}] No mixing factor yielded improvement.")
            break

        # occasional random jitter to avoid tiny cycles
        if np.random.rand() < 0.01:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a[src] * 0.3
            a[src] -= delta
            a[dst] += delta
            a = _project_onto_simplex(a)

        iteration += 1

    # --------------------------------------------------------------
    # Phase‑2: soft‑max descent refinement (uses remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"[refine] Starting soft‑max refinement, {remaining:.1f}s left.")
        refined = _softmax_descent(best_a, deadline)
        conv_ref = _full_convolution_fft(refined, refined)
        refined_val = float(2.0 * n * conv_ref.max())
        if refined_val < best_val:
            best_val = refined_val
            best_a = refined
            print(f"[refine] improvement to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final reporting (optional – the outer framework may also call it)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
