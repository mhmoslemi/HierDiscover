assistantfinal**Solution Overview**  
The objective to minimise is  

\[
F(a)=\frac{2n\;\max_k\;(a*a)[k]}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

After clipping and normalising the sequence (`a` is projected onto the probability
simplex) the denominator becomes 1, therefore we simply have to minimise  

\[
\max_k (a*a)[k] .
\]

The programme below combines three ideas:

1. **Projected Adam optimisation** – a fast first‑order method on the simplex.
   The gradient of the soft‑max approximation of the max‑operator is obtained
   by one FFT‑based convolution.

2. **Periodic linear‑programming boost** – for the current iterate `a` we solve  

   \[
   \max\;\mathbf 1^\top g\quad\text{s.t.}\;a*g\le M,\;g\ge0,
   \]

   where `M = max_k (a*a)[k]`.  
   The solution `g` is the “most spread’’ distribution that respects the current
   bound. After normalising `g` we mix it a little with `a`.  This step
   reproduces the LP‑based update described in the statement and typically
   yields a large improvement in a single call.

3. **Multi‑restart & random jitter** – if no improvement has been observed for
   a while the algorithm restarts from a fresh random Dirichlet vector.  Small
   random mass‑transfer steps occasionally help to escape plateaux.

All steps are timed; the routine stops a few seconds before the 1000 s budget
and returns the best sequence discovered.

```python
"""Propose a step‑function with a low max auto‑convolution.

The search uses
  – Adam optimisation on the probability simplex with a soft‑max
    approximation of the max‑operator,
  – occasional exact linear‑programming updates (the LP described in the
    statement) that produce a strong descent direction,
  – random restarts and jitter to avoid local minima.

The routine runs until the time budget expires and returns the best
non‑negative sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex  (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Full convolution via FFT (used many times)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x*y (length len(x)+len(y)-1)."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max_k (a*a)[k]
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented via a single convolution: reverse w, convolve with a,
    then pick the slice that corresponds to the required indices.
    """
    n = a.shape[0]
    # reverse the weight vector (full convolution)
    u = w[::-1]
    conv = _full_convolution_fft(u, a)          # length = 3n‑2
    # indices that correspond to g_i (see derivation in the statement)
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (LP‑boost)
# ----------------------------------------------------------------------
def _lp_optimize(a: np.ndarray, M: float, B: np.ndarray) -> np.ndarray:
    """
    Solve   max 1ᵀ g   s.t.   a * g <= M,   g >= 0.
    ``B`` is the binary matrix with B[k,i]=1 iff a_i contributes to (a*g)[k].
    The actual constraint matrix is A = B * diag(a).
    """
    from scipy.optimize import linprog

    # A = B * a  (broadcast columnwise)
    A = B * a
    rows = A.shape[0]
    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(a.shape[0], dtype=np.float64)   # maximise sum(g) -> minimise -sum
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=(0, None), method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x

                    if new_val < best_val:
                        best_val = new_val
                        best_a = a.copy()
                        print(f"[LP {iteration:7d}] improved to {best_val:.6f}")
            except Exception as e:                       # pragma: no cover
                print("[LP] exception:", e)

        # ----------------------------------------------------------------
        # Restart if we have been stagnant for a long time
        # ----------------------------------------------------------------
        if iteration - last_improve > 8000:
            # fresh random Dirichlet start
            a = np.random.dirichlet(alpha=np.ones(n))
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"[restart] iteration {iteration}")

    # ----------------------------------------------------------------------
    # Return the best sequence discovered
    # ----------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best_score={final_score:.6f}, "
          f"iterations={iteration}, LP_calls={lp_calls}")
    return best_a.tolist()
```
