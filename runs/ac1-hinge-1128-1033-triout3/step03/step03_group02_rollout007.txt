def _full_convolution_fft(x, y, size=None):
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    if size is None:
        size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:total]
    np.maximum(conv, 0, out=conv)
    return conv

"""
Propose a step‑function (non‑negative vector) that minimizes the
evaluate_sequence objective.

Algorithm
---------
* Parameterise the coefficients with logits u and obtain the
  probabilities a = softmax(u) → Σ a = 1, a_i ≥ 0.
* At each iteration compute the full auto‑convolution a*a using a single
  FFT (size chosen once for the whole run).  The maximum M = max(a*a)
  gives the exact objective value 2·n·M.
* Approximate the non‑smooth max by a soft‑max with temperature τ.
  The soft‑max weights w form a probability distribution over the 2n‑1
  convolution entries.
* The gradient of the smoothed objective w.r.t. a is
      g = 2 * (w_rev * a)               (full convolution, slice)
  which is obtained with another FFT (re‑using the transform of a).
* Map the gradient to the logit space:
      grad_u_i = a_i·(g_i – ⟨a,g⟩)
  and update u with the Adam optimizer (first‑order adaptive steps).
* Temperature τ and learning‑rate η follow linear schedules from a
  relatively large start to a tiny end, turning the smoothed objective
  into the true max in the final phase.
* Occasional random jitter on u encourages escaping plateaux.
* The best vector (by the exact objective) is kept throughout the run.
* When the time budget is almost exhausted a cheap local‑search
  (random mass transfers) performs a final refinement.

The whole routine respects the 1000 s budget, works on any length n,
and returns a plain Python list of floats ready for evaluate_sequence().
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full convolution via a real FFT (single call per operand)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Return the full convolution of real vectors x and y using FFT size `size`."""
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.shape[0] + y.shape[0] - 1]
    # eliminate tiny negatives caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the evaluation
    function.  Uses an Adam‑optimised soft‑max‑approximation of the max‑
    convolution objective, with temperature annealing and a quick local
    refinement at the end.
    Returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin – we stop a couple of seconds before the hard limit
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a start length n and an initial shape (use the best known if
    # height_sequence_1 is available, otherwise a smooth sine shape).
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined in the outer scope
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = a0.shape[0]
    except Exception:
        n = 1800                     # a modestly large length
        x = np.arange(n, dtype=float)
        # a smooth half‑cosine bump (zero at the ends, 1 at the centre)
        a0 = 0.5 * (1 - np.cos(np.pi * x / (n - 1)))

    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0:
        a0 = np.full(n, 1.0 / n)
    a0 = a0 / a0.sum()               # normalise to Σa = 1

    # ------------------------------------------------------------------
    # Logit parametrisation: a = softmax(u)
    # ------------------------------------------------------------------
    eps = 1e-12
    u = np.log(np.maximum(a0, eps))

    # ------------------------------------------------------------------
    # Adam state
    # ------------------------------------------------------------------
    beta1 = 0.90
    beta2 = 0.999
    eps_adam = 1e-12
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # ------------------------------------------------------------------
    # Pre‑computed FFT size (covers both a*a and a*w_rev)
    # For a length n we need size >= 3n‑2 (worst‑case length of w_rev*a)
    # ------------------------------------------------------------------
    size = 1 << ((3 * n - 2) - 1).bit_length()   # next power of two

    # ------------------------------------------------------------------
    # Book‑keeping of the best vector (exact objective)
    # ------------------------------------------------------------------
    # compute initial exact objective
    a = a0.copy()
    conv_a = _full_convolution_fft(a, a, size)[: 2 * n - 1]
    M = conv_a.max()
    best_val = float(2.0 * n * M)
    best_a = a.copy()
    best_iter = 0

    # display initial status
    print(f"[init] n={n}, initial objective={best_val:.6f}")

    iteration = 0
    # hyper‑parameters for schedules
    lr_start = 0.5
    lr_end = 1e-3
    jitter_prob = 0.02
    jitter_start_sigma = 0.05

    # ------------------------------------------------------------------
    # Main optimisation loop (soft‑max / Adam)
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- obtain a from logits -----
        u_max = np.max(u)
        exp_u = np.exp(u - u_max)
        a = exp_u / exp_u.sum()

        # ----- FFT of a (re‑used for both convolutions) -----
        fft_a = np.fft.rfft(a, n=size)

        # ----- auto‑convolution a*a -----
        conv_full = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]
        np.maximum(conv_full, 0, out=conv_full)
        M = conv_full.max()
        cur_val = float(2.0 * n * M)

        # keep the best exact solution
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            best_iter = iteration
            print(f"[{iteration:07d}] new best = {best_val:.6f}")

        # ----- progress for schedules -----
        progress = (now - start_time) / (deadline - start_time)

        # ----- temperature (soft‑max smoothing) -----
        # start with a relatively large temperature (≈0.5·M) and cool to ~1e‑6·M
        tau = M * (0.5 * (1.0 - progress) + 1e-6 * progress)

        # ----- soft‑max weights w over the convolution -----
        if tau > 1e-12:
            # shift by the max to keep exponentials bounded
            exp_args = (conv_full - M) / tau
            max_arg = np.max(exp_args)
            exp_vals = np.exp(exp_args - max_arg)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv_full)
                w[np.argmax(conv_full)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv_full)
            w[np.argmax(conv_full)] = 1.0

        # ----- gradient of the smoothed objective w.r.t. a -----
        w_rev = w[::-1]
        fft_w = np.fft.rfft(w_rev, n=size)
        conv_w_full = np.fft.irfft(fft_a * fft_w, n=size)[: 3 * n - 2]
        np.maximum(conv_w_full, 0, out=conv_w_full)
        start_idx = 2 * n - 2
        idx = start_idx - np.arange(n)          # slice length n
        grad_a = 2.0 * conv_w_full[idx]

        # ----- map to logit gradient (softmax Jacobian) -----
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # ----- Adam update on logits -----
        iteration += 1
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u * grad_u)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional random jitter on logits to escape plateaux -----
        if np.random.rand() < jitter_prob:
            sigma = jitter_start_sigma * (1.0 - progress) + 1e-4 * progress
            u += np.random.randn(n) * sigma

        # ----- periodic status ----- (lightweight, every 2000 iters)
        if iteration % 2000 == 0:
            print(f"[{iteration:07d}] current = {cur_val:.6f}, best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final local refinement (random mass transfers) while time permits
    # ------------------------------------------------------------------
    remaining = deadline - time.time()

        a = best_a.copy()
        # pre‑allocate FFT size (same as before)
        move_frac_start = 0.2
        move_frac_end = 0.001
        local_start = time.time()
        local_iter = 0
        while True:
            if time.time() - local_start > remaining - 0.5:
                break
            # pick two distinct indices
            i, j = np.random.choice(n, size=2, replace=False)
            # fraction of mass to move (decays with time)
            frac = move_frac_start * (1.0 - local_iter / 50000) + move_frac_end * (local_iter / 50000)
            frac = max(frac, move_frac_end)
            delta = np.random.rand() * a[i] * frac
            if delta == 0:
                continue
            a[i] -= delta
            a[j] += delta

            # evaluate new objective (exact)
            conv = _full_convolution_fft(a, a, size)[: 2 * n - 1]
            M_new = conv.max()
            val_new = float(2.0 * n * M_new)

            if val_new < best_val:
                best_val = val_new
                best_a = a.copy()
                print(f"[local {local_iter:07d}] improved to {best_val:.6f}")
                # keep the improvement
                # (no need to revert)
            else:
                # revert the move
                a[i] += delta
                a[j] -= delta

            local_iter += 1

    # Final reporting (the framework may call evaluate_sequence itself)
    # ------------------------------------------------------------------

        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
