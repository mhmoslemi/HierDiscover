def simplex_projection(v):
    # returns projection on simplex sum=1, v>=0
    # Algorithm from Duchi et al. 2008.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    # Duchi et al. algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # Should not happen; return uniform
        return np.full_like(v, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (list of non‑negative heights) that minimises the
objective used in ``evaluate_sequence``.

Algorithm
---------

1. **Initialisation**  
   * Use the previously found construction ``height_sequence_1`` if it is
     available, otherwise start with a uniform vector.
   * Normalise the vector so that its sum is exactly 1 (the value of the
     sum does not affect the objective because of scale‑invariance).

2. **Projected sub‑gradient descent on the exact max‑convolution**
   * At the current point ``a`` compute the full convolution ``c = a*a``.
   * Let ``k*`` be the index where ``c`` attains its maximum.
   * A sub‑gradient of ``f(a)=2·n·max(c)`` is
        g_i = 4·n·a_{k*‑i}  (zero when the index is out of range).
   * Take a step ``a←a‑η·g`` with a linearly decreasing step size ``η`` and
     then project back onto the probability simplex (non‑negative, sum=1).
   * Accept the new point if it strictly improves the objective; otherwise
     accept it with a small probability given by an exponential cooling schedule
     (simulated‑annealing style) to escape shallow plateaus.

3. **Exploration**
   * With a tiny probability we add Gaussian noise to the current vector
     (followed by the simplex projection) to diversify the search.
   * If no improvement is seen for a large number of iterations we restart
     from the best solution found so far plus a small random perturbation.

4. **Termination**
   * The loop runs until the allotted time budget (≈1000 s) is exhausted.
   * The best vector observed during the run is returned (as a Python list);
     its exact value is also printed using the provided ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself using a real FFT.
    Length of the result is 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise can create tiny negative values; clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum(a)=1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


def _grad_max_conv(a: np.ndarray, k_star: int) -> np.ndarray:
    """
    Sub‑gradient of 2·n·max(conv) w.r.t. a when the maximal index is k_star.
    The formula is g_i = 4·n·a_{k_star-i} (zero if the partner index is out of range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    # i must satisfy 0 ≤ i < n and 0 ≤ k_star‑i < n.
    i_lo = max(0, k_star - (n - 1))
    i_hi = min(k_star, n - 1)
    if i_lo <= i_hi:
        i = np.arange(i_lo, i_hi + 1, dtype=int)
        j = k_star - i
        g[i] = 4.0 * n * a[j]
    return g


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    Implementation of the algorithm from Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sort descending
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # Degenerate case – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    # ------------------------------------------------------------------
    n = 4000
