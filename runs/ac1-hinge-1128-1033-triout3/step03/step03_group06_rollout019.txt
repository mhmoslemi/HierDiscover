def propose_candidate(...):
    start = time.time()
    deadline = start + budget_s - 0.5
    best_seq = None
    best_score = np.inf

    # seed
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        s = external.sum()
        if s > 0:
            external = external/s
        else:
            external = np.ones_like(external)/len(external)
        cur_score = 2*external.size*max_conv_fft(external)
        best_seq = external.copy()
        best_score = cur_score
        print(f"Seeded from height_sequence_1: length={external.size}, score≈{cur_score:.6f}")
    except NameError:
        pass

    # Parameter sweep lengths
    lengths = [4000, 6000, 8000, 10000, 12000, 16000, 20000]
    # increase later if time.

    for n in lengths:
        if time.time() >= deadline: break

        # define distance array
        d = np.arange(n)
        d = np.minimum(d, n-1-d)  # distances to nearest edge

        # 1) power law sweep
        for p in np.linspace(0.1, 2.5, 35):
            if time.time() >= deadline: break
            a = (d+1)**(-p) + 1e-12
            a = a / a.sum()
            # maybe add tiny epsilon to avoid zero.
            cur = 2*n*max_conv_fft(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"new best len={n} p={p:.3f} score≈{cur:.6f}")

        # 2) sqrt diff shape (p approx .5 but we include multi scaling)
        a = (np.sqrt(d+1) - np.sqrt(d)) + 1e-12
        a = a / a.sum()
        cur = 2*n*max_conv_fft(a)
        if cur < best_score:
            best_score = cur; best_seq = a.copy()

            a = np.exp(-lam * d) + 1e-12
            a = a / a.sum()
            cur = 2*n*max_conv_fft(a)
            if cur < best_score:
                best_score = cur; best_seq = a.copy()

def simulated_annealing(a, time_deadline):
    best = a.copy()
    best_obj = objective(best)
    current = a.copy()
    cur_obj = best_obj

    iter =0
    while time.time() < time_deadline:
        iter+=1
        # propose neighbor
        i,j = np.random.choice(len(a), 2, replace=False)
        # compute delta as small fraction of a_i
        delta = np.random.rand() * min(current[i], 0.1)  # up to 0.1
        new = current.copy()
        new[i] -= delta
        new[j] += delta
        # ensure positivity
        if new[i] < 0:
            continue
        # compute objective
        maxb = max_conv_fft(new)
        new_obj = 2.0 * len(a) * maxb
        # compute acceptance
        if new_obj < cur_obj:
            current = new
            cur_obj = new_obj
            if new_obj < best_obj:
                best = new.copy()
                best_obj = new_obj
        else:
            # temperature schedule
            T = max(0.01, 1.0 - iter / 200000) # linear cooldown
            prob = np.exp(-(new_obj - cur_obj) / T)
            if np.random.rand() < prob:
                current = new
                cur_obj = new_obj
        # optional print
        if iter % 10000 == 0:
            print(f"SA iter {iter}: cur_obj={cur_obj:.6f}, best={best_obj:.6f}")

def propose_candidate(budget_s=1000, seed=12345):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    # seed
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        if external.sum() == 0:
            external = np.ones_like(external) / external.size
        else:
            external = external / external.sum()
        cur_score = 2.0 * external.size * max_conv_fft(external)
        best_seq = external.copy()
        best_score = cur_score
        print(f"Seeded from height_sequence_1: n={external.size}, approx_score={cur_score:.6f}")
    except NameError:
        pass

    if best_seq is None:
        # fallback uniform
        n0 = 4000
        best_seq = np.full(n0, 1.0/n0)
        best_score = 2.0 * n0 * max_conv_fft(best_seq)
        print(f"Fallback uniform: n={n0}, approx_score={best_score:.6f}")

    # Parameter sweep: power law
    lengths = [4000, 6000, 8000, 10000, 12000, 16000, 20000]
    p_vals = np.linspace(0.1, 2.0, 40)  # 40 values
    for n in lengths:
        if time.time() >= deadline:
            break
        d = np.arange(n)
        d = np.minimum(d, n-1-d)
        # iterate p
        for p in p_vals:

            a = (d + 1) ** (-p) + 1e-14
            a = a / a.sum()
            cur = 2.0 * n * max_conv_fft(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"New best: n={n}, p={p:.3f}, score≈{cur:.6f}")
                if best_score < 1.503:
                    break
        # maybe sqrt shape
        a = (np.sqrt(d+1) - np.sqrt(d)) + 1e-14
        a = a / a.sum()
        cur = 2.0 * n * max_conv_fft(a)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"New best sqrt: n={n}, score≈{cur:.6f}")
        if best_score < 1.503:
            break

    # Stage refinement
    # Use the best length found and a few projected gradient steps
    if time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        # FFT size for gradient: need >= 3n-2
        fft_size = 1 << ((3 * n - 2).bit_length())
        step = 0.1
        min_step = 1e-5
        max_step = 1.0
        iter = 0
        max_iters = 4000
        print(f"\nRefinement on n={n} start, initial approx score={best_score:.6f}")
        while time.time() < deadline and iter < max_iters:
            iter += 1

                best_seq = a.copy()

            # Determine active indices (within tolerance)

            active = np.where(conv >= max_val - tol)[0]
            if active.size == 0:

Propose a step function (non‑negative vector) that minimises the metric
used by `evaluate_sequence`.

Algorithm
---------

1.  Start from a seed if `height_sequence_1` is present, otherwise use a
    simple uniform distribution.

2.  Systematically explore *decreasing* power‑law shapes
    a_i ∝ (n‑i)^p for several lengths n (up to 20000) and several

    end, which reduces the central convolution term (the usual bottleneck)
    while keeping the edge squares tiny.  For large n the objective
    2·n·max(conv) tends to 2·(p+1)^2·B(p+1,p+1) (B = beta function);

3.  Keep the best distribution seen in the sweep (using fast FFT‑based

4.  Refine the best candidate with a few rounds of **projected gradient
    descent** on the exact max‑convolution.  The sub‑gradient of the
    max is obtained by a uniform weight on all indices attaining the
    current maximum; after an additive step the iterate is projected onto
    the probability simplex.

5.  Return the final vector.  The function respects the given time budget
    (default 1000 s) and prints progress information.

# FFT‑based convolution helpers

def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a (real FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full convolution a*a using a supplied FFT length."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a⋆a)_k with respect to a.
    a : vector (length n)
    w : weight vector (length 2n‑1, sum to 1)
    size : FFT length (≥ 3n‑2)
    Returns gradient of length n.
    """
    n = a.shape[0]
    L = 2 * n - 1
    A = np.fft.rfft(a, n=size)

    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    idx = L - 1 - np.arange(n)                     # 2n‑2 … n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Input must be 1‑D")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def decreasing_power_law(n: int, p: float) -> np.ndarray:
