"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Deterministic sweep** – try several parametric families
   (U‑shapes, beta, exponential, left‑hand power law) on a handful of
   lengths.  The best candidate becomes the seed for the continuous phase.

2. **Smooth‑gradient refinement** – minimise the smoothed max‑convolution
   using an Adam optimiser on the simplex.
   The smoothing uses a soft‑max weight
       w_k = exp(β·(c_k – max(c)))      (c = self‑convolution)
   which interpolates between a full gradient (small β) and an
   active‑set sub‑gradient (large β).  β is increased slowly,
   the learning‑rate is decayed, and after each step the vector is
   projected onto the probability simplex by simple normalisation.

3. **Upsampling & second refinement** – if time permits the best vector
   is interpolated to a larger length (up to 20000) and refined again.
   Longer vectors can lower the product `n·max(conv)` further.

The whole search runs inside the time budget (≈1000 s) and returns the
best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def next_power_of_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """
    Project v onto the probability simplex by renormalising.
    This is cheaper than the full O(n log n) projection and is sufficient
    because v is already non‑negative after the gradient step.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using an FFT length that is a power of two.
    Returns the full convolution of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = next_power_of_two(2 * n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (a is assumed to sum to 1)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a normalised vector."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Parametric families (all return a vector that already sums to 1)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge: (min(i,n‑1‑i)+1)^(−p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1.0) ** (-p) + 1e-14
    return a / a.sum()


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))^(−p)."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    return a / a.sum()


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}: a[i] ∝ (i+1)^{α‑1} (n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta - 1.0)) + 1e-14
    return a / a.sum()


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges: exp(−γ·dist)."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    return a / a.sum()


def harmonic_left_distribution(n: int, p: float) -> np.ndarray:
    """One‑sided power law: a[i] ∝ (i+1)^{−p} (mass concentrated on the left)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p) + 1e-14
    return a / a.sum()


# ----------------------------------------------------------------------
# Core refinement routine (Adam on a smoothed max‑convolution)
# ----------------------------------------------------------------------


def refine_sequence(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine `start_seq` (already normalised) using Adam with a soft‑max
    approximation of the max‑convolution.
    Returns the best sequence found and its score.
    """
    a = start_seq.copy()
    n = a.shape[0]

    # FFT sizes (powers of two) for the two convolutions we need
    size_conv = next_power_of_two(2 * n)          # for a*a
    size_grad = next_power_of_two(3 * n)          # for cross‑correlation

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedules
    step = 0.3           # initial learning‑rate
    step_decay = 0.9999  # geometric decay per iteration
    beta_smooth = 0.5    # initial smoothing coefficient
    beta_inc = 1.001     # multiplicative increase each iteration

    best_local_a = a.copy()
    best_local_score = score_from_vector(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ---------- self‑convolution ----------
        fa = np.fft.rfft(a, n=size_conv)
        conv_full = np.fft.irfft(fa * fa, n=size_conv)
        conv = conv_full[: 2 * n - 1]
        max_c = conv.max()

        # ---------- smooth weights (soft‑max) ----------
        # diff ≤ 0, exp(beta·diff) ∈ (0,1]
        w = np.exp(beta_smooth * (conv - max_c))

        # ---------- gradient via cross‑correlation ----------
        a_rev = a[::-1]

        fw = np.fft.rfft(w, n=size_grad)
        grad_full = np.fft.irfft(fra * fw, n=size_grad)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice length n

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        t_corr = 1.0 - beta1 ** it
        v_corr = 1.0 - beta2 ** it
        m_hat = m / t_corr
        v_hat = v / v_corr
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- projection onto simplex ----------
        a = simplex_normalise(a)

        # ---------- evaluate new score ----------
        fa_new = np.fft.rfft(a, n=size_conv)
        conv_new = np.fft.irfft(fa_new * fa_new, n=size_conv)[: 2 * n - 1]
        score = 2.0 * n * conv_new.max()

        if score < best_local_score - 1e-12:
            best_local_score = score

        beta_smooth *= beta_inc

            a = simplex_normalise(a + noise)

            v.fill(0.0)

            print(f"[refine] iter {it:7d} – injected noise, reset Adam state")

        if step < 1e-12:
            break

    return best_local_a, best_local_score


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of `seq` onto a longer grid of size `new_n`.
    Returned vector is renormalised to sum‑1.
    """
    old_n = seq.shape[0]
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    return simplex_normalise(new_seq)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 3.0   # a small safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 1. Deterministic sweep over several families
    # --------------------------------------------------------------
    lengths = [4000, 6000, 8000, 12000, 16000]   # modest set of lengths
    p_grid = np.arange(0.2, 2.01, 0.2)
    gamma_grid = np.arange(0.001, 0.021, 0.004)
    alpha_grid = np.arange(0.2, 2.01, 0.4)
    beta_grid = np.arange(0.2, 2.01, 0.4)

    print("=== Deterministic sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break
        # U‑shape (min distance)
        for p in p_grid:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (u_min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # U‑shape (product distance)
        for p in p_grid:
            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (u_prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # One‑sided power law
        for p in p_grid:
            a = harmonic_left_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (left)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # Exponential decay (symmetric)
        for g in gamma_grid:
            a = exp_decay_distribution(n, g)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (exp)    n={n:5d} γ={g:5.3f}  score={sc:.6f}")

        # Asymmetric Beta distributions (α,β)
        for alpha in alpha_grid:
            for beta in beta_grid:
                a = beta_distribution(n, alpha, beta)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"new best (beta)   n={n:5d} α={alpha:4.2f} β={beta:4.2f}  score={sc:.6f}")

        # occasional time check
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 2. Random Dirichlet seeds (few quick samples)
    # --------------------------------------------------------------
    if time.time() < deadline:
        for _ in range(8):
            n = np.random.choice(lengths)
            # concentration <1 gives spikier vectors, >1 gives smoother
            conc = np.random.uniform(0.2, 1.5)
            a = np.random.gamma(conc, 1.0, size=n)
            a = simplex_normalise(a)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (rand)   n={n:5d} conc={conc:.2f}  score={sc:.6f}")

    # --------------------------------------------------------------
    # 3. Fallback if nothing succeeded (unlikely)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 8000
        best_seq = np.random.rand(n)
        best_seq = simplex_normalise(best_seq)
        best_score = score_from_vector(best_seq)
        print(f"fallback random   n={n}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Continuous refinement (Adam + smooth max)
    # --------------------------------------------------------------
    # First refinement on the best length found
    print("\n=== Continuous refinement (phase 1) ===")
    refined_seq, refined_score = refine_sequence(best_seq, deadline)
    if refined_score < best_score:
        best_score, best_seq = refined_score, refined_seq.copy()
        print(f"Phase‑1 improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Upsample and second refinement (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # target a larger length, but not exceeding 20000
        new_n = min(20000, len(best_seq) * 2)
        if new_n > len(best_seq):
            print("\n=== Upsampling & second refinement (phase 2) ===")
            up_seq = upsample_sequence(best_seq, new_n)
            refined_seq2, refined_score2 = refine_sequence(up_seq, deadline)
            if refined_score2 < best_score:
                best_score, best_seq = refined_score2, refined_seq2.copy()
                print(f"Phase‑2 improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Return result
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator will again clip/normalise, so we return the raw vector
    return best_seq.tolist()
