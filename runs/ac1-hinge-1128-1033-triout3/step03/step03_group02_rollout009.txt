def run_optim(a, time_budget):
    n = len(a)
    best_a = a.copy()
    best_val = float('inf')
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    iteration = 0
    start_time = time.time()
    while True:
        # check time
        now = time.time()
        if now - start_time >= time_budget:
            break
        # conv
        conv_a = _full_convolution_fft(a, a)
        M = conv_a.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[run] iter {iteration} best {best_val:.6f}")
        # compute progress for schedule
        # Use total time y = (now - start_time) / time_budget
        prog = (now - start_time) / time_budget
        # tau schedule
        tau_start = M * 0.3  # maybe
        tau_end = M * 1e-6 + 1e-12
        tau = tau_start * (1 - prog) + tau_end * prog
        # compute softmax weights
        if tau > 0:
            exp_args = (conv_a - M) / tau
            # shift to avoid overflow
            # compute max arg
            max_arg = np.max(exp_args)  # should be 0
            exp_vals = np.exp(exp_args - max_arg)  # subtract max to avoid overflow
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv_a)
                w[np.argmax(conv_a)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0

        # gradient
        grad = _gradient_from_weights(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        # step size schedule
        eta_start = 0.5
        eta_end = 1e-5
        eta = eta_start * (1 - prog) + eta_end * prog

        step = m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - eta * step

        # jitter
        if np.random.rand() < 0.01:
            src = np.random.randint(n)
            dst = np.random.randint(n-1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a_candidate[src] * 0.5
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        # maintain symmetry (optional)
        # a_candidate = (a_candidate + a_candidate[::-1]) * 0.5

        # projection onto simplex
        a = _project_onto_simplex(a_candidate)

        iteration += 1

        # maybe every few thousand iteration recompute conv to avoid drift (but already recompute each)
    return best_a, best_val

"""
Propose a step‑function sequence that minimises the evaluation function.

The algorithm is a long‑run Adam‑style optimisation on the simplex.
We start from several diverse initialisations (the known best
`height_sequence_1`, a sinusoidal “U” shape, an exponential edge‑heavy
shape and a random Dirichlet).  For each initialisation we run an
Adam‐optimiser that minimises the smooth approximation
``max_k (a*a)[k]`` using a soft‑max (temperature τ) over the
auto‑convolution entries.  The iterates are projected onto the
probability simplex after every step.  A small random “jitter” occasionally
moves a fraction of mass between two positions to help escape plateaux.
The optimisation runs until the allotted 1000 s budget expires; the best
sequence seen over all runs is returned.

The code is deliberately written with only top‑level functions (no closures,
no lambdas) and uses only numpy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full convolution using a real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x*y (length len(x)+len(y)-1)."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient from soft‑max weights
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented via a single convolution: reverse w and convolve with a,
    then pick the appropriate slice.
    """
    n = a.shape[0]
    u = w[::-1]                       # reversed weight vector
    conv = _full_convolution_fft(u, a)  # length = (2n-1)+n-1 = 3n-2
    start = 2 * n - 2
    idx = start - np.arange(n)        # indices for i = 0..n-1
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Objective (fast internal version, identical to evaluate_sequence when Σa=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Compute 2·n·max(conv(a,a))  (assumes Σa = 1)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _init_sin(n: int) -> np.ndarray:
    """U‑shape: 1 - sin(π x/(n-1)), projected onto the simplex."""
    x = np.arange(n, dtype=float)
    a = 1.0 - np.sin(np.pi * x / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _init_exp(n: int, alpha: float = 0.008) -> np.ndarray:
    """Edge‑heavy exponential decay from both ends, normalised."""
    i = np.arange(n, dtype=float)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _init_uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=float)


def _init_random(n: int, concentration: float = 0.5) -> np.ndarray:
    """Dirichlet random start."""
    a = np.random.gamma(concentration, 1.0, size=n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _init_from_global() -> np.ndarray | None:
    """If the global variable height_sequence_1 exists, use it."""
    try:
        a = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if a.ndim != 1:
            return None
        if (a < 0).any():
            a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            return None
        return a / s
    except Exception:
        return None


# ----------------------------------------------------------------------
# Core optimisation routine (Adam + soft‑max)
# ----------------------------------------------------------------------
def _run_adam_opt(
    a0: np.ndarray,
    time_budget: float,
    *,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eta_start: float = 0.5,
    eta_end: float = 1e-5,
    tau_factor: float = 0.3,
    tau_end_factor: float = 1e-6,
    jitter_prob: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Run Adam‑style optimisation for at most ``time_budget`` seconds.
    Returns the best sequence found and its internal objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _objective(best_a)

    start = time.time()
    iteration = 0

    while True:
        now = time.time()
        if now - start >= time_budget:
            break

        # --- convolution and objective -------------------------------------------------
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)

        # store the best candidate
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter {iteration:8d} new best = {best_val:.6f}")

        # --- schedule (progress between 0 and 1) ---------------------------------------
        prog = (now - start) / time_budget

        # temperature for the soft‑max
        tau_start = max(M * tau_factor, 1e-12)
        tau_end = max(M * tau_end_factor, 1e-14)
        tau = tau_start * (1.0 - prog) + tau_end * prog

        # --- soft‑max weights ---------------------------------------------------------
        if tau > 0.0:
            # (conv - M) ≤ 0, shift by the maximum (=0) to avoid overflow
            exp_args = (conv - M) / tau
            # use a stable exponentiation
            exp_shift = np.max(exp_args)  # should be 0 but keep for safety
            exp_vals = np.exp(exp_args - exp_shift)
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # --- gradient ---------------------------------------------------------------
        grad = _gradient_from_weights(a, w)

        # --- Adam update ------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        # learning‑rate schedule
        eta = eta_start * (1.0 - prog) + eta_end * prog

        step = m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - eta * step

        # --- occasional random mass transfer (jitter) ---------------------------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a_candidate[src] * 0.5
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        # --- project back onto the simplex -------------------------------------------
        a = _project_onto_simplex(a_candidate)

        iteration += 1

    return best_a, best_val
