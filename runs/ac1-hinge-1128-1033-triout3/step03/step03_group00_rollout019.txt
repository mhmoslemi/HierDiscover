def propose_candidate(seed=42, budget_s=1000, **kwargs):
    # initialize
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    
    n = 5000   # maybe 5000 to provide more degrees
    # attempt to initialize from height_sequence_1 if present else uniform.
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        old_x = np.linspace(0,1,old.shape[0])
        new_x = np.linspace(0,1,n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0)
    except Exception:
        a = np.full(n, 1.0/n)
    # Ensure sum=1
    a = a / a.sum()
    
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    # best_val is objective value (2 * n * max conv).
    
    T0 = best_val * 0.2 # set initial temp.
    T_end = best_val * 0.001
    # Temperature schedule: exponential.
    
    iter = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute conv and max using FFT (since we need full conv each step)
        conv = _full_convolution(a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max  # same as evaluate_sequence without extra division.
        # To compute exact value using evaluate_sequence if needed:
        # cur_val = evaluate_sequence(a.tolist()) # slower.
        # Determine temperature based on progress.
        progress = (now - start) / (deadline - start)
        T = T0 * (1 - progress) + T_end * progress   # linear schedule.
        
        # decide move
        if np.random.rand() < 0.7:
            # targeted move
            k = int(conv.argmax())
            # indices i where both exist
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # compute products for indices in range
            idx_range = np.arange(i_start, i_end + 1, dtype=int)
            prod = a[idx_range] * a[k - idx_range]
            # choose i with max product
            max_idx = idx_range[np.argmax(prod)]
            counterpart = k - max_idx
            # which one to reduce?
            if a[max_idx] >= a[counterpart]:
                reduce_idx = max_idx
            else:
                reduce_idx = counterpart
            # increase_idx choose random low-mass index
            low_candidates = np.argpartition(a, kth= int(0.05 * n))[:int(0.05 * n)]  # bottom 5%
            increase_idx = int(np.random.choice(low_candidates))
        else:
            # random move
            reduce_idx, increase_idx = np.random.choice(n, 2, replace=False)
        # Determine delta
        # step size proportional to current max and temperature
        step = 0.01 * a[reduce_idx] * (T / (T0 + 1e-12) + 0.1)  # adapt
        # ensure not too large, maybe also cap to a[reduce_idx]/2
        max_delta = min(step, a[reduce_idx] * 0.5, 0.1)  # don't move too big
        delta = np.random.rand() * max_delta
        if delta <= 0:
            continue
        # create new a candidate
        a_new = a.copy()
        a_new[reduce_idx] -= delta
        a_new[increase_idx] += delta
        # Ensure nonnegativity (a_new[reduce_idx] >=0)
        # compute new conv
        conv_new = _full_convolution(a_new)
        max_new = conv_new.max()
        val_new = 2.0 * n * max_new
        # accept?
        if val_new < cur_val or np.random.rand() < np.exp(-(val_new - cur_val) / (T + 1e-12)):
            a = a_new
            cur_val = val_new
            cur_max = max_new
        # track best
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # print progress
            print(f"[{iter:7d}] new best = {best_val:.6f}")
        iter += 1
        # occasional reset?
        # maybe after certain no improvement steps.
    # final evaluation using evaluate_sequence
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported = {final_score:.6f}")
    return best_a.tolist()

from scipy.optimize import linprog
import scipy.sparse as sp

def lp_optimize(f, M):
    n = len(f)
    # Build constraint matrix A_ub shape (2*n-1, n)
    # row k: sum_i f_i * b_{k-i} <= M
    # This is convolution matrix of f.
    rows = 2*n-1
    # Build sparse matrix
    diags = []
    offsets = []
    for i, fi in enumerate(f):
        if fi == 0:
            continue
        # diagonal offset i in rows
        data = np.full(n, fi)
        diags.append(data)
        offsets.append(i)
    A = sp.diags(diags, offsets, shape=(rows, n), format='csr')
    # Build RHS: M repeated rows times
    b_ub = np.full(rows, M)
    # Objective: maximize sum b => minimize -sum b
    c = -np.ones(n)
    # Bounds: b_j >=0
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        raise RuntimeError('LP failed')
    b = result.x
    return b
```

def project_simplex(v):
    # https://arxiv.org/abs/1101.6081; simple O(n log n) algorithm.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import time, numpy as np
    from numpy.fft import rfft, irfft

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # problem size
    n = 4000

    # function for full convolution using FFT
    def full_conv(x):
        m = x.shape[0]
        out_len = 2*m - 1
        size = 1 << (out_len - 1).bit_length()
        fx = np.fft.rfft(x, size)
        conv = np.fft.irfft(fx*fx, size)[:out_len]
        return conv

    # simplex projection
    def proj_simplex(v):
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # fallback: uniform
            return np.full_like(v, 1.0/len(v))
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

    # candidate generator
    def make_sequence(shape, param):
        if shape == 'uniform':
            a = np.full(n, 1.0/n)
        elif shape == 'linear':
            # param = p exponent for (1 - i/(n-1))^p
            p = param
            x = np.arange(n) / (n-1)
            a = (1 - x) ** p
            a = a / a.sum()
        elif shape == 'exp':
            lam = param
            i = np.arange(n)
            a = np.exp(-lam * i)
            a = a / a.sum()
        elif shape == 'power':
            # a_i ∝ (i+1)^p (increasing)
            p = param
            i = np.arange(1, n+1)
            a = i ** p
            a = a / a.sum()
        elif shape == 'sin':
            # a_i ∝ sin(pi * i / (n-1))
            i = np.arange(n)
            a = np.sin(np.pi * i / (n-1))
            a = a / a.sum()
        else:
            a = np.full(n, 1.0/n)
        return a

    # Explore a small set of parametric families
    best_a = None
    best_val = float('inf')
    # list of families and parameter ranges
    families = [
        ('uniform', [None]),
        ('linear', [0.2, 0.5, 1.0, 1.5, 2.0]),
        ('exp', [1e-4, 2e-4, 5e-4, 1e-3, 2e-3]),
        ('power', [0.2, 0.5, 1.0, 2.0]),
        ('sin', [None])
    ]

    print("[search] start parametric scan")
    for shape, params in families:
        for p in params:
            a = make_sequence(shape, p)
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[scan] new best {val:.6f} shape={shape} param={p}")
            # check time
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # Local subgradient descent to refine best_a
    if best_a is None:
        # fallback to uniform
        best_a = np.full(n, 1.0/n)
        best_val = evaluate_sequence(best_a.tolist())
    a = best_a.copy()
    best_val = evaluate_sequence(a.tolist())
    best_seq = a.copy()

    # hyperparameters for subgradient descent
    lr0 = 0.5   # initial learning rate (large)
    lr_end = 0.01
    t_no_improve = 0
    MAX_NO_IMPROVE = 2000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = full_conv(a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            t_no_improve = 0
            print(f"[{iteration:7d}] refine best = {best_val:.6f}")
        else:
            t_no_improve += 1

        # stop if already very good
        if best_val <= 1.503:
            print("[info] target reached, early exit")
            break

        # learning rate schedule
        progress = (now - start) / (deadline - start)
        lr = lr0 * (1.0 - progress) + lr_end * progress

        # find index of max conv
        k_max = int(conv.argmax())

        # subgradient
        g = np.zeros_like(a)
        # fill using loop (fast enough)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                g[i] = 2.0 * a[j]
        # gradient descent step
        a_new = a - lr * g
        # project onto simplex
        a = proj_simplex(a_new)

        # occasional random perturbation to escape plateaus
        if t_no_improve > MAX_NO_IMPROVE:
            print(f"[{iteration:7d}] random restart after {t_no_improve} idle steps")
            a = np.random.rand(n)
            a = proj_simplex(a)
            t_no_improve = 0

        iteration += 1

        # periodic recompute with exact evaluate (already done)
        if iteration % 5000 == 0:
            pass

    # final evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best evaluation reported = {final_score:.6f}")
    return best_seq.tolist()

    without any gradient tricks.

Phase 2 – **Sub‑gradient descent on the true max‑convolution**:
    The objective is 2·n·max(conv(a,a)).  The gradient of the active term
    conv_k is simply 2·a shifted by k, so we can take a sub‑gradient step
    that directly reduces the current worst‑case convolution entry.  After
    each step the vector is projected back onto the probability simplex
    (a_i ≥ 0, Σa_i = 1).  An annealing schedule on the step‑size and a
    restart mechanism avoid stagnation.

Both phases respect the 1000‑second budget, use only NumPy and the already
provided ``evaluate_sequence`` and return the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x * x using a real FFT (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # tiny negative values caused by round‑off are clipped
    np.maximum(conv, 0, out=conv)
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = 1}.
    Implements the O(n log n) algorithm from
    Wang & Carreira‑Perpinán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1        # index (0‑based)
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_sequence(shape: str, param, n: int) -> np.ndarray:
    """Return a non‑negative vector of length n according to a simple parametric family."""
    if shape == "uniform":
        a = np.full(n, 1.0 / n)
    elif shape == "linear":
        # a_i ∝ (1 - i/(n‑1))**p    (p ≥ 0)
        p = param
        x = np.arange(n) / (n - 1)
        a = (1.0 - x) ** p
        a = a / a.sum()
    elif shape == "exp":
        # a_i ∝ exp(‑λ·i)
        lam = param
        i = np.arange(n)
        a = np.exp(-lam * i)
        a = a / a.sum()
    elif shape == "power":
        # a_i ∝ (i+1)**p   (p ≥ 0, increasing shape)
        p = param
        i = np.arange(1, n + 1)
        a = i.astype(float) ** p
        a = a / a.sum()
    elif shape == "sin":
        # a_i ∝ sin(π·i/(n‑1)), zero at the edges
        i = np.arange(n)
        a = np.sin(np.pi * i / (n - 1))
        a = a / a.sum()
    else:
        # safety fallback
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine first scans a few parametric families,
    then refines the best candidate with a sub‑gradient descent on the exact
    max‑convolution objective.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Problem size – 4000 provides a good trade‑off between quality and speed.
    n = 4000

    # --------------------------------------------------------------
    # Phase 1 – cheap parametric scan
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)            # uniform fallback
    best_val = float("inf")

    families = [
        ("uniform", [None]),
        ("linear",   [0.2, 0.5, 1.0, 1.5, 2.0]),
        ("exp",      [5e-5, 1e-4, 2e-4, 5e-4, 1e-3]),
        ("power",    [0.2, 0.5, 1.0, 2.0]),
        ("sin",      [None]),
    ]

    print("[search] start parametric scan")
    for shape, params in families:
        for p in params:
            a = _make_sequence(shape, p, n)
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[scan] new best {best_val:.6f} – shape={shape} param={p}")
            # stop early if we run out of time
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase 2 – sub‑gradient descent on max(conv)
    # --------------------------------------------------------------
    a = best_a.copy()
    # Re‑evaluate with the exact formula (avoid double work later)
    conv = _full_convolution(a)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_val = cur_val
    best_seq = a.copy()

    # hyper‑parameters for the descent
    lr_start = 0.5          # big step at the beginning
    lr_end = 0.01           # tiny step near the end
    max_no_improve = 3000   # steps without improvement before a restart
    no_improve_cnt = 0
    it = 0

    print("[refine] start sub‑gradient refinement")
    while True:
        now = time.time()
        if now >= deadline:
            break

        # compute full convolution and current objective
        conv = _full_convolution(a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve_cnt = 0
            print(f"[{it:7d}] refine best = {best_val:.6f}")
        else:
            no_improve_cnt += 1

        # early exit if we already beat the target
        if best_val <= 1.503:
            print("[refine] target reached – stopping early")
            break

        # learning‑rate annealing (linear)
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (1.0 - progress) + lr_end * progress

        # index attaining the maximum convolution entry
        k_max = int(conv.argmax())

        # sub‑gradient of the active term:  g_i = 2·a_{k_max‑i} (zero outside)
        g = np.zeros_like(a)
        # vectorised fill using NumPy slicing (still O(n))
        if k_max < n:
            g[: k_max + 1] = 2.0 * a[k_max :: -1]
        else:
            start = k_max - (n - 1)
            g[start:] = 2.0 * a[: n - start][::-1]

        # gradient step followed by projection onto the simplex
        a_new = a - lr * g
        a = _project_simplex(a_new)

        # occasional random restart to escape plateaus
        if no_improve_cnt >= max_no_improve:
            print(f"[{it:7d}] restart after {no_improve_cnt} idle steps")
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve_cnt = 0

        it += 1

        # lightweight progress output
        if it % 2000 == 0:
            print(f"[{it:7d}] current best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final reporting (use the provided ``evaluate_sequence`` to be safe)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    return best_seq.tolist()
