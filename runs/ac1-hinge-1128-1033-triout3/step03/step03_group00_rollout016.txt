def _project_onto_simplex(v, z=1):
    # v is a real numpy array
    # From https://arxiv.org/abs/1309.1541
    # returns w such that w_i >= 0 and sum w_i = z and w is closest to v in Euclidean distance
    if v.sum() == z and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
A fast, multi‑restart optimizer for the step‑function objective
``evaluate_sequence``.

The algorithm works directly on the feasible simplex
    a_i ≥ 0 ,  Σ a_i = √(2·n)          (the same normalisation used by the
    reference solution).
A soft‑max re‑parameterisation is avoided – we use a simple Euclidean
projection onto the simplex after each gradient step.  The gradient is
computed for a smooth log‑sum‑exp approximation of the max‑convolution
and the update follows the Adam scheme.  In addition:

* a cheap linear‑programming “sum‑increase” step (LP) is performed every
  few thousand iterations.  The LP maximises Σg_j under the linear
  constraints (f * g)_k ≤ M (the current max‑convolution) and g ≥ 0.
  The optimal g is then mixed with the current iterate; this typically
  reduces the max‑convolution further.

* the whole search is split into short independent runs with different
  random seeds and (optionally) different problem sizes.  The best
  sequence seen before the time budget expires is returned.

The code respects the required signature ``propose_candidate`` and
contains no closures or file I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: FFT based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x * x (length 2·len(x)‑1) using a real FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex Σ a_i = z , a_i ≥ 0
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Projects v onto the simplex {a ≥ 0, Σ a = z}."""
    if z < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # fast path – already feasible
    if np.all(v >= 0) and np.isclose(v.sum(), z):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):  # all entries will be zero
        theta = cssv[-1] / len(v)
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optional: linear programming step (maximise Σ g_j s.t. f*g ≤ M)
# ----------------------------------------------------------------------
def _lp_max_sum_given_bound(f: np.ndarray, M: float, target_sum: float):
    """
    Solve:
        maximise   Σ g_j
        subject to (f * g)_k ≤ M   for all k
                  g ≥ 0
    Returns a feasible g (scaled to have Σ g = target_sum) or None.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import diags, csr_matrix
    except Exception:   # pragma: no cover – SciPy not available
        return None

    n = len(f)
    # Build the Toeplitz matrix: each diagonal i contains the constant f[i]
    # data has shape (n, n): row i is filled with f[i]
    data = np.tile(f[:, np.newaxis], (1, n))
    offsets = np.arange(n)               # 0 … n‑1
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g → minimise −Σ g
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, 1000.0) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None

    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s <= 0:
        return None
    # scale to the required total mass
    g = g * (target_sum / s)
    return g

                a_lp = _project_onto_simplex(a_lp, target_sum)
                conv_lp = _full_convolution(a_lp)
                val_lp = float(2.0 * n_primary * conv_lp.max() / (np.sum(a_lp) ** 2))
                if val_lp < best_val:
                    best_val = val_lp
                    best_a = a_lp.copy()
                    print(f"[{iteration:7d}] LP boost improved to {best_val:.6f}")
                a0 = a_lp

        # ----- restart if stuck for a long time --------------------------------
        if no_improve > RESTART_LIMIT:
            print(f"[{iteration:7d}] restart after {no_improve} idle steps")
            a0 = np.random.rand(n_primary)
            a0 = _project_onto_simplex(a0, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    # ----------------------------------------------------------------------
    # Final reporting (use the supplied evaluate_sequence for safety)
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
