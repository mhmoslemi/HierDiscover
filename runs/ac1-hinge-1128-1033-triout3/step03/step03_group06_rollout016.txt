def propose_candidate(budget_s=1000, seed=42):

        fft_size = 1 << ((3 * n - 2).bit_length())
        # hyperparameters
        eta = 0.5
        eta_min = 1e-6
        eta_max = 2.0
        max_iters = 20000
        for it in range(max_iters):
            if time.time() >= deadline:
                break
            # compute convolution
            conv = full_conv_fft(a, fft_size)  # length 2*n-1
            max_b = conv.max()
            cur_score = 2.0 * n * max_b
            if cur_score < global_best_score:
                global_best_score = cur_score
                global_best_seq = a.copy()
                print(f"n={n}, iter={it}, new global best={cur_score:.6f}")
                if global_best_score < 1.503:
                    print("Target achieved early.")
                    break
            # entropy gradient
            w = np.log(conv + 1e-30) + 1.0
            grad = gradient_entropy(a, w, fft_size)  # array length n
            # multiplicative update
            exp_arg = -eta * grad
            exp_arg = np.clip(exp_arg, -50.0, 50.0)
            a_candidate = a * np.exp(exp_arg)
            # renormalize to simplex
            a_candidate = project_to_simplex(a_candidate)
            # evaluate candidate
            conv_c = full_conv_fft(a_candidate, fft_size)
            cand_score = 2.0 * n * conv_c.max()
            if cand_score < cur_score:

                eta = min(eta * 1.05, eta_max)

Propose a step‑function (non‑negative vector) that minimises the
evaluation metric used by `evaluate_sequence`.

Algorithm
---------

1.  Initialise the distribution with a symmetric “arcsine’’ shape:
    a_i ∝ 1/√((i+½)·(n‑i‑½)).  This distribution is known to give a
    nearly flat self‑convolution (the continuous analogue makes the
    sum of two i.i.d. draws almost uniform).

    The self‑convolution b = a∗a is a probability distribution on
    {0,…,2n‑2}.  Maximising its entropy H(b)=−∑b log b makes b as
    flat as possible, which directly reduces the objective

    An exponentiated (multiplicative) update keeps a non‑negative and
    a projection onto the simplex enforces ∑a=1.

3.  Several length values are tried (the search is time‑bounded, so
    later lengths may be skipped).  The best vector seen overall is
    finally evaluated with the official `evaluate_sequence` and returned.

The routine respects a 1000 s budget, prints progress information and
never performs any file/network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a (real FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using supplied FFT size."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("input must be 1‑D")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Symmetric arcsine‑type distribution:
        a_i ∝ 1 / sqrt( (i+0.5)·(n‑i‑0.5) )
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    denom = np.sqrt((i + 0.5) * (n - i - 0.5))
    a = 1.0 / denom
    a /= a.sum()
    return a


def gradient_entropy(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the *negative* entropy –H(a∗a) w.r.t. a.
    w = log(b) + 1 where b = a∗a.
    The result is `2 * conv_rev[L-1-i]` with L=2n‑1.
    """
    n = a.shape[0]
    L = 2 * n - 1
    w_rev = w[::-1]                     # reverse for the required convolution
    A = np.fft.rfft(a, n=size)
    W_rev = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]   # length 3n‑2
    idx = L - 1 - np.arange(n)           # positions that give the gradient
    grad = 2.0 * conv[idx]
    return grad


# ----------------------------------------------------------------------
#  Main search routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """
    Search for a non‑negative step function that minimises the metric
    used by `evaluate_sequence`.  Returns the best list of floats found
    before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 0.  Optional external seed (height_sequence_1) if it exists
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    try:
        ext = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if ext.size:
            ext = np.maximum(ext, 0.0)
            ext_sum = ext.sum()
            if ext_sum > 0:
                ext = ext / ext_sum
                cur = 2.0 * ext.size * max_conv_fft(ext)
                best_seq = ext.copy()
                best_score = cur
                print(f"Seeded from height_sequence_1: n={ext.size}, score≈{cur:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths we will try (larger n gives a larger theoretical lower bound)
    # ------------------------------------------------------------------
    lengths = [8000, 12000, 20000, 30000]   # sorted automatically
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = np.inf

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- initial distribution for this length --------------------
        if best_seq is not None and best_seq.shape[0] == n:
            a = best_seq.copy()
        else:
            a = arcsine_distribution(n)

        # ---- FFT size – must be ≥ 3·n‑2 for the entropy gradient -------
        fft_size = 1 << ((3 * n - 2).bit_length())

        # ---- optimisation hyper‑parameters --------------------------------
        eta = 0.5            # multiplicative step size
        eta_min = 1e-6
        eta_max = 2.0
        max_iters = 20000    # generous upper bound; real stop is time‑based

        print(f"\n=== Optimising length n={n} ===")
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # ----- exact convolution and current objective -----------------
            conv = full_conv_fft(a, fft_size)          # length 2·n‑1
            max_b = conv.max()
            cur_score = 2.0 * n * max_b

            # ----- update global best ------------------------------------
            if cur_score < global_best_score:
                global_best_score = cur_score
                global_best_seq = a.copy()
                print(f"Iter {it:5d} – new global best: {cur_score:.6f}")
                if global_best_score < 1.503:
                    print("Target reached – terminating early.")
                    break

            # ----- entropy gradient (−H) ---------------------------------
            w = np.log(conv + 1e-30) + 1.0                # shape (2n‑1,)
            grad = gradient_entropy(a, w, fft_size)      # shape (n,)

            # ----- multiplicative mirror‑descent step --------------------
            exp_arg = -eta * grad
            exp_arg = np.clip(exp_arg, -50.0, 50.0)        # protect from overflow
            a_candidate = a * np.exp(exp_arg)

            # project back onto the probability simplex
            a_candidate = project_to_simplex(a_candidate)

            # ----- evaluate candidate (exact max) -------------------------
            conv_cand = full_conv_fft(a_candidate, fft_size)
            cand_score = 2.0 * n * conv_cand.max()

            # ----- acceptance rule ----------------------------------------
            if cand_score < cur_score:
                a = a_candidate
                eta = min(eta * 1.05, eta_max)   # success → be a bit bolder
            else:
                eta = max(eta * 0.70, eta_min)   # failure → shrink step

            # occasional random jitter to escape tiny plateaus
            if (it + 1) % 5000 == 0:
                noise = np.random.randn(n) * 0.005
                a = a + noise
                a = np.maximum(a, 0.0)
                a = project_to_simplex(a)

        # use the final a as a seed for the next length
        best_seq = a.copy()

        if global_best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3.  Final official evaluation & return
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # fallback – uniform distribution of required length
        n_fallback = lengths[-1] if lengths else 8000
        global_best_seq = np.full(n_fallback, 1.0 / n_fallback)

    try:
        exact_score = evaluate_sequence(global_best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")
        exact_score = global_best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {global_best_score:.6f}")
    print(f"Best length            : {global_best_seq.shape[0]}")
    return global_best_seq.tolist()
