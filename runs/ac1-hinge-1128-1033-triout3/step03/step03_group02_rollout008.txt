"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------

* Initialise a symmetric probability vector on the simplex.
  If a previous construction ``height_sequence_1`` exists we up‑sample it,
  otherwise a simple “U‑shape’’ (high at the ends, low in the middle) is used.

* Optimise the vector with a projected Adam optimiser.
  The objective for a normalised vector (∑a=1) is 2·n·max(a∗a).
  At each iteration we
    – compute the full auto‑convolution via a real FFT,
    – build a soft‑max weight vector over the convolution (temperature τ
      decays linearly to a very small value),
    – obtain a (smoothed) sub‑gradient `g = 2·conv(w_rev, a)`,
    – perform an Adam step,
    – enforce symmetry `a←(a+a[::-1])/2`,
    – project back onto the probability simplex.

* Occasionally a random “mass‑transfer’’ move is applied (probability
  ``jitter_prob``) and, if no improvement is seen for many iterations,
  the current iterate is re‑initialised to a random symmetric vector
  (a simple simulated‑annealing restart).

* The best vector seen so far (lowest value of ``2·n·max(conv)``) is kept
  and, after the time budget, its exact score is reported via
  ``evaluate_sequence``.

The combination of a smoother temperature schedule, Adam’s adaptive
step‑sizes, symmetry enforcement and occasional random restarts yields
a tighter upper bound than the previous soft‑max momentum descent.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al.)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Full (linear) convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x*y using an FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Clip tiny negative values that appear due to round‑off.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max weighted convolution
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i using one FFT.
    `a` has length n, `w` length 2n‑1.
    """
    n = a.shape[0]
    u = w[::-1]                     # reversed weight vector
    conv = _full_convolution_fft(u, a)   # length 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)          # selects the needed slice
    return 2.0 * conv[idx]


# ----------------------------------------------------------------------
# Helper: initialise a symmetric probability vector of length n
# ----------------------------------------------------------------------
def _initial_vector(n: int, seed: int) -> np.ndarray:
    rng = np.random.default_rng(seed)
    # Use the previously discovered construction if present.
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple, np.ndarray)):
        base = np.asarray(base, dtype=float)
        base = np.maximum(base, 0.0)
    else:
        # Simple U‑shape: large at the ends, small in the centre.
        x = np.arange(n, dtype=float)
        base = 1.0 - np.sin(np.pi * x / (n - 1))
        base = np.maximum(base, 0.0)

    # Upsample / downsample to the target length.
    if base.shape[0] != n:
        old_x = np.linspace(0.0, 1.0, base.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        base = np.interp(new_x, old_x, base)

    # Enforce symmetry and normalise onto the simplex.
    a = (base + base[::-1]) / 2.0
    a = _project_onto_simplex(a)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Adam based projected optimisation with soft‑max smoothing,
    symmetry enforcement and occasional random restarts.
    Returns the best non‑negative step‑function found before the time budget
    expires (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the final evaluation call.
    deadline = start_time + budget_s - 1.5

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    # ------------------------------------------------------------------
    # Larger n gives a finer discretisation of the limiting shape.
    # 4096 is a good compromise between resolution and speed.
    n = 4096
    a = _initial_vector(n, seed)

    # ------------------------------------------------------------------
    # Adam hyper‑parameters ---------------------------------------------
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)   # first moment estimate
    v = np.zeros_like(a)   # second moment estimate
    beta1_pow = 1.0
    beta2_pow = 1.0

    # ------------------------------------------------------------------
    # Learning‑rate & temperature schedules -----------------------------
    # ------------------------------------------------------------------
    # The schedules are linear in the elapsed proportion of the budget.
    # They are defined after the first convolution (to obtain M0).
    conv_a = _full_convolution_fft(a, a)
    M0 = conv_a.max()
    # Initial learning rate (large enough to move quickly) and final tiny value.
    eta_start = 0.5
    eta_end = 1e-5
    # Temperature for the soft‑max weight vector.
    tau_start = max(M0 * 5.0, 1e-5)   # start fairly “hot”
    tau_end = max(M0 * 1e-5, 1e-12)  # end “cold”

    # ------------------------------------------------------------------
    # Miscellaneous -----------------------------------------------------
    # ------------------------------------------------------------------
    jitter_prob = 0.02          # probability of a random mass‑transfer move
    restart_after = 150_000     # iterations without improvement → restart
    no_improve_cnt = 0

    best_a = a.copy()
    best_val = float("inf")
    iteration = 0

    print(f"[init] n={n}, initial M0={M0:.6e}, best bound (≈2·n·M) = {2*n*M0:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- current convolution & objective --------------------------------
        conv_a = _full_convolution_fft(a, a)          # length 2n‑1
        M = conv_a.max()
        cur_val = float(2.0 * n * M)

        # ---- record best ----------------------------------------------------
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve_cnt = 0
            print(f"[{iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve_cnt += 1

        # ---- schedule -------------------------------------------------------
        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress
        tau = tau_start * (1.0 - progress) + tau_end * progress
        tau = max(tau, 1e-14)        # avoid division by zero

        # ---- soft‑max weights ------------------------------------------------
        # weight = exp((conv - M) / τ)  (all ≤ 0 in exponent)
        exp_args = (conv_a - M) / tau
        exp_vals = np.exp(exp_args)
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            # fallback to hard sub‑gradient (single active index)
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0
        else:
            w = exp_vals / sum_exp

        # ---- gradient (softened sub‑gradient) -------------------------------
        grad = _gradient_from_weights(a, w)

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = m_hat / (np.sqrt(v_hat) + eps)

        # optional clipping to keep updates well‑behaved
        step = np.clip(step, -10.0, 10.0)

        a_candidate = a - eta * step

        # ---- occasional random “jitter’’ (mass transfer) --------------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # Transfer a random fraction (up to 50%) of the source mass.
            delta = np.random.rand() * a_candidate[src] * 0.5
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        # ---- enforce symmetry ------------------------------------------------
        a_candidate = (a_candidate + a_candidate[::-1]) * 0.5

        # ---- projection onto the simplex ------------------------------------
        a = _project_onto_simplex(a_candidate)

        iteration += 1

        # ---- simple restart if stuck ----------------------------------------
        if no_improve_cnt >= restart_after:
            print(f"[{iteration:7d}] restart (no improvement for {restart_after} iters)")
            rng = np.random.default_rng(seed + iteration)
            rnd = rng.random(n)
            a = (rnd + rnd[::-1]) * 0.5
            a = _project_onto_simplex(a)
            # reset Adam statistics
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            no_improve_cnt = 0

    # ------------------------------------------------------------------
    # Final reporting ----------------------------------------------------
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
