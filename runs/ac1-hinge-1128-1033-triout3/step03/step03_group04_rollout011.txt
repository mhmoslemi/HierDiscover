"""
Propose a low‑score step‑function using a smooth‑max gradient descent (Adam)
with a simplex projection.  The algorithm works in two phases:

1. Deterministic sweep of several parametric families (U‑shape, beta, exponential)
   to collect a few promising seeds (the best 10 across all tested lengths).

2. For each seed a refined Adam‑based optimisation is performed.
   The objective is a smooth approximation of the infinity‑norm
   L( a ) = (1/β)·log ∑_k exp(β·(a*a)[k]).
   The gradient of L can be written as a single correlation of the softmax
   weights with the current vector, which we evaluate with an FFT.  The
   smoothing parameter β is slowly increased, so the optimisation first
   follows a smooth landscape and finally concentrates on the true maximum.

The method is fast (O(n log n) per iteration) and works well for vectors of
size 5 000‑12 000 within the 1000 s budget.  The best sequence found is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution / correlation utilities
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length 2*len(a)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]                 # length 2n‑1


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is non‑negative, sum=1)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a vector already normalized."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def correlation_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear correlation of x and y (both real vectors).
    Returns an array of length len(x)+len(y)-1 where index i corresponds
    to lag (i - (len(y)-1)).
    """
    L = len(x) + len(y) - 1
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    corr = np.fft.irfft(X * np.conj(Y), n=L)
    return corr


def gradient_from_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Given a weight vector w of length 2·len(a)‑1 (e.g. softmax or binary mask),
    compute the sub‑gradient of  ∑_k w_k·(a*a)[k]  with respect to a.
    The result is 2·correlation(w, a) evaluated at the appropriate lags.
    """
    n = a.shape[0]
    corr = correlation_fft(w, a)                # length 3n‑2
    # gradient[i] = 2 * Σ_k w_k * a_{k-i}  -> lag = -i
    grad = 2.0 * corr[(n - 1)::-1][:n]
    return grad


# ----------------------------------------------------------------------
# Parametric families (used for the deterministic seed sweep)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge: (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the two edges."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def harmonic_left_distribution(n: int, p: float) -> np.ndarray:
    """One‑sided power law: a[i] ∝ (i+1)^{-p}."""
    i = np.arange(n) + 1
    a = i.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Refinement using smooth‑max Adam optimiser
# ----------------------------------------------------------------------
def refine_with_adam(
    a0: np.ndarray,
    deadline: float,
    *,
    init_lr: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    beta_start: float = 5.0,
    beta_inc: float = 0.004,
    beta_max: float = 200.0,
    eval_every: int = 100
) -> tuple[np.ndarray, float]:
    """
    Adam‑based optimisation of the smooth‑max loss.
    Returns the best vector discovered and its true score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    lr = init_lr
    beta = beta_start

    best_seq = a.copy()
    best_score = score_from_vector(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # 1) self‑convolution
        conv = full_convolution_fft(a)
        max_c = conv.max()

        # 2) smooth log‑sum‑exp (soft‑max) weights
        shifted = conv - max_c                # avoid overflow
        exps = np.exp(beta * shifted)
        sum_exps = exps.sum()
        w = exps / sum_exps                   # w sums to 1, length 2n‑1

        # 3) gradient of the smooth loss
        grad = gradient_from_weights(w, a)    # shape (n,)

        # 4) Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the probability simplex (maintains sum=1)
        a = simplex_projection(a)

        # 5) schedule updates
        lr *= 0.9998                # gentle decay
        beta = min(beta_max, beta + beta_inc)

        # 6) occasional evaluation of the true score
        if it % eval_every == 0:
            cur_score = 2.0 * n * max_c
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"   [Iter {it:7d}] new best (true score) = {best_score:.6f}")

        # 7) small random jitter to escape plateaus (every few thousand iters)
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-4
            a = simplex_projection(a + jitter)

    return best_seq, best_score
