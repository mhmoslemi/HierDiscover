"""
search for a step‑function (non‑negative vector) that minimises
evaluate_sequence().  The routine uses a projected Adam optimiser
with a smoothly‑approximated L∞‑norm of the self‑convolution (a
log‑sum‑exp).  It runs a multiscale schedule: start with a modest
length, optimise, then up‑sample the current best vector and keep
optimising at larger lengths.  Simple adaptive step‑size and a
periodic jitter avoid premature stagnation.  The best vector found
within the 1000 s budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full convolution a*a (length 2·n‑1) using a pre‑computed FFT size.
    """
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Normalised soft‑max weights for a vector `conv`:
    w_k = exp(alpha·conv_k) / Σ exp(alpha·conv).
    The exponent is stabilised by subtraction of the max entry.
    """
    mx = conv.max()
    # avoid overflow/underflow
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution (∂Sα/∂a).
    Returned vector has the same length as a.
    """
    n = a.shape[0]
    L = 2 * n - 1  # length of w
    # FFT of a
    A = np.fft.rfft(a, n=size)

    # FFT of weights and its reversal
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # a * w (full length n+L‑1 = 3·n‑2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    # gradient = first n entries of conv_aw  + reversed slice of conv_aw_rev
    idx = L - 1 - np.arange(n)               # 2·n‑2, …, n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x≥0, Σx=1}.
    Implementation follows Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a normalized distribution `old` to a new length.
    The result is re‑normalised to sum to 1.
    """
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new += 1e-14                # avoid exact zeros
    new /= new.sum()
    return new


def u_shaped_distribution(n: int, p: float = 0.8) -> np.ndarray:
    """
    Symmetric U‑shape: a_i ∝ (min(i, n‑1‑i)+1)^{-p}.
    Normalised to sum 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)               # distance to nearest edge

"""
search for a step‑function (non‑negative vector) that minimises
evaluate_sequence().  The routine uses a projected Adam optimiser
with a smoothly‑approximated L∞‑norm of the self‑convolution (a
log‑sum‑exp).  It runs a multiscale schedule: start with a modest
length, optimise, then up‑sample the current best vector and keep
optimising at larger lengths.  Simple adaptive step‑size and a
periodic jitter avoid premature stagnation.  The best vector found
within the 1000 s budget is returned.
"""

import time

# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using a pre‑computed FFT size."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Normalised soft‑max weights for a vector `conv`:
    w_k = exp(alpha·conv_k) / Σ exp(alpha·conv).
    The exponent is stabilised by subtraction of the max entry.
    """
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution (∂Sα/∂a).
    Returned vector has the same length as a.
    """
    n = a.shape[0]
    L = 2 * n - 1                # length of w
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    idx = L - 1 - np.arange(n)   # 2·n‑2 … n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x≥0, Σx = 1}.
    Implementation follows Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a normalised distribution `old` to a new length.
    The result is re‑normalised to sum to 1.
    """
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new += 1e-14                # avoid exact zeros
    new /= new.sum()
    return new

    Symmetric U‑shape: a_i ∝ (min(i, n‑1‑i)+1)^{-p}.
