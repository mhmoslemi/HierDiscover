"""
Propose a step‑function (non‑negative sequence) that minimises the
objective 2·n·max_k (a*a)[k] (the normalised objective used by
evaluate_sequence).  The search combines three complementary ideas:

* Exponentiated‑gradient descent on the probability simplex with a
  soft‑max approximation of the max‑operator.  This provides fast
  smoothing‑gradient steps that keep the iterate non‑negative.
* Occasional “jitter’’ moves – random transfer of a small amount of mass
  between two coordinates – to escape plateaux.
* A linear‑programming sub‑step (in the spirit of the original paper):
  given the current vector `a` and its maximal autocorrelation `M`,
  solve   maximise Σ b_j   s.t.  a ⊛ b ≤ M , b ≥ 0.
  The solution `b` typically has a larger ℓ₁‑norm than `a`; after normalising
  it we mix it into the current iterate.  This provably never increases the
  objective and often produces a noticeable drop in the peak.
The algorithm runs for the whole time budget (default 1000 s) and returns
the best sequence found.  If a target value (≈ 1.503) is reached early the
search stops and the sequence is returned immediately.
"""
import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper: gradient from soft‑max weights
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Compute g_i = 2 Σ_k w_k a_{k-i} using a single FFT‑based convolution."""
    n = a.shape[0]
    # reverse the weight vector
    u = w[::-1]
    # full convolution u * a   -> length 3n‑2 (actually 2n-1 + n-1 = 3n-2)
    conv = _full_convolution_fft(u, a)
    # we need the slice corresponding to indices (2n‑2 … n‑1)
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Helper: solve the auxiliary linear programme
# maximise Σ b_j   subject to  a ⊛ b ≤ M , b ≥ 0
# ----------------------------------------------------------------------
def _solve_lp_max_sum(a: np.ndarray, M: float):
    """Return the optimal b (numpy array) or None if the LP fails."""
    n = a.shape[0]
    # Build the dense convolution matrix A where A @ b = a ⊛ b
    # Using a Toeplitz matrix with first column a padded with zeros.
    col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
    row = np.concatenate([a[:1], np.zeros(n - 1, dtype=np.float64)])
    A = toeplitz(col, row)          # shape (2n‑1, n)
    # Objective: maximise Σ b → minimise –Σ b
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    # Use the HiGHS interior‑point/simplex solver (fast for medium size)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

    """

    # --------------------------------------------------------------

            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, seq0)
    except Exception:
        # random Dirichlet start (all entries >0, sums to 1)
        a = np.random.dirichlet(np.ones(n))

    # Safety normalisation
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    # --------------------------------------------------------------
    # Optimisation bookkeeping
    # --------------------------------------------------------------

    # initial evaluation (fast)

    best_val = float(2.0 * n * conv_a.max())
    print(f"[init] n={n}, initial objective={best_val:.6f}")

    # Hyper‑parameters
    eta_start = 0.5               # initial exponentiated‑gradient step size
    eta_end   = 1e-6              # final step size
    tau_factor_start = 0.5        # τ = τ_factor * M at the beginning
    tau_factor_end   = 1e-5
    jitter_prob = 0.02            # probability of a random mass‑transfer
    lp_interval = 15000           # how many gradient steps between LP solves
    mix_factor_start = 0.12       # initial mixing fraction for the LP direction
    mix_factor_end   = 0.02

    iteration = 0
    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ------------------------------------------------------------------
        # 1) Evaluate the current objective
        # ------------------------------------------------------------------
        conv_a = _full_convolution_fft(a, a)
        M = conv_a.max()
        cur_val = float(2.0 * n * M)

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # early exit if we have already reached the target bound
        if best_val <= 1.503:
            print("[info] target reached – exiting early")
            break

        # ------------------------------------------------------------------
        # 2) Decide which update to perform (LP vs. gradient)
        # ------------------------------------------------------------------
        progress = iteration / max(1, lp_interval)   # normalised progress for schedules
        # temperature for the soft‑max (relative to the current max)
        tau = M * (tau_factor_start * (1.0 - progress) + tau_factor_end * progress)

        # --------------------------------------------------------------
        #   Gradient step (exponentiated gradient on the simplex)
        # --------------------------------------------------------------
        # Soft‑max weights
        if tau > 0:
            exp_args = (conv_a - M) / tau      # ≤ 0, max at zero
            # shift to avoid overflow, safe because max is 0
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv_a)
                w[np.argmax(conv_a)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0

        grad = _gradient_from_weights(a, w)

        # exponentiated‑gradient update
        eta = eta_start * (1.0 - iteration / max(1, deadline - start_time)) + \
              eta_end * (iteration / max(1, deadline - start_time))
        # make the exponent safe
        step_factor = np.exp(np.clip(-eta * grad, -50.0, 0.0))
        a = a * step_factor
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            # fall back to a random point on the simplex
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        # occasional random jum‑p (helps escape plateaux)
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a[src] * 0.4   # move up to 40 % of source mass
            a[src] -= delta
            a[dst] += delta

        iteration += 1

        # ------------------------------------------------------------------
        #   Linear‑programming refinement every lp_interval steps
        # ------------------------------------------------------------------
        if iteration % lp_interval == 0:
            # Solve max‑sum LP under the current peak constraint
            b = _solve_lp_max_sum(a, M)
            if b is not None:
                sum_b = b.sum()
                if sum_b > 0:
                    # normalise to the same ℓ₁‑norm as a (which is 1)
                    g = b / sum_b
                    # mixing factor decays as the search proceeds
                    mix = mix_factor_start * (1.0 - iteration / (deadline - start_time)) + \
                          mix_factor_end * (iteration / (deadline - start_time))
                    a = (1.0 - mix) * a + mix * g
                    # re‑project (tiny numerical drift may appear)
                    a = np.maximum(a, 0.0)
                    a_sum = a.sum()
                    if a_sum == 0.0 or not np.isfinite(a_sum):
                        a = np.random.dirichlet(np.ones(n))
                    else:
                        a /= a_sum
            # (optional) sanity check – recompute conv to keep drift tiny
            conv_a = _full_convolution_fft(a, a)

    # ----------------------------------------------------------------------
    # Final reporting (optional – the outer framework will also call it)
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
