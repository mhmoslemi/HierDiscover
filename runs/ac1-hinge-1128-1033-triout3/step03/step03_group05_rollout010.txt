def max_conv_subgrad(p, max_idx):
    n = len(p)
    g = np.zeros_like(p)
    if max_idx < n:
        g[:max_idx+1] = 2.0 * p[max_idx::-1]   # reversed prefix
    else:
        start = max_idx - n + 1
        # tail reversed: p[n-1:start-1:-1] is reversed segment from start to n-1
        g[start:] = 2.0 * p[n-1:start-1:-1]
    return g
```

"""
Propose a step‑function (non‑negative vector) that minimises
    evaluate_sequence(p) = 2·n·max(conv(p,p)) / (sum(p)^2)

The algorithm combines several ideas:

*   A rich pool of symmetric initial candidates (arcsine, Beta‑family,
    triangular, random Dirichlet draws and the historic construction).
*   A long sub‑gradient phase that directly reduces the maximal
    convolution entry while keeping the vector on the probability simplex.
*   An Adam optimiser on a smooth log‑sum‑exp approximation of the
    max‑convolution.  The smoothing factor is increased slowly (up to 200)
    while the Adam learning‑rate is kept stable.
*   A final stochastic “swap‑hill‑climber’’ that moves tiny amounts of
    probability between two positions if it improves the true objective.
    This phase also performs a tiny simulated‑annealing walk to escape
    shallow local minima.
*   All steps respect the 1000 s time budget and always return a valid
    sequence (non‑negative, sum > 0.01, entries clipped to [0,1000]).

The code is written with only NumPy and the standard library, uses FFT‑based
convolutions for speed and projects every intermediate vector onto the
probability simplex.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Simplex projection  (Euclidean projection onto {x >=0, sum x = 1})
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Return the full (linear) self‑convolution of p using FFT of size m_fft."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _score(p: np.ndarray, m_fft: int) -> float:
    """Exact objective value 2·n·max(conv(p,p)) for a probability vector."""
    n = p.size
    conv = _conv_self(p, m_fft)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _max_conv_subgrad(p: np.ndarray, max_idx: int) -> np.ndarray:
    """Sub‑gradient of the (unique) maximal entry of conv(p,p)."""
    n = p.size
    g = np.zeros_like(p)
    if max_idx < n:
        # overlapping segment is [0 .. max_idx]
        g[: max_idx + 1] = 2.0 * p[max_idx :: -1]
    else:
        start = max_idx - n + 1
        # overlapping segment is [start .. n‑1]
        g[start:] = 2.0 * p[n - 1 : start - 1 : -1]
    return g


# ----------------------------------------------------------------------
# Generate families of symmetric initial probability vectors
# ----------------------------------------------------------------------
def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like density on {0,…,n‑1} : (i+½)^{a‑1} (n‑½‑i)^{b‑1}."""
    i = np.arange(n, dtype=float)
    # shift to avoid zero at the borders
    x = (i + 0.5) / n
    # use safe pow – values stay finite for a,b>0
    v = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v[:] = 1.0 / n
    else:
        v /= s
    return v


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular shape, peak in the centre."""
    i = np.arange(n, dtype=float)
    v = np.minimum(i + 1.0, n - i)
    v /= v.sum()
    return v


def _arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine distribution (Beta(½,½))."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    inv = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    v = inv / inv.sum()
    return v
