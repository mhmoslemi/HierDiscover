def proj_simplex(v):
    # v: np.ndarray
    # algorithm from https:...
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - 1
    ind = np.arange(len(v)) + 1
    cond = v_sorted - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step function minimizing the evaluation metric.

Algorithm
---------
1. Warm‑start from a global ``height_sequence_1`` if it exists.
2. Deterministic sweep over several parametric families (U‑shaped,
   tails, exponential, product‑power‑law and a combined power‑law/
   exponential family).  The best ~5 candidates (according to the
   cheap Fourier‑based bound 2·n·max(conv)) are kept.
3. For each of those candidates a smooth‑approximation gradient descent
   is performed:
   * The current self‑convolution is computed.
   * A soft‑max with temperature τ turns the max‑norm into a smooth
     differentiable surrogate.
   * The gradient of the surrogate w.r.t. the coefficients is obtained
     analytically and evaluated with two FFT‑based convolutions.
   * A projected gradient step (projection onto the probability simplex)
     is taken.  τ and the step size are cooled exponentially.
   The best vector seen during the optimisation is stored.
4. After the time budget the best vector is evaluated with the official
   ``evaluate_sequence`` routine and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    # Works for arbitrary dimensionality, O(n log n) due to sorting.
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero; return uniform simplex.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all return a normalized vector with sum == 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges: a[i] ∝ exp(-α·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """
    Product‑power‑law shape:
        a[i] ∝ ((i+1)*(n-i))^{‑p}
    (very small values in the middle, larger at the ends).
    """
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-12
    a /= a.sum()
    return a


def combined_power_exp_distribution(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Product of a power‑law and an exponential decay, symmetrised.
    a[i] ∝ (i+1)^{‑p}·exp(-α·i) + (n‑i)^{‑p}·exp(-α·(n‑1‑i)).
    """
    i = np.arange(n)
    left = ((i + 1) ** (-p)) * np.exp(-alpha * i)
    right = ((n - i) ** (-p)) * np.exp(-alpha * (n - 1 - i))
    a = left + right + 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate of max(conv)
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Return a probability vector w_i ∝ exp(conv_i / τ)."""
    if tau <= 0.0:
        tau = 1e-12
    # subtract max for numerical stability
    shifted = conv - conv.max()
    exps = np.exp(shifted / tau)
    w = exps / exps.sum()
    return w


def gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a ( Σ_i w_i (a*a)_i ) = Σ_i w_i ∂(a*a)_i / ∂a .
    The formula reduces to two convolutions that are evaluated via FFT.
    """
    n = a.shape[0]
    L = len(w)                 # L = 2*n-1
    # size for linear convolution (next power of two for speed)
    size = 1 << ((n + L - 1).bit_length())

    # FFT of a and w
    A = np.fft.fft(a, n=size)
    W = np.fft.fft(w, n=size)

    # First term: g2_j = Σ_i w_i a_{j-i}
    conv_w_a = np.fft.ifft(W * A).real
    g2 = conv_w_a[:n]

    # Second term: g1_j = Σ_i w_i a_{i-j}
    # This is a cross‑correlation; we reuse the same FFTs with reversed w.
    w_rev = w[::-1]
    W_rev = np.fft.fft(w_rev, n=size)
    conv_a_wrev = np.fft.ifft(A * W_rev).real
    # index = (L‑1) – j  (L = 2*n‑1)
    idx = (L - 1) - np.arange(n)
    g1 = conv_a_wrev[idx]

    grad = g1 + g2
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    print("=== Parametric sweep ===")
    for n in lengths:

            if time.time() > deadline:
                break
            a = u_shaped_distribution(n, p)
            approx = 2.0 * n * max_conv_fft(a)
            maybe_push(a, approx)

        # sum‑of‑tails
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sum_tails_distribution(n, p)
            approx = 2.0 * n * max_conv_fft(a)
            maybe_push(a, approx)

        # exponential

            if time.time() > deadline:
                break
            a = exponential_distribution(n, a_alpha)
            approx = 2.0 * n * max_conv_fft(a)
            maybe_push(a, approx)

        # product power‑law
        for p in p_vals:
            if time.time() > deadline:
                break
            a = product_powerlaw_distribution(n, p)
            approx = 2.0 * n * max_conv_fft(a)
            maybe_push(a, approx)

        # combined power‑law + exponential (more expressive)
        for p in p_vals[::3]:               # sample a subset to keep time reasonable
            for a_alpha in alphas[::2]:
                if time.time() > deadline:
                    break
                a = combined_power_exp_distribution(n, p, a_alpha)
                approx = 2.0 * n * max_conv_fft(a)
                maybe_push(a, approx)

    # If we somehow ended up without any candidate, fall back to a random vector.
    if not top_candidates:
        n = 8000
        a = np.random.rand(n)
        a /= a.sum()
        approx = 2.0 * n * max_conv_fft(a)
        maybe_push(a, approx)

    # ------------------------------------------------------------------
    # 2. Gradient‑descent refinement on the simplex.
    # ------------------------------------------------------------------
    print("\n=== Gradient‑descent refinement ===")
    for idx, (init_score, a0) in enumerate(top_candidates):
        if time.time() > deadline:
            break
        print(f"Refining candidate {idx + 1}/{len(top_candidates)} (init approx={init_score:.6f})")
        a = a0.copy()
        n = a.shape[0]

        # initialise temperature τ as a fraction of the current maximal convolution entry
        conv = full_conv_fft(a)
        cur_max = conv.max()
        tau = max(0.5 * cur_max, 1e-3)

        # base step size (will be scaled by τ)
        eta_base = 0.25

        # simple cooling schedules
        tau_decay = 0.997
        eta_decay = 0.999

        # iteration counter for this candidate

        while time.time() < deadline:
            it += 1

            # recompute convolution and current score
            conv = full_conv_fft(a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep global best if improved
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"Iter {it:5d} (cand {idx + 1}): new global best = {best_score:.6f}")

                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    break

            # construct soft‑max weights
            w = softmax_weights(conv, tau)

            # gradient of the smooth surrogate
            grad = gradient_softmax(a, w)

            # step size adaption – we shrink it together with τ
            eta = eta_base * max(tau, 1e-8)

            # projected gradient step

            # evaluate candidate
            conv_new = full_conv_fft(a_new)
            new_max = conv_new.max()
            new_score = 2.0 * n * new_max

            # accept if we made progress; otherwise keep the old point

                a = a_new
            else:

                if np.random.rand() < 0.001:

            # cool the temperature and step size

            eta_base *= eta_decay

            # early stop if we performed many iterations without improvement
            if it % 200 == 0 and time.time() > deadline - 30.0:
                # leave a few seconds for final exact evaluation
                break

    # ------------------------------------------------------------------
    # 3. Final exact evaluation (once) and report
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())  # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()

"""
Propose a candidate step function minimizing the evaluation metric.

Algorithm
---------
1. Warm‑start from a global ``height_sequence_1`` if it exists.
2. Deterministic sweep over several parametric families (U‑shaped,
   tails, exponential, product‑power‑law and a combined power‑law/
   exponential family).  The best ~5 candidates (according to the
   cheap Fourier‑based bound 2·n·max(conv)) are kept.
3. For each of those candidates a smooth‑approximation gradient descent
   is performed:
   * The current self‑convolution is computed.
   * A soft‑max with temperature τ turns the max‑norm into a smooth
     differentiable surrogate.
   * The gradient of the surrogate w.r.t. the coefficients is obtained
     analytically and evaluated with two FFT‑based convolutions.
   * A projected gradient step (projection onto the probability simplex)
     is taken.  τ and the step size are cooled exponentially.
   The best vector seen during the optimisation is stored.
4. After the time budget the best vector is evaluated with the official
   ``evaluate_sequence`` routine and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all return a normalized vector with sum == 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges: a[i] ∝ exp(-α·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """
    Product‑power‑law shape:
        a[i] ∝ ((i+1)*(n-i))^{‑p}
    (very small values in the middle, larger at the ends).
    """
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-12
    a /= a.sum()
    return a


def combined_power_exp_distribution(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Product of a power‑law and an exponential decay, symmetrised.
    a[i] ∝ (i+1)^{‑p}·exp(-α·i) + (n‑i)^{‑p}·exp(-α·(n‑1‑i)).
    """
    i = np.arange(n)
    left = ((i + 1) ** (-p)) * np.exp(-alpha * i)
    right = ((n - i) ** (-p)) * np.exp(-alpha * (n - 1 - i))
    a = left + right + 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Return a probability vector w_i ∝ exp(conv_i / τ)."""
    if tau <= 0.0:
        tau = 1e-12
    shifted = conv - conv.max()          # for numerical stability
    exps = np.exp(shifted / tau)
    w = exps / exps.sum()
    return w


def gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_i w_i (a*a)_i.
    The result is g = 2·(a * w) + 2·corr(w, a) (both evaluated via FFT).
    """
    n = a.shape[0]
    L = w.shape[0]               # = 2*n-1
    size = 1 << ((n + L - 1).bit_length())   # next power of two for the linear convolution

    A = np.fft.fft(a, n=size)
    W = np.fft.fft(w, n=size)

    # term g2_j = Σ_i w_i a_{j-i}
    conv_w_a = np.fft.ifft(W * A).real
    g2 = conv_w_a[:n]

    # term g1_j = Σ_i w_i a_{i-j}
    w_rev = w[::-1]
    W_rev = np.fft.fft(w_rev, n=size)
    conv_a_wrev = np.fft.ifft(A * W_rev).real
    idx = (L - 1) - np.arange(n)          # pick positions L‑1‑j
    g1 = conv_a_wrev[idx]

    grad = g1 + g2
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimizing the evaluation metric.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0. Warm‑start from an externally supplied sequence if available.
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)  # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external /= s
            approx = 2.0 * external.size * max_conv_fft(external)
            if approx < best_score:
                best_score = approx
                best_seq = external.copy()
                print(f"Seeded from height_sequence_1: approx_score={approx:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Deterministic parametric sweep – keep the few best candidates.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000, 12000, 16000,
               20000, 24000, 28000, 32000]
    p_vals = np.linspace(0.2, 3.0, 27)          # 0.2 … 3.0
    alphas = np.linspace(0.0001, 0.02, 20)      # exponential decay rates

    K = 5                         # keep top‑K candidates from the sweep
    top_candidates = []           # list of (approx_score, ndarray)

    def maybe_push(seq: np.ndarray, approx: float):
        nonlocal best_seq, best_score, top_candidates
        if approx < best_score:
            best_score = approx
            best_seq = seq.copy()
            print(f"New global best (approx) = {approx:.6f}")
        if len(top_candidates) < K:
            top_candidates.append((approx, seq.copy()))
            top_candidates.sort(key=lambda x: x[0])
        else:
            if approx < top_candidates[-1][0]:
                top_candidates[-1] = (approx, seq.copy())
                top_candidates.sort(key=lambda x: x[0])

    print("=== Parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # U‑shaped power‑law
        for p in p_vals:
            if time.time() > deadline:
                break
            a = u_shaped_distribution(n, p)
            approx = 2.0 * n * max_conv_fft(a)
            maybe_push(a, approx)

        # sum‑of‑tails
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sum_tails_distribution(n, p)
            approx = 2.0 * n * max_conv_fft(a)
            maybe_push(a, approx)

        # exponential
        for a_alpha in alphas:
            if time.time() > deadline:
                break
            a = exponential_distribution(n, a_alpha)
            approx = 2.0 * n * max_conv_fft(a)
            maybe_push(a, approx)

        # product power‑law
        for p in p_vals:
            if time.time() > deadline:
                break
            a = product_powerlaw_distribution(n, p)
            approx = 2.0 * n * max_conv_fft(a)
            maybe_push(a, approx)

        # combined power‑law + exponential (sampled sparsely to keep budget)
        for p in p_vals[::3]:
            for a_alpha in alphas[::2]:
                if time.time() > deadline:
                    break
                a = combined_power_exp_distribution(n, p, a_alpha)
                approx = 2.0 * n * max_conv_fft(a)
                maybe_push(a, approx)

    if not top_candidates:
        # fallback: random simplex vector
        n = 8000
        a = np.random.rand(n)
        a /= a.sum()
        approx = 2.0 * n * max_conv_fft(a)
        maybe_push(a, approx)

    # ------------------------------------------------------------------
    # 2. Gradient‑descent refinement on the simplex.
    # ------------------------------------------------------------------
    print("\n=== Gradient‑descent refinement ===")
    for cand_idx, (init_approx, a0) in enumerate(top_candidates):
        if time.time() > deadline:
            break
        print(f"Refining candidate {cand_idx + 1}/{len(top_candidates)} "
              f"(init approx={init_approx:.6f})")
        a = a0.copy()
        n = a.shape[0]

        # initialise temperature τ
        conv = full_conv_fft(a)
        cur_max = conv.max()
        tau = max(0.5 * cur_max, 1e-3)

        # base step size (scaled by τ each iteration)
        eta_base = 0.25
        tau_decay = 0.997
        eta_decay = 0.999

        it = 0
        while time.time() < deadline:
            it += 1

            # current convolution & score
            conv = full_conv_fft(a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # update global best if improved
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"Iter {it:5d} (cand {cand_idx + 1}): new global best = {best_score:.6f}")
                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    break

            # soft‑max weights and gradient
            w = softmax_weights(conv, tau)
            grad = gradient_softmax(a, w)

            # projected gradient step
            eta = eta_base * max(tau, 1e-9)
            a_new = proj_simplex(a - eta * grad)

            # evaluate the candidate
            conv_new = full_conv_fft(a_new)
            new_max = conv_new.max()
            new_score = 2.0 * n * new_max

            # accept if we made progress; occasional uphill moves help escape
            if new_score < cur_score or np.random.rand() < 0.001:
                a = a_new

            # cooling schedules
            tau = max(tau * tau_decay, 1e-6)
            eta_base *= eta_decay

            # periodic time‑check
            if it % 200 == 0 and time.time() > deadline - 30.0:
                break

    # ------------------------------------------------------------------
    # 3. Final exact evaluation (once) and report
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())  # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()
