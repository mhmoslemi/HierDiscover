    for it in range(max_iter):

"""
Propose a non‑negative step‑function (probability vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

where conv denotes the linear self‑convolution.
The algorithm is a hybrid of:

1.  A linear‑programming “mixing” step (inspired by the paper’s
    iterative LP).  For a current distribution `a` we maximise the
    total mass of a second distribution `b` while keeping the
    convolution `a*b` bounded by the current worst autocorrelation.
    The optimiser (scipy.optimize.linprog) is cheap for n≈2000.

2.  A few rounds of multiplicative mirror‑descent on a smooth
    approximation of the max‑convolution (soft‑max).  This fine‑tunes
    the shape found by the LP.

The routine works for a range of lengths (≈1200–2500) and respects the
time budget (default 1000 s).  It can be seeded with the global
variable `height_sequence_1` if present; otherwise it falls back to a
U‑shaped distribution and a random Dirichlet start.
"""

import time
import numpy as np

# optional SciPy – required for the LP step
try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:          # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic convolution helpers
# ----------------------------------------------------------------------
def max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return float(np.convolve(a, a).max())


def full_conv(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1)."""
    return np.convolve(a, a)


# ----------------------------------------------------------------------
# Soft‑max upper‑bound utilities (used in the final polishing stage)
# ----------------------------------------------------------------------
def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities w_k ∝ exp(alpha·conv_k)."""
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of a*a w.r.t. a.
    Implements the formula from the original implementation.
    """
    n = a.shape[0]
    L = 2 * n - 1                          # length of w
    A = np.fft.rfft(a, n=size)

    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    idx = L - 1 - np.arange(n)            # 2·n‑2 … n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    if v.ndim != 1:                     # pragma: no cover
        raise ValueError("Input must be a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple constructions for initial seeds
# ----------------------------------------------------------------------
def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from an old probability vector to length new_n."""
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    # a tiny floor avoids exact zeros
    new += 1e-14
    new /= new.sum()
    return new


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape (p<1 gives higher mass near the edges)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)               # distance to nearest border
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming mixing step (core of the optimisation)
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2·n‑1) × n) such that A @ b = conv(f,b).
    Row k contains the coefficients f_i for i where 0 ≤ k‑i < n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i, val in enumerate(f):
        A[i:i + n, i] = val
    return A


def lp_mix_step(a: np.ndarray, M: float, t: float) -> np.ndarray:
    """
    One LP‑based mixing step:
        maximise Σ b_i  subject to  conv(a,b) ≤ M  and  b_i ≥ 0.
    The returned vector g has Σ g_i = 1 and is mixed with `a` using
    the mixing factor `t` (i.e. new_a = (1‑t)·a + t·g).
    """
    n = a.shape[0]

    # Build the (2n‑1)×n constraint matrix
    A = build_conv_matrix(a)

    # Linear programme: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)               # minimise -∑b_i
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Solve with HiGHS (fast interior‑point algorithm)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if not res.success:                            # pragma: no cover
        raise RuntimeError(f"LP failed: {res.message}")

    b0 = res.x
    s0 = b0.sum()
    if s0 <= 0.0:
        # degenerate – fall back to the current distribution
        return a.copy()

    # Scale to obtain a proper probability vector
    g = b0 / s0
    # convex combination
    new_a = (1.0 - t) * a + t * g
    # Re‑project to the simplex to cure tiny numerical drift
    new_a = project_to_simplex(new_a)
    return new_a


def lp_based_optimisation(a0: np.ndarray,
                          max_iters: int,
                          deadline: float,
                          t_start: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Repeated LP mixing (Algorithm 1 of the paper) with a slowly decaying
    mixing rate.  Returns the best vector seen and its approximate score
    (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * max_conv(a)

    t = t_start
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = max_conv(a)                     # current worst autocorrelation
        a_new = lp_mix_step(a, M, t)

        cur_score = 2.0 * n * max_conv(a_new)

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"   LP‑iter {it+1:3d}: improved score {best_score:.6f}")

        # simple schedule for the mixing factor
        if cur_score < 2.0 * n * M:         # improvement
            t = min(t * 1.10, 0.9)
        else:
            t = max(t * 0.70, 0.02)

        a = a_new

        # stop early if we are already very close to the theoretical lower bound
        if best_score < 1.30:                # very aggressive early exit
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Soft‑max mirror‑descent refinement (optional polishing)
# ----------------------------------------------------------------------
def softmax_refine(a0: np.ndarray,
                  deadline: float,
                  max_iters: int = 3000) -> np.ndarray:
    """
    A few dozen mirror‑descent steps on the smooth max‑convolution.
    Works well as a final polish after the LP phase.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT size sufficient for all convolutions that appear (3·n‑2)
    fft_size = 1 << ((3 * n - 2).bit_length())

    step_eta = 0.3
    eta_min, eta_max = 5e-5, 5.0
    alpha = 2.0
    alpha_max = 250.0
    alpha_mul = 1.005

    best_a = a.copy()
    best_score = 2.0 * n * max_conv(a)

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        conv = full_conv(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"   soft‑max iter {iteration:4d}: new best {best_score:.6f}")

        # soft‑max weights
        w = softmax_weights(conv, alpha)

        # gradient
        grad = gradient_softmax(a, w, fft_size)

        # multiplicative mirror‑descent step (with safeguards)
        exp_arg = -step_eta * grad
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a_candidate = a * np.exp(exp_arg)
        a_candidate = project_to_simplex(a_candidate)

        cand_score = 2.0 * n * max_conv(a_candidate)

        if cand_score < cur_score:
            a = a_candidate
            step_eta = min(step_eta * 1.02, eta_max)
        else:
            step_eta = max(step_eta * 0.90, eta_min)

        alpha = min(alpha * alpha_mul, alpha_max)

        # occasional diagnostic
        if iteration % 500 == 0:
            print(f"   soft‑max iter {iteration:4d}: score≈{cur_score:.6f}, α={alpha:.1f}, η={step_eta:.4f}")

        # exit if already near the lower bound 1.0
        if best_score < 1.05:
            break

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) that minimises the
    evaluation metric used by `evaluate_sequence`.  The routine respects
    the allocated time budget, combines an LP‑based mixing phase with a
    short soft‑max refinement, and returns the best vector found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # 0.  Seed from the previous best construction if it exists
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    external = None
    try:
        external = np.array(height_sequence_1, dtype=float)      # noqa: F821
    except Exception:
        pass

    if external is not None and external.size:
        external = np.maximum(external, 0.0)
        if external.sum() < 1e-12:
            external = np.ones_like(external) / external.size
        external = external / external.sum()
        cur_score = 2.0 * external.size * max_conv(external)
        best_seq = external.copy()
        best_score = cur_score
        print(f"Seeded from height_sequence_1: n={external.size}, score≈{cur_score:.6f}")

    # ------------------------------------------------------------------
    # 1.  Lengths to explore (moderate sizes – LP stays cheap)
    # ------------------------------------------------------------------
    lengths = [1200, 1600, 2000, 2500]          # feel free to adjust
    lengths = [n for n in lengths if n > 0]

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop
    # ------------------------------------------------------------------
    global_best_seq = best_seq if best_seq is not None else None
    global_best_score = best_score

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # prepare a few different initial seeds for this length
        seeds = []

        if best_seq is not None:
            seeds.append(resample_distribution(best_seq, n))

        # classic symmetric U‑shape
        seeds.append(u_shaped_distribution(n, p=0.8))

        # random Dirichlet‑like start
        rand_vec = np.random.rand(n) + 1e-14
        rand_vec /= rand_vec.sum()
        seeds.append(rand_vec)

        for init_idx, a0 in enumerate(seeds, start=1):
            if time.time() >= deadline:
                break
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            print(f"  Seed {init_idx}/{len(seeds)} (n={n})")

            # ----------------------------------------------------------
            # 2a. LP‑based mixing phase
            # ----------------------------------------------------------
            if SCIPY_AVAILABLE:
                try:
                    a_lp, lp_score = lp_based_optimisation(
                        a0,
                        max_iters=30,
                        deadline=deadline,
                        t_start=0.4,
                    )
                except Exception as exc:                     # pragma: no cover
                    print(f"    LP failed ({exc}); falling back to soft‑max only")
                    a_lp = a0
                    lp_score = 2.0 * n * max_conv(a_lp)
            else:
                # if SciPy is unavailable we skip the LP phase
                a_lp = a0
                lp_score = 2.0 * n * max_conv(a_lp)

            # keep best across seeds
            if lp_score < global_best_score:
                global_best_score = lp_score
                global_best_seq = a_lp.copy()
                print(f"    New global best after LP: score≈{global_best_score:.6f}")

            # ----------------------------------------------------------
            # 2b. Short soft‑max polishing (few hundred iterations)
            # ----------------------------------------------------------
            if time.time() < deadline:
                a_refined = softmax_refine(
                    a_lp,
                    deadline=deadline,
                    max_iters=2000,
                )
                final_score = 2.0 * n * max_conv(a_refined)
                if final_score < global_best_score:
                    global_best_score = final_score
                    global_best_seq = a_refined.copy()
                    print(f"    New global best after polish: score≈{global_best_score:.6f}")

            if global_best_score < 1.503:                # reached target early
                print("Target achieved – terminating early.")
                break

        if global_best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3.  Final official evaluation (if the function is available)
    # ------------------------------------------------------------------
    final_val = None
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {final_val:.6f}")
    except Exception as exc:                                    # pragma: no cover
        print(f"Could not call evaluate_sequence ({exc})")
        final_val = None

    print("\n=== Search finished ===")
    print(f"Best approximate score : {global_best_score:.6f}")
    if final_val is not None:
        print(f"Best exact score       : {final_val:.6f}")
    print(f"Best length            : {global_best_seq.shape[0]}")
    return global_best_seq.tolist()
