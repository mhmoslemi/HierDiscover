"""
Improved optimization of step‑function heights.

The algorithm works on the probability simplex (Σ a = 1, a_i ≥ 0) and
minimises the quantity 2·n·max_k (a*a)[k] which is exactly the value
returned by ``evaluate_sequence`` after normalisation.
The variables are kept in the log‑space ``x`` (a = softmax(x)) so that
the simplex constraint is satisfied automatically.  A smoothed
sub‑gradient of the max‑convolution is obtained with a soft‑max weighting
over all convolution entries; the resulting gradient is fed to an Adam
optimizer with a simple time‑based learning‑rate schedule.
Occasional random “jitter’’ moves help to escape plateaux.
The routine runs for the whole time budget (default 1000 s) and returns
the best coefficient vector seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    np.maximum(conv, 0, out=conv)                # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Gradient from soft‑max weights (sub‑gradient of the smoothed max)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented via a single convolution: reverse w and convolve with a.
    """
    n = a.shape[0]
    u = w[::-1]                              # reverse weight vector
    conv = _full_convolution_fft(u, a)       # length = 3n‑2
    # indices that correspond to the original length n:
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises the step‑function coefficients with Adam on a softened
    max‑convolution objective.  Returns the best non‑negative sequence
    found before the time budget expires (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin so the framework does not kill us early
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a length (a few thousand works well within the time limit)
    # ------------------------------------------------------------------
    n = 1800                                    # modestly large, still fast
    # start from a uniform distribution on the simplex
    a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Log‑space representation (softmax guarantees non‑negativity + sum=1)
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.90, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps_adam = 1e-8
    base_lr = 0.30                               # initial learning rate

    # bookkeeping of the best solution seen so far
    conv = _full_convolution_fft(a, a)
    cur_val = float(2.0 * n * conv.max())
    best_val = cur_val
    best_a = a.copy()

    print(f"[init] n={n}, initial objective={best_val:.6f}")

    iteration = 0
    # ------------------------------------------------------------------
    # Main optimisation loop (runs until the time budget is exhausted)
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1

        # ----- full auto‑convolution of the current a -----
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # ----- schedule parameters (linear in elapsed time) -----
        progress = (now - start_time) / (deadline - start_time)

        # temperature for the soft‑max (relative to the current max)
        tau_start = M * 0.40
        tau_end   = M * 1e-6
        tau = tau_start * (1.0 - progress) + tau_end * progress

        # ----- compute soft‑max weights over the convolution -----
        if tau > 0.0:
            # shift by the maximum to avoid overflow in the exponentials
            exp_args = (conv - M) / tau
            max_arg = exp_args.max()
            exp_vals = np.exp(exp_args - max_arg)
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # ----- gradient w.r.t. a (smoothed sub‑gradient) -----
        grad_a = _gradient_from_weights(a, w)

        # ----- gradient w.r.t. log‑parameters x (softmax chain rule) -----
        dot = np.dot(grad_a, a)                    # <grad_a , a>
        grad_x = a * (grad_a - dot)                # ∇_x (softmax) = diag(a)*(grad_a - <grad_a,a>)

        # ----- Adam update for x -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate schedule (linear decay)
        lr = base_lr * (1.0 - progress) + 1e-5 * progress
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- obtain a from the updated logits (softmax) -----
        max_x = x.max()
        exp_x = np.exp(x - max_x)            # shift for numerical stability
        a = exp_x / exp_x.sum()               # always on the simplex

        # ----- occasional random mass‑transfer (helps escape plateaux) -----
        if np.random.rand() < 0.03:            # ~3 % chance per iteration
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            # move a random fraction of mass from i to j
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta
            # re‑normalise and reset Adam state (a sudden change needs a fresh start)
            a = a / a.sum()
            x = np.log(np.maximum(a, eps))
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0

    # ------------------------------------------------------------------
    # Final reporting (framework will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
