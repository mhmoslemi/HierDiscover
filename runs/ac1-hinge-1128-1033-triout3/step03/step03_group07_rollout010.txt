"""
Search for a non‑negative step function that minimizes the
``evaluate_sequence`` metric.
The algorithm:
  1. Warm‑start from external sequence if present.
  2. Exhaustive parametric sweep over several asymmetric families
     (decreasing power‑law, decreasing exponential, linear decline,
     two‑block mixture, mixed power‑law/exponential).  For each length
     a grid of parameters is evaluated using a fast FFT based
     approximation 2·n·max(a*a).  The best candidate is kept.
  3. Local optimisation: simulated‑annealing / hill‑climbing on the
     simplex.  Three move types are used:
        – “mass transfer” (move a fraction of mass from a source index
          to a destination index),
        – “gradient step” (sub‑gradient of the current maximal
          convolution entry, possibly aggregated over several near‑max
          entries),
        – “smooth window” (redistribute the mass inside a short window
          uniformly).  Moves are accepted if they improve the score or,
          with a temperature‑dependent probability, otherwise.
  4. Final exact evaluation with ``evaluate_sequence``.
The routine respects the global time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""
import time
import numpy as np

def max_conv_fft(a: np.ndarray) -> float:
    """Return max entry of self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())

def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv

# Parametric families -----------------------------------------------------

def decreasing_powerlaw(n: int, p: float, beta: float = 0.0) -> np.ndarray:
    """Monotonically decreasing power‑law a[i] ∝ (i+1+β)^{-p}."""
    i = np.arange(n) + 1.0 + beta
    a = i ** (-p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n) / n
    else:
        a /= s
    return a

def decreasing_exponential(n: int, alpha: float) -> np.ndarray:
    """Monotonically decreasing exponential a[i] ∝ exp(-α·i)."""
    i = np.arange(n)
    a = np.exp(-alpha * i)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n) / n
    else:
        a /= s
    return a

def linear_decreasing(n: int, cutoff: int) -> np.ndarray:
    """
    Linear decreasing shape: a[i] = (cutoff - i) for i < cutoff,
    zero afterwards.  The vector is normalised to unit L1.
    """
    a = np.zeros(n)
    c = min(cutoff, n)
    if c > 0:
        a[:c] = (c - np.arange(c)).astype(float)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n) / n
    else:
        a /= s
    return a

def two_block_mixture(n: int, block_len: int, ratio: float) -> np.ndarray:
    """
    Two‑block mixture: first `block_len` entries have weight 1,
    remaining entries have weight 1/ratio (ratio>1 ⇒ first block larger).
    The vector is normalised to L1=1.
    """
    a = np.empty(n, dtype=float)
    block_len = int(min(max(block_len, 0), n))
    if block_len == n:
        a[:] = 1.0
    else:
        a[:block_len] = 1.0
        a[block_len:] = 1.0 / max(ratio, 1e-12)
    s = a.sum()
    a /= s
    return a

def mix_power_exponential(n: int, p: float, beta: float, alpha: float, lam: float) -> np.ndarray:
    """
    Convex combination of a decreasing power‑law and a decreasing exponential:
        a = λ·powerlaw + (1-λ)·exponential.
    Parameters: 0 ≤ λ ≤ 1.
    """
    pow_part = decreasing_powerlaw(n, p, beta)
    exp_part = decreasing_exponential(n, alpha)
    a = lam * pow_part + (1.0 - lam) * exp_part
    s = a.sum()
    if s == 0.0:
        a = np.ones(n) / n
    else:
        a /= s
    return a

Search for a non‑negative step function that minimizes the metric returned by
``evaluate_sequence``.  The algorithm consists of four stages:

1. **Warm‑start** – If a global variable ``height_sequence_1`` exists, it is
   normalised, scored (fast FFT based) and kept if it is the best so far.

2. **Deterministic parametric sweep** – Several *asymmetric* families are
   generated for many lengths (up to 24 000) and for a grid of shape
   parameters.  Each candidate is normalised to L1 = 1 and scored with the
   cheap approximation ``2·n·max(a*a)`` (computed by a real FFT).  The best
   candidate becomes the starting point for the refinement stage.

   Families used:
   * decreasing power‑law         a[i] ∝ (i+1+β)⁻ᵖ  
   * decreasing exponential       a[i] ∝ exp(−α·i)  
   * linear decreasing (truncated)   a[i] = max(0, c−i)  
   * two‑block mixture            a[i]=1 for i<c, =1/ratio otherwise  
   * convex mixture of power‑law & exponential.

3. **Local stochastic refinement** – Simulated‑annealing / hill‑climbing
   on the simplex.  Three move types are employed:
   * *gradient step* – sub‑gradient of the current maximal convolution entry
     (aggregated over a small set of near‑max entries) and a mirror‑descent
     update.
   * *mass‑transfer* – move a random fraction of the mass from a source index
     to a destination index; the effect on the convolution peak is estimated
     with a cheap O(n) incremental formula and the best improving move is
     applied.
   * *smooth window* – pick a short window and redistribute its total mass
     uniformly, thereby flattening local spikes.
   Moves are accepted if they improve the score or with an annealing
   probability.  The temperature decays slowly (factor 0.998 per iteration).

4. **Exact evaluation** – The best vector found is finally evaluated with the
   official ``evaluate_sequence`` routine and returned as a plain Python list.

The routine respects the global time budget (default 1000 s) and prints

# Fast convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------

# ----------------------------------------------------------------------


def decreasing_powerlaw(n: int, p: float, beta: float = 0.0) -> np.ndarray:
    """Monotonically decreasing power‑law: a[i] ∝ (i+1+β)⁻ᵖ."""
    i = np.arange(n) + 1.0 + beta
    a = i ** (-p)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def decreasing_exponential(n: int, alpha: float) -> np.ndarray:
    """Monotonically decreasing exponential: a[i] ∝ exp(−α·i)."""
    i = np.arange(n)
    a = np.exp(-alpha * i)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def linear_decreasing(n: int, cutoff: int) -> np.ndarray:
    """
    Linear decreasing shape: a[i] = (cutoff - i) for i < cutoff,
    zero afterwards.  Normalised to L1 = 1.
    """
    a = np.zeros(n, dtype=float)
    c = int(min(max(cutoff, 0), n))
    if c > 0:
        a[:c] = (c - np.arange(c)).astype(float)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def two_block_mixture(n: int, block_len: int, ratio: float) -> np.ndarray:
    """
    Two‑block mixture: first ``block_len`` entries have weight 1,
    the remaining entries have weight 1/ratio (ratio>1 ⇒ first block larger).
    Normalised to L1 = 1.
    """
    block_len = int(min(max(block_len, 0), n))
    a = np.empty(n, dtype=float)
    if block_len == n:
        a[:] = 1.0
    else:
        a[:block_len] = 1.0
        a[block_len:] = 1.0 / max(ratio, 1e-12)
    a /= a.sum()
    return a


def mix_power_exponential(
    n: int, p: float, beta: float, alpha: float, lam: float
) -> np.ndarray:
    """
    Convex combination of a decreasing power‑law and a decreasing exponential:
        a = λ·powerlaw + (1‑λ)·exponential.

    s = a.sum()
    if s == 0.0:

    else:

    return a
