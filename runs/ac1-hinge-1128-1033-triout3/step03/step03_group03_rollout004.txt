import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >=0, sum x = 1}."""
    # Duchi et al. (2008) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative step‑function (list of heights) that minimises
    `2 * n * max(conv(a, a)) / (sum(a) ** 2)`.

    The algorithm performs a smooth‑max gradient descent on the simplex.
    Convolutions are evaluated with FFTs for O(n log n) speed.
    The smoothness parameter β is annealed from a small start to a large
    value, turning the smooth approximation of the max into the true max.
    Occasional random restarts avoid getting trapped in poor local minima.
    """
    deadline = time.time() + budget_s - 5.0  # keep 5 s safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem dimension. 5000 is a good compromise.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 2. Initialise a.  If a previous construction is available we reuse it,
    #    otherwise we start from the Catalan sequence (the “constant‑convolution’’
    #    construction).
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be provided by the outer harness.
        a0 = np.array(height_sequence_1, dtype=np.float64)
        if a0.size >= n:
            a = a0[:n].copy()
        else:
            # pad with zeros and renormalise
            a = np.concatenate([a0, np.zeros(n - a0.size, dtype=np.float64)])
    except Exception:
        # Catalan‑style start: a_k = C(2k,k)/4**k
        a = np.empty(n, dtype=np.float64)
        cur = 1.0
        a[0] = cur
        for k in range(1, n):
            cur = cur * (2 * k - 1) / (2 * k)
            a[k] = cur
    # Normalise to the probability simplex
    a = a / np.sum(a)

    # ------------------------------------------------------------------
    # Helper kernels for FFT‑based convolutions.
    # ------------------------------------------------------------------
    L = 2 * n - 1               # length of a*a
    M = n + L - 1               # length of a * w_rev ( = 3n‑2 )
    N_fft = _next_pow2(M)       # power‑of‑two size for all FFTs in this run

    # ------------------------------------------------------------------
    # 3. Parameters for the annealing / gradient descent loop.
    # ------------------------------------------------------------------
    eta = 0.5                     # initial step size
    eta_min = 1e-6
    eta_decay = 0.9997
    beta = 0.5                    # start smoothness (softmax temperature)
    beta_max = 80.0
    beta_mul = 1.005              # geometric increase per iteration
    it = 0
    no_improve = 0
    best_val = np.inf
    best_seq = a.copy()

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---- FFT of the current a (shared for both convolutions) ----------
        a_pad = np.zeros(N_fft, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        # ---- Self‑convolution a*a ----------------------------------------
        conv_full = np.fft.irfft(A * A)[:L]               # length 2n‑1
        # smooth approximation of max via softmax
        maxc = conv_full.max()
        # using beta * (conv - maxc) prevents overflow
        exp_shift = np.exp(beta * (conv_full - maxc))
        w = exp_shift / exp_shift.sum()                  # weights, sum to 1

        # ---- Gradient of the smooth max ---------------------------------
        # gradient: g_j = 2 * sum_k a_k * w_{j+k}
        w_rev = w[::-1]                                   # reversed weight vector
        w_pad = np.zeros(N_fft, dtype=np.float64)
        w_pad[:L] = w_rev
        W = np.fft.rfft(w_pad)
        conv_rev = np.fft.irfft(A * W)[:M]                # length 3n‑2
        # pick the slice that corresponds to j = 0..n‑1
        g = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # ---- Gradient descent step on the simplex ------------------------
        a_candidate = a - eta * g
        a_candidate = _project_simplex(a_candidate)

        # ---- Evaluate the true objective (non‑smoothed) ------------------
        a_pad2 = np.zeros(N_fft, dtype=np.float64)
        a_pad2[:n] = a_candidate
        A2 = np.fft.rfft(a_pad2)
        conv2 = np.fft.irfft(A2 * A2)[:L]
        current_val = 2.0 * n * conv2.max()          # sum(a)==1 ⇒ denominator =1

        # ---- Keep the best solution --------------------------------------
        if current_val < best_val - 1e-12:
            best_val = current_val
            best_seq = a_candidate.copy()
            a = a_candidate
            no_improve = 0
            print(f"[iter {it:6d}] score={best_val:.10f}  eta={eta:.6f}  beta={beta:.3f}")
        else:
            a = a_candidate
            no_improve += 1

        # ---- Parameter schedules -----------------------------------------
        eta = max(eta_min, eta * eta_decay)
        beta = min(beta_max, beta * beta_mul)

        # ---- Occasional housekeeping --------------------------------------
        if it % 500 == 0:
            # re‑normalise to avoid drift caused by numerical errors
            a = a / a.sum()

        # ---- Random restart if we are stuck --------------------------------
        if no_improve > 4000:
            a = rng.random(n)
            a = a / a.sum()
            eta = 0.5
            beta = 0.5
            no_improve = 0
            print(f"[restart] iteration {it}")

    # 4. Return the best sequence (clamped to the limits required by the
    #    official evaluator).  The official evaluator is called once more
    #    to obtain the final score that will be reported.
    # ------------------------------------------------------------------
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(best_seq)
    print(f"[final] best_score={best_val:.10f}, evaluator_score={final_score:.10f}, length={len(best_seq)}")
    return best_seq

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project `v` onto the set {x | x ≥ 0, sum(x)=1}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:                               # all entries are non‑positive
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The algorithm performs a smooth‑max gradient descent on the probability
    simplex.  Convolutions are evaluated with FFTs (O(n log n)).  A
    temperature‑like parameter β anneals from a small value (soft‑max
    approximation) to a large one (true max), while the step size η decays.
    Random restarts prevent premature convergence.  The best sequence found
    within the allotted wall‑clock time is returned (clamped to the limits
    enforced by the judge).
    """
    # ------------------------------------------------------------------
    # 0. Timing / RNG
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # a few seconds safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 2. Initial vector `a` on the simplex.
    #    Prefer the classic “Catalan’’ (constant‑convolution) start;
    #    fall back to a random vector if a previous solution is unavailable.
    # ------------------------------------------------------------------

        if a0.size >= n:

        else:

    except Exception:
        # Catalan coefficients a_k = C(2k,k) / 4**k
        a = np.empty(n, dtype=np.float64)
        cur = 1.0
        a[0] = cur
        for k in range(1, n):
            cur = cur * (2 * k - 1) / (2 * k)
            a[k] = cur
    a = a / a.sum()               # project onto the simplex

    # ------------------------------------------------------------------
    # 3. Pre‑compute FFT size (large enough for every convolution we need)
    # ------------------------------------------------------------------
    L = 2 * n - 1                 # length of a*a
    M = n + L - 1                 # length of a * w_rev  ( = 3n‑2 )
    N_fft = _next_pow2(M)         # power‑of‑two for FFTs

    # ------------------------------------------------------------------
    # 4. Hyper‑parameters for the annealing/gradient descent
    # ------------------------------------------------------------------
    eta = 0.5                     # initial step size
    eta_min = 1e-6
    eta_decay = 0.9997
    beta = 0.5                    # start soft‑max temperature
    beta_max = 80.0
    beta_mul = 1.005              # geometric increase per iteration
    it = 0
    no_improve = 0
    best_val = np.inf
    best_seq = a.copy()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- FFT of the current a (shared for both convolutions) -----
        a_pad = np.zeros(N_fft, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        # ----- Self‑convolution a*a ----------------------------------
        conv = np.fft.irfft(A * A)[:L]                     # length 2n‑1

        # ----- Smooth approximation of max (soft‑max) ---------------
        #   max_soft = (1/β)·log Σ exp(β·conv_i)
        #   Its gradient w.r.t. a is a weighted sum of the gradients of each
        #   entry, where the weights are the soft‑max probabilities.
        maxc = conv.max()
        exp_shift = np.exp(beta * (conv - maxc))          # shift for stability
        w = exp_shift / exp_shift.sum()                  # Σ w_i = 1

        # ----- Gradient of the smooth max ----------------------------
        #   ∂max_soft/∂a_j = Σ_i w_i·∂conv_i/∂a_j,
        #   and ∂conv_i/∂a_j = 2·a_{i-j} (if index in range).
        #   This can be written as a convolution of a with the reversed weight
        #   vector w_rev.
        w_rev = w[::-1]

        # pick slice that corresponds to j = 0,…,n‑1

        a_new = a - eta * g
