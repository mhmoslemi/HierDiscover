"""
Propose a candidate step‑function (non‑negative vector) that minimises

    2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm consists of

1.  A fast deterministic sweep over several parametric, symmetric families
    (power‑law U‑shapes, shifted U‑shapes, product‑distance, exponential,
    cosine‑based) for a variety of lengths.  The best vector from this sweep
    seeds the stochastic phase.

2.  Stochastic refinement on the probability simplex using an
    *exponential‑gradient* update (mirror descent w.r.t. KL) driven by a
    sub‑gradient of the maximal convolution entry, combined with a simple
    simulated‑annealing acceptance rule, occasional random mass‑transfer
    moves and a re‑heat strategy to escape stagnation.

All helper functions are top‑level (no closures) and the routine respects
the given time budget (default 1000 s).  The returned list sums to one,
so the external evaluator will only rescale it as required.
"""

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex { w >= 0 , sum(w) = 1 }.
    Implementation follows Wang & Carreira‑Perpiñán (O(n log n)).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a * a using a real FFT.
    Returns a NumPy array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector with sum(a)=1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    For each index k attaining (or almost attaining) the maximum, we add
    the contribution 2*a[j] to the gradient entry i where i+j = k.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent / exponentiated‑gradient step on the simplex:
        a' ∝ a * exp(-η * grad)
    The result is exactly normalised to sum = 1.
    """
    # avoid under‑/overflow by clipping the exponent
    exp_term = np.exp(-eta * grad)
    a_new = a * exp_term
    s = a_new.sum()
    if s == 0.0:
        # fallback to uniform distribution if everything vanished
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new

def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))^-p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a
