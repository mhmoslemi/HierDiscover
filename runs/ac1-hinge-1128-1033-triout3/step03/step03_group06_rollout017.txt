    evaluated with the official ``evaluate_sequence`` and returned.

The routine respects a 1000 s time budget, prints progress, and never uses
file or network I/O.
"""

import time
import numpy as np
from typing import List

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------

def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)−1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector ``old`` (sum=1) to a new length.
    """
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    # tiny epsilon keeps the gradient well‑behaved
    new += 1e-14
    new /= new.sum()
    return new


def subgradient_max_convolution(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal entry of the self‑convolution.
    Returns a vector g with the same shape as ``a``.
    """
    n = a.shape[0]
    max_val = conv.max()
    # all indices attaining the maximum – usually a single one
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        # contribution 2·a_{k−i}
        g[idx] += 2.0 * a[k - idx]
    return g


def objective_from_conv(n: int, conv_max: float) -> float:
    """
    Fast proxy for the evaluation function when the vector is normalised
    to have L1‑norm = 1.
    """
    return 2.0 * n * conv_max


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------

def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function returns the best list of floats it
    discovered within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5            # safety margin

    # --------------------------------------------------------------------
    # 0.  Load seed if supplied (global variable ``height_sequence_1``)
    # --------------------------------------------------------------------
    best_seq = None
    best_obj = np.inf
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external /= s                                 # normalise to 1
            conv = conv_full_fft(external)
            cur_obj = objective_from_conv(len(external), conv.max())
            best_seq = external.copy()
            best_obj = cur_obj
            print(f"Seeded from height_sequence_1: n={len(external)}, "
                  f"objective≈{cur_obj:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------------
    # 1.  Fallback initialisation: uniform distribution on a modest size
    # --------------------------------------------------------------------
    if best_seq is None:
        n0 = 8000
        best_seq = np.full(n0, 1.0 / n0, dtype=float)
        cur_obj = objective_from_conv(n0, 1.0 / n0)
        best_obj = cur_obj
        print(f"Fallback initialisation: n={n0}, objective≈{cur_obj:.6f}")

    # --------------------------------------------------------------------
    # 2.  Lengths we will explore (larger n gives a larger factor 2·n)
    # --------------------------------------------------------------------
    lengths = [4000, 8000, 12000, 16000, 20000]
    lengths = sorted(set(lengths))

    # --------------------------------------------------------------------
    # 3.  Main optimisation loop (projected sub‑gradient descent)
    # --------------------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_obj = best_obj

    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise candidate for this length
        if global_best_seq.shape[0] != n:
            a = resample_distribution(global_best_seq, n)
        else:
            a = global_best_seq.copy()

        # optimisation parameters
        eta = 0.5                # initial multiplicative step size
        eta_min = 1e-7
        eta_max = 5.0
        no_improve_cnt = 0
        max_no_improve = 2000   # after this we add jitter
        iteration = 0

        print(f"\n=== Optimising length n={n} ===")
        while time.time() < deadline:
            iteration += 1

            # --- evaluate current point ---
            conv = conv_full_fft(a)
            cur_max = conv.max()
            cur_obj = objective_from_conv(n, cur_max)

            # keep global best
            if cur_obj < global_best_obj:
                global_best_obj = cur_obj
                global_best_seq = a.copy()
                no_improve_cnt = 0
                print(f"Iter {iteration:5d} – new global best: {cur_obj:.6f}")

                # early stop if we breach the target
                if global_best_obj < 1.503:
                    print("Target reached – exiting early.")
                    break
            else:
                no_improve_cnt += 1

            # --- compute sub‑gradient of the maximal convolution entry ---
            grad = subgradient_max_convolution(a, conv)

            # multiplicative mirror‑descent step (exponential update)
            exp_arg = -eta * grad
            # avoid overflow / underflow
            exp_arg = np.clip(exp_arg, -50.0, 50.0)
            a_candidate = a * np.exp(exp_arg)
            a_candidate = project_to_simplex(a_candidate)

            # evaluate candidate (exact max)
            conv_cand = conv_full_fft(a_candidate)
            cand_obj = objective_from_conv(n, conv_cand.max())

            # acceptance rule
            if cand_obj < cur_obj:
                a = a_candidate
                eta = min(eta * 1.02, eta_max)   # successful – be a bit bolder
                no_improve_cnt = 0
            else:
                # reject – shrink step size
                eta = max(eta * 0.90, eta_min)

            # --- occasional random jitter to escape plateaus ---
            if no_improve_cnt >= max_no_improve:
                jitter = np.random.randn(n) * 1e-3
                a = a + jitter
                a = np.maximum(a, 0.0)
                a = project_to_simplex(a)
                no_improve_cnt = 0
                print(f"Iter {iteration:5d}: jitter applied, η={eta:.4f}")

            # optional progress output every 2000 iterations
            if iteration % 2000 == 0:
                print(f"Iter {iteration:5d}: cur_obj≈{cur_obj:.6f}, η={eta:.4f}")

        # keep the last vector as seed for the next (larger) length
        global_best_seq = global_best_seq.copy()   # tiny safety copy
        if global_best_obj < 1.503:
            break

    # --------------------------------------------------------------------
    # 4.  Final official evaluation & return
    # --------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(global_best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")
        exact_score = global_best_obj

    print("\n=== Search finished ===")
    print(f"Best objective (approx.) : {global_best_obj:.6f}")
    print(f"Best length               : {global_best_seq.shape[0]}")
    return global_best_seq.tolist()
