```
    n = a.shape[0]
```

```
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0
    # Step 1: small deterministic sweep
    lengths = [2000, 4000, 6000, 8000]
    exponents = [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0]
    lam_vals = [1e-4, 1e-3, 0.01, 0.05, 0.1, 0.2, 0.3]
    best_seed = None
    best_score = np.inf
    # sweep
    for n in lengths:
        for p in exponents:
            a = u_min_distribution(n, p)
            sc = score(a)
            if sc < best_score:
                best_score, best_seed = sc, a.copy()
        for p in exponents:
            a = u_prod_distribution(n, p)
            sc = score(a)
            if sc < best_score:
                best_score, best_seed = sc, a.copy()
        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score(a)
            if sc < best_score:
                best_score, best_seed = sc, a.copy()

    # incorporate external seed if exists
    try:
        ext = np.array(height_sequence_1, dtype=float)
        ext = np.maximum(ext,0.0)
        s = ext.sum()
        if s > 0:
            ext = ext / s
            sc = score(ext)
            if sc < best_score:
                best_score, best_seed = sc, ext.copy()
    except NameError:
        pass

    if best_seed is None:
        # fallback simple uniform
        n = 5000
        best_seed = np.ones(n) / n
        best_score = score(best_seed)

    n = best_seed.shape[0]

    # Build initial population
    pop_size = 30
    population = np.empty((pop_size, n), dtype=np.float64)
    pop_scores = np.empty(pop_size, dtype=np.float64)

    # Fill first entries with some structured seeds
    seeds = [best_seed]
    # add some extra param seeds
    for p in [0.5, 1.5, 2.5]:
        seeds.append(u_min_distribution(n, p))
        seeds.append(u_prod_distribution(n, p))
    for lam in [0.001, 0.01, 0.05]:

    # ensure we have at most pop_size seeds
    for i, seed_vec in enumerate(seeds):

    # fill rest random Dirichlet
    for i in range(len(seeds), pop_size):
        rand = np.random.rand(n)
        population[i] = rand / rand.sum()

    # track overall best
    best_idx = pop_scores.argmin()
    best_vec = population[best_idx].copy()
    best_score = pop_scores[best_idx]

    # DE hyperparameters
    F = 0.8
    CR = 0.9

    # Main DE loop
    generation = 0
    while time.time() < deadline:

        # maybe adjust F,CR over time: we can linearly decay F a bit
```

```
def compute_subgradient(a, conv):
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k-(n-1))
        end_i = min(n-1, k)
        i_vals = np.arange(start_i, end_i+1)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad
```

The evaluator computes  

\[
\text{score}(a)=\frac{2\,|a|\;\max (a*a)}{(\sum a)^2},
\]

so the objective is *scale‑invariant*.  We may therefore restrict ourselves to
probability vectors `∑a = 1`.  The problem is highly non‑convex – the
objective is the maximum entry of a quadratic form – and simple gradient
descent can get stuck in a local minimum.

The new search combines two complementary ideas  

1. **A quick deterministic sweep** over a few standard “U‑shaped’’ families
   (inverse distance, inverse product distance, exponential decay).  
   The best vector from this sweep (or the external seed `height_sequence_1`
   if it exists) is used as the **base length** `n`.

2. **Differential Evolution (DE)** on the simplex.  
   DE is a population‑based, gradient‑free optimiser that works well on
   highly non‑linear problems.  Each candidate is projected onto the simplex
   after mutation & crossover, guaranteeing feasibility at every step.
   A small population (≈30) and a moderate number of generations keep the
   runtime well below the 1000 s budget while exploring the search space far
   beyond a single hill‑climber.

3. **Local sub‑gradient polishing** – after DE terminates we run a few
   projected sub‑gradient steps (the same gradient used in the previous
   solution) to squeeze out a final improvement.

All helper functions are top‑level, use only `numpy` (and the already‑imported
`evaluate_sequence`), and respect the time budget.

```python
"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimises

    2 * n * max_k (a * a)[k]

where * denotes full linear convolution.  The algorithm works in three
phases:

1. Deterministic sweep over a handful of parametric families
   (inverse‑distance, inverse‑product‑distance, exponential decay) to obtain
   a good initial length and a seed vector.

2. Differential Evolution on the probability simplex to explore the space
   globally.  After every mutation/crossover the trial vector is projected
   onto the simplex (Wang & Carreira‑Perpiñán O(n log n) algorithm).

3. Short projected sub‑gradient polishing to obtain a final local improvement.

The routine respects a 1000 s time budget, prints progress, and returns the
best vector found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex:
        { w >= 0 , sum(w) = 1 }
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution & scoring utilities
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n                 # zero‑padding length
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)   # element‑wise product in frequency domain
    return conv[: 2 * n - 1]


def score_vector(a: np.ndarray) -> float:
    """
    Exact value returned by evaluate_sequence for a vector with sum == 1.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices where the convolution attains (or almost attains) its
    maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        # each (i,j) pair contributes 2*a[j] to ∂/∂a[i]
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Parametric “U‑shaped’’ families (used for the deterministic seed)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i, n‑1‑i)+1)^{-p}   (symmetric, edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^{-p}   (edge‑heavy, product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i) + exp(-lam*(n‑1‑i))   ( symmetric exponential )."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the given time
    budget. Returns a plain Python list of floats (∑a ≈ 1).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin for final return

    # ------------------------------------------------------------------
    # 1. Deterministic sweep – cheap but gives a good base length n
    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf

    # modest set of lengths & parameters (covers small, medium, large)
    lengths = [2000, 4000, 6000, 8000]
    exponents = [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0]
    lambdas = [1e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1, 3e-1]

    print("=== Deterministic sweep start ===")
    for n in lengths:
        if time.time() > deadline:
            break
        for p in exponents:
            a = u_min_distribution(n, p)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f" new best (min)  n={n:5d} p={p:4.2f}  score={sc:.6f}")
            a = u_prod_distribution(n, p)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f" new best (prod) n={n:5d} p={p:4.2f}  score={sc:.6f}")
        for lam in lambdas:
            a = exp_distribution(n, lam)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f" new best (exp)  n={n:5d} λ={lam:.5e}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate external construction if it exists
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        s = external.sum()
        if s > 0.0:
            external /= s
            sc = score_vector(external)
            if sc < best_score:
                best_score, best_vec = sc, external.copy()
                print(f" external seed used, length {len(external)} score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 5000
        best_vec = np.random.rand(n)
        best_vec /= best_vec.sum()
        best_score = score_vector(best_vec)
        print(f" fallback random seed, n={n} score={best_score:.6f}")

    n = best_vec.shape[0]           # final problem dimension
    print(f"\n=== Selected problem size: n = {n} ===")
    print(f" Seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Differential Evolution on the simplex
    # ------------------------------------------------------------------
    pop_size = 30                       # modest size – fits time budget
    F = 0.8                              # mutation factor
    CR = 0.9                             # crossover probability

    # initialise population
    population = np.empty((pop_size, n), dtype=np.float64)
    pop_scores = np.empty(pop_size, dtype=np.float64)

    # first a few curated seeds (the best one + a few parametric variations)
    seeds = [best_vec]
    for p in [0.8, 2.0, 4.0]:
        seeds.append(u_min_distribution(n, p))
        seeds.append(u_prod_distribution(n, p))
    for lam in [1e-3, 5e-2, 2e-1]:
        seeds.append(exp_distribution(n, lam))

    for i, vec in enumerate(seeds[:pop_size]):
        population[i] = vec
        pop_scores[i] = score_vector(vec)

    # fill the rest with random Dirichlet draws
    for i in range(len(seeds), pop_size):
        rnd = np.random.rand(n)
        population[i] = rnd / rnd.sum()
        pop_scores[i] = score_vector(population[i])

    # keep a global best copy
    best_idx = int(pop_scores.argmin())
    best_vec = population[best_idx].copy()
    best_score = float(pop_scores[best_idx])

    generation = 0
    print("\n=== DE optimisation start ===")
    while time.time() < deadline:
        generation += 1
        # one DE generation
        for i in range(pop_size):
            # select three distinct individuals a,b,c ≠ i
            idxs = np.arange(pop_size)
            idxs = idxs[idxs != i]
            a, b, c = np.random.choice(idxs, 3, replace=False)
            donor = population[a] + F * (population[b] - population[c])
            donor = np.maximum(donor, 0.0)          # keep non‑negative

            # binomial crossover
            crossover_mask = np.random.rand(n) < CR
            if not np.any(crossover_mask):
                crossover_mask[np.random.randint(n)] = True
            trial = np.where(crossover_mask, donor, population[i])

            # project back onto the simplex
            trial = simplex_projection(trial)
            trial /= trial.sum()                     # tiny renorm for safety

            trial_score = score_vector(trial)

            # selection
            if trial_score < pop_scores[i]:
                population[i] = trial
                pop_scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    print(f"Gen {generation:4d} ind {i:2d}: new best = {best_score:.6f}")

        # occasional time‑check
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 5. Short projected sub‑gradient polish (fine‑tuning)
    # ------------------------------------------------------------------
    print("\n=== Sub‑gradient polishing start ===")
    current = best_vec.copy()
    current_score = best_score
    step = 0.15            # initial step size
    step_decay = 0.9995
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution_fft(current)
        grad = compute_subgradient(current, conv)

        cand = current - step * grad
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)
        cand /= cand.sum()
        cand_score = score_vector(cand)

        if cand_score < current_score:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                last_improve = it
                print(f"Subgrad it {it:6d}: new best = {best_score:.6f}")

        step *= step_decay
        if step < 1e-7:
            break
        # re‑heat if stuck
        if it - last_improve > 3000:
            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            current /= current.sum()
            current_score = score_vector(current)
            step = 0.15
            last_improve = it
            print(f"Subgrad it {it:6d}: re‑heat")

    # ------------------------------------------------------------------
    # 6. Return the best sequence found
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {len(best_vec)})")
    return best_vec.tolist()
```
