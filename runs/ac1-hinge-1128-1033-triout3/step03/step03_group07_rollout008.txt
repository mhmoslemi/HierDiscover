def project_simplex(v):
    # ensure sum(v) = 1 and v >=0
    # using algorithm from Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step function that (approximately) minimizes

    score = 2·n·max(a*a) / (sum(a)^2)

The function is scale‑invariant, so we keep the vector normalised to sum = 1.
The search proceeds in three stages:

1. **Warm‑start** – if a global variable ``height_sequence_1`` exists it is
   normalised and taken as the initial best candidate.

2. **Parametric sweep** – a compact family of symmetric “U‑shaped” power‑law
   distributions is exhaustively scanned for several lengths (up to 30 000)
   with a fine grid of exponents.  The best candidate from this sweep is
   added to the list of starting points.

3. **Smooth‑max gradient descent on the simplex** –  
   For a current vector ``a`` we compute its full self‑convolution
   ``c = a*a`` (FFT, O(n log n)).  The maximum entry is approximated by the
   smooth‑max (soft‑max) of ``c`` with an inverse temperature ``β``.
   The gradient of the smooth‑max is a weighted correlation between ``a``
   and the soft‑max weights and can be obtained with a second FFT.
   A projected gradient step (onto the probability simplex) is taken,
   the step size ``η`` is adapted on the fly, and ``β`` is slowly increased
   so that the smooth‑max converges to the true max.
   The loop runs until the time budget (≈ 1000 s) is exhausted or the
   target score < 1.503 is reached.

The best vector found (exactly evaluated by ``evaluate_sequence``) is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    if v.ndim != 1:
        raise ValueError("project_simplex expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho where condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                     # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors a and b using FFT.
    Returns a 1‑D array of length len(a)+len(b)-1.
    """
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << (n.bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)[:n]
    return conv


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimisation routine (smooth‑max gradient descent)
# ----------------------------------------------------------------------


def _smooth_max_gradient_step(a: np.ndarray,
                              beta: float,
                              eta: float) -> tuple[np.ndarray, float]:
    """
    Perform one gradient descent step on the smooth‑max of the convolution.
    Returns the new vector (already projected onto the simplex) and the
    resulting approximate score 2·n·max(conv).
    """
    n = a.shape[0]

    # ----- 1. self‑convolution (the smooth‑max argument) -----------------
    size_conv = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size_conv)
    conv = np.fft.irfft(A * A, n=size_conv)[: 2 * n - 1]
    cmax = conv.max()

    # ----- 2. soft‑max weights ------------------------------------------
    # w_k ≈ exp(β·(c_k−cmax)) / Σ exp(...)
    w = np.exp(beta * (conv - cmax))
    w_sum = w.sum()
    if w_sum == 0.0:          # numerical safeguard
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # ----- 3. gradient of the smooth‑max (weighted correlation) ----------
    # linear correlation r[i] = Σ_j a[j]·w[i+j] for i=0..n‑1
    len_w = w.shape[0]                     # = 2·n‑1
    size_corr = 1 << ((n + len_w - 1).bit_length())
    A_corr = np.fft.rfft(a, n=size_corr)
    W_corr = np.fft.rfft(w, n=size_corr)
    corr_full = np.fft.irfft(A_corr * np.conj(W_corr), n=size_corr)
    # keep only the first n entries (lags 0 … n‑1)
    r = corr_full[:n]
    grad = 2.0 * r                        # ∂/∂a of Σ_k w_k·c_k

    # ----- 4. projected gradient step ------------------------------------
    cand = a - eta * grad
    cand = project_simplex(cand)

    # ----- 5. evaluate new approximated score ----------------------------
    A2 = np.fft.rfft(cand, n=size_conv)
    conv2 = np.fft.irfft(A2 * A2, n=size_conv)[: 2 * n - 1]
    new_score = 2.0 * n * conv2.max()
    return cand, new_score


def _run_gradient_descent(start: np.ndarray,
                         time_deadline: float,
                         target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Run the smooth‑max gradient descent from a starting point.
    Returns the best vector and its approximate score encountered.
    """
    a = start.copy()
    n = a.shape[0]

    # hyper‑parameters (tuned empirically)
    eta = 0.15               # initial step size
    eta_min = 1e-7
    beta = 5.0               # start with a soft approximation
    beta_inc = 0.04
    beta_max = 150.0
    best_a = a.copy()
    # initial score
    best_score = 2.0 * n * max_conv_fft(a)

    it = 0
    while time.time() < time_deadline:
        it += 1
        cand, cand_score = _smooth_max_gradient_step(a, beta, eta)

        # accept if improvement, otherwise shrink step size
        if cand_score < best_score:
            a = cand
            best_a = a.copy()
            best_score = cand_score
            # occasionally enlarge step size after a success
            eta = min(eta * 1.05, 0.5)
            # progress report
            print(f"Iter {it:6d}: new best approx score = {best_score:.6f}")
            if best_score < target_score:
                print("Target met – exiting early from gradient descent.")
                break
        else:
            # reject: keep the old a but reduce the step size
            eta = max(eta * 0.6, eta_min)

        # increase the soft‑max temperature slowly
        beta = min(beta + beta_inc, beta_max)

        # occasional safety‑check for stagnation
        if it % 200 == 0:
            # re‑evaluate exact convolution to guard against drift
            cur_score = 2.0 * n * max_conv_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"Iter {it:6d}: occasional re‑eval best ≈ {best_score:.6f}")

    return best_a, best_score
