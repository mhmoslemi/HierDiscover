def local_refine(seq, time_deadline):
    best = seq.copy()
    best_score = 2 * len(seq) * max_conv_fft(best)
    current = seq.copy()
    current_score = best_score
    n = len(seq)
    temperature = 1.0
    while time.time() < time_deadline:
        # compute conv
        conv = full_conv_fft(current)  # maybe use max_conv_fft only
        kmax = np.argmax(conv)
        # compute subgradient
        grad = np.zeros_like(current)
        # compute mapping
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = kmax - idx_i
        grad[idx_i] = 2.0 * current[idx_j]  # hold
        # step size:
        step = temperature * 0.05  # delta fraction
        cand = current - step * grad
        cand = np.clip(cand, 0, None)
        cand = normalize(cand)
        cand_score = 2 * n * max_conv_fft(cand)
        # SA acceptance
        if cand_score < current_score or np.random.rand() < np.exp((current_score - cand_score) / max(temperature, 1e-12)):
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"Refine: new best {best_score:.6f}")
        # random mass transfer step:
        # sample src proportional to current
        src = np.random.choice(n, p=current/current.sum())
        dst = np.random.randint(0, n-1)
        if dst >= src:
            dst += 1
        delta = current[src] * np.random.rand() * 0.2  # move fraction
        new_seq = current.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta
        new_score = 2 * n * max_conv_fft(new_seq)
        if new_score < current_score or np.random.rand() < np.exp((current_score - new_score)/max(temperature, 1e-12)):
            current = new_seq
            current_score = new_score
            if new_score < best_score:
                best = new_seq.copy()
                best_score = new_score
                print(f"Refine: new best after random move {best_score:.6f}")
        temperature *= 0.9995
    return best, best_score
```

"""
Advanced search for a non‑negative step function minimizing

    score = 2 * n * max(self‑convolution) / (sum(a)^2)

Because the score is scale‑invariant we keep all candidates normalised
to sum = 1 and minimise 2·n·max(conv) where conv = a∗a.
The algorithm consists of three stages:

1️⃣ Warm‑start from a provided global ``height_sequence_1`` (if any).

2️⃣ Deterministic sweep over several expressive parametric families:
   * symmetric U‑shapes (power‑law)
   * sum‑of‑tails (two power‑laws)
   * pure exponential decay from the edges
   * product power‑law
   * Beta‑type ( (i+1)^{α‑1}·(n‑i)^{β‑1} )
   * sin‑power ( (sin‑shape)^{‑p} )
   * asymmetric double‑exponential (exp(‑α·i)+exp(‑β·(n‑1‑i)))
   * two‑block piecewise‑constant (high weight on the ends)

   The sweep explores many lengths (up to 64 000) and a grid of
   parameters.  The best discovered vector (approximate score) is kept.

3️⃣ Stochastic refinement starting from the current best vector.
   Each iteration performs:
   • a full convolution to locate the peak index kₘₐₓ,
   • a projected sub‑gradient step (moving away from the sub‑gradient
     gᵢ = 2·a_{kₘₐₓ‑i}),
   • a random “mass‑transfer’’ move,
   • a Metropolis acceptance test with a slowly cooling temperature.
   The best vector found during refinement is stored.

Finally the candidate is evaluated with the official ``evaluate_sequence``.
The whole search respects a 1000 s time budget (5 s safety margin).

All helper functions are defined at top level, no I/O is performed, and
the returned object is a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Normalisation helpers
# ----------------------------------------------------------------------


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    s = v.sum()
    if s <= 0.0:
        v = np.ones_like(v)
        s = v.size
    return v / s


# ----------------------------------------------------------------------
# Parametric families (all return a normalised vector with sum == 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    return normalize(a)


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    return normalize(a)


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    return normalize(a)


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """Product‑power‑law: a[i] ∝ ((i+1)*(n-i))^{‑p}."""
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-12
    return normalize(a)


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape: a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(n)
    a = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1) + 1e-12
    return normalize(a)


def sin_power_distribution(n: int, p: float) -> np.ndarray:
    """U‑shaped via a negative power of a sine‑shape."""
    i = np.arange(n)
    s = np.sin(np.pi * (i + 0.5) / n)          # ∈ (0,1]
    a = s ** (-p) + 1e-12
    return normalize(a)


def double_exponential_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric sum of two exponentials from both edges."""
    i = np.arange(n)
    a = np.exp(-alpha * i) + np.exp(-beta * (n - 1 - i)) + 1e-12
    return normalize(a)


def two_block_distribution(n: int, L_frac: float, high_ratio: float) -> np.ndarray:
    """
    Simple two‑block shape.
    The first L = int(L_frac·n) indices at each end receive weight
    high_ratio × the middle weight.
    """
    L = max(1, int(L_frac * n))
    if 2 * L >= n:
        # fall back to uniform
        return np.full(n, 1.0 / n)
    # unknowns: x = weight at an end site, y = weight at a middle site.
    # Constraints:
    #   2·L·x + (n‑2·L)·y = 1   and   x = high_ratio·y
    y = 1.0 / (2 * L * high_ratio + (n - 2 * L))
    x = high_ratio * y
    a = np.full(n, y)
    a[:L] = x
    a[-L:] = x
    return normalize(a)

# Local stochastic refinement (simulated annealing)
# ----------------------------------------------------------------------


def local_refine(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a simple simulated‑annealing / projected‑gradient search.
    Returns the best sequence found and its approximate score (2·n·max(conv)).
    """
    current = start_seq.copy()
    n = current.shape[0]
    # initial full convolution (exact) – needed for the first kmax
    conv = full_conv_fft(current)
    current_score = 2.0 * n * conv.max()
    best_seq = current.copy()
    best_score = current_score

    temperature = 1.0
    cooling = 0.9995
    max_frac = 0.25               # max fraction of source mass that may be moved
    grad_step = 0.07              # base step size for gradient move (scaled by temp)

    it = 0
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # a) locate the current peak of the convolution
        # --------------------------------------------------------------
        conv = full_conv_fft(current)           # O(n log n)
        kmax = int(conv.argmax())
        conv_max = conv.max()
        current_score = 2.0 * n * conv_max

        # --------------------------------------------------------------
        # b) generate a gradient (sub‑gradient) move
        # --------------------------------------------------------------
        grad = np.zeros_like(current)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        idx_j = kmax - idx_i
        grad[idx_i] = 2.0 * current[idx_j]

        step = grad_step * temperature
        cand_grad = current - step * grad
        cand_grad = np.clip(cand_grad, 0.0, None)
        cand_grad = normalize(cand_grad)
        cand_grad_score = 2.0 * n * max_conv_fft(cand_grad)

        # --------------------------------------------------------------
        # c) generate a random mass‑transfer move
        # --------------------------------------------------------------
        src = np.random.choice(n, p=current / current.sum())
        dst = np.random.randint(0, n - 1)
        if dst >= src:
            dst += 1
        delta = current[src] * np.random.rand() * max_frac
        cand_rand = current.copy()
        cand_rand[src] -= delta
        cand_rand[dst] += delta
        cand_rand = normalize(cand_rand)   # re‑normalise to guard against rounding
        cand_rand_score = 2.0 * n * max_conv_fft(cand_rand)

        # --------------------------------------------------------------
        # d) pick the better of the two candidates
        # --------------------------------------------------------------
        if cand_grad_score <= cand_rand_score:
            cand = cand_grad
            cand_score = cand_grad_score
        else:
            cand = cand_rand
            cand_score = cand_rand_score

        # --------------------------------------------------------------
        # e) Metropolis acceptance test
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            # we already know cand_score, no need to recompute conv
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"Iter {it:6d}: new global best ≈ {best_score:.6f}")

                # early exit if we have crossed a plausible target
                if best_score < 1.503:
                    print("Target threshold reached – stopping refinement.")
                    break

        # --------------------------------------------------------------
        # f) temperature schedule
        # --------------------------------------------------------------
        temperature = max(temperature * cooling, 1e-8)

    return best_seq, best_score
