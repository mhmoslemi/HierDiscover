"""
Propose a candidate step‑function using a smoothed‑max gradient descent
(Adam) on the self‑convolution.  The objective
    2 * n * max(conv(a,a)) / (sum(a)**2)
is invariant under scaling, so the sequence is kept on the
probability simplex (∑a=1, a_i≥0).  At each iteration we:

1. Compute the self‑convolution c = a*a.
2. Form a soft‑max weight vector w over the entries of c with a
   temperature that slowly decreases; when the temperature is large
   w is almost uniform (smooth objective) and when it is small w
   concentrates on the maximal entries (approaches the true max).
3. The gradient of the weighted sum  Σ w_k c_k w.r.t. a is
        g_i = 2 * Σ_k w_k a_{k-i} .
   This can be evaluated efficiently by a single convolution.
4. Apply an Adam update, then project back onto the simplex.
5. Keep the best sequence seen.  After the time budget a short
   greedy hill‑climbing phase fine‑tunes the result.  The final
   sequence is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (∑x=1, x≥0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                      # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:                                      # all non‑positive -> uniform
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of weighted convolution using a single FFT convolution
# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k (a*a)_k = 2 * Σ_k w_k a_{k-i}.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current sequence (non‑negative, sum = 1).
    w : np.ndarray, shape (2n-1,)
        Non‑negative weights (soft‑max over the convolution).

    Returns
    -------
    g : np.ndarray, shape (n,)
        Gradient vector.
    """
    n = a.shape[0]
    # reverse weights for the correlation‑style convolution
    w_rev = w[::-1]
    # full convolution a * w_rev  (length 3n‑2)
    full = np.convolve(a, w_rev)
    # indices that give Σ_j w_{i+j} a_j  for each i
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * full[idx]
    return g
