Implement projection_onto_simplex(v): this ensures sum=1 and non-negative. Use algorithm from "Efficient Projections onto the L1-Ball" by Duchi.

Pseudo:

def project_onto_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    n = a.shape[0]

    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())

    return conv

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, sum x = 1}."""
    # Implementation of the algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    a /= a.sum()
    return a

def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a

def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a

def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-12
    a /= a.sum()
    return a

        print(f"Fallback random init: n={n}, approx_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement – mixed projected‑gradient + random moves
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]
    conv = full_conv_fft(current)
    conv_max = conv.max()
    current_score = 2.0 * n * conv_max

    # Simulated‑annealing parameters
    temperature = 1.0
    cooling_rate = 0.998
    prob_gradient = 0.25        # chance to perform a gradient step
    max_frac = 0.4             # max fraction of source mass moved in a random step
    step0 = 0.5                # initial gradient step size
    it = 0

    print("\n=== Stochastic refinement (gradient + random) ===")
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Choose move type
        # --------------------------------------------------------------
        if np.random.rand() < prob_gradient:
            # ----- Projected gradient step -----
            step = step0 / np.sqrt(it + 1.0)
            g = subgradient_max_conv(current)
            cand = current - step * g
            cand = project_onto_simplex(cand)
            cand_score = 2.0 * n * max_conv_fft(cand)
        else:
            # ----- Random mass‑transfer move -----
            if np.all(current == 0):
                break
            src = np.random.choice(n, p=current / current.sum())
            if current[src] <= 0.0:
                continue
            dst = np.random.randint(0, n - 1)
            if dst >= src:
                dst += 1
            delta = current[src] * np.random.rand() * max_frac
            if delta <= 0.0:
                continue
            cand = current.copy()
            cand[src] -= delta
            cand[dst] += delta
            # Renormalise to guard against minute floating‑point drift
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand /= cand_sum
            cand_score = 2.0 * n * max_conv_fft(cand)

        # --------------------------------------------------------------
        # Simulated‑annealing acceptance
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if current_score < best_score:
                best_score = current_score
                best_seq = current.copy()
                print(f"Iter {it:6d}: new global best = {best_score:.6f}")
                if best_score < 1.503:   # generous early‑stop target
                    print("Target reached – exiting early.")
                    break

        # Cool down
        temperature = max(temperature * cooling_rate, 1e-7)

    # ------------------------------------------------------------------
    # 4. Exact evaluation and return
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())  # type: ignore
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()
