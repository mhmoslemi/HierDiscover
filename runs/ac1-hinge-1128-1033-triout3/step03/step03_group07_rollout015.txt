def local_refine(a, deadline):

    n = len(a)

    cur = a.copy()

    cur_score = 2 * n * max_conv_fft(cur)

    best_seq_local = cur.copy()

    best_score_local = cur_score

    temperature = 1.0

    max_frac = 0.3

    iter = 0

    while time.time() < deadline:

        iter += 1

        conv = full_conv_fft(cur)

        max_val = conv.max()

        kmax = int(conv.argmax())

        # compute partner contributions
        # see for i in [0, n-1]; partner = cur[kmax-i] if index in range else 0

        idx = np.arange(n)

        partner = np.where((kmax - idx >= 0) & (kmax - idx < n), cur[kmax - idx], 0.0)

        # find source and destination
        src = np.argmax(partner)
        dst = np.argmin(partner)

        if src == dst:
            # random perturb then continue
            src = np.random.randint(0, n)
            dst = (src + np.random.randint(1, n)) % n

        # magnitude of transfer
        delta = cur[src] * max_frac * np.random.rand()  # random fraction of max_frac

        if delta <= 0:
            continue

        new_a = cur.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        # compute new score
        new_conv_max = full_conv_fft(new_a).max()
        new_score = 2 * n * new_conv_max

        # accept?
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / max(temperature, 1e-12)):
            # accept move
            cur = new_a
            cur_score = new_score
            # update best
            if cur_score < best_score_local:
                best_score_local = cur_score
                best_seq_local = cur.copy()
                print(f"Iter {iter:6d}: new global best = {best_score_local:.6f}")
                # optional early stop if < target
                if best_score_local < 1.5:
                    print("Target reached.")
                    break

        # temperature decay
        temperature *= 0.9995

        # occasionally random small perturb (maybe every 10 iter)
        if iter % 10 == 0:
            # random jitter: add small noise and renorm
            noise = np.random.rand(n) * 1e-4
            cur += noise
            cur = np.maximum(cur, 0)
            cur_sum = cur.sum()
            if cur_sum > 0:
                cur /= cur_sum

    return best_seq_local, best_score_local

"""
propose_candidate – search for a non‑negative step function that minimizes
the provided evaluation metric.

Algorithm
---------
1.  Warm‑start from an external sequence `height_sequence_1` if it exists.
2.  Deterministic parametric sweep:
    * Several analytic families (U‑shaped power law, sum‑of‑tails,
      exponential, product‑power‑law, truncated power‑law) are generated
      for a range of lengths (up to 120 000) and a dense grid of
      parameters.
    * The candidate with the smallest *approximate* score
      ``2·n·max(conv)`` (where ``conv = a*a``) is kept as the initial
      solution.
3.  Local refinement:
    * Repeatedly move a small fraction of mass from an index that
      contributes most to the current maximal convolution entry to an
      index that contributes the least.
    * The move is accepted if it improves the score or with a
      temperature‑dependent probability (simulated annealing).
    * Temperature and the maximal transferable fraction decay
      geometrically; occasional random noise is injected to escape
      plateaus.
4.  The best vector found is finally evaluated with the official
    ``evaluate_sequence`` routine and returned as a plain ``list``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, power‑of‑2 size)
# ----------------------------------------------------------------------


def _fft_size(m):
    """Next power of two >= m."""
    return 1 << ((m - 1).bit_length())


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _fft_size(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = _fft_size(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def approx_score(a: np.ndarray) -> float:
    """Approximate objective for a vector normalised to sum = 1."""
    n = a.shape[0]
    return 2.0 * n * max_conv_fft(a)


# ----------------------------------------------------------------------
# Parametric families (return a normalised vector, sum = 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1.0) ** (-p) + 1e-15
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1.0) ** (-p) + (n - i) ** (-p) + 1e-15
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges: a[i] ∝ exp(-α·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-15
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """
    Product‑power‑law shape:
        a[i] ∝ ((i+1)*(n-i))^{‑p}
    (large at the ends, small in the middle).
    """
    i = np.arange(n)
    a = ((i + 1.0) * (n - i)) ** (-p) + 1e-15
    a /= a.sum()
    return a


def truncated_powerlaw_distribution(n: int, p: float, frac: float) -> np.ndarray:
    """
    Truncated symmetric power‑law: mass is placed only on the two
    edge blocks of length L = floor(frac·n) (mirrored).  Inside the
    middle block the entries are zero.  The shape on each edge block
    follows (i+1)^{‑p}.
    """
    L = int(frac * n)
    if L == 0:
        L = 1
    L = min(L, n // 2)        # avoid overlap of the two blocks
    a = np.zeros(n, dtype=np.float64)
    i = np.arange(L)
    vals = (i + 1.0) ** (-p) + 1e-15
    a[:L] = vals
    a[-L:] = vals[::-1]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Local refinement – directed mass‑transfer moves
# ----------------------------------------------------------------------


def _refine_sequence(seq: np.ndarray, deadline: float, target: float = 1.5):
    """Hill‑climbing / simulated annealing on the simplex."""
    n = seq.shape[0]
    cur = seq.copy()
    cur_sum = cur.sum()
    if cur_sum == 0.0:
        cur = np.ones_like(cur) / n
    else:
        cur /= cur_sum

    conv = full_conv_fft(cur)
    cur_score = 2.0 * n * conv.max()
    best = cur.copy()
    best_score = cur_score

    temperature = 1.0
    max_frac = 0.25               # maximal fraction of source mass to move
    min_frac = 1e-4
    it = 0

    print("\n=== Local refinement (directed mass transfer) ===")
    while time.time() < deadline:
        it += 1

        # current maximal convolution entry
        conv = full_conv_fft(cur)
        max_val = conv.max()
        kmax = int(conv.argmax())

        # partner values a[kmax‑i] for each i (zero if out of range)
        idx = np.arange(n)
        partner = np.where(
            (kmax - idx >= 0) & (kmax - idx < n),
            cur[kmax - idx],
            0.0,
        )

        src = int(partner.argmax())
        dst = int(partner.argmin())

        # occasionally perform a random move to increase exploration
        if np.random.rand() < 0.07:
            src = np.random.choice(n, p=cur / cur.sum())
            dst = np.random.randint(0, n - 1)
            if dst >= src:
                dst += 1

        if src == dst:
            continue

        # amount of mass to transfer
        delta = cur[src] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta

        # evaluate candidate (exact convolution)
        cand_score = 2.0 * n * full_conv_fft(cand).max()

        # acceptance test (simulated annealing)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"Iter {it:6d}: new global best = {best_score:.6f}"
                )
                if best_score < target:
                    print(f"Target {target:.3f} reached – stopping refinement.")
                    break

        # temperature / step‑size schedule
        temperature = max(temperature * 0.9995, 1e-6)
        max_frac = max(max_frac * 0.9995, min_frac)

        # occasional jitter to shake out of plateaus
        if it % 250 == 0:
            noise = (np.random.rand(n) - 0.5) * 1e-4 * cur.max()
            cur += noise
            np.clip(cur, 0.0, None, out=cur)
            cur /= cur.sum()
            # reset temperature a little to allow acceptance of small changes
            temperature = max(temperature, 1e-5)

    return best, best_score
