def full_convolution_fft_general(x, y):
    size = len(x) + len(y) - 1
    nfft = 1 << (size - 1).bit_length()
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(fx * fy, n=nfft)
    return conv[:size]

"""
Propose a low‑score step‑function.

The algorithm works in two phases.

1️⃣ Deterministic sweep over a few parametric families (U‑shape and
   symmetric Beta) for several lengths.  The best distribution found
   (including the optional external `height_sequence_1`) is taken as a
   seed.

2️⃣ Continuous refinement via projected gradient descent on a smooth
   approximation of the ℓ∞‑norm of the self‑convolution.  The smooth
   max is realised with a soft‑max (temperature β).  Each iteration
   computes the full convolution and the correlation needed for the
   gradient using a single FFT of length ≈ 3·n (n = sequence length).
   An additive update on the simplex is applied, with a slowly decaying
   step size and a slowly increasing β.  The current best-ever vector
   (according to the true objective) is tracked and returned.

The whole search stops before the supplied time budget expires (≈ 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w ≥ 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Convolution / correlation utilities (FFT based, linear length)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper to get the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(x: np.ndarray, y: np.ndarray, n_fft: int) -> np.ndarray:
    """
    Linear convolution of real vectors x and y using FFT of length n_fft.
    Returns the first len(x)+len(y)-1 entries.
    """
    fx = np.fft.rfft(x, n=n_fft)
    fy = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(fx * fy, n=n_fft)
    return conv[: x.shape[0] + y.shape[0] - 1]


def linear_correlation_fft(w: np.ndarray, a: np.ndarray, n_fft: int) -> np.ndarray:
    """
    Linear correlation r[l] = Σ_k w_k * a_{k+l}  (lags from -(len(a)-1) … len(w)-1).
    Returned array has length len(w)+len(a)-1 and is ordered by increasing lag.
    """
    W = np.fft.rfft(w, n=n_fft)
    A = np.fft.rfft(a, n=n_fft)
    # correlation = ifft(conj(W) * A)
    r = np.fft.irfft(np.conj(W) * A, n=n_fft)
    return r[: w.shape[0] + a.shape[0] - 1]  # linear (non‑circular) part


# ----------------------------------------------------------------------
#  Parametric families (basic shapes)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge: (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete symmetric Beta(α,β) on {0,…,n‑1}.  a[i] ∝ (i+1)^{α‑1} (n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
#  Score helper (exact objective for a normalised vector)
# ----------------------------------------------------------------------


def max_convolution(a: np.ndarray, n_fft: int) -> float:
    """Maximum entry of a*a (a is non‑negative, already normalised)."""
    conv = full_convolution_fft(a, a, n_fft)
    return float(conv.max())


def exact_score(a: np.ndarray, n_fft: int) -> float:
    """Exact value returned by evaluate_sequence for a vector with sum=1."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a, n_fft)


# ----------------------------------------------------------------------
#  Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) discovered before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1. Deterministic sweep over a few families
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # candidate lengths (a compromise between quality and speed)
    lengths = [8000, 12000, 16000, 20000]

    # coarse grids for the two families
    p_grid = np.arange(0.1, 2.01, 0.1)          # for u_min
    alpha_grid = np.arange(0.2, 2.01, 0.2)     # symmetric beta (α=β)

    print("=== Deterministic sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break
        n_fft = _next_pow2(3 * n - 2)

        # ---- U‑shape -------------------------------------------------
        for p in p_grid:
            a = u_min_distribution(n, p)
            sc = exact_score(a, n_fft)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (U‑shape)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # ---- Symmetric Beta -------------------------------------------
        for alpha in alpha_grid:
            a = beta_distribution(n, alpha, alpha)
            sc = exact_score(a, n_fft)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (Beta)     n={n:5d} α={alpha:4.2f}  score={sc:.6f}")

    # --------------------------------------------------------------
    # 2. Incorporate external construction if present
    # --------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            total = external.sum()
            if total > 0.0:
                external /= total
                n = external.shape[0]
                n_fft = _next_pow2(3 * n - 2)
                sc_ext = exact_score(external, n_fft)
                if sc_ext < best_score:
                    best_score, best_seq = sc_ext, external.copy()
                    print(f" external height_sequence_1  n={n:5d}  score={sc_ext:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 3. Fallback (unlikely needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 10000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        n_fft = _next_pow2(3 * n - 2)
        best_score = exact_score(best_seq, n_fft)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Continuous refinement (smooth‑max gradient descent)
    # --------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]
    n_fft = _next_pow2(3 * n - 2)

    # hyper‑parameters
    beta = 5.0                     # start temperature for soft‑max
    beta_inc = 1.01                # multiplicative increase per iteration
    step = 0.25                    # initial additive step size
    step_decay = 0.9995            # multiplicative decay per iteration
    eval_every = 10                # evaluate true score every this many iters
    max_no_imp = 5000              # iterations without improvement before re‑heat

    best_iter = 0
    last_improve = 0
    iteration = 0

    print("\n=== Starting continuous refinement ===")
    while time.time() < deadline:
        iteration += 1

        # ---------- convolution of current with itself ----------
        A = np.fft.rfft(current, n=n_fft)
        conv_full = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
        max_conv = conv_full.max()

        # ---------- smooth max weights ----------
        # numerically stable soft‑max
        exp_vals = np.exp(beta * (conv_full - max_conv))
        w_sum = exp_vals.sum()
        if w_sum == 0.0:
            w = np.full_like(conv_full, 1.0 / conv_full.size)
        else:
            w = exp_vals / w_sum

        # ---------- correlation w ⋆ a (gradient) ----------
        W = np.fft.rfft(w, n=n_fft)
        # correlation = ifft(conj(W) * A)
        corr_full = np.fft.irfft(np.conj(W) * A, n=n_fft)[: 3 * n - 2]

        # gradient: g_i = 2 * r[l = -i]  with lag -i located at index (n-1-i)
        grad = 2.0 * corr_full[n - 1 :: -1]    # first n entries reversed
        # (corr_full[n-1::-1] is clearer but a little slower)

        # ---------- additive projected step ----------
        cand = current - step * grad
        # enforce non‑negativity before projection
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)

        # ---------- update ------------------------------------------------
        current = cand
        step *= step_decay
        beta *= beta_inc

        # ---------- occasional exact evaluation ----------------------------
        if iteration % eval_every == 0:
            # compute true max convolution for the candidate
            n_fft_eval = _next_pow2(2 * n - 1)
            score = exact_score(current, n_fft_eval)
            if score < best_score - 1e-12:
                best_score, best_seq = score, current.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ---------- stagnation handling ------------------------------------
        if iteration - last_improve > max_no_imp:
            # small random jitter + re‑heat
            jitter = np.random.rand(n) * 1e-4
            current = simplex_projection(current + jitter)
            step = 0.25
            beta = 5.0
            last_improve = iteration
            print(f" * Re‑heat at iter {iteration}")

    # ------------------------------------------------------------------
    # 5. Return the best found sequence (raw vector, sum=1)
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator will clip & re‑normalise, so we return the raw vector.
    return best_seq.tolist()
