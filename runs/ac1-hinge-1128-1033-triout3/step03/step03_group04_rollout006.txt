def optimize_distribution(initial_a, n, time_limit):
    # adapt iteration count.
    start = time.time()
    a = initial_a.copy()
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.random.rand(n)
    a /= a.sum()
    best_local_score = np.inf
    best_local_a = a.copy()
    # hyperparameters
    alpha = 5.0
    alpha_max = 300.0
    alpha_growth = 1.005
    lr = 0.2
    lr_decay = 0.9995
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    # Precompute power for Adam
    pow_beta1 = 1.0
    pow_beta2 = 1.0

    while time.time() < time_limit:
        # Compute convolution of a with itself (through FFT)
        size_conv = 1 << ((2*n - 1).bit_length())
        # Use rfft
        fa = np.fft.rfft(a, n=size_conv)
        conv = np.fft.irfft(fa * fa, n=size_conv)[:2*n-1]
        max_c = conv.max()
        # Evaluate score
        score = 2.0 * n * max_c
        if score < best_local_score:
            best_local_score = score
            best_local_a = a.copy()
            # print progress
            print(f"n={n} best local score={best_local_score:.6f} time={time.time():.1f}")

        # softmax weights
        # shift by max to avoid overflow
        shifted = alpha * (conv - max_c)
        # Clip extreme values to avoid overflow on exp: maybe -700 < exponent < 700. Use np.clip
        shifted = np.clip(shifted, -700, 700)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0:
            # Use uniform
            w = np.full_like(w, 1.0/len(w))
        else:
            w /= w_sum

        # Compute gradient via correlation
        # compute FFT size for correlation
        len_w = len(w)
        len_rev = n
        size_corr = 1 << ((len_w + len_rev - 1).bit_length())
        fw = np.fft.rfft(w, n=size_corr)
        frev = np.fft.rfft(a[::-1], n=size_corr)
        conv_rev = np.fft.irfft(fw * frev, n=size_corr)
        # slice gradient
        grad = 2.0 * conv_rev[n-1:n-1+n]

        # Adam moments
        t += 1
        pow_beta1 *= beta1
        pow_beta2 *= beta2
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - pow_beta1)
        v_hat = v / (1 - pow_beta2)
        # compute exponent update
        step = - lr * m_hat / (np.sqrt(v_hat) + eps)
        # Prevent too large steps
        step = np.clip(step, -5, 5)  # limit exponent scale
        # Update a multiplicatively
        a = a * np.exp(step)
        # Ensure positivity
        a = np.maximum(a, 0.0)
        # Renormalise to sum 1
        total = a.sum()
        if total == 0:
            a = np.random.rand(n)
            a /= a.sum()
        else:
            a /= total

        # Update hyperparameters
        alpha = min(alpha * alpha_growth, alpha_max)
        lr *= lr_decay

        # occasionally add random noise to avoid stagnation
        if t % 5000 == 0:
            perturb = np.random.rand(n) * 1e-5
            a += perturb
            a = np.maximum(a, 0.0)
            a /= a.sum()
            # reset Adam moments
            m = np.zeros(n)
            v = np.zeros(n)
            pow_beta1 = pow_beta2 = 1.0

        # Optionally early break if improvement is tiny for many steps

    return best_local_a, best_local_score

"""
Propose a low‑score step function.

Algorithm
---------
1.  Load the best known construction (``height_sequence_1``) if it exists.
2.  For a series of lengths (increasing from a few thousand up to ~30k)
    *   Interpolate the current best sequence to the new length.
    *   Run a smooth‑max Adam / exponentiated‑gradient descent:
        –  The objective is 2·n·max(a*a) (sum of a is kept equal to 1).
        –  A softmax with a temperature (α) gives a differentiable
           approximation of the max‑entry; α is slowly increased.
        –  The gradient of the soft‑max is obtained via a correlation
           (FFT‑based) –  O(n log n) per iteration.
        –  An Adam‑style moment estimate is used, the update is
           multiplicative (mirror‑descent) and the vector is finally
           renormalised (simplex projection is trivial after the
           multiplicative step).
        –  Periodic tiny random perturbations (“re‑heating”) avoid
           stagnation.
    *   Keep the best vector found for the current length and, if it
        improves the global best, store it.
3.  After the allotted time expires return the best vector discovered.
   The returned list is a raw probability vector; ``evaluate_sequence``
   will clip and renormalise it again if needed.

The whole search runs under the 1000 s budget and makes heavy use of
FFT‑based convolutions so that millions of candidates can be examined.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (returns length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 2 * n                     # zero‑pad to avoid circular effects
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def correlate_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the (non‑circular) correlation needed for the gradient:
    grad_k = 2 * Σ_i w_i * a_{i‑k} .
    This equals twice the slice (n‑1 … 2n‑2) of the linear convolution
    of w with the reversed a.
    """
    n = a.shape[0]
    size = w.shape[0] + n - 1        # length of full linear convolution
    fw = np.fft.rfft(w, n=size)
    frev = np.fft.rfft(a[::-1], n=size)
    conv = np.fft.irfft(fw * frev, n=size)
    # slice = conv[n‑1 : n‑1 + n]
    return 2.0 * conv[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Scoring (sum of a is assumed to be 1)
# ----------------------------------------------------------------------


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a probability vector."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_val = conv.max()
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Simple parametric families (used only for occasional random restarts)
# ----------------------------------------------------------------------


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1} (unnormalised)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0))
    a += 1e-14                     # avoid exact zeros
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Interpolation helper – resize a distribution to a new length
# ----------------------------------------------------------------------


def resample_distribution(v: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative vector to length n."""
    m = v.shape[0]
    if m == n:
        return v.copy()
    x_old = np.arange(m)
    x_new = np.linspace(0, m - 1, n)
    new = np.interp(x_new, x_old, v)
    new = np.maximum(new, 0.0)
    total = new.sum()
    if total == 0.0:
        new = np.random.rand(n)
        new /= new.sum()
    else:
        new /= total
    return new


# ----------------------------------------------------------------------
# Core optimiser for a single length
# ----------------------------------------------------------------------


def optimize_distribution(init_a: np.ndarray,
                          max_time: float,
                          verbose: bool = True):
    """
    Smooth‑max Adam / mirror‑descent on a probability vector.
    Returns the best vector and its score found before max_time.
    """
    a = init_a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.random.rand(a.shape[0])
        a /= a.sum()
    else:
        a /= s

    n = a.shape[0]

    # Adam‑style moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    pow_beta1, pow_beta2 = 1.0, 1.0

    # schedules
    alpha = 1.0          # soft‑max temperature
    alpha_max = 300.0
    alpha_growth = 1.001

    lr = 0.2             # base learning‑rate (in the exponent)
    lr_decay = 0.9995

    best_local_a = a.copy()
    best_local_score = np.inf
    iteration = 0
    last_improve_iter = 0

    while time.time() < max_time:
        iteration += 1

        # ----- self‑convolution -------------------------------------------------
        conv = full_convolution_fft(a)
        max_c = conv.max()
        cur_score = 2.0 * n * max_c
        if cur_score < best_local_score - 1e-12:
            best_local_score = cur_score
            best_local_a = a.copy()
            last_improve_iter = iteration
            if verbose:
                print(f" n={n:5d}  iter={iteration:7d}  new best={best_local_score:.6f}")

        # ----- soft‑max weights over the convolution entries --------------------
        shifted = alpha * (conv - max_c)          # max entry gets 0, others ≤0
        # protect against overflow / underflow
        shifted = np.clip(shifted, -700, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ----- gradient of the smooth max (correlation) -------------------------
        grad = correlate_fft(w, a)                # shape (n,)

        # ----- Adam moment updates ---------------------------------------------
        pow_beta1 *= beta1
        pow_beta2 *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - pow_beta1)
        v_hat = v / (1.0 - pow_beta2)

        # ----- exponentiated (mirror‑descent) step -----------------------------
        step = - lr * m_hat / (np.sqrt(v_hat) + eps)
        # limit the exponent to avoid overflow of np.exp
        step = np.clip(step, -5.0, 5.0)
        a = a * np.exp(step)

        # ----- renormalise to the simplex --------------------------------------
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:
            a = np.random.rand(n)
            a /= a.sum()
        else:
            a /= total

        # ----- schedules --------------------------------------------------------
        alpha = min(alpha * alpha_growth, alpha_max)
        lr *= lr_decay

        # ----- occasional re‑heat to escape plateaus ---------------------------
        if iteration - last_improve_iter > 30000:
            perturb = np.random.rand(n) * 1e-5
            a = a + perturb
            a = np.maximum(a, 0.0)
            a /= a.sum()
            # reset Adam state
            m[:] = 0.0
            v[:] = 0.0
            pow_beta1 = pow_beta2 = 1.0
            last_improve_iter = iteration
            if verbose:
                print(f" *** re‑heat at iter {iteration}")

    return best_local_a, best_local_score
