    final_score = evaluate_sequence(best_a.tolist())

"""
Step‑function optimiser.

The objective is                     2·n·max(conv(a,a)) / (sum(a)^2)
with a_i ≥ 0.  Using a soft‑max parametrisation a = softmax(w)
enforces the constraints automatically.  We minimise a smooth
log‑sum‑exp surrogate of the max‑convolution via Adam, start from a
U‑shaped “arcsine” distribution (which is a good analytical guess) and
switch to a pure max‑based sub‑gradient as the smoothing vanishes.
The routine runs until the supplied time budget is exhausted and returns
the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negatives caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

        a0 = np.maximum(a0, 0.0)
