def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time
    import numpy as np
    deadline = time.time() + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # choose n
    n = 8000 (maybe 8000 for good tradeoff)
    # compute FFT length for self convolution
    L_self = 1 << ((2*n - 1).bit_length())
    L_comb = 1 << ((3*n - 2).bit_length())

    # initialization: try Catalan
    coeffs = np.empty(n, dtype=np.float64)
    cur=1.0
    coeffs[0]=cur
    for k in range(1,n):
        cur = cur * (2*k - 1) / (2*k)
        coeffs[k] = cur
    a = coeffs / coeffs.sum()

    # If height_sequence_1 exists, maybe try both
    init_candidates = [a.copy()]

    if 'height_sequence_1' in globals():
        hs1 = np.array(height_sequence_1, dtype=np.float64)
        if hs1.size >= n:
            a1 = hs1[:n]
        else:

    # also random uniform and random simplex

    best_val = np.inf

    # define optimizer function for a given start
    def optimize_start(a0):
        nonlocal best_val, best_seq, deadline
        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999

        patience = 5000
        while time.time() < deadline:
            t += 1
            # compute self conv via FFT
            fa = np.fft.rfft(a, n=L_self)

"""
propose_candidate
-----------------

Search for a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a deterministic‑annealing style optimisation:
  * variables live on the probability simplex (∑a = 1, a_i ≥ 0)
  * the non‑smooth max is approximated by a smooth log‑sum‑exp
    which yields a dense gradient that can be computed by one extra FFT.
  * an Adam‑style update with projection onto the simplex is applied.
  * the temperature (α) of the log‑sum‑exp starts small and is
    exponentially increased, gradually turning the smooth surrogate into the true max.
  * several sensible initialisations (Catalan, uniform, a random simplex,
    and the optional global `height_sequence_1`) are tried; the best sequence
    found before the time budget expires is returned.

The routine respects the 1000 s wall‑clock limit, prints occasional progress
messages and finally calls `evaluate_sequence` to obtain the exact score that
the judge will see.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} using Duchi et al. (2008)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: convolution via FFT (real inputs).
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """Return the linear convolution of a and b using FFT of given size."""
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function (list of heights) that minimises the evaluation
    function.  The algorithm runs a smooth‑approximation Adam optimisation
    on the maximal entry of the self‑convolution, trying several initial
    points and keeping the best solution seen before the deadline.
    """
    # ----- time handling -------------------------------------------------
    deadline = time.time() + budget_s - 5.0          # safety margin
    rng = np.random.default_rng(seed)

    # ----- problem size -------------------------------------------------
    n = 8000                                         # length of the step‑function
    # FFT sizes (powers of two)
    size_self = _next_pow2(2 * n - 1)                # for a*a
    size_comb = _next_pow2(3 * n - 2)                # for w * rev(a)

    # ----- build a few sensible starting points -------------------------
    init_candidates = []

    # 1) Catalan (central‑binomial) coefficients, normalised to sum 1
    coeffs = np.empty(n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, n):
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    a_catalan = coeffs / coeffs.sum()
    init_candidates.append(a_catalan.copy())

    # 2) Uniform simplex point
    a_uniform = np.ones(n, dtype=np.float64) / n
    init_candidates.append(a_uniform.copy())

    # 3) Random simplex point
    a_random = rng.random(n)
    a_random = a_random / a_random.sum()
    init_candidates.append(a_random.copy())

    # 4) Optional global sequence (if supplied by the framework)
    if 'height_sequence_1' in globals():
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size >= n:
            a_hs = hs[:n].copy()
        else:
            a_hs = np.concatenate([hs, np.zeros(n - hs.size)])
        a_hs = np.maximum(a_hs, 0.0)
        if a_hs.sum() < 1e-12:
            a_hs = np.ones(n, dtype=np.float64)
        a_hs = a_hs / a_hs.sum()
        init_candidates.append(a_hs.copy())

    # ----- bookkeeping ---------------------------------------------------
    best_val = np.inf
    best_seq = None

    # ----- core optimiser -----------------------------------------------
    def _optimise_start(a_start: np.ndarray):
        nonlocal best_val, best_seq, deadline

        a = a_start.copy()
        m = np.zeros_like(a)          # first moment (Adam)
        v = np.zeros_like(a)          # second moment (Adam)

        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr = 0.01                     # base learning rate for Adam
        alpha = 5.0                   # start temperature for log‑sum‑exp
        alpha_max = 200.0
        alpha_factor = 1.001          # slow increase
        t = 0
        no_improve = 0
        patience = 5000               # restart if no improvement for this many iters

        while time.time() < deadline:
            t += 1

            # ---- self‑convolution a*a (FFT) --------------------------------
            conv_self = _conv_fft(a, a, size_self)
            max_conv = conv_self.max()

            # ---- smooth surrogate weights (log‑sum‑exp) --------------------
            # w_k ∝ exp(alpha * (conv_k - max_conv))  → sum(w)=1
            shift = conv_self - max_conv
            w = np.exp(alpha * shift)
            w_sum = w.sum()
            if w_sum == 0.0:                     # extreme underflow → fallback
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ---- gradient of the smooth surrogate -------------------------
            rev_a = a[::-1]
            conv_wa = _conv_fft(w, rev_a, size_comb)          # length = 3n‑2
            # slice the portion that corresponds to indices 0 … n‑1
            g = 2.0 * conv_wa[n - 1 : n - 1 + n]

            # ---- Adam update ------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a_new = _project_simplex(a_new)

            # ---- evaluate the true objective (max‑convolution) -------------
            conv_new = _conv_fft(a_new, a_new, size_self)
            max_conv_new = conv_new.max()
            sum_a = a_new.sum()
            if sum_a < 1e-12:
                # avoid division by (almost) zero – unlikely but guard anyway
                continue
            val = 2.0 * n * max_conv_new / (sum_a * sum_a)

            # ---- keep the best seen so far ---------------------------------
            if val < best_val - 1e-12:
                best_val = val
                best_seq = a_new.copy()
                no_improve = 0
                print(f"[iter {t:8d}] new_best={best_val:.12f} "
                      f"alpha={alpha:.3f} lr={lr:.6f}")
            else:
                no_improve += 1

            a = a_new

            # ---- schedule for α (temperature) -------------------------------
            if alpha < alpha_max:
                alpha = min(alpha_max, alpha * alpha_factor)

            # ---- occasional re‑normalisation (numerical safety) -------------
            if t % 5000 == 0:
                a = a / a.sum()

            # ---- restart if stuck -------------------------------------------
            if no_improve >= patience:
                a = rng.random(n)
                a = a / a.sum()
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                no_improve = 0
                alpha = 5.0
                print("[restart] no improvement for a while – re‑initialising.")

    # ----- run optimiser from each starting point -----------------------
    for idx, a0 in enumerate(init_candidates):
        # quick evaluation of the start point (may already be good)
        conv0 = _conv_fft(a0, a0, size_self)
        init_val = 2.0 * n * conv0.max() / (a0.sum() ** 2)
        if init_val < best_val:
            best_val = init_val
            best_seq = a0.copy()
            print(f"[init {idx}] starting_val={best_val:.12f}")

        _optimise_start(a0)

    # ----- final post‑processing -----------------------------------------
    # clamp to the limits enforced by the official evaluator
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(best_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(best_seq)}")
    return best_seq
