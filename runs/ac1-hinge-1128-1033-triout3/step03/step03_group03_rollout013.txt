"""
propose_candidate
-----------------

Implements an Adam‑based projected gradient descent on a *soft‑max*
approximation of the maximal self‑convolution entry.
The variables live on the probability simplex (∑a = 1, a_i ≥ 0);
with this normalisation the evaluation function reduces to

    score = 2 * n * max(conv(a, a))

The algorithm proceeds as follows:

1. Initialise the step‑function with the (truncated) Catalan sequence
   (the classic “constant‑convolution’’ construction).  If a global
   variable ``height_sequence_1`` is present it is used as a warm‑start.

2. At each iteration
   * compute the full self‑convolution,
   * build a soft‑max weight vector `w` from the convolution entries
     (temperature controlled by a slowly increasing `alpha`);
   * compute the gradient of the smoothed objective:
         g_i = 2 * Σ_j w_{i+j} * a_j          (correlation of w and a)
   * perform an Adam update and project back onto the simplex.

3. Keep track of the best sequence seen so far, decay the learning
   rate, increase the soft‑max temperature and occasionally restart
   from a random point when progress stalls.

The whole routine stops a few seconds before the given time budget,
clamps the final coefficients to the limits required by
`evaluate_sequence`, and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} using the algorithm of
    Duchi et al. (2008). Works for 1‑D arrays."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------

    Adam optimiser on a smoothed max‑convolution objective and runs
    until the wall‑clock budget expires.
    """

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    n = 5000                     # length of the step‑function

    # --------------------------------------------------------------

    #    otherwise fall back to the Catalan (constant‑convolution) start.
    # --------------------------------------------------------------

            if init.size > n:
                init = init[:n]
            else:
                init = np.pad(init, (0, n - init.size), constant_values=0.0)
        a = init / init.sum()
        print("[init] warm‑start from height_sequence_1")
    except Exception:
        # Catalan coefficients a_k = C(2k,k) / 4**k
        coeffs = np.empty(n, dtype=np.float64)
        cur = 1.0
        coeffs[0] = cur
        for k in range(1, n):
            cur = cur * (2 * k - 1) / (2 * k)
            coeffs[k] = cur
        a = coeffs / coeffs.sum()
        print("[init] Catalan start")

    # --------------------------------------------------------------
    # 3) Adam hyper‑parameters
    # --------------------------------------------------------------
    lr = 0.08                     # base learning rate
    beta1, beta2 = 0.9, 0.999

    # soft‑max temperature (α) – starts low, grows slowly

    print(f"[state] n={n}, initial_score={best_score:.10f}")

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---- convolution and soft‑max weights ----
        conv = np.convolve(a, a)
        max_val = conv.max()

        # numerically stable soft‑max on the convolution entries
        # w_k ∝ exp(α * (c_k - max_val))
        exp_arg = alpha * (conv - max_val)          # max → 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                            # should not happen
