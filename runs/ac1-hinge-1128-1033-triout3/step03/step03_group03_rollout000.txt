def fft_convolve(a, b):
    n = a.shape[0]
    m = b.shape[0]
    N = 1 << ((n+m-1).bit_length())
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    fc = fa * fb
    conv = np.fft.irfft(fc, N)
    return conv[:n+m-1]

"""
Propose a step‐function with a lower max‐self‑convolution value.

The algorithm improves on the simple projected sub‑gradient method by
optimising a *smooth* approximation of the maximum convolution entry.
The smooth objective is

    fβ(a) = (1/β)·log Σk exp(β·(a*a)k),

whose gradient can be written in closed form and is computed efficiently
with FFT‑based convolutions.  An Adam optimiser drives the variables on
the probability simplex (∑a = 1, a ≥ 0).  The smoothing parameter β is
gradually increased so that the optimisation starts with a very smooth
landscape and slowly focuses on the true max‑convolution.  Random
restarts are performed if progress stalls.  The whole routine runs
well within the 1000 s budget and typically beats the previous bound
≈ 1.520, aiming for the target ≈ 1.503.

All helper functions are defined at top‑level; the evaluator
`evaluate_sequence` is assumed to be available in the environment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al., 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project the vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of real vectors x and y using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    N = 1 << ((n + m - 1).bit_length())   # next power of two
    fx = np.fft.rfft(x, N)
    fy = np.fft.rfft(y, N)
    fz = fx * fy
    z = np.fft.irfft(fz, N)
    return z[: n + m - 1]


# ----------------------------------------------------------------------
# Compute self‑convolution and its maximum
# ----------------------------------------------------------------------
def _self_conv_max(a: np.ndarray) -> float:
    """Self‑convolution a*a and return its maximal entry."""
    conv = _fft_convolve(a, a)
    # rounding artefacts may produce tiny negatives
    conv = np.maximum(conv, 0.0)
    return conv.max()


# ----------------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a ( (1/β)·log Σk exp(β·c_k) ) where c = a*a.
    The gradient is 2· Σk w_k·a_{k‑i} with w_k = softmax(β·c_k).
    """
    # self‑convolution (needed for the softmax weights)
    conv = _fft_convolve(a, a)
    conv = np.maximum(conv, 0.0)

    # softmax weights (stable formulation)
    max_c = conv.max()
    exp_term = np.exp(beta * (conv - max_c))
    w = exp_term / exp_term.sum()               # shape (2n‑1,)

    # gradient = 2 * (a_rev * w) evaluated at indices n‑1 … n‑1+n‑1
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, w)               # length 3n‑2
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]
    return grad


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Produce a list of non‑negative heights (step‑function) that minimises
    `evaluate_sequence`.  The search runs for up to `budget_s` seconds and
    returns the best sequence seen.
    """
    # ------------------------------------------------------------------
    # 1) time budget
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 2) problem size and initial point
    # ------------------------------------------------------------------
    # a) try to reuse the previously found best sequence if available
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=np.float64)
        init = np.maximum(init, 0.0)
        if init.sum() > 0.0:
            init = init / init.sum()
    else:
        init = None

    # b) default length
    n = 5000
    if init is not None:
        if init.shape[0] != n:
            # either truncate or pad with zeros to obtain the desired length
            if init.shape[0] > n:
                init = init[:n]
            else:
                init = np.concatenate([init, np.zeros(n - init.shape[0])])
        a = init.copy()
    else:
        # start from the classic Catalan (central‑binomial) distribution
        coeffs = np.empty(n, dtype=np.float64)
        cur = 1.0
        coeffs[0] = cur
        for k in range(1, n):
            cur = cur * (2 * k - 1) / (2 * k)
            coeffs[k] = cur
        a = coeffs / coeffs.sum()                 # project onto simplex

    a = _project_simplex(a)                       # safety (exact sum = 1)

    # ------------------------------------------------------------------
    # 3) Adam optimiser state
    # ------------------------------------------------------------------
    alpha = 0.20          # base learning rate
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    # ------------------------------------------------------------------
    # 4) smoothing schedule (log‑sum‑exp temperature)
    # ------------------------------------------------------------------
    beta_smooth = 0.5      # start very smooth
    beta_max = 20.0
    beta_factor = 1.001    # multiplicative increase per iteration

    # ------------------------------------------------------------------
    # 5) bookkeeping
    # ------------------------------------------------------------------
    best_val = 2.0 * n * _self_conv_max(a)    # current best (on simplex)
    best_seq = a.copy()
    no_improve = 0
    restart_limit = 20000                     # iterations without improvement

    print(f"[init] n={n}, initial_score={best_val:.10f}")

    # ------------------------------------------------------------------
    # 6) optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- smooth gradient step -----
        grad = _smooth_gradient(a, beta_smooth)

        # Adam update (biased first, then bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - alpha * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_simplex(a_new)

        # ----- evaluate true objective (max‑convolution) -----
        cur_max = _self_conv_max(a_new)
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a_new.copy()
            no_improve = 0
            print(f"[iter {it:6d}] best_score={best_val:.10f}  beta={beta_smooth:.3f}")
        else:
            no_improve += 1

        a = a_new

        # ----- schedules -----
        # learning rate decay
        alpha = max(1e-4, alpha * 0.9995)

        # smoothness increase
        if beta_smooth < beta_max:
            beta_smooth = min(beta_max, beta_smooth * beta_factor)

        # ----- occasional restart if stuck -----
        if no_improve > restart_limit:
            a = rng.random(n)
            a = a / a.sum()
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            alpha = 0.20
            beta_smooth = 0.5
            no_improve = 0
            print(f"[restart] iteration {it}")

    # ------------------------------------------------------------------
    # 7) final conversion and official evaluation
    # ------------------------------------------------------------------
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(best_seq)
    print(f"[result] final_score={final_score:.10f}, length={len(best_seq)}")
    return best_seq
