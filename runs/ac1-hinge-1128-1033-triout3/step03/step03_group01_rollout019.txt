def adam_optimize(initial, max_iter, deadline, n):
    a = initial.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    best_a = a.copy()
    best_score = score_from_vector(a)
    step_num = 0
    last_improve = 0
    while time.time() < deadline:
        step_num += 1
        # compute gradient of objective
        conv = full_convolution_fft(a)
        grad = compute_subgradient(a, conv)  # subgradient of max_conv
        # we multiply by 2*n? Not needed for direction as scaling influences step size.
        # update m, v
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**step_num)
        v_hat = v / (1 - beta2**step_num)

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises the evaluation score

    2 * n * max_k (a * a)[k]

where * denotes the full linear convolution.  The routine consists of
four stages:

1.  A fairly dense deterministic sweep over many parametric families.
    In addition to the three families used in the reference solution
    (inverse‑distance, product‑inverse‑distance and symmetric
    exponential) we also explore monotone power‑law families,
    two‑parameter “beta‑like’’ families and a combined power‑law /
    exponential family.  This yields a high‑quality seed.

2.  If an external construction `height_sequence_1` is present we test
    it as a possible seed.

3.  Stochastic refinement based on a projected Adam optimiser
    (first‑moment / second‑moment estimates) on the simplex.  The
    optimiser follows the (sub‑)gradient of the maximal convolution
    entry, projects onto the probability simplex after each step and
    uses a simulated‑annealing acceptance rule to escape shallow
    local minima.  Periodic “re‑heat’’ moves are performed when the
    optimiser stagnates for a long stretch.

4.  The best sequence found (as a plain Python list) is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex { x >= 0 , sum(x) = 1 }.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact value returned by ``evaluate_sequence`` for a vector that
    already satisfies sum(a) == 1 (the denominator becomes 1).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices where the convolution attains (or almost attains) its
    maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

# Parametric families

def u_min_distribution(n: int, p: float) -> np.ndarray:
    """Edge‑heavy “U‑shape’’ based on the distance to the nearest edge."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """Edge‑heavy shape based on the product of distances to both edges."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def power_left_distribution(n: int, p: float) -> np.ndarray:
    """Monotone increasing power‑law a[i] ∝ (i+1)^p, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + 1e-14
    a /= a.sum()
    return a


def power_right_distribution(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law a[i] ∝ (n-i)^p, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Two‑parameter Beta‑like family:
        a[i] ∝ (i+1)^α * (n-i)^β .
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** alpha) * ((n - i + 1) ** beta) + 1e-14
    a /= a.sum()
    return a


def power_exp_distribution(n: int, p: float, lam: float) -> np.ndarray:
    """
    Combination of a power‑law (edge distance) and an exponential term.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) * np.exp(-lam * d) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------


def adam_refine(initial: np.ndarray,
                deadline: float,
                max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector using a projected Adam optimiser.
    Returns the best vector seen and its score.
    """
    a = initial.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5                # initial learning‑rate
    iteration = 0
    best_a = a.copy()
    best_score = score_from_vector(a)
    last_improve = 0

    # simple SA schedule
    temperature = 1.0
    temp_decay = 0.9999

    while time.time() < deadline and iteration < max_iter:
        iteration += 1

        # ---- compute sub‑gradient of the current max‑conv entry
        conv = full_convolution_fft(a)
        grad = compute_subgradient(a, conv)

        # ---- Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- gradient descent candidate (projected on simplex)
        cand = a - step
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)
        cand /= cand.sum()

        # ---- evaluate
        cand_score = score_from_vector(cand)

        # ---- SA acceptance
        accept = False
        cur_score = score_from_vector(a)
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best = {best_score:.6f}")

        # ---- schedule updates
        temperature *= temp_decay
        lr *= 0.9999          # gentle decay

        # ---- stagnation handling
        if iteration - last_improve > 200_000:
            # re‑heat: tiny random bump + reset optimiser state
            bump = np.random.rand(n) * 1e-5
            a = simplex_projection(a + bump)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            temperature = 1.0
            lr = 0.5
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for the best non‑negative height sequence within the given
    time budget.  Returns the sequence as a list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    best_seq: np.ndarray | None = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over many families
    # ------------------------------------------------------------------
    print("=== Deterministic sweep start ===")
    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000]

    # power‑law exponents for families that use a single scalar
    exponents = np.arange(0.1, 6.01, 0.1)

    # extended range for the monotone families (they can tolerate larger p)
    exponents_extended = np.arange(0.1, 12.01, 0.2)

    # exponential decay parameters (log‑spaced)
    lam_vals = np.logspace(-4, -0.3, num=40)   # from 1e‑4 to ~0.5

    # beta‑like parameters (two‑parameter grid)
    beta_params = np.arange(0.1, 4.1, 0.5)

    for n in lengths:
        if time.time() > deadline:
            break
        # --- symmetric families (same as reference) -----------------
        for p in exponents:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (exp)     n={n:5d} λ={lam:.5f}  score={sc:.6f}")

        # --- monotone power‑law families --------------------------------
        for p in exponents_extended:
            a = power_left_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (powL)    n={n:5d} p={p:5.2f}  score={sc:.6f}")

            a = power_right_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (powR)    n={n:5d} p={p:5.2f}  score={sc:.6f}")

        # --- beta‑like two‑parameter families -------------------------
        for alpha in beta_params:
            for beta in beta_params:
                a = beta_distribution(n, alpha, beta)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f" new best (beta)    n={n:5d} α={alpha:3.1f} β={beta:3.1f}  score={sc:.6f}")

        # --- combined power‑law / exponential family --------------------
        for p in np.arange(0.5, 5.1, 0.5):
            for lam in np.logspace(-3, -0.5, num=10):
                a = power_exp_distribution(n, p, lam)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f" new best (p*e)    n={n:5d} p={p:3.1f} λ={lam:.4f}  score={sc:.6f}")

        # occasional time check
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. Randomised search over the same families (exploration)
    # ------------------------------------------------------------------
    print("\n=== Randomised exploration start ===")
    random_families = ['u_min', 'u_prod', 'exp', 'powL', 'powR', 'beta', 'pexp']
    while time.time() < deadline:
        n = np.random.choice(lengths)
        fam = np.random.choice(random_families)
        if fam == 'u_min':
            p = 10 ** np.random.uniform(-2, 1)               # ~0.01 – 10
            a = u_min_distribution(n, p)
        elif fam == 'u_prod':
            p = 10 ** np.random.uniform(-2, 1)
            a = u_prod_distribution(n, p)
        elif fam == 'exp':
            lam = 10 ** np.random.uniform(-5, -0.2)
            a = exp_distribution(n, lam)
        elif fam == 'powL':
            p = 10 ** np.random.uniform(-2, 1)
            a = power_left_distribution(n, p)
        elif fam == 'powR':
            p = 10 ** np.random.uniform(-2, 1)
            a = power_right_distribution(n, p)
        elif fam == 'beta':
            alpha = 10 ** np.random.uniform(-1, 1)          # 0.1 – 10
            beta = 10 ** np.random.uniform(-1, 1)
            a = beta_distribution(n, alpha, beta)
        else:  # combined power‑exp
            p = 10 ** np.random.uniform(-1, 0.5)            # 0.1 – 3
            lam = 10 ** np.random.uniform(-4, -0.3)
            a = power_exp_distribution(n, p, lam)

        sc = score_from_vector(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f" random best  fam={fam:5s} n={n:5d}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Incorporate an external construction if available
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0.0:
                external /= s
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score, best_seq = sc, external.copy()
                    print(f" using external height_sequence_1  score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Stochastic refinement (projected Adam + SA)
    # ------------------------------------------------------------------
    print("\n=== Stochastic refinement start ===")
    best_seq, best_score = adam_refine(best_seq, deadline)

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
