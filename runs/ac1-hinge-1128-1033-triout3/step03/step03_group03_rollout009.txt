"""
Exponentiated‑gradient descent on the probability simplex.

The objective is

    2 * n * max(conv(a, a)) / (sum(a)**2)

with a_i ≥ 0.  The value is scale‑invariant, therefore we keep
∑ a_i = 1 (simplex) at all times.  The algorithm works directly on the
simplex using a mirror‑descent update (exponentiated‑gradient).  At each
iteration the sub‑gradient of the maximal convolution entry is taken,
the update is performed as

    a_i ← a_i * exp( - η (g_i - ⟨a, g⟩) )

and the vector is renormalised.  η (step size) follows a 1/√t schedule.
If no improvement is observed for a few thousand iterations the current
 iterate is jittered (small Gaussian noise) to escape possible plateaus.
The routine tries a few sensible initialisations (Catalan coefficients,
discrete arcsine law and uniform) and keeps the best one found before the
time budget expires.  The final sequence is clamped to the limits required
by `evaluate_sequence`.
"""

import time
from typing import List

import numpy as np


def _catalan_initial(n: int) -> np.ndarray:
    """Catalan‑like coefficients a_k = C(2k,k)/4**k, normalised to sum 1."""
    coeffs = np.empty(n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, n):
        cur = cur * (2 * k - 1) / (2 * k)  # (2k‑1)/(2k) recurrence
        coeffs[k] = cur
    coeffs /= coeffs.sum()
    return coeffs


def _arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine law: a_i ∝ 1/sqrt((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n - 0.5 - i))
    a /= a.sum()
    return a


def _uniform_initial(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _subgradient_max_conv(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal self‑convolution entry at position ``idx``.
    The gradient is 2 * a shifted by ``idx`` (zero‑padded outside the range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, idx - (n - 1))
    hi = min(n - 1, idx)
    i_range = np.arange(lo, hi + 1, dtype=np.int64)
    j = idx - i_range
    g[i_range] = 2.0 * a[j]
    return g


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The method uses exponentiated‑gradient descent on the simplex,
    occasional jittering to escape stagnation and three simple initialisers.
    """
    # ------------------------------------------------------------------
    # 0)  Time budget
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0  # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1)  Problem size
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 2)  Choose the best of a few deterministic starts
    # ------------------------------------------------------------------
    starters = [
        _catalan_initial(n),
        _arcsine_initial(n),
        _uniform_initial(n),
    ]

    best_val = np.inf
    best_seq = None
    for start in starters:
        conv = np.convolve(start, start)
        val = 2.0 * n * conv.max() / (np.sum(start) ** 2)
        if val < best_val:
            best_val = val
            best_seq = start.copy()
    # ``a`` will hold the current iterate
    a = best_seq.copy()

    print(f"[init] n={n}, start_score={best_val:.10f}")

    # ------------------------------------------------------------------
    # 3)  Optimisation parameters
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0               # counter for stagnation
    jitter_thr = 3000            # iterations without improvement before jitter
    eta0 = 0.5                    # base step size for the 1/√t schedule

    # ------------------------------------------------------------------
    # 4)  Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---- compute convolution and locate a maximal entry ------------
        conv = np.convolve(a, a)
        max_val = conv.max()
        # pick a random index among (numerically) maximal positions
        max_idxs = np.flatnonzero(conv >= max_val - 1e-14)
        idx = int(rng.choice(max_idxs))

        # ---- sub‑gradient of the chosen entry -------------------------
        g = _subgradient_max_conv(a, idx)

        # expectation of the gradient under the current distribution
        G = np.dot(a, g)

        # ---- exponentiated‑gradient update -------------------------------
        eta = eta0 / np.sqrt(it)            # classic 1/√t schedule
        # factor = exp(-η (g - G))  (element‑wise)
        factor = np.exp(-eta * (g - G))
        a = a * factor
        # numerical safety – avoid exact zeros
        a = np.maximum(a, 1e-16)
        a /= a.sum()

        # ---- evaluate new candidate ------------------------------------
        conv_new = np.convolve(a, a)
        val = 2.0 * n * conv_new.max() / (np.sum(a) ** 2)   # ‑ sum(a) is 1

        # ---- record improvement ----------------------------------------
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            no_improve = 0
            print(f"[iter {it:6d}] score={best_val:.10f}  step={eta:.6e}")
        else:
            no_improve += 1

        # ---- occasional jitter to escape plateaus -----------------------
        if no_improve >= jitter_thr:
            # perturb the best solution a little and continue from there
            pert = 0.02 * rng.standard_normal(n)   # 2 % Gaussian noise
            a = best_seq * (1.0 + pert)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0
            print(f"[jitter] iteration {it}")

        # ---- very rare full restart (prevents pathological drift) ------
        if it % 200_000 == 0:
            a = rng.random(n)
            a /= a.sum()
            print(f"[restart] iteration {it}")

    # ------------------------------------------------------------------
    # 5)  Return the best sequence (clamped to the evaluator limits)
    # ------------------------------------------------------------------
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(best_seq)  # final official evaluation
    print(
        f"[result] final_score={final_score:.10f}, "
        f"iterations={it}, length={len(best_seq)}"
    )
    return best_seq
