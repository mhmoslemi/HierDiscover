def solve_lp_for_g(f, M):
    n = len(f)
    # Build constraints: conv(f, g) <= M  for each k (0..2n-2)
    # Represent using sparse matrix A where A[k, j] = f[k - j] if 0 <= k - j < n else 0.
    # Use scipy.sparse.csr_matrix.
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # for j in [lo..hi], i = k - j
        # A[k, j] = f[i]
        i_range = range(k - hi, k - lo + 1)  # but simpler: for j in lo..hi: i = k - j
        for j in range(lo, hi+1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    # Build csr
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    # Bounds: g_j >= 0 (lower bound 0, no upper bound)
    bounds = [(0, None)] * n
    # constraints: A g <= M
    # Use scipy.optimize.linprog with method='highs' (which uses HiGHS for sparse)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    # Return solution g (if success)
    return res.x

"""
Propose a step‑function (vector of non‑negative heights) that minimises
evaluate_sequence().  

The algorithm is a two‑phase optimiser:
  1.  Adam on a soft‑max parametrisation of the coefficients, minimising a
      smooth log‑sum‑exp approximation of the max‑convolution.  The smoothing
      (ε) decays exponentially to a very small value (≈1e‑12) so the final
      updates are essentially sub‑gradient steps on the true max.
      Learning‑rate and Adam parameters also decay exponentially.
  2.  A short hill‑climbing / simulated‑annealing refinement that repeatedly
      transfers a tiny amount of mass between two random indices.  Each move
      is accepted only if it does not increase the exact objective
      (or with a tiny probability that vanishes towards the deadline).

Both phases keep the vector on the probability simplex (a_i ≥ 0,
∑a_i = 1) – the objective is invariant under scaling, so this normalisation
does not affect the true score.  The best exact value seen (computed with
the original evaluate_sequence formula) is remembered and finally returned.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)               # clip spurious negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = rfft(x, size)
    fy = rfft(y, size)
    conv = irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv
