"""
Propose a non‑negative step‑function (vector) that minimises the metric
implemented by ``evaluate_sequence``.

Algorithm (high‑level)
----------------------
* The search works on a probability simplex (∑a_i = 1) – the objective
  (2·n·max(conv)) is then exactly the value returned by
  ``evaluate_sequence`` (up to the clamping that the evaluator applies).

* For each chosen length n we start from a “seed” distribution:
  – an externally supplied best construction (``height_sequence_1``) if it
    exists, otherwise a discretised arcsine law (∝ 1/√(x(1−x))) which puts
    more mass near the ends.
  – the seed is resampled to the current length.

* The objective ``max(a*a)`` is smooth‑approximated by the log‑sum‑exp
  (soft‑max) of the full self‑convolution:
      S_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k) .
  As α → ∞ the soft‑max converges to the true maximum.

* We minimise S_α using the Adam optimiser on the gradient obtained
  via the convolution theorem (three FFTs per iteration).  After each
  Adam update the vector is projected back onto the probability simplex.

* The smoothing parameter α is increased slowly (α←α·1.005) so the bound
  tightens while optimisation proceeds.

* If the optimisation stalls for a while we inject a small random
  perturbation (to escape possible local minima) and reset Adam’s
  moments.

* The best vector encountered across all lengths (and across the whole
  run) is finally evaluated with ``evaluate_sequence`` and returned.

The routine respects the 1000 s budget, prints progress information,
and uses only NumPy – no file or network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n−1) using supplied FFT size."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_k = exp(α·conv_k) / Σ exp(α·conv)."""
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the self‑convolution.
    Implements ∂S/∂a = Σ_k w_k ∂(a*a)_k/∂a using FFTs.
    """
    n = a.shape[0]
    L = 2 * n - 1  # length of w / conv
    # FFT of a
    A = np.fft.rfft(a, n=size)

    # FFT of w and reversed w
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # Convolution a * w  (length n + L − 1 = 3·n−2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    idx = L - 1 - np.arange(n)          # indices for the reversed part
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x_i ≥ 0, Σx_i = 1}.
    O(n log n) implementation (Wang & Carreira‑Perpiñán, 2013).
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation resampling of a normalised distribution."""
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new += 1e-14                # avoid exact zeros
    new /= new.sum()
    return new


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Discretised arcsine law on {0,…,n‑1}:
        a_i ∝ 1 / sqrt( x_i·(1−x_i) ),  x_i = (i+0.5)/n .
    The distribution is symmetric and heavily weighted near the edges.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                      # points in (0,1)
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a += 1e-14
    a /= a.sum()
    return a

def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the metric used by
    ``evaluate_sequence``.  The function respects a time budget (seconds)
    and returns the best list of floats found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ----------------------------------------------------
    # 0.  Load external seed if it exists
    # ----------------------------------------------------
    best_seq = None
    best_score = np.inf
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size > 0:
            external = np.maximum(external, 1e-12)
            external /= external.sum()
            cur_score = 2.0 * external.size * max_conv_fft(external)
            best_seq = external.copy()
            best_score = cur_score
            print(f"Seeded from height_sequence_1: n={external.size}, "
                  f"score≈{cur_score:.6f}")
    except NameError:
        pass

    # ----------------------------------------------------
    # 1.  Fallback initialisation (arcsine law)
    # ----------------------------------------------------
    if best_seq is None:
        n0 = 8000
        best_seq = arcsine_distribution(n0)
        best_score = 2.0 * n0 * max_conv_fft(best_seq)
        print(f"Fallback initialisation (arcsine): n={n0}, "
              f"score≈{best_score:.6f}")

    # ----------------------------------------------------
    # 2.  Lengths to explore – larger n inflates the factor 2·n
    # ----------------------------------------------------
    lengths_to_try = [8000, 12000, 20000, 30000]   # adapt if time runs out
    lengths_to_try = sorted(set(lengths_to_try))

    global_best_seq = best_seq.copy()
    global_best_score = best_score

    # ----------------------------------------------------
    # 3.  Optimisation loop over lengths
    # ----------------------------------------------------
    for n in lengths_to_try:
        if time.time() >= deadline:
            break

        # ----- initialise vector for this length -----
        if global_best_seq.shape[0] != n:
            a = resample_distribution(global_best_seq, n)
        else:
            a = global_best_seq.copy()
        a = np.maximum(a, 1e-12)
        a /= a.sum()   # ensure we stay on the simplex

        # ----- Adam state -----
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr = 0.5                     # base learning rate for Adam
        alpha = 1.0                  # start with a soft smoothing
        alpha_max = 250.0
        alpha_mul = 1.005
        max_iters = 40000            # generous; actual stop is time‑based
        last_improve = 0
        best_local_score = np.inf
        best_local_a = a.copy()

        # FFT size needed for gradient (≥ 3·n−2)
        fft_size = 1 << ((3 * n - 2).bit_length())

        print(f"\n=== Optimising length n={n} (FFT size={fft_size}) ===")

        it = 0
        while time.time() < deadline and it < max_iters:
            it += 1

            # ----- exact convolution and current objective -----
            conv = full_conv_fft(a, fft_size)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # ----- update best if improved -----
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_a = a.copy()
                last_improve = it
                if cur_score < global_best_score:
                    global_best_score = cur_score
                    global_best_seq = a.copy()
                    print(f"Iter {it:5d} – new global best: score≈{cur_score:.6f}")

                # early exit if target already reached
                if global_best_score < 1.503:
                    print("Target threshold reached – terminating early.")
                    break

            # ----- smooth objective (soft‑max) -----
            w = softmax_weights(conv, alpha)

            # ----- gradient of soft‑max w.r.t. a -----
            grad = gradient_softmax(a, w, fft_size)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- keep feasibility -----
            a = np.maximum(a, 1e-12)            # positivity
            a = project_to_simplex(a)           # enforce Σa = 1

            # ----- schedule -----
            alpha = min(alpha * alpha_mul, alpha_max)

            # ----- stagnation handling -----
            if it % 2000 == 0:
                # if no improvement in the last 2000 iterations, jitter
                if it - last_improve > 2000:
                    noise = np.random.randn(n) * 0.01
                    a = a + noise
                    a = np.maximum(a, 1e-12)
                    a /= a.sum()
                    m.fill(0.0)
                    v.fill(0.0)
                    print(f"Iter {it:5d}: stagnation – random perturbation")
                    last_improve = it   # reset counter

        # store the best vector found for this length as seed for larger n
        global_best_seq = best_local_a.copy()
        global_best_score = best_local_score
        print(f"Finished n={n}: best local score {best_local_score:.6f}")

        # stop completely if we already hit the target
        if global_best_score < 1.503:
            break

    # ----------------------------------------------------
    # 4.  Final official evaluation (if the function is available)
    # ----------------------------------------------------
    try:
        exact_score = evaluate_sequence(global_best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
        global_best_score = exact_score
    except Exception as exc:
        print(f"Could not call evaluate_sequence (will return approximate): {exc}")

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {global_best_score:.6f}")
    print(f"Best length          : {global_best_seq.shape[0]}")
    return global_best_seq.tolist()
