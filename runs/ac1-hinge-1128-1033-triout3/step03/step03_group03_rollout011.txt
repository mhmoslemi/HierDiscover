"""
Propose a step‑function with a low evaluation score.

The algorithm works in two stages within a 1000 s time budget:

1. **Smoothed Adam descent** – a differentiable approximation of the
   `max` in the objective is built with softmax weights
   `w_k ∝ exp(β·(conv_k−max(conv)))`.  The gradient of the weighted
   convolution is obtained by a second convolution
   `g = 2·(w * a_rev)` (where `a_rev` is `a` reversed).  An Adam‑type
   update with projection onto the probability simplex moves the
   coefficients.

2. **Hard‑max sub‑gradient descent** – the classic sub‑gradient step
   using a (random) index where the convolution attains its maximum.
   This refines the solution found by the smoothed phase and includes
   occasional random restarts to escape plateaus.

Both phases keep track of the best vector seen so far.  The final
sequence is clipped to the limits enforced by `evaluate_sequence`
and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} using the algorithm of
    Duchi et al. (2008).  Works for 1‑D arrays.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.arange(n) + 1
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  A two‑phase optimisation (smoothed Adam
    descent + hard‑max sub‑gradient descent) is performed on a vector
    of length `n` (default 5000, or the length of `height_sequence_1`
    if it exists).  The best sequence found before the deadline is
    returned.
    """
    # ------------------------------------------------------------------
    # 0) General set‑up.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0          # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Determine problem size and initialise a.
    # --------------------------------------------------------------
    try:
        # global variable supplied by the harness (may not exist)
        init_seq = np.array(height_sequence_1, dtype=float)  # type: ignore
        n = init_seq.size
        a = np.maximum(init_seq, 0.0)
    except Exception:
        # fallback: Catalan‑type coefficients (central binomial / 4**k)
        n = 5000
        coeffs = np.empty(n, dtype=float)
        cur = 1.0
        coeffs[0] = cur
        for k in range(1, n):
            cur = cur * (2 * k - 1) / (2 * k)          # recurrence
            coeffs[k] = cur
        a = coeffs

    # Project onto the simplex (∑a = 1, a_i ≥ 0)
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / a.sum()
    a = _project_simplex(a)

    # ----------------------------------------------------------------
    # Helper to evaluate the current vector (sum is 1 ⇒ objective = 2·n·max(conv)).
    # ----------------------------------------------------------------
    def _objective(conv_arr: np.ndarray) -> float:
        return 2.0 * n * conv_arr.max()

    # ----------------------------------------------------------------
    # Initial evaluation.
    # ----------------------------------------------------------------
    conv = np.convolve(a, a)                 # full convolution, length 2n-1
    best_val = _objective(conv)
    best_seq = a.copy()
    print(f"[init] n={n}, initial_score={best_val:.10f}")

    # ==============================================================
    # 2) Phase 1 – Smoothed Adam descent (≈ 40 % of the time budget).
    # ==============================================================

    phase1_deadline = time.time() + (deadline - time.time()) * 0.40
    # Adam hyper‑parameters
    lr = 0.08
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    it = 0
    smooth_beta_start = 0.5
    smooth_beta_inc = 0.001
    smooth_beta_cap = 30.0

    while time.time() < phase1_deadline:
        it += 1

        # ----- (a) current convolution and soft‑max weights -----
        # conv already holds the convolution of the current `a`.
        max_conv = conv.max()
        smooth_beta = min(smooth_beta_cap,
                          smooth_beta_start + it * smooth_beta_inc)
        # stable soft‑max
        w = np.exp(smooth_beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # ----- (b) gradient of the smoothed objective ----------
        a_rev = a[::-1]
        # weighted convolution w * a_rev  (full length = 3n‑2)
        conv_wa = np.convolve(w, a_rev)
        start = n - 1
        g = 2.0 * conv_wa[start:start + n]   # shape (n,)

        # ----- (c) Adam update ---------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # ----- (d) evaluate new point ---------------------------
        conv_new = np.convolve(a_new, a_new)
        val = _objective(conv_new)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a_new.copy()
            print(f"[phase1 {it:6d}] score={best_val:.10f}")

        # ----- (e) prepare for next iteration -------------------
        a = a_new
        conv = conv_new

        # ------ simple learning‑rate decay ----------------------
        lr *= 0.9999

        # ------ occasional jitter to avoid stagnation -----------
        if it % 2000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = _project_simplex(a + noise)
            conv = np.convolve(a, a)

    # ==============================================================
    # 3) Phase 2 – Hard‑max projected sub‑gradient descent.
    # ==============================================================

    eta = 0.05                     # initial step size
    min_eta = 1e-6
    decay = 0.9998
    no_improve = 0
    restart_threshold = 2000

    while time.time() < deadline:
        # conv already holds the convolution of the current `a`.
        max_val = conv.max()
        # Indices where the convolution is (almost) maximal.
        max_idxs = np.flatnonzero(conv >= max_val - 1e-12)
        idx = int(rng.choice(max_idxs))

        # Gradient of the selected maximal entry.
        g = np.zeros_like(a)
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=int)
        j = idx - i_range
        g[i_range] = 2.0 * a[j]

        # Gradient step + projection.
        a_new = a - eta * g
        a_new = _project_simplex(a_new)

        # Evaluate.
        conv_new = np.convolve(a_new, a_new)
        val = _objective(conv_new)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a_new.copy()
            no_improve = 0
            a = a_new
            conv = conv_new
            print(f"[phase2 {it:6d}] score={best_val:.10f}  step={eta:.6f}")
        else:
            a = a_new
            conv = conv_new
            no_improve += 1

        # Decay the step size.
        eta = max(min_eta, eta * decay)

        # Re‑normalise every so often (numerical drift).
        if (it + 1) % 1000 == 0:
            a = a / a.sum()
            conv = np.convolve(a, a)

        # Random restart if we have been stuck too long.
        if no_improve > restart_threshold:
            a = rng.random(n)
            a = a / a.sum()
            conv = np.convolve(a, a)
            eta = 0.05
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[restart] iteration {it}")

        it += 1

    # -----------------------------------------------------------------
    # 4) Return the best sequence (clamped to the limits of the evaluator).
    # -----------------------------------------------------------------
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(best_seq)
    print(f"[result] final_score={final_score:.10f}, length={len(best_seq)}")
    return best_seq
