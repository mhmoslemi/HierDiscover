"""
Adam‑based smoothed max‑convolution minimisation.

The objective to minimise is
    F(a) = 2·n·max(a⋆a)      with   a_i ≥ 0 ,  Σ a_i = 1 .
(Scaling of a does not change the value, therefore we keep the vector on the
probability simplex.)

We minimise a smooth surrogate
    F_τ(a) = 2·n·τ·log Σ_k exp((a⋆a)[k]/τ)
using an Adam optimiser, a temperature τ that is linearly reduced
(high τ ⇒ smooth loss, low τ ⇒ closer to the true max), and occasional
random “mass‑transfer’’ jumps.  After each update we project back onto the
simplex and enforce symmetry a_i = a_{n‑1‑i} (the problem is symmetric,
so this reduces the search space).  The routine runs for the given time
budget and returns the best sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σ x = 1} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full linear convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of Σ_k w_k·(a⋆a)[k] with respect to a.
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented via a single convolution: reverse w, convolve with a,
    then pick the appropriate slice.
    """
    n = a.shape[0]                     # length of a
    # reverse the weight vector (length = 2n-1)
    u = w[::-1]
    # full convolution u * a  -> length 3n‑2
    conv = _full_convolution_fft(u, a)
    # indices: start = 2n‑2, then go backwards n steps
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Adam‑based optimisation of the max‑auto‑convolution objective.
    Returns the best non‑negative step‑function found before the time
    budget expires (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the outer framework
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a length n and an initial vector a
    # ------------------------------------------------------------------
    try:
        # Use the previously discovered construction if available.
        init_seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init_seq.shape[0]
        a = init_seq.copy()
    except Exception:
        # modestly large, edge‑heavy shape
        n = 1742                     # length observed in the previous run
        x = np.arange(n, dtype=float)
        a = 1.0 - np.sin(np.pi * x / (n - 1))
        a = np.maximum(a, 0.0)

    # Normalise onto the probability simplex and enforce symmetry
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    a = (a + a[::-1]) * 0.5
    a = _project_onto_simplex(a)

    # ------------------------------------------------------------------
    # Adam bookkeeping
    # ------------------------------------------------------------------
    beta1 = 0.90
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros_like(a)           # first moment
    v = np.zeros_like(a)           # second moment

    # hyper‑parameters (linear schedules)
    eta_start = 0.30   # initial step size
    eta_end   = 5e-6   # final step size

    # we will set the temperature schedule after the first convolution
    conv = _full_convolution_fft(a, a)
    M0 = conv.max()
    tau_start = max(M0 * 0.25, 1e-12)   # fairly high temperature
    tau_end   = max(M0 * 1e-6, 1e-12)   # very low temperature (≈ hard max)

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_val = float('inf')
    iteration = 0
    last_improve_iter = 0
    jitter_prob = 0.02

    print(f"[init] n={n}, initial max‑conv={M0:.6e}")

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- full autocorrelation -----
        conv = _full_convolution_fft(a, a)   # length 2n‑1
        M = conv.max()
        cur_val = float(2.0 * n * M)         # Σa = 1  ⇒  evaluate ≈ 2·n·max

        # keep best so far
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve_iter = iteration
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # progress (0 … 1) used for schedules
        progress = (now - start_time) / (deadline - start_time)

        # ----- temperature (soft‑max smoothing) -----
        tau = tau_start * (1.0 - progress) + tau_end * progress
        if tau > 0:
            # compute stable soft‑max weights ∝ exp((conv−M)/τ)
            exp_args = (conv - M) / tau   # ≤ 0, max at 0
            exp_vals = np.exp(exp_args)   # ≤ 1
            sum_exp = exp_vals.sum()
            if sum_exp > 0 and np.isfinite(sum_exp):
                w = exp_vals / sum_exp
            else:
                # fallback to a hard sub‑gradient
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # ----- gradient of the smoothed objective -----
        grad = _gradient_from_weights(a, w)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = eta_start * (1.0 - progress) + eta_end * progress
        step = eta * m_hat / (np.sqrt(v_hat) + epsilon)

        a_candidate = a - step

        # ----- occasional random mass‑transfer (jitter) -----
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a_candidate[src] * 0.5
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        # ----- projection onto simplex and enforce symmetry -----
        a = _project_onto_simplex(a_candidate)
        a = (a + a[::-1]) * 0.5
        a = _project_onto_simplex(a)

        iteration += 1

        # ----- simple stagnation detection – restart with fresh random seed -----
        if iteration - last_improve_iter > 150_000 and (deadline - now) > 30.0:
            print(f"[{iteration:7d}] stagnation detected – restarting")
            a = np.random.rand(n)
            a = _project_onto_simplex(a)
            a = (a + a[::-1]) * 0.5
            a = _project_onto_simplex(a)
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            last_improve_iter = iteration

        # occasional sanity‑check (no effect on the algorithm)
        if iteration % 5000 == 0:
            _ = _full_convolution_fft(a, a)   # keep NumPy’s internal caches happy

    # ------------------------------------------------------------------
    # Final reporting (the outer framework will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
