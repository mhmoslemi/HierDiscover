"""
propose_candidate
-----------------

Search for a step‑function (non‑negative vector) that minimises the
evaluation metric returned by ``evaluate_sequence``.
The metric equals the maximum entry of the self‑convolution when the
vector is normalised to have ℓ₁‑norm √(2 n).  The algorithm combines three
ideas:

1. **Linear‑programming (LP) tightening** –  given a current vector f,
   solve a linear program that finds a vector b with maximal total mass
   under the constraints  f∗b ≤ max(f∗f).  Scaling b back to the
   prescribed ℓ₁‑norm and taking a small convex combination with f
   provably never raises the maximum convolution value; iterating the
   step reduces the objective.

2. **Gradient‑based refinements** –  after a few LP steps a few
   sub‑gradient iterations on the *actual* max entry, followed by Adam
   optimisation on a log‑sum‑exp (smooth‑max) surrogate, are applied.
   The updates are projected onto the ℓ₁‑simplex of the required total
   mass.

3. **Local swap hill‑climber** –  tiny mass transfers between two
   positions are tested and kept if they improve the objective.

The whole procedure respects the 1000 s budget, tries several
initialisations (arcsine shape, uniform, random Dirichlet draws and the
historic sequence if available) and always returns a valid sequence.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection onto {x ≥ 0, Σ x = target}
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray, target: float = 1.0) -> np.ndarray:
    """Project v onto the probability simplex with given target sum."""
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), target):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, n + 1)
    # Find the last index where u_i > cssv_i / i
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based self‑convolution utilities
# ----------------------------------------------------------------------
def conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Return the full self‑convolution of p using FFT (length 2·len(p)-1)."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def conv_max(p: np.ndarray, m_fft: int) -> float:
    """Maximum entry of the self‑convolution of p."""
    return float(conv_self(p, m_fft).max())


# ----------------------------------------------------------------------
# Build the lower‑triangular Toeplitz matrix A such that (A @ b) = f∗b
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return matrix A (size (2·n‑1)×n) with A@b = f∗b."""
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    # column j contributes f[0…n‑1] starting at row j
    for j in range(n):
        A[j : j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Solve the LP: maximise Σ b_j s.t. b ≥ 0, f∗b ≤ M
# ----------------------------------------------------------------------
def lp_optimize(f: np.ndarray, M: float) -> np.ndarray:
    """Return the optimal b for the LP (maximise sum(b))."""
    n = f.shape[0]
    A = build_conv_matrix(f)          # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M)      # RHS = M for every convolution entry
    c = -np.ones(n)                   # maximise sum(b)  <=> minimise -sum(b)
    bounds = [(0.0, None)] * n
    # Using the HiGHS solver (fast, robust)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *actual* max entry of the convolution
# ----------------------------------------------------------------------
def subgradient_descent(p: np.ndarray, m_fft: int, target_sum: float,
                        max_iter: int = 5_000, step0: float = 0.2,
                        decay: float = 0.9995, min_step: float = 1e-5
                        ) -> tuple[np.ndarray, float]:
    """Perform cheap sub‑gradient steps on max(conv(p,p))."""
    n = p.shape[0]
    best_p = p.copy()
    best_M = conv_max(p, m_fft)
    step = step0

    for it in range(max_iter):
        conv = conv_self(p, m_fft)
        max_idx = int(conv.argmax())

        # sub‑gradient of the maximal entry
        g = np.zeros_like(p)
        if max_idx < n:
            g[: max_idx + 1] = 2.0 * p[max_idx :: -1]
        else:
            start = max_idx - n + 1
            g[start:] = 2.0 * p[n - 1 : start - 1 : -1]

        p = p - step * g
        p = simplex_projection(p, target_sum)

        cur_M = conv_max(p, m_fft)
        if cur_M < best_M - 1e-12:
            best_M = cur_M
            best_p = p.copy()

        step = max(step * decay, min_step)
    return best_p, best_M


# ----------------------------------------------------------------------
# Adam optimiser on a log‑sum‑exp (smooth‑max) surrogate
# ----------------------------------------------------------------------
def adam_optimize(p: np.ndarray, m_fft: int, target_sum: float,
                  max_iter: int = 20_000,
                  lr: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps_adam: float = 1e-8,
                  alpha_start: float = 1.0,
                  alpha_mul: float = 1.001,
                  alpha_max: float = 50.0) -> tuple[np.ndarray, float]:
    """Adam optimisation using a smoothed max (log‑sum‑exp)."""
    n = p.shape[0]
    m1 = np.zeros_like(p)
    m2 = np.zeros_like(p)

    alpha = alpha_start
    best_p = p.copy()
    best_M = conv_max(p, m_fft)

    for it in range(max_iter):
        # FFT of current vector
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

        # smooth max via log‑sum‑exp
        z = alpha * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)          # stabilise
        w = exp_z / np.sum(exp_z)          # soft‑max weights (size 2n‑1)

        # gradient of the smooth surrogate
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
        corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]

        idx = 2 * n - 2 - np.arange(n)      # mapping to gradient indices
        grad = 2.0 * corr_full[idx]

        # Adam moment updates
        m1 = beta1 * m1 + (1.0 - beta1) * grad
        m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

        m1_hat = m1 / (1.0 - beta1 ** (it + 1))
        m2_hat = m2 / (1.0 - beta2 ** (it + 1))

        step_vec = lr * m1_hat / (np.sqrt(m2_hat) + eps_adam)
        p = p - step_vec
        p = simplex_projection(p, target_sum)

        # tighten the smooth approximation
        alpha = min(alpha * alpha_mul, alpha_max)

        cur_M = conv_max(p, m_fft)
        if cur_M < best_M - 1e-12:
            best_M = cur_M
            best_p = p.copy()
    return best_p, best_M


# ----------------------------------------------------------------------
# Tiny local‑swap hill‑climber
# ----------------------------------------------------------------------
def local_swap(p: np.ndarray, m_fft: int, target_sum: float,
               attempts: int = 5_000,
               move_scale: float = 1e-4,
               rnd: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """Move a tiny amount of mass between two positions if it improves the score."""
    if rnd is None:
        rnd = np.random.default_rng()
    cur_M = conv_max(p, m_fft)
    best_p = p.copy()
    for _ in range(attempts):
        i, j = rnd.integers(0, len(p), size=2)
        if i == j:
            continue
        eps_move = move_scale * rnd.random()
        if p[i] <= eps_move:
            continue
        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move
        # keep the sum unchanged automatically
        new_M = conv_max(new_p, m_fft)
        if new_M < cur_M - 1e-12:
            p = new_p
            cur_M = new_M
            best_p = p.copy()
    return best_p, cur_M


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the length n (fallback to 1153 if unknown)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 1153

    # --------------------------------------------------------------
    # Normalisation: we keep ℓ₁‑norm = sqrt(2 n) so that the
    # evaluation metric returned by evaluate_sequence equals max(conv).
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)

    # FFT size – next power of two ≥ 2 n‑1
    m_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Helper for timing
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 2.0   # a tiny safety margin

    # --------------------------------------------------------------
    # Build a pool of promising initial vectors
    # --------------------------------------------------------------
    init_vectors = []

    # 1) Arcsine‑shaped distribution (continuous arcsine density discretised)
    i = np.arange(n, dtype=float)
    eps = 1e-12
    inv = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    p_arcsine = inv / inv.sum()
    init_vectors.append(p_arcsine * target_sum)

    # 2) Uniform distribution
    init_vectors.append(np.full(n, 1.0 / n) * target_sum)

    # 3) A few random Dirichlet draws
    for _ in range(3):
        rand = rng.random(n)
        rand = rand / rand.sum() * target_sum
        init_vectors.append(rand)

    # 4) Historic construction (if present)
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        seq = np.clip(seq, 0.0, 1000.0)
        s = seq.sum()
        if s > 0.0:
            init_vectors.append(seq * (target_sum / s))

    best_p = None
    best_M = float("inf")

    # --------------------------------------------------------------
    # Core optimisation loops
    # --------------------------------------------------------------
    for start_id, p0 in enumerate(init_vectors):
        if time.time() >= deadline:
            break
        print(f"\n=== Start {start_id + 1}/{len(init_vectors)} =====================")
        p = p0.copy()
        cur_M = conv_max(p, m_fft)

        if cur_M < best_M:
            best_M = cur_M
            best_p = p.copy()
            print(f"  initial max(conv) = {best_M:.9f}")

        # ---------- LP tightening phase ----------
        max_lp_iters = 25
        for lp_it in range(max_lp_iters):
            if time.time() >= deadline:
                break

            # Solve LP: maximise Σ b_j under f∗b ≤ cur_M
            try:
                b = lp_optimize(p, cur_M)
            except Exception as e:
                print(f"  LP failed at iter {lp_it}: {e}")
                break

            sum_b = b.sum()
            if sum_b <= 0.0:
                break

            # Scale to the required ℓ₁‑norm
            g = b * (target_sum / sum_b)

            # Try to mix p and g.  Reduce mixing factor if it does not help.
            t = 0.2
            improved = False
            while t > 1e-5:
                p_new = (1.0 - t) * p + t * g
                new_M = conv_max(p_new, m_fft)
                if new_M < cur_M - 1e-12:
                    p = p_new
                    cur_M = new_M
                    improved = True
                    break
                t *= 0.5

            if not improved:
                # No improvement possible with this LP solution
                break

            if cur_M < best_M - 1e-12:
                best_M = cur_M
                best_p = p.copy()
                print(f"  LP iter {lp_it:02d} – new best max(conv) = {best_M:.9f}")

            # Early exit if already very good
            if best_M <= 0.00066:   # corresponds to score ≈ 1.52, keep room for further gain
                break

        # ---------- Sub‑gradient phase ----------
        p, cur_M = subgradient_descent(p, m_fft, target_sum,
                                      max_iter=5_000, step0=0.2)
        if cur_M < best_M - 1e-12:
            best_M = cur_M
            best_p = p.copy()
            print(f"  after sub‑grad – new best max(conv) = {best_M:.9f}")

        # ---------- Adam phase ----------
        p, cur_M = adam_optimize(p, m_fft, target_sum,
                                 max_iter=15_000,
                                 lr=0.05, alpha_start=1.0,
                                 alpha_mul=1.001, alpha_max=30.0)
        if cur_M < best_M - 1e-12:
            best_M = cur_M
            best_p = p.copy()
            print(f"  after Adam – new best max(conv) = {best_M:.9f}")

        # ---------- Local swap refinement ----------
        p, cur_M = local_swap(p, m_fft, target_sum,
                              attempts=5_000, rnd=rng)
        if cur_M < best_M - 1e-12:
            best_M = cur_M
            best_p = p.copy()
            print(f"  after swaps – new best max(conv) = {best_M:.9f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_p is None:
        best_p = np.full(n, target_sum / n)
        best_M = conv_max(best_p, m_fft)

    print(f"\nFinal best max(conv) = {best_M:.9f} → score {2*n*best_M/(target_sum**2):.9f}")

    # Clip to the limits enforced by evaluate_sequence and return as list.
    return np.clip(best_p, 0.0, 1000.0).tolist()
