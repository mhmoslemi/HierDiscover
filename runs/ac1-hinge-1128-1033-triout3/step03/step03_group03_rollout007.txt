"""
propose_candidate
-----------------

Search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

with the additional clamps performed by the official evaluator.
The algorithm consists of three stages:

1. **Warm‑start** – try a handful of simple parametric families
   (Catalan, power‑law, geometric) and keep the best one (or a
   previously discovered sequence if the global ``height_sequence_1``
   variable is present).

2. **Smoothed sub‑gradient descent** – instead of using the exact
   max‑entry gradient we minimise a smooth approximation
   `τ·log Σ exp(conv/τ)`.  The gradient is a cross‑correlation and can
   be obtained with a single convolution, so each iteration costs only
   O(n log n).  A momentum term and a geometric decay of the step size
   and the temperature `τ` provide fast convergence.

3. **Stagnation handling** – when no improvement has been observed for
   a while we inject a small random perturbation and reset the momentum
   (a cheap form of restart).

The whole loop runs until the 1000 s wall‑clock budget is exhausted.
The best sequence seen is finally clamped to the limits enforced by
``evaluate_sequence`` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x | x ≥ 0, sum(x)=1}.

    Implements the algorithm of Duchi et al. (2008).  Works for 1‑D
    arrays.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a step‑function (list of heights) that minimises the
    evaluation function.  The method combines a quick parametric
    warm‑start with a smoothed projected gradient descent on the
    maximal self‑convolution entry.
    """
    # safety margin for the harness
    deadline = time.time() + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size.
    # --------------------------------------------------------------
    # Use the length of a previously found sequence if available,
    # otherwise fall back to a default size.
    try:
        init_global = np.array(height_sequence_1, dtype=np.float64)
        n = len(init_global)
        if n < 2000:               # very short globals are not useful here
            raise NameError
    except Exception:
        init_global = None
        n = 5000                     # default length

    # --------------------------------------------------------------
    # 2) Warm‑start: evaluate a few simple families and keep the best.
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    def _eval(seq: np.ndarray):
        # seq is already on the simplex, so the official evaluator
        # will give the same value (except for the clamping which never
        # triggers for our tiny numbers).
        return evaluate_sequence(seq.tolist())

    # a) previously discovered sequence
    if init_global is not None:
        a0 = init_global.copy()
        a0 = a0 / a0.sum()
        val = _eval(a0)
        best_seq = a0.copy()
        best_val = val
        print(f"[init] global sequence length={n}, score={val:.10f}")

    # b) Catalan (central binomial) coefficients
    coeffs = np.empty(n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, n):
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    a_catalan = coeffs / coeffs.sum()
    val = _eval(a_catalan)
    if val < best_val:
        best_seq = a_catalan.copy()
        best_val = val
        print(f"[candidate] Catalan, score={val:.10f}")

    # c) Power‑law families a_i ∝ (i+1)^{-p}
    p_vals = np.linspace(0.3, 1.3, 11)          # 0.3 … 1.3
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        a = idx ** (-p)
        a = a / a.sum()
        val = _eval(a)
        if val < best_val:
            best_seq = a.copy()
            best_val = val
            print(f"[candidate] power p={p:.3f}, score={val:.10f}")

    # d) Geometric families a_i ∝ r^{i}
    r_vals = np.linspace(0.85, 0.99, 8)
    for r in r_vals:
        a = r ** np.arange(n, dtype=np.float64)
        a = a / a.sum()
        val = _eval(a)
        if val < best_val:
            best_seq = a.copy()
            best_val = val
            print(f"[candidate] geometric r={r:.3f}, score={val:.10f}")

    # --------------------------------------------------------------
    # 3) Initialise the optimisation variables.
    # --------------------------------------------------------------
    a = best_seq.copy()                # current iterate (on the simplex)
    momentum = np.zeros_like(a)        # for Nesterov‑type momentum

    # hyper‑parameters
    eta = 0.5                          # initial step size (Euclidean)
    min_eta = 1e-5
    decay_eta = 0.9999                 # geometric decay per iteration

    tau = 0.02                         # temperature for the smooth max
    tau_min = 1e-7
    decay_tau = 0.9997

    beta = 0.9                         # momentum coefficient

    it = 0
    no_improve = 0
    best_approx = 2.0 * n * np.convolve(a, a).max()   # approximation of the objective (sum=1)

    # --------------------------------------------------------------
    # 4) Main optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # current convolution and (approx.) objective
        conv = np.convolve(a, a)
        max_conv = conv.max()
        approx_val = 2.0 * n * max_conv

        # track improvement
        if approx_val < best_approx - 1e-12:
            best_approx = approx_val
            best_seq = a.copy()
            # evaluate exact score only when we have a new record
            exact_val = _eval(best_seq)
            best_val = exact_val
            print(f"[iter {it:6d}] new best approx={approx_val:.10f}, exact={exact_val:.10f}, "
                  f"eta={eta:.5f}, tau={tau:.2e}")
            no_improve = 0
        else:
            no_improve += 1

        # ----------------------------------------------------------
        # Smooth max – compute weights w_k = exp((c_k - max_c)/tau)
        # ----------------------------------------------------------
        shifted = conv - max_conv                # ≤ 0
        w = np.exp(shifted / tau)               # 0 < w_k ≤ 1

        # ----------------------------------------------------------
        # Gradient of the smooth max: g = 2 * (a ⋆ w)
        # ----------------------------------------------------------
        w_rev = w[::-1]
        g_full = np.convolve(a, w_rev)          # length = 3n‑2
        L = len(w)                              # = 2n‑1
        # cross‑correlation index transform
        g = 2.0 * g_full[L - 1 - np.arange(n)]

        # ----------------------------------------------------------
        # Momentum step and projection onto the simplex
        # ----------------------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * g
        a_new = a - eta * momentum
        a_new = _project_simplex(a_new)

        a = a_new

        # decay parameters
        eta = max(min_eta, eta * decay_eta)
        tau = max(tau_min, tau * decay_tau)

        # ----------------------------------------------------------
        # Stagnation handling – inject a tiny random perturbation
        # ----------------------------------------------------------
        if no_improve > 20000:
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
            momentum.fill(0.0)
            no_improve = 0
            print(f"[restart] iter {it}, added noise")

        # occasional status report (does not affect timing)
        if it % 5000 == 0:
            print(f"[status] iter {it}, best≈{best_approx:.10f}, eta={eta:.5f}, tau={tau:.2e}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clamped to the evaluator limits).
    # --------------------------------------------------------------
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(best_seq)
    print(f"[result] final_score={final_score:.10f}, length={len(best_seq)}")
    return best_seq
