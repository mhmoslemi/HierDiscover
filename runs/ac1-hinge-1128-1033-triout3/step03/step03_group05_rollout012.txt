"""
Propose a candidate step‐function (probability vector) that minimizes
the evaluation score 2·n·max(conv_self)  (the same quantity returned by
``evaluate_sequence`` when the vector is normalised to sum = 1).

Approach
--------
* Initialise a modest pool of reasonable symmetric probability vectors:
  – arcsine‑shaped families  p_i ∝ ((i+½)(n‑½‑i))^{-a} with a∈[0.3, 0.9]
  – a uniform vector
  – a triangular “tent’’ shape
  – a few random Dirichlet draws.
* For every start vector run three optimisation phases
  1. **Sub‑gradient descent** on the true max–convolution entry.
     The sub‑gradient is computed analytically; a decaying step size
     together with projection onto the probability simplex keeps the
     iterate feasible.  Symmetry (p_i = p_{n‑1‑i}) is enforced after each
     projection.
  2. **Adam on a smooth log‑sum‑exp approximation** of the max.
     The soft‑max weights are computed from the self‑convolution; the
     exact gradient of the smoothed objective is obtained via a single
     FFT‑based correlation.  The smoothing parameter α is increased
     slowly up to a large value (≈ 200) so that the smooth loss becomes
     the true max.
  3. **Local swap hill‑climber** that moves a tiny amount of probability
     from one index to another whenever it reduces the objective.
* The best vector seen over all starts is returned (clipped to the
  [0, 1000] range required by ``evaluate_sequence``).

The whole routine respects the 1000 s budget, prints progress and always
returns a valid non‑negative sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negativity and unit sum.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x | x_i≥0, Σx_i=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Enforce left/right symmetry (p_i = p_{n-1-i}) while preserving the sum.
# ----------------------------------------------------------------------
def _symmetrize(p: np.ndarray) -> np.ndarray:
    """Return a symmetrised copy of p (average with its reverse)."""
    return (p + p[::-1]) * 0.5

# ----------------------------------------------------------------------
# Self‑convolution via FFT (linear convolution, length 2·n‑1).
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Linear self‑convolution of p using an FFT of size ``m_fft``."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv

# ----------------------------------------------------------------------
# Objective value (2·n·max(conv_self)) for a probability vector.
# ----------------------------------------------------------------------
def _score(p: np.ndarray, m_fft: int) -> float:
    """Return the evaluation score for a probability vector (sum(p)=1)."""
    conv = _conv_self(p, m_fft)
    return float(2.0 * p.size * np.max(conv))

# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑entry of the convolution.
# ----------------------------------------------------------------------
def _max_subgradient(p: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Compute a sub‑gradient of max(conv_self) w.r.t. p."""
    n = p.size
    max_val = conv.max()
    # All locations attaining the maximum (within numerical tolerance)
    active = np.where(conv >= max_val - 1e-15)[0]
    g = np.zeros_like(p)
    for idx in active:
        if idx < n:
            # Overlap size = idx+1, reversed segment of p[0:idx+1]
            g[: idx + 1] += 2.0 * p[idx :: -1]
        else:
            start = idx - n + 1
            # Tail part (indices start … n‑1) reversed
            g[start:] += 2.0 * p[n - 1 : start - 1 : -1]
    if active.size > 1:
        g /= float(active.size)
    return g

# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp smoothed max (parameter α).
# ----------------------------------------------------------------------
def _smooth_max_gradient(p: np.ndarray, conv: np.ndarray,
                         alpha: float, m_fft: int,
                         idx_map: np.ndarray) -> np.ndarray:
    """
    Compute ∇[(1/α)·log Σ_k exp(α·conv_k)] for a probability vector p.
    The gradient is 2·corr(p, w_rev) on the appropriate indices,
    where w are the soft‑max weights.
    """
    # soft‑max weights
    z = alpha * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / exp_z.sum()

    # correlation p * w_rev (linear, not circular)
    w_rev = w[::-1]
    p_fft = np.fft.rfft(p, n=m_fft)
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * p.size - 2]

    # Extract the n components that correspond to ∂conv_k/∂p_i
    grad = 2.0 * corr_full[idx_map]
    return grad

# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 1153                       # fallback length used in the benchmark

    # FFT size must accommodate linear convolutions of length 3·n‑2
    needed = max(2 * n - 1, 3 * n - 2)
    m_fft = 1 << ((needed - 1).bit_length())

    # Mapping from p‑index to the linear‑convolution index used in the
    # gradient of the smoothed loss.
    idx_map = 2 * n - 2 - np.arange(n)

    # --------------------------------------------------------------
    # Helper: timing
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_score = float("inf")
    best_p = None

    # --------------------------------------------------------------
    # Build a pool of promising initial vectors (all symmetric)
    # --------------------------------------------------------------
    init_vectors = []

    i = np.arange(n, dtype=float)

    # 1) Arcsine‑type family p_i ∝ ((i+½)(n‑½‑i))^{-a}
    for a in np.linspace(0.30, 0.90, 7):          # a=0.5 is the classic arcsine
        den = (i + 0.5) * (n - 0.5 - i) + 1e-12
        p = den ** (-a)
        p = _symmetrize(p / p.sum())
        init_vectors.append(p)

    # 2) Uniform distribution (already symmetric)
    p_uniform = np.full(n, 1.0 / n)
    init_vectors.append(p_uniform)

    # 3) Tent (triangular) shape
    centre = (n - 1) / 2.0
    tent = 1.0 - np.abs(i - centre) / centre
    p_tent = _symmetrize(tent / tent.sum())
    init_vectors.append(p_tent)

    # 4) Random Dirichlet draws (symmetrised)
    for _ in range(3):

        rnd /= rnd.sum()
        p = _symmetrize(rnd)
        init_vectors.append(p)

    # Core optimisation loops – each start is processed independently
    # --------------------------------------------------------------
    for start_id, p0 in enumerate(init_vectors):
        if time.time() >= deadline:

            best_p = p.copy()

        min_step = 1e-7

            cur_score = 2.0 * n * conv.max()

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                print(f"  Subgrad iter {it:5d} – new best {best_score:.6f}")

                    break

            p = p - step * g
            p = _simplex_projection(p)
            p = _symmetrize(p)

            step = max(step * decay, min_step)

        # ----------------------------------------------------------
        # (b) Adam optimisation on a log‑sum‑exp smooth max
        # ----------------------------------------------------------
        # Adam state
        m1 = np.zeros_like(p)
        m2 = np.zeros_like(p)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lr = 0.05

        # Smoothing schedule
        alpha = 1.0
        alpha_mul = 1.0015
        alpha_max = 200.0

        max_adam_iters = 100_000
        for it in range(max_adam_iters):
            if time.time() >= deadline:
                break

            # current convolution and smooth gradient
            p_fft = np.fft.rfft(p, n=m_fft)
            conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

            # smooth max (log‑sum‑exp)
            grad = _smooth_max_gradient(p, conv, alpha, m_fft, idx_map)

            # Adam biased‑moment updates
            m1 = beta1 * m1 + (1.0 - beta1) * grad
            m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

            it1 = it + 1
            m_hat = m1 / (1.0 - beta1 ** it1)
            v_hat = m2 / (1.0 - beta2 ** it1)

            step_vec = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            p = p - step_vec
            p = _simplex_projection(p)
            p = _symmetrize(p)

            # increase smoothness
            alpha = min(alpha * alpha_mul, alpha_max)

            # evaluate
            cur_score = 2.0 * n * np.max(conv)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                print(f"  Adam iter {it:5d} – new best {best_score:.6f}")

                if best_score <= 1.503:
                    break

        # (c) Tiny local swap hill‑climber
        # ----------------------------------------------------------
        swap_attempts = 30_000
        cur_score = _score(p, m_fft)

        for _ in range(swap_attempts):
            if time.time() >= deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # move at most 10 % of p[i] (but never more than p[i])
            eps_move = p[i] * rng.random() * 0.1
            if eps_move <= 0.0:
                continue
            new_p = p.copy()
            new_p[i] -= eps_move
            new_p[j] += eps_move
            new_score = _score(new_p, m_fft)
            if new_score < cur_score - 1e-12:
                p = new_p
                cur_score = new_score
                best_p = p.copy()
                best_score = cur_score
                print(f"  Swap improvement → {best_score:.6f}")
                if best_score <= 1.503:
                    break

    # ------------------------------------------------------------------
    # Final global perturbation & refinement (if time permits)
    # ------------------------------------------------------------------
    if best_p is not None and time.time() < deadline:
        p = best_p.copy()
        cur_score = best_score
        extra_swaps = 40_000
        for _ in range(extra_swaps):
            if time.time() >= deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:

            eps_move = p[i] * rng.random() * 0.2   # larger jumps than before

            new_p = p.copy()
            new_p[i] -= eps_move
            new_p[j] += eps_move
            new_score = _score(new_p, m_fft)
            if new_score < cur_score - 1e-12:
                p = new_p
                cur_score = new_score
                best_p = p.copy()
                best_score = cur_score
                print(f"  Final swap improvement → {best_score:.6f}")
                if best_score <= 1.503:
                    break

    print(f"\n=== Finished. Best score achieved: {best_score:.6f} ===")
    # Clip to the range enforced by evaluate_sequence and return a plain list
    return np.clip(best_p, 0.0, 1000.0).tolist()
