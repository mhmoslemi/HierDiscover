Propose candidate sequence using alternating linear programming and convex mixing.
The algorithm starts from an arcsine‑shaped distribution (and a few random starts),
solves a linear program that maximises the ℓ1‑norm of a partner vector g under the
constraint ‖f∗g‖∞ ≤ M where M = ‖f∗f‖∞, mixes f and the normalised optimiser g,
and repeats.  After the LP phase a tiny random‑swap hill‑climber polish the result.
The whole search respects the 1000 s budget and returns a valid list of non‑negative
heights.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: score for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _score(p: np.ndarray) -> float:
    """Return the evaluation score 2·n·max(p∗p) for a probability vector."""
    conv = np.convolve(p, p)
    return float(2.0 * p.size * conv.max())

        * Try a few mixing coefficients t∈{0.05,0.1,0.2,0.3,0.5,0.02} and keep the

3.  Optionally polish the best vector with a few hundred Adam steps on a
    log‑sum‑exp smooth approximation of the max (this mirrors the reference
    sub‑gradient/Adam code but with a tighter schedule).

4.  Finish with a tiny random‑swap hill‑climber that moves a small amount of
    probability mass between two indices whenever it improves the score.

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
