def conv_fft(a):
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n-1]
    return conv

def proj_simplex(v):
    if np.sum(v) == 1 and np.all(v>=0): return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1

### Function to project onto simplex

```python
def proj_simplex(v):
    """Projects v onto the probability simplex."""
    # ensure numpy array
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is all zeros?
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    # Normalize to sum=1 (some rounding)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w
```

def gradient_smooth_max(a, t):
    """Return gradient of smoothed max (log-sum-exp) w.r.t a."""
    n = a.shape[0]
    conv = conv_fft(a)  # length 2n-1
    max_conv = conv.max()
    # compute weights
    w = np.exp(t * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(conv) / conv.size
    else:
        w /= w_sum
    # compute gradient: 2 * (w * a_rev) convolution slice
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    grad_full = np.fft.irfft(W * A_rev, n=size)[:len(w) + len(a_rev) - 1]
    start = len(a_rev) - 1
    grad = 2.0 * grad_full[start:start + n]
    return grad, conv

"""
Propose a non‑negative step function that minimizes the provided evaluation metric.

Algorithm Overview
------------------
1. **Warm‑start** – If a global variable ``height_sequence_1`` exists it is used,
   normalised to a probability vector (sum = 1).

2. **Parametric pre‑search** – For a handful of lengths (up to 24 000) a small
   family of symmetric distributions (U‑shaped, sum‑of‑tails, exponential,
   product‑power‑law) is swept.  The member with the smallest *approximate*
   score 2·n·max(a∗a) becomes the seed for the next phase.

3. **Smooth max‑gradient descent** –  
   For the chosen seed we minimise the smooth approximation of the max‑norm

        f_t(a) = (1/t)·log∑_k exp(t·(a∗a)_k)

   whose gradient is
        ∇f_t(a)_i = 2·∑_k w_k·a_{k‑i},
   where w_k ∝ exp(t·(a∗a)_k).  The gradient is obtained by a single FFT
   convolution.  After each step the vector is projected onto the probability
   simplex (non‑negative, sum = 1).  The smoothing parameter `t` starts small
   (≈1) and is multiplied by a factor ≈1.01 each iteration, gradually turning
   the surrogate into the true max.

4. **Best‑of‑all** – The best candidate over all lengths and random restarts
   (measured by the exact objective 2·n·max(a∗a) ) is kept.  Finally the
   official ``evaluate_sequence`` routine is called once.

The routine respects a 1000 s wall‑clock budget and prints progress
information.  All helper functions are top‑level as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT, zero‑padding to the next power of 2)
# ----------------------------------------------------------------------


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a using a real FFT.  Result length = 2·len(a)‑1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of a∗a (computed via FFT)."""
    return float(conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise to exactly 1
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Parametric families (all return a normalised vector with sum = 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges: a[i] ∝ exp(-α·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """Product‑power‑law shape: a[i] ∝ ((i+1)*(n‑i))^{‑p}."""
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max‑gradient (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def gradient_smooth_max(a: np.ndarray, t: float):
    """
    Compute gradient of the smoothed max (log‑sum‑exp) of the self‑convolution.
    Returns (grad, conv) where conv = a∗a.
    """
    n = a.shape[0]
    conv = conv_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    # weights proportional to exp(t * (conv - max_conv))
    w = np.exp(t * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # gradient: 2 * (w ∗ a_rev) sliced to length n
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    grad_full = np.fft.irfft(W * A_rev, n=size)[: len(w) + len(a_rev) - 1]
    start = len(a_rev) - 1                 # = n‑1
    grad = 2.0 * grad_full[start:start + n]
    return grad, conv


# ----------------------------------------------------------------------
# Optimisation driver for a single length
# ----------------------------------------------------------------------


def optimise_length(
        n: int,
        init_a: np.ndarray,
        max_iters: int,
        t0: float,
        t_factor: float,
        step0: float,
        deadline: float):
    """
    Projected gradient descent on the smoothed max‑objective.

    Parameters
    ----------
    n            – length of the vector.
    init_a       – starting probability vector (sum = 1, non‑negative).
    max_iters    – maximum number of iterations (hard stop on time limit).
    t0           – initial smoothing parameter.
    t_factor     – factor by which t is multiplied each iteration.
    step0        – base step size (scaled by 1/√iter).
    deadline     – absolute time after which the routine must stop.
    """
    a = proj_simplex(init_a.copy())
    best_a = a.copy()
    conv = conv_fft(a)
    best_score = 2.0 * n * conv.max()
    t = t0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        grad, conv = gradient_smooth_max(a, t)

        # diminishing step size
        step = step0 / np.sqrt(it)

        a_candidate = a - step * grad
        a_candidate = proj_simplex(a_candidate)

        # evaluate exact (via FFT) max‑convolution for the candidate
        conv_cand = conv_fft(a_candidate)
        cand_score = 2.0 * n * conv_cand.max()

        # accept if improvement (no simulated‑annealing here – we keep a downhill walk)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            # occasional progress report
            if it % 200 == 0:
                print(f"    [n={n}] iter {it:5d}, t={t:.3f}, score={cand_score:.6f}")

        a = a_candidate
        t *= t_factor  # slowly make the surrogate sharper

    return best_a, best_score
