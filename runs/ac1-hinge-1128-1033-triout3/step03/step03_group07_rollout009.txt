def grad_wrt_a(w, a):
    # w: array length 2n-1
    # a: length n
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)[: len(w) + len(a_rev) - 1]
    grad = 2.0 * conv[n-1 : n-1 + n]  # slice length n
    return grad

"""
Propose a non‑negative step function minimizing the provided evaluation metric.

Algorithm
---------
1. Warm‑start: use global `height_sequence_1` if present.  
2. Deterministic parametric sweep over several families (U‑shaped,
   tail‑sum, exponential, product‑power‑law, symmetric Beta‑like) for a
   range of lengths up to 32000 and a wide range of shape parameters.
   The best candidate (according to the cheap proxy `2·n·max(a∗a)`)
   is kept as the starting vector.
3. Exponentiated‑gradient descent on a smoothed version of the
   `max(conv)` objective.
   * The smooth surrogate is `Fβ(a)= (1/β)·log Σ_k exp(β·(a∗a)_k)`.
   * The gradient of `Fβ` w.r.t. the probability vector `a`
     can be written as `2·(w∗a_rev)` where `w_k ∝
     exp(β·(a∗a)_k)` and `a_rev` is `a` reversed.
   * The update is multiplicative, `a←a·exp(−η·grad)`,
     followed by a renormalisation to keep Σ a = 1.
   * β is slowly increased (sharpening the max) while η is
     slowly decreased.  An occasional small random perturbation is
     added to escape shallow local minima.  The loop stops when the
     time budget is exhausted or the target score 1.503 is reached.
4. The best vector found is finally evaluated with the official
   `evaluate_sequence` routine and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def grad_wrt_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution objective.
    Returns the vector `g` of length len(a) with g_i = 2·Σ_k w_k·a_{k‑i}.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed copy of a
    # linear convolution of w (len 2n‑1) with a_rev (len n)
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A, n=size)[: len(w) + n - 1]
    # slice the part that corresponds to indices i = 0 … n‑1
    start = n - 1
    grad = 2.0 * conv[start : start + n]
    return grad


# ----------------------------------------------------------------------
# Parametric families (all return a vector normalised to sum == 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-18
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-18
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges: a[i] ∝ exp(-α·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-18
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """
    Product‑power‑law shape:
        a[i] ∝ ((i+1)*(n-i))^{‑p}
    """
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-18
    a /= a.sum()
    return a


def beta_like_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta‑like shape:
        a[i] ∝ (x_i)^{α-1}·(1‑x_i)^{β‑1},
    where x_i = i/(n‑1).  When α,β<1 the distribution is U‑shaped.
    """
    x = np.linspace(0.0, 1.0, n)
    # avoid zeros to keep the vector finite
    a = (np.power(x + 1e-18, alpha - 1.0) *
         np.power(1.0 - x + 1e-18, beta - 1.0) + 1e-18)
    a /= a.sum()
    return a

    beta_params = [(0.2, 0.2), (0.3, 0.3), (0.4, 0.4), (0.6, 0.6)]

    for n in lengths:

        for p in p_vals:

            score = 2.0 * n * max_conv_fft(a)

                print(f"New best: n={n:5d}, shape=exp, α={a_alpha:6.4f}, approx_score={score:.6f}")

        # ---- Product power‑law ----
        for p in p_vals:
            if time.time() > deadline:
                break
            a = product_powerlaw_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"New best: n={n:5d}, shape=prod‑pow, p={p:5.2f}, approx_score={score:.6f}")

        for (alpha, beta) in beta_params:
            if time.time() > deadline:

            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"New best: n={n:5d}, shape=beta, α={alpha:.2f}, β={beta:.2f}, approx_score={score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        a = np.random.rand(n)

        best_score = 2.0 * n * max_conv_fft(a)

"""
Propose a non‑negative step function minimizing the provided evaluation metric.

Algorithm
---------
1. Warm‑start: use the global variable ``height_sequence_1`` if it exists.
2. Deterministic parametric sweep over several families (U‑shaped power‑law,
   sum‑of‑tails, exponential, product‑power‑law, symmetric Beta‑like) for a
   range of lengths (up to 32 000) and shape parameters.  The cheapest proxy
   ``2·n·max(a∗a)`` is used to keep the best candidate.
3. Exponentiated‑gradient descent on a smoothed version of the ``max(conv)``
   objective.
   * The smooth surrogate is   Fβ(a)= (1/β)·log Σ exp(β·(a∗a)ₖ).
   * Its gradient can be written as  g = 2·(w∗a_rev) where
     wₖ ∝ exp(β·(a∗a)ₖ) and a_rev is a reversed copy of a.
   * The update is multiplicative: a ← a·exp(−η·g), followed by renormalisation
     to keep Σ a = 1.
   * β is slowly increased (sharpening the max) while η is slowly
     decreased.  A tiny simulated‑annealing acceptance rule and occasional
     random jitter help escape shallow plateaus.
4. The best vector found is finally evaluated with the official
   ``evaluate_sequence`` routine and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def grad_wrt_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution objective.
    Returns g of length len(a) with
        g_i = 2·Σ_k w_k·a_{k‑i}.
    """
    n = a.shape[0]
    a_rev = a[::-1]                         # reversed a
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A, n=size)[: len(w) + n - 1]
    start = n - 1
    grad = 2.0 * conv[start : start + n]
    return grad


# ----------------------------------------------------------------------
# Parametric families (all return a normalised vector with sum == 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-18
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-18
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges: a[i] ∝ exp(−α·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-18
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """
    Product‑power‑law shape:
        a[i] ∝ ((i+1)*(n‑i))^{‑p}.
    """
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-18
    a /= a.sum()
    return a


def beta_like_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta‑like shape:
        a[i] ∝ x_i^{α‑1}·(1‑x_i)^{β‑1},
    with x_i = i/(n‑1).  Choose α,β<1 for a U‑shape.
    """
    x = np.linspace(0.0, 1.0, n)
    a = (np.power(x + 1e-18, alpha - 1.0) *
         np.power(1.0 - x + 1e-18, beta - 1.0) + 1e-18)
    a /= a.sum()
    return a
