    # compute softmax weights w
    max_c = conv.max()

    offset = len(w) - 1
    grad = 2.0 * conv2[offset - np.arange(n)]
    return grad

def smooth_gradient(a, conv, t):
    n = a.shape[0]
    max_c = conv.max()
    # Scale
    scaled = t * (conv - max_c)
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    if sum_exp == 0:
        w = np.zeros_like(exp_vals)
        w[conv.argmax()] = 1.0
    else:
        w = exp_vals / sum_exp
    # correlation conv

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises the evaluation score

        2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm consists of
two phases:

1.  Deterministic sweep over several parametric “U‑shaped’’ families
    (inverse distance, inverse‑product‑distance, exponential decay,
    offset‑U‑shape).  A large set of lengths and shape parameters are
    tried; the best vector from this sweep becomes the seed.

2.  Stochastic refinement using a smoothed‑max gradient (soft‑max)
    followed by a projected (onto the probability simplex) momentum
    update.  The soft‑max temperature is increased slowly; after it
    exceeds a modest value the exact sub‑gradient of the max‑convolution
    entry is used.  Simulated‑annealing acceptance, occasional random
    mass‑transfer moves and a reheating scheme are employed to escape
    stagnation.

The whole search runs inside the 1000 s budget and returns the best
sequence found (as a plain Python list).  All numbers are non‑negative;
the external evaluator will rescale the vector as needed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level as required)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex
        { w >= 0 , sum(w) = 1 }.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact value returned by ``evaluate_sequence`` for a vector that
    already satisfies sum(a) == 1 (the denominator becomes 1).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices where the convolution attains (or almost attains) its
    maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        # contribution: 2 * a[j]
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def smooth_gradient(a: np.ndarray, conv: np.ndarray, t: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation
        φ_t(a) = (1/t) * log Σ_k exp(t * (a*a)[k])
    Returns a vector of the same size as ``a``.
    """
    n = a.shape[0]
    max_c = conv.max()
    # soft‑max weights (stable)
    scaled = t * (conv - max_c)
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    if sum_exp == 0:
        w = np.zeros_like(exp_vals)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_vals / sum_exp

    # cross‑correlation of a with w (grad_i = 2 Σ_j a_j w_{i+j})
    w_rev = w[::-1]
    total_len = n + len(w_rev) - 1          # = 3n‑2
    size = 1 << ((total_len - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w_rev, n=size)
    conv2 = np.fft.irfft(fa * fw, n=size)
    conv2 = conv2[:total_len]               # linear convolution result

    offset = len(w) - 1                     # = 2n‑2
    grad = 2.0 * conv2[offset - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Parametric families of U‑shaped sequences
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on the distance to the nearest edge:
        a[i] = (min(i, n‑1‑i) + 1)^{-p}
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on the product of distances to both edges:
        a[i] = ((i+1)*(n‑i))^{-p}
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """
    Symmetric exponential decay from the edges:
        a[i] = exp(-lam*i) + exp(-lam*(n‑1‑i))
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def u_min_offset_distribution(n: int, p: float, c: float) -> np.ndarray:
    """
    Offset version of the distance‑based U‑shape:
        a[i] = (min(i, n‑1‑i) + c)^{-p}
    Normalised to sum = 1.  The offset c > 0 allows a gentler
    behaviour near the centre.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the given
    time budget.  Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over parametric families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # Lengths to explore – a compromise between quality and speed.
    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000,
               24000, 30000]

    # Parameters for the families
    exponents = np.arange(0.2, 6.01, 0.2)            # for power‑laws
    lam_vals = np.logspace(-5, -0.3, num=25)       # for exponential decay
    offsets = [0.1, 0.5, 1.0, 2.0, 5.0]            # for offset U‑shape

    print("=== Deterministic sweep start ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # --- plain distance‑based U‑shape ---------------------------------
        for p in exponents:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best (min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # --- product‑distance U‑shape ------------------------------------
        for p in exponents:
            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best (prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # --- offset distance U‑shape --------------------------------------
        for p in exponents[::2]:        # coarser grid for the extra dimension
            for c in offsets:
                a = u_min_offset_distribution(n, p, c)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"New best (off)   n={n:5d} p={p:4.2f} c={c:.2f}  score={sc:.6f}")

        # --- exponential decay --------------------------------------------
        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best (exp)   n={n:5d} λ={lam:.5f}  score={sc:.6f}")

        # time guard
        if time.time() > deadline:
            break

    # 2. Incorporate an external construction if available

            s = external.sum()
            if s > 0.0:
                external /= s
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score = sc
                    best_seq = external.copy()
                    print(f"Using external height_sequence_1   score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f"Fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement (smooth‑max gradient + momentum + SA)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    a /= a.sum()
    conv = full_convolution_fft(a)
    cur_score = float(2.0 * n * conv.max())

    # hyper‑parameters
    t = 0.5                     # soft‑max temperature (starts small)
    t_inc = 1.02                # geometric increase
    t_max = 12.0                # after this we switch to exact sub‑gradient
    step = 0.6                  # initial step length
    step_decay = 0.9995
    momentum = 0.9
    vel = np.zeros_like(a)      # momentum buffer
    temperature = 0.8           # SA temperature
    temp_decay = 0.9999
    iteration = 0
    last_improve = 0

    print("\n=== Stochastic refinement start ===")
    while time.time() < deadline:
        iteration += 1

        # ----- 4a. Gradient (smooth‑max or sub‑gradient) -----------------
        if t < t_max:
            grad = smooth_gradient(a, conv, t)
        else:
            grad = compute_subgradient(a, conv)

        # ----- 4b. Momentum / projected step ----------------------------
        vel = momentum * vel - step * grad
        cand = a + vel
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)
        cand /= cand.sum()

        # ----- 4c. Evaluate candidate ------------------------------------
        cand_conv = full_convolution_fft(cand)

        accept = False

            accept = True
        else:

            conv = cand_conv
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()

                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

    # ------------------------------------------------------------------

    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
