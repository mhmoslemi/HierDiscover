assistantfinal**Solution Overview**

The evaluation function is homogeneous – scaling the whole vector does not change the score.  
Hence we can (and do) restrict ourselves to probability vectors  
`a ≥ 0 ,  Σ a = 1`.  

For a vector `a` (length `n`) let

```
b = a * a                     # full linear convolution, length 2·n‑1
score(a) = 2·n·max(b)         # the value returned by evaluate_sequence
```

`max(b)` is the ℓ∞‑norm of the quadratic form `aᵀQₖa`.  
Minimising the maximum convolution entry is a convex problem, but solving it with a generic
convex optimiser is far too slow for the dimension we need (several thousand).

The algorithm below combines three ideas:

1. **Fast deterministic scan** – a few classic “U‑shaped’’ families (inverse‑distance,
   inverse‑product‑distance, exponential decay) are evaluated for several lengths;
   the best of them is kept as a *seed*.

2. **Smoothed‐max gradient descent** – instead of the non‑smooth `max`, we minimise a smooth
   approximation

```
Lβ(a) = (1/β)·log( Σ_k exp( β·b_k ) )           (β ≥ 1)
```

   `Lβ → max(b)` when `β` becomes large.  The gradient of `Lβ` can be written
   analytically and computed with two FFTs (one for `a*a` and one for the weighted
   convolution needed in the gradient).  An Adam optimiser on the simplex (projected
   after each step) drives the search.

3. **Stagnation handling** – after a long period without improvement a tiny random
   perturbation is added and the Adam moments are reset (a simple “re‑heat’’).

All work is performed inside `propose_candidate`; the function respects the supplied
time‑budget (default 1000 s) and finally returns the best vector it has found.

```python
"""
propose_candidate
-----------------
Search for a non‑negative step‑function (probability vector) that minimises

    2 * n * max_k (a * a)[k]

where * is full linear convolution.  The method is:

1.  deterministic sweep over a few parametric families (U‑shapes,
    product‑U‑shapes, exponentials) for several lengths;
2.  initialise a gradient‑descent phase with the best candidate;
3.  minimise a smoothed approximation of the ℓ∞‑norm of the convolution
    using Adam on the simplex (projected after each step);
4.  occasional re‑heat when progress stalls.

The optimiser works purely with NumPy/FFT, needs no external solver and fits
the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w ≥ 0, Σw = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def score_from_vector(a: np.ndarray) -> float:
    """Exact score for a probability vector (Σa = 1)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i,n‑1‑i)+1)^(-p)   (U‑shape based on distance to edge)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1.0) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^(-p)   (U‑shape based on product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1.0) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam·i) + exp(-lam·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def resample_to_length(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a vector to a new length."""
    old_n = v.shape[0]
    if old_n == new_n:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, v)


def smoothed_max_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of Lβ(a) = (1/β)·log Σ exp(β·(a*a)[k]).
    Returns ∇Lβ (same shape as a).
    """
    n = a.shape[0]
    max_val = conv.max()
    # Soft‑max weights (sum to 1)
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    w /= w_sum
    # Gradient: g_i = 2 * Σ_j w_{i+j} a_j
    w_rev = w[::-1]                       # reverse for convolution
    size = 1 << ((n + len(w_rev) - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    Fwr = np.fft.rfft(w_rev, size)
    conv_aw_rev = np.fft.irfft(Fa * Fwr, size)[: n + len(w_rev) - 1]
    idx = (len(w_rev) - 1) - np.arange(n)   # pick the right slice
    grad = 2.0 * conv_aw_rev[idx]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the given
    time budget.  Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # leave a safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep – quick exploration of parametric families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # lengths we are willing to examine during the sweep
    sweep_lengths = [4000, 6000, 8000, 12000, 16000]
    # exponent range – go a little beyond the previous 6.0
    exponents = np.arange(0.1, 8.01, 0.2)
    # exponential parameters (log‑spaced)
    lam_vals = np.logspace(-5, -0.3, num=40)   # from 1e‑5 to ≈0.5

    print("=== Deterministic sweep start ===")
    for n in sweep_lengths:
        if time.time() > deadline:
            break
        for p in exponents:
            # inverse‑distance family
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"  new best (min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

            # product‑distance family
            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"  new best (prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"  new best (exp)   n={n:5d} λ={lam:.5e}  score={sc:.6f}")

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. External construction, if available (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size:
            external = np.maximum(external, 0.0)
            external_sum = external.sum()
            if external_sum > 0.0:
                external /= external_sum
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score = sc
                    best_seq = external.copy()
                    print(f"  using external height_sequence_1  score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – random seed (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 6000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f"  fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Prepare the optimisation problem (fixed final length)
    # ------------------------------------------------------------------
    final_n = 12000                     # a compromise between quality & speed
    best_seq = resample_to_length(best_seq, final_n)
    best_seq /= best_seq.sum()

    # Adam state
    m = np.zeros(final_n, dtype=np.float64)
    v = np.zeros(final_n, dtype=np.float64)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.05                        # base learning rate

    # smoothing (β) schedule
    beta_smooth = 1.0
    beta_growth = 1.001
    max_beta = 80.0

    current = best_seq.copy()
    n = final_n
    last_improve = 0
    iteration = 0

    print("\n=== Stochastic refinement (Adam + smoothed max) start ===")

    while time.time() < deadline:
        iteration += 1

        # ----- 4a. compute convolution and current score -----
        conv = full_convolution_fft(current)
        cur_score = float(2.0 * n * conv.max())

        # keep the globally best vector
        if cur_score < best_score:
            best_score = cur_score
            best_seq = current.copy()
            last_improve = iteration
            print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ----- 4b. update smoothing parameter -----
        beta_smooth = min(beta_smooth * beta_growth, max_beta)

        # ----- 4c. gradient of the smoothed max -----
        grad = smoothed_max_gradient(current, conv, beta_smooth)

        # ----- 4d. Adam update (projected onto simplex) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        current = current - step
        current = np.maximum(current, 0.0)
        current = simplex_projection(current)
        current /= current.sum()                     # re‑normalise (tiny drift)

        # ----- 4e. stagnation handling (re‑heat) -----
        if iteration - last_improve > 50000:
            # add a tiny random bump and reset Adam moments
            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            current /= current.sum()
            m[:] = 0.0
            v[:] = 0.0
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

        # occasional safety break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 5. Final answer
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
```
