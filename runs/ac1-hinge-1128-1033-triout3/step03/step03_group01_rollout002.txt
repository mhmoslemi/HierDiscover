"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises the evaluation score

    2 * n * max_k (a * a)[k]

where * denotes the full linear convolution.  The algorithm works
in three phases:

1.  Deterministic sweep over several parametric “U‑shaped’’ families
    (inverse‑distance, inverse‑product‑distance, exponential decay)
    for a range of lengths and shape parameters.  The best vector from
    this sweep is taken as a seed.

2.  Stochastic refinement on the simplex:
    * projected additive sub‑gradient descent,
    * occasional pairwise mass‑transfer moves,
    * simulated‑annealing acceptance,
    * a periodic linear‑programming “direction’’ step that solves
      max ∑b_j subject to (a * b)[k] ≤ M for the currently active
      convolution constraints (M = max(a * a)),
    * re‑heating after long stagnation.

3.  Return the best sequence discovered (as a plain Python list).  The
    vector is normalised to sum = 1 – the external evaluator rescales
    it automatically, so the objective is invariant to a global scale.

The search respects the supplied time budget (default 1000 s) and
prints progress information.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper functions (top‑level as required)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex:
        { w >= 0 , sum(w) = 1 }.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact value returned by ``evaluate_sequence`` for a vector that
    already satisfies sum(a) == 1 (the denominator becomes 1).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] with respect to a.
    Only indices achieving (or almost achieving) the maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        # each pair contributes 2*a[j]
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on the distance to the nearest edge:
        a[i] = (min(i, n‑1‑i) + 1)^{-p}
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on the product of distances to both edges:
        a[i] = ((i+1)*(n‑i))^{-p}
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """
    Symmetric exponential decay from the edges:
        a[i] = exp(-lam*i) + exp(-lam*(n‑1‑i))
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_conv = full_convolution_fft(current)
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ---- 4e. schedule updates
        temperature *= temp_decay
        eta *= eta_decay

        # ---- 4f. periodic LP‑based direction step
        if iteration % 5000 == 0:
            M = current_conv.max()
            eps_active = M * 1e-9 + 1e-15
            active_idx = np.where(current_conv >= M - eps_active)[0]

            # Build a sparse constraint matrix A such that (current * b)[k] = A[k]·b
            rows = []
            cols = []
            data = []
            for r, k in enumerate(active_idx):
                start_i = max(0, k - (n - 1))
                end_i = min(n - 1, k)
                i_vals = np.arange(start_i, end_i + 1)
                j_vals = k - i_vals
                coeff = current[i_vals]        # a_i appearing in (a*b)[k]
                rows.extend([r] * len(i_vals))
                cols.extend(j_vals.tolist())
                data.extend(coeff.tolist())
            if rows:  # safety – should always be true
                A = sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
                c = -np.ones(n)               # maximise sum(b)  →  minimise -sum(b)
                bounds = [(0, None)] * n
                try:
                    res = linprog(c, A_ub=A, b_ub=np.full(len(active_idx), M),
                                  bounds=bounds, method='highs',
                                  options={'presolve': True})
                    if res.success and res.x.sum() > 0:
                        b = res.x
                        b_norm = b / b.sum()
                        # convex combination of current and the LP direction
                        new = (1 - t_mix) * current + t_mix * b_norm
                        new = simplex_projection(new)
                        new /= new.sum()
                        new_conv = full_convolution_fft(new)
                        new_score = score_from_vector(new)

                        # SA acceptance for the LP move
                        accept_lp = False
                        if new_score < current_score:
                            accept_lp = True
                        else:
                            prob_lp = np.exp((current_score - new_score) / max(temperature, 1e-12))
                            if np.random.rand() < prob_lp:
                                accept_lp = True
                        if accept_lp:
                            current = new
                            current_conv = new_conv
                            current_score = new_score
                            if new_score < best_score:
                                best_score = new_score
                                best_seq = new.copy()
                                last_improve = iteration
                                print(f"Iter {iteration:7d}: LP step new best = {best_score:.6f}")
                except Exception:
                    # LP failures are ignored – we simply continue
                    pass

        # ---- 4g. stagnation handling (re‑heat)
        if iteration - last_improve > 200_000:
            bump = np.random.rand(n) * 1e-4
            current = simplex_projection(current + bump)
            current /= current.sum()
            current_conv = full_convolution_fft(current)
            current_score = score_from_vector(current)
            temperature = 1.0
            eta = 0.4
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
