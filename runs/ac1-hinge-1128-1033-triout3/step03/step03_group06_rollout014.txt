"""
Propose a step‑function (non‑negative vector) that minimises the
evaluation metric used by ``evaluate_sequence``.

Algorithm
---------

1. **Initialisation**
   * If a previous construction ``height_sequence_1`` exists, use it.
   * Otherwise start from a few simple parametric families:
        –  U‑shaped power law  a_i ∝ (min(i,n‑1‑i)+1)^(-p)
        –  product‑law        a_i ∝ ((i+1)*(n-i))^(-p)
        –  exponential decay a_i ∝ exp(-γ·min(i,n‑1‑i))
   * For each family we sweep the free parameter (p or γ) on a modest
     size (n≈4000) and keep the shape that gives the smallest
     ``2·n·max(conv)`` (the scaling factor ``sum(a)`` is 1).

2. **Refinement**
   * The best shape found above (and a few random Dirichlet starts) are
     fed to a mirror‑descent optimiser that minimises a smooth
     approximation of the max‑convolution:
         S_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
     The gradient of S_α is obtained with three FFTs per iteration.
   * The update is multiplicative (exponentiated gradient) and the
     vector is re‑projected onto the probability simplex.
   * The smoothness parameter ``α`` is slowly increased (tightening
     the bound) while the step size ``η`` is adapted according to the
     success of the last move.  Occasionally a small random perturbation
     is injected if no improvement has been seen for a while.
   * The optimisation is run on the final length (default 20 000)
     until the time budget (≈1000 s) is exhausted.

3. **Final evaluation**
   * The best vector seen is evaluated with the official
     ``evaluate_sequence`` and returned as a plain Python list.

The routine prints progress information, respects the time budget and
never performs any file or network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a (real FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution (length 2·n‑1) using supplied FFT size."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max and its gradient (used as a smooth upper bound of max)
# ----------------------------------------------------------------------
def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities w_k = exp(α·conv_k)/∑exp."""
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution w.r.t. a.
    Implements the formula: ∇S = a * w + a * rev(w).
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of self‑convolution
    A = np.fft.rfft(a, n=size)

    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    idx = L - 1 - np.arange(n)                     # 2·n‑2, …, n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (Σx_i=1, x_i≥0)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    if v.ndim != 1:
        raise ValueError("Input must be 1‑D")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Resampling helpers (linear interpolation)
# ----------------------------------------------------------------------
def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a probability vector (sum=1) by linear interpolation."""
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new += 1e-14                     # avoid exact zeros
    new /= new.sum()
    return new


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape: a_i ∝ (dist_to_edge+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)               # distance to nearest edge
    a = (d + 1.0) ** (-p) + 1e-14
    a /= a.sum()
    return a


def product_distribution(n: int, p: float) -> np.ndarray:
    """Product‑law: a_i ∝ ((i+1)*(n-i))^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i * (n + 1 - i)) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exponential_distribution(n: int, gamma: float) -> np.ndarray:
    """Exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def generate_shape(shape_type: str, n: int, param: float) -> np.ndarray:
    """Factory for the three families."""
    if shape_type == "u":
        return u_shaped_distribution(n, param)
    if shape_type == "prod":
        return product_distribution(n, param)
    if shape_type == "exp":
        return exponential_distribution(n, param)
    raise ValueError(f"unknown shape_type {shape_type}")


# ----------------------------------------------------------------------
# Mirror‑descent optimisation of a single starting vector
# ----------------------------------------------------------------------
def refine_sequence(a_init: np.ndarray,
                    max_iters: int,
                    deadline: float,
                    debug_tag: str = "") -> tuple[np.ndarray, float]:
    """
    Runs the multiplicative mirror‑descent optimiser starting from ``a_init``.
    Returns the best vector found and its approx. score 2·n·max(conv).
    ``debug_tag`` is printed in progress messages.
    """
    a = a_init.copy()
    n = a.shape[0]
    fft_size = 1 << ((3 * n - 2).bit_length())

    # optimiser hyper‑parameters
    step_eta = 0.4
    eta_min, eta_max = 1e-6, 5.0
    alpha = 2.0
    alpha_max = 200.0
    alpha_mul = 1.001

    best_local_seq = a.copy()
    best_local_score = 2.0 * n * max_conv_fft(a)

    last_improve = 0
    iteration = 0

    while iteration < max_iters and time.time() < deadline:
        iteration += 1

        # current convolution & score
        conv = full_conv_fft(a, fft_size)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep the best seen locally
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
            last_improve = iteration
            print(f"[{debug_tag}] iter {iteration:5d} – new local best: {cur_score:.6f}")

        w = softmax_weights(conv, alpha)

        # gradient of the smooth max
        grad = gradient_softmax(a, w, fft_size)

        exp_arg = -step_eta * grad
        exp_arg = np.clip(exp_arg, -50.0, 50.0)   # avoid overflow
        a_candidate = a * np.exp(exp_arg)
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s > 0:
            a_candidate /= s
        else:
            a_candidate = a.copy()

        # evaluate candidate (exact max)
        conv_cand = full_conv_fft(a_candidate, fft_size)
        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        # acceptance rule
        if cand_score < cur_score:
            a = a_candidate
            step_eta = min(step_eta * 1.02, eta_max)
        else:
            step_eta = max(step_eta * 0.90, eta_min)

        # tighten the smooth‑max approximation
        alpha = min(alpha * alpha_mul, alpha_max)

        # occasional random jitter if stagnating
        if iteration - last_improve > 800:
            eps = 0.02 * np.random.rand(n)
            a = a + eps * a.mean()
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve = iteration
            step_eta = 0.3
            print(f"[{debug_tag}]  jitter injected at iter {iteration}")

        # cheap progress report
        if iteration % 500 == 0:
            print(f"[{debug_tag}] iter {iteration:5d}: score≈{cur_score:.6f}, α={alpha:.1f}, η={step_eta:.4f}")

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` metric.  The function respects the supplied
    time budget (seconds) and returns the best list of floats found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0.   Take the previously supplied construction if it exists
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            if external.sum() < 1e-12:
                external = np.ones_like(external) / external.size
            external /= external.sum()
            cur_score = 2.0 * external.size * max_conv_fft(external)
            best_seq = external.copy()
            best_score = cur_score
            print(f"Seeded from height_sequence_1: n={external.size}, score≈{cur_score:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.   Fast parametric sweep (small n) to obtain a good shape
    # ------------------------------------------------------------------
    if best_seq is None:
        # start from a simple U‑shape if no seed is available
        best_seq = u_shaped_distribution(2000, p=0.8)
        best_score = 2.0 * best_seq.size * max_conv_fft(best_seq)
        print(f"Fallback init: n={best_seq.size}, score≈{best_score:.6f}")

    # parameter grids for the three families
    param_grid = {
        "u":   np.linspace(0.2, 1.8, 16),          # p‑values
        "prod":np.linspace(0.1, 1.0, 10),          # p‑values
        "exp": np.linspace(0.0, 5.0, 26),          # γ‑values
    }

    # we evaluate everything at a modest size (n≈4000) for speed
    n_probe = 4000
    print("\n=== Parametric sweep ===")
    for shape_type, grid in param_grid.items():
        for param in grid:
            if time.time() >= deadline:
                break
            cand = generate_shape(shape_type, n_probe, float(param))
            cur_max = max_conv_fft(cand)
            cur_score = 2.0 * n_probe * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cand.copy()
                print(f"New sweep best: shape={shape_type}, param={param:.3f}, n={n_probe}, score≈{cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2.   Upsample to the final working length
    # ------------------------------------------------------------------
    final_n = 20000
    if best_seq.shape[0] != final_n:
        best_seq = resample_distribution(best_seq, final_n)

    # ------------------------------------------------------------------
    # 3.   Refine with mirror‑descent starting from several seeds
    # ------------------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_score = 2.0 * final_n * max_conv_fft(global_best_seq)

    restarts = [
        ("seed", best_seq),
        ("rand1", np.random.rand(final_n) + 1e-14),
        ("rand2", np.random.rand(final_n) + 1e-14),
    ]

    print("\n=== Refinement phase ===")
    for tag, init_vec in restarts:
        if time.time() >= deadline:
            break
        # normalise the start vector
        init_vec = np.maximum(init_vec, 0.0)
        if init_vec.sum() == 0:
            init_vec = np.full(final_n, 1.0 / final_n)
        else:
            init_vec /= init_vec.sum()

        # run the optimiser; allocate a modest iteration budget per restart
        max_iters = 3000
        seq, score = refine_sequence(init_vec, max_iters, deadline, debug_tag=tag)

        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(f"\n>>> Global best improved by {tag}: score≈{score:.6f}")

    # ------------------------------------------------------------------
    # 4.   Final official evaluation & return
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(global_best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluate_sequence result: {exact_score:.6f}")
    except Exception as exc:
        print(f"Warning – evaluate_sequence failed: {exc}")
        exact_score = global_best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {global_best_score:.6f}")
    print(f"Best exact score (if evaluated) : {exact_score:.6f}")
    print(f"Best length : {global_best_seq.shape[0]}")
    return global_best_seq.tolist()
