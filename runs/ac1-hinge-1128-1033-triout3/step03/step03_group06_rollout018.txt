"""
    propose_candidate()
    ==================

    Goal
    ----
    Find a non‑negative vector `a` (step‑function) that minimises

        2 * n * max(conv(a,a)) / (sum(a)**2)

    where `conv` is the full discrete self‑convolution.
    The metric is homogeneous, therefore we may (and do) normalise the
    vector to sum to one and simply minimise `2 * n * max(conv)`.

    Strategy
    --------
    1. **Construct a large “Sidon‑like’’ support.**  
       By placing the non‑zero entries on indices whose pairwise sums are
       (almost) unique we keep the worst convolution entry low.
       A greedy, randomised construction quickly yields a support of size
       ≈200 for a length of 20 000 – far larger than the uniform support
       and with a much smaller theoretical bound on the objective.

    2. **Refine the values on that support** with a first‑order optimiser.
       The smooth upper bound
           Sα(a) = (1/α)·log Σ exp(α·conv_k)
       is differentiable; its gradient w.r.t. `a` is obtained cheaply by
       three FFTs per iteration.  An Adam‑style update (additive,
       projected onto the probability simplex) is used – this is more
       robust than the simple multiplicative mirror‑descent from the
       previous baseline.

    3. **Multiple restarts** are performed while time permits, each time
       starting from a fresh random Dirichlet vector (or from the Sidon
       support) and keeping the best solution seen.

    4. The best vector is finally evaluated with the official
       `evaluate_sequence` and returned as a plain Python list.

    The whole routine respects a 1000 s wall‑clock budget and prints
    progress information.  No file or network I/O is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (real FFT, O(n log n))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution using the supplied FFT length."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities w_k = exp(α·conv_k) / Σ exp(·)."""
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution w.r.t. a.
    Implements the formula from the analysis.
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of the convolution
    # FFT of a
    A = np.fft.rfft(a, n=size)

    # FFT of the soft‑max weight vector and its reversal
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # a * w  (linear convolution, length 3·n‑2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    # Build the gradient (see derivation in the analysis)
    idx = L - 1 - np.arange(n)                 # indices for the reversed part
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    O(n log n) algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all are zero after projection – give uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def greedy_sum_unique_set(n: int, rng: np.random.Generator) -> list[int]:
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that all sums i+j
    (i ≤ j, i,j∈S) are distinct.
    The order in which candidates are examined is randomised by `rng`.
    Returns the list of selected indices.
    """
    used = set()
    selected = []
    order = np.arange(n, dtype=np.int64)
    rng.shuffle(order)

    for i in order:
        # quick discard – self‑sum already used?
        if i + i in used:
            continue
        conflict = False
        for j in selected:
            if i + j in used:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        for j in selected:
            used.add(i + j)
        used.add(i + i)
        selected.append(i)
    return selected


def adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_alpha: float = 2.0,
    max_alpha: float = 500.0,
    alpha_mul: float = 1.004,
    eta0: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the soft‑max upper bound.
    Parameters are deliberately generous – the routine runs until the
    `deadline` is reached or no improvement is seen for a while.
    Returns the best vector (normalised) and its associated objective
    value 2·n·max(conv).
    """
    a = a0.copy()
    size = 1 << ((3 * n - 2).bit_length())   # >= 3·n‑2
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_score = float("inf")
    step_eta = eta0
    alpha = start_alpha
    last_improve = time.time()

    while time.time() < deadline:
        t += 1
        conv = full_conv_fft(a, size)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep the best seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"Adam it {t:5d} – new best: {best_score:.6f}")

            # optional early exit
            if best_score < 1.40:         # well below the target
                break

        # soft‑max and its gradient
        w = softmax_weights(conv, alpha)
        grad = gradient_softmax(a, w, size)

        # Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - step_eta * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)                 # stay on the simplex

        # tighten the smooth‑max approximation slowly
        alpha = min(alpha * alpha_mul, max_alpha)

        # simple learning‑rate adaptation
        if cur_score < best_score * 0.9999:
            step_eta = min(step_eta * 1.02, 5.0)
        else:
            step_eta = max(step_eta * 0.94, 1e-4)

        # stop if we have been idle for a while
        if time.time() - last_improve > 30.0:
            break

    return best_a, best_score
