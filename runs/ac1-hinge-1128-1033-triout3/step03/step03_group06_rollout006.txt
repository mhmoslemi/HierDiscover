def full_conv_fft(a, size):
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*a.shape[0]-1]
    return conv
```

def gradient_of_weighted_conv(a, w, size):
    n = len(a)
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)
    conv_aw = np.fft.irfft(A * W, n=size)[:n + len(w) - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[:n + len(w) - 1]
    L = len(w)           # 2n-1
    idx = L - 1 - np.arange(n)
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad
```

"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` metric.

Algorithm
---------

1.  Seed from a previously found construction (`height_sequence_1`) if it
    exists; otherwise start from a simple U‑shaped distribution.

2.  For several candidate lengths (from 1500 up to 6000) perform a fast
    **max‑convolution gradient descent**:
    *   Compute the full self‑convolution with an FFT.
    *   Identify the indices where the convolution attains its maximum.
    *   The sub‑gradient of `max(conv)` is the average gradient of those
        active convolutions; this gradient is obtained by two FFT‑based
        convolutions (O(n log n)).
    *   Update the probability vector by a multiplicative mirror‑descent
        step `a ← a·exp(-η·grad)` and project back onto the simplex.
    *   The step size follows `η = η0 / sqrt(t)`.  Improvements are kept
        as the global best; the loop stops when time runs out or the
        target score `1.503` is reached.

3.  After the length loop a final long refinement (up to the remaining
    budget) is performed on the best sequence found.

4.  The best sequence is evaluated with the official `evaluate_sequence`
    and returned.

All heavy work is done with FFTs; the routine respects the 1000 s
budget, prints progress messages and never uses file or network I/O.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ------------------------------------------------------------

def max_conv_fft(a: np.ndarray) -> float:
    """
    Maximum entry of the self‑convolution a*a using a real FFT.
    a must be a 1‑D non‑negative array (its sum is irrelevant for the
    computation of the maximum).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·n‑1) using a supplied FFT size.
    Returned array is real.
    """
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def gradient_of_weighted_conv(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    w – weight vector of length 2·n‑1 (sum to 1).
    size – FFT length, must be ≥ 3·n‑2.
    Returns a vector of length n.
    """
    n = a.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    # FFT of a
    A = np.fft.rfft(a, n=size)

    # FFT of w and the reversed w
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # Convolution a * w  (length n+L‑1 = 3·n‑2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    # Gradient = conv_aw[:n] + conv_aw_rev[L-1 - np.arange(n)]
    idx = L - 1 - np.arange(n)                     # 2·n‑2, …, n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries become zero after projection – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resample a probability vector `old` (sum=1) to length `new_n`
    using linear interpolation and renormalise.
    """
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new += 1e-14          # avoid exact zeros
    new /= new.sum()
    return new


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Simple symmetric U‑shape: a_i ∝ (min(i, n‑1‑i)+1)^{-p}.
    Returns a probability vector (sum = 1).
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)               # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

# Main search routine
# ------------------------------------------------------------

def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects the given time budget and
    returns the best list of floats it found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # --------------------------------------------------------
    # 0.  Initialise from a possible external seed
    # --------------------------------------------------------
    best_seq = None
    best_score = np.inf
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            external_sum = external.sum()
            if external_sum < 1e-12:
                external = np.ones_like(external) / external.size
            else:
                external = external / external_sum
            cur_score = 2.0 * external.size * max_conv_fft(external)
            best_seq = external.copy()
            best_score = cur_score
            print(f"[Seed] length={external.size}, approx_score≈{cur_score:.6f}")
    except NameError:
        # No external variable – we'll start from scratch
        pass
    except Exception as exc:
        print(f"[Seed] error while processing external seed: {exc}")

    # --------------------------------------------------------
    # 1.  Fallback initialisation if needed
    # --------------------------------------------------------
    if best_seq is None:
        n0 = 2000
        best_seq = u_shaped_distribution(n0, p=0.8)
        best_score = 2.0 * n0 * max_conv_fft(best_seq)
        print(f"[Init] fallback length={n0}, approx_score≈{best_score:.6f}")

    # --------------------------------------------------------
    # 2.  Lengths to explore – a moderate range is enough for the target
    # --------------------------------------------------------
    lengths_to_try = [1500, 2000, 2500, 3000, 4000, 5000, 6000]
    lengths_to_try = [n for n in lengths_to_try if n > 0]

    # --------------------------------------------------------
    # 3.  Main optimisation: max‑convolution gradient descent
    # --------------------------------------------------------
    for n in lengths_to_try:
        if time.time() >= deadline:
            break
        print(f"\n--- Optimising length n={n} ---")

        # Resample current best to the new length
        if best_seq.shape[0] != n:
            a = resample_distribution(best_seq, n)
        else:
            a = best_seq.copy()
        a = a / a.sum()                     # ensure a is a probability vector

        # FFT sizes
        conv_fft_size = 1 << ((2 * n - 1).bit_length())
        grad_fft_size = 1 << ((3 * n - 2).bit_length())

        # hyper‑parameters
        eta0 = 0.8                          # initial step‑size multiplier
        max_iters = 8000

        for it in range(1, max_iters + 1):
            if time.time() >= deadline:
                break

            # --- current score ------------------------------------------------
            conv = full_conv_fft(a, conv_fft_size)
            maxc = conv.max()
            score = 2.0 * n * maxc

            # update global best
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[Iter {it:4d}] n={n} NEW BEST ≈ {score:.6f}")
                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    break

            # --- gradient of max(conv) (sub‑gradient) -------------------------
            # identify the active set (entries attaining the maximum)
            tol = 1e-12
            active = conv >= maxc - tol
            n_active = int(active.sum())
            if n_active == 0:
                break
            w = np.zeros_like(conv)
            w[active] = 1.0 / n_active

            grad = gradient_of_weighted_conv(a, w, grad_fft_size)

            # --- step‑size schedule -------------------------------------------
            eta = eta0 / np.sqrt(it)

            # --- multiplicative mirror‑descent update -------------------------
            exp_arg = -eta * grad
            exp_arg = np.clip(exp_arg, -50.0, 50.0)   # protect against overflow
            a_candidate = a * np.exp(exp_arg)
            s = a_candidate.sum()
            if s > 0:
                a_candidate /= s
            else:
                a_candidate = a.copy()               # fall back if something went wrong

            # --- candidate evaluation (exact max) -----------------------------
            conv_cand = full_conv_fft(a_candidate, conv_fft_size)
            cand_score = 2.0 * n * conv_cand.max()

            # --- accept if we improved, otherwise keep the old vector ---------
            if cand_score < score:
                a = a_candidate
                # (optional) make the step a little more aggressive
                eta0 = min(eta0 * 1.03, 5.0)
            else:
                # failed step – shrink the base step size a little
                eta0 = max(eta0 * 0.90, 5e-4)

            # occasional progress report
            if it % 1000 == 0:
                print(f"[Iter {it:4d}] score≈{score:.6f}, eta0≈{eta0:.4f}")

        # end of length‑specific loop – continue with next length
        if best_score < 1.503:
            break

    # --------------------------------------------------------
    # 4.  Final long refinement on the best sequence found so far
    # --------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        n = best_seq.shape[0]
        conv_fft_size = 1 << ((2 * n - 1).bit_length())
        grad_fft_size = 1 << ((3 * n - 2).bit_length())
        eta0 = 0.6
        extra_iters = 15000
        print(f"\n[Refine] Starting final refinement on length {n}")
        for it in range(1, extra_iters + 1):
            if time.time() >= deadline:
                break

            conv = full_conv_fft(best_seq, conv_fft_size)
            maxc = conv.max()
            score = 2.0 * n * maxc
            if score < best_score:
                best_score = score
                print(f"[Refine {it:5d}] NEW BEST ≈ {score:.6f}")
                if best_score < 1.503:
                    print("Target reached during refinement.")
                    break

            # gradient of max
            tol = 1e-12
            active = conv >= maxc - tol
            w = np.zeros_like(conv)
            n_active = int(active.sum())
            if n_active == 0:
                break
            w[active] = 1.0 / n_active
            grad = gradient_of_weighted_conv(best_seq, w, grad_fft_size)

            # step size decays slowly
            eta = eta0 / np.sqrt(it)

            exp_arg = -eta * grad
            exp_arg = np.clip(exp_arg, -50.0, 50.0)
            cand = best_seq * np.exp(exp_arg)
            cand /= cand.sum()
            cand_score = 2.0 * n * full_conv_fft(cand, conv_fft_size).max()
            if cand_score < score:
                best_seq = cand
                eta0 = min(eta0 * 1.02, 5.0)
            else:
                eta0 = max(eta0 * 0.93, 5e-5)

            if it % 3000 == 0:
                print(f"[Refine {it:5d}] score≈{score:.6f}, eta0≈{eta0:.5f}")

    # --------------------------------------------------------
    # 5.  Final official evaluation & return
    # --------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {best_score:.6f}")
    print(f"Best length            : {best_seq.shape[0]}")
    return best_seq.tolist()
