"""
Hybrid search for low‑score non‑negative step functions.

1.  Deterministic sweep of several parametric families to obtain a good
    seed (U‑shapes, Beta, exponential, raised‑cosine, harmonic).
2.  Refined stochastic optimisation that minimises a *smooth* approximation
    of the max‑convolution using projected (additive) and multiplicative
    (mirror‑descent) updates.  The smooth max is the log‑sum‑exp with a
    temperature β that is slowly increased, giving a more informative
    gradient than the hard active‑set sub‑gradient.
3.  Simple simulated‑annealing acceptance and occasional reheating to
    escape plateaus.

All convolutions are performed by FFTs, allowing millions of iterations
within the 1000 s time budget.  The best vector found is returned as a
plain list; the external variable ``height_sequence_1`` is used if it
exists.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using an FFT (real‑valued inputs)."""
    total_len = a.shape[0] + b.shape[0] - 1
    size = 1 << (total_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[:total_len]


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (linear) via FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    size = 1 << (total_len - 1).bit_length()
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[:total_len]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is non‑negative, sum=1)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a vector already normalised."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Parametric families (used for the deterministic sweep)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the two edges: ((i+1)*(n-i))^(-p)."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}. a[i] ∝ (i+1)^{α‑1} (n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges: exp(-γ·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def harmonic_left_distribution(n: int, p: float) -> np.ndarray:
    """One‑sided power law: a[i] ∝ (i+1)^{-p} (mass concentrated on the left)."""
    i = np.arange(n) + 1
    a = i.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def raised_cosine_distribution(n: int, alpha: float) -> np.ndarray:
    """Raised‑cosine shape a[i] = (1‑cos(π·i/(n‑1)))^{α}."""
    i = np.arange(n)
    a = (1.0 - np.cos(np.pi * i / (n - 1))) ** alpha + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence (list of floats) discovered before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # --------------------------------------------------------------
    # 1. Deterministic sweep over several parametric families
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    lengths = [4000, 8000, 12000, 16000]          # modest lengths for the sweep
    p_grid = np.linspace(0.2, 2.0, 19)             # for U‑shapes
    beta_grid = np.linspace(0.3, 2.0, 18)          # symmetric beta
    gamma_grid = np.linspace(0.001, 0.02, 20)      # exponential decay
    alpha_cos_grid = np.linspace(0.2, 2.0, 19)     # raised‑cosine
    p_harm_grid = np.linspace(0.2, 2.0, 19)       # harmonic left

    print("=== Deterministic sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # U‑shape families
        for p in p_grid:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # Beta family (symmetric)
        for a_param in beta_grid:
            a = beta_distribution(n, a_param, a_param)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (beta)    n={n:5d} α={a_param:4.2f}  score={sc:.6f}")

        # Exponential decay
        for gamma in gamma_grid:
            a = exp_decay_distribution(n, gamma)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (exp)     n={n:5d} γ={gamma:5.3f}  score={sc:.6f}")

        # Raised‑cosine family
        for alpha in alpha_cos_grid:
            a = raised_cosine_distribution(n, alpha)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (rcos)    n={n:5d} α={alpha:4.2f}  score={sc:.6f}")

        # Harmonic left‑biased
        for p_h in p_harm_grid:
            a = harmonic_left_distribution(n, p_h)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (harmL)   n={n:5d} p={p_h:4.2f}  score={sc:.6f}")

        # occasional time check
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 2. Use external construction if available
    # --------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            total = external.sum()
            if total > 0.0:
                external /= total
                sc_ext = score_from_vector(external)
                if sc_ext < best_score:
                    best_score, best_seq = sc_ext, external.copy()
                    print(f" using external height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 3. Fallback (very unlikely)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 8000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Refined stochastic optimisation (smooth‑max gradient)
    # --------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]
    current_score = best_score

    # hyper‑parameters
    temperature = 1.0
    cooling_rate = 0.9997
    base_step = 0.25
    step_size = base_step
    beta = 5.0                         # temperature of the soft‑max (smoothness)
    beta_growth = 1.001                # slowly make the approximation sharper
    max_beta = 200.0
    iteration = 0
    last_improve_iter = 0

    print("\n=== Starting stochastic refinement ===")
    while time.time() < deadline:
        iteration += 1

        # ----------------------------------------------------------
        # a) current convolution and smooth‑max weights
        # ----------------------------------------------------------
        conv = full_convolution_fft(current)          # shape (2n-1,)
        max_c = conv.max()
        # soft‑max weights (numerically stable)
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # ----------------------------------------------------------
        # b) gradient of the smooth max (2 * (a * w_rev) evaluated at indices)
        # ----------------------------------------------------------
        w_rev = w[::-1]
        conv_rev = conv_fft(current, w_rev)            # length 3n-2
        idx = (2 * n - 2) - np.arange(n)               # indices for gradient
        grad = 2.0 * conv_rev[idx]

        # ----------------------------------------------------------
        # c) stochastic move (additive or multiplicative)
        # ----------------------------------------------------------
        r = np.random.rand()
        if r < 0.35:
            # multiplicative (mirror‑descent) update
            cand = current * np.exp(-step_size * grad)
            cand_sum = cand.sum()
            if cand_sum == 0.0:
                continue
            cand = cand / cand_sum
        else:
            # additive projected sub‑gradient step
            cand = current - step_size * grad
            cand = np.maximum(cand, 0.0)
            cand = simplex_projection(cand)

        # ----------------------------------------------------------
        # d) evaluate candidate (fast internal score)
        # ----------------------------------------------------------
        cand_score = score_from_vector(cand)

        # ----------------------------------------------------------
        # e) simulated‑annealing acceptance
        # ----------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve_iter = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # f) anneal temperature, step size and smoothness parameter
        # ----------------------------------------------------------
        temperature *= cooling_rate
        step_size *= cooling_rate
        beta = min(beta * beta_growth, max_beta)

        # ----------------------------------------------------------
        # g) stagnation handling (re‑heat)
        # ----------------------------------------------------------
        if iteration - last_improve_iter > 80000:
            # add a tiny random bump and reset annealing parameters
            bump = np.random.rand(n) * 5e-4
            current = current + bump
            current = simplex_projection(current)
            current_score = score_from_vector(current)
            last_improve_iter = iteration
            temperature = 1.0
            step_size = base_step
            beta = 5.0
            print(f" *** Re‑heat at iter {iteration}")

    # --------------------------------------------------------------
    # 5. Return the best sequence discovered
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator will clip, trim and renormalise again,
    # so we return the raw simplex‑normalised vector.
    return best_seq.tolist()
