assistantfinal**Solution Overview**

The evaluation metric is  

\[
F(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a*a` is the self‑convolution of the non‑negative vector `a`
(`n = len(a)`).  The value is invariant under scaling, therefore we may
always keep the vector on the probability simplex (`∑a_i = 1`).  
The problem is to minimise the infinity‑norm of the convolution
(`max_k (a*a)_k`).

The code below performs a **global stochastic search** with many
restarts and a **local Adam‑optimiser** that works on the
*soft‑max* upper bound

\[
S_\alpha(a)=\frac{1}{\alpha}\log\!\Bigl(\sum_k
e^{\alpha\,(a*a)_k}\Bigr) .
\]

`Sα` is a smooth approximation of `max(a*a)`.  
Its gradient can be written in closed form using three FFTs, which
keeps each optimisation step `O(n log n)`.

Main ideas added compared with the previous attempt  

* **Random restarts** with different lengths and very different initial
  shapes (U‑shaped, Dirichlet, random spikes).  
* Optimisation is performed **in log‑space** (`p = log a`) and the
  probabilities are obtained by a stable soft‑max, guaranteeing the
  simplex constraint without an extra projection step.  
* **Adam** (adaptive moments) replaces the simple multiplicative step
  size, giving automatic per‑coordinate learning‑rate adaptation.  
* A **schedule for the smoothing parameter** `α` (slowly increased up to
  a large value) and occasional **Gaussian noise** added to the logits
  help the optimiser escape shallow local minima.  
* After the stochastic phase a **dedicated refinement** phase runs on
  the current best vector with a much larger `α` (≈ 2000) and a tiny
  learning‑rate, trying to squeeze out the last improvements.  

All helper functions are top‑level, there is no file or network I/O,
and the whole routine stops automatically when the 1000 s budget is
exhausted.

"""
propose_candidate – stochastic search + Adam on a smooth max‑convolution
=======================================================================

The routine searches for a non‑negative vector a (∑a_i = 1) that
minimises the evaluation metric used by `evaluate_sequence`.  It
combines many random restarts (different lengths and initial shapes)
with a local Adam optimiser on the soft‑max approximation of the
convolution peak.  The algorithm respects the given time budget and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·n‑1) using FFT of length ``size``.
    The result is real and returned as a 1‑D array.
    """
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * a.shape[0] - 1]


def max_conv_fft(a: np.ndarray, size: int) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return float(full_conv_fft(a, size).max())


# ----------------------------------------------------------------------
# Soft‑max and its gradient
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max probabilities w_k = exp(α·conv_k) / Σ exp(α·conv).
    Numerically stabilised by subtracting the maximum entry.
    """
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the self‑convolution with respect to ``a``.
    Implements the formula   ∇S = 2·(a * w) (using FFTs).
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of w  (conv size)

    # FFT of a (already real)
    A = np.fft.rfft(a, n=size)

    # FFT of w and of its reversal (needed for the two symmetric terms)
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # Convolution a * w  (length n + L – 1 = 3·n – 2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    # Gradient = first n entries + the mirrored tail
    idx = L - 1 - np.arange(n)               # 2·n‑2 … n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


# ----------------------------------------------------------------------
# Simple probability utilities
# ----------------------------------------------------------------------


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max: returns probabilities that sum to 1."""
    shift = logits - np.max(logits)
    exps = np.exp(shift)
    return exps / exps.sum()


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law “U” shape (used as a quick fallback)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

            s = ext.sum()
            if s < 1e-12:
                ext = np.ones_like(ext) / ext.size
                s = 1.0
            ext /= s
            # Compute a quick bound (2·n·max) – no need for full evaluate_sequence yet
            size = 1 << ((3 * ext.size - 2).bit_length())
            tmp_score = 2.0 * ext.size * max_conv_fft(ext, size)
            best_seq = ext.copy()
            best_score = tmp_score
            print(f"Seeded from height_sequence_1: n={ext.size}, approx_score={tmp_score:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Fallback initialisation if needed
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 8000
        best_seq = u_shaped_distribution(n0, p=0.8)
        size = 1 << ((3 * n0 - 2).bit_length())
        best_score = 2.0 * n0 * max_conv_fft(best_seq, size)
        print(f"Fallback init: n={n0}, approx_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Parameter pools
    # ------------------------------------------------------------------
    # Lengths that will be sampled (larger n gives a larger factor 2·n)
    length_candidates = np.array([5000, 8000, 12000, 16000, 20000, 25000, 30000])

    # Adam hyper‑parameters (common for all runs)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.07                     # base learning rate (will be scaled)

    # Smoothing schedule for the soft‑max
    alpha_start = 5.0
    alpha_mul = 1.025
    alpha_max = 2000.0

    # ------------------------------------------------------------------
    # 3.  Main stochastic‑restart loop
    # ------------------------------------------------------------------
    runs = 0
    while time.time() < deadline:
        runs += 1

        # --------------------------------------------------------------
        # Choose a length (biased towards larger sizes)
        # --------------------------------------------------------------
        n = int(np.random.choice(length_candidates))
        fft_size = 1 << ((3 * n - 2).bit_length())   # >= 3·n‑2

        # --------------------------------------------------------------
        # Random initial vector (mixture of U‑shape, Dirichlet, spikes)
        # --------------------------------------------------------------
        if np.random.rand() < 0.5:
            # U‑shape with a random exponent
            a0 = u_shaped_distribution(n, p=np.random.uniform(0.4, 1.2))
        else:
            # Dirichlet‑like draw (gamma with shape < 1 gives more sparsity)
            shape = np.random.uniform(0.2, 0.8)
            raw = np.random.gamma(shape, 1.0, size=n)
            a0 = raw / raw.sum()

        # Occasionally inject a few random spikes
        if np.random.rand() < 0.2:
            spikes = np.random.choice(n, size=int(0.01 * n), replace=False)
            a0[spikes] += np.random.rand(spikes.size) * 0.5
            a0 = np.maximum(a0, 1e-14)

        # Normalise to the simplex and work in log‑space
        a0 = np.maximum(a0, 1e-14)
        a0 /= a0.sum()
        p = np.log(a0)                     # logits (unconstrained)

        # Adam accumulators
        m = np.zeros_like(p)
        v = np.zeros_like(p)

        # Smoothing parameter for this run
        alpha = alpha_start

        # Local loop
        max_iters = 3000
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # ---------------------------------------------------------
            # Current probability vector and its convolution
            # ---------------------------------------------------------
            a = softmax_logits(p)                # stays on the simplex
            conv = full_conv_fft(a, fft_size)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # Update global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[run {runs:02d}] n={n}, iter={it:04d} – new best "
                      f"score≈{cur_score:.6f}")

                # Early exit if the target is already beaten
                if best_score < 1.503:
                    print("Target reached – terminating search.")
                    break

            # ---------------------------------------------------------
            # Soft‑max weights and gradient w.r.t. a
            # ---------------------------------------------------------
            w = softmax_weights(conv, alpha)
            grad_a = gradient_softmax(a, w, fft_size)      # ∂S/∂a

            # ---------------------------------------------------------
            # Convert gradient to the log‑space (logits) using the Jacobian
            # of the soft‑max mapping.
            # ---------------------------------------------------------
            dot_ag = np.dot(a, grad_a)                     # a·∇S
            grad_p = a * (grad_a - dot_ag)                 # ∂S/∂p

            # ---------------------------------------------------------
            # Adam update on the logits (we move opposite to the gradient)
            # ---------------------------------------------------------
            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_p
            v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # Simple 1/√t decay for the learning‑rate keeps the steps
            # aggressive at the beginning and conservative later.
            lr = base_lr / np.sqrt(t)

            p -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ---------------------------------------------------------
            # Exploration boost – occasional random jitter
            # ---------------------------------------------------------
            if (it + 1) % 250 == 0:
                p += np.random.normal(scale=0.015, size=n)

            # ---------------------------------------------------------
            # Increase smoothing (tighten the bound)
            # ---------------------------------------------------------
            alpha = min(alpha * alpha_mul, alpha_max)

            # ---------------------------------------------------------
            # Ultra‑short stop if no improvement for a while
            # ---------------------------------------------------------
            if (it + 1) % 1000 == 0:
                print(f"[run {runs:02d}] n={n}, iter={it+1:04d} – "
                      f"score≈{cur_score:.6f}, α={alpha:.1f}")

        # early global termination
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 4.  Final refinement on the best sequence found so far
    # ------------------------------------------------------------------
    if best_seq is not None:
        n_best = best_seq.shape[0]
        fft_size = 1 << ((3 * n_best - 2).bit_length())

        # Initialise logits from the best vector
        p = np.log(np.maximum(best_seq, 1e-30))
        m = np.zeros_like(p)
        v = np.zeros_like(p)

        # Strong smoothing and a tiny learning‑rate
        alpha = 500.0
        alpha_mul = 1.0          # keep it fixed
        base_lr = 0.02

        for it in range(1500):
            if time.time() >= deadline:
                break

            a = softmax_logits(p)
            conv = full_conv_fft(a, fft_size)
            cur_max = conv.max()
            cur_score = 2.0 * n_best * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"Refine iter {it:04d} – improved score≈{cur_score:.6f}")

            w = softmax_weights(conv, alpha)
            grad_a = gradient_softmax(a, w, fft_size)

            dot_ag = np.dot(a, grad_a)
            grad_p = a * (grad_a - dot_ag)

            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_p
            v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = base_lr / np.sqrt(t)
            p -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # tiny jitter every 300 steps
            if (it + 1) % 300 == 0:
                p += np.random.normal(scale=0.005, size=n_best)

    # ------------------------------------------------------------------
    # 5.  Official evaluation and return
    # ------------------------------------------------------------------
    try:
        exact = evaluate_sequence(best_seq.tolist())          # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact:.6f}")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")
        exact = best_score

    print("\n=== Search finished ===")
    print(f"Best (approx.) score : {best_score:.6f}")
    print(f"Best length          : {best_seq.shape[0] if best_seq is not None else 'N/A'}")
    print(f"Total runs performed : {runs}")

    return best_seq.tolist()
