"""
propose_candidate
-----------------
Search for a non‑negative step function that minimizes the value returned by
``evaluate_sequence``.  The algorithm combines three ideas:

1. **Arcsine start** – a known shape that already flattens the self‑convolution.
2. **Linear‑programming step** – given a current distribution `f`,
   we solve the LP

        max   Σ g_i
        s.t.  (f * g)_k ≤ M   for all k,
              g_i ≥ 0,

   where `M = max(f*f)`.  The optimizer returns a vector `g`
   with the largest possible total mass that still respects the current
   maximal convolution height.  Mixing `f` with the normalised `g`
   (convex combination) provably never increases `M`.

3. **Smooth Adam refinement** – after a few LP updates we switch to a
   differentiable log‑sum‑exp approximation of the max and run Adam
   optimisation on the simplex.  Finally a tiny pair‑wise swap hill‑climber
   cleans up the remaining tiny improvements.

The whole procedure respects a 1000 s time budget, prints progress,
and always returns a valid sequence (non‑negative, entries ≤ 1000).

# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σ x = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

            # Build the (2n‑1) × n constraint matrix A such that (A @ g)_k = (f * g)_k
            col_idx = np.arange(n)[:, None]               # shape (n,1)
            row_idx = col_idx + np.arange(n)               # shape (n,n)
            A = np.zeros((2 * n - 1, n), dtype=float)
            A[row_idx, col_idx] = f                       # broadcast f into each column

            # Solve the LP: maximise Σ g_i   ⇔   minimise –Σ g_i
            c = -np.ones(n, dtype=float)
            bounds = [(0.0, None)] * n
            b_ub = np.full(2 * n - 1, M, dtype=float)

            lp_res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'disp': False})
            if not lp_res.success:
                print(f"  LP failed (outer {outer + 1}): {lp_res.message}")
                break

            g0 = lp_res.x
            sum_g0 = g0.sum()
            if sum_g0 <= 1e-12:
                print("  LP returned zero mass – aborting this start.")
                break

            # Normalise g to keep the simplex constraint (sum = 1)
            g = g0 / sum_g0

            # Convex combination (the mixing weight t slowly decays)
            f = (1.0 - t) * f + t * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()   # re‑project onto the simplex (numerical safety)

            # Evaluate
            cur_score = evaluate_sequence(f.tolist())
            print(f"  LP‑mix {outer + 1:02d} – t={t:.4f} – score={cur_score:.6f}")

            # Update global best
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_f = f.copy()
                print(f"    ** New global best: {best_score:.6f}")

            # Simple schedule for t (decrease if no progress)
            if cur_score < best_score + 1e-8:
                t = min(t * 1.10, 0.60)
            else:
                t = max(t * 0.70, 0.01)

            # Early stop if we already reached the target region
            if best_score <= 1.503 + 5e-4:
                break

        # ----------------------------------------------------------
        # 2) Adam optimisation on a smooth approximation of max
        # ----------------------------------------------------------
        # Use the best vector found for this start (or the current f)
        f_adam = best_f.copy() if best_f is not None else f.copy()

        # Adam state
        m1 = np.zeros_like(f_adam)
        m2 = np.zeros_like(f_adam)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lr = 0.05

        # Smoothing schedule
        alpha = 5.0          # start smoothing factor
        alpha_mul = 1.01
        alpha_max = 50.0

        for it in range(8000):
            if time.time() >= deadline:
                break

            # FFT‑based self‑convolution
            f_fft = np.fft.rfft(f_adam, n=m_fft)
            conv = np.fft.irfft(f_fft * f_fft, n=m_fft)[: 2 * n - 1]

            # log‑sum‑exp smooth max
            z = alpha * conv
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)          # stabilised exponentials
            w = exp_z / np.sum(exp_z)          # soft‑max weights (sum to 1)

            # Gradient of the smooth max w.r.t. f
            w_rev = w[::-1]
            w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
            corr_full = np.fft.irfft(f_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]
            idx = 2 * n - 2 - np.arange(n)      # mapping back to gradient components
            grad = 2.0 * corr_full[idx]

            # Adam update
            m1 = beta1 * m1 + (1.0 - beta1) * grad
            m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

            m_hat = m1 / (1.0 - beta1 ** (it + 1))
            v_hat = m2 / (1.0 - beta2 ** (it + 1))

            step_vec = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            f_adam = f_adam - step_vec
            f_adam = _simplex_projection(f_adam)   # stay on the probability simplex

            # increase smoothing (approach true max)
            alpha = min(alpha * alpha_mul, alpha_max)

            # occasional evaluation
            if (it % 1000) == 0 or it == 7999:
                cur_score = evaluate_sequence(f_adam.tolist())
                if cur_score < best_score - 1e-9:
                    best_score = cur_score
                    best_f = f_adam.copy()
                    print(f"  Adam iter {it:5d} – new best {best_score:.6f}")
                if best_score <= 1.503 + 1e-4:
                    break

        # ----------------------------------------------------------
        # 3) Tiny local‑swap hill‑climber
        # ----------------------------------------------------------
        if best_f is None:
            best_f = f_adam.copy()
        p = best_f.copy()
        cur_score = evaluate_sequence(p.tolist())
        swap_attempts = 3000
        for _ in range(swap_attempts):
            if time.time() >= deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            eps_move = 1e-4 * rng.random()
            if p[i] <= eps_move:
                continue
            new_p = p.copy()
            new_p[i] -= eps_move
            new_p[j] += eps_move
            new_score = evaluate_sequence(new_p.tolist())
            if new_score < cur_score - 1e-12:
                p = new_p
                cur_score = new_score
                best_f = p.copy()
                best_score = cur_score
                print(f"  Swap improvement → {best_score:.6f}")
                if best_score <= 1.503:
                    break

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_f is None:
        best_f = np.full(n, 1.0 / n)

    print(f"\nFinal best score: {best_score:.6f}")
    # Clip to the limits enforced by evaluate_sequence and return as list
    return np.clip(best_f, 0.0, 1000.0).tolist()
```

This implementation follows the three‑phase strategy described above, respects the time budget, and always returns a valid non‑negative sequence.
