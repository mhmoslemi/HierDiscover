"""
Propose a candidate step‑function using Adam on the soft‑max approximation
of the L∞‑norm of the self‑convolution.  The variables are kept on the
probability simplex via a softmax parameterisation, guaranteeing non‑negative
coefficients and unit sum.  The algorithm starts from the best known sequence
(if provided) and performs many cheap FFT‑based updates within the time budget.
It anneals a temperature τ that turns the soft‑max into a hard max, and uses
a linearly decaying learning‑rate.  Random jitter occasionally perturbs the
log‑weights to escape plateaux.  The best sequence seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Eliminate tiny negative rounding errors.
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient from soft‑max weights
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    For a probability vector a (len=n) and weight vector w (len=2n-1) that
    sums to 1, compute the gradient of Σ_k w_k·(a*a)[k] with respect
    to a.  The formula is g_i = 2·Σ_k w_k·a_{k-i}.
    Implemented via a single FFT‑based convolution.
    """
    n = a.shape[0]
    # reverse weights
    u = w[::-1]
    conv = _full_convolution_fft(u, a)               # length 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)                       # n indices
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Softmax for probabilities (used to map log‑weights to a)
# ----------------------------------------------------------------------
def _softmax(p: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    maxp = np.max(p)
    exp_p = np.exp(p - maxp)
    return exp_p / exp_p.sum()
