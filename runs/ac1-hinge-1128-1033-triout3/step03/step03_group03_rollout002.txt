Propose a low‑score step‑function.

The score of a non‑negative vector a of length n (with
Σ a = 1) is 2·n·max(conv(a,a)), where conv is the
discrete self‑convolution.  The optimisation problem is
highly non‑convex.  This implementation combines three ideas
that proved useful in practice:

1. Initialise with the classic Catalan (constant‑convolution)

2. Perform a mirror‑descent (exponentiated‑gradient) on the

   smooth maximum is

   Its gradient can be written as a single convolution,
   allowing an O(n log n) update.  β is increased gradually so
   that the algorithm first smooths the objective and later
   concentrates on the true maximum.

3. After the main loop we keep the best vector ever seen.
   The loop runs until the wall‑clock budget expires, with
   occasional random restarts when no improvement is observed.

All operations stay on the probability simplex (a_i ≥ 0,
Σ a_i = 1), so the evaluation function is invariant to scaling.
The final sequence is clamped to the limits imposed by the
judge (0 ≤ a_i ≤ 1000) and returned.

The method works comfortably within the 1000 s time limit and

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast convolution using FFT (real‑valued input).
# ----------------------------------------------------------------------
def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*y using an FFT of length power‑of‑2."""
    size = x.size + y.size - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(fx * fy, nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Helper: soft‑max weights for a vector (numerically stable).
# ----------------------------------------------------------------------
def _softmax_weights(v: np.ndarray, beta: float) -> np.ndarray:
    """Return soft‑max weights w_i ∝ exp(beta * v_i)."""
    # subtract the maximum for numerical stability
    vmax = v.max()
    ex = np.exp(beta * (v - vmax))
    return ex / ex.sum()


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The algorithm runs a soft‑max exponentiated‑gradient
    descent on the maximal self‑convolution entry, with simple restart
    logic, and returns the best sequence found before the time budget ends.
    """
    # --------------------------------------------------------------
    # 1) Timing control
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0     # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 2) Problem size – a larger n gives more freedom but costs a bit
    #    more time per iteration.  20000 is a good compromise for the
    #    1000 s budget.
    # --------------------------------------------------------------
    n = 20000

    # --------------------------------------------------------------
    # 3) Initialise with the truncated Catalan sequence.
    #    a_k = (2k‑1)!!/(2k)!!  (k≥0)   –>  a_k ≈ 1/√(πk)
    # --------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    cur = 1.0
    a[0] = cur
    for k in range(1, n):
        cur = cur * (2 * k - 1) / (2 * k)
        a[k] = cur
    a = a / a.sum()                     # project onto the simplex

    # --------------------------------------------------------------
    # 4) Track the best solution ever seen.
    # --------------------------------------------------------------
    conv = _convolve_fft(a, a)
    best_val = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[init] n={n}, initial_score={best_val:.10f}")

    # --------------------------------------------------------------
    # 5) Hyper‑parameters
    # --------------------------------------------------------------
    eta = 1.0               # initial learning rate for exponentiated update
    min_eta = 1e-4
    eta_decay = 0.9998

    beta = 5.0              # start with a smooth approximation of max
    beta_inc = 0.02
    beta_max = 200.0

    it = 0
    no_improve = 0
    max_no_improve = 8000   # after this many stale iterations we restart

    # --------------------------------------------------------------
    # 6) Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- a) compute convolution and current score -----------------
        conv = _convolve_fft(a, a)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c

        # ----- b) keep the best solution --------------------------------
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[iter {it:6d}] new best {best_val:.10f} (max={max_c:.6e})")
        else:
            no_improve += 1

        # ----- c) soft‑max over convolution entries --------------------
        w = _softmax_weights(conv, beta)          # shape (2n‑1,)

        # ----- d) gradient of the soft‑maxed objective -----------------
        #   ∂S/∂a_i = 2 * (w * a_rev)[i + n‑1]   where a_rev = a[::-1]
        a_rev = a[::-1]
        w_conv_rev = _convolve_fft(w, a_rev)     # length 3n‑2
        g = 2.0 * w_conv_rev[n - 1 : n - 1 + n]   # shape (n,)

        # ----- e) exponentiated‑gradient (mirror descent) ---------------
        #   a_i ← a_i * exp(-η * g_i)  then renormalise.
        #   Clip g to avoid extreme exponentials.
        g_clipped = np.clip(g, -50.0, 50.0)
        factor = np.exp(-eta * g_clipped)
        a = a * factor
        s = a.sum()
        if s == 0.0:                         # numerical safety net
            a = rng.random(n)
            a = a / a.sum()
        else:
            a = a / s

        # ----- f) schedule updates --------------------------------------
        eta = max(min_eta, eta * eta_decay)
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # ----- g) occasional random restart -----------------------------
        if no_improve > max_no_improve:
            a = rng.random(n)
            a = a / a.sum()
            eta = 1.0
            beta = 5.0
            no_improve = 0
            print(f"[restart] iteration {it}")

        # ----- h) keep the loop responsive --------------------------------
        if it % 5000 == 0:
            elapsed = time.time() - (deadline - budget_s + 5.0)
            print(f"[status] iter {it}, best={best_val:.10f}, elapsed={elapsed:.1f}s")

    # --------------------------------------------------------------
    # 7) Return the best sequence (clipped to the evaluator’s limits)
    # --------------------------------------------------------------

    print(f"[result] final_score={final_score:.10f}, length={len(best_seq)}")
    return best_seq
