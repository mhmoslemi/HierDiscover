def adam_optimize(n, time_limit, init_a=None):
    start = time.time()
    deadline = start + time_limit
    if init_a is None:
        a = np.full(n, 1.0/n)
    else:
        a = init_a.copy()
        a = a / a.sum()
    w = np.log(a + 1e-12)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    # hyperparameters:
    lr_start = 0.2
    lr_end = 0.001
    eps_start = 0.1
    eps_end = 1e-8
    best_a = a.copy()
    best_val = float('inf')
    no_improve_cnt = 0
    IMPROVE_LIMIT = 30000
    while True:
        now = time.time()
        if now >= deadline:
            break
        # softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()
        # compute conv
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve_cnt = 0
            print(f"[adam] n={n} iter={t} new best={best_val:.6f}")
        else:
            no_improve_cnt += 1
        # schedule
        progress = (now - start) / (deadline - start)
        lr = lr_start * (1.0 - progress) + lr_end * progress
        eps = eps_start * ( (eps_end/eps_start) ** progress ) # exponential decay
        # compute p vector
        shifted = (conv - max_conv) / eps
        # avoid overflow: large negative numbers produce zero weights
        # Maybe clip shifted to avoid extreme exponentials? Using np.exp safe.
        weights = np.exp(shifted)
        p = weights / weights.sum()
        # gradient wrt a
        # compute p_rev = p[::-1]
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev) # length n + (2n-1) -1 = 3n-2
        # mid segment
        segment = conv_ab[n-1:2*n-1]
        grad_a = 4.0 * n * segment[::-1]  # shape (n,)
        # gradient wrt w
        d = np.dot(grad_a, a)
        grad_w = a * (grad_a - d)
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w * grad_w)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # apply random perturbation occasionally
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)
        # restart if stagnated
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[adam] n={n} restart after {no_improve_cnt} iterations")
            w = np.random.randn(n) * 0.1
            m.fill(0)
            v.fill(0)
            t = 0
            no_improve_cnt = 0
    return best_a, best_val
```

def subgradient_finetune(a, time_limit):
    start = time.time()
    deadline = start + time_limit
    n = a.shape[0]
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a).max()
    eta0 = 0.5  # step size base
    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # get max index (any)
        k_star = int(np.argmax(conv))
        # compute gradient
        g = np.zeros_like(a)
        # compute reversed indices
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        # for i in [low, high]: j = k_star - i
        # then g[i] += 2 * a[j]
        # Equivalent to: g[low:high+1] = 2 * a[k_star - np.arange(low, high+1)]
        idx = np.arange(low, high+1, dtype=int)
        g[idx] = 2.0 * a[k_star - idx]
        # step size
        eta = eta0 / np.sqrt(it + 1)
        # multiplicative update: a = a * exp(-eta * g)
        a = a * np.exp(-eta * g)
        # renormalize
        total = a.sum()
        if total <= 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total
        it += 1
        # occasional random jitter for escaping

"""
Propose a step‑function (vector of non‑negative heights) that minimises the
provided ``evaluate_sequence`` objective.

The search proceeds in three stages:

1. **Coarse Adam optimisation (n≈2000)** – a smoothed version of the
   max‑convolution (log‑sum‑exp) is optimised with Adam on the soft‑max
   parametrisation `a = softmax(w)`.  Hyper‑parameters (learning rate,
   temperature ε) are aggressively annealed and occasional random restarts
   avoid stagnation.  If a previous construction ``height_sequence_1`` is
   available it is interpolated to the current size and used as a start.

2. **Upsampling & fine Adam optimisation (n≈6000–8000)** – the best vector
   from stage 1 is linearly interpolated to a larger dimension, giving the
   algorithm more degrees of freedom.  The same Adam scheme is run again,
   but with a slower learning‑rate schedule and a tighter final ε to focus
   on the true ℓ∞‑objective.

3. **Sub‑gradient mirror descent** – finally a simple exponentiated‑gradient
   (mirror‑descent) scheme is applied directly on the ℓ∞‑objective.
   At each iteration the index `k*` achieving the maximal convolution is
   located; the sub‑gradient is `g_i = 2·a_{k*‑i}` (zero outside the array).
   The update `a_i ← a_i·exp(−η·g_i)` (followed by renormalisation) shrinks
   the mass at positions that contribute most to the peak and yields a very
   fine local improvement.

All stages respect the global time budget (≈1000 s) and keep the best exact
objective value seen (computed with the original ``evaluate_sequence``).  The
returned list satisfies the non‑negativity and sum‑normalisation constraints
required by the evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x*x (length 2*len(x)-1) using real FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # tiny negative round‑off can appear; clip to zero
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Adam optimiser for the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Run Adam on the log‑sum‑exp approximation of `max(conv(a,a))`.
    Returns the best vector found (sum = 1) and its exact objective value.
    """
    start = time.time()
    deadline = start + time_limit

    # initialise a (probability vector)
    if init_a is None:
        a = np.full(n, 1.0 / n)
    else:
        a = np.copy(init_a)
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()

    # soft‑max parametrisation
    w = np.log(a + 1e-12)

    # Adam state
    beta1, beta2 = 0.90, 0.999
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    # schedules
    lr_start, lr_end = 0.20, 0.001          # learning‑rate (linear)
    eps_start, eps_end = 0.10, 1e-8        # temperature (exponential)

    # bookkeeping
    best_a = a.copy()
    best_val = float('inf')
    no_improve_cnt = 0
    IMPROVE_LIMIT = 30000  # restart after many idle steps

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- soft‑max to feasible point -----
        w_shift = w - np.max(w)               # numerical stability
        a = np.exp(w_shift)
        a /= a.sum()                           # Σa = 1, a_i ≥ 0

        # ----- exact objective (no smoothing) -----
        conv = _full_convolution(a)            # length 2n‑1
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        # ----- record best -----
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve_cnt = 0
            print(f"[adam n={n:4d}] iter={iteration:7d}  new best = {best_val:.6f}")
        else:
            no_improve_cnt += 1

        # ----- schedules -----
        progress = (now - start) / (deadline - start)
        lr = lr_start * (1.0 - progress) + lr_end * progress
        # exponential decay for ε
        eps = eps_start * (eps_end / eps_start) ** progress

        # ----- smoothed max via log‑sum‑exp -----
        # p_k = softmax(conv_k / ε)
        shifted = (conv - max_conv) / eps        # ≤ 0
        weights = np.exp(shifted)               # under‑flow gives 0
        p = weights / weights.sum()             # length 2n‑1, Σp = 1

        # ----- gradient wrt a (see derivation in description) -----
        p_rev = p[::-1]                         # reversed p
        conv_ab = _conv_fft(a, p_rev)           # length 3n‑2
        segment = conv_ab[n - 1 : 2 * n - 1]    # middle n entries
        grad_a = 4.0 * n * segment[::-1]        # (n,)

        # ----- chain‑rule through soft‑max -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ----- occasional perturbation (helps escape plateaus) -----
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # ----- random restart if stuck too long -----
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[adam n={n:4d}] restart after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the true ℓ∞‑objective
# ----------------------------------------------------------------------
def _subgradient_finetune(
    a: np.ndarray,
    time_limit: float,
) -> np.ndarray:
    """
    Simple exponentiated‑gradient (mirror descent) using the exact max‑convolution.
    Works directly on the probability vector a (Σa = 1, a_i ≥ 0).
    """
    start = time.time()
    deadline = start + time_limit
    n = a.shape[0]

    # keep the best seen
    conv = _full_convolution(a)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    eta0 = 0.50   # base step size; decays like 1/√t
    it = 0

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # index of the current maximal convolution entry
        k_star = int(np.argmax(conv))

        # sub‑gradient g_i = 2 * a_{k*‑i} (zero outside [0,n‑1])
        g = np.zeros_like(a)
        lo = max(0, k_star - (n - 1))
        hi = min(k_star, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] = 2.0 * a[k_star - idx]

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * g)

        # renormalise to keep Σa = 1 (and avoid all zeros)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        it += 1

        # occasional tiny jitter to escape flat regions
        if np.random.rand() < 0.001:
            jitter = np.random.rand(n)
            a = 0.5 * a + 0.5 * jitter
            a = a / a.sum()

    return best_a
