def simplex_projection(v):
    """Project vector v onto probability simplex (nonnegative sum=1)."""
    # Using algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (list of non‑negative heights) that minimizes
evaluate_sequence.  

The algorithm consists of two stages:

1. **Smoothed‑max Adam** (≈70 % of the time budget).  
   We keep the coefficients a_i on the probability simplex (Σa_i=1) by
   projecting after each update.  The loss 2·n·max(conv(a,a)) is
   approximated by a log‑sum‑exp with a temperature ε that decays to
   almost zero, giving a smooth gradient that drives the maximum of the
   auto‑convolution down.

2. **Exact‑max sub‑gradient refinement** (remaining time).  
   After the temperature has become tiny we switch to the true
   sub‑gradient of the max‑function: the probability vector p is 1 on the
   active convolution indices (uniformly normalised).  The same Adam
   machinery is used, but now the gradient points directly at the
   worst‑case convolution entries.  The iterates are projected onto the
   simplex after each step.  Random perturbations and occasional restarts
   avoid stagnation.

During both phases we keep the best sequence seen (exact objective
computed with the original evaluate_sequence) and finally return it.
"""

import time
import numpy as np
from scipy.optimize import linprog  # not used directly but imported for safety

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, full length)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise can produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x ≥ 0, Σx = 1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projects v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection only works on 1‑D arrays")
    n = v.shape[0]
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (the same formula used by evaluate_sequence)
# ----------------------------------------------------------------------
def _exact_objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a already normalised Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()

# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Optimise step‑function heights using a two‑stage Adam scheme
    (smooth then exact).  Returns the best list of floats found before
    the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin for the caller

    # Problem size – 4000 works well on a single core.
    # --------------------------------------------------------------
    n = 4000

    # Initialise from the previous construction if available,
    # otherwise start with a uniform distribution.

    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.clip(a, 0.0, None)
    except Exception:
        a = np.full(n, 1.0 / n)

    a = a / a.sum()                               # enforce Σa = 1

    # --------------------------------------------------------------
    # Variables for Adam (first‑order moment m, second‑order v)
    # --------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    t_iter = 0

    # Hyper‑parameters (learning‑rate and temperature schedule)
    lr_start = 0.05
    lr_end = 0.001
    eps_start = 5e-3
    eps_end = 1e-8

    best_val = _exact_objective(a)

    # --------------------------------------------------------------
    phase1_deadline = start + 0.70 * (deadline - start)

        conv_a = _full_convolution(a)

        cur_val = 2.0 * n * max_conv

    # --------------------------------------------------------------

        max_conv = conv_a.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{t_iter:7d}] new best (phase‑2) = {best_val:.6f}")

        # ---- build the probability vector p that puts mass on all active maxima ----
        max_idxs = np.where(conv_a == max_conv)[0]
        p = np.zeros_like(conv_a)
        if len(max_idxs) > 0:
            p[max_idxs] = 1.0 / len(max_idxs)   # uniform over active indices

        # ---- sub‑gradient (same formula as before, now with binary p) ----
        p_rev = p[::-1]
        conv_ap = _conv_fft(a, p_rev)
        segment = conv_ap[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # ---- Adam‑style update (still with projection) ----
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad_a
        v = beta2 * v + (1.0 - beta2) * (grad_a * grad_a)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # learning‑rate schedule – slower than Phase‑1
        now = time.time()
        progress = (now - start) / (deadline - start)
        lr = lr_end * (0.5 + 0.5 * (1.0 - progress))   # from 0.0005 → 0.001

        a_tmp = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = _project_simplex(a_tmp)

        # ---- very occasional random local move (helps exploratory jumps) ----
        if np.random.rand() < 0.01:
            i, j = np.random.choice(n, size=2, replace=False)
            delta = min(a[i], 0.001) * np.random.rand()
            a[i] -= delta
            a[j] += delta
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Final report – run the official evaluator on the best candidate
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    return best_a.tolist()
