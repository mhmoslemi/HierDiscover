"""
Propose a step‑function (non‑negative height vector) that minimises

    2 * n * max_k (a * a)[k] / (sum(a))**2

where * denotes the full linear convolution.  The algorithm consists of
two stages:

1. **Deterministic sweep** – a collection of parametric families
   (symmetric/monotone power‑law, exponential, and Beta‑like shapes)
   are instantiated for several lengths and parameters.
   The best vector (normalised to sum = 1) is kept as a seed.

2. **Refinement** – starting from the seed we perform a stochastic
   optimisation on the probability simplex.  The main move is an
   exponentiated‑gradient (mirror‑descent) step using the sub‑gradient of
   the maximal convolution entry, combined with a simple simulated‑annealing
   acceptance rule and occasional random mass‑transfer perturbations.
   The optimiser runs until the global time budget expires, keeping the
   best vector seen.

The whole search is self‑contained, respects the 1000 s budget and
returns a plain Python list of floats.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w≥0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 2 * n
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact score returned by ``evaluate_sequence`` for a vector that is
    already normalised (sum(a) == 1).  The denominator therefore equals 1.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices belonging to (almost) maximal entries contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)

    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        # each pair (i,j) contributes 2*a[j] to the derivative w.r.t. a[i]
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Parametric families (both symmetric and monotone)
# ----------------------------------------------------------------------


def sym_u_min_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric “U‑shape’’ based on distance to the nearest edge."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def sym_u_prod_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric “U‑shape’’ based on product of distances to both edges."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def sym_exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def mono_power_distribution(n: int, p: float) -> np.ndarray:
    """Monotone power‑law decay a[i] ∝ (i+1)^(-p)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p)) + 1e-14
    a /= a.sum()
    return a


def mono_exp_distribution(n: int, lam: float) -> np.ndarray:
    """Monotone exponential decay a[i] ∝ exp(-lam*i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑like shape on a grid of size n."""
    t = np.linspace(0.0, 1.0, n)
    # avoid 0^negative by adding epsilon
    a = (np.power(t + 1e-14, alpha - 1.0) *
         np.power(1.0 - t + 1e-14, beta - 1.0)) + 1e-14
    a /= a.sum()
    return a

# Refinement routine (exponentiated‑gradient + SA)
# ----------------------------------------------------------------------


def refine_sequence(a0: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Stochastic refinement of a normalised vector a0.
    Uses an exponentiated‑gradient (mirror‑descent) step on the simplex
    combined with a simulated‑annealing acceptance rule and occasional
    random mass‑transfer perturbations.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = full_convolution_fft(a)
    cur_score = float(2.0 * n * conv.max())

    best = a.copy()
    best_score = cur_score

    start = time.time()
    deadline = start + time_limit

    temperature = 1.0
    temp_decay = 0.9997
    eta = 0.8          # learning rate for the exponentiated‑gradient step
    min_eta = 1e-6
    stagnation_counter = 0
    last_improve = 0
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ---- sub‑gradient of the maximal convolution entry
        grad = compute_subgradient(a, conv)

        # ---- exponentiated‑gradient step (always stays non‑negative)
        cand = a * np.exp(-eta * grad)
        cand = np.maximum(cand, 1e-16)
        cand /= cand.sum()

        # ---- occasional random mass‑transfer move (helps escape)
        if iteration % 200 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            if cand[i] > 0.0:
                delta = cand[i] * np.random.rand() * 0.5
                cand[i] -= delta
                cand[j] += delta
                cand = np.maximum(cand, 0.0)
                cand /= cand.sum()

        # ---- evaluate candidate
        cand_conv = full_convolution_fft(cand)
        cand_score = float(2.0 * n * cand_conv.max())

        # ---- Simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            conv = cand_conv
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ---- schedule updates
        temperature *= temp_decay
        eta = max(eta * 0.9999, min_eta)

        # ---- re‑heat / learning‑rate reset on long stagnation
        if iteration - last_improve > 60000:
            temperature = 1.0
            eta = max(eta, 0.3)
            # small random bump to shake the state
            bump = np.random.rand(n) * 1e-5
            a = simplex_projection(a + bump)
            a /= a.sum()
            conv = full_convolution_fft(a)
            cur_score = float(2.0 * n * conv.max())
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

    return best

        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f"fallback random seed n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement (exponentiated‑gradient + SA)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n=== Refinement start (remaining {:.1f}s) ===".format(remaining))
        refined = refine_sequence(best_seq, remaining - 2.0)
        refined_score = score_from_vector(refined)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined
            print(f"Refinement improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return best vector as a plain Python list
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
