Search for a non‑negative step vector that minimises the evaluation
function used in the competition.  The algorithm combines three ideas:

1. **Linear‑programming polishing** – Starting from the well‑known
   “arcsine’’ shape, we solve a small LP that maximises the total mass
   of a new vector b under the constraint (f * b) ≤ M, where M is the
   current maximal self‑convolution of f.  The solution is rescaled to
   the same ℓ₁‑norm as f and mixed with f.  Repeating this step drives
   the maximal convolution down monotonically.

2. **Smooth‑max Adam optimisation** – The exact objective
   ``2·n·max(conv)`` is non‑smooth.  We replace the max by a
   log‑sum‑exp with a gradually increasing temperature
   α (α→∞ ⇒ max).  Using the analytical gradient of the smoothed
   objective we run Adam on the soft‑max logits of the vector,
   projecting only at the very end.  α is increased from 0.5 to 200,
   providing a homotopy from a very smooth landscape to the true
   objective.

3. **Local swap hill‑climber** – Finally a cheap random‑swap greedy
   search moves tiny amounts of probability between two coordinates if
   it improves the true objective.

All steps respect the 1000 s time budget, keep the vector in the
probability simplex (sum = 1, entries ≥ 0) and return a list of floats
ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0 , Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities (re‑used many times)
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution of p using a pre‑computed FFT size."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _score(p: np.ndarray, m_fft: int) -> float:
    """Exact evaluation score 2·n·max(conv_self)."""
    conv = _conv_self(p, m_fft)
    return float(2.0 * p.size * np.max(conv))


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth max) w.r.t. the probability vector.
# ----------------------------------------------------------------------
def _grad_a(p: np.ndarray, w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. p.
    w is the soft‑max weight vector (len = 2·n‑1).
    """
    n = p.shape[0]
    p_fft = np.fft.rfft(p, n=m_fft)
    w_rev_fft = np.fft.rfft(w[::-1], n=m_fft)
    corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]
    idx = 2 * n - 2 - np.arange(n)          # mapping from full correlation to gradient
    return 2.0 * corr_full[idx]


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)       # type: ignore[name-defined]
    except Exception:
        n = 1153                           # fallback length used in the benchmark

    # FFT size – next power‑of‑two that can contain a convolution of length 2·n‑1
    m_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Timing
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 1.0     # safety margin

    best_score = float("inf")
    best_a = None

    # --------------------------------------------------------------
    # Initial candidate: arcsine shape (known good starting point)
    # --------------------------------------------------------------
    i = np.arange(n, dtype=float)
    eps = 1e-12
    inv = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)   # arcsine density
    f = inv / inv.sum()                                    # normalised (ℓ₁‑norm = 1)

    cur_score = _score(f, m_fft)
    best_score = cur_score
    best_a = f.copy()
    print(f"Initial score (arcsine) : {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – Linear‑programming polishing
    # --------------------------------------------------------------
    max_lp_iters = 12                     # enough to see monotonic improvement
    mix_factor = 0.5                      # new = (1-t)*old + t*new, t∈(0,1)

    for lp_iter in range(max_lp_iters):
        if time.time() >= deadline:
            break

        # current maximal self‑convolution
        conv_f = _conv_self(f, m_fft)
        M = np.max(conv_f)

        # Build the Toeplitz matrix representing convolution with f.
        # First column = f padded with zeros, first row = [f0,0,0,…].
        col = np.concatenate([f, np.zeros(n - 1)])   # length 2·n‑1
        row = np.concatenate([np.array([f[0]]), np.zeros(n - 1)])
        A = toeplitz(col, row)                       # shape (2·n‑1, n)

        # Linear programme: maximise Σ b  ⇔  minimise −Σ b
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)                  # (f * b) ≤ M
        bounds = [(0.0, None)] * n

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={"presolve": True})

        if not res.success:
            print(f"LP {lp_iter}: failed – {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Rescale to keep ℓ₁‑norm = 1 and mix.
        g = b / sum_b
        f_new = (1.0 - mix_factor) * f + mix_factor * g

        new_score = _score(f_new, m_fft)
        print(f"LP {lp_iter:02d} – score {new_score:.6f}")

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = f_new.copy()

        # Stop if LP did not improve the objective.
        if new_score >= cur_score - 1e-12:
            print("LP stopped – no further improvement.")
            break

        f = f_new
        cur_score = new_score

    # --------------------------------------------------------------
    # Stage 2 – Adam optimisation on a smooth approximation of max.
    # --------------------------------------------------------------
    # Work in the soft‑max log‑it space to keep the simplex constraints.
    eps_small = 1e-12
    x = np.log(f + eps_small)        # logits such that softmax(x) = f

    # Adam state
    m1 = np.zeros_like(x)
    m2 = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    # Smoothing schedule (log‑sum‑exp)
    alpha = 0.5
    alpha_mul = 1.02
    alpha_max = 200.0

    it = 0
    while time.time() < deadline:
        # ---------- soft‑max ----------
        mx = np.max(x)
        exp_x = np.exp(x - mx)
        a = exp_x / exp_x.sum()          # probability vector, sum = 1

        # ---------- convolution ----------
        a_fft = np.fft.rfft(a, n=m_fft)
        conv = np.fft.irfft(a_fft * a_fft, n=m_fft)[: 2 * n - 1]

        # ---------- smooth max ----------
        z = alpha * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        w = exp_z / exp_z.sum()          # soft‑max weights (len = 2·n‑1)

        # ---------- gradient ----------
        grad_a = _grad_a(a, w, m_fft)    # ∂L/∂a
        dot = np.dot(grad_a, a)          # aᵀ·∇L_a
        grad_x = a * (grad_a - dot)      # chain rule through soft‑max

        # ---------- Adam update ----------
        m1 = beta1 * m1 + (1.0 - beta1) * grad_x
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m1 / (1.0 - beta1 ** (it + 1))
        v_hat = m2 / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        x = x - step

        # ---------- schedule ----------
        alpha = min(alpha * alpha_mul, alpha_max)

        # ---------- periodic evaluation ----------
        if (it & 2047) == 0:               # every ~2 k iterations
            cur_score = 2.0 * n * np.max(conv)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"Adam iter {it:06d} – new best {best_score:.6f}")
            if best_score <= 1.503 + 1e-9:
                print("Target reached during Adam, exiting early.")
                break

        it += 1

    # --------------------------------------------------------------
    # Stage 3 – Tiny local swap hill‑climber
    # --------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = _score(best_a, m_fft)

    p = best_a.copy()
    cur_score = _score(p, m_fft)
    swap_attempts = 20000

    for _ in range(swap_attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a tiny random fraction of mass from i to j
        eps_move = 1e-5 * rng.random() * p[i]
        if eps_move <= 0.0:
            continue
        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move
        new_score = _score(new_p, m_fft)
        if new_score < cur_score:
            p = new_p
            cur_score = new_score
            best_a = p.copy()
            best_score = cur_score
            print(f"Swap improvement → {best_score:.6f}")
            if best_score <= 1.503:
                break

    print(f"Final best score: {best_score:.6f}")

    # Clip to the range enforced by evaluate_sequence and return as list
    return np.clip(best_a, 0.0, 1000.0).tolist()
