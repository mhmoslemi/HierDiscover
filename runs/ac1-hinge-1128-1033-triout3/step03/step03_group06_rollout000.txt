"""
Propose a step‑function (probability vector) that minimises the metric
`evaluate_sequence`.  The optimiser works on several scales, each time
optimising a probability vector via Adam on a smooth *log‑sum‑exp* upper
bound of the self‑convolution maximum.  The vector is parametrised by a
log‑probability (soft‑max) to enforce non‑negativity and unit sum
automatically.  The smoothness parameter α is slowly increased so the
objective converges to the true L∞ norm.  After each scale the best
vector is up‑sampled and used as the seed for the next larger length.
A simple stagnation detector adds a small Gaussian perturbation when
no improvement is seen for many iterations.  The routine respects the
global time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """
    Exact maximum entry of the linear self‑convolution a*a using a real FFT.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a (length 2·n‑1) using a supplied FFT size.
    """
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Normalised soft‑max weights w_k = exp(α·conv_k) / Σ exp(α·conv).
    The exponent is stabilised by subtracting the maximum entry.
    """
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    if s == 0.0:                     # numerical safety
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of Σ w_k·conv_k (= soft‑max of the convolution) w.r.t. a.
    a : probability vector (length n)
    w : soft‑max weights (length 2·n‑1, sum to 1)
    size : FFT length (≥ 3·n‑2)
    Returns a vector of length n.
    """
    n = a.shape[0]
    L = 2 * n - 1                       # length of w
    # FFT of a (already normalised)
    A = np.fft.rfft(a, n=size)

    # FFT of w and its reversal
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # Convolution a * w (length n+L‑1 = 3·n‑2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    # Gradient = conv_aw[:n] + conv_aw_rev[L-1 - np.arange(n)]
    idx = L - 1 - np.arange(n)          # 2·n‑2, …, n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    (Not used in the main routine – kept for completeness.)
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a new length.
    The result is re‑normalised to sum to one.
    """
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new += 1e-14                     # avoid exact zeros
    new /= new.sum()
    return new


def u_shaped_distribution(n: int, p: float = 0.8) -> np.ndarray:
    """
    Simple symmetric U‑shaped distribution: a_i ∝ (min(i, n‑1‑i)+1)^{‑p}.
    Returns a normalised probability vector.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)               # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

    try:
        exact_score = evaluate_sequence(final_list)   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")
        exact_score = None

    print("\n=== Search finished ===")
    print(f"Best approximated score : {global_best_score:.6f}")
    print(f"Best length             : {global_best_seq.shape[0]}")
    return final_list
