"""
Propose a candidate step‑function (non‑negative vector) with a low
evaluate_sequence score.

The algorithm works in three stages:

1. **Initialization** – use the classic arcsine distribution, scale it
   so that Σa = sqrt(2 n) (the scaling is irrelevant for the score but
   matches the reference paper).

2. **LP polishing** – solve a linear program
   max Σg   s.t.   f * g ≤ M,   g ≥ 0,
   where * denotes convolution, M = max(f * f).  The solution g is
   scaled back to the required ℓ1‑norm and mixed with the current f.
   A few LP steps (normally ≤ 10) flatten the self‑convolution considerably.

3. **Gradient refinement** – starting from the best vector seen so far,
   run a long projected Adam optimisation on a smooth log‑sum‑exp
   approximation of the max‑convolution, followed by a tiny random‑swap
   local search.  The whole procedure respects the 1000 s budget and
   always returns a valid list of non‑negative floats.

The implementation uses only NumPy, SciPy (optional for the LP) and
standard Python libraries.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0 , Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_self_fft(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution of p using FFT (length 2·len(p)‑1)."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _conv_max(p: np.ndarray, m_fft: int) -> float:
    """Maximum entry of the self‑convolution of p."""
    return float(_conv_self_fft(p, m_fft).max())


# ----------------------------------------------------------------------
# Sub‑gradient of the *actual* max entry of the convolution.
# ----------------------------------------------------------------------
def _subgrad_max_conv(p: np.ndarray, max_idx: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry w.r.t. p."""
    n = p.size
    g = np.zeros_like(p)
    if max_idx < n:
        # overlap size = max_idx + 1
        rev = p[max_idx::-1]                 # reversed slice
        sl = p[: max_idx + 1]
        g[: max_idx + 1] = rev + sl
    else:
        # tail overlap
        start = max_idx - n + 1
        sl1 = p[start:]                      # forward part
        sl2 = p[n - 1 : start - 1 : -1]      # reversed tail part
        g[start:] = sl1 + sl2
    return g

# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    # --------------------------------------------------------------
    # Problem size (use historic length if available)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 1153  # fallback length used in the benchmark

    sqrt2n = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Initial arcsine distribution (probability vector) and scaling.
    # --------------------------------------------------------------
    i = np.arange(n, dtype=float)
    eps = 1e-12
    inv = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    p0 = inv / inv.sum()               # probability vector (Σ=1)
    f = p0 * sqrt2n                    # scaled vector (Σ = √(2 n))

    # --------------------------------------------------------------
    # Scoring helper (scale‑invariant, works with any ℓ1‑norm)
    # --------------------------------------------------------------
    def _score_vec(vec: np.ndarray) -> float:
        """Return evaluate_sequence score for a vector."""
        conv = _conv_self_fft(vec, m_fft)
        return float(2.0 * vec.size * conv.max() / (vec.sum() ** 2))

    # --------------------------------------------------------------
    # Pre‑compute FFT size (next power‑of‑two for safe convolution)
    # --------------------------------------------------------------
    m_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Keep the best vector seen so far.
    # --------------------------------------------------------------
    best_f = f.copy()

        have_scipy = True

        row_idx = (np.arange(n)[:, None] + np.arange(n)).ravel()

        # ----------------------------------------------------------

            # on rows i … i+n‑1.
            data = np.repeat(f, n)        # one block per column

                              shape=(2 * n - 1, n))

            c = -np.ones(n)
            b_ub = np.full(2 * n - 1, M, dtype=float)

                          A_ub=A_ub,
                          b_ub=b_ub,
                          bounds=(0, None),
                          method='highs',

                print(f"LP iteration {it_lp}: solver failed (status {res.status})")
                break

            g = res.x
            sum_g = g.sum()
            if sum_g <= 0:
                break

            # Scale g back to the required ℓ1‑norm (√(2 n)).
            scale = sqrt2n / sum_g
            g_scaled = g * scale

            # Mix with the current vector.
            f = (1.0 - t_mix) * f + t_mix * g_scaled

            # Evaluate new score.
            cur_score = _score_vec(f)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_f = f.copy()
                print(f"LP iter {it_lp}: improved score → {best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    # that the final score is the same.
    p = best_f.copy()
    p = p / p.sum()                     # probability vector
    m1 = np.zeros_like(p)
    m2 = np.zeros_like(p)

    # Hyper‑parameters
    sub_step = 0.2
    sub_decay = 0.9995
    sub_min_step = 1e-6
    sub_max_iters = 20000

    adam_lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    alpha = 1.0
    alpha_mul = 1.001
    alpha_max = 200.0

    # Time management
    deadline = time.time() + budget_s - 5.0   # 5 s safety

    best_p = p.copy()
    best_score_grad = 2.0 * n * _conv_max(p, m_fft)   # because Σp = 1

    # ---- Sub‑gradient descent on the *actual* max entry ----
    for it in range(sub_max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self_fft(p, m_fft)
        max_val = conv.max()
        max_idx = conv.argmax()

        cur_score = 2.0 * n * max_val   # Σp = 1 ⇒ denominator = 1
        if cur_score < best_score_grad - 1e-12:
            best_score_grad = cur_score
            best_p = p.copy()
            print(f"Sub‑grad iter {it:5d} – new best {best_score_grad:.6f}")

        # Sub‑gradient of the max‑entry.
        g = _subgrad_max_conv(p, max_idx)

        # Gradient step + projection onto simplex.
        p = p - sub_step * g
        p = _simplex_projection(p)
        sub_step = max(sub_step * sub_decay, sub_min_step)

        # Early stop if we already beat the target.
        if best_score_grad <= 1.503:
            break

    # ---- Adam optimisation on a log‑sum‑exp smooth approximation ----
    it = 0
    while time.time() < deadline:
        # Smoothed max (log‑sum‑exp) and its gradient.
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

        m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)
