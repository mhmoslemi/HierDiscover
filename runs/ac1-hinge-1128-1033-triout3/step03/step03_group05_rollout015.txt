"""
Propose a height sequence that minimizes the evaluation score.

The algorithm combines several complementary optimisation ideas:

1. **Diverse initialisations** – arcsine‐shaped, uniform,
   exponential, sinusoidal and many random Dirichlet / Gamma draws.
2. **Sub‑gradient descent** – moves probability away from the
   currently maximal convolution entry (the exact sub‑gradient of the
   max‑convolution).
3. **Adam on a log‑sum‑exp smooth max** – a differentiable surrogate of
   the max; the gradient is obtained via FFT‑based correlation.
   The smoothing parameter α is increased slowly so the optimisation
   converges towards the true max.
4. **Linear‑programming (LP) step** – for the current vector `p`,
   solve a LP that finds a vector `g` with maximal total mass such that
   the convolution `p * g` does not exceed the current maximal
   self‑convolution.  A convex combination `p ← (1‑t)p + t·g` (Frank‑Wolfe
   style) yields a new feasible direction.
5. **Local swap hill‑climber** – tiny mass transfers between two
   coordinates are accepted if they improve the exact objective.
6. **Time‑aware loop** – the whole routine respects the 1000 s budget,
   repeatedly restarting from fresh initial vectors while time permits.

All vectors are kept on the probability simplex (non‑negative, sum = 1),
which makes the evaluation scale‑invariant.  The final sequence is
clipped to the bounds enforced by `evaluate_sequence` and returned as a
plain list of floats.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Self‑convolution of p using a pre‑computed FFT size."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _score(p: np.ndarray, m_fft: int) -> float:
    """Exact evaluation score: 2·n·max(conv_self) for a probability vector."""
    conv = _conv_self(p, m_fft)
    return float(2.0 * p.size * np.max(conv))


def _build_conv_matrix(p: np.ndarray) -> np.ndarray:
    """Build the (2n‑1)×n matrix A such that A @ g = p * g."""
    n = p.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = p
    return A

        lr = 0.05

        alpha = 1.0
        alpha_mul = 1.001
        max_alpha = 300.0
        max_adam_iters = 40000

        for it in range(max_adam_iters):
            if time.time() >= deadline:
                return

            p_fft = np.fft.rfft(p, n=m_fft)
            conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

            # smoothed max via log‑sum‑exp
            z = alpha * conv
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)            # stability trick
            w = exp_z / np.sum(exp_z)            # soft‑max weights

            # gradient of smoothed objective
            w_rev = w[::-1]
            w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
            corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: (3 * n - 2)]
            idx = 2 * n - 2 - np.arange(n)
            grad = 2.0 * corr_full[idx]

            # Adam update
            m1 = beta1 * m1 + (1.0 - beta1) * grad
            m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

            m1_hat = m1 / (1.0 - beta1 ** (it + 1))
            m2_hat = m2 / (1.0 - beta2 ** (it + 1))

            step_vec = lr * m1_hat / (np.sqrt(m2_hat) + eps_adam)
            p = p - step_vec
            p = _simplex_projection(p)

            # increase smoothing
            alpha = min(alpha * alpha_mul, max_alpha)

            # occasional exact evaluation
            if (it % 1000 == 0) or (it == max_adam_iters - 1):
                cur_score = 2.0 * n * np.max(conv)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_p = p.copy()
                    print(f"[Adam]   iter {it:5d} – new best {best_score:.6f}")
                if cur_score <= 1.503:
                    return

        # --------------------------------------------------------------
        # (c) Linear‑programming “Frank‑Wolfe” step
        # --------------------------------------------------------------
        if time.time() < deadline - 0.5:
            conv = _conv_self(p, m_fft)
            M = np.max(conv)

            A = _build_conv_matrix(p)
            c = -np.ones(n)                     # maximise sum(g)
            b_ub = np.full(A.shape[0], M)
            bounds = [(0, None) for _ in range(n)]

            try:
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
                if res.success:
                    g0 = res.x
                    s = g0.sum()
                    if s > 0:
                        g_norm = g0 / s
                        t = 0.2                       # convex blend factor
                        p = (1.0 - t) * p + t * g_norm
                        p = _simplex_projection(p)
                        cur_score = _score(p, m_fft)
                        if cur_score < best_score - 1e-12:
                            best_score = cur_score
                            best_p = p.copy()
                            print(f"[LP]        blend new best {best_score:.6f}")
            except Exception as e:
                print("[LP] exception:", e)

        # --------------------------------------------------------------
        # (d) Tiny local swap hill‑climber
        # --------------------------------------------------------------
        if time.time() < deadline - 0.5:
            p_swap = best_p.copy() if best_p is not None else p.copy()
            cur_score_swap = _score(p_swap, m_fft)
            swap_attempts = 5000
            for _ in range(swap_attempts):
                if time.time() >= deadline:
                    break
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                # move at most a tiny fraction of the source mass
                max_move = min(p_swap[i], 1e-3)
                if max_move <= 0:
                    continue
                eps_move = max_move * rng.random()
                new_p = p_swap.copy()
                new_p[i] -= eps_move
                new_p[j] += eps_move
                new_score = _score(new_p, m_fft)
                if new_score < cur_score_swap:
                    p_swap = new_p
                    cur_score_swap = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_p = new_p.copy()
                        print(f"Swap improvement → {best_score:.6f}")
                    if best_score <= 1.503:
                        return

    # ----------------------------------------------------------------------
    # Driver: repeatedly launch optimisation from distinct starts
    # ----------------------------------------------------------------------
    start_id = 0
    while time.time() < deadline - 0.5:
        # rotate through the initial pool; if exhausted, generate fresh randoms
        if start_id < len(init_pool):
            p0 = init_pool[start_id]
        else:
            # fresh random Gamma vector (occasionally with a different shape)
            shape = rng.choice([0.2, 0.5, 1.0, 2.0, 5.0])
            p0 = rng.gamma(shape, size=n)
            p0 /= p0.sum()
        start_id += 1

        _optimise_start(p0)

        if best_score <= 1.503:
            break

    # ----------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    # ----------------------------------------------------------------------
    if best_p is None:
        best_p = np.full(n, 1.0 / n)
        best_score = _score(best_p, m_fft)
        print(f"Fallback uniform – score {best_score:.6f}")

    print(f"Final best score: {best_score:.6f}")

    # Clip to limits required by evaluate_sequence and return
    return np.clip(best_p, 0.0, 1000.0).tolist()
