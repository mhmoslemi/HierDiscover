"""Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence().

The method works in three phases:

1. **Initial candidates** – several deterministic and random probability
   vectors (arcsine shape, uniform, Beta‑like families and the historic
   construction if available).

2. **Projected Adam on a smooth max** – the objective
   `max(conv(p,p))` is approximated by a log‑sum‑exp (soft‑max) with a
   temperature `α`.  The vector `p` is represented by an unconstrained
   parameter `z` (soft‑max of `z`), which guarantees positivity and unit
   sum.  Adam updates are performed on `z` while `α` is increased
   geometrically, so the smoothed objective becomes tighter and the
   gradient converges to a sub‑gradient of the true max.  Symmetry
   (`p[i]=p[n−1−i]`) is enforced after each projection, because the
   optimal solution is known to be symmetric.

3. **Local refinement** – a cheap hill‑climber moves a tiny probability
   mass between two indices whenever it strictly reduces the real
   evaluation score `2·n·max(conv(p,p))`.

All operations use FFTs (size ≥ 3·n) to keep each iteration fast, allowing
hundreds of thousands of iterations within the 1000 s budget.  The best
vector found is returned (clipped to the limits enforced by
`evaluate_sequence`)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: self‑convolution via FFT (linear, not circular)
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of p using a pre‑allocated FFT length."""
    p_fft = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(p_fft * p_fft, n=fft_len)[: 2 * p.size - 1]
    return conv

# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 1153                     # fallback size used in the benchmark

    # FFT size must be large enough for a convolution of lengths n and (2n‑1)
    # (required when we convolve p with the soft‑max weight vector).
    fft_len = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Timing
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_score = float("inf")
    best_p = None

    # --------------------------------------------------------------
    # Build a pool of promising initial vectors (they are already normalised)
    # --------------------------------------------------------------
    init_vectors = []

    # 1) Arcsine shape (the classic construction)
    i = np.arange(n, dtype=float)
    inv = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + 1e-12)
    p_arcsine = inv / inv.sum()
    init_vectors.append(p_arcsine)

    # 2) Uniform distribution (perfectly symmetric)
    init_vectors.append(np.full(n, 1.0 / n))

    # 3) Random Dirichlet‑like draws
    for _ in range(3):
        r = rng.random(n) + 1e-12
        init_vectors.append(r / r.sum())

    # 4) Random symmetric Beta‑like shapes: p_i ∝ (i+½)^(a-1)*(n-½-i)^(b-1)
    for _ in range(2):
        a = rng.uniform(0.2, 2.5)
        b = rng.uniform(0.2, 2.5)
        shape = ((i + 0.5) ** (a - 1.0)) * ((n - 0.5 - i) ** (b - 1.0))
        shape = shape.clip(min=1e-12)
        shape = shape / shape.sum()
        init_vectors.append(shape)

    # 5) Historic construction if it exists
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        seq = np.clip(seq, 0.0, 1000.0)
        s = seq.sum()
        if s > 0.0:
            init_vectors.append(seq / s)

    # --------------------------------------------------------------
    # Core optimisation: projected Adam on the smoothed max
    # --------------------------------------------------------------
    for start_id, p0 in enumerate(init_vectors):
        if time.time() >= deadline:
            break

        print(f"\n=== optimisation start {start_id + 1}/{len(init_vectors)} ===")
        # Ensure positivity (log will be taken)
        eps = 1e-12
        p0 = np.maximum(p0, eps)
        p0 = p0 / p0.sum()

        # Parameterisation: p = softmax(z)
        z = np.log(p0)

        # Adam state
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lr = 0.07                     # initial learning rate
        lr_decay = 0.99995

        # Smooth‑max temperature
        alpha = 1.0
        alpha_mul = 1.002
        alpha_max = 300.0

        iteration = 0
        while time.time() < deadline:
            iteration += 1

            # ----- obtain probability vector (always on the simplex) -----
            z_shift = z.max()
            p = np.exp(z - z_shift)
            p = p / p.sum()
            # enforce exact symmetry (optimal solutions are symmetric)
            p = 0.5 * (p + p[::-1])

            # ----- self‑convolution and real score -----
            p_fft = np.fft.rfft(p, n=fft_len)
            conv = np.fft.irfft(p_fft * p_fft, n=fft_len)[: 2 * n - 1]
            cur_score = 2.0 * n * conv.max()

            if cur_score < best_score - 1e-12:
                best_score = float(cur_score)
                best_p = p.copy()
                print(f" iter {iteration:6d} – new best {best_score:.6f}")
                if best_score <= 1.503:
                    break

            # ----- smooth‑max (log‑sum‑exp) weights -----
            z_alpha = alpha * conv
            max_z = z_alpha.max()
            exp_z = np.exp(z_alpha - max_z)                # avoid overflow
            w = exp_z / exp_z.sum()                        # soft‑max over 2n‑1 entries

            # ----- gradient of the smooth max w.r.t. p -----
            w_rev = w[::-1]                                # reversal required for the gradient
            w_rev_fft = np.fft.rfft(w_rev, n=fft_len)
            corr_full = np.fft.irfft(p_fft * w_rev_fft, n=fft_len)
            idx = (2 * n - 2) - np.arange(n)               # mapping to the first n indices
            grad_p = 2.0 * corr_full[idx]

            # ----- transform gradient to the z‑space (softmax Jacobian) -----
            dot = np.dot(p, grad_p)
            grad_z = p * (grad_p - dot)                    # ∇_z f = J_softmax^T ∇_p f

            # ----- Adam update on z -----
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            z -= step
            # re‑impose symmetry on the parameters
            z = 0.5 * (z + z[::-1])

            # ----- schedule updates -----
            lr *= lr_decay
            alpha = min(alpha * alpha_mul, alpha_max)

            # early exit if we already hit the target bound
            if best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    # Final local refinement (tiny random mass swaps)
    # ------------------------------------------------------------------
    if best_p is None:
        best_p = np.full(n, 1.0 / n)
        best_score = 2.0 * n * np.max(_conv_self(best_p, fft_len))

    p = best_p.copy()
    cur_score = best_score
    swap_attempts = 20000

    while time.time() < deadline and swap_attempts > 0:
        swap_attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # tiny transfer of probability mass
        delta = 1e-4 * rng.random()
        if p[i] <= delta:
            continue
        new_p = p.copy()
        new_p[i] -= delta
        new_p[j] += delta
        # keep symmetry (optional, but helpful)
        new_p = 0.5 * (new_p + new_p[::-1])
        new_score = 2.0 * n * np.max(_conv_self(new_p, fft_len))
        if new_score < cur_score:
            p = new_p
            cur_score = new_score
            best_p = p.copy()
            best_score = cur_score
            print(f" Swap improvement → {best_score:.6f}")
            if best_score <= 1.503:
                break

    print(f"\nFinal best score: {best_score:.6f}")

    # Clip to the range enforced by evaluate_sequence and return
    return np.clip(best_p, 0.0, 1000.0).tolist()
