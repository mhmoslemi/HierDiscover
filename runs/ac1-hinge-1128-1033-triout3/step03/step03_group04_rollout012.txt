"""
Propose a low‑score step‑function.

The algorithm works in three phases:

1. **Deterministic sweep** – explores several parametric families
   (U‑shape, product‑U, Beta, exponential, and the exact arcsine/Beta(½,½)
   distribution) on a range of lengths.  The best vector found is kept
   as a seed.

2. **Stochastic refinement** – a fast sub‑gradient / mirror‑descent /
   random‑mass‑transfer loop (projected onto the simplex) with simulated‑
   annealing acceptance.  The active set for the sub‑gradient is enlarged
   slightly (relative tolerance 1e‑6) and the step‑size / temperature are
   decayed very gradually.

3. **Linear‑programming boost** – when the current length is ≤ 2000 the
   current vector `f` is used to solve the LP

        max   Σ y_j
        s.t.  (f * y)_k ≤ M   for all k,
              y_j ≥ 0 ,

   where `M = max(f*f)`.  The LP solution `g` is then mixed with the
   current vector (`f←(1‑t)f+t g`).  This operation strictly reduces the
   maximum of the self‑convolution and therefore improves the score.

All operations use FFT‑based convolution for speed, the simplex projection
implementation follows Wang & Carreira‑Perpiñán (O(n log n)), and the code
stops automatically before the 1000 s wall‑clock limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w ≥ 0, Σw = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length 2·len(a)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is non‑negative)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by `evaluate_sequence` for a normalised vector."""
    n = a.shape[0]
    # a is already non‑negative; keep the L1 sum for safety.
    s = a.sum()
    if s == 0:
        return np.inf
    return float(2.0 * n * max_convolution(a) / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge: (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the two edges."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}: a[i] ∝ (i+1)^{α‑1} (n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def arcsine_distribution(n: int) -> np.ndarray:
    """Discrete approximation of Beta(½,½) (the arcsine law)."""
    i = np.arange(n, dtype=np.float64) + 0.5           # centre of each bin
    a = 1.0 / np.sqrt(i * (n - i)) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# LP‑based improvement (optional, requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _LP_AVAILABLE = True
except Exception:                # pragma: no cover
    _LP_AVAILABLE = False


def solve_lp_bound(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ y_j   s.t.   (f * y)_k ≤ M  ∀k,   y ≥ 0 .
    Returns the normalised solution (Σ y = 1) or None on failure.
    """
    if not _LP_AVAILABLE:
        return None

    n = f.shape[0]
    rows = 2 * n - 1
    # Build the constraint matrix A_ub (rows × n)
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        # rows i … i+n‑1 receive f[i]
        A[i:i + n, i] = f[i]
    b = np.full(rows, M, dtype=np.float64)

    # Maximise Σ y   ↔   minimise −Σ y
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    # Use the HiGHS solver – fast for moderate sizes.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 5.0})
    if not res.success or res.x is None:
        return None
    g = res.x
    total = g.sum()
    if total <= 0.0:
        return None
    g /= total
    return g


# ----------------------------------------------------------------------
# Random candidate generator (used in the stochastic phase)
# ----------------------------------------------------------------------


def generate_random_candidate(n: int) -> np.ndarray:
    """Create a random distribution from one of the families."""
    choice = np.random.rand()
    if choice < 0.18:
        p = 10.0 ** np.random.uniform(-2.0, 0.5)      # p ∈ [0.01, 3]
        return u_min_distribution(n, p)
    if choice < 0.36:
        p = 10.0 ** np.random.uniform(-2.0, 0.5)
        return u_prod_distribution(n, p)
    if choice < 0.54:
        a = 10.0 ** np.random.uniform(-2.0, 0.5)      # α ∈ [0.01, 3]
        b = 10.0 ** np.random.uniform(-2.0, 0.5)      # β ∈ [0.01, 3]
        return beta_distribution(n, a, b)
    if choice < 0.72:
        gamma = 10.0 ** np.random.uniform(-4.0, -0.2)  # γ ∈ [1e‑4, 0.63]
        return exp_decay_distribution(n, gamma)
    if choice < 0.86:
        return arcsine_distribution(n)
    # fallback: small random perturbation of a uniform vector
    a = np.random.rand(n)
    a += 1e-3
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    Returns the best sequence discovered before the time budget expires.
    """

    deadline = time.time() + budget_s - 5.0     # small safety margin

    # ------------------------------------------------------------------
    best_seq = None

        np.arange(0.05, 0.51, 0.05),
        np.arange(0.6, 4.01, 0.2)

        np.arange(0.05, 0.51, 0.05),

        np.arange(0.001, 0.011, 0.001),
        np.arange(0.015, 0.051, 0.005)
    ))

    print("=== Deterministic sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # --- U‑shape family -------------------------------------------------
        for p in p_grid:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_min)   n={n:5d} p={p:5.3f}  score={sc:.6f}")

            beta_off = alpha * 0.7 + 0.3   # arbitrary offset to get asymmetry

            sc = score_from_vector(a)
            if sc < best_score:

                print(f" new best (beta_asym) n={n:5d} α={alpha:5.3f} β={beta_off:5.3f}  score={sc:.6f}")

        for gamma in gamma_grid:
            a = exp_decay_distribution(n, gamma)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (exp)    n={n:5d} γ={gamma:6.4f}  score={sc:.6f}")

        # --- Arcsine (Beta ½,½) ------------------------------------------------
        a = arcsine_distribution(n)
        sc = score_from_vector(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f" new best (arcsine) n={n:5d}                score={sc:.6f}")

        # occasional time check
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. Use external construction if it exists (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            total = external.sum()
            if total > 0.0:
                external /= total
                sc_ext = score_from_vector(external)
                if sc_ext < best_score:
                    best_score, best_seq = sc_ext, external.copy()
                    print(f" using external height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement (gradient / mirror / random moves)
    # ------------------------------------------------------------------
    current = best_seq.copy()
    current_score = best_score
    n = len(current)

    # hyper‑parameters
    temperature = 0.015          # initial temperature for simulated annealing
    cooling_rate = 0.99995       # very slow cooling
    base_step = 0.30
    step_size = base_step

    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------

        max_val = conv.max()
        # entries within a tiny relative tolerance of the maximum

        active_idx = np.where(conv >= max_val * (1.0 - rel_tol))[0]

        # --------------------------------------------------------------
        # 4b – Assemble sub‑gradient from the active set (exact)
        # --------------------------------------------------------------
        grad = np.zeros_like(current)
        for k in active_idx:

            end_i = min(n - 1, k)
            i_vals = np.arange(start_i, end_i + 1, dtype=int)
            j_vals = k - i_vals
            grad[i_vals] += 2.0 * current[j_vals]

        if r < 0.35:

            dst_idx = np.argpartition(grad, low_k)[:low_k]
            src = np.random.choice(src_idx)
            dst = np.random.choice(dst_idx)
            if src != dst and current[src] > 1e-14:

                cand[src] -= delta
                cand[dst] += delta

            else:

            cand = current - step_size * grad
            cand = np.maximum(cand, 0.0)

        # --------------------------------------------------------------

        # --------------------------------------------------------------

        # --------------------------------------------------------------

            accept = True
        else:

            if np.random.rand() < prob:

            current_score = cand_score
            # update global best if improvement is significant

                best_score = cand_score
                best_seq = cand.copy()
                last_improve_iter = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # --------------------------------------------------------------

        step_size *= cooling_rate

        # --------------------------------------------------------------

            rand_score = score_from_vector(rand_cand)
            if rand_score < best_score:
                best_score = rand_score
                best_seq = rand_cand.copy()
                current = rand_cand.copy()
                current_score = rand_score
                last_improve_iter = iteration
                print(f"Iter {iteration:7d}: random param hit score = {best_score:.6f}")

        # 4h – LP boost (only for moderate n)
        # --------------------------------------------------------------
        if _LP_AVAILABLE and n <= 2000 and iteration % 5000 == 0:
            M = max_convolution(current)
            g = solve_lp_bound(current, M)
            if g is not None:
                t = 0.20                     # mixing weight
                cand = (1.0 - t) * current + t * g
                cand = simplex_projection(cand)
                cand_score = score_from_vector(cand)
                # accept if it improves (always accept improvements)
                if cand_score < current_score:
                    current = cand
                    current_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    last_improve_iter = iteration
                    print(f"Iter {iteration:7d}: LP‑boost improved score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 4i – Re‑heat if stagnation persists
        # --------------------------------------------------------------
        if iteration - last_improve_iter > 120000:
            bump = np.random.rand(n) * 1e-4
            current = current + bump
            current = simplex_projection(current)
            current_score = score_from_vector(current)
            last_improve_iter = iteration
            temperature = 0.015
            step_size = base_step
            print(f" * Re‑heat at iter {iteration}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator will once again apply clipping and normalisation,
    # therefore we return the raw (already normalised) vector.
    return best_seq.tolist()
