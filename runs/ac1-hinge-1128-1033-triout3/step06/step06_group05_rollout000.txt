"""
Propose candidate step‑function heights.

Algorithm
---------
1. **Simplex projection** – All vectors are kept on the probability
   simplex (non‑negative, sum = 1) so that `evaluate_sequence`
   simply returns `2·n·max(a*a)`.

2. **Phase 0 – Warm‑start** – If a previous construction
   `height_sequence_1` exists it is interpolated to the working size;
   otherwise a random simplex point is used.

3. **Phase A – Soft‑max ADAM** – minimise a smooth surrogate of the
   max‑convolution:
        f_p(a) = (1/p)·log Σ_k exp(p·(a*a)_k)
   The gradient is
        ∇f_p = 2·conv_rev( a_rev , softmax_weights ) / Σw ,
   where the softmax weights are `w_k = exp(p·(b_k‑max(b)))`.
   `p` is increased from 1 to ~30 during this phase and the update is
   performed with the Adam optimiser.  After each step the vector is
   projected back onto the simplex.

4. **Phase B – Sub‑gradient descent on the exact max** – ordinary
   sub‑gradient steps for the true objective `max(a*a)` with a linearly
   decreasing step‑size.

5. **Phase C – L²‑flattening** – minimise the variance of the
   convolution (smooth surrogate) using a plain gradient step and a
   simplex projection.

6. **Phase D – Greedy hill‑climbing** – deterministic local search
   that moves a small amount of mass from the index contributing most
   to the current peak of the convolution to the index contributing
   least.  The full convolution is updated incrementally in O(n).

7. **Final evaluation** – the best vector encountered is evaluated
   once with the supplied `evaluate_sequence` and returned.

The three optimisation phases together respect the 1000 s wall‑time
limit.  Empirically the soft‑max ADAM phase discovers a better basin
than the previous sub‑gradient‑only approach, allowing the subsequent
phases to push the evaluation below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself.
    Length = 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    Length = len(x)+len(y)‑1.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0.0] = 0.0
    w_sum = w.sum()
    if w_sum > 0.0:
        w /= w_sum
    else:                           # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase‑A – Soft‑max ADAM minimisation of a smooth max‑approximation
# ----------------------------------------------------------------------
def _softmax_adam_phase(a: np.ndarray,
                        n: int,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> tuple[np.ndarray, float]:
    """
    Minimise f_p(a) = (1/p)·log Σ exp(p·(a*a))
    using Adam and a linear increase of p from 1 to 30.
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0
    start = time.time()

    # pre‑allocate reversed version of a (will be updated each iteration)
    while time.time() < deadline:
        # convolution a*a (full)
        b = _full_convolution(a)                     # length 2n‑1
        b_max = b.max()

        # schedule p  (1 → 30)
        phase_frac = (time.time() - start) / (deadline - start)
        p = 1.0 + 29.0 * phase_frac

        # compute soft‑max weights in a numerically safe way
        # w_k = exp(p·(b_k - b_max))
        w = np.exp(p * (b - b_max))
        sum_w = w.sum()
        if sum_w == 0.0:          # degenerate – fall back to uniform weights
            w = np.ones_like(b) / b.shape[0]
            sum_w = 1.0

        # gradient = 2·conv_rev( a_rev , w ) / sum_w
        a_rev = a[::-1]                               # length n
        conv_rev = _conv_fft(a_rev, w)                # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n] / sum_w

        # Adam update
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # learning‑rate schedule (larger at the beginning, shrink later)
        lr = 0.6 / (1.0 + 5.0 * phase_frac)          # ~0.6 → 0.07

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a = _project_simplex(a_candidate)

        # occasional exact evaluation to keep the best vector
        if t_iter % 250 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SoftAdam {t_iter:7d}] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Sub‑gradient descent on the exact objective (max b)
# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """Reduce max(b) by sub‑gradient steps with a linear step‑size schedule."""
    start = time.time()
    max_step = 0.10   # initial step size
    min_step = 1e-4   # final step size
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        k = int(np.argmax(b))  # index of the current worst convolution
        cur_max = b[k]
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")

        # gradient of max(b) w.r.t a (sub‑gradient at the worst index)
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * a[mid]

        # linear step‑size schedule
        progress = (time.time() - start) / (deadline - start)
        step = max_step * (1.0 - progress) + min_step * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – L²‑flattening (smooth variance of b)
# ----------------------------------------------------------------------
def _l2_flatten_phase(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """Minimise Σ(b−b̄)² to flatten the convolution."""
    start = time.time()
    base_step = 0.5
    min_step = 0.001
    avg_target = 1.0 / (2 * n - 1)    # mean of b when Σa = 1
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[L2 {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        r = b - avg_target
        a_rev = a[::-1]
        conv1 = _conv_fft(r, a)          # length 3n‑2
        conv2 = _conv_fft(r, a_rev)
        start_idx = n - 1
        end_idx = 2 * n - 1
        grad = 2.0 * (conv1[start_idx:end_idx] + conv2[start_idx:end_idx])

        # step‑size schedule
        progress = (time.time() - start) / (deadline - start)
        step = base_step * (1.0 - progress) + min_step * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        if no_improve > IMPROVE_LIMIT:
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[L2 {it}] restart after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – Greedy hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """
    Local search moving a fraction of mass from the index most
    responsible for the current peak to the index least responsible.
    Full convolution is updated incrementally in O(n).
    """
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # contribution of each a_i to the current worst convolution entry
        k = max_idx
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            # pick another source with non‑zero mass
            nonzeros = np.where(a > 1e-14)[0]
            if nonzeros.size == 0:
                break
            source = int(np.random.choice(nonzeros))

        # move up to 30 % of the source mass (randomised)
        delta = a[source] * 0.30 * np.random.rand()
        if delta <= 0.0:
            continue

        # --- Incremental update of the full convolution -----------------
        b_new = b.copy()
        # add/subtract the cross terms 2·δ·a
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # pure δ² terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept move
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # perturb after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1

    return best_a

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between expressiveness
    # and speed.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialisation – reuse previous construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # pylint: disable=undefined-variable
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)   # enforce Σa = 1
    best_a = a.copy()
    # initial evaluation – cheap and gives a baseline
    best_val = evaluate_sequence(best_a.tolist())

    # ------------------------------------------------------------------
    # Split time budget among the four phases.
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    t_softmax_end = start_time + 0.30 * total_time   # 30 %
    t_subgrad_end = start_time + 0.55 * total_time   # next 25 %
    t_l2_end      = start_time + 0.80 * total_time   # next 25 %
    # remaining ≈20 % for hill climbing

    # ----------------------------
    # Phase A – Soft‑max ADAM
    # ----------------------------
    best_a, best_val = _softmax_adam_phase(

    # ----------------------------
    best_a, best_val = _subgradient_phase(

    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported = {final_score:.6f}")

    return best_a.tolist()
