"""
Propose a step‑function (non‑negative heights) that (approximately) minimizes
``evaluate_sequence``.  The strategy combines three ideas:

1. **Large problem size** –  use n≈8000 (the bound improves roughly as
   1/√n, so a larger n gives a lower achievable objective).

2. **Smooth surrogate optimisation** –  minimise a log‑sum‑exp
   approximation of ``max(conv)`` with L‑BFGS‑B.  The parameters are kept in a
   soft‑max representation, guaranteeing non‑negativity and a fixed ℓ₁‑norm.
   The smoothing parameter ε is decreased on a continuation schedule.

3. **Aggressive mass‑transfer hill‑climbing** –  repeatedly locate the lag where
   the convolution attains its maximum, pick the pairs with the highest
   contribution to that lag and move a fraction of their mass to a set of
   low‑mass positions.  The step size is adapted and a move is kept only if
   the *exact* objective (computed via fast FFT convolution) improves.

The routine respects the given time budget (default 1000 s) and prints progress.
The best sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient w.r.t. soft‑max parameters
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Return the smooth surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv−max)/ε)

    together with its gradient w.r.t. the soft‑max parameters ``w``.
    """
    # soft‑max to obtain a feasible a (a_i≥0, Σa_i = 1)
    w_shift = w - np.max(w)                 # improve numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # full auto‑convolution of a
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient of L w.r.t. a (closed‑form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                 # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)               # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]              # middle n entries
    grad_a = 4.0 * n * seg[::-1]                # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Hill‑climbing mass‑transfer refinement
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              best_val: float,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climbing: at each step pick the lag k where the
    convolution is maximal, identify the pairs (i,j) that contribute most to
    that lag and move a small fraction of their mass to low‑mass sites.
    The move is kept only if the true objective improves.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.08            # start with a modest fraction

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue

        # how many pairs to manipulate?  up to 5 strongest ones
        top_n = min(5, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]

        donors = []      # list of (i,j) tuples (j may == i)
        for idx in top_idx:
            i = int(i_range[idx])
            j = int(j_range[idx])
            donors.append((i, j))

        # choose destination positions – low‑mass tail
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                # self‑pair a_i²
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue

        # re‑normalise to keep ℓ₁‑norm = 1 (scale‑invariant)
        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            # slightly increase step after success
            step_frac = min(step_frac * 1.02, 0.5)
        else:
            # shrink step when no improvement
            step_frac = max(step_frac * 0.94, 1e-5)

    return cur_a, cur_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function heights.

    1. Initialise from ``height_sequence_1`` if present, otherwise draw a
       random Dirichlet vector on a moderately large grid (n≈8000).
    2. Run a continuation L‑BFGS‑B on a log‑sum‑exp surrogate with a
       finer ε‑schedule (30 stages).  The optimisation is performed on the
       soft‑max parametrisation to keep the ℓ₁‑norm fixed.
    3. Perform several random restarts (up to 5) to escape possible minima.
    4. Finish with an aggressive hill‑climbing mass‑transfer phase that
       directly reduces the true ``evaluate_sequence`` score.

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a lower possible bound.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise (reuse the previous construction when available)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        # Dirichlet(1) → uniform on the simplex
        a0 = rng.random(n)
        a0 /= a0.sum()

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    w = np.log(a0 + 1e-12)                 # soft‑max parameters

    best_a = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Continuation schedule for ε (log‑space, finer towards zero)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-2, -12, num=30)

    # ------------------------------------------------------------------
    # Primary optimisation pass (L‑BFGS‑B on the smooth surrogate)
    # ------------------------------------------------------------------
    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        res = minimize(fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 800,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # evaluate exact objective
        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Random restarts – repeat a shortened continuation from perturbed starts
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0 and restart < 5:
        restart += 1
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.8, size=n)

        # only the last, finer part of the ε‑schedule is needed
        for eps in eps_schedule[-12:]:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 400,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart {restart}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _mass_transfer_hill_climb(best_a, best_val,
                                                    deadline, rng)

    # ------------------------------------------------------------------
    # Final report (use the provided evaluator for the final score)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] objective reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
