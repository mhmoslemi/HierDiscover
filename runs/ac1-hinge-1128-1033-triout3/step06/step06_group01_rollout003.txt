"""
Hybrid optimizer for the step‑function evaluation problem.

The algorithm proceeds in four stages:

1. **Smooth surrogate L‑BFGS** – we minimise a log‑sum‑exp
   approximation of the max‑convolution while keeping the
   ℓ₁‑norm fixed (soft‑max parametrisation).  Multiple restarts
   with a decreasing smoothing parameter ε give a good continuous
   shape.

2. **Ends‑large re‑ordering** – the multiset of heights is reordered
   so that the largest values are placed at the two ends of the
   sequence.  This heuristic reduces the peak of the self‑convolution
   for the same multiset.

3. **Projected sub‑gradient descent on the true max‑convolution** –
   starting from the reordered sequence we repeatedly identify the
   index k where (a*a) is maximal, move a in the negative sub‑gradient
   direction of that peak and project back onto the simplex (non‑negative,
   ℓ₁‑norm = 1).  The step size decays as 1/√t.  This refines the
   solution beyond the smooth surrogate.

4. **Short simulated‑annealing (SA) + hill‑climbing** – a final
   permutation search (swap moves) removes any remaining local
   irregularities.

All phases respect a total wall‑time budget of 1000 s and the function
returns the best list of non‑negative floats found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate wrapper (no closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip rounding artefacts
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast evaluator that returns exactly the same value as
    ``evaluate_sequence(a.tolist())`` but without the safety checks.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Convolution used inside the gradient of the smooth surrogate
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where L is the smoothed surrogate

        L = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)

    and grad_w is the gradient w.r.t. the soft‑max parameters w.
    """
    # ---------- soft‑max (positivity, ℓ₁ = 1) ----------
    w_shift = w - np.max(w)                     # numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                               # now Σa = 1

    # ---------- convolution a * a ----------
    conv = _fft_convolve_self(a)                 # length 2·n‑1
    max_conv = conv.max()

    # ---------- smooth approximation of the max ----------
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)                # log∑exp((c−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    weights = np.exp(shifted)                   # exp((c−max)/ε)
    p = weights / np.sum(weights)               # probability distribution over k

    # g_i = 4·n·∑_k p_k·a_{k−i}
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    # The central n entries of conv_ap correspond to indices [n‑1, 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                # ∂L/∂a

    # ---------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# Bind the helper convolution to the function object so the wrapper can see it
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristic – put the largest values at the two ends
# ----------------------------------------------------------------------
def _ends_large_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic re‑ordering that places the largest values at the
    extreme ends of the sequence and the smallest near the centre.
    This often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    left_ptr = 0
    right_ptr = n - 1
    i = 0
    while i < n:
        # place a large value
        new_a[i] = sorted_vals[right_ptr]
        right_ptr -= 1
        i += 1
        if i >= n:
            break
        # place a small value
        new_a[i] = sorted_vals[left_ptr]
        left_ptr += 1
        i += 1
    return new_a


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap moves only)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """Fast SA that swaps two positions; works directly on the fast evaluator."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # initialise temperature as a small fraction of the current value
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99994

    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        a[i], a[j] = a[j], a[i]            # propose swap
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            a[i], a[j] = a[j], a[i]        # revert

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Refine by simple hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:          # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                           # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Sub‑gradient method that directly minimises  max(a*a) while
    keeping a on the simplex (non‑negative, ℓ₁ = 1).
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = _eval_fast(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    step0 = 0.12                      # initial step size (empirically good)
    it = 0
    while time.time() < deadline:
        # convolution and location of the current maximum
        conv = _fft_convolve_self(cur_a)
        k_max = int(np.argmax(conv))

        # gradient of the max entry w.r.t. a : 2 * a_{k_max - i}  (if within range)
        idx_arr = k_max - np.arange(n)
        mask = (idx_arr >= 0) & (idx_arr < n)
        grad = np.zeros_like(cur_a)
        grad[mask] = 2.0 * cur_a[idx_arr[mask]]

        # diminishing step size
        alpha = step0 / np.sqrt(it + 1.0)

        new_a = cur_a - alpha * grad
        new_a = np.maximum(new_a, 0.0)
        sum_new = np.sum(new_a)
        if sum_new < 1e-12:
            # fallback – re‑initialise uniformly to avoid stagnation
            new_a = np.full(n, 1.0 / n)
            sum_new = 1.0
        new_a /= sum_new                        # project onto ℓ₁ = 1 simplex

        new_val = _eval_fast(new_a)

        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[subgrad] iter {it} new best = {best_val:.6f}")

        # accept only improving steps – this guarantees monotonic descent
        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val

        it += 1
    return best_a, best_val

# Main entry point required by the challenge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 1.5
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # --------------------------------------------------------------
    # Problem size – 4000 works well with the FFT based code
    # --------------------------------------------------------------
    n = 4000
    global _GLOBAL_N
    _GLOBAL_N = n

    # --------------------------------------------------------------
    # Initialise – try to reuse a previous construction if possible
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to requested size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)      # enforce non‑negativity
    a0 = a0 / np.sum(a0)          # normalise to ℓ₁ = 1

    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial fast evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Smooth surrogate L‑BFGS (few restarts, fine ε‑schedule)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-11), num=18)

    restart = 0
    while time.time() < deadline - 180.0:      # reserve ~3 min for later phases
        restart += 1
        print(f"[restart {restart}]")
        # initialise soft‑max parameters – use logarithm of current a plus noise
        w = np.log(best_a + 1e-12) + 0.5 * np.random.randn(n)

        for eps in eps_schedule:
            if time.time() >= deadline - 150.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate the current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= np.sum(a)
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # --- End of one restart -------------------------------------------------
        # apply ends‑large re‑ordering
        a_ord = _ends_large_reorder(best_a)
        a_ord /= np.sum(a_ord)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Projected sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline - 90.0:
        print("[subgrad] starting projected sub‑gradient descent")
        sub_deadline = deadline - 60.0
        best_a, best_val = _subgradient_descent(best_a, sub_deadline)

    # --------------------------------------------------------------
    # 3. Short SA on the permutation of the best sequence
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[SA] final simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 15.0)

    # --------------------------------------------------------------
    # 4. Final cheap hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ----------------------------------------------------------------------
    # Report final value using the official evaluator
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required by the framework
    return best_a.tolist()
