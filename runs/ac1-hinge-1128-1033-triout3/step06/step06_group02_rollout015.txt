import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Uses FFT for O(n log n) speed, returns a vector of length 2·n‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate negligible negatives caused by rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values on the left half and the smallest on the
    right half – a cheap heuristic that often reduces the central
    convolution peak.
    """
    n = a.shape[0]
    vals = np.sort(a)                         # ascending
    new = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new[i] = vals[n - 1 - i]              # large → left side
        new[n - 1 - i] = vals[i]              # small → right side
    if n % 2:
        new[half] = vals[half]
    return new


def _interleaved_reorder(a: np.ndarray) -> np.ndarray:
    """
    Alternate the largest and smallest entries at opposite ends,
    i.e. [max, min, 2nd‑max, 2nd‑min, …].  This often yields a
    flatter self‑convolution than pure sorting.
    """
    n = a.shape[0]
    vals = np.sort(a)[::-1]                   # descending
    new = np.empty_like(a)
    left, right = 0, n - 1
    for i, v in enumerate(vals):
        if i % 2 == 0:
            new[left] = v
            left += 1
        else:
            new[right] = v
            right -= 1
    return new


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex Σ v_i = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-12:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _gradient_step(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    One mirror‑descent step that reduces the current maximal convolution
    entry.  The gradient for the active lag k* is g_i = a_{k*-i}.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k_star = int(conv.argmax())
    # build gradient vector g_i = a_{k*-i} (zero outside the valid range)
    g = np.zeros_like(a)
    i_start = max(0, k_star - (n - 1))
    i_end = min(n - 1, k_star)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    jdx = k_star - idx
    g[idx] = a[jdx]
    # exponential mirror‑descent update
    a_new = a * np.exp(-alpha * g)
    return _project_to_simplex(a_new)


def _evaluate(seq: np.ndarray) -> float:
    """Wrapper around the provided evaluate_sequence."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        return float("inf")


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: random swaps (20 %) and
    small mass transfers (80 %).  Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:  # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                # mass transfer from i to j
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_a = _project_to_simplex(new_a)
        val = _evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


def _gen_arcsine_distribution(n: int) -> np.ndarray:
    """
    Discretise the continuous arcsine density
        f(x) ∝ 1 / sqrt(x (1-x)),   x∈(0,1)
    using mid‑point rule and normalise to unit sum.
    """
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n
    w = 1.0 / np.sqrt(x * (1.0 - x))
    a = w / w.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence`` within the allotted time budget.
    The routine mixes random shape sampling, deterministic reorderings,
    a few mirror‑descent iterations and a final hill‑climbing phase.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a 5 s safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Baseline (uniform) – gives a reference value.
    # ------------------------------------------------------------------
    a_uniform = np.full(n, 1.0 / n)
    best_a = a_uniform.copy()
    best_val = _evaluate(best_a)
    print(f"[baseline] uniform evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Shape exploration:
    #   – Dirichlet samples with several concentration parameters.
    #   – a deterministic arcsine‑like density.
    # For each shape we try a few clever reorderings and a few random
    # permutations, keeping the best candidate.
    # ------------------------------------------------------------------
    concentration_set = [0.2, 0.5, 1.0, 2.0, 5.0]   # low ⇒ spiky, high ⇒ smooth
    shape_deadline = deadline - 200.0               # reserve time for refinement

    while time.time() < shape_deadline:
        # pick a concentration at random
        alpha = rng.choice(concentration_set)
        # Dirichlet sample via Gamma(α,1) variables
        a = rng.gamma(alpha, 1.0, size=n)
        a /= a.sum()

        # a few deterministic orderings
        candidates = [
            a,                                 # raw order
            _anti_symmetric_reorder(a),        # large‑left / small‑right
            _interleaved_reorder(a),           # alternating large/small
        ]

        # add a few random permutations
        for _ in range(2):
            perm = rng.permutation(n)
            candidates.append(a[perm])

        # evaluate all
        for cand in candidates:
            val = _evaluate(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                print(f"[shape] new best ({alpha:.2f}) = {best_val:.6f}")

        # occasional deterministic arcsine shape (no need to repeat too often)
        if rng.random() < 0.05:
            a_arc = _gen_arcsine_distribution(n)
            for cand in (a_arc,
                         _anti_symmetric_reorder(a_arc),
                         _interleaved_reorder(a_arc)):
                val = _evaluate(cand)
                if val < best_val:
                    best_val = val
                    best_a = cand.copy()
                    print(f"[arcsine] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent on the (non‑smooth) max‑convolution.
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    alpha0 = 0.30            # initial step size (larger than before)
    # decay schedule: geometric, slowed down a little
    while time.time() < deadline - 30.0:
        a = _gradient_step(a, alpha0 * (0.999 ** (it // 100)))
        # occasional bookkeeping
        if it % 100 == 0:
            cur_val = _evaluate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[grad] iter {it} new best = {best_val:.6f}")
            # try a cheap re‑ordering every 500 steps
            if it % 500 == 0:
                a_as = _anti_symmetric_reorder(a)
                val_as = _evaluate(a_as)
                if val_as < best_val:
                    best_val = val_as
                    best_a = a_as.copy()
                    a = a_as.copy()
                    print(f"[grad reorder] improved to {best_val:.6f}")
        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – final stochastic hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final verification & return
    # ------------------------------------------------------------------
    final_score = _evaluate(best_a)
    print(f"[final] best evaluation reported = {final_score:.6f}")

    return best_a.tolist()
