"""
Simple two‑phase optimizer for the step–function bound.

Phase 0 – Very fine sweep of the one‑parameter U‑shaped family
           a_i ∝ ((i+½)(n‑i‑½))^{-γ} to obtain a deterministic seed.

Phase 1 – Sub‑gradient descent on the exact objective
           (projected exponential‑gradient on the simplex).
           The sub‑gradient is available analytically because the
           objective is 2·n·max_k (a∗a)[k] with Σ a = 1.
           A decreasing multiplicative step‑size together with occasional
           random perturbations drives the sequence towards a lower
           maximal autocorrelation.

If a previous construction (`height_sequence_1`) is available it is used
as an additional warm start.  The routine respects the 1000 s budget,
prints progress and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with sum 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {w ≥ 0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – fine sweep of the U‑shaped power‑law family
# ----------------------------------------------------------------------
def _initial_shape_fine(n: int, steps: int = 401) -> tuple[np.ndarray, float]:
    """Search a fine grid of γ ∈ [0,4] for the U‑shape family."""
    best_a = None
    best_val = np.inf
    x = np.arange(n, dtype=np.float64)
    eps = 1e-12
    for gamma in np.linspace(0.0, 4.0, steps):
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _gradient_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of 2·n·max(conv(a,a)) w.r.t. a at the current max index."""
    n = a.shape[0]
    g = np.zeros_like(a)
    # indices i for which a[k_max‑i] is defined
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] = 4.0 * n * a[k_max - idx]
    return g

    Returns the best sequence seen before *deadline*.

    best_local_val = best_val

        else:
            no_improve += 1

        # sub‑gradient at the (any) maximising index
        k_max = int(np.argmax(b))
        g = _gradient_max(a, k_max)

        # step‑size schedule (multiplicative scale decays)
        progress = (now - start) / (deadline - start)
        start_scale = 0.5
        end_scale = 0.001
        scale = start_scale * (1.0 - progress) + end_scale * progress

        max_g = g.max()
        if max_g > 0:
            lam = scale / max_g                      # ensures max change ≈ scale
            a = a * np.exp(-lam * g)                # multiplicative update
            a = np.maximum(a, 1e-20)                # avoid exact zeros
            a /= a.sum()                            # re‑normalise (simple soft‑max)
        else:
            # no useful gradient – fall back to a tiny random walk
            a += np.random.normal(scale=0.001, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

        # occasional random perturbation to escape stagnation
        if no_improve >= IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            no_improve = 0
            print(f"[SubGrad {t:7d}] stagnation – perturb")
        t += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Targeted hill‑climbing (small refinement after sub‑gradient phase)
# ----------------------------------------------------------------------
def _hillclimb_refine(a_start: np.ndarray,
                      n: int,
                      start: float,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Fast deterministic hill‑climb, moving a tiny amount of mass each step."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    max_frac_start, max_frac_end = 0.60, 0.02
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (biased by contribution)
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index (biased towards *small* contribution)
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # fallback uniform among all except i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept only strictly improving moves
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[HillRefine {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and restart from current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillRefine {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8000 entries give a good trade‑off between quality
    # and speed.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previous construction if available.
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – fine parametric sweep for a deterministic seed.
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf
    if a0 is not None:
        best_a, best_val = a0.copy(), _objective(a0)
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    try:
        shape_a, shape_val = _initial_shape_fine(n, steps=401)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] fine shape sweep improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] shape sweep failed:", exc)

    # fallback random start if everything else failed
    if best_a is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_a, best_val = a, _objective(a)
        print("[init] random start, value=", best_val)

    # ------------------------------------------------------------------
    # Phase 1 – sub‑gradient (exponential‑gradient) descent.
    # Allocate roughly 50 % of the remaining budget.
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.50 * budget_s
    best_a, best_val = _subgrad_phase(best_a, n,
                                      start=start_time,
                                      deadline=phase1_deadline,
                                      best_a=best_a,
                                      best_val=best_val,
                                      target_val=1.503)

    # ------------------------------------------------------------------
    # Phase 2 – quick hill‑climbing refinement (remaining time).
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_refine(best_a, n,
                                         start=phase1_deadline,
                                         deadline=deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
