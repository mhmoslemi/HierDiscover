"""
propose_candidate
-----------------

A fast mirror‑descent optimizer for the non‑negative step‑function problem.
The algorithm works in three stages:

1.  Build a small portfolio of diverse starting points (uniform, previous
    best, random Dirichlet, simple beta‑shapes and an anti‑symmetric
    re‑ordering).  The one with the smallest *max‑convolution* is kept.

2.  Run an exponential‑mirror‑descent loop on the *sub‑gradient* of the
    current maximal lag of the self‑convolution.  The step‑size is
    adapted automatically (reduced when no improvement is seen).  Every
    few hundred iterations an anti‑symmetric reorder is tried and a tiny
    random mass‑transfer perturbation is attempted to escape plateaus.
    The loop stops a few seconds before the overall budget expires.

3.  A short hill‑climbing refinement (random swaps / mass transfers)
    finishes the search.

All arithmetic is performed with the sequence normalised to sum = 1,
so the objective can be evaluated cheaply as ``2·n·max(conv)``.
The final list is returned and also verified with the supplied
``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Eliminate tiny negative round‑off errors
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest values on the left with the smallest on the right.
    This heuristic often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _compute_objective(a: np.ndarray) -> float:
    """
    Compute the evaluation function for a normalised sequence (sum = 1).

    Because the sequence is already normalised, the value returned by
    ``evaluate_sequence`` is exactly ``2·n·max(conv)``.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Simple hill‑climb: random swaps (20 %) and mass transfers (80 %).
    Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Re‑project onto the simplex (sum to 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _compute_objective(new_a)
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


def _gradient_descent(a0: np.ndarray, deadline: float):
    """
    Mirror‑descent on the sub‑gradient of the max‑lag of the self‑convolution.
    Returns the best sequence and its objective value found before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _full_convolution(a)
    val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = val

    it = 0
    base_step = 5.0          # initial mirror‑descent step (in the exponent)
    step = base_step
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # --- gradient from the current maximal lag(s) -----------------
        max_val = conv.max()
        # include all lags that are numerically tied with the maximum
        thresh = max_val - 1e-14
        top_idxs = np.where(conv >= thresh)[0]

        # accumulate sub‑gradient (sum over all active lags)
        g = np.zeros_like(a)
        for k in top_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            g[i] += a[j]

        # --- exponential mirror‑descent update ------------------------
        a_new = a * np.exp(-step * g)

            val = val_new
            no_improve = 0
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
                print(f"[gd iter {it}] new best = {best_val:.6f}")
        else:

            no_improve += 1

        # --- occasional random perturbation ---------------------------
        if it % 200 == 0:

            if i != j and a[i] > 1e-12:

                a_pert = a.copy()
                a_pert[i] -= delta
                a_pert[j] += delta
                a_pert = np.maximum(a_pert, 0.0)
                a_pert /= a_pert.sum()
                conv_pert = _full_convolution(a_pert)

                if val_pert < val - 1e-12:

                    conv = conv_pert
                    val = val_pert
                    if val_pert < best_val:
                        best_val = val_pert
                        best_a = a_pert.copy()
                        print(f"[perturb] new best = {best_val:.6f}")

            if val_ord < val - 1e-12:
