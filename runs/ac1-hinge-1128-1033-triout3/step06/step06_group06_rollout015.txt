def simplex_projection(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]

    return w

    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    # shift v by its min?
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
        theta = 0

"""
Propose a step‑function (list of non‑negative heights) that minimizes
`evaluate_sequence`.  The algorithm proceeds in four stages:

1.   Quick parametric sweep (uniform, linear, quadratic, sine, exponential)
     to obtain a decent deterministic start.
2.   Random sparse‑support initializations (few non‑zero entries, equal weight)
     – many candidates are tried and the best one is kept.
3.   Sub‑gradient descent on the exact objective 2·n·max(conv(a,a)),
     where the sub‑gradient is taken at the current maximal convolution entry.
     After each gradient step the vector is projected back onto the simplex.
4.   A short targeted hill‑climber and simulated‑annealing refinement
     (re‑using the robust implementations from the previous baseline).

All phases respect the overall 1000 s budget, use only NumPy
(and the already‑provided `evaluate_sequence`) and never write to disk.
The final sequence is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    # next power of two for speed
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa ≈ 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector.  The sum of a is assumed ≈1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x ≥ 0, Σx = 1})
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    # condition u_i - (c_i / i) > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric initialisation (uniform / linear / quadratic / sine / expo)
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector (Σ=1) and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    # linear decreasing
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # quadratic decreasing
    a = (np.arange(n, 0, -1, dtype=float) ** 2)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # sine half‑wave
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # exponential families
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    return best_a, best_val


# ----------------------------------------------------------------------
# Random sparse support initialisation
# ----------------------------------------------------------------------
def _random_sparse_phase(n: int, deadline: float, best_a: np.ndarray,
                        best_val: float) -> tuple[np.ndarray, float]:
    """Sample many vectors that have equal weight on a random sparse support."""
    while time.time() < deadline:
        s = np.random.randint(70, 180)          # support size
        idx = np.random.choice(n, s, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Sparse] new best = {best_val:.6f} (support={s})")
        # a tiny break to avoid tight loops when time is almost up
        if time.time() > deadline - 0.1:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact objective (projected onto simplex)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int, deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on  f(a)=2·n·max(conv(a,a)).
    At each iteration we take the sub‑gradient at the current maximal entry,
    step with a diminishing learning‑rate and project back onto the simplex.
    """
    a = a0.copy()
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 18000

    while time.time() < deadline:
        # current convolution and the index attaining the maximum
        conv = _full_convolution(a)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2 * a_{k-i} for valid i
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        # diminishing learning‑rate (roughly 0.8 / sqrt(t))
        lr = 0.8 / np.sqrt(it + 1)

        # gradient step + simplex projection
        a_new = _simplex_projection(a - lr * g)

        # occasional random shake to escape plateaux
        if np.random.rand() < 0.015:
            noise = np.random.normal(scale=0.015, size=n)
            a_new = _simplex_projection(np.maximum(a_new + noise, 0.0))

        a = a_new
        cur_val = _objective_value(a)

        if cur_val + 1e-12 < best_val:          # strict improvement
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SubGrad {it:6d}] new best = {best_val:.6f}")
            # early exit if we already beat the target
            if best_val <= 1.503:
                break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # re‑center around the current best with a small random perturbation
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = _simplex_projection(np.maximum(a + noise, 0.0))
            no_improve = 0
            print(f"[SubGrad {it:6d}] perturb after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted hill climbing (exact objective) – reused from the baseline
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Fast local search moving a tiny amount of mass between indices."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 180_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the current maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (biased towards large contributions)
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – occasionally favour a location with little contribution
        if np.random.rand() < 0.30:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept only strictly improving moves
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:6d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – perturb and restart from the best seen
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:6d}] perturbing after stagnation")
        it += 1

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement – also reused from the baseline
# ----------------------------------------------------------------------
def _annealing_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Metropolis‑style search that occasionally accepts worsening moves."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    T_start, T_end = 5e-4, 5e-8
    IMPROVE_LIMIT = 100_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source index – bias as in hill‑climber
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # simple uniform target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer (≤ 0.5 % of source mass)
        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        # incremental convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal {it:6d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # small random shake
            noise = np.random.normal(scale=0.003, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:6d}] perturbing after stagnation")
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,

    """

    `evaluate_sequence`.  The routine respects the given time budget (≈1000 s)
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a short safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries balances quality and speed
    n = 8000

    # ------------------------------------------------------------------
    # 0. deterministic parametric start (or reuse a known construction)
    # ------------------------------------------------------------------
    try:
        # if a previous solution is available, interpolate it

        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
